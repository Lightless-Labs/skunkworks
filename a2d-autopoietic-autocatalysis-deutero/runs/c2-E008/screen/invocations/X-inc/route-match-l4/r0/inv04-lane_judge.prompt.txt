        Self {

            routes: Vec::new(),

    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

            if route.method == method
                && same_shape(&route.segments, &segments)
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
            {
                return Err(Error::Conflict { with: id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());

        let parsed = parse_request_path(head)?;

        for (id, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((id, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates
                .iter()
                .any(|(id, _)| self.routes[*id].method == "HEAD")
            {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(id, _)| self.routes[*id].method == eligible_method)
            .collect();

                .routes
                .iter()

                    // The route must have matched the path; recompute because candidates
                    // were consumed only after the method decision.

                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });

                    .then_with(|| idb.cmp(ida))

                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {

        };

            route: id,
            params,
            redirect,
        })
    }

            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = Vec::with_capacity(params.len());
        let mut seen = HashSet::new();
        for (key, value) in params {
            if !seen.insert(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.push((*key, *value));
        }

        let required: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &required {
            if !supplied.iter().any(|(given, _)| given == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .iter()
            .map(|(key, _)| *key)
            .filter(|key| !required.iter().any(|required| required == key))
            .collect();
        extras.sort();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for key in &required {
            let value = supplied
                .iter()
                .find(|(given, _)| given == key)
                .map(|(_, value)| *value)
                .expect("required keys have already been checked");
            let wildcard = route
                .segments
                .iter()
                .any(|segment| matches!(segment, Segment::Wildcard(n) if n == key));
            if !valid_build_value(value, wildcard) {
                return Err(Error::InvalidValue {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut output = String::from("/");
        let mut emitted = 0usize;
        for segment in &route.segments {

                    if emitted > 0 {
                        output.push('/');
                    }
                    encode_segment(value, &mut output);
                    emitted += 1;
                }
                Segment::Param(key) => {
                    if emitted > 0 {

                    }

                        .map(|(_, value)| *value)
                        .expect("required parameter exists");
                    encode_segment(value, &mut output);
                    emitted += 1;
                }
                Segment::Wildcard(key) => {

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = pattern.as_bytes();
    let mut trailing = false;
    let end = if bytes.last() == Some(&b'/') {
        trailing = true;
        bytes.len() - 1
    } else {
        bytes.len()
    };

    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1usize;

    while start < end {
        let slash = bytes[start..end]
            .iter()
            .position(|b| *b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);
        if slash == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..slash];
        let first = raw.as_bytes()[0];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' && slash != end {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if first == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
        start = slash + 1;
    }

    if segments.is_empty() {
        return Err(Error::InvalidPattern { at: 1 });
    }
    Ok((segments, trailing))
}

fn parse_request_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1usize;

    while start < end {
        let slash = bytes[start..end]
            .iter()
            .position(|b| *b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);
        if slash == start {
            return Err(Error::InvalidPath { at: start });
        }
        let raw = &path[start..slash];
        let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);
        start = slash + 1;
    }

    if segments.is_empty() {
        return Err(Error::InvalidPath { at: 1 });
    }
    Ok(ParsedPath { segments, trailing })
}

fn decode_segment(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(offset + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(offset + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(left, right)| match (left, right) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    let fixed_len = if wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };

    if (!wildcard && path.len() != fixed_len) || (wildcard && path.len() < fixed_len) {
        return None;
    }

    let mut params = Vec::new();
    for (segment, value) in route.segments[..fixed_len].iter().zip(path.iter()) {
        match segment {
            Segment::Static(expected) if expected != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if wildcard {
        let Segment::Wildcard(name) = route.segments.last().expect("wildcard exists") else {
            unreachable!()
        };
        params.push((name.clone(), path[fixed_len..].join("/")));
    }
    Some(params)
}

fn rank(segments: &[Segment]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut has_wildcard = false;
    for segment in segments {
        match segment {
            Segment::Static(_) => result.push(2),
            Segment::Param(_) => result.push(1),
            Segment::Wildcard(_) => {
                result.push(0);
                has_wildcard = true;
                break;
            }
        }
    }
    if !has_wildcard {
        result.push(3);
    }
    result
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
        return false;
    }
    if wildcard {
        value.is_empty()
            || value
                .split('/')
                .all(|part| !part.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn encode_segment(value: &str, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for b in value.bytes() {
        if is_unreserved(b) {
            output.push(b as char);
        } else {
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
}

pub struct Router {

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_path(pattern, ParseKind::Pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let rank = make_rank(&segments);
        let id = self.routes.len();
        let candidate = Route {
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
            id,
        };

        let mut conflict: Option<usize> = None;
        for route in &self.routes {
            if route.method == candidate.method
                && (self.policy != SlashPolicy::Strict || route.trailing == candidate.trailing)
                && same_shape(&route.segments, &candidate.segments)
            {
                conflict = Some(conflict.map_or(route.id, |old| old.min(route.id)));
            }
        }
        if let Some(with) = conflict {
            return Err(Error::Conflict { with });
        }

        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head, ParseKind::Path)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &request_segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<(&Route, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .map(|(route, params)| (*route, params.clone()))
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank
                .cmp(&a.rank)
                .then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing != request_trailing
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = match self.routes.iter().find(|route| route.name == name) {
            Some(route) => route,
            None => return Err(Error::UnknownName),
        };

        let mut supplied: Vec<(&str, &str)> = Vec::with_capacity(params.len());
        for &(key, value) in params {
            if supplied.iter().any(|(prior, _)| *prior == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_owned(),
                });
            }
            supplied.push((key, value));
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.iter().any(|(supplied_key, _)| supplied_key == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .iter()
            .map(|(key, _)| *key)
            .filter(|key| !required.iter().any(|required_key| required_key == key))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let (key, wildcard) = match segment {
                Segment::Param(key) => (key, false),
                Segment::Wildcard(key) => (key, true),
                Segment::Static(_) => continue,
            };
            let value = supplied
                .iter()
                .find(|(supplied_key, _)| *supplied_key == key)
                .map(|(_, value)| *value)
                .unwrap_or("");
            if !valid_build_value(value, wildcard) {
                return Err(Error::InvalidValue { name: key.clone() });
            }
        }

        let mut output = String::from("/");
        for (index, segment) in route.segments.iter().enumerate() {
            if index > 0 {
                output.push('/');
            }
            match segment {
                Segment::Static(text) => output.push_str(&encode_component(text)),
                Segment::Param(key) => {
                    let value = supplied
                        .iter()
                        .find(|(supplied_key, _)| *supplied_key == key)
                        .map(|(_, value)| *value)
                        .unwrap_or("");
                    output.push_str(&encode_component(value));
                }
                Segment::Wildcard(key) => {
                    let value = supplied
                        .iter()
                        .find(|(supplied_key, _)| *supplied_key == key)
                        .map(|(_, value)| *value)
                        .unwrap_or("");
                    if !value.is_empty() {
                        for (part_index, part) in value.split('/').enumerate() {
                            if part_index > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part));
                        }
                    }
                }
            }
        }

        if route.trailing && output != "/" && !output.ends_with('/') {
            output.push('/');
        }
        Ok(output)
    }
}

fn parse_path(path: &str, kind: ParseKind) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| match kind {
        ParseKind::Pattern => Error::InvalidPattern { at },
        ParseKind::Path => Error::InvalidPath { at },
    };

    if !path.starts_with('/') {
        return Err(invalid(0));
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = path.as_bytes();
    let mut starts = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            starts.push((start, i));
            start = i + 1;
        }
    }
    let trailing = bytes.last() == Some(&b'/');
    if trailing {
        starts.push((start, bytes.len() - 1));
    } else {
        starts.push((start, bytes.len()));
    }

    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();

    for (index, (begin, end)) in starts.iter().copied().enumerate() {
        if begin == end {
            return Err(invalid(begin));
        }
        let raw = &bytes[begin..end];
        let is_last = index + 1 == starts.len();

        if kind == ParseKind::Pattern && (raw[0] == b':' || raw[0] == b'*') {
            let wildcard = raw[0] == b'*';
            if wildcard && !is_last {
                return Err(invalid(begin));
            }
            let name_bytes = &raw[1..];
            if !valid_capture_name(name_bytes) {
                return Err(invalid(begin));
            }
            let name = std::str::from_utf8(name_bytes)
                .expect("capture names are ASCII")
                .to_owned();
            if names.iter().any(|prior| prior == &name) {
                return Err(invalid(begin));
            }
            names.push(name.clone());
            segments.push(if wildcard {
                Segment::Wildcard(name)
            } else {
                Segment::Param(name)
            });
        } else {
            let decoded = decode_segment(raw, begin).map_err(invalid)?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok((segments, trailing))
}

fn decode_segment(raw: &[u8], offset: usize) -> Result<String, usize> {
    let mut decoded = Vec::with_capacity(raw.len());
    let mut i = 0;
    while i < raw.len() {
        let byte = raw[i];
        if byte == b'%' {
            if i + 2 >= raw.len() {
                return Err(offset + i);
            }
            let hi = hex_upper(raw[i + 1]).ok_or(offset + i)?;
            let lo = hex_upper(raw[i + 2]).ok_or(offset + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(byte) {
                return Err(offset + i);
            }
            decoded.push(byte);
            i += 1;
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    let mut wildcard = false;
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                wildcard = true;
                break;
            }
        }
    }
    if !wildcard {
        rank.push(3);
    }
    rank
}

fn same_shape(left: &[Segment], right: &[Segment]) -> bool {
    left.len() == right.len()
        && left.iter().zip(right).all(|(a, b)| match (a, b) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn match_segments(
    pattern: &[Segment],
    request: &[Segment],
) -> Option<Vec<(String, String)>> {
    let wildcard_index = pattern
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));

    let fixed_count = wildcard_index.unwrap_or(pattern.len());
    if request.len() < fixed_count {
        return None;
    }
    if wildcard_index.is_none() && request.len() != pattern.len() {
        return None;
    }

    let mut params = Vec::new();
    for index in 0..fixed_count {
        let request_value = match request.get(index)? {
            Segment::Static(value) => value,
            _ => return None,
        };
        match &pattern[index] {
            Segment::Static(expected) if expected == request_value => {}
            Segment::Param(name) => params.push((name.clone(), request_value.clone())),
            _ => return None,
        }
    }

    if let Some(index) = wildcard_index {
        let name = match &pattern[index] {
            Segment::Wildcard(name) => name.clone(),
            _ => unreachable!(),
        };
        let tail = request[fixed_count..]
            .iter()
            .map(|segment| match segment {
                Segment::Static(value) => Some(value.as_str()),
                _ => None,
            })
            .collect::<Option<Vec<_>>>()?
            .join("/");
        params.push((name, tail));
    }
    Some(params)
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
        return false;
    }
    if wildcard {
        if value.is_empty() {
            return true;
        }
        value
            .split('/')
            .all(|part| !part.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn encode_component(value: &str) -> String {
    let mut output = String::new();
    for byte in value.bytes() {
        if is_unreserved(byte) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn parse_segments(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut pieces: Vec<(usize, usize)> = Vec::new();
    let mut start = 1usize;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == start {
                if i == bytes.len() - 1 {
                    break;
                }
                return Err(invalid(i));
            }
            pieces.push((start, i));
            start = i + 1;
        }
    }

    let trailing = bytes.last() == Some(&b'/');
    if !trailing {
        if start == bytes.len() {
            return Err(invalid(start));
        }
        pieces.push((start, bytes.len()));
    } else if start < bytes.len() - 1 {
        pieces.push((start, bytes.len() - 1));
    }

    let mut segments = Vec::with_capacity(pieces.len());
    let mut names = std::collections::HashSet::new();

    for (a, b) in pieces {
        let raw = &bytes[a..b];

        if pattern && raw.first() == Some(&b':') {
            let name = &raw[1..];
            let name = std::str::from_utf8(name).map_err(|_| invalid(a))?;
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(invalid(a));
            }
            segments.push(Segment::Param(name.to_owned()));
            continue;
        }

        if pattern && raw.first() == Some(&b'*') {
            let name = &raw[1..];
            let name = std::str::from_utf8(name).map_err(|_| invalid(a))?;
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(invalid(a));
            }
            if b != bytes.len() - usize::from(trailing) {
                return Err(invalid(a));
            }
            segments.push(Segment::Wildcard(name.to_owned()));
            continue;
        }

        let mut decoded = Vec::with_capacity(raw.len());
        let mut i = 0usize;
        while i < raw.len() {
            let c = raw[i];
            if c == b'%' {
                if i + 2 >= raw.len() {
                    return Err(invalid(a + i));
                }
                let Some(hi) = hex_value(raw[i + 1]) else {
                    return Err(invalid(a + i));
                };
                let Some(lo) = hex_value(raw[i + 2]) else {
                    return Err(invalid(a + i));
                };
                let value = (hi << 4) | lo;
                if !(0x20..=0x7e).contains(&value)
                    || value == b'/'
                    || is_unreserved(value)
                {
                    return Err(invalid(a + i));
                }
                decoded.push(value);
                i += 3;
            } else if is_unreserved(c) {
                decoded.push(c);
                i += 1;
            } else {
                return Err(invalid(a + i));
            }
        }
        segments.push(Segment::Static(
            String::from_utf8(decoded).expect("decoded bytes are ASCII"),
        ));
    }

    Ok((segments, trailing))
}

        .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
        .unwrap_or(input.len());
    let head = &input[..head_end];
    let (segments, trailing) = parse_segments(head, false)?;
    let mut decoded = Vec::with_capacity(segments.len());
    let bytes = head.as_bytes();

    let mut start = 1usize;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i > start {
                decoded.push(decode_request_segment(&bytes[start..i]).ok_or(
                    Error::InvalidPath { at: start },
                )?);
            }
            start = i + 1;
        }
    }
    if !trailing && start < bytes.len() {
        decoded.push(
            decode_request_segment(&bytes[start..]).ok_or(Error::InvalidPath { at: start })?,
        );
    } else if trailing && start < bytes.len() - 1 {
        decoded.push(
            decode_request_segment(&bytes[start..bytes.len() - 1])
                .ok_or(Error::InvalidPath { at: start })?,
        );
    }
    Ok((decoded, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_segments(pattern, true)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let has_wildcard = segments
            .iter()
            .any(|s| matches!(s, Segment::Wildcard(_)));
        let candidate = Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            has_wildcard,
        };

        if let Some(existing) = self
            .routes
            .iter()
            .filter(|r| shape_conflicts(r, &candidate, self.policy))
            .map(|r| r.id)
            .min()
        {
            return Err(Error::Conflict { with: existing });
        }

        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (request, request_trailing) = parse_request(path)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
            .unwrap_or(path.len());
        let original_head = &path[..head_end];

        let mut candidates: Vec<(&Route, Vec<(String, String)>, Vec<u8>)> = Vec::new();
        for route in &self.routes {
            let mut params = Vec::new();
            let mut ri = 0usize;
            let mut qi = 0usize;
            let mut matched = true;
            let mut rank = Vec::new();

            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Static(text) => {
                        if request.get(qi) != Some(text) {
                            matched = false;
                            break;
                        }
                        rank.push(2);
                        ri += 1;
                        qi += 1;
                    }
                    Segment::Param(name) => {
                        let Some(value) = request.get(qi) else {
                            matched = false;
                            break;
                        };
                        params.push((name.clone(), value.clone()));
                        rank.push(1);
                        ri += 1;
                        qi += 1;
                    }
                    Segment::Wildcard(name) => {
                        let value = request[qi..].join("/");
                        params.push((name.clone(), value));
                        rank.push(0);
                        ri += 1;
                        qi = request.len();
                    }
                }
            }
            if matched && qi == request.len() {
                if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                    continue;
                }
                if !route.has_wildcard {
                    rank.push(3);
                }
                candidates.push((route, params, rank));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(r, _, _)| r.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(r, _, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for route in self.routes.iter().filter(|route| {
                route_matches_path(route, &request, request_trailing, self.policy)
            }) {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_owned());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let mut best = 0usize;
        for i in 1..eligible.len() {
            if eligible[i].2 > eligible[best].2 {
                best = i;
            }
        }
        let (route, params, _) = &eligible[best];

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !request.is_empty()
        {
            if route.trailing {
                Some(format!("{}/", original_head))
            } else {
                Some(original_head.trim_end_matches('/').to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(route) = self.routes.iter().find(|r| r.name == name) else {
            return Err(Error::UnknownName);
        };

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let captures: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &captures {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !captures.contains(key))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let name = match segment {
                Segment::Param(name) | Segment::Wildcard(name) => name,
                Segment::Static(_) => continue,
            };
            let value = supplied[name.as_str()];
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid = match segment {
                Segment::Param(_) => valid_ascii && !value.is_empty() && !value.contains('/'),
                Segment::Wildcard(_) => {
                    valid_ascii
                        && (value.is_empty()
                            || value
                                .split('/')
                                .all(|part| !part.is_empty()))
                }
                Segment::Static(_) => true,
            };
            if !valid {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut output = String::new();
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    output.push('/');
                    output.push_str(&encode_segment(value));
                }
                Segment::Param(name) => {
                    output.push('/');
                    output.push_str(&encode_segment(supplied[name.as_str()]));
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        output.push('/');
                        let mut first = true;
                        for part in value.split('/') {
                            if !first {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(part));
                            first = false;
                        }
                    }
                }
            }
        }

        if output.is_empty() {
            output.push('/');
        } else if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn route_matches_path(
    route: &Route,
    request: &[String],
    request_trailing: bool,
    policy: SlashPolicy,
) -> bool {
    if policy == SlashPolicy::Strict && route.trailing != request_trailing {
        return false;
    }

    let mut qi = 0usize;
    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if request.get(qi) != Some(value) {
                    return false;
                }
                qi += 1;
            }
            Segment::Param(_) => {
                if qi >= request.len() {
                    return false;
                }
                qi += 1;
            }
            Segment::Wildcard(_) => return true,
        }
    }
    qi == request.len()
}
