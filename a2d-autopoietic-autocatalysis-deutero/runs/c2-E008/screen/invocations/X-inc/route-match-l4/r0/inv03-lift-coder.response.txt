#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, Debug, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone, Debug)]
struct Route {
    id: usize,
    name: String,
    method: String,
    pattern: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(
        &mut self,
        name: &str,
        method: &str,
        pattern: &str,
    ) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method == method
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
                && same_shape(&route.segments, &segments)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        let rank = rank_for(&segments);
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            pattern: pattern.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, trailing) = parse_path_head(head)?;

        let mut path_candidates = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &request_segments) {
                path_candidates.push((route, params));
            }
        }
        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let effective_method = if method == "HEAD" {
            if path_candidates.iter().any(|(route, _)| route.method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };

        let eligible: Vec<_> = path_candidates
            .iter()
            .filter(|(route, _)| route.method == effective_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = path_candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (route, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id))
            })
            .unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != trailing
            && !request_segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        for segment in &route.segments {
            if let Segment::Param(key) | Segment::Wildcard(key) = segment {
                if !supplied.contains_key(key.as_str()) {
                    return Err(Error::MissingParam { name: key.clone() });
                }
            }
        }

        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !route_has_capture(route, key))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            if let Segment::Param(key) | Segment::Wildcard(key) = segment {
                if !valid_build_value(supplied[key.as_str()], matches!(segment, Segment::Wildcard(_))) {
                    return Err(Error::InvalidValue { name: key.clone() });
                }
            }
        }

        let mut output = String::from("/");
        let mut emitted = 0usize;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(value.as_bytes()));
                    emitted += 1;
                }
                Segment::Param(key) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(supplied[key.as_str()].as_bytes()));
                    emitted += 1;
                }
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    if !value.is_empty() {
                        if emitted > 0 {
                            output.push('/');
                        }
                        for (index, component) in value.split('/').enumerate() {
                            if index > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(component.as_bytes()));
                        }
                        emitted += value.split('/').count();
                    }
                }
            }
        }

        if output != "/" && route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    parse_segments(pattern, true)
}

fn parse_path_head(path: &str) -> Result<(Vec<Segment>, bool), Error> {
    let (segments, trailing) = parse_segments(path, false)?;
    Ok((segments, trailing))
}

fn parse_segments(input: &str, is_pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut segments = Vec::new();
    let mut start = 1usize;
    let mut trailing = false;
    while start <= bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|b| *b == b'/')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if end == start {
            if end == bytes.len() && start == bytes.len() {
                trailing = true;
                break;
            }
            return Err(invalid(start));
        }

        let raw = &input[start..end];
        if is_pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(invalid(start));
            }
            if segments.iter().any(|segment| match segment {
                Segment::Param(existing) | Segment::Wildcard(existing) => existing == name,
                Segment::Static(_) => false,
            }) {
                return Err(invalid(start));
            }
            if marker == b'*' {
                if end < bytes.len() && end + 1 < bytes.len() {
                    return Err(invalid(start));
                }
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            let decoded = decode_segment(raw.as_bytes(), start).map_err(|at| invalid(at))?;
            segments.push(Segment::Static(decoded));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            trailing = true;
            break;
        }
    }

    if let Some(index) = segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)))
    {
        if index + 1 != segments.len() {
            let offset = segment_offset(input, index);
            return Err(invalid(offset));
        }
    }
    Ok((segments, trailing))
}

fn segment_offset(input: &str, index: usize) -> usize {
    let mut offset = 1usize;
    for _ in 0..index {
        let next = input[offset..].find('/').unwrap();
        offset += next + 1;
    }
    offset
}

fn decode_segment(bytes: &[u8], base: usize) -> Result<String, usize> {
    let mut decoded = Vec::new();
    let mut i = 0usize;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let high = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let low = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(left, right)| match (left, right) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn rank_for(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn match_segments(pattern: &[Segment], request: &[Segment]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let wildcard_index = pattern
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_count = wildcard_index.unwrap_or(pattern.len());

    if request.len() < fixed_count || (wildcard_index.is_none() && request.len() != pattern.len()) {
        return None;
    }

    for (pat, req) in pattern[..fixed_count].iter().zip(request.iter()) {
        match pat {
            Segment::Static(expected) => match req {
                Segment::Static(actual) if expected == actual => {}
                _ => return None,
            },
            Segment::Param(name) => {
                let Segment::Static(value) = req else {
                    return None;
                };
                params.push((name.clone(), value.clone()));
            }
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_index {
        if let Segment::Wildcard(name) = &pattern[index] {
            let value = request[fixed_count..]
                .iter()
                .map(|segment| match segment {
                    Segment::Static(value) => value.as_str(),
                    _ => "",
                })
                .collect::<Vec<_>>()
                .join("/");
            params.push((name.clone(), value));
        }
    }
    Some(params)
}

fn route_has_capture(route: &Route, key: &str) -> bool {
    route.segments.iter().any(|segment| match segment {
        Segment::Param(name) | Segment::Wildcard(name) => name == key,
        Segment::Static(_) => false,
    })
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|byte| (0x20..=0x7e).contains(&byte)) {
        return false;
    }
    if wildcard {
        value.is_empty()
            || value
                .split('/')
                .all(|component| !component.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn encode_component(bytes: &[u8]) -> String {
    let mut encoded = String::new();
    for &byte in bytes {
        if is_unreserved(byte) {
            encoded.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    encoded
}
