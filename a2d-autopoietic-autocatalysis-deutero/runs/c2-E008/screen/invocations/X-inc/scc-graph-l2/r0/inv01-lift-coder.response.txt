pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative first pass of Kosaraju's algorithm.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);
    for start in 0..n {
        if visited[start] {
            continue;
        }
        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.pop() {
            if next_index < adjacency[vertex].len() {
                stack.push((vertex, next_index + 1));
                let next = adjacency[vertex][next_index];
                if !visited[next] {
                    visited[next] = true;
                    stack.push((next, 0));
                }
            } else {
                finish_order.push(vertex);
            }
        }
    }

    // Second pass assigns provisional component IDs.
    let mut provisional_of = vec![usize::MAX; n];
    let mut provisional_components: Vec<Vec<usize>> = Vec::new();
    for &start in finish_order.iter().rev() {
        if provisional_of[start] != usize::MAX {
            continue;
        }
        let id = provisional_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        provisional_of[start] = id;
        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &next in &reverse[vertex] {
                if provisional_of[next] == usize::MAX {
                    provisional_of[next] = id;
                    stack.push(next);
                }
            }
        }
        members.sort_unstable();
        provisional_components.push(members);
    }

    let mut order: Vec<usize> = (0..provisional_components.len()).collect();
    order.sort_unstable_by(|&a, &b| {
        provisional_components[b]
            .len()
            .cmp(&provisional_components[a].len())
            .then_with(|| provisional_components[a][0].cmp(&provisional_components[b][0]))
    });

    let mut remap = vec![0usize; provisional_components.len()];
    let mut components = Vec::with_capacity(provisional_components.len());
    for (new_id, &old_id) in order.iter().enumerate() {
        remap[old_id] = new_id;
        components.push(provisional_components[old_id].clone());
    }

    let mut component_of = vec![0usize; n];
    for vertex in 0..n {
        component_of[vertex] = remap[provisional_of[vertex]];
    }

    let component_count = components.len();
    let mut cyclic: Vec<bool> = components
        .iter()
        .map(|members| members.len() > 1)
        .collect();
    let mut internal_edges = vec![0usize; component_count];
    let mut arc_stats: Vec<(usize, usize, usize, usize)> = Vec::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let from_component = component_of[from];
        let to_component = component_of[to];
        if from_component == to_component {
            internal_edges[from_component] += 1;
            if from == to {
                cyclic[from_component] = true;
            }
        } else {
            arc_stats.push((from_component, to_component, edge_index, 1));
        }
    }

    arc_stats.sort_unstable_by_key(|&(from, to, edge_index, _)| (from, to, edge_index));
    let mut arcs = Vec::new();
    let mut index = 0;
    while index < arc_stats.len() {
        let (from, to, first_edge, _) = arc_stats[index];
        let mut multiplicity = 0usize;
        while index < arc_stats.len() && arc_stats[index].0 == from && arc_stats[index].1 == to {
            multiplicity += arc_stats[index].3;
            index += 1;
        }
        arcs.push(Arc {
            from,
            to,
            multiplicity,
            first_edge,
        });
    }
    arcs.sort_unstable_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
