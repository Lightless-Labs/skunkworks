#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semi,
    Query,
    Amp,
}

#[derive(Debug, Clone)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    pos: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let operator = match bytes[content_start] {
        b'+' => { content_start += 1; Operator::Plus }
        b'#' => { content_start += 1; Operator::Hash }
        b'.' => { content_start += 1; Operator::Dot }
        b'/' => { content_start += 1; Operator::Slash }
        b';' => { content_start += 1; Operator::Semi }
        b'?' => { content_start += 1; Operator::Query }
        b'&' => { content_start += 1; Operator::Amp }
        _ => Operator::Simple,
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut specs = Vec::new();
    let mut used = std::collections::HashSet::new();
    let mut part_start = content_start;

    loop {
        let part_end = bytes[part_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|i| part_start + i)
            .unwrap_or(end);

        if part_start == part_end {
            return Err(Error::EmptyVariable { pos: part_start });
        }

        let mut name_end = part_start;
        while name_end < part_end && bytes[name_end] != b'*' && bytes[name_end] != b':' {
            name_end += 1;
        }

        if name_end == part_start {
            return Err(Error::InvalidName { pos: part_start });
        }

        if !is_name_start(bytes[part_start]) {
            return Err(Error::InvalidName { pos: part_start });
        }
        for i in part_start + 1..name_end {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
        }

        let modifier = if name_end == part_end {
            Modifier::None
        } else {
            let modifier_pos = name_end;
            let suffix = &bytes[name_end..part_end];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.len() >= 2
                && suffix[0] == b':'
                && suffix.len() <= 5
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut n = 0usize;
                for &digit in &suffix[1..] {
                    n = n * 10 + (digit - b'0') as usize;
                }
                if (1..=9999).contains(&n) {
                    Modifier::Prefix(n)
                } else {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        let name = template[part_start..name_end].to_string();
        if used.contains(&name) {
            return Err(Error::DuplicateVariable { pos: part_start });
        }
        used.insert(name.clone());
        specs.push(Specifier { name, pos: part_start, modifier });

        if part_end == end {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression { operator, specs })
}

fn reserved(b: u8) -> bool {
    matches!(b, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'=')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn encode(input: &str, allow_reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h1), Some(h2)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(HEX[h1 as usize] as char);
                out.push(HEX[h2 as usize] as char);
                i += 3;
                continue;
            }
        }

        if unreserved(b) || (allow_reserved && reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn select_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn encoded_scalar(value: &str, modifier: &Modifier, allow_reserved: bool) -> String {
    let selected = match modifier {
        Modifier::Prefix(n) => value.chars().take(*n).collect::<String>(),
        _ => value.to_string(),
    };
    encode(&selected, allow_reserved)
}

fn named_field(name: &str, value: String, operator: &Operator) -> String {
    match operator {
        Operator::Semi if value.is_empty() => name.to_string(),
        _ => format!("{name}={value}"),
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let allow_reserved = matches!(expr.operator, Operator::Plus | Operator::Hash);
    let mut fields = Vec::new();

    for spec in &expr.specs {
        let Some(value) = select_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(s) => {
                let v = encoded_scalar(s, &spec.modifier, allow_reserved);
                fields.push(match expr.operator {
                    Operator::Semi | Operator::Query | Operator::Amp => {
                        named_field(&spec.name, v, &expr.operator)
                    }
                    _ => v,
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        let v = encode(item, allow_reserved);
                        fields.push(match expr.operator {
                            Operator::Semi | Operator::Query | Operator::Amp => {
                                named_field(&spec.name, v, &expr.operator)
                            }
                            _ => v,
                        });
                    }
                } else {
                    let v = items.iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(match expr.operator {
                        Operator::Semi | Operator::Query | Operator::Amp => {
                            named_field(&spec.name, v, &expr.operator)
                        }
                        _ => v,
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }
                if entries.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, val) in entries {
                        let k = encode(key, allow_reserved);
                        let v = encode(val, allow_reserved);
                        let field = if matches!(expr.operator, Operator::Semi) && v.is_empty() {
                            k
                        } else {
                            format!("{k}={v}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, allow_reserved));
                        components.push(encode(val, allow_reserved));
                    }
                    let v = components.join(",");
                    fields.push(match expr.operator {
                        Operator::Semi | Operator::Query | Operator::Amp => {
                            named_field(&spec.name, v, &expr.operator)
                        }
                        _ => v,
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expr.operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Hash => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semi => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Amp => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let Some(relative_end) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = i + 1 + relative_end;
                if let Some(inner) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: start + 1 + inner });
                }
                let expr = parse_expression(template, start, end)?;
                expressions.push((start, end + 1, expr));
                i = end + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for (start, end, expr) in expressions {
        out.push_str(&template[cursor..start]);
        out.push_str(&expand_expression(&expr, vars)?);
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
