#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Ampersand,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier<'a> {
    name: &'a str,
    start: usize,
    modifier: Modifier,
}

struct Expression<'a> {
    operator: Operator,
    specs: Vec<Specifier<'a>>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    content_start: usize,
    content_end: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut operator = Operator::Simple;
    let mut start = content_start;

    if start < content_end {
        operator = match bytes[start] {
            b'+' => Operator::Plus,
            b'#' => Operator::Fragment,
            b'.' => Operator::Dot,
            b'/' => Operator::Slash,
            b';' => Operator::Semicolon,
            b'?' => Operator::Query,
            b'&' => Operator::Ampersand,
            _ => Operator::Simple,
        };
        if operator != Operator::Simple {
            start += 1;
        } else if !is_name_start(bytes[start]) && bytes[start] != b',' {
            return Err(Error::InvalidName { pos: start });
        }
    }

    if start == content_end {
        return Err(Error::EmptyExpression { pos: content_end });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut spec_start = start;

    loop {
        let mut spec_end = spec_start;
        while spec_end < content_end && bytes[spec_end] != b',' {
            spec_end += 1;
        }

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let raw = &template[spec_start..spec_end];
        let raw_bytes = raw.as_bytes();
        let modifier_at = raw_bytes
            .iter()
            .position(|b| *b == b'*' || *b == b':')
            .unwrap_or(raw_bytes.len());

        if modifier_at == 0 {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let name_bytes = &raw_bytes[..modifier_at];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (offset, &b) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(b) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let modifier = if modifier_at == raw_bytes.len() {
            Modifier::None
        } else {
            let modifier_pos = spec_start + modifier_at;
            let suffix = &raw_bytes[modifier_at..];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.first() == Some(&b':')
                && (2..=5).contains(&suffix.len())
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut n = 0usize;
                for &digit in &suffix[1..] {
                    n = n * 10 + (digit - b'0') as usize;
                }
                if n <= 9999 {
                    Modifier::Prefix(n)
                } else {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        let name = &template[spec_start..spec_start + modifier_at];
        if !seen.insert(name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        specs.push(Specifier {
            name,
            start: spec_start,
            modifier,
        });

        if spec_end == content_end {
            break;
        }
        spec_start = spec_end + 1;
        if spec_start > content_end {
            break;
        }
    }

    Ok(Expression { operator, specs })
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(n: u8) -> u8 {
    if n < 10 {
        b'0' + n
    } else {
        b'A' + n - 10
    }
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h1), Some(h2)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(hex_digit(h1) as char);
                out.push(hex_digit(h2) as char);
                i += 3;
                continue;
            }
        }
        if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4) as char);
            out.push(hex_digit(b & 0x0f) as char);
        }
        i += 1;
    }
    out
}

fn prefix_chars(value: &str, n: usize) -> String {
    value.chars().take(n).collect()
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(
    expression: &Expression<'_>,
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let reserved = matches!(expression.operator, Operator::Plus | Operator::Fragment);
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) {
            match value {
                Value::Scalar(_) => {}
                Value::List(_) | Value::Map(_) => {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
            }
        }

        match value {
            Value::Scalar(s) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(n) => prefix_chars(s, n),
                    _ => s.clone(),
                };
                fields.push(if matches!(
                    expression.operator,
                    Operator::Semicolon | Operator::Query | Operator::Ampersand
                ) {
                    named_field(spec.name, encode(&selected, reserved), expression.operator)
                } else {
                    encode(&selected, reserved)
                });
            }
            Value::List(items) => {
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(
                            expression.operator,
                            Operator::Semicolon | Operator::Query | Operator::Ampersand
                        ) {
                            named_field(spec.name, encoded, expression.operator)
                        } else {
                            encoded
                        });
                    }
                } else if !items.is_empty() {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(
                        expression.operator,
                        Operator::Semicolon | Operator::Query | Operator::Ampersand
                    ) {
                        named_field(spec.name, joined, expression.operator)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, val) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(val, reserved);
                        let field = if expression.operator == Operator::Semicolon
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else if !entries.is_empty() {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(val, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(
                        expression.operator,
                        Operator::Semicolon | Operator::Query | Operator::Ampersand
                    ) {
                        named_field(spec.name, joined, expression.operator)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let mut close = cursor + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }
                if let Some(offset) = bytes[open + 1..close].iter().position(|b| *b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let expression = parse_expression(template, open + 1, close)?;
                output.push_str(&expand_expression(&expression, vars)?);
                cursor = close + 1;
            }
            _ => {
                let start = cursor;
                while cursor < bytes.len() && bytes[cursor] != b'{' && bytes[cursor] != b'}' {
                    cursor += 1;
                }
                output.push_str(&template[start..cursor]);
            }
        }
    }

    Ok(output)
}
