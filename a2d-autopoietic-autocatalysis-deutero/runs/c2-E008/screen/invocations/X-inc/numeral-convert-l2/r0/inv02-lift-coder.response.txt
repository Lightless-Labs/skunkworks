#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = tag_end + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let mut digits = Vec::new();
    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => return Err(BaseError::InvalidDigit { operand, index }),
        }
    }

    for index in payload_start..bytes.len() {
        if bytes[index] == b'_'
            && (index == payload_start
                || index + 1 == bytes.len()
                || bytes[index - 1] == b'_'
                || bytes[index + 1] == b'_')
        {
            return Err(BaseError::Separator { operand, index });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for digit in value.iter_mut() {
        let n = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (n % 10) as u8;
        carry = n / 10;
    }
    while carry != 0 {
        value.push((carry % 10) as u8);
        carry /= 10;
    }
}

fn to_decimal(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut decimal = vec![0];
    for &digit in digits {
        multiply_add(&mut decimal, radix, digit);
    }
    while decimal.len() > 1 && decimal.last() == Some(&0) {
        decimal.pop();
    }
    decimal
}

fn divide_small(value: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for digit in value.iter_mut().rev() {
        let n = remainder * 10 + *digit as u16;
        *digit = (n / divisor as u16) as u8;
        remainder = n % divisor as u16;
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn render(digits: Vec<u8>, radix: u8, style: BaseStyle) -> String {
    let mut decimal = digits;
    let mut output = Vec::new();

    if decimal.iter().all(|&d| d == 0) {
        output.push(0);
    } else {
        while decimal.iter().any(|&d| d != 0) {
            output.push(divide_small(&mut decimal, radix));
        }
        output.reverse();
    }

    while output.len() < style.min_width {
        output.insert(0, 0);
    }

    let mut chars = Vec::new();
    for (index, &digit) in output.iter().enumerate() {
        if style.group != 0 && index != 0 && (output.len() - index) % style.group == 0 {
            chars.push('_');
        }
        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + digit - 10,
            _ => b'a' + digit - 10,
        };
        chars.push(byte);
    }
    String::from_utf8(chars).expect("output consists of ASCII characters")
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let decimal = to_decimal(&digits, from);
    Ok(render(decimal, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut sum = vec![0u8];
    for digit in left_digits {
        multiply_add(&mut sum, base, digit);
    }

    let right_value = to_decimal(&right_digits, base);
    let mut carry = 0u16;
    let len = sum.len().max(right_value.len());
    sum.resize(len, 0);
    for i in 0..len {
        let n = sum[i] as u16 + right_value.get(i).copied().unwrap_or(0) as u16 + carry;
        sum[i] = (n % 10) as u8;
        carry = n / 10;
    }
    if carry != 0 {
        sum.push(carry as u8);
    }

    Ok(render(sum, base, style))
}
