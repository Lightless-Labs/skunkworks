#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone)]
struct Component {
    value: String,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn raw_components(path: &str, offset: usize) -> Vec<Component> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push(Component {
            value: path[start..i].to_string(),
            at: offset + start,
        });
    }
    result
}

fn push_component(stack: &mut Vec<String>, component: &Component) -> Result<(), PathError> {
    match component.value.as_str() {
        "." => {}
        ".." => {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at: component.at });
            }
        }
        _ => stack.push(component.value.clone()),
    }
    Ok(())
}

fn render_absolute(stack: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent && !stack.is_empty() {
        result.push('/');
    }
    result
}

fn render_normalized(stack: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return render_absolute(stack, intent);
    }
    if stack.is_empty() {
        return ".".to_string();
    }
    let mut result = stack.join("/");
    if intent {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    for component in raw_components(path, 0) {
        match component.value.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component.at });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(component.value),
        }
    }

    Ok(render_normalized(&stack, absolute, intent))
}

fn validate_context(value: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !value.starts_with('/') {
        return Err(error);
    }
    match normalize(value) {
        Ok(normalized) => {
            let stack = normalized
                .split('/')
                .filter(|part| !part.is_empty())
                .map(str::to_string)
                .collect();
            Ok(stack)
        }
        Err(_) => Err(error),
    }
}

fn request_parts<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(Vec<String>, &'a str, usize, bool), PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);

    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let offset = if bytes.len() > 1 { 2 } else { 1 };
            let rest = if bytes.len() > 1 { &path[2..] } else { "" };
            return Ok((Vec::new(), rest, offset, intent));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    Ok((Vec::new(), path, 0, intent))
}

fn resolve_core(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_stack = validate_context(cwd, PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path)
            && link.path != "/";
        let valid_target = validate_raw(&link.target).is_ok();
        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(first) = links[..index].iter().position(|earlier| earlier.path == link.path) {
            return Err(PathError::DuplicateLink { first, second: index });
        }
    }

    validate_raw(path)?;
    let intent = directory_intent(path);
    let (start, rest, offset, absolute) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home_stack = home_stack.ok_or(PathError::MissingHome)?;
            let rest_offset = if bytes.len() > 1 { 2 } else { 1 };
            let rest = if bytes.len() > 1 { &path[2..] } else { "" };
            (home_stack, rest, rest_offset, false)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_stack, path, 0, false)
    };

    let mut stack = start;
    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    enum Work<'a> {
        Request(Vec<Component>),
        Target(Vec<Component>, usize),
    }

    let mut work = vec![Work::Request(raw_components(rest, offset))];

    while let Some(item) = work.pop() {
        match item {
            Work::Target(components, link_index) => {
                for component in components {
                    match component.value.as_str() {
                        "." => {}
                        ".." => {
                            if stack.pop().is_none() {
                                return Err(PathError::Escape { at: component.at });
                            }
                        }
                        _ => {
                            stack.push(component.value.clone());
                            let key = format!("/{}", stack.join("/"));
                            if let Some(index) = links.iter().position(|link| link.path == key) {
                                if active.contains(&index) {
                                    return Err(PathError::Loop { index });
                                }
                                if expansions >= max_expansions {
                                    return Err(PathError::Limit { index });
                                }
                                expansions += 1;
                                active.push(index);
                                let target = &links[index].target;
                                let target_components = raw_components(target, 0);
                                work.push(Work::Target(Vec::new(), index));
                                work.push(Work::Target(target_components, index));
                            }
                        }
                    }
                }
                if active.last() == Some(&link_index) {
                    active.pop();
                }
            }
            Work::Request(components) => {
                for component in components {
                    match component.value.as_str() {
                        "." => {}
                        ".." => {
                            if stack.pop().is_none() {
                                return Err(PathError::Escape { at: component.at });
                            }
                        }
                        _ => {
                            stack.push(component.value.clone());
                            let key = format!("/{}", stack.join("/"));
                            if let Some(index) = links.iter().position(|link| link.path == key) {
                                if active.contains(&index) {
                                    return Err(PathError::Loop { index });
                                }
                                if expansions >= max_expansions {
                                    return Err(PathError::Limit { index });
                                }
                                expansions += 1;
                                active.push(index);
                                let target = &links[index].target;
                                work.push(Work::Target(Vec::new(), index));
                                work.push(Work::Target(raw_components(target, 0), index));
                            }
                        }
                    }
                }
            }
        }
    }

    let _ = absolute;
    Ok(render_absolute(&stack, intent))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home, &[], 0)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home, links, max_expansions)
}

fn components_of_absolute(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_string(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = components_of_absolute(&resolved_base);
    let target_components = components_of_absolute(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probed) = probe else {
            continue;
        };
        let alias_components = components_of_absolute(&probed);
        if alias_components.len() <= target_components.len()
            && alias_components
                .iter()
                .zip(target_components.iter())
                .all(|(a, b)| a == b)
        {
            let mut candidate = route(&base_components, &components_of_absolute(&link.path));
            let suffix = &target_components[alias_components.len()..];
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }
        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if components_of_absolute(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Empty)
}
