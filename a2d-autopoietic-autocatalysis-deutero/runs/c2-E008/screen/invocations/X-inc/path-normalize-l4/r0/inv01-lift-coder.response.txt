use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn render_absolute(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return "/".to_owned();
    }
    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn render_normalized(parts: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return render_absolute(parts, intent);
    }
    if parts.is_empty() {
        return ".".to_owned();
    }
    let mut result = parts.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn process_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<bool, PathError> {
    match component {
        "" | "." => Ok(false),
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_owned());
            }
            Ok(false)
        }
        _ => {
            stack.push(component.to_owned());
            Ok(true)
        }
    }
}

fn normalize_impl(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    let mut offset = 0;
    for component in path.split('/') {
        if !component.is_empty() {
            process_component(&mut stack, component, absolute, offset)?;
        }
        offset += component.len() + 1;
    }

    Ok(render_normalized(&stack, absolute, intent))
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    normalize_impl(path)
}

fn normalized_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize_impl(path) {
        Ok(normalized) => Ok(components(&normalized)),
        Err(_) => Err(error),
    }
}

fn expand_tilde<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(Option<&'a str>, &'a str, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((None, path, 0));
    }
    if path.len() == 1 {
        return home
            .map(|h| (Some(h), "", 1))
            .ok_or(PathError::MissingHome);
    }
    if path.as_bytes()[1] == b'/' {
        return home
            .map(|h| (Some(h), &path[2..], 2))
            .ok_or(PathError::MissingHome);
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(normalized_context(h, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let (tilde_home, remainder, _) = expand_tilde(path, home)?;

    let (mut stack, request, request_offset, absolute) = if let Some(h) = tilde_home {
        (components(h), remainder, path.len() - remainder.len(), true)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_parts, path, 0, true)
    };

    if tilde_home.is_some() {
        if let Some(parts) = home_parts {
            stack = parts;
        }
    }

    let mut offset = request_offset;
    for component in request.split('/') {
        if !component.is_empty() {
            process_component(&mut stack, component, absolute, offset)?;
        }
        offset += component.len() + 1;
    }

    Ok(render_absolute(&stack, directory_intent(path)))
}

#[derive(Clone)]
struct WorkComponent {
    value: String,
    at: usize,
}

struct Traversal {
    stack: Vec<String>,
    active: Vec<usize>,
    expansions: usize,
}

fn link_table(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut table = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize_impl(&link.path).is_ok_and(|n| n == link.path);
        let valid_target = validate_raw(&link.target).is_ok();
        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        table.insert(link.path.clone(), index);
    }
    Ok(table)
}

fn stack_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn run_components(
    traversal: &mut Traversal,
    mut queue: Vec<WorkComponent>,
    links: &[Link],
    table: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<(), PathError> {
    while let Some(item) = queue.first().cloned() {
        queue.remove(0);

        match item.value.as_str() {
            "" | "." => continue,
            ".." => {
                if traversal.stack.pop().is_none() {
                    return Err(PathError::Escape { at: item.at });
                }
                continue;
            }
            _ => {}
        }

        traversal.stack.push(item.value);
        let key = stack_path(&traversal.stack);
        let Some(&index) = table.get(&key) else {
            continue;
        };

        if traversal.active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if traversal.expansions >= max_expansions {
            return Err(PathError::Limit { index });
        }
        traversal.expansions += 1;
        traversal.active.push(index);

        traversal.stack.pop();
        let target = &links[index].target;
        let mut target_queue = Vec::new();
        let mut offset = 0;
        for component in target.split('/') {
            if !component.is_empty() {
                target_queue.push(WorkComponent {
                    value: component.to_owned(),
                    at: offset,
                });
            }
            offset += component.len() + 1;
        }

        if target.starts_with('/') {
            traversal.stack.clear();
        }
        target_queue.extend(queue);
        run_components(traversal, target_queue, links, table, max_expansions)?;
        traversal.active.pop();

        // The recursive call consumed the remaining queue as well.
        return Ok(());
    }
    Ok(())
}

fn prepare_request(path: &str, home: Option<&str>) -> Result<(String, usize, bool), PathError> {
    let (tilde_home, remainder, consumed) = expand_tilde(path, home)?;
    if let Some(home) = tilde_home {
        Ok((home.to_owned() + "/" + remainder, consumed, true))
    } else {
        Ok((path.to_owned(), 0, path.starts_with('/')))
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(normalized_context(h, PathError::InvalidHome)?),
        None => None,
    };
    let table = link_table(links)?;
    validate_raw(path)?;

    let (tilde_home, remainder, consumed) = if path.starts_with('~') {
        let (h, r, c) = expand_tilde(path, home)?;
        (h, r, c)
    } else {
        (None, path, 0)
    };

    let absolute = path.starts_with('/') || tilde_home.is_some();
    let mut traversal = Traversal {
        stack: if absolute {
            if let Some(h) = tilde_home {
                components(h)
            } else {
                Vec::new()
            }
        } else {
            cwd_parts
        },
        active: Vec::new(),
        expansions: 0,
    };

    if absolute && tilde_home.is_some() {
        if let Some(parts) = home_parts {
            traversal.stack = parts;
        }
    }

    let mut queue = Vec::new();
    let mut offset = consumed;
    for component in remainder.split('/') {
        if !component.is_empty() {
            queue.push(WorkComponent {
                value: component.to_owned(),
                at: offset,
            });
        }
        offset += component.len() + 1;
    }

    run_components(&mut traversal, queue, links, &table, max_expansions)?;
    Ok(render_absolute(
        &traversal.stack,
        directory_intent(path),
    ))
}

fn components_of_absolute(path: &str) -> Vec<String> {
    components(path)
}

fn route(base: &[String], target: &[String]) -> String {
    let mut common = 0;
    while common < base.len()
        && common < target.len()
        && base[common] == target[common]
    {
        common += 1;
    }
    let mut result = vec!["..".to_owned(); base.len() - common];
    result.extend(target[common..].iter().cloned());
    if result.is_empty() {
        ".".to_owned()
    } else {
        result.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = components_of_absolute(&resolved_base);
    let target_parts = components_of_absolute(&resolved_target);

    let mut candidates = vec![route(&base_parts, &target_parts)];
    for link in links {
        let Ok(probed) = resolve_links(&link.path, "/", None, links, max_expansions) else {
            continue;
        };
        let alias_parts = components_of_absolute(&probed);
        if alias_parts.len() <= target_parts.len()
            && alias_parts
                .iter()
                .zip(target_parts.iter())
                .all(|(a, b)| a == b)
        {
            let mut candidate = route(&base_parts, &components(&link.path));
            let suffix = &target_parts[alias_parts.len()..];
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let target_is_dir = resolved_target.ends_with('/');
    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if candidate != "." && target_is_dir && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let Ok(result) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) else {
            continue;
        };
        if components(&result) == target_parts {
            valid.push(candidate);
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
