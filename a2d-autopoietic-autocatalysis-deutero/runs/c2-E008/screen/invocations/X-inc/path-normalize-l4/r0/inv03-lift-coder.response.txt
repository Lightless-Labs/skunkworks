use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn format_components(components: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    } else if components.is_empty() {
        ".".to_owned()
    } else {
        components.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce_components(
    components: &mut Vec<String>,
    raw: &str,
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let mut cursor = 0;
    for component in raw.split('/') {
        let start = cursor;
        cursor += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if components.last().is_some_and(|last| last != "..") {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape { at: offset + start });
            } else {
                components.push("..".to_owned());
            }
        } else {
            components.push(component.to_owned());
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut components = Vec::new();
    reduce_components(&mut components, path, 0, absolute)?;
    Ok(format_components(&components, absolute, intent))
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    let normalized = normalize(path).map_err(|_| error)?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

fn expand_tilde<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(Option<&'a str>, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((Some(path), 0));
    }

    let bytes = path.as_bytes();
    if bytes.len() == 1 || bytes[1] == b'/' {
        if home.is_none() {
            return Err(PathError::MissingHome);
        }
        let consumed = if bytes.len() > 1 { 2 } else { 1 };
        return Ok((Some(&path[consumed..]), consumed));
    }

    Err(PathError::UnsupportedTilde { at: 1 })
}

fn resolve_plain(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);
    let (remainder, consumed) = expand_tilde(path, home)?;
    let remainder = remainder.unwrap_or("");
    let tilde_expanded = consumed != 0;
    let absolute = !tilde_expanded && path.starts_with('/');

    let mut components = if absolute {
        Vec::new()
    } else if tilde_expanded {
        home_components.unwrap_or_default()
    } else {
        cwd_components
    };

    reduce_components(&mut components, remainder, consumed, true)?;
    Ok(format_components(&components, true, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_plain(path, cwd, home)
}

fn canonical_components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut indices = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();

        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

    Ok(indices)
}

#[derive(Clone)]
struct Work {
    raw: String,
    offset: usize,
    cursor: usize,
    active_link: Option<usize>,
    is_context: bool,
}

impl Work {
    fn new(raw: &str, offset: usize, is_context: bool) -> Self {
        Self {
            raw: raw.to_owned(),
            offset,
            cursor: 0,
            active_link: None,
            is_context,
        }
    }

    fn next_component(&mut self) -> Option<(String, usize)> {
        while self.cursor <= self.raw.len() {
            if self.cursor == self.raw.len() {
                self.cursor += 1;
                return None;
            }

            let rest = &self.raw[self.cursor..];
            let start = self.cursor;
            let len = rest.find('/').unwrap_or(rest.len());
            let end = start + len;
            self.cursor = if end == self.raw.len() { end + 1 } else { end + 1 };

            let component = &self.raw[start..end];
            if !component.is_empty() {
                return Some((component.to_owned(), self.offset + start));
            }
        }
        None
    }
}

fn absolute_key(components: &[String]) -> String {
    if components.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", components.join("/"))
    }
}

fn traverse_links(
    initial: Vec<String>,
    work: Vec<Work>,
    links: &[Link],
    keys: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    let mut work = work;
    let mut active = Vec::new();
    let mut expansions = 0usize;

    while let Some(item) = work.last_mut() {
        let Some((component, at)) = item.next_component() else {
            let finished = work.pop().unwrap();
            if let Some(index) = finished.active_link {
                if active.last() == Some(&index) {
                    active.pop();
                } else if let Some(position) = active.iter().rposition(|&n| n == index) {
                    active.remove(position);
                }
            }
            continue;
        };

        if component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some() {
                stack.pop();
            } else {
                return Err(PathError::Escape { at });
            }
            continue;
        }

        stack.push(component);
        let key = absolute_key(&stack);
        if let Some(&index) = keys.get(&key) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);

            let link_parent = {
                let mut parent = stack.clone();
                parent.pop();
                parent
            };
            stack.pop();

            let target = &links[index].target;
            let target_absolute = target.starts_with('/');
            if target_absolute {
                stack.clear();
            } else {
                stack = link_parent;
            }

            let target_work = Work {
                raw: target.clone(),
                offset: 0,
                cursor: 0,
                active_link: Some(index),
                is_context: false,
            };
            work.push(target_work);
        }
    }

    Ok(stack)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home, PathError::InvalidHome)?),
        None => None,
    };
    let keys = validate_links(links)?;

    validate_raw(path)?;
    let intent = directory_intent(path);
    let (remainder, consumed) = expand_tilde(path, home)?;
    let remainder = remainder.unwrap_or("");
    let tilde_expanded = consumed != 0;
    let absolute = !tilde_expanded && path.starts_with('/');

    let initial = if absolute {
        Vec::new()
    } else if tilde_expanded {
        home_components.unwrap_or_default()
    } else {
        cwd_components
    };

    let mut work = Vec::new();
    if !absolute {
        let context = if tilde_expanded {
            home.unwrap_or("")
        } else {
            cwd
        };
        let context = normalize(context).map_err(|_| {
            if tilde_expanded {
                PathError::InvalidHome
            } else {
                PathError::InvalidCwd
            }
        })?;
        let context_components = canonical_components(&context);
        work.push(Work::new(&context_components.join("/"), 0, true));
    }
    work.push(Work::new(remainder, consumed, false));

    let result = traverse_links(
        if absolute { Vec::new() } else { Vec::new() },
        work,
        links,
        &keys,
        max_expansions,
    )?;

    Ok(format_components(&result, true, intent))
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;

    let base_components = canonical_components(&resolved_base);
    let target_components = canonical_components(&resolved_target);

    let common = base_components
        .iter()
        .zip(&target_components)
        .take_while(|(left, right)| left == right)
        .count();

    let mut route = Vec::new();
    route.extend((common..base_components.len()).map(|_| "..".to_owned()));
    route.extend(target_components[common..].iter().cloned());

    let direct = if route.is_empty() {
        ".".to_owned()
    } else {
        route.join("/")
    };

    let target_is_directory = resolved_target.ends_with('/');
    let mut candidates = vec![direct];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let prefix = canonical_components(&probe);
        if prefix.len() > target_components.len()
            || prefix
                .iter()
                .zip(&target_components)
                .any(|(left, right)| left != right)
        {
            continue;
        }

        let key_components = canonical_components(&link.path);
        let key_common = base_components
            .iter()
            .zip(&key_components)
            .take_while(|(left, right)| left == right)
            .count();
        let mut shortcut = Vec::new();
        shortcut.extend((key_common..base_components.len()).map(|_| "..".to_owned()));
        shortcut.extend(key_components[key_common..].iter().cloned());

        if shortcut.is_empty() {
            shortcut.push(".".to_owned());
        }
        let suffix = &target_components[prefix.len()..];
        if !suffix.is_empty() {
            if shortcut.as_slice() == [String::from(".")] {
                shortcut.clear();
            }
            shortcut.extend(suffix.iter().cloned());
        }
        let candidate = if shortcut.is_empty() {
            ".".to_owned()
        } else {
            shortcut.join("/")
        };
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_is_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if canonical_components(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|left, right| {
        left.len()
            .cmp(&right.len())
            .then_with(|| left.as_bytes().cmp(right.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Empty)
}
