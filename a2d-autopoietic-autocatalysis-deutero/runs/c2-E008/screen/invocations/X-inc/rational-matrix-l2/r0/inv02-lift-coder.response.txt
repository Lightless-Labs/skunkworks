use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (num, den) = if den < 0 { (-num, -den) } else { (num, den) };
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let divisor = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    let reduced_num = num / divisor;
    let reduced_den = den / divisor;

    if reduced_num < i64::MIN as i128
        || reduced_num > i64::MAX as i128
        || reduced_den < 1
        || reduced_den > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: reduced_num as i64,
        den: reduced_den as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        let divisor = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        rational_from_i128(n / divisor, d / divisor)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(parse_error(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if b.is_ascii_digit() || (b == b'-' && i == 0) {
                // Valid grammar byte.
            } else {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let numerator_end = slash.unwrap_or(bytes.len());
        let numerator_start = usize::from(bytes[0] == b'-');
        if numerator_start == numerator_end {
            return Err(parse_error(text.len(), ParseKind::InvalidByte));
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(parse_error(text.len(), ParseKind::InvalidByte));
            }
        }

        // The scan above validates all present bytes; ensure numerator and denominator
        // contain only digits in their required positions.
        for i in numerator_start..numerator_end {
            if !bytes[i].is_ascii_digit() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }
        if let Some(s) = slash {
            for i in s + 1..bytes.len() {
                if !bytes[i].is_ascii_digit() {
                    return Err(parse_error(i, ParseKind::InvalidByte));
                }
            }
        }

        let num_digits = numerator_end - numerator_start;
        if num_digits >= 2 && bytes[numerator_start] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == 1 && bytes[numerator_start] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let Some(s) = slash {
            if bytes[s + 1] == b'0' && bytes.len() - s - 1 >= 2 {
                return Err(parse_error(s + 2, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0u128;
        for &b in &bytes[numerator_start..numerator_end] {
            numerator_magnitude = numerator_magnitude
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
            if numerator_magnitude > limit {
                return Err(parse_error(0, ParseKind::OutOfRange));
            }
        }

        let mut denominator_value = 1u128;
        if let Some(s) = slash {
            denominator_value = 0;
            for &b in &bytes[s + 1..] {
                denominator_value = denominator_value
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
                if denominator_value > i64::MAX as u128 {
                    return Err(parse_error(s + 1, ParseKind::OutOfRange));
                }
            }
            if denominator_value == 0 {
                return Err(parse_error(s + 1, ParseKind::ZeroDenominator));
            }
            if denominator_value == 1 {
                return Err(parse_error(s, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_error(s, ParseKind::NonCanonical));
            }

            if gcd_u128(numerator_magnitude, denominator_value) > 1 {
                return Err(parse_error(s, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if numerator_magnitude == 1u128 << 63 {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };
        let denominator = denominator_value as i64;
        Ok(Self {
            num: signed_num,
            den: denominator,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero;
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
