#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<u8>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.values[field_index(field)].clone()
    }
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

const DOMAINS: [(u8, u8); 5] = [(0, 59), (0, 23), (1, 31), (1, 12), (0, 6)];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
    step_max: u32,
    domain_min: u8,
    domain_max: u8,
) -> Result<u32, ParseError> {
    let error = |kind| match kind {
        0 => ParseError::LeadingZero { field, offset: start },
        1 => ParseError::NumberOverflow { field, offset: start },
        2 => ParseError::OutOfRange { field, offset: start },
        _ => ParseError::BadStep { field, offset: start },
    };

    if end <= start {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(0));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| error(1))?;
    }

    if is_step {
        if value == 0 || value > step_max {
            return Err(error(3));
        }
    } else if value < domain_min as u32 || value > domain_max as u32 {
        return Err(error(2));
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field, min: u8, max: u8) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; (max - min + 1) as usize];
    let mut item_start = 0;

    for item_end in (0..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_range) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some((*slash + 1, item.len()))),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if is_star {
            0
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            1
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen > 0
                && hyphen + 1 < base.len()
                && base[..hyphen].iter().all(u8::is_ascii_digit)
                && base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                2
            } else {
                -1
            }
        } else {
            -1
        };

        if base_kind < 0 || step_range.is_some_and(|(s, e)| s == e) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (start, end, end_number_offset) = match base_kind {
            0 => (min as u32, max as u32, item_start),
            1 => {
                let n = parse_number(
                    item,
                    0,
                    base.len(),
                    field,
                    false,
                    0,
                    min,
                    max,
                )?;
                (n, if step_range.is_some() { max as u32 } else { n }, item_start)
            }
            _ => {
                let hyphen = hyphens[0];
                let start_num = parse_number(item, 0, hyphen, field, false, 0, min, max)?;
                let end_offset = item_start + hyphen + 1;
                let end_num = parse_number(
                    item,
                    hyphen + 1,
                    base.len(),
                    field,
                    false,
                    0,
                    min,
                    max,
                )?;
                if start_num > end_num {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
                (start_num, end_num, end_offset)
            }
        };

        let step = if let Some((step_start, step_end)) = step_range {
            let step_offset = item_start + step_start;
            parse_number(
                item,
                step_start,
                step_end,
                field,
                true,
                (max - min + 1) as u32,
                min,
                max,
            )
            .map_err(|e| match e {
                ParseError::LeadingZero { .. } => ParseError::LeadingZero {
                    field,
                    offset: step_offset,
                },
                ParseError::NumberOverflow { .. } => ParseError::NumberOverflow {
                    field,
                    offset: step_offset,
                },
                ParseError::BadStep { .. } => ParseError::BadStep {
                    field,
                    offset: step_offset,
                },
                other => other,
            })?
        } else {
            1
        };

        let _ = end_number_offset;
        let mut value = start;
        while value <= end {
            selected[(value - min as u32) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(i, &yes)| yes.then_some(min + i as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut values: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_stars = [false; 2];

    for i in 0..5 {
        values[i] = parse_field(parts[i], FIELDS[i], DOMAINS[i].0, DOMAINS[i].1)?;
        if i == 2 {
            day_stars[0] = parts[i].as_bytes() == b"*";
        } else if i == 4 {
            day_stars[1] = parts[i].as_bytes() == b"*";
        }
    }

    Ok(Cron { values, day_stars })
}
