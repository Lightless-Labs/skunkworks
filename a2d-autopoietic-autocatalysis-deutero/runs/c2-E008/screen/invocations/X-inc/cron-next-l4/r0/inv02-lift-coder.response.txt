#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn allowed_name_value(field: Field, token: &str) -> Option<u8> {
    let names = names(field)?;
    names
        .iter()
        .position(|name| name.eq_ignore_ascii_case(token))
        .map(|i| {
            if field == Field::Month {
                (i + 1) as u8
            } else {
                i as u8
            }
        })
}

fn is_legal_byte(b: u8) -> bool {
    b.is_ascii_digit()
        || b.is_ascii_alphabetic()
        || matches!(b, b'*' | b'-' | b'/' | b',')
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(
    field: Field,
    token: &str,
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    let bytes = token.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(invalid_syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if is_step {
        let (_, max) = bounds(field);
        let width = (max as u32) + 1 - bounds(field).0 as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = bounds(field);
        if value < min as u32 || value > max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(value)
}

fn parse_endpoint(field: Field, token: &str, offset: usize) -> Result<u8, ParseError> {
    if token.as_bytes().iter().all(u8::is_ascii_digit) {
        return Ok(parse_number(field, token, offset, false)? as u8);
    }
    if let Some(value) = allowed_name_value(field, token) {
        return Ok(value);
    }
    Err(invalid_syntax(field, offset))
}

fn parse_field(field: Field, text: &str) -> Result<(Vec<bool>, bool), ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !is_legal_byte(byte) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; max as usize + 1];
    let unrestricted = text == "*";

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|i| item_start + i)
            .unwrap_or(bytes.len());

        if item_end == item_start {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &text[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .as_bytes()
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base, step_token, step_offset) = match slash_positions.as_slice() {
            [] => (item, None, 0usize),
            [pos] => (&item[..*pos], Some(&item[*pos + 1..]), *pos + 1),
            _ => return Err(invalid_syntax(field, item_start)),
        };

        if base.is_empty() || step_token.is_some_and(str::is_empty) {
            return Err(invalid_syntax(field, item_start));
        }

        let hyphen_positions: Vec<usize> = base
            .as_bytes()
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let (base_kind, left, right, left_offset, right_offset) = match hyphen_positions.as_slice() {
            [] if base == "*" => (0u8, "", "", 0usize, 0usize),
            [] => (1u8, base, "", 0usize, 0usize),
            [pos] => {
                if base.starts_with('*') || base.ends_with('*') {
                    return Err(invalid_syntax(field, item_start));
                }
                (
                    2u8,
                    &base[..*pos],
                    &base[*pos + 1..],
                    0usize,
                    *pos + 1,
                )
            }
            _ => return Err(invalid_syntax(field, item_start)),
        };

        if base_kind == 0 && base != "*" {
            return Err(invalid_syntax(field, item_start));
        }
        if base_kind != 0
            && (left.is_empty()
                || (base_kind == 2 && right.is_empty())
                || left == "*"
                || right == "*")
        {
            return Err(invalid_syntax(field, item_start));
        }

        // Validate the structure and token categories before interpreting values.
        if base_kind != 0 {
            for token in if base_kind == 1 {
                vec![left]
            } else {
                vec![left, right]
            } {
                if !(token.as_bytes().iter().all(u8::is_ascii_digit)
                    || allowed_name_value(field, token).is_some())
                {
                    return Err(invalid_syntax(field, item_start));
                }
            }
        }

        if let Some(step) = step_token {
            if !step.as_bytes().iter().all(u8::is_ascii_digit) || step.is_empty() {
                return Err(invalid_syntax(field, item_start));
            }
        }

        let mut start_value = min;
        let mut end_value = max;

        if base_kind != 0 {
            start_value = parse_endpoint(field, left, item_start + left_offset)?;
            end_value = if base_kind == 2 {
                parse_endpoint(field, right, item_start + right_offset)?
            } else {
                start_value
            };
            if base_kind == 2 && start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + right_offset,
                });
            }
        }

        let step = if let Some(token) = step_token {
            Some(parse_number(field, token, item_start + step_offset, true)? as u8)
        } else {
            None
        };

        if step.is_none() && base_kind == 1 {
            selected[start_value as usize] = true;
        } else {
            let stride = step.unwrap_or(1);
            let mut value = start_value as u16;
            let end = end_value as u16;
            while value <= end {
                selected[value as usize] = true;
                value += stride as u16;
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
        if item_start == bytes.len() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }
    }

    Ok((selected, unrestricted))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.starts_with('@') {
        match input {
            "@hourly" => Some("0 * * * *"),
            "@daily" => Some("0 0 * * *"),
            "@weekly" => Some("0 0 * * 0"),
            "@monthly" => Some("0 0 1 * *"),
            "@yearly" => Some("0 0 1 1 *"),
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        None
    };
    let source = expanded.unwrap_or(input);
    let parts: Vec<&str> = source.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (_, max) = bounds(FIELDS[i]);
        vec![false; max as usize + 1]
    });
    let mut flags = [false; 5];
    for i in 0..5 {
        let (set, unrestricted) = parse_field(FIELDS[i], parts[i])?;
        sets[i] = set;
        flags[i] = unrestricted;
    }

    Ok(Cron {
        sets,
        unrestricted_dom: flags[2],
        unrestricted_dow: flags[4],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

    fn has(&self, field: Field, value: u8) -> bool {
        self.sets[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn date_matches(&self, date: CivilDateTime, weekday: u8) -> bool {
        let dom = self.has(Field::DayOfMonth, date.day);
        let dow = self.has(Field::DayOfWeek, weekday);
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if date.day == days_in_month(date.year, date.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.has(Field::Minute, at.minute)
            && self.has(Field::Hour, at.hour)
            && self.has(Field::Month, at.month)
            && self.date_matches(at, weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        Ok(self.search_forward(after, false))
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        Ok(self.search_backward(before, false))
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(fire) = self.search_forward(cursor, true) {
            if fire >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(fire);
            cursor = fire;
        }

        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for field in FIELDS {
            let values = self.allowed(field);
            let (_, max) = bounds(field);
            let full = values.len() == (max as usize + 1 - bounds(field).0 as usize)
                && values.first().copied() == Some(bounds(field).0)
                && values.last().copied() == Some(max);

            if field == Field::DayOfMonth && self.unrestricted_dom {
                fields.push("*".to_string());
            } else if field == Field::DayOfWeek && self.unrestricted_dow {
                fields.push("*".to_string());
            } else if full {
                if matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                    fields.push("*/1".to_string());
                } else {
                    fields.push("*".to_string());
                }
            } else {
                fields.push(
                    values
                        .iter()
                        .map(u8::to_string)
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        fields.join(" ")
    }

    fn search_forward(&self, from: CivilDateTime, inclusive: bool) -> Option<CivilDateTime> {
        let mut date = CivilDateTime {
            year: from.year,
            month: from.month,
            day: from.day,
            hour: 0,
            minute: 0,
        };
        loop {
            if self.has(Field::Month, date.month) {
                let wd = weekday(date.year, date.month, date.day);
                if self.date_matches(date, wd) {
                    for hour in self.allowed(Field::Hour) {
                        if date == CivilDateTime {
                            year: from.year,
                            month: from.month,
                            day: from.day,
                            hour: 0,
                            minute: 0,
                        } && hour < from.hour
                        {
                            continue;
                        }
                        for minute in self.allowed(Field::Minute) {
                            let candidate = CivilDateTime {
                                year: date.year,
                                month: date.month,
                                day: date.day,
                                hour,
                                minute,
                            };
                            if candidate > from || (inclusive && candidate == from) {
                                return Some(candidate);
                            }
                        }
                    }
                }
            }

            if date.year == 2399 && date.month == 12 && date.day == 31 {
                return None;
            }
            date = increment_date(date);
        }
    }

    fn search_backward(&self, from: CivilDateTime, inclusive: bool) -> Option<CivilDateTime> {
        let mut date = CivilDateTime {
            year: from.year,
            month: from.month,
            day: from.day,
            hour: 0,
            minute: 0,
        };
        loop {
            if self.has(Field::Month, date.month) {
                let wd = weekday(date.year, date.month, date.day);
                if self.date_matches(date, wd) {
                    for hour in self.allowed(Field::Hour).into_iter().rev() {
                        for minute in self.allowed(Field::Minute).into_iter().rev() {
                            let candidate = CivilDateTime {
                                year: date.year,
                                month: date.month,
                                day: date.day,
                                hour,
                                minute,
                            };
                            if candidate < from || (inclusive && candidate == from) {
                                return Some(candidate);
                            }
                        }
                    }
                }
            }
            if date.year == 2000 && date.month == 1 && date.day == 1 {
                return None;
            }
            date = decrement_date(date);
        }
    }
}

fn validate_datetime(value: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&value.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&value.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if value.day == 0 || value.day > days_in_month(value.year, value.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if value.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if value.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_date(mut date: CivilDateTime) -> CivilDateTime {
    if date.day < days_in_month(date.year, date.month) {
        date.day += 1;
    } else if date.month < 12 {
        date.month += 1;
        date.day = 1;
    } else {
        date.year += 1;
        date.month = 1;
        date.day = 1;
    }
    date.hour = 0;
    date.minute = 0;
    date
}

fn decrement_date(mut date: CivilDateTime) -> CivilDateTime {
    if date.day > 1 {
        date.day -= 1;
    } else if date.month > 1 {
        date.month -= 1;
        date.day = days_in_month(date.year, date.month);
    } else {
        date.year -= 1;
        date.month = 12;
        date.day = 31;
    }
    date.hour = 0;
    date.minute = 0;
    date
}
