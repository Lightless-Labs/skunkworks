use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [BTreeSet<u8>; 5],
    dom_unrestricted: bool,
    dow_unrestricted: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn month_name(s: &str) -> Option<u8> {
    match s.to_ascii_uppercase().as_str() {
        "JAN" => Some(1),
        "FEB" => Some(2),
        "MAR" => Some(3),
        "APR" => Some(4),
        "MAY" => Some(5),
        "JUN" => Some(6),
        "JUL" => Some(7),
        "AUG" => Some(8),
        "SEP" => Some(9),
        "OCT" => Some(10),
        "NOV" => Some(11),
        "DEC" => Some(12),
        _ => None,
    }
}

fn weekday_name(s: &str) -> Option<u8> {
    match s.to_ascii_uppercase().as_str() {
        "SUN" => Some(0),
        "MON" => Some(1),
        "TUE" => Some(2),
        "WED" => Some(3),
        "THU" => Some(4),
        "FRI" => Some(5),
        "SAT" => Some(6),
        _ => None,
    }
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_decimal(s: &str, field: Field, offset: usize, step: bool) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(invalid_syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n.checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if step {
        let (lo, hi) = domain(field);
        let width = (hi - lo + 1) as u32;
        if n == 0 || n > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (lo, hi) = domain(field);
        if n < lo as u32 || n > hi as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(n)
}

fn parse_atom(s: &str, field: Field, offset: usize) -> Result<(u8, bool), ParseError> {
    if s.as_bytes().iter().all(u8::is_ascii_digit) && !s.is_empty() {
        return Ok((parse_decimal(s, field, offset, false)? as u8, true));
    }
    let named = match field {
        Field::Month => month_name(s),
        Field::DayOfWeek => weekday_name(s),
        _ => None,
    };
    named.map(|v| (v, false)).ok_or(invalid_syntax(field, offset))
}

fn parse_item(item: &str, field: Field, item_offset: usize, out: &mut BTreeSet<u8>) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes.iter().enumerate().filter_map(|(i, &b)| (b == b'/').then_some(i)).collect();
    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }

    let (base, step_part, step_offset) = if let Some(&p) = slash_positions.first() {
        if p == 0 || p + 1 == bytes.len() {
            return Err(invalid_syntax(field, item_offset));
        }
        (&item[..p], Some(&item[p + 1..]), item_offset + p + 1)
    } else {
        (item, None, item_offset)
    };

    let hyphens: Vec<usize> = base.as_bytes().iter().enumerate().filter_map(|(i, &b)| (b == b'-').then_some(i)).collect();
    if hyphens.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }

    let (is_star, left, right, right_offset) = if base == "*" {
        (true, None, None, 0)
    } else if let Some(&p) = hyphens.first() {
        if p == 0 || p + 1 == base.len() || base[..p] == "*" || base[p + 1..] == "*" {
            return Err(invalid_syntax(field, item_offset));
        }
        (false, Some(&base[..p]), Some(&base[p + 1..]), item_offset + p + 1)
    } else {
        if base.is_empty() || base.contains('*') {
            return Err(invalid_syntax(field, item_offset));
        }
        (false, Some(base), None, 0)
    };

    if is_star && step_part.is_none() {
        let (lo, hi) = domain(field);
        out.extend(lo..=hi);
        return Ok(());
    }

    let step = match step_part {
        Some(s) => Some(parse_decimal(s, field, step_offset, true)? as u8),
        None => None,
    };

    let (start, end) = if is_star {
        domain(field)
    } else {
        let (a, _) = parse_atom(left.unwrap(), field, item_offset)?;
        if let Some(r) = right {
            let (b, _) = parse_atom(r, field, right_offset)?;
            if a > b {
                return Err(ParseError::ReversedRange { field, offset: right_offset });
            }
            (a, b)
        } else {
            let (_, hi) = domain(field);
            (a, if step.is_some() { hi } else { a })
        }
    };

    if let Some(step) = step {
        let mut v = start;
        while v <= end {
            out.insert(v);
            match v.checked_add(step) {
                Some(next) => v = next,
                None => break,
            }
        }
    } else {
        out.extend(start..=end);
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<BTreeSet<u8>, ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut set = BTreeSet::new();
    let mut start = 0usize;
    for (i, part) in text.split(',').enumerate() {
        let offset = start;
        if part.is_empty() {
            return Err(ParseError::EmptyItem { field, offset });
        }
        parse_item(part, field, offset, &mut set)?;
        start += part.len() + if i + 1 < text.split(',').count() { 1 } else { 0 };
    }
    Ok(set)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let mut sets: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    for i in 0..5 {
        sets[i] = parse_field(parts[i], FIELDS[i])?;
    }
    Ok(Cron {
        sets,
        dom_unrestricted: parts[2] == "*",
        dow_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].iter().copied().collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.sets[0].contains(&at.minute)
            || !self.sets[1].contains(&at.hour)
            || !self.sets[3].contains(&at.month)
        {
            return false;
        }
        let dom = self.sets[2].contains(&at.day);
        let dow = self.sets[4].contains(&weekday(at));
        match (self.dom_unrestricted, self.dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = CivilDateTime { hour: 0, minute: 0, ..after };
        loop {
            for &hour in self.sets[1].range((after.hour.min(23))..) {
                for &minute in &self.sets[0] {
                    let candidate = CivilDateTime { hour, minute, ..date };
                    if candidate > after && self.matches_valid(candidate) {
                        return Ok(Some(candidate));
                    }
                }
            }
            if !advance_day(&mut date) {
                return Ok(None);
            }
            date.hour = 0;
            date.minute = 0;
            if self.sets[1].is_empty() || self.sets[0].is_empty() {
                return Ok(None);
            }
            for &hour in &self.sets[1] {
                for &minute in &self.sets[0] {
                    let candidate = CivilDateTime { hour, minute, ..date };
                    if self.matches_valid(candidate) {
                        return Ok(Some(candidate));
                    }
                }
            }
            // The first day after `after` was checked above; move on from it.
            while advance_day(&mut date) {
                for &hour in &self.sets[1] {
                    for &minute in &self.sets[0] {
                        let candidate = CivilDateTime { hour, minute, ..date };
                        if self.matches_valid(candidate) {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            return Ok(None);
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = CivilDateTime { hour: 0, minute: 0, ..before };
        loop {
            for &hour in self.sets[1].iter().rev() {
                for &minute in self.sets[0].iter().rev() {
                    let candidate = CivilDateTime { hour, minute, ..date };
                    if candidate < before && self.matches_valid(candidate) {
                        return Ok(Some(candidate));
                    }
                }
            }
            if !retreat_day(&mut date) {
                return Ok(None);
            }
            date.hour = 0;
            date.minute = 0;
            loop {
                for &hour in self.sets[1].iter().rev() {
                    for &minute in self.sets[0].iter().rev() {
                        let candidate = CivilDateTime { hour, minute, ..date };
                        if self.matches_valid(candidate) {
                            return Ok(Some(candidate));
                        }
                    }
                }
                if !retreat_day(&mut date) {
                    return Ok(None);
                }
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(next) = self.next_fire(cursor).map_err(|_| WindowError::InvalidStart {
            component: DatePart::Year,
        })? {
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow { fires, truncated: true });
            }
            fires.push(next);
            cursor = next;
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for i in 0..5 {
            let field = FIELDS[i];
            let (lo, hi) = domain(field);
            let full = self.sets[i].len() == (hi - lo + 1) as usize;
            if full && (i == 0 || i == 1 || i == 3) {
                fields.push("*".to_string());
            } else if i == 2 && self.dom_unrestricted {
                fields.push("*".to_string());
            } else if i == 4 && self.dow_unrestricted {
                fields.push("*".to_string());
            } else if full && (i == 2 || i == 4) {
                fields.push("*/1".to_string());
            } else {
                fields.push(
                    self.sets[i]
                        .iter()
                        .map(u8::to_string)
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        fields.join(" ")
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if leap(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += at.day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn advance_day(at: &mut CivilDateTime) -> bool {
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
    } else if at.month < 12 {
        at.month += 1;
        at.day = 1;
    } else if at.year < 2399 {
        at.year += 1;
        at.month = 1;
        at.day = 1;
    } else {
        return false;
    }
    true
}

fn retreat_day(at: &mut CivilDateTime) -> bool {
    if at.day > 1 {
        at.day -= 1;
    } else if at.month > 1 {
        at.month -= 1;
        at.day = days_in_month(at.year, at.month);
    } else if at.year > 2000 {
        at.year -= 1;
        at.month = 12;
        at.day = 31;
    } else {
        return false;
    }
    true
}
