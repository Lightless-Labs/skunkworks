use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Clone, Copy)]
struct Domain {
    min: u8,
    max: u8,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn parse_error(field: Field, offset: usize, which: u8) -> ParseError {
    match which {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn parse_number(bytes: &[u8], start: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, start, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, start, 1));
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, start, 2))?;
    }
    Ok(value)
}

fn name_value(bytes: &[u8], field: Field) -> Option<u32> {
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    let names: &[&[u8]] = match field {
        Field::Month => &[
            b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN",
            b"JUL", b"AUG", b"SEP", b"OCT", b"NOV", b"DEC",
        ],
        Field::DayOfWeek => &[b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT"],
        _ => return None,
    };
    names.iter().position(|n| *n == upper.as_slice()).map(|i| {
        if field == Field::Month {
            (i + 1) as u32
        } else {
            i as u32
        }
    })
}

fn valid_atom(bytes: &[u8], field: Field) -> bool {
    !bytes.is_empty()
        && (bytes.iter().all(u8::is_ascii_digit) || name_value(bytes, field).is_some())
}

fn atom_value(bytes: &[u8], start: usize, field: Field) -> Result<u32, ParseError> {
    let value = if bytes.iter().all(u8::is_ascii_digit) {
        parse_number(bytes, start, field)?
    } else {
        name_value(bytes, field).ok_or_else(|| parse_error(field, start, 0))?
    };
    let d = domain(field);
    if value < d.min as u32 || value > d.max as u32 {
        return Err(parse_error(field, start, 3));
    }
    Ok(value)
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, b)| (*b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(parse_error(field, item_offset, 0));
    }
    let (base, step_part, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, 0)
    };

    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, b)| (*b == b'-').then_some(i))
        .collect();

    let range_parts = if is_star {
        None
    } else if hyphens.is_empty() {
        if !valid_atom(base, field) {
            return Err(parse_error(field, item_offset, 0));
        }
        None
    } else if hyphens.len() == 1 {
        let pos = hyphens[0];
        let left = &base[..pos];
        let right = &base[pos + 1..];
        if !valid_atom(left, field) || !valid_atom(right, field) {
            return Err(parse_error(field, item_offset, 0));
        }
        Some((left, right, pos))
    } else {
        return Err(parse_error(field, item_offset, 0));
    };

    if let Some(step_bytes) = step_part {
        if step_bytes.is_empty() || !step_bytes.iter().all(u8::is_ascii_digit) {
            return Err(parse_error(field, item_offset, 0));
        }
    }

    let d = domain(field);
    let (start, end, end_offset) = if is_star {
        (d.min as u32, d.max as u32, item_offset)
    } else if let Some((left, right, hyphen_pos)) = range_parts {
        let start = atom_value(left, item_offset, field)?;
        let right_start = item_offset + hyphen_pos + 1;
        let end = atom_value(right, right_start, field)?;
        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: right_start,
            });
        }
        (start, end, right_start)
    } else {
        let value = atom_value(base, item_offset, field)?;
        let end = if step_part.is_some() {
            d.max as u32
        } else {
            value
        };
        (value, end, item_offset)
    };

    let step = if let Some(step_bytes) = step_part {
        let n = parse_number(step_bytes, step_offset, field)?;
        let width = (d.max - d.min + 1) as u32;
        if n == 0 || n > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        n
    } else {
        1
    };

    let _ = end_offset;
    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let source = if input.as_bytes().first() == Some(&b'@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let parts: Vec<&str> = source.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let d = domain(FIELDS[i]);
        vec![false; d.max as usize + 1]
    });
    let mut unrestricted_days = [false; 2];

    for (i, text) in parts.iter().enumerate() {
        let field = FIELDS[i];
        let bytes = text.as_bytes();
        for (offset, &b) in bytes.iter().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if field == Field::DayOfMonth && *text == "*" {
            unrestricted_days[0] = true;
        }
        if field == Field::DayOfWeek && *text == "*" {
            unrestricted_days[1] = true;
        }

        let mut offset = 0;
        for item in text.split(',') {
            parse_item(item.as_bytes(), offset, field, &mut selected[i])?;
            offset += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(i as u8))
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.selected[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn day_matches(&self, year: u16, month: u8, day: u8) -> bool {
        let dom = self.contains(Field::DayOfMonth, day);
        let dow = self.contains(Field::DayOfWeek, weekday(year, month, day));
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if day == days_in_month(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn time_matches(&self, at: CivilDateTime) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.day_matches(at.year, at.month, at.day)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.time_matches(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut date = (after.year, after.month, after.day);
        loop {
            let (year, month, day) = date;
            if self.contains(Field::Month, month) && self.day_matches(year, month, day) {
                for hour in self.allowed(Field::Hour) {
                    for minute in self.allowed(Field::Minute) {
                        let candidate = CivilDateTime {
                            year,
                            month,
                            day,
                            hour,
                            minute,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            match next_date(year, month, day) {
                Some(next) => date = next,
                None => return Ok(None),
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut date = (before.year, before.month, before.day);
        loop {
            let (year, month, day) = date;
            if self.contains(Field::Month, month) && self.day_matches(year, month, day) {
                for hour in self.allowed(Field::Hour).into_iter().rev() {
                    for minute in self.allowed(Field::Minute).into_iter().rev() {
                        let candidate = CivilDateTime {
                            year,
                            month,
                            day,
                            hour,
                            minute,
                        };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            match previous_date(year, month, day) {
                Some(prev) => date = prev,
                None => return Ok(None),
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(next) = self.next_fire(cursor).map_err(|_| unreachable!())? {
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(next);
            cursor = next;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for field in FIELDS {
            let d = domain(field);
            let values = self.allowed(field);
            let full: Vec<u8> = (d.min..=d.max).collect();
            let text = if field == Field::DayOfMonth && self.unrestricted_days[0] {
                "*".to_string()
            } else if field == Field::DayOfWeek && self.unrestricted_days[1] {
                "*".to_string()
            } else if values == full {
                match field {
                    Field::DayOfMonth | Field::DayOfWeek => "*/1".to_string(),
                    _ => "*".to_string(),
                }
            } else {
                values
                    .iter()
                    .map(u8::to_string)
                    .collect::<Vec<_>>()
                    .join(",")
            };
            fields.push(text);
        }
        fields.join(" ")
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn next_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day < days_in_month(year, month) {
        Some((year, month, day + 1))
    } else if month < 12 {
        Some((year, month + 1, 1))
    } else if year < 2399 {
        Some((year + 1, 1, 1))
    } else {
        None
    }
}

fn previous_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day > 1 {
        Some((year, month, day - 1))
    } else if month > 1 {
        let prev_month = month - 1;
        Some((year, prev_month, days_in_month(year, prev_month)))
    } else if year > 2000 {
        Some((year - 1, 12, 31))
    } else {
        None
    }
}
