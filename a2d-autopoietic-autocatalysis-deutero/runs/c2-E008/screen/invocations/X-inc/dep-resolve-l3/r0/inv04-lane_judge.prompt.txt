fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn satisfies(v: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|&c| matches(v, c))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", b));
        }
    }
    out
}

fn collect_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    assigned: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (package_name, version) in assigned {
        if let Some(package) = catalog.iter().find(|p| p.name == *package_name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_leaf_error: Option<ResolveError>,
    limited: bool,
}

impl<'a> Search<'a> {
    fn visit(&mut self, assigned: &mut Vec<(String, Version)>) {
        if self.limited {
            return;
        }

        let active = collect_constraints(self.roots, self.catalog, assigned);

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .next()
        {
            self.record_error(ResolveError::Missing {
                package: name.clone(),
            });
            return;
        }

        let mut sorted_assigned = assigned.clone();
        sorted_assigned.sort_by(|a, b| a.0.cmp(&b.0));
        for (name, version) in &sorted_assigned {
            if let Some(constraints) = active.get(name) {
                if !satisfies(*version, constraints) {
                    self.record_error(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                    return;
                }
            }
        }

        let mut unselected: Vec<String> = active
            .keys()
            .filter(|name| !assigned.iter().any(|(n, _)| n == *name))
            .cloned()
            .collect();
        unselected.sort();

        for name in &unselected {
            let constraints = active.get(name).cloned().unwrap_or_default();
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !package
                .releases
                .iter()
                .any(|r| satisfies(r.version, &constraints))
            {
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints,
                });
                return;
            }
        }

        let Some(name) = unselected.first().cloned() else {
            let mut solution = assigned.clone();
            solution.sort_by(|a, b| a.0.cmp(&b.0));
            let cost = self
                .locks
                .iter()
                .filter(|lock| {
                    !solution
                        .iter()
                        .any(|(name, version)| name == &lock.package && *version == lock.version)
                })
                .count();

            let replace = match &self.best {
                None => true,
                Some((best_cost, best_solution)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        self.lex_solution_cmp(&solution, best_solution).is_gt()
                    }
                }
            };
            if replace {
                self.best = Some((cost, solution));
            }
            return;
        };

        let constraints = active.get(&name).cloned().unwrap_or_default();
        let package = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|&v| satisfies(v, &constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limited = true;
                return;
            }
            self.attempts += 1;
            assigned.push((name.clone(), version));
            self.visit(assigned);
            assigned.pop();
            if self.limited {
                return;
            }
        }
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_leaf_error.is_none() {
            self.first_leaf_error = Some(error);
        }
    }

    fn lex_solution_cmp(
        &self,
        left: &[(String, Version)],
        right: &[(String, Version)],
    ) -> std::cmp::Ordering {
        let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        names.sort_unstable();
        names.dedup();

        for name in names {
            let lv = left.iter().find(|(n, _)| n == name).map(|(_, v)| *v);
            let rv = right.iter().find(|(n, _)| n == name).map(|(_, v)| *v);
            let ord = match (lv, rv) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(a), Some(b)) => a.cmp(&b),
            };
            if !ord.is_eq() {
                return ord;
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_leaf_error: None,
        limited: false,
    };
    let mut assigned = Vec::new();
    search.visit(&mut assigned);

    if search.limited {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, solution)) = search.best else {
        return Err(search.first_leaf_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = collect_constraints(roots, catalog, &solution);
    let mut before = std::collections::BTreeMap::<String, Version>::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let after: std::collections::BTreeMap<String, Version> = solution.iter().cloned().collect();

    let mut names: std::collections::BTreeSet<String> = before.keys().cloned().collect();
    names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&package)
                    .map(|constraints| !satisfies(old, constraints))
                    .unwrap_or(false)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

fn canonical_requirements(
    roots: &[Requirement],
    selected: &[(String, Version)],
    catalog: &[Package],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in selected {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn inspect(&mut self, selected: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = canonical_requirements(self.roots, selected, self.catalog);

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let mut branch: Option<String> = None;
        for name in active.keys() {
            if selected.iter().any(|(selected_name, _)| selected_name == name) {
                continue;
            }
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            let constraints = active.get(name).unwrap();
            if !package.releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, release.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
            if branch.is_none() {
                branch = Some(name.clone());
            }
        }

        let Some(name) = branch else {
            let mut solution = selected.clone();
            solution.sort_by(|a, b| a.0.cmp(&b.0));
            let cost = self
                .locks
                .iter()
                .filter(|lock| {
                    !solution
                        .iter()
                        .any(|(package, version)| package == &lock.package && *version == lock.version)
                })
                .count();

            let replace = match &self.best {
                None => true,
                Some((best_cost, best_solution)) => {
                    cost < *best_cost
                        || (cost == *best_cost
                            && self.lex_better(&solution, best_solution))
                }
            };
            if replace {
                self.best = Some((cost, solution));
            }
            return Ok(());
        };

        let package = self.catalog.iter().find(|p| p.name == name).unwrap();
        let constraints = active.get(&name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|release| {
                constraints
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, release.version))
            })
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(index) = candidates.iter().position(|version| *version == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.push((name.clone(), candidate));
            match self.inspect(selected) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    selected.pop();
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
            selected.pop();
        }

        Ok(())
    }

    fn lex_better(
        &self,
        left: &[(String, Version)],
        right: &[(String, Version)],
    ) -> bool {
        let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        names.sort_unstable();
        names.dedup();

        for name in names {
            let a = left.iter().find(|(package, _)| package == name).map(|x| x.1);
            let b = right.iter().find(|(package, _)| package == name).map(|x| x.1);
            match (a, b) {
                (None, None) => {}
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    let result = search.inspect(&mut Vec::new());
    if let Err(ResolveError::Limit { attempts }) = result {
        return Err(ResolveError::Limit { attempts });
    }

    let Some((_, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let final_active = canonical_requirements(roots, &chosen, catalog);
    let mut selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selected.sort_by(|a, b| a.package.cmp(&b.package));

    let mut names = std::collections::BTreeSet::new();
    for lock in locks {
        names.insert(lock.package.clone());
    }
    for selection in &selected {
        names.insert(selection.package.clone());
    }

    let mut changes = Vec::new();
    for name in names {
        let before = locks
            .iter()
            .find(|lock| lock.package == name)
            .map(|lock| lock.version);
        let after = chosen
            .iter()
            .find(|(package, _)| package == &name)
            .map(|(_, version)| *version);
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|package| package.name == name)
                    .is_some_and(|package| package.releases.iter().any(|r| r.version == locked));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active.get(&name).is_some_and(|constraints| {
                    !constraints
                        .iter()
                        .all(|constraint| constraint_matches(*constraint, locked))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Assignments = std::collections::BTreeMap<String, Version>;
type Active = std::collections::BTreeMap<String, Vec<Constraint>>;

fn active_constraints(
    roots: &[Requirement],
    catalog: &std::collections::BTreeMap<String, &Package>,
    assigned: &Assignments,
) -> Active {
    let mut active = Active::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assigned {
        if let Some(package) = catalog.get(name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for cs in active.values_mut() {
        *cs = canonical(std::mem::take(cs));
    }
    active
}

fn node_error(
    active: &Active,
    catalog: &std::collections::BTreeMap<String, &Package>,
    assigned: &Assignments,
) -> Option<ResolveError> {
    if let Some(name) = active.keys().find(|name| !catalog.contains_key(*name)) {
        return Some(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in assigned {
        if let Some(cs) = active.get(name) {
            if !satisfies(*version, cs) {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: cs.clone(),
                });
            }
        }
    }

    for (name, cs) in active {
        if !assigned.contains_key(name) {
            let has_match = catalog[name]
                .releases
                .iter()
                .any(|r| satisfies(r.version, cs));
            if !has_match {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.clone(),
                });
            }
        }
    }
    None
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: std::collections::BTreeMap<String, &'a Package>,
    locks: std::collections::BTreeMap<String, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Version>, Assignments)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn visit(&mut self, assigned: &mut Assignments) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, &self.catalog, assigned);
        if let Some(error) = node_error(&active, &self.catalog, assigned) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active.keys().find(|name| !assigned.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| assigned.get(*name) != Some(*version))
                .count();
            let all_names: Vec<_> = self.catalog.keys().cloned().collect();
            let vector = all_names
                .iter()
                .map(|n| assigned.get(n).copied().unwrap_or(Version {
                    major: 0,
                    minor: 0,
                    patch: 0,
                }))
                .collect::<Vec<_>>();
            let better = match &self.best {
                None => true,
                Some((best_cost, best_vector, _)) => {
                    cost < *best_cost || (cost == *best_cost && vector > *best_vector)
                }
            };
            if better {
                self.best = Some((cost, vector, assigned.clone()));
            }
            return Ok(());
        };

        let cs = &active[&name];
        let package = self.catalog[&name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|r| satisfies(r.version, cs))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), version);
            self.visit(assigned)?;
            assigned.remove(&name);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let catalog_map: std::collections::BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: catalog_map,
        locks: lock_map,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut Assignments::new())?;

    let Some((_, _, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
        }));
    };

    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = active_constraints(roots, &search.catalog, &chosen);
    let mut before = std::collections::BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().cloned());
    names.extend(chosen.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let old = before.get(&package).copied();
        let new = chosen.get(&package).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = search.catalog.get(&package).is_some_and(|p| {
                    p.releases.iter().any(|r| r.version == old_version)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&package)
                    .is_some_and(|cs| !satisfies(old_version, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
