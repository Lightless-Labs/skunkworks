```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| b.arrival.cmp(&a.arrival).then_with(|| a.id.cmp(&b.id)));

    let mut events = Vec::with_capacity(customers.len().saturating_mul(3));
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut current_time = 0u64;
    let mut finish_time: Option<u64> = None;
    let mut previous_finish: Option<u64> = None;
    let mut first_dispatch = true;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        let next_customer_arrival = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_time = match (finish_time, next_customer_arrival) {
            (Some(f), Some(a)) => f.min(a),
            (Some(f), None) => f,
            (None, Some(a)) => a,
            (None, None) => break,
        };
        current_time = next_time;

        if finish_time == Some(current_time) {
            let id = waiting
                .front()
                .map(|_| 0)
                .unwrap_or(0);
            let _ = id;
            // The active customer is tracked separately below through its ID.
        }

        // Finish the active occupation before processing arrivals at this tick.
        if finish_time == Some(current_time) {
            let id = ACTIVE_CUSTOMER.with(|_| 0);
            let _ = id;
        }

        // The active customer ID is maintained locally to preserve event ordering.
        if let Some((active_id, active_finish)) = ACTIVE.get() {
            if active_finish == current_time {
                events.push(Event {
                    time: current_time,
                    customer: active_id,
                    kind: Kind::Finish,
                });
                ACTIVE.clear();
                finish_time = None;
                previous_finish = Some(current_time);
                finished_count += 1;
            }
        }

        // Emit the whole arrival cohort in descending ID order.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == current_time {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: current_time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        // Dispatch while idle. Zero-duration occupations finish inline.
        while finish_time.is_none() {
            let Some(customer) = waiting.pop_front() else {
                break;
            };
            let wake = first_dispatch
                || previous_finish.is_some_and(|finished_at| current_time > finished_at);
            let base_finish = current_time.checked_add(customer.service).ok_or(
                Error::TimeOverflow {
                    customer: customer.id,
                    time: current_time,
                },
            )?;
            let finish = base_finish
                .checked_add(if wake { wake_cost } else { 0 })
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time: current_time,
                })?;

            events.push(Event {
                time: current_time,
                customer: customer.id,
                kind: Kind::Start { finish, wake },
            });
            first_dispatch = false;

            if finish == current_time {
                events.push(Event {
                    time: current_time,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                previous_finish = Some(current_time);
                finished_count += 1;
            } else {
                finish_time = Some(finish);
                ACTIVE.set((customer.id, finish));
            }
        }
    }

    Ok(events)
}

struct ActiveCustomer;

impl ActiveCustomer {
    fn set(&self, value: (u32, u64)) {
        ACTIVE_STATE.with(|state| *state.borrow_mut() = Some(value));
    }

    fn get(&self) -> Option<(u32, u64)> {
        ACTIVE_STATE.with(|state| *state.borrow())
    }

    fn clear(&self) {
        ACTIVE_STATE.with(|state| *state.borrow_mut() = None);
    }
}

thread_local! {
    static ACTIVE_STATE: std::cell::RefCell<Option<(u32, u64)>> = const { std::cell::RefCell::new(None) };
}

static ACTIVE: ActiveCustomer = ActiveCustomer;
```
