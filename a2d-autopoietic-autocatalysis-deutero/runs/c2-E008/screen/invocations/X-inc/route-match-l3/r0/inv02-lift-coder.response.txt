#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (segments, trailing_slash) = parse_pattern(pattern)?;
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        let mut rank = segments
            .iter()
            .map(|s| match s {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            })
            .collect::<Vec<_>>();
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (path_segments, request_trailing) = parse_path(head)?;

        let mut candidates = Vec::new();
        for route in &self.routes {
            if let Some(params) = match_route(route, &path_segments) {
                let slash_ok = match self.policy {
                    SlashPolicy::Strict => route.trailing_slash == request_trailing,
                    SlashPolicy::Ignore | SlashPolicy::Redirect => true,
                };
                if slash_ok {
                    candidates.push((route, params));
                }
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && !candidates.iter().any(|(r, _)| r.method == "HEAD")
        {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(r, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = candidates
                .iter()
                .map(|(r, _)| r.method.clone())
                .collect::<Vec<_>>();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (route, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id)))
            .unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != request_trailing
            && !path_segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing_slash {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    parse_segments(input, true).map_err(|at| Error::InvalidPattern { at })
}

fn parse_path(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    parse_segments(input, false).map_err(|at| Error::InvalidPath { at })
}

fn parse_segments(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let body = &input[1..end];
    if body.is_empty() {
        return Ok((Vec::new(), trailing));
    }

    let pieces: Vec<(usize, &str)> = body
        .split('/')
        .scan(1usize, |offset, piece| {
            let result = (*offset, piece);
            *offset += piece.len() + 1;
            Some(result)
        })
        .collect();

    let mut segments = Vec::with_capacity(pieces.len());
    let mut names = std::collections::HashSet::new();

    for (start, piece) in pieces {
        if piece.is_empty() {
            return Err(start);
        }

        if pattern && (piece.starts_with(':') || piece.starts_with('*')) {
            let marker = piece.as_bytes()[0];
            let name = &piece[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(start);
            }
            if marker == b'*' && start + piece.len() != end {
                return Err(start);
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_static(piece, start)?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok((segments, trailing))
}

fn decode_static(input: &str, start: usize) -> Result<String, usize> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(start + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(start + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(start + i)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(start + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(start + i);
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[Segment]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));

    if let Some(index) = wildcard_index {
        if path.len() < index {
            return None;
        }
        let mut params = Vec::new();
        for i in 0..index {
            match_segment(&route.segments[i], &path[i], &mut params)?;
        }
        let name = match &route.segments[index] {
            Segment::Wildcard(name) => name.clone(),
            _ => unreachable!(),
        };
        let rest = path[index..]
            .iter()
            .map(|s| match s {
                Segment::Static(v) => v.as_str(),
                _ => unreachable!(),
            })
            .collect::<Vec<_>>()
            .join("/");
        params.push((name, rest));
        Some(params)
    } else {
        if route.segments.len() != path.len() {
            return None;
        }
        let mut params = Vec::new();
        for (pattern, request) in route.segments.iter().zip(path) {
            match_segment(pattern, request, &mut params)?;
        }
        Some(params)
    }
}

fn match_segment(
    pattern: &Segment,
    request: &Segment,
    params: &mut Vec<(String, String)>,
) -> Option<()> {
    let value = match request {
        Segment::Static(value) => value,
        _ => unreachable!(),
    };
    match pattern {
        Segment::Static(expected) if expected == value => Some(()),
        Segment::Param(name) => {
            params.push((name.clone(), value.clone()));
            Some(())
        }
        _ => None,
    }
}
