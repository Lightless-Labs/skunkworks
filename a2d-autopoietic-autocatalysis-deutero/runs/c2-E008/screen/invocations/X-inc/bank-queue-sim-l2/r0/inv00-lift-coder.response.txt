#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct SimServer {
    id: u32,
    opens: u64,
    overhead: u64,
    is_open: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<Customer> = customers.to_vec();
    ordered_customers.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut sim_servers: Vec<SimServer> = servers
        .iter()
        .map(|server| SimServer {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            is_open: false,
            busy_until: None,
            idle_since: server.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;

    loop {
        // Finish events precede all other events at this tick.
        let mut finishing: Vec<usize> = sim_servers
            .iter()
            .enumerate()
            .filter_map(|(i, server)| (server.busy_until == Some(now)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| sim_servers[i].id);
        for i in finishing {
            let server = &mut sim_servers[i];
            if let Some(customer) = server.busy_customer.take() {
                events.push(Event::Finish {
                    time: now,
                    customer,
                    server: server.id,
                });
            }
            server.busy_until = None;
            server.idle_since = now;
        }

        let mut opening: Vec<usize> = sim_servers
            .iter()
            .enumerate()
            .filter_map(|(i, server)| (!server.is_open && server.opens == now).then_some(i))
            .collect();
        opening.sort_by(|&a, &b| sim_servers[b].id.cmp(&sim_servers[a].id));
        for i in opening {
            let server = &mut sim_servers[i];
            server.is_open = true;
            events.push(Event::Open {
                time: now,
                server: server.id,
            });
        }

        let arrival_start = next_customer;
        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == now
        {
            next_customer += 1;
        }
        for customer in &ordered_customers[arrival_start..next_customer] {
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(customer_index(customers, customer.id));
        }

        loop {
            if waiting.is_empty() {
                break;
            }

            let selected = sim_servers
                .iter()
                .enumerate()
                .filter(|(_, server)| server.is_open && server.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server_id = sim_servers[server_index].id;
            let finish = now
                .checked_add(customer.service)
                .and_then(|time| time.checked_add(sim_servers[server_index].overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server_id,
                finish,
            });

            let server = &mut sim_servers[server_index];
            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                server.idle_since = now;
            } else {
                server.busy_until = Some(finish);
                server.busy_customer = Some(customer.id);
            }
        }

        if next_customer == ordered_customers.len()
            && waiting.is_empty()
            && sim_servers.iter().all(|server| server.busy_until.is_none())
            && sim_servers.iter().all(|server| server.is_open)
        {
            break;
        }

        let next_arrival = ordered_customers
            .get(next_customer)
            .map(|customer| customer.arrival);
        let next_open = sim_servers
            .iter()
            .filter(|server| !server.is_open)
            .map(|server| server.opens)
            .min();
        let next_finish = sim_servers.iter().filter_map(|server| server.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
