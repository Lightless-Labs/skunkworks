#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<bool>; 5],
    day_of_month_star: bool,
    day_of_week_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    let offset = start;
    let number_bytes = &bytes[start..end];

    if number_bytes.len() > 1 && number_bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value: u32 = 0;
    for &byte in number_bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if step {
        let (_, max) = field_bounds(field);
        if value < 1 || value > max - field_bounds(field).0 + 1 {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = field_bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }

    Ok(value)
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();
    let hyphen_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
        .collect();

    if slash_positions.len() > 1 || hyphen_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
        (slash, Some(slash + 1))
    } else {
        (item.len(), None)
    };
    let base = &item[..base_end];

    if base.is_empty() || step_start == Some(item.len()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let star = base == b"*";
    let range_hyphen = base.iter().position(|&b| b == b'-');

    let (base_kind, first_start, first_end, second_start, second_end) = if star {
        (0u8, 0, 0, 0, 0)
    } else if let Some(hyphen) = range_hyphen {
        if hyphen == 0 || hyphen + 1 == base.len() || base[hyphen + 1..].contains(&b'-') {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        if base[..hyphen].contains(&b'*') || base[hyphen + 1..].contains(&b'*') {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        (2u8, 0, hyphen, hyphen + 1, base.len())
    } else {
        if base.contains(&b'*') || base.contains(&b'-') || base.is_empty() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        (1u8, 0, base.len(), 0, 0)
    };

    if base_kind == 0 && !star {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }
    if base_kind != 0
        && (base[..first_end].iter().any(|b| !b.is_ascii_digit())
            || (base_kind == 2
                && base[second_start..second_end]
                    .iter()
                    .any(|b| !b.is_ascii_digit())))
    {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (min, max) = field_bounds(field);
    let step = if let Some(start) = step_start {
        let raw_start = base_end + 1;
        Some(parse_number(
            item,
            raw_start,
            item.len(),
            field,
            true,
        )?)
    } else {
        None
    };
    let _ = start;

    let (start_value, end_value) = match base_kind {
        0 => (min, max),
        1 => {
            let start = parse_number(item, first_start, first_end, field, false)?;
            let end = if step.is_some() { max } else { start };
            (start, end)
        }
        2 => {
            let start = parse_number(item, first_start, first_end, field, false)?;
            let end = parse_number(item, second_start, second_end, field, false)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_offset + second_start,
                });
            }
            (start, end)
        }
        _ => unreachable!(),
    };

    let stride = step.unwrap_or(1);
    let mut value = start_value;
    while value <= end_value {
        selected[(value - min) as usize] = true;
        match value.checked_add(stride) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = field_bounds(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut item_start = 0;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(&bytes[item_start..i], item_start, field, &mut selected)?;
            item_start = i + 1;
        }
    }
    parse_item(
        &bytes[item_start..],
        item_start,
        field,
        &mut selected,
    )?;
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut allowed: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, field) in FIELDS.iter().copied().enumerate() {
        allowed[i] = parse_field(pieces[i], field)?;
    }

    Ok(Cron {
        allowed,
        day_of_month_star: pieces[2].as_bytes() == b"*",
        day_of_week_star: pieces[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        let (min, _) = field_bounds(field);
        self.allowed[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &is_allowed)| {
                if is_allowed {
                    Some(min + i as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let (min, max) = field_bounds(field);
        value >= min
            && value <= max
            && self.allowed[field_index(field)][(value - min) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.contains(Field::DayOfMonth, at.day)
            && self.contains(Field::DayOfWeek, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;

        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            current = next_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn next_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
