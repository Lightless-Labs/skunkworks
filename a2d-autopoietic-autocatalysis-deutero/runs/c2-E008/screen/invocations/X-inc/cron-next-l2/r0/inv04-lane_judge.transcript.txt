pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = Vec::with_capacity(5);
    for (index, field) in FIELDS.into_iter().enumerate() {
        selected.push(parse_field(parts[index], field)?);
    }

    Ok(Cron {
        selected: selected.try_into().unwrap(),
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };

        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| {
                if is_selected {
                    Some(value as u8)
                } else {
                    None
                }
            })
            .collect()

        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);

        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            current = match increment_minute(current) {
                Some(next) => next,
                None => return Ok(None),
            };

            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(value: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&value.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&value.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if value.day < 1 || value.day > days_in_month(value.year, value.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if value.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if value.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn increment_minute(value: CivilDateTime) -> Option<CivilDateTime> {

        return Some(CivilDateTime {
            minute: value.minute + 1,

            minute: 0,

            minute: 0,
            hour: 0,

        return Some(CivilDateTime {
            minute: 0,
            hour: 0,
            day: 1,
            month: 1,

        let mut start_value = min;
        let mut end_value = max;

        if !is_star {
            if let Some(hyphen) = range_pos {
                start_value = parse_number(&item, 0, hyphen, field, false)?;
                end_value = parse_number(&item, hyphen + 1, base_end, field, false)?;
                if start_value > end_value {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
            } else {
                start_value = parse_number(&item, 0, base_end, field, false)?;
                end_value = if slash_pos.is_some() { max } else { start_value };
            }
        }

        let step = if let Some(slash) = slash_pos {
            let step_start = slash + 1;
            parse_number(&item, step_start, item.len(), field, true)?
        } else {
            1
        };

        if is_star && slash_pos.is_none() && field == Field::DayOfMonth && text == "*" {
            exact_star = true;
        }
        if is_star && slash_pos.is_none() && field == Field::DayOfWeek && text == "*" {
            exact_star = true;
        }

        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    // A day field is exactly "*" only when its entire text is that one byte.
    if field == Field::DayOfMonth || field == Field::DayOfWeek {
        exact_star = text.as_bytes() == b"*";
    }

    Ok((selected, exact_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_of_month_is_star = false;
    let mut day_of_week_is_star = false;

    for (index, (&field, text)) in FIELDS.iter().zip(pieces.iter()).enumerate() {
        let (values, exact_star) = parse_field(text, field)?;
        selected[index] = values;
        if field == Field::DayOfMonth {
            day_of_month_is_star = exact_star;
        } else if field == Field::DayOfWeek {
            day_of_week_is_star = exact_star;
        }
    }

    Ok(Cron {
        selected,
        day_of_month_is_star,
        day_of_week_is_star,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    if current.day < days_in_month(current.year, current.month) {
                        current.day += 1;
                    } else {
                        current.day = 1;
                        if current.month < 12 {
                            current.month += 1;
                        } else {
                            current.month = 1;
                            if current.year == 2399 {
                                return Ok(None);
                            }
                            current.year += 1;
                        }
                    }
                }
            }

            if self.matches(current)? {
                return Ok(Some(current));
            }
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
        }
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}
