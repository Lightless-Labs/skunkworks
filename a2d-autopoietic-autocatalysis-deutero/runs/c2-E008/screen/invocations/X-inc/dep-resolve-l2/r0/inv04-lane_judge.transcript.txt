fn valid_name(name: &str) -> bool {

        || !bytes[0].is_ascii_lowercase()

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn constraint_matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

    constraints

fn constraints_match(version: Version, constraints: &[Constraint]) -> bool {

fn encode_text(text: &str) -> String {
    let mut result = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {

fn active_constraints(
    roots: &[Requirement],
    catalog_by_name: &std::collections::BTreeMap<&str, &Package>,
    assignments: &std::collections::BTreeMap<String, Version>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in assignments {
        if let Some(package) = catalog_by_name.get(name.as_str()) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical_constraints(std::mem::take(constraints));
    }
    active
}

fn inspect_node(
    active: &std::collections::BTreeMap<String, Vec<Constraint>>,
    catalog_by_name: &std::collections::BTreeMap<&str, &Package>,
    assignments: &std::collections::BTreeMap<String, Version>,
) -> Result<Option<String>, ResolveError> {
    if let Some(name) = active
        .keys()
        .filter(|name| !catalog_by_name.contains_key(name.as_str()))
        .next()
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in assignments {
        if let Some(constraints) = active.get(name) {
            if !constraints_match(*version, constraints) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in active {
        if assignments.contains_key(name) {
            continue;
        }
        let package = catalog_by_name[name.as_str()];
        if !package
            .releases
            .iter()
            .any(|release| constraints_match(release.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    Ok(active
        .keys()
        .find(|name| !assignments.contains_key(*name))
        .cloned())
}

fn search(
    roots: &[Requirement],
    catalog_by_name: &std::collections::BTreeMap<&str, &Package>,
    assignments: &mut std::collections::BTreeMap<String, Version>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let active = active_constraints(roots, catalog_by_name, assignments);
    let next = inspect_node(&active, catalog_by_name, assignments)?;

    let Some(package_name) = next else {
        return Ok(assignments
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect());
    };

    let constraints = &active[&package_name];
    let package = catalog_by_name[package_name.as_str()];
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|release| constraints_match(release.version, constraints))
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    let mut first_error = None;
    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        assignments.insert(package_name.clone(), version);
        match search(roots, catalog_by_name, assignments, attempts, max_attempts) {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => return Err(error),
            Err(error) => {
                assignments.remove(&package_name);
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
        package: package_name,
        requirements: constraints.clone(),
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let catalog_by_name: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut attempts = 0;
    let mut assignments = std::collections::BTreeMap::new();
    search(
        roots,
        &catalog_by_name,
        &mut assignments,
        &mut attempts,
        max_attempts,
    )
}

        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(

            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))

        ResolveError::NoMatch {
            package,
            requirements,

    }
}

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut std::collections::BTreeMap<String, Version>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in selected.iter() {
        if let Some(package) = catalog.iter().find(|p| &p.name == name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }

    if let Some(name) = active.keys().find(|name| !catalog.iter().any(|p| &p.name == *name)) {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in selected.iter() {
        if let Some(constraints) = active.get(name) {
            if constraints.iter().any(|c| !matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in &active {
        if selected.contains_key(name) {
            continue;
        }
        let package = catalog.iter().find(|p| &p.name == name).unwrap();
        if !package
            .releases
            .iter()
            .any(|r| constraints.iter().all(|c| matches(r.version, *c)))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
    let Some(name) = next else {
        return Ok(selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect());
    };

    let package = catalog.iter().find(|p| p.name == name).unwrap();
    let mut releases: Vec<&Release> = package
        .releases
        .iter()
        .filter(|r| {
            active[&name]
                .iter()
                .all(|c| matches(r.version, *c))
        })
        .collect();
    releases.sort_by(|a, b| b.version.cmp(&a.version));

    let mut first_error = None;
    for release in releases {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.insert(name.clone(), release.version);
        match search(catalog, roots, selected, attempts, max_attempts) {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => return Err(error),
            Err(error) => {
                selected.remove(&name);
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }
    Err(first_error.unwrap_or(ResolveError::NoMatch {
        package: name.clone(),
        requirements: active[&name].clone(),
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    search(
        catalog,
        roots,
        &mut std::collections::BTreeMap::new(),
        &mut 0,
        max_attempts,
    )
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", byte));
        }
    }
    out
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x
                && if x.major > 0 {
                    version.major == x.major
                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

            for (k, dependency) in release.deps.iter().enumerate() {

struct Search<'a> {
    catalog: &'a [Package],
    max_attempts: usize,
    attempts: usize,
}

impl Search<'_> {
    fn inspect(
        &mut self,
        roots: &[Requirement],
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<Vec<Selection>, ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();

        for root in roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }

        for (name, version) in selected.iter() {
            if let Some(package) = self.catalog.iter().find(|p| p.name == *name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dependency in &release.deps {
                        active
                            .entry(dependency.package.clone())
                            .or_default()
                            .push(dependency.constraint);
                    }
                }
            }
        }

        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if !matches_all(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !package
                .releases
                .iter()
                .any(|release| matches_all(release.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(package_name) = next else {
            return Ok(selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect());
        };

        let constraints = active.get(&package_name).unwrap();
        let package = self
            .catalog
            .iter()
            .find(|p| p.name == package_name)
            .unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|release| matches_all(release.version, constraints))
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        let mut first_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), candidate);
            let result = self.inspect(roots, selected);
            selected.remove(&package_name);

            match result {
                Ok(solution) => return Ok(solution),
                Err(error @ ResolveError::Limit { .. }) => return Err(error),
                Err(error) => {
                    if first_error.is_none() {
                        first_error = Some(error);
                    }
                }
            }
        }

        Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: package_name,
            requirements: constraints.clone(),
        }))
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;

    let mut search = Search {
        catalog,
        max_attempts,
        attempts: 0,
    };
    let mut selected = std::collections::BTreeMap::new();
    search.inspect(roots, &mut selected)
}
