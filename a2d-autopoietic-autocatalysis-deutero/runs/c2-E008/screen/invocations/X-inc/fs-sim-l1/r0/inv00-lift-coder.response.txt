```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let (parent_components, name) = components.split_at(components.len() - 1);
        let parent_path = Self::canonical(parent_components);

        let parent = self
            .lookup_dir_mut(parent_components, &parent_path)?;

        let name = &name[0];
        if let Entry::Dir(children) = parent {
            if children.contains_key(name) {
                return Err(Error::Exists(Self::canonical(&components)));
            }
            children.insert(name.clone(), Entry::Dir(BTreeMap::new()));
            Ok(())
        } else {
            Err(Error::NotDir(parent_path))
        }
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let (parent_components, name) = components.split_at(components.len() - 1);
        let parent_path = Self::canonical(parent_components);
        let target_path = Self::canonical(&components);

        let parent = self
            .lookup_dir_mut(parent_components, &parent_path)?;
        let name = &name[0];

        if let Entry::Dir(children) = parent {
            match children.get_mut(name) {
                Some(Entry::Dir(_)) => Err(Error::IsDir(target_path)),
                Some(Entry::File(contents)) => {
                    *contents = data.to_vec();
                    Ok(())
                }
                None => {
                    children.insert(name.clone(), Entry::File(data.to_vec()));
                    Ok(())
                }
            }
        } else {
            Err(Error::NotDir(parent_path))
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let (parent_components, name) = components.split_at(components.len() - 1);
        let parent_path = Self::canonical(parent_components);
        let target_path = Self::canonical(&components);

        let parent = self.lookup_dir(parent_components, &parent_path)?;
        let Entry::Dir(children) = parent else {
            return Err(Error::NotDir(parent_path));
        };

        match children.get(&name[0]) {
            Some(Entry::File(contents)) => Ok(contents.clone()),
            Some(Entry::Dir(_)) => Err(Error::IsDir(target_path)),
            None => Err(Error::Missing(target_path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let canonical_path = Self::canonical(&components);
        let entry = self.lookup(&components, &canonical_path)?;

        let Entry::Dir(children) = entry else {
            return Err(Error::NotDir(canonical_path));
        };

        Ok(children
            .iter()
            .map(|(name, entry)| {
                let kind = match entry {
                    Entry::Dir(_) => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                (name.clone(), kind)
            })
            .collect())
    }

    fn validate(path: &str) -> Result<Vec<String>, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }

        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        if !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }

        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut component_start = 1;

        for (index, component) in path[1..].split('/').enumerate() {
            let start = component_start;
            if component.is_empty() || component == "." || component == ".." {
                let offset = if component.is_empty() {
                    if start == path.len() {
                        path.len() - 1
                    } else {
                        start - 1
                    }
                } else {
                    start
                };
                return Err(Error::InvalidPath(offset));
            }

            components.push(component.to_string());
            component_start = start + component.len() + 1;

            // Keep the loop index used so the iteration is explicit and stable.
            let _ = index;
        }

        Ok(components)
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn lookup<'a>(&'a self, components: &[String], full_path: &str) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut prefix = String::new();

        for (index, component) in components.iter().enumerate() {
            if !matches!(current, Entry::Dir(_)) {
                return Err(Error::NotDir(Self::canonical(&components[..index])));
            }

            prefix.push('/');
            prefix.push_str(component);

            let Entry::Dir(children) = current else {
                unreachable!();
            };

            match children.get(component) {
                Some(entry) => current = entry,
                None => return Err(Error::Missing(prefix)),
            }
        }

        let _ = full_path;
        Ok(current)
    }

    fn lookup_dir<'a>(
        &'a self,
        components: &[String],
        path: &str,
    ) -> Result<&'a Entry, Error> {
        let entry = self.lookup(components, path)?;
        if matches!(entry, Entry::Dir(_)) {
            Ok(entry)
        } else {
            Err(Error::NotDir(path.to_string()))
        }
    }

    fn lookup_dir_mut<'a>(
        &'a mut self,
        components: &[String],
        path: &str,
    ) -> Result<&'a mut Entry, Error> {
        let mut current = &mut self.root;
        let mut prefix = String::new();

        for (index, component) in components.iter().enumerate() {
            if !matches!(current, Entry::Dir(_)) {
                return Err(Error::NotDir(Self::canonical(&components[..index])));
            }

            prefix.push('/');
            prefix.push_str(component);

            let Entry::Dir(children) = current else {
                unreachable!();
            };

            match children.get_mut(component) {
                Some(entry) => current = entry,
                None => return Err(Error::Missing(prefix)),
            }
        }

        if matches!(current, Entry::Dir(_)) {
            Ok(current)
        } else {
            Err(Error::NotDir(path.to_string()))
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
