```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let c = self.den as i128;
        let d = rhs.den as i128;
        Self::from_i128(a * d + b * c, c * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let c = self.den as i128;
        let d = rhs.den as i128;
        Self::from_i128(a * d - b * c, c * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse {
                        at: i,
                        kind: ParseKind::InvalidByte,
                    });
                }
                slash = Some(i);
            } else if b.is_ascii_digit() {
                // Digit syntax is checked below with the sign and token boundaries.
            } else if b == b'-' && i == 0 {
                // Only a leading minus is permitted.
            } else {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes.first() == Some(&b'-'));
        if num_start == num_end {
            return Err(RationalError::Parse {
                at: bytes.len(),
                kind: ParseKind::InvalidByte,
            });
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse {
                    at: bytes.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_digits = &bytes[num_start..num_end];
        let den_digits = slash.map(|s| &bytes[s + 1..]);

        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if bytes[0] == b'-' && num_digits == b"0" {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }
        if let Some(dd) = den_digits {
            if dd.len() >= 2 && dd[0] == b'0' {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 2,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        fn magnitude(digits: &[u8], limit: u128) -> Option<u128> {
            let mut value = 0u128;
            for &b in digits {
                let digit = (b - b'0') as u128;
                value = value.checked_mul(10)?.checked_add(digit)?;
                if value > limit {
                    return None;
                }
            }
            Some(value)
        }

        let negative = bytes[0] == b'-';
        let nlimit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let nmag = magnitude(num_digits, nlimit).ok_or(RationalError::Parse {
            at: 0,
            kind: ParseKind::OutOfRange,
        })?;

        let den_value = if let Some(dd) = den_digits {
            Some(
                magnitude(dd, i64::MAX as u128).ok_or(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: ParseKind::OutOfRange,
                })?,
            )
        } else {
            None
        };

        if den_value == Some(0) {
            return Err(RationalError::Parse {
                at: slash.unwrap() + 1,
                kind: ParseKind::ZeroDenominator,
            });
        }

        if let Some(d) = den_value {
            if d == 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if nmag == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(nmag, d) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let signed_num = if negative {
            if nmag == (1u128 << 63) {
                i64::MIN as i128
            } else {
                -(nmag as i128)
            }
        } else {
            nmag as i128
        };
        Self::from_i128(signed_num, den_value.unwrap_or(1) as i128)
            .map_err(|_| RationalError::Parse {
                at: 0,
                kind: ParseKind::OutOfRange,
            })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
