```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn normalized(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128
        || n > i64::MAX as i128
        || d < 1
        || d > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        normalized(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        normalized(
            (self.num as i128) * (rhs.den as i128)
                + (rhs.num as i128) * (self.den as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        normalized(
            (self.num as i128) * (rhs.den as i128)
                - (rhs.num as i128) * (self.den as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        normalized(
            (self.num as i128) * (rhs.num as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        normalized(
            (self.num as i128) * (rhs.den as i128),
            (self.den as i128) * (rhs.num as i128),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        if n < i64::MIN as i128 || n > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| Err(RationalError::Parse { at, kind });

        if bytes.is_empty() {
            return fail(0, ParseKind::Empty);
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            if i == bytes.len() {
                return fail(i, ParseKind::InvalidByte);
            }
            return fail(i, ParseKind::InvalidByte);
        }
        let num_end = i;

        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return fail(i, ParseKind::InvalidByte);
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return fail(i, ParseKind::InvalidByte);
            }
            den_end = Some(i);
            if i < bytes.len() {
                return fail(i, ParseKind::InvalidByte);
            }
        }

        let negative = bytes[0] == b'-';
        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return fail(if negative { 2 } else { 1 }, ParseKind::NonCanonical);
        }
        if negative && num_digits == b"0" {
            return fail(0, ParseKind::NonCanonical);
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return fail(ds + 1, ParseKind::NonCanonical);
            }
        }

        let mut nmag: u128 = 0;
        for &b in num_digits {
            nmag = nmag * 10 + (b - b'0') as u128;
            let limit = if negative {
                1u128 << 63
            } else {
                i64::MAX as u128
            };
            if nmag > limit {
                return fail(0, ParseKind::OutOfRange);
            }
        }

        let mut dmag: u128 = 1;
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            dmag = 0;
            for &b in &bytes[ds..de] {
                dmag = dmag * 10 + (b - b'0') as u128;
                if dmag > i64::MAX as u128 {
                    return fail(ds, ParseKind::OutOfRange);
                }
            }
            if dmag == 0 {
                return fail(ds, ParseKind::ZeroDenominator);
            }
        }

        if let (Some(ds), Some(_)) = (den_start, den_end) {
            let slash = ds - 1;
            if dmag == 1 {
                return fail(slash, ParseKind::NonCanonical);
            }
            if nmag == 0 {
                return fail(slash, ParseKind::NonCanonical);
            }
            if gcd_u128(nmag, dmag) > 1 {
                return fail(slash, ParseKind::NonCanonical);
            }
        }

        let n = if negative {
            if nmag == (1u128 << 63) {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };
        let d = dmag as i64;
        Ok(Self { num: n, den: d })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
