```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_fraction(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

fn from_unsigned_fraction(negative: bool, magnitude: u128, den: u128) -> Result<Rational, RationalError> {
    let g = gcd_u128(magnitude, den);
    let mag = magnitude / g;
    let d = den / g;
    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    let n = if negative {
        if mag > (1u128 << 63) {
            return Err(RationalError::Overflow);
        }
        if mag == (1u128 << 63) {
            i64::MIN
        } else {
            -(mag as i64)
        }
    } else {
        if mag > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        mag as i64
    };
    Ok(Rational { num: n, den: d as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        from_i128_fraction(n, d)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let d1 = self.den as i128;
        let d2 = rhs.den as i128;
        from_i128_fraction(a * d2 + b * d1, d1 * d2)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let d1 = self.den as i128;
        let d2 = rhs.den as i128;
        from_i128_fraction(a * d2 - b * d1, d1 * d2)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_i128_fraction(
            (self.num as i128) * (rhs.num as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_i128_fraction(
            (self.num as i128) * (rhs.den as i128),
            (self.den as i128) * (rhs.num as i128),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }
        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let mut slash = None;
        let mut denominator_start = None;
        let mut denominator_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            denominator_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start.unwrap() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            denominator_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let numerator_digits = &bytes[numerator_start..numerator_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_err(if negative { 2 } else { 1 }, ParseKind::NonCanonical));
        }
        if negative && numerator_digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (denominator_start, denominator_end) {
            let digits = &bytes[ds..de];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let num_mag = parse_magnitude(numerator_digits);
        let num_limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag > num_limit {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }

        let den_mag = if let (Some(ds), Some(de)) = (denominator_start, denominator_end) {
            let digits = &bytes[ds..de];
            let value = parse_magnitude(digits);
            if value > i64::MAX as u128 {
                return Err(parse_err(ds, ParseKind::OutOfRange));
            }
            Some(value)
        } else {
            None
        };

        if let Some(d) = den_mag {
            let ds = denominator_start.unwrap();
            if d == 0 {
                return Err(parse_err(ds, ParseKind::ZeroDenominator));
            }
            if d == 1 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, d) > 1 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            from_unsigned_fraction(negative, num_mag, d)
        } else {
            from_unsigned_fraction(negative, num_mag, 1)
        }
    }
}

fn parse_magnitude(digits: &[u8]) -> u128 {
    let mut value = 0u128;
    for &b in digits {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u128);
    }
    value
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
