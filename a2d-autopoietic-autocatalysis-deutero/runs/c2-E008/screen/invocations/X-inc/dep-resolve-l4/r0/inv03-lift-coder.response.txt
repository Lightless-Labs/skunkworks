fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_rank(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b as char, '.' | '_' | '-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut e = evidence.to_vec();
    e.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    e
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

    path: Vec<String>,

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    features: BTreeSet<String>,
    incoming: Vec<Vec<String>>,
}

use std::collections::{BTreeMap, BTreeSet, HashMap, HashSet};

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn sorted_feature_list(fs: &[String]) -> String {
    let mut f = fs.to_vec();
    f.sort();
    f.dedup();
    f.join(",")
}

fn root_token(i: usize, root: &Requirement) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        root.package,
        constraint_text(root.constraint),
        sorted_feature_list(&root.features)
    )
}

fn dep_token(
    source: &str,
    version: Version,
    j: usize,
    dep: &Dependency,
) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        j,
        dep.package,
        constraint_text(dep.constraint),
        sorted_feature_list(&dep.features),
        dep.gate.as_deref().unwrap_or("*"),
        match dep.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn canonical_path(paths: &[Vec<String>]) -> Vec<String> {
    paths
        .iter()
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
        .cloned()
        .unwrap_or_default()
}

fn active_state(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> BTreeMap<String, Active> {
    let by_name: HashMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut active: BTreeMap<String, Active> = BTreeMap::new();
    let mut queue: Vec<(String, String, Vec<String>)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        let a = active.entry(root.package.clone()).or_default();
        a.origins.push(Origin {
            constraint: root.constraint,
            path: path.clone(),
        });
        a.incoming.push(path.clone());
        for f in &root.features {
            if a.features.insert(f.clone()) {
                queue.push((root.package.clone(), f.clone(), path.clone()));
            }
        }
    }

    // Repeatedly process enabled feature requests. Since feature sets only grow,
    // this computes the least fixed point over the selected branch.
    let mut processed: HashSet<(String, String)> = HashSet::new();
    loop {
        let next = queue.pop();
        if let Some((package_name, feature_name, _request_path)) = next {
            if !processed.insert((package_name.clone(), feature_name.clone())) {
                continue;
            }
            let Some(version) = selected.get(&package_name).copied() else {
                continue;
            };
            let Some(package) = by_name.get(package_name.as_str()) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            // Undeclared features are retained for diagnostics but enable nothing.
            if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                let paths = active
                    .get(&package_name)
                    .map(|a| a.incoming.clone())
                    .unwrap_or_default();
                let path = canonical_path(&paths);
                for enabled in &feature.enables {
                    let a = active.entry(package_name.clone()).or_default();
                    if a.features.insert(enabled.clone()) {
                        queue.push((package_name.clone(), enabled.clone(), path.clone()));
                    }
                }
            }
        } else {
            // Once feature queue is drained, activate all currently enabled deps.
            // New destination feature requests restart feature processing.
            let mut added = false;
            let selected_names: Vec<String> = selected.keys().cloned().collect();
            for source in selected_names {
                let Some(version) = selected.get(&source).copied() else {
                    continue;
                };
                let Some(package) = by_name.get(source.as_str()) else {
                    continue;
                };
                let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                    continue;
                };
                let source_active = active.get(&source).cloned().unwrap_or_default();
                let source_path = canonical_path(&source_active.incoming);
                for (j, dep) in release.deps.iter().enumerate() {
                    if !target_matches(dep.platform, target)
                        || dep.gate.as_ref().is_some_and(|g| !source_active.features.contains(g))
                    {
                        continue;
                    }
                    let token = dep_token(&source, version, j, dep);
                    let mut path = source_path.clone();
                    path.push(token);
                    let a = active.entry(dep.package.clone()).or_default();
                    if !a.origins.iter().any(|o| o.constraint == dep.constraint && o.path == path) {
                        a.origins.push(Origin {
                            constraint: dep.constraint,
                            path: path.clone(),
                        });
                        a.incoming.push(path.clone());
                        added = true;
                    }
                    for f in &dep.features {
                        if a.features.insert(f.clone()) {
                            queue.push((dep.package.clone(), f.clone(), path.clone()));
                            added = true;
                        }
                    }
                }
            }
            if !queue.is_empty() || added {
                continue;
            }
            break;
        }
    }

    active
}

fn requirements_for(a: &Active) -> Vec<Constraint> {
    canonical_constraints(a.origins.iter().map(|o| o.constraint).collect())
}

fn evidence_for(a: &Active) -> Vec<Evidence> {
    let mut ev = a
        .origins
        .iter()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    ev.sort_by(|x, y| {
        constraint_rank(x.constraint)
            .cmp(&constraint_rank(y.constraint))
            .then_with(|| x.path.cmp(&y.path))
    });
    ev
}

fn validate_names(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    let mut check = |site: String, name: &str| {
        if !valid_name(name) {
            Err(ResolveError::InvalidName {
                site,
                name: name.to_string(),
            })
        } else {
            Ok(())
        }
    };

    for (i, root) in roots.iter().enumerate() {
        check(format!("root[{}]", i), &root.package)?;
        for (f, feature) in root.features.iter().enumerate() {
            check(format!("root[{}].feature[{}]", i, f), feature)?;
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        check(format!("catalog[{}]", i), &package.name)?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                check(
                    format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                    &dep.package,
                )?;
                if let Some(gate) = &dep.gate {
                    check(
                        format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                        gate,
                    )?;
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    check(
                        format!(
                            "catalog[{}].release[{}].dep[{}].feature[{}]",
                            i, j, k, f
                        ),
                        feature,
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                check(
                    format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                    &feature.name,
                )?;
                for (e, enabled) in feature.enables.iter().enumerate() {
                    check(
                        format!(
                            "catalog[{}].release[{}].feature[{}].enable[{}]",
                            i, j, f, e
                        ),
                        enabled,
                    )?;
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        check(format!("lock[{}]", i), &lock.package)?;
    }
    Ok(())
}

fn validate_catalog(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut seen = HashSet::new();
    for p in catalog {
        if !seen.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = HashSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut features = HashSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn validate_locks(locks: &[Lock]) -> Result<(), ResolveError> {
    let mut seen = HashSet::new();
    for lock in locks {
        if !seen.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    budget: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, BTreeMap<String, Version>)>,
}

impl<'a> Search<'a> {
    fn candidates(&self, package: &str, cs: &[Constraint]) -> Vec<Version> {
        let Some(p) = self.catalog.iter().find(|p| p.name == package) else {
            return Vec::new();
        };
        let mut vs: Vec<Version> = p
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| cs.iter().all(|c| matches(*v, *c)))
            .collect();
        vs.sort();
        vs.dedup();
        vs.reverse();
        if let Some(lock) = self.locks.iter().find(|l| l.package == package) {
            if let Some(i) = vs.iter().position(|v| *v == lock.version) {
                let v = vs.remove(i);
                vs.insert(0, v);
            }
        }
        vs
    }

    fn inspect(&self, selected: &BTreeMap<String, Version>) -> Result<BTreeMap<String, Active>, ResolveError> {
        let active = active_state(self.catalog, self.roots, selected, self.target);
        let names: Vec<String> = active.keys().cloned().collect();

        for name in &names {
            if !self.catalog.iter().any(|p| p.name == *name) {
                let a = active.get(name).unwrap();
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: evidence_for(a),
                });
            }
        }

        for (name, version) in selected {
            if let Some(a) = active.get(name) {
                let cs = requirements_for(a);
                if !cs.iter().all(|c| matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs,
                        evidence: evidence_for(a),
                    });
                }
            }
        }

        for name in &names {
            if selected.contains_key(name) {
                continue;
            }
            let a = active.get(name).unwrap();
            let cs = requirements_for(a);
            if self.candidates(name, &cs).is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs,
                    evidence: evidence_for(a),
                });
            }
        }

        for name in selected.keys() {
            let Some(a) = active.get(name) else {
                continue;
            };
            let version = selected[name];
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            let release = package.releases.iter().find(|r| r.version == version).unwrap();
            for feature in &a.features {
                if !release.features.iter().any(|f| f.name == *feature) {
                    let path = canonical_path(&a.incoming);
                    return Err(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: version,
                        feature: feature.clone(),
                        path,
                    });
                }
            }
        }

        Ok(active)
    }

    fn solution_cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|l| selected.get(&l.package) != Some(&l.version))
            .count()
    }

    fn better(&self, candidate: &BTreeMap<String, Version>, cost: usize) -> bool {
        let Some((best_cost, best)) = &self.best else {
            return true;
        };
        if cost != *best_cost {
            return cost < *best_cost;
        }
        let names: BTreeSet<String> = self
            .catalog
            .iter()
            .map(|p| p.name.clone())
            .chain(candidate.keys().cloned())
            .chain(best.keys().cloned())
            .collect();
        for name in names {
            match (candidate.get(&name), best.get(&name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = match self.inspect(selected) {
            Ok(a) => a,
            Err(e) => {
                if self.first_error.is_none() {
                    self.first_error = Some(e);
                }
                return Ok(());
            }
        };

        let next = active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(package) = next else {
            let cost = self.solution_cost(selected);
            if self.better(selected, cost) {
                self.best = Some((cost, selected.clone()));
            }
            return Ok(());
        };

        let cs = requirements_for(active.get(&package).unwrap());
        let candidates = self.candidates(&package, &cs);
        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), version);
            self.visit(selected)?;
            selected.remove(&package);
        }
        Ok(())
    }
}

fn final_constraints(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> BTreeMap<String, Vec<Constraint>> {
    active_state(catalog, roots, selected, target)
        .into_iter()
        .map(|(name, active)| (name, requirements_for(&active)))
        .collect()
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;
    validate_catalog(catalog)?;
    validate_locks(locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        budget: max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
    };
    search.visit(&mut BTreeMap::new())?;

    let Some((_, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_cs = final_constraints(catalog, roots, &chosen, target);
    let mut before: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let all_names: BTreeSet<String> = before
        .keys()
        .cloned()
        .chain(chosen.keys().cloned())
        .collect();
    let mut changes = Vec::new();

    for package in all_names {
        let old = before.remove(&package);
        let new = chosen.get(&package).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old_version));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_cs
                    .get(&package)
                    .is_some_and(|cs| !cs.iter().all(|c| matches(old_version, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
