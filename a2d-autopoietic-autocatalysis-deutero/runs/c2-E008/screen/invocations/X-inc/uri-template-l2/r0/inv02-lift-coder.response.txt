#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy)]
struct Variable<'a> {
    name: &'a str,
}

#[derive(Debug, Clone, Copy)]
struct Expression<'a> {
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();

    if let Some(relative) = bytes[open + 1..close].iter().position(|&byte| byte == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + relative,
        });
    }

    let mut start = open + 1;
    let operator = match bytes[start] {
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' => {
            let operator = bytes[start];
            start += 1;
            if operator != b'+' && operator != b'#' {
                return Err(Error::UnsupportedOperator { pos: open + 1 });
            }
            Some(operator)
        }
        _ => None,
    };

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut item_start = start;

    loop {
        let item_end = bytes[item_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |relative| item_start + relative);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let name_bytes = &bytes[item_start..item_end];
        for (index, &byte) in name_bytes.iter().enumerate() {
            let valid = if index == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: item_start + index,
                });
            }
        }

        let name = &template[item_start..item_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        seen.push(name);
        variables.push(Variable { name });

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;
                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close + 1, expression));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn uppercase_hex(byte: u8) -> u8 {
    if byte.is_ascii_lowercase() {
        byte - (b'a' - b'A')
    } else {
        byte
    }
}

fn append_encoded(output: &mut String, value: &str, allow_reserved: bool) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            output.push('%');
            output.push(uppercase_hex(bytes[index + 1]) as char);
            output.push(uppercase_hex(bytes[index + 2]) as char);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::with_capacity(template.len());
    let mut literal_start = 0;

    for (open, end, expression) in expressions {
        output.push_str(&template[literal_start..open]);

        let mut fields: Vec<&str> = Vec::new();
        for variable in expression.variables {
            if let Some((_, value)) = vars.iter().rev().find(|(name, _)| *name == variable.name) {
                fields.push(value);
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }

            let allow_reserved = matches!(expression.operator, Some(b'+') | Some(b'#'));
            for (index, field) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                append_encoded(&mut output, field, allow_reserved);
            }
        }

        literal_start = end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
