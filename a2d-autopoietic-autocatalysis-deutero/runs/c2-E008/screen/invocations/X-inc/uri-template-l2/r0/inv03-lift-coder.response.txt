#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let fields: Vec<&str> = expression
            .names
            .iter()
            .filter_map(|name| vars.iter().rev().find(|(key, _)| key == name).map(|(_, value)| *value))
            .collect();

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            let allow_reserved = matches!(expression.operator, Some('+') | Some('#'));
            for (index, value) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                encode_value(value, allow_reserved, &mut output);
            }
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let rest = &template[cursor..];
        let Some(relative_open) = rest.find(['{', '}']) else {
            break;
        };
        let brace_pos = cursor + relative_open;
        let brace = template.as_bytes()[brace_pos];

        if brace == b'}' {
            return Err(Error::UnexpectedClose { pos: brace_pos });
        }

        let content_start = brace_pos + 1;
        let Some(relative_close) = template[content_start..].find('}') else {
            return Err(Error::Unclosed { pos: brace_pos });
        };
        let close_pos = content_start + relative_close;

        if let Some(relative_nested) = template[content_start..close_pos].find('{') {
            return Err(Error::NestedOpen {
                pos: content_start + relative_nested,
            });
        }

        let content = &template[content_start..close_pos];
        let (operator, body, body_start) = match content.as_bytes().first().copied() {
            Some(byte @ (b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')) => {
                let op = byte as char;
                if op != '+' && op != '#' {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                }
                (Some(op), &content[1..], content_start + 1)
            }
            _ => (None, content, content_start),
        };

        if body.is_empty() {
            return Err(Error::EmptyExpression { pos: close_pos });
        }

        let mut names = Vec::new();
        let mut seen: Vec<&str> = Vec::new();
        let mut part_start = 0;

        loop {
            let next_comma = body[part_start..].find(',');
            let part_end = next_comma.map_or(body.len(), |offset| part_start + offset);
            let name = &body[part_start..part_end];
            let name_pos = body_start + part_start;

            if name.is_empty() {
                return Err(Error::EmptyVariable { pos: name_pos });
            }

            if let Some(invalid_offset) = invalid_name_offset(name) {
                return Err(Error::InvalidName {
                    pos: name_pos + invalid_offset,
                });
            }

            if seen.contains(&name) {
                return Err(Error::DuplicateVariable { pos: name_pos });
            }
            seen.push(name);
            names.push(name);

            match next_comma {
                Some(_) => part_start = part_end + 1,
                None => break,
            }
        }

        expressions.push(Expression {
            start: brace_pos,
            end: close_pos,
            operator,
            names,
        });
        cursor = close_pos + 1;
    }

    Ok(expressions)
}

fn invalid_name_offset(name: &str) -> Option<usize> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Some(0);
    }
    if !is_name_start(bytes[0]) {
        return Some(0);
    }
    bytes
        .iter()
        .enumerate()
        .skip(1)
        .find_map(|(index, byte)| (!is_name_continue(*byte)).then_some(index))
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn encode_value(value: &str, allow_reserved: bool, output: &mut String) {
    let bytes = value.as_bytes();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            output.push('%');
            output.push((bytes[index + 1] as char).to_ascii_uppercase());
            output.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
