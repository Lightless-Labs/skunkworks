pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let op = bytes[content_start];
        if op != b'+' && op != b'#' {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        content_start += 1;
        Some(op)
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        for (offset, &byte) in name_bytes.iter().enumerate() {
            let valid = if offset == 0 {
                is_name_start(byte)
            } else {
                is_name_continue(byte)
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open = pos;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;
                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close + 1, expression));
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|&&(binding_name, _)| binding_name == name)
        .map(|&(_, value)| value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'('
            | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn push_encoded(output: &mut String, value: &str, allow_reserved: bool) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        i += 1;
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[literal_start..start]);

        let allow_reserved = matches!(expression.operator, Some(b'+' | b'#'));
        let mut fields = Vec::new();

        for variable in expression.variables {
            let _ = variable.start;
            if let Some(value) = lookup(variable.name, vars) {
                let mut field = String::new();
                push_encoded(&mut field, value, allow_reserved);
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }
            for (index, field) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                output.push_str(field);
            }
        }

        literal_start = end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[open + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let content_start = open + 1;
                let mut part_start = content_start;
                let mut operator = None;

                if content_start < close
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let op = bytes[content_start] as char;
                    if op != '+' && op != '#' {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    operator = Some(op);
                    part_start += 1;
                }

                if part_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen = std::collections::HashSet::new();
                let mut spec_start = part_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &bytes[spec_start..spec_end];
                    if !is_name_start(spec[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    for (offset, &byte) in spec.iter().enumerate().skip(1) {
                        if !is_name_continue(byte) {
                            return Err(Error::InvalidName {
                                pos: spec_start + offset,
                            });
                        }
                    }

                    let name = std::str::from_utf8(spec).expect("validated ASCII name");
                    if !seen.insert(name.to_owned()) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    names.push(name.to_owned());

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push((open, close + 1, Expression { operator, names }));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for (open, end, expression) in expressions {
        output.push_str(&template[literal_start..open]);

        let mut fields = Vec::new();
        for name in &expression.names {
            if let Some(value) = vars
                .iter()
                .rev()
                .find_map(|(binding, value)| (*binding == name).then_some(*value))

                fields.push(encode(value, expression.operator.is_some()));
            }

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }
        literal_start = end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let fields: Vec<&str> = expression
            .names
            .iter()
            .filter_map(|name| vars.iter().rev().find(|(key, _)| key == name).map(|(_, value)| *value))
            .collect();

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            let allow_reserved = matches!(expression.operator, Some('+') | Some('#'));
            for (index, value) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                encode_value(value, allow_reserved, &mut output);
            }
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let rest = &template[cursor..];
        let Some(relative_open) = rest.find(['{', '}']) else {
            break;
        };
        let brace_pos = cursor + relative_open;
        let brace = template.as_bytes()[brace_pos];

        if brace == b'}' {
            return Err(Error::UnexpectedClose { pos: brace_pos });
        }

        let content_start = brace_pos + 1;
        let Some(relative_close) = template[content_start..].find('}') else {
            return Err(Error::Unclosed { pos: brace_pos });
        };
        let close_pos = content_start + relative_close;

        if let Some(relative_nested) = template[content_start..close_pos].find('{') {
            return Err(Error::NestedOpen {
                pos: content_start + relative_nested,
            });
        }

        let content = &template[content_start..close_pos];
        let (operator, body, body_start) = match content.as_bytes().first().copied() {
            Some(byte @ (b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')) => {
                let op = byte as char;
                if op != '+' && op != '#' {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                }
                (Some(op), &content[1..], content_start + 1)
            }
            _ => (None, content, content_start),
        };

        if body.is_empty() {
            return Err(Error::EmptyExpression { pos: close_pos });
        }

        let mut names = Vec::new();
        let mut seen: Vec<&str> = Vec::new();
        let mut part_start = 0;

        loop {
            let next_comma = body[part_start..].find(',');
            let part_end = next_comma.map_or(body.len(), |offset| part_start + offset);
            let name = &body[part_start..part_end];
            let name_pos = body_start + part_start;

            if name.is_empty() {
                return Err(Error::EmptyVariable { pos: name_pos });
            }

            if let Some(invalid_offset) = invalid_name_offset(name) {
                return Err(Error::InvalidName {
                    pos: name_pos + invalid_offset,
                });
            }

            if seen.contains(&name) {
                return Err(Error::DuplicateVariable { pos: name_pos });
            }
            seen.push(name);
            names.push(name);

            match next_comma {
                Some(_) => part_start = part_end + 1,
                None => break,
            }
        }

        expressions.push(Expression {
            start: brace_pos,
            end: close_pos,
            operator,
            names,
        });
        cursor = close_pos + 1;
    }

    Ok(expressions)
}

fn invalid_name_offset(name: &str) -> Option<usize> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Some(0);
    }
    if !is_name_start(bytes[0]) {
        return Some(0);
    }
    bytes
        .iter()
        .enumerate()
        .skip(1)
        .find_map(|(index, byte)| (!is_name_continue(*byte)).then_some(index))
}
