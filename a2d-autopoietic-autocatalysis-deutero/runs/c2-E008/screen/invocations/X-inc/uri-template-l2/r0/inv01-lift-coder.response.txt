#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let op = bytes[content_start];
        if op != b'+' && op != b'#' {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        content_start += 1;
        Some(op)
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        for (offset, &byte) in name_bytes.iter().enumerate() {
            let valid = if offset == 0 {
                is_name_start(byte)
            } else {
                is_name_continue(byte)
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open = pos;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;
                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close + 1, expression));
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|&&(binding_name, _)| binding_name == name)
        .map(|&(_, value)| value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'('
            | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn push_encoded(output: &mut String, value: &str, allow_reserved: bool) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        i += 1;
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[literal_start..start]);

        let allow_reserved = matches!(expression.operator, Some(b'+' | b'#'));
        let mut fields = Vec::new();

        for variable in expression.variables {
            let _ = variable.start;
            if let Some(value) = lookup(variable.name, vars) {
                let mut field = String::new();
                push_encoded(&mut field, value, allow_reserved);
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }
            for (index, field) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                output.push_str(field);
            }
        }

        literal_start = end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
