#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: &Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn canonicalize(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by(|a, b| {
        constraint_rank(a)
            .cmp(&constraint_rank(b))
            .then_with(|| match (a, b) {
                (Constraint::Exact(x), Constraint::Exact(y))
                | (Constraint::AtLeast(x), Constraint::AtLeast(y))
                | (Constraint::Before(x), Constraint::Before(y))
                | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(y),
                _ => std::cmp::Ordering::Equal,
            })
    });
    constraints.dedup();
    constraints
}

fn constraint_text(c: &Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(*v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(*v)),
        Constraint::Before(v) => format!("<{}", version_text(*v)),
        Constraint::Caret(v) => format!("^{}", version_text(*v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonicalize(constraints.to_vec())
        .iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut out = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{:02X}", byte));
        }
    }
    out
}

fn permits(constraint: Constraint, v: Version) -> bool {
    match constraint {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, req) in roots.iter().enumerate() {
        if !valid_name(&req.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: req.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    for i in 0..catalog.len() {
        if catalog[..i]
            .iter()
            .any(|p| p.name == catalog[i].name)
        {
            return Err(ResolveError::DuplicatePackage {
                package: catalog[i].name.clone(),
            });
        }
    }

    for package in catalog {
        for i in 0..package.releases.len() {
            let version = package.releases[i].version;
            if package.releases[..i]
                .iter()
                .any(|release| release.version == version)
            {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version,
                });
            }
        }
    }
    Ok(())
}

fn active_constraints(
    roots: &[Requirement],
    selected: &[(String, Version)],
    catalog_by_name: &std::collections::BTreeMap<&str, &Package>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::new();

    for req in roots {
        active
            .entry(req.package.clone())
            .or_insert_with(Vec::new)
            .push(req.constraint);
    }

    for (name, version) in selected {
        if let Some(package) = catalog_by_name.get(name.as_str()) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_insert_with(Vec::new)
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonicalize(std::mem::take(constraints));
    }
    active
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;
    validate_duplicates(catalog)?;

    let catalog_by_name: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();
    let mut selected: Vec<(String, Version)> = Vec::new();

    loop {
        let active = active_constraints(roots, &selected, &catalog_by_name);

        if let Some(name) = active
            .keys()
            .filter(|name| !catalog_by_name.contains_key(name.as_str()))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in &selected {
            if let Some(constraints) = active.get(name) {
                if !constraints
                    .iter()
                    .all(|constraint| permits(*constraint, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let mut next: Option<(String, Version)> = None;
        for (name, constraints) in &active {
            if selected.iter().any(|(selected_name, _)| selected_name == name) {
                continue;
            }
            let package = catalog_by_name[name.as_str()];
            let best = package
                .releases
                .iter()
                .map(|release| release.version)
                .filter(|version| {
                    constraints
                        .iter()
                        .all(|constraint| permits(*constraint, *version))
                })
                .max();

            match best {
                None => {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
                Some(version) if next.is_none() => next = Some((name.clone(), version)),
                Some(_) => {}
            }
        }

        match next {
            Some(choice) => selected.push(choice),
            None => {
                selected.sort_by(|a, b| a.0.cmp(&b.0));
                return Ok(selected
                    .into_iter()
                    .map(|(package, version)| Selection { package, version })
                    .collect());
            }
        }
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
    }
}
