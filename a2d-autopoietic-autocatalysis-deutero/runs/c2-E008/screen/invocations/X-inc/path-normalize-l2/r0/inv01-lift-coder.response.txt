#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a, I>(
    components: I,
    stack: &mut Vec<&'a str>,
    absolute: bool,
) -> Result<(), usize>
where
    I: IntoIterator<Item = (&'a str, usize)>,
{
    for (component, offset) in components {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|last| *last != "..") {
                stack.pop();
            } else if absolute {
                return Err(offset);
            } else {
                stack.push(component);
            }
        } else {
            stack.push(component);
        }
    }
    Ok(())
}

fn components_with_offsets(path: &str, base_offset: usize) -> Vec<(&str, usize)> {
    let mut result = Vec::new();
    let mut start = 0;
    for (index, byte) in path.bytes().enumerate() {
        if byte == b'/' {
            if start < index {
                result.push((&path[start..index], base_offset + start));
            }
            start = index + 1;
        }
    }
    if start < path.len() {
        result.push((&path[start..], base_offset + start));
    }
    result
}

fn render(stack: &[&str], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        result = "/".to_string();
    }

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let components = components_with_offsets(path, 0);
    let mut stack = Vec::new();

    if let Err(at) = reduce_components(components, &mut stack, absolute) {
        return Err(PathError::Escape { at });
    }

    Ok(render(&stack, absolute, intent))
}

fn normalize_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_string)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = normalize_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(normalize_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let expanded_tilde = bytes.first() == Some(&b'~');
    let tilde_slash = expanded_tilde && bytes.get(1) == Some(&b'/');

    if expanded_tilde && !tilde_slash && path.len() != 1 {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    let absolute = !expanded_tilde && bytes.first() == Some(&b'/');
    let intent = directory_intent(path);

    let (mut stack, remainder, offset) = if expanded_tilde {
        let home_stack = home_components.ok_or(PathError::MissingHome)?;
        let start = if tilde_slash { 2 } else { 1 };
        (home_stack.iter().map(String::as_str).collect(), &path[start..], start)
    } else if absolute {
        (Vec::new(), path, 0)
    } else {
        (cwd_components.iter().map(String::as_str).collect(), path, 0)
    };

    let components = components_with_offsets(remainder, offset);
    if let Err(at) = reduce_components(components, &mut stack, true) {
        return Err(PathError::Escape { at });
    }

    Ok(render(&stack, true, intent))
}
