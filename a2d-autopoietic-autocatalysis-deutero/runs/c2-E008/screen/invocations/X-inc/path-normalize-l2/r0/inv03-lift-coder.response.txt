#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|component| *component == "." || *component == "..")
}

fn reduce<'a>(
    stack: &mut Vec<&'a str>,
    path: &'a str,
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let mut cursor = 0;
    for component in path.split('/') {
        let at = offset + cursor;
        cursor += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|last| *last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push(component);
            }
        } else {
            stack.push(component);
        }
    }
    Ok(())
}

fn render(stack: &[&str], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let directory = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute)?;

    Ok(render(&stack, absolute, directory))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let normalized_cwd = if cwd.starts_with('/') {
        normalize(cwd).map_err(|_| PathError::InvalidCwd)?
    } else {
        return Err(PathError::InvalidCwd);
    };
    let cwd_components = components(&normalized_cwd);

    let normalized_home = match home {
        Some(home_path) => {
            if !home_path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(normalize(home_path).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate_raw(path)?;

    let mut stack: Vec<&str>;
    let mut absolute;
    let mut remainder = path;
    let mut offset = 0;

    if path.starts_with('~') {
        if path.len() > 1 && !path.as_bytes()[1..].starts_with(b"/") {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }

        let home_path = normalized_home.as_deref().ok_or(PathError::MissingHome)?;
        stack = components(home_path);
        absolute = true;

        if path == "~" {
            remainder = "";
            offset = path.len();
        } else {
            remainder = &path[2..];
            offset = 2;
        }
    } else if path.starts_with('/') {
        stack = Vec::new();
        absolute = true;
    } else {
        stack = cwd_components;
        absolute = true;
    }

    reduce(&mut stack, remainder, offset, absolute)?;

    Ok(render(&stack, absolute, directory_intent(path)))
}
