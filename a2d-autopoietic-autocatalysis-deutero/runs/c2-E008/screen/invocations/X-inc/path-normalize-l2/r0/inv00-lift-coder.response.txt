#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|part| *part == "." || *part == "..")
}

fn normalize_components<'a>(
    parts: impl IntoIterator<Item = (&'a str, usize)>,
    absolute: bool,
) -> Result<Vec<String>, PathError> {
    let mut result: Vec<String> = Vec::new();

    for (part, at) in parts {
        match part {
            "." => {}
            ".." => {
                if result.last().is_some_and(|last| last != "..") {
                    result.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    result.push("..".to_owned());
                }
            }
            _ => result.push(part.to_owned()),
        }
    }

    Ok(result)
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = parts.join("/");

    if absolute {
        result.insert(0, '/');
    } else if result.is_empty() {
        result.push('.');
    }

    if intent && result != "/" && result != "." {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let parts = components(path)
        .into_iter()
        .map(|part| {
            let at = part.as_ptr() as usize - path.as_ptr() as usize;
            (part, at)
        });
    let reduced = normalize_components(parts, absolute)?;
    Ok(render(&reduced, absolute, directory_intent(path)))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;

    let home_parts = match home {
        Some(home) => Some(validate_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let expanded_tilde = bytes[0] == b'~';
    let (absolute, mut result, remainder, offset) = if expanded_tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let parts = home_parts.ok_or(PathError::MissingHome)?;
        let offset = if bytes.len() > 1 { 2 } else { 1 };
        (true, parts, &path[offset..], offset)
    } else if bytes[0] == b'/' {
        (true, Vec::new(), path, 0)
    } else {
        (true, cwd_parts, path, 0)
    };

    for part in remainder.split('/').filter(|part| !part.is_empty()) {
        let at = part.as_ptr() as usize - path.as_ptr() as usize;
        match part {
            "." => {}
            ".." => {
                if result.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => result.push(part.to_owned()),
        }
    }

    let _ = (offset, absolute);
    Ok(render(&result, true, directory_intent(path)))
}
