fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128).unsigned_abs();
    let dy = (a.y as i128 - b.y as i128).unsigned_abs();
    let scaled_squared = (dx * dx + dy * dy) * 1_000_000;

    // Integer square root, rounded to the nearest integer with half ties upward.
    let mut low = 0u128;
    let mut high = 2_000_000_001u128;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid * mid <= scaled_squared {
            low = mid;
        } else {
            high = mid;
        }
    }

    let lower_square = low * low;
    let upper_square = (low + 1) * (low + 1);
    if scaled_squared - lower_square >= upper_square - scaled_squared {
        (low + 1) as u64
    } else {
        low as u64
    }
}

    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn prepare(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT

        }

        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    Ok(points[..effective_len].to_vec())
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (i128::from(a.x) - i128::from(b.x)).unsigned_abs();
    let dy = (i128::from(a.y) - i128::from(b.y)).unsigned_abs();
    let squared = dx * dx + dy * dy;
    let scaled_squared = squared * 1_000_000;

    // Find floor(sqrt(scaled_squared)) with integer arithmetic.
    let mut low = 0u128;
    let mut high = 2_000_000_000_000u128;
    while low < high {
        let mid = low + (high - low + 1) / 2;
        if mid * mid <= scaled_squared {
            low = mid;
        } else {
            high = mid - 1;
        }
    }

    let floor_root = low;
    let remainder = scaled_squared - floor_root * floor_root;
    // Round up iff sqrt(scaled_squared) is at least floor_root + 1/2.
    let round_up = remainder * 4 >= floor_root * 4 + 1;
    (floor_root + u128::from(round_up)) as u64
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare(points)?;
    let n = polygon.len();

    let mut signed_area2 = 0i128;
    for i in 0..n {
        let current = polygon[i];
        let next = polygon[(i + 1) % n];
        signed_area2 += i128::from(current.x) * i128::from(next.y)
            - i128::from(current.y) * i128::from(next.x);
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];
        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace_longest = match longest_edge {
            None => true,
            Some(best_index) => {
                if length > edge_milli[best_index] {
                    true
                } else if length < edge_milli[best_index] {
                    false
                } else {
                    let edge_key = if a <= b { (a, b) } else { (b, a) };
                    let best_a = polygon[best_index];
                    let best_b = polygon[(best_index + 1) % n];

                    } else {
                        (best_b, best_a)
                    };
                    edge_key < best_key || (edge_key == best_key && i < best_index)
                }
            }
        };

            longest_edge = Some(i);
        }

        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

    OutOfRange { index: usize },

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_sqrt(n: u128) -> u128 {
    // Integer square root, using the restoring bit-by-bit algorithm.
    let mut remainder = n;
    let mut root = 0u128;
    let mut bit = 1u128 << 126; // Largest power of four representable in u128.
    while bit > remainder {
        bit >>= 2;
    }
    while bit != 0 {
        if remainder >= root + bit {
            remainder -= root + bit;
            root = (root >> 1) + bit;
        } else {
            root >>= 1;
        }
        bit >>= 2;
    }

    // Round to the nearest integer; ties round upward. The midpoint comparison
    // avoids squaring a potentially large candidate.
    if remainder > root {
        root + 1
    } else {
        root
    }
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = (i128::from(a.x) - i128::from(b.x)).unsigned_abs();
    let dy = (i128::from(a.y) - i128::from(b.y)).unsigned_abs();
    let scaled_squared = (dx * dx + dy * dy) * 1_000_000;
    rounded_sqrt(scaled_squared) as u64
}

    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT

        }

        points.len() - 1
    } else {
        points.len()
    };

    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0i128;

    for index in 0..effective_len {
        let a = polygon[index];
        let b = polygon[(index + 1) % effective_len];

        signed_area2 += i128::from(a.x) * i128::from(b.y)
            - i128::from(a.y) * i128::from(b.x);

        edge_milli.push(edge_length_milli(a, b));
    }

    let perimeter_milli = edge_milli.iter().copied().sum();
    let longest_edge = (0..effective_len).max_by(|&i, &j| {

                    let b = polygon[(index + 1) % effective_len];
                    if a <= b { (a, b) } else { (b, a) }
                };
                pair(j).cmp(&pair(i))
            })

    });

        area2: signed_area2.unsigned_abs(),

        edge_milli,
        perimeter_milli,
        longest_edge,
    })

fn round_sqrt_nearest(value: u128) -> u64 {
    // Integer square root by binary search. All valid squared edge lengths
    // after scaling are at most 8 * 10^24.
    let mut low = 0_u128;
    let mut high = value + 1;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid <= value / mid {
            low = mid;
        } else {
            high = mid;
        }
    }

    let lower_sq = low * low;
    let remainder = value - lower_sq;
    let gap = 2 * low + 1;
    let rounded = if remainder * 2 >= gap { low + 1 } else { low };
    rounded as u64
}
