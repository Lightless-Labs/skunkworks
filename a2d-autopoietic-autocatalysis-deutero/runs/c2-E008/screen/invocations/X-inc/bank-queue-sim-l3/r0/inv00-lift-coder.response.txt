#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    pause_time: Option<u64>,
    remaining: u64,
}

#[derive(Clone)]
struct Running {
    job_index: usize,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(at), Some(bt)) => at.cmp(&bt).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers
        .iter()
        .cloned()
        .map(|customer| Job {
            remaining: customer.service,
            customer,
            started: false,
            pause_time: None,
        })
        .collect();

    let mut trace = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0;
    let mut completed = 0;
    let mut now = customers[order[0]].arrival;

    while completed < customers.len() {
        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let job_index = active.job_index;
                trace.push(Event {
                    time: now,
                    customer: jobs[job_index].customer.id,
                    kind: Kind::Finish,
                });
                running = None;
                completed += 1;
            }
        }

        let cohort_start = next_arrival;
        while next_arrival < order.len()
            && customers[order[next_arrival]].arrival == now
        {
            let job_index = order[next_arrival];
            trace.push(Event {
                time: now,
                customer: jobs[job_index].customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(job_index);
            next_arrival += 1;
        }
        let _ = cohort_start;

        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                let best = waiting
                    .iter()
                    .copied()
                    .min_by(|&a, &b| waiter_cmp(&jobs[a], &jobs[b]));
                if let Some(best_index) = best {
                    let active_index = active.job_index;
                    if jobs[best_index].customer.priority < jobs[active_index].customer.priority {
                        let remaining = active.finish - now;
                        trace.push(Event {
                            time: now,
                            customer: jobs[active_index].customer.id,
                            kind: Kind::Preempt {
                                remaining,
                                by: jobs[best_index].customer.id,
                            },
                        });
                        jobs[active_index].remaining = remaining;
                        jobs[active_index].pause_time = Some(now);
                        waiting.push(active_index);
                        running = None;
                    }
                }
            }
        }

        if running.is_none() {
            loop {
                let Some((position, &job_index)) = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, &a), (_, &b)| waiter_cmp(&jobs[a], &jobs[b]))
                else {
                    break;
                };
                waiting.remove(position);

                let job = &jobs[job_index];
                let customer_id = job.customer.id;
                let remaining = job.remaining;
                let is_resume = job.started;
                let cost = if last_dispatched == Some(customer_id) {
                    0
                } else if last_dispatched.is_none() {
                    0
                } else {
                    switch_cost
                };

                let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: customer_id,
                    time: now,
                })?;
                let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                    customer: customer_id,
                    time: now,
                })?;

                jobs[job_index].started = true;
                jobs[job_index].pause_time = None;
                last_dispatched = Some(customer_id);

                trace.push(Event {
                    time: now,
                    customer: customer_id,
                    kind: if is_resume {
                        Kind::Resume {
                            remaining,
                            work_at,
                            finish,
                        }
                    } else {
                        Kind::Start {
                            remaining,
                            work_at,
                            finish,
                        }
                    },
                });

                if finish == now {
                    trace.push(Event {
                        time: now,
                        customer: customer_id,
                        kind: Kind::Finish,
                    });
                    completed += 1;
                    continue;
                }

                running = Some(Running {
                    job_index,
                    work_at,
                    finish,
                });
                break;
            }
        }

        if completed == customers.len() {
            break;
        }

        let mut next_time = None;
        if next_arrival < order.len() {
            next_time = Some(customers[order[next_arrival]].arrival);
        }
        if let Some(active) = running.as_ref() {
            if active.finish > now {
                next_time = Some(next_time.map_or(active.finish, |t| t.min(active.finish)));
            }
            if active.work_at > now {
                next_time = Some(next_time.map_or(active.work_at, |t| t.min(active.work_at)));
            }
        }

        match next_time {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(trace)
}
