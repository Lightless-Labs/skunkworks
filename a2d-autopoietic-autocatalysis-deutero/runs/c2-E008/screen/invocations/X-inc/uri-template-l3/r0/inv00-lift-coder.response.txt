#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Variable<'a> {
    name: &'a str,
    name_pos: usize,
    value: &'a str,
}

#[derive(Clone, Copy)]
struct Expression<'a> {
    operator: Option<char>,
    variables: Vec<Variable<'a>>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;

                if let Some(relative_nested) =
                    bytes[open + 1..close].iter().position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_nested,
                    });
                }

                expressions.push(parse_expression(template, open, close)?);
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    cursor = 0;
    let mut expression_index = 0;

    while cursor < bytes.len() {
        if bytes[cursor] == b'{' {
            let expression = &expressions[expression_index];
            expression_index += 1;

            let mut fields = Vec::new();
            for variable in &expression.variables {
                if let Some(value) = vars
                    .iter()
                    .rev()
                    .find_map(|(name, value)| (*name == variable.name).then_some(*value))
                {
                    fields.push((variable.name, value));
                }
            }

            if !fields.is_empty() {
                match expression.operator {
                    Some('#') => output.push('#'),
                    Some('.') => output.push('.'),
                    Some('/') => output.push('/'),
                    Some(';') => output.push(';'),
                    Some('?') => output.push('?'),
                    Some('&') => output.push('&'),
                    _ => {}
                }

                let separator = match expression.operator {
                    Some('.') => '.',
                    Some('/') => '/',
                    Some(';') | Some('&') | Some('?') => '&',
                    _ => ',',
                };

                for (index, (name, value)) in fields.iter().enumerate() {
                    if index > 0 {
                        output.push(separator);
                    }

                    match expression.operator {
                        Some(';') => {
                            output.push_str(name);
                            let encoded = encode(value, false);
                            if !encoded.is_empty() {
                                output.push('=');
                                output.push_str(&encoded);
                            }
                        }
                        Some('?') | Some('&') => {
                            output.push_str(name);
                            output.push('=');
                            output.push_str(&encode(value, false));
                        }
                        _ => output.push_str(&encode(
                            value,
                            matches!(expression.operator, Some('+') | Some('#')),
                        )),
                    }
                }
            }

            cursor += 1;
            while cursor < bytes.len() && bytes[cursor] != b'}' {
                cursor += 1;
            }
            cursor += 1;
        } else {
            let ch = template[cursor..].chars().next().unwrap();
            output.push(ch);
            cursor += ch.len_utf8();
        }
    }

    Ok(output)
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;
    let operator = match bytes[content_start] {
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' => {
            let operator = bytes[content_start] as char;
            content_start += 1;
            Some(operator)
        }
        _ => None,
    };

    if let Some(operator) = operator {
        if !matches!(operator, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            return Err(Error::UnsupportedOperator { pos: content_start - 1 });
        }
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map_or(close, |relative| spec_start + relative);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        for (index, &byte) in name_bytes.iter().enumerate() {
            let valid = if index == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: spec_start + index,
                });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        variables.push(Variable {
            name,
            name_pos: spec_start,
            value: "",
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

const HEX: &[u8; 16] = b"0123456789ABCDEF";

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
