#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn parse(byte: u8, pos: usize) -> Result<Option<Self>, Error> {
        let op = match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Query),
            b'&' => Some(Self::Continuation),
            b'!' | b'=' | b',' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos });
            }
            _ => None,
        };
        Ok(op)
    }

    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon | Self::Continuation | Self::Query => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continuation => Some('&'),
        }
    }

    fn named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continuation)
    }

    fn semicolon(self) -> bool {
        matches!(self, Self::Semicolon)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::with_capacity(bytes.len());

    let mut i = 0;
    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(HEX[high as usize] as char);
                out.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || (reserved && is_reserved(byte))
        {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }

        i += 1;
    }

    out
}

fn expand_expression(
    content: &str,
    content_pos: usize,
    closing_pos: usize,
    vars: &[(&str, &str)],
) -> Result<String, Error> {
    let bytes = content.as_bytes();
    let (operator, start) = if let Some(&first) = bytes.first() {
        match Operator::parse(first, content_pos)? {
            Some(operator) => (operator, 1),
            None => (Operator::Simple, 0),
        }
    } else {
        return Err(Error::EmptyExpression { pos: closing_pos });
    };

    if start == bytes.len() {
        return Err(Error::EmptyExpression { pos: closing_pos });
    }

    let mut fields: Vec<(String, String)> = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = start;

    for index in start..=bytes.len() {
        if index != bytes.len() && bytes[index] != b',' {
            continue;
        }

        let spec_pos = content_pos + spec_start;
        if index == spec_start {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let spec = &content[spec_start..index];
        let spec_bytes = spec.as_bytes();

        if !is_name_start(spec_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_pos });
        }
        for (offset, &byte) in spec_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: spec_pos + offset,
                });
            }
        }

        if seen.iter().any(|name| *name == spec) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }
        seen.push(spec);

        if let Some((_, value)) = vars.iter().rev().find(|(name, _)| *name == spec) {
            let encoded = encode(value, operator.reserved());
            fields.push((spec.to_string(), encoded));
        }

        spec_start = index + 1;
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let separator = operator.separator();
    let mut out = String::new();
    if let Some(prefix) = operator.prefix() {
        out.push(prefix);
    }

    for (index, (name, value)) in fields.iter().enumerate() {
        if index > 0 {
            out.push(separator);
        }
        if operator.named() {
            out.push_str(name);
            if operator.semicolon() && value.is_empty() {
                continue;
            }
            out.push('=');
        }
        out.push_str(value);
    }

    Ok(out)
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[cursor + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = cursor + 1 + relative_close;

                if let Some(relative_open) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_open,
                    });
                }

                expressions.push((open, close));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::with_capacity(template.len());
    let mut literal_start = 0;

    for (open, close) in expressions {
        output.push_str(&template[literal_start..open]);
        output.push_str(&expand_expression(
            &template[open + 1..close],
            open + 1,
            close,
            vars,
        )?);
        literal_start = close + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
