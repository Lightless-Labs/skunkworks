#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Query),
            b'&' => Some(Self::Continuation),
            _ => None,
        }
    }

    fn separator(self) -> u8 {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => b',',
            Self::Dot => b'.',
            Self::Slash => b'/',
            Self::Semicolon | Self::Query | Self::Continuation => b'&',
        }
    }

    fn prefix(self) -> Option<u8> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some(b'#'),
            Self::Dot => Some(b'.'),
            Self::Slash => Some(b'/'),
            Self::Semicolon => Some(b';'),
            Self::Query => Some(b'?'),
            Self::Continuation => Some(b'&'),
        }
    }

    fn is_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }

    fn is_named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continuation)
    }
}

struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let operator = if content_start < end {
        if let Some(op) = Operator::from_byte(bytes[content_start]) {
            content_start += 1;
            op
        } else if matches!(bytes[content_start], b'=' | b'!' | b'@' | b'|' | b'$' | b'%' | b'^' | b'<' | b'>' | b'\\') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        } else {
            Operator::Simple
        }
    } else {
        Operator::Simple
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut part_start = content_start;

    loop {
        let mut part_end = part_start;
        while part_end < end && bytes[part_end] != b',' {
            part_end += 1;
        }

        if part_start == part_end {
            return Err(Error::EmptyVariable { pos: part_start });
        }

        let name_bytes = &bytes[part_start..part_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: part_start });
        }
        for (offset, byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(*byte) {
                return Err(Error::InvalidName {
                    pos: part_start + offset,
                });
            }
        }

        let name = &template[part_start..part_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: part_start });
        }
        seen.push(name);
        names.push((part_start, part_end));

        if part_end == end {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        names,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[start + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + offset,
                    });
                }

                expressions.push(parse_expression(template, start, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn push_percent(output: &mut String, byte: u8) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    output.push('%');
    output.push(HEX[(byte >> 4) as usize] as char);
    output.push(HEX[(byte & 0x0f) as usize] as char);
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(b"0123456789ABCDEF"[high as usize] as char);
                output.push(b"0123456789ABCDEF"[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            push_percent(&mut output, byte);
        }
        i += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        let operator = expression.operator;
        let mut fields = Vec::new();

        for (name_start, name_end) in expression.names {
            let name = &template[name_start..name_end];
            if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
                fields.push((name, encode(value, operator.is_reserved())));
            }
        }

        if !fields.is_empty() {
            if let Some(prefix) = operator.prefix() {
                output.push(prefix as char);
            }

            for (index, (name, value)) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(operator.separator() as char);
                }
                if operator.is_named() {
                    output.push_str(name);
                    if operator == Operator::Semicolon && value.is_empty() {
                        continue;
                    }
                    output.push('=');
                }
                output.push_str(value);
            }
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
