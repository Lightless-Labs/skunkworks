```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_owned()),
            }
        }
        Self::canonical(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_owned();
        }
        match path.rfind('/') {
            Some(0) => "/".to_owned(),
            Some(i) => path[..i].to_owned(),
            _ => "/".to_owned(),
        }
    }

    fn basename(path: &str) -> &str {
        path.rsplit('/').next().unwrap_or("")
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn contains(parent: &str, child: &str) -> bool {
        parent == child
            || (parent == "/" && child.starts_with('/'))
            || (parent != "/" && child.starts_with(&format!("{}/", parent)))
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| *p != path && Self::parent(p) == path)
            .cloned()
            .collect()
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        let mut todo: Vec<String> = Self::components(path);
        let mut physical = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        while !todo.is_empty() {
            let component = todo.remove(0);
            physical.push(component);
            let candidate = Self::canonical(&physical);
            let is_final = todo.is_empty();

            let Some(entry) = self.entries.get(&candidate) else {
                if create_final && is_final {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            };

            if let Entry::Link(target) = entry {
                if !is_final || follow_final {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;

                    let mut expanded = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        physical[..physical.len() - 1].to_vec()
                    };
                    for part in target.split('/') {
                        match part {
                            "" | "." => {}
                            ".." => {
                                expanded.pop();
                            }
                            other => expanded.push(other.to_owned()),
                        }
                    }
                    expanded.extend(todo.into_iter());
                    todo = expanded;
                    physical.clear();
                    continue;
                }
            }

            if !is_final && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
        }

        Ok(Self::canonical(&physical))
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        Ok((resolved_parent, Self::basename(path).to_owned()))
    }

    fn remove_subtree(&mut self, path: &str) {
        let prefix = if path == "/" {
            "/".to_owned()
        } else {
            format!("{}/", path)
        };
        self.entries
            .retain(|p, _| p != path && !p.starts_with(&prefix));
    }

    fn subtree_size(&self, path: &str) -> u64 {
        match self.entries.get(path) {
            Some(Entry::File(data)) => data.len() as u64,
            Some(Entry::Link(target)) => target.len() as u64,
            Some(Entry::Dir) => self
                .children(path)
                .iter()
                .map(|child| self.subtree_size(child))
                .sum(),
            None => 0,
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = Self::join(&parent, &name);
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, true)?;
        if matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::IsDir(resolved));
        }
        self.entries
            .insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        let mut result: Vec<(String, Kind)> = self
            .children(&resolved)
            .into_iter()
            .filter_map(|child| {
                self.entries
                    .get(&child)
                    .map(|entry| (Self::basename(&child).to_owned(), entry.kind()))
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        Self::validate(to)?;
        let source = Self::normalize(from, &self.cwd);
        let destination = Self::normalize(to, &self.cwd);
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::join(&source_parent, &source_name);
        let Some(source_entry) = self.entries.get(&source_path).cloned() else {
            return Err(Error::Missing(source_path));
        };

        if matches!(source_entry, Entry::Dir) && Self::contains(&source_path, &self.cwd) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = Self::join(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::contains(&destination_path, &self.cwd)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains(&source_path, &destination_path) {
            return Err(Error::Cycle(destination_path));
        }

        let dest_entry = self.entries.get(&destination_path).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref dest) = dest_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let source_prefix = format!("{}/", source_path);
        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| *p == &source_path || p.starts_with(&source_prefix))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if dest_entry.is_some() {
            self.remove_subtree(&destination_path);
        }
        self.remove_subtree(&source_path);

        for (old_path, entry) in moved {
            let suffix = &old_path[source_path.len()..];
            let new_path = format!("{}{}", destination_path, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        if normalized == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = Self::join(&parent, &name);
        let Some(entry) = self.entries.get(&target) else {
            return Err(Error::Missing(target));
        };

        if matches!(entry, Entry::Dir) && Self::contains(&target, &self.cwd) {
            return Err(Error::Busy(target));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        self.remove_subtree(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let normalized = Self::normalize(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (parent, name) = self.resolve_parent(&normalized)?;
        let link_path = Self::join(&parent, &name);
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries
            .insert(link_path, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        let mut rows = Vec::new();
        let prefix = if resolved == "/" {
            "/".to_owned()
        } else {
            format!("{}/", resolved)
        };
        for entry_path in self.entries.keys() {
            if entry_path == &resolved || entry_path.starts_with(&prefix) {
                rows.push((entry_path.clone(), self.subtree_size(entry_path)));
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
