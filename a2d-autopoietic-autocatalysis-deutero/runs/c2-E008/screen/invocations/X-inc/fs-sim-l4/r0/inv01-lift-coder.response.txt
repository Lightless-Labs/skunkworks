```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate_path(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }
        Self::canonical(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            _ => "/".to_string(),
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            "/".to_string()
        } else {
            path.rsplit('/').next().unwrap_or("").to_string()
        }
    }

    fn is_within(path: &str, ancestor: &str) -> bool {
        if ancestor == "/" {
            return true;
        }
        path == ancestor
            || path
                .strip_prefix(ancestor)
                .is_some_and(|rest| rest.starts_with('/'))
    }

    fn join_child(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        allow_missing_final: bool,
    ) -> Result<String, Error> {
        let mut todo: Vec<String> = Self::components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;
        let mut idx = 0usize;

        while idx < todo.len() {
            let part = todo[idx].clone();
            let candidate_parts = {
                let mut p = resolved.clone();
                p.push(part.clone());
                p
            };
            let candidate = Self::canonical(&candidate_parts);
            let is_final = idx + 1 == todo.len();

            match self.entries.get(&candidate) {
                None => {
                    if is_final && allow_missing_final {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
                Some(Entry::Dir) => {
                    resolved.push(part);
                    idx += 1;
                }
                Some(Entry::File(_)) => {
                    if !is_final {
                        return Err(Error::NotDir(candidate));
                    }
                    resolved.push(part);
                    idx += 1;
                }
                Some(Entry::Link(target)) if follow_final || !is_final => {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let target_base = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        resolved.clone()
                    };
                    let mut target_parts = target_base;
                    for piece in target.split('/') {
                        match piece {
                            "" | "." => {}
                            ".." => {
                                target_parts.pop();
                            }
                            _ => target_parts.push(piece.to_string()),
                        }
                    }
                    let suffix = todo[idx + 1..].to_vec();
                    let mut combined = target_parts;
                    for piece in suffix {
                        match piece.as_str() {
                            "" | "." => {}
                            ".." => {
                                combined.pop();
                            }
                            _ => combined.push(piece),
                        }
                    }
                    let expanded = Self::canonical(&combined);
                    todo = Self::components(&expanded);
                    resolved.clear();
                    idx = 0;
                }
                Some(Entry::Link(_)) => {
                    resolved.push(part);
                    idx += 1;
                }
            }
        }
        Ok(Self::canonical(&resolved.split('/').filter(|s| !s.is_empty()).map(str::to_owned).collect::<Vec<_>>().as_slice()))
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let name = Self::basename(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn direct_children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| *p != path && Self::parent(p) == path)
            .cloned()
            .collect()
    }

    fn subtree_entries(&self, root: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| Self::is_within(p, root))
            .cloned()
            .collect()
    }

    fn validate_target(target: &str) -> Result<(), Error> {
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        Ok(())
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let (parent, name) = self.resolve_parent(&canonical)?;
        let final_path = Self::join_child(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, true)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }
        let (parent, name) = self.resolve_parent(&resolved)?;
        let final_path = Self::join_child(&parent, &name);
        match self.entries.get(&final_path) {
            Some(Entry::Dir) => Err(Error::IsDir(final_path)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(final_path, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                self.entries
                    .insert(final_path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut result: Vec<(String, Kind)> = self
                    .direct_children(&resolved)
                    .into_iter()
                    .filter_map(|p| {
                        self.entries
                            .get(&p)
                            .map(|entry| (Self::basename(&p), entry.kind()))
                    })
                    .collect();
                result.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(result)
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate_path(from)?;
        Self::validate_path(to)?;
        let source = Self::normalize(from, &self.cwd);
        let destination = Self::normalize(to, &self.cwd);
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&source)?;
        let src = Self::join_child(&src_parent, &src_name);
        let source_entry = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&destination)?;
        let dst = Self::join_child(&dst_parent, &dst_name);

        if matches!(self.entries.get(&dst), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let destination_entry = self.entries.get(&dst).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }

        if overwrite {
            if let Some(ref dest_entry) = destination_entry {
                let src_is_dir = matches!(source_entry, Entry::Dir);
                let dst_is_dir = matches!(dest_entry, Entry::Dir);
                match (src_is_dir, dst_is_dir) {
                    (true, false) => return Err(Error::NotDir(dst)),
                    (false, true) => return Err(Error::IsDir(dst)),
                    (true, true) => {
                        if !self.direct_children(&dst).is_empty() {
                            return Err(Error::NotEmpty(dst));
                        }
                    }
                    _ => {}
                }
            }
        }

        if destination_entry.is_some() {
            for p in self.subtree_entries(&dst) {
                self.entries.remove(&p);
            }
        }

        let moving = self.subtree_entries(&src);
        let mut moved_entries = Vec::new();
        for old in &moving {
            let suffix = if old == &src {
                String::new()
            } else {
                old[src.len()..].to_string()
            };
            let new_path = format!("{dst}{suffix}");
            if let Some(entry) = self.entries.remove(old) {
                moved_entries.push((new_path, entry));
            }
        }
        for (path, entry) in moved_entries {
            self.entries.insert(path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = Self::join_child(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        if matches!(entry, Entry::Dir)
            && !recursive
            && !self.direct_children(&target).is_empty()
        {
            return Err(Error::NotEmpty(target));
        }

        if matches!(entry, Entry::Dir) && recursive {
            for p in self.subtree_entries(&target) {
                self.entries.remove(&p);
            }
        } else {
            self.entries.remove(&target);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate_path(link)?;
        let canonical_link = Self::normalize(link, &self.cwd);
        Self::validate_target(target)?;
        let (parent, name) = self.resolve_parent(&canonical_link)?;
        let final_path = Self::join_child(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries
            .insert(final_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate_path(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let paths = match entry {
            Entry::Dir => self.subtree_entries(&resolved),
            _ => vec![resolved.clone()],
        };
        let mut result = Vec::with_capacity(paths.len());
        for p in paths {
            let size = match self.entries.get(&p).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => self
                    .subtree_entries(&p)
                    .into_iter()
                    .filter_map(|child| {
                        if child == p {
                            None
                        } else {
                            self.entries.get(&child).map(|entry| match entry {
                                Entry::File(data) => data.len() as u64,
                                Entry::Link(target) => target.len() as u64,
                                Entry::Dir => 0,
                            })
                        }
                    })
                    .sum(),
            };
            result.push((p, size));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }
}
```
