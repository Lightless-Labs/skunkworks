#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
            })
            .collect();

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head, false)?;

        let mut best: Option<(usize, &Route)> = None;
        for (route_id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != request_segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;
            for (pattern_segment, request_segment) in
                route.segments.iter().zip(request_segments.iter())
            {
                match pattern_segment {
                    Segment::Static(value) if value == request_segment => {}
                    Segment::Static(_) => {
                        matches = false;
                        break;
                    }
                    Segment::Param(name) => {
                        params.push((name.clone(), request_segment.clone()));
                    }
                }
            }

            if !matches {
                continue;
            }

            let replace = match best {
                None => true,
                Some((best_id, best_route)) => {
                    route.rank > best_route.rank
                        || (route.rank == best_route.rank && route_id < best_id)
                }
            };
            if replace {
                best = Some((route_id, route));
            }
        }

        let (route_id, route) = best.ok_or(Error::NotFound)?;
        let mut params = Vec::new();
        for (pattern_segment, request_segment) in
            route.segments.iter().zip(request_segments.iter())
        {
            if let Segment::Param(name) = pattern_segment {
                params.push((name.clone(), request_segment.clone()));
            }
        }

        Ok(Match {
            route: route_id,
            params,
        })
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    if pattern == "/" {
        return Ok(Vec::new());
    }

    if pattern.ends_with('/') {
        let slash_at = pattern.len() - 1;
        parse_path(&pattern[..slash_at], true)?;
        return Err(Error::InvalidPattern { at: slash_at });
    }

    let raw_segments = split_segments(pattern, true)?;
    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();

    for (start, raw) in raw_segments {
        if raw.starts_with(':') {
            let name = &raw[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }
    }

    Ok(segments)
}

fn parse_path(path: &str, pattern: bool) -> Result<Vec<String>, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !path.starts_with('/') {
        return Err(invalid(0));
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    if path.ends_with('/') {
        let slash_at = path.len() - 1;
        let preceding = &path[..slash_at];
        let _ = parse_nontrailing_segments(preceding, pattern)?;
        return Err(invalid(slash_at));
    }

    parse_nontrailing_segments(path, pattern)
}

fn parse_nontrailing_segments(path: &str, pattern: bool) -> Result<Vec<String>, Error> {
    let raw_segments = split_segments(path, pattern)?;
    raw_segments
        .into_iter()
        .map(|(start, segment)| decode_segment(segment, start, pattern))
        .collect()
}

fn split_segments<'a>(
    path: &'a str,
    pattern: bool,
) -> Result<Vec<(usize, &'a str)>, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let mut result = Vec::new();
    let mut start = 1;
    for (offset, byte) in path.as_bytes().iter().enumerate().skip(1) {
        if *byte == b'/' {
            if offset == start {
                return Err(invalid(start));
            }
            result.push((start, &path[start..offset]));
            start = offset + 1;
        }
    }

    if start == path.len() {
        return Err(invalid(start));
    }
    result.push((start, &path[start..]));
    Ok(result)
}

fn decode_segment(segment: &str, start: usize, pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(start + i));
            }
            let high = uppercase_hex(bytes[i + 1]).ok_or_else(|| invalid(start + i))?;
            let low = uppercase_hex(bytes[i + 2]).ok_or_else(|| invalid(start + i))?;
            let value = (high << 4) | low;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(start + i));
            }

            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid(start + i));
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segment is ASCII"))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}
