#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }

    let Some(size) = board.width.checked_mul(board.height) else {
        return Err(Error::Shape);
    };
    if board.cells.len() != size {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let lines = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..lines {
        let indices: Vec<usize> = match direction {
            Direction::Left => (0..board.width).map(|col| line * board.width + col).collect(),
            Direction::Right => (0..board.width)
                .rev()
                .map(|col| line * board.width + col)
                .collect(),
            Direction::Up => (0..board.height)
                .map(|row| row * board.width + line)
                .collect(),
            Direction::Down => (0..board.height)
                .rev()
                .map(|row| row * board.width + line)
                .collect(),
        };

        let tiles: Vec<(usize, u32)> = indices
            .into_iter()
            .filter_map(|index| {
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut output = Vec::new();
        let mut i = 0;
        while i < tiles.len() {
            if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
                let (first, value) = tiles[i];
                let (second, _) = tiles[i + 1];
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first, second });
                }
                let merged_value = value * 2;
                let to = indices_for_output(direction, board.width, board.height, line, output.len());
                output.push((to, merged_value));
                merges.push(Merge {
                    from: [first, second],
                    to,
                    value: merged_value,
                });
                score += u64::from(merged_value);
                i += 2;
            } else {
                let to = indices_for_output(direction, board.width, board.height, line, output.len());
                output.push((to, tiles[i].1));
                i += 1;
            }
        }

        for (index, value) in output {
            cells[index] = value;
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

fn indices_for_output(
    direction: Direction,
    width: usize,
    height: usize,
    line: usize,
    offset: usize,
) -> usize {
    match direction {
        Direction::Left => line * width + offset,
        Direction::Right => line * width + (width - 1 - offset),
        Direction::Up => offset * width + line,
        Direction::Down => (height - 1 - offset) * width + line,
    }
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let directions = [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ];
    let mut legal = Vec::new();

    for direction in directions {
        if let Ok(result) = move_board(board, direction) {
            if result.changed {
                legal.push(direction);
            }
        }
    }

    Ok(legal)
}
