use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128_parts(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        Self::from_i128_parts(a + b, self.den as i128 * rhs.den as i128)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        Self::from_i128_parts(a - b, self.den as i128 * rhs.den as i128)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128_parts(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128_parts(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| Err(RationalError::Parse { at, kind });
        if bytes.is_empty() {
            return fail(0, ParseKind::Empty);
        }

        let mut slash = None;
        let mut i = 0;
        if bytes[0] == b'-' { i = 1; }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
        if i == num_start {
            if i == bytes.len() { return fail(i, ParseKind::InvalidByte); }
            return fail(i, ParseKind::InvalidByte);
        }
        let num_end = i;
        if i < bytes.len() {
            if bytes[i] != b'/' { return fail(i, ParseKind::InvalidByte); }
            slash = Some(i);
            i += 1;
            let den_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
            if i == den_start {
                return fail(i, ParseKind::InvalidByte);
            }
            if i < bytes.len() { return fail(i, ParseKind::InvalidByte); }
        }

        let negative = bytes[0] == b'-';
        let digits = &text[num_start..num_end];
        if digits.len() >= 2 && digits.as_bytes()[0] == b'0' {
            return fail(num_start + 1, ParseKind::NonCanonical);
        }
        if negative && digits == "0" {
            return fail(0, ParseKind::NonCanonical);
        }

        let den_start = slash.map(|s| s + 1);
        if let Some(ds) = den_start {
            if bytes.len() - ds >= 2 && bytes[ds] == b'0' {
                return fail(ds + 1, ParseKind::NonCanonical);
            }
        }

        let nmag = match digits.parse::<u128>() {
            Ok(v) => v,
            Err(_) => return fail(0, ParseKind::OutOfRange),
        };
        let nlim = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag > nlim { return fail(0, ParseKind::OutOfRange); }

        let dmag = if let Some(ds) = den_start {
            match text[ds..].parse::<u128>() {
                Ok(v) if v <= i64::MAX as u128 => v,
                _ => return fail(ds, ParseKind::OutOfRange),
            }
        } else {
            1
        };

        if let Some(ds) = den_start {
            if dmag == 0 { return fail(ds, ParseKind::ZeroDenominator); }
            if dmag == 1 { return fail(slash.unwrap(), ParseKind::NonCanonical); }
            if nmag == 0 { return fail(slash.unwrap(), ParseKind::NonCanonical); }
            if gcd_u128(nmag, dmag) > 1 { return fail(slash.unwrap(), ParseKind::NonCanonical); }
        }

        let n = if negative { -(nmag as i128) } else { nmag as i128 };
        Self::from_i128_parts(n, dmag as i128)
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) }
        else { write!(f, "{}/{}", self.num, self.den) }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = vec![Rational { num: 0, den: 1 }; self.data.len()];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[c * self.rows + r] = self.data[r * self.cols + c];
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let lhs = self.data[r * self.cols + k];
                    let rv = rhs.data[k * rhs.cols + c];
                    let term = lhs.mul(&rv).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut work = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == work.rows { break; }
            let mut chosen: Option<usize> = None;
            for r in p..work.rows {
                let v = work.data[r * work.cols + col];
                if v.num == 0 { continue; }
                chosen = match chosen {
                    None => Some(r),
                    Some(q) => {
                        let qv = work.data[q * work.cols + col];
                        let key = (v.den, v.num.unsigned_abs(), r);
                        let qkey = (qv.den, qv.num.unsigned_abs(), q);
                        if key < qkey { Some(r) } else { Some(q) }
                    }
                };
            }
            let Some(q) = chosen else { continue; };
            if q != p {
                for c in 0..work.cols {
                    work.data.swap(p * work.cols + c, q * work.cols + c);
                }
                swaps.push((p, q));
            }
            let pivot = work.data[p * work.cols + col];
            for c in 0..work.cols {
                let value = work.data[p * work.cols + c]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
                work.data[p * work.cols + c] = value;
            }
            for r in 0..work.rows {
                if r == p { continue; }
                let factor = work.data[r * work.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..work.cols {
                    let product = factor.mul(&work.data[p * work.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                    let value = work.data[r * work.cols + c].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                    work.data[r * work.cols + c] = value;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix: work, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = Rational { num: 1, den: 1 };
        }
        let augmented = Self { rows: n, cols: width, data };
        let red = augmented.eliminate(n)?;
        let rank = red.rank();
        if rank < n { return Err(MatrixError::Singular { rank }); }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                out.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: out })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}
