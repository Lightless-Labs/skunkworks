fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {

    let operator = if content_start < end && b"+#./;?&".contains(&bytes[content_start]) {
        let op = bytes[content_start] as char;

            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        content_start += 1;
        op

            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        '\0'
    } else {
        '\0'
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut seen = Vec::<String>::new();
    let mut pos = content_start;

    loop {
        let comma = bytes[pos..end].iter().position(|&b| b == b',')
            .map(|i| pos + i).unwrap_or(end);
        if comma == pos {
            return Err(Error::EmptyVariable { pos });
        }

            return Err(Error::InvalidName { pos });
        }

            let valid = if i == pos { is_name_start(bytes[i]) } else { is_name_continue(bytes[i]) };
            if !valid {
                return Err(Error::InvalidName { pos: i });
            }
        }

        let name = template[pos..mod_at].to_owned();
        let modifier = if mod_at == comma {
            Modifier::None
        } else if bytes[mod_at] == b'*' && mod_at + 1 == comma {
            Modifier::Explode
        } else if bytes[mod_at] == b':' {
            let suffix = &bytes[mod_at + 1..comma];
            if suffix.is_empty()
                || suffix.len() > 4
                || !suffix.iter().all(u8::is_ascii_digit)
                || suffix[0] == b'0'

                return Err(Error::InvalidModifier { pos: mod_at });

                return Err(Error::InvalidModifier { pos: mod_at });

            return Err(Error::DuplicateVariable { pos: spec_start });

            break;

        pos = comma + 1;
        if pos == end {
            return Err(Error::EmptyVariable { pos: end });
        }
    }

    Ok(Expression { start, end, operator, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }
        let unreserved = b.is_ascii_alphanumeric() || b"-._~".contains(&b);
        let reserved_byte = b":/?#[]@!$&'()*+,;=".contains(&b);
        if unreserved || (reserved && reserved_byte) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
        i += 1;
    }
    out
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.operator;
    let reserved = op == '+' || op == '#';
    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let Some(value) = selected(&spec.name, vars) else { continue };
        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        let name = &spec.name;
        let named = matches!(op, ';' | '?' | '&');
        match value {
            Value::Scalar(s) => {
                let text = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                let encoded = encode(&text, reserved);
                if named {
                    if op == ';' && encoded.is_empty() {
                        fields.push(name.clone());
                    } else {
                        fields.push(format!("{name}={encoded}"));
                    }
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() { continue; }
                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if named { format!("{name}={encoded}") } else { encoded });
                    }
                } else {
                    let joined = items.iter().map(|s| encode(s, reserved)).collect::<Vec<_>>().join(",");
                    fields.push(if named { format!("{name}={joined}") } else { joined });
                }
            }
            Value::Map(entries) => {
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(k, _)| k == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if entries.is_empty() { continue; }
                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, val) in entries {
                        let k = encode(key, reserved);
                        let v = encode(val, reserved);
                        if op == ';' && v.is_empty() {
                            fields.push(k);
                        } else {
                            fields.push(format!("{k}={v}"));
                        }
                    }
                } else {
                    let mut parts = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        parts.push(encode(key, reserved));
                        parts.push(encode(val, reserved));
                    }
                    let joined = parts.join(",");
                    fields.push(if named { format!("{name}={joined}") } else { joined });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let separator = match op {
        '.' | '/' | ';' | '?' | '&' => op,
        _ => ',',
    };
    let prefix = match op {
        '#' | '.' | '/' | ';' | '?' | '&' => op.to_string(),
        _ => String::new(),
    };
    Ok(format!("{prefix}{}", fields.join(&separator.to_string())))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let Some(offset) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = i + 1 + offset;
                if let Some(inner) = bytes[i + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + inner });
                }
                expressions.push(parse_expression(template, start, end)?);
                i = end + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for expr in &expressions {
        out.push_str(&template[cursor..expr.start]);
        out.push_str(&expand_expression(expr, vars)?);
        cursor = expr.end + 1;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let mut operator = None;
    let mut offset = 0;

    if let Some(&b) = bytes.first() {
        if b"+#./;?&".contains(&b) {
            operator = Some(b);
            offset = 1;
        }
    }

    let body = &content[offset..];
    let body_pos = content_pos + offset;
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = 0;

    loop {
        let end = body.as_bytes()[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| start + n)
            .unwrap_or(body.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: body_pos + start,
            });
        }

        let raw = &body[start..end];
        let raw_bytes = raw.as_bytes();
        let modifier_at = raw_bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_at.unwrap_or(raw.len());
        let name = &raw[..name_end];

        if name.is_empty() {
            return Err(Error::InvalidName {
                pos: body_pos + start,
            });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName {
                pos: body_pos + start,
            });
        }
        for (i, &b) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(b) {
                return Err(Error::InvalidName {
                    pos: body_pos + start + i,
                });
            }
        }

        let modifier = if let Some(m) = modifier_at {
            let modifier_pos = body_pos + start + m;
            let suffix = &raw[m..];
            if suffix == "*" {
                Modifier::Explode
            } else if let Some(digits) = suffix.strip_prefix(':') {
                let d = digits.as_bytes();
                if !(1..=4).contains(&d.len())
                    || d[0] == b'0'
                    || !d.iter().all(u8::is_ascii_digit)
                {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                let n = digits.parse::<usize>().unwrap_or(0);
                if !(1..=9999).contains(&n) {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Modifier::Prefix(n)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        } else {
            Modifier::None
        };

        if seen.iter().any(|s| s == name) {
            return Err(Error::DuplicateVariable {
                pos: body_pos + start,
            });
        }
        seen.push(name.to_owned());
        specs.push(Specifier {
            name: name.to_owned(),
            start: body_pos + start,
            modifier,
        });

        if end == body.len() {
            break;
        }
        start = end + 1;
        if start == body.len() {
            return Err(Error::EmptyVariable {
                pos: body_pos + start,
            });
        }
    }

    Ok(Expression { operator, specs })
}

        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || b"-._~".contains(&b);
        let reserved_byte = b":/?#[]@!$&'()*+,;=".contains(&b);
        if unreserved || (reserved && reserved_byte) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(b"0123456789ABCDEF"[(b >> 4) as usize] as char);
            out.push(b"0123456789ABCDEF"[(b & 15) as usize] as char);
        }
        i += 1;
    }
    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(n, _)| *n == name).map(|(_, v)| v)
}

fn named_field(name: &str, value: String, semicolon: bool) -> String {
    if semicolon && value.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + value.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&value);
        field
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.operator;
    let reserved = matches!(op, Some(b'+') | Some(b'#'));
    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        let scalar_prefix = match spec.modifier {
            Modifier::Prefix(n) => Some(n),
            _ => None,
        };
        if scalar_prefix.is_some() && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(s) => {
                let selected = if let Some(n) = scalar_prefix {
                    s.chars().take(n).collect::<String>()
                } else {
                    s.clone()
                };
                fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                    named_field(&spec.name, encode(&selected, reserved), op == Some(b';'))
                } else {
                    encode(&selected, reserved)
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                            named_field(&spec.name, encoded, op == Some(b';'))
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                        named_field(&spec.name, joined, op == Some(b';'))
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }

                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let field = if op == Some(b';') && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                        named_field(&spec.name, joined, op == Some(b';'))
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match op {
        None | Some(b'+') => (",", ""),
        Some(b'#') => (",", "#"),
        Some(b'.') => (".", "."),
        Some(b'/') => ("/", "/"),
        Some(b';') => (";", ";"),
        Some(b'?') => ("&", "?"),
        Some(b'&') => ("&", "&"),
        _ => (",", ""),
    };

    let mut result = String::from(prefix);
    result.push_str(&fields.join(separator));
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<(usize, usize, Expression)>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                if let Some(relative_nested) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_nested,
                    });
                }
                let content = &template[open + 1..close];
                let expr = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expr));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (start, end, expr) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expr, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}

fn operator_from_byte(b: u8) -> Option<Operator> {
    match b {
        b'+' => Some(Operator::Plus),
        b'#' => Some(Operator::Hash),
        b'.' => Some(Operator::Dot),
        b'/' => Some(Operator::Slash),
        b';' => Some(Operator::Semicolon),
        b'?' => Some(Operator::Question),
        b'&' => Some(Operator::Ampersand),
        _ => None,
    }

fn parse_expression(
    template: &str,
    open_pos: usize,
    close_pos: usize,
) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_pos = open_pos + 1;
    let operator = if content_pos < close_pos {
        if let Some(operator) = operator_from_byte(bytes[content_pos]) {
            content_pos += 1;
            operator
        } else {
            Operator::Simple
        }
    } else {
        Operator::Simple
    };

    if content_pos == close_pos {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = content_pos;

    loop {
        let comma_pos = bytes[spec_start..close_pos]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close_pos);

        if spec_start == comma_pos {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let modifier_pos = bytes[spec_start..comma_pos]
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .map(|offset| spec_start + offset)
            .unwrap_or(comma_pos);

        if modifier_pos == spec_start {
            return Err(Error::InvalidName { pos: spec_start });
        }

        if !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for pos in spec_start + 1..modifier_pos {
            if !is_name_continue(bytes[pos]) {
                return Err(Error::InvalidName { pos });
            }
        }

        let name = template[spec_start..modifier_pos].to_owned();
        let modifier = if modifier_pos == comma_pos {
            Modifier::None
        } else if bytes[modifier_pos] == b'*' {
            if modifier_pos + 1 != comma_pos {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Explode
        } else {
            let digits = &bytes[modifier_pos + 1..comma_pos];
            if digits.is_empty()
                || digits.len() > 4
                || digits[0] == b'0'
                || !digits.iter().all(u8::is_ascii_digit)
            {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }

            let value = digits
                .iter()
                .fold(0usize, |acc, b| acc * 10 + (b - b'0') as usize);
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Prefix(value)
        };

        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            pos: spec_start,
            modifier,
        });

        if comma_pos == close_pos {
            break;
        }
        spec_start = comma_pos + 1;
        if spec_start == close_pos {
            return Err(Error::EmptyVariable { pos: close_pos });
        }
    }

    Ok(Expression {
        start: open_pos,
        end: close_pos + 1,
        operator,
        specs,
    })
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }

        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(input: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn expand_expression(
    expression: &Expression,
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Operator::Plus | Operator::Hash);
    let named = matches!(
        operator,
        Operator::Semicolon | Operator::Question | Operator::Ampersand
    );
    let semicolon = operator == Operator::Semicolon;
    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(scalar) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(count) => scalar.chars().take(count).collect::<String>(),
                    Modifier::None | Modifier::Explode => scalar.clone(),
                };
                let encoded = encode(&selected, allow_reserved);
                fields.push(if named {
                    named_field(&spec.name, encoded, semicolon)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }

                if spec.modifier == Modifier::Explode {
                    for item in items {
                        let encoded = encode(item, allow_reserved);
                        fields.push(if named {
                            named_field(&spec.name, encoded, semicolon)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = join_components(
                        items.iter().map(|item| encode(item, allow_reserved)),
                    );
                    fields.push(if named {
                        named_field(&spec.name, joined, semicolon)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                if spec.modifier == Modifier::Explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(value, allow_reserved);
                        let field = if semicolon && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let joined = join_components(entries.iter().flat_map(|(key, value)| {
                        [
                            encode(key, allow_reserved),
                            encode(value, allow_reserved),
                        ]
                    }));
                    fields.push(if named {
                        named_field(&spec.name, joined, semicolon)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Hash => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<Expression>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open_pos = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open_pos });
                };
                let close_pos = i + 1 + relative_close;

                if let Some(relative_open) =
                    bytes[i + 1..close_pos].iter().position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_open,
                    });
                }

                expressions.push(parse_expression(template, open_pos, close_pos)?);
                i = close_pos + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for expression in &expressions {
        output.push_str(&template[cursor..expression.start]);
        output.push_str(&expand_expression(expression, vars)?);
        cursor = expression.end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = if let Some(&b) = bytes.first() {
        let op = match b {
            b'+' => Some(Operator::Plus),
            b'#' => Some(Operator::Hash),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Question),
            b'&' => Some(Operator::Ampersand),
            _ => None,
        };
        if let Some(op) = op {
            (op, 1)
        } else {
            (Operator::Simple, 0)
        }
    } else {
        (Operator::Simple, 0)
    };

    if content.is_empty() || offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut start = offset;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| start + n)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: if start == bytes.len() { close_pos } else { content_pos + start },
            });
        }

        let spec = &content[start..end];
        let spec_pos = content_pos + start;
        let spec_bytes = spec.as_bytes();
        let name_end = spec_bytes
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(spec_bytes.len());

        if name_end == 0 {
            return Err(Error::InvalidName { pos: spec_pos });
        }

        let name_bytes = &spec_bytes[..name_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_pos });
        }
        if let Some(i) = name_bytes.iter().position(|&b| !is_name_continue(b)) {
            return Err(Error::InvalidName { pos: spec_pos + i });
        }

        let modifier = if name_end == spec_bytes.len() {
            Modifier::None
        } else {
            let modifier_pos = spec_pos + name_end;
            let suffix = &spec[name_end..];
            if suffix == "*" {
                Modifier::Explode
            } else if let Some(digits) = suffix.strip_prefix(':') {
                let valid = !digits.is_empty()
                    && digits.len() <= 4
                    && digits.as_bytes()[0].is_ascii_digit()
                    && digits.as_bytes()[0] != b'0'
                    && digits.bytes().all(|b| b.is_ascii_digit());
                if !valid {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                let value = digits.bytes().fold(0usize, |n, b| n * 10 + (b - b'0') as usize);
                if !(1..=9999).contains(&value) {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Modifier::Prefix(value)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        let name = spec[..name_end].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }
        specs.push(Specifier { name, modifier, pos: spec_pos });

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::EmptyVariable { pos: close_pos });
        }
    }

    Ok(Expression { operator, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() && hex_value(bytes[i + 1]).is_some() && hex_value(bytes[i + 2]).is_some() {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[hex_value(bytes[i + 1]).unwrap() as usize] as char);
            out.push(HEX[hex_value(bytes[i + 2]).unwrap() as usize] as char);
            i += 3;
        } else if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
            i += 1;
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
            i += 1;
        }
    }
    out
}

fn truncate_chars(s: &str, n: usize) -> String {
    s.chars().take(n).collect()
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn named_field(name: &str, value: String, semicolon: bool) -> String {
    if semicolon && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(expression.operator, Operator::Plus | Operator::Hash);
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        if let Modifier::Prefix(n) = spec.modifier {
            match value {
                Value::Scalar(_) => {}
                Value::List(_) | Value::Map(_) => return Err(Error::TypeMismatch { pos: spec.pos }),
            }
            let Value::Scalar(s) = value else { unreachable!() };
            let truncated = truncate_chars(s, n);
            fields.push(match expression.operator {
                Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                    named_field(&spec.name, encode(&truncated, reserved), expression.operator == Operator::Semicolon)
                }
                _ => encode(&truncated, reserved),
            });
            continue;
        }

        match value {
            Value::Scalar(s) => {
                let encoded = encode(s, reserved);
                fields.push(match expression.operator {
                    Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                        named_field(&spec.name, encoded, expression.operator == Operator::Semicolon)
                    }
                    _ => encoded,
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if spec.modifier == Modifier::Explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(match expression.operator {
                            Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                                named_field(&spec.name, encoded, expression.operator == Operator::Semicolon)
                            }
                            _ => encoded,
                        });
                    }
                } else {
                    let joined = items.iter().map(|item| encode(item, reserved)).collect::<Vec<_>>().join(",");
                    fields.push(match expression.operator {
                        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                            named_field(&spec.name, joined, expression.operator == Operator::Semicolon)
                        }
                        _ => joined,
                    });
                }
            }
            Value::Map(entries) => {
                let mut seen = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !seen.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }
                if entries.is_empty() {
                    continue;
                }

                if spec.modifier == Modifier::Explode {
                    for (key, value) in entries {
                        let key = encode(key, reserved);
                        let value = encode(value, reserved);
                        fields.push(if expression.operator == Operator::Semicolon && value.is_empty() {
                            key
                        } else {
                            format!("{key}={value}")
                        });
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(match expression.operator {
                        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                            named_field(&spec.name, joined, expression.operator == Operator::Semicolon)
                        }
                        _ => joined,
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Hash => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut parsed = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                if let Some(relative_open) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + relative_open });
                }
                let content = &template[open + 1..close];
                parsed.push((open, close + 1, parse_expression(content, open + 1, close)?));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for (start, end, expression) in parsed {
        out.push_str(&template[cursor..start]);
        out.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
