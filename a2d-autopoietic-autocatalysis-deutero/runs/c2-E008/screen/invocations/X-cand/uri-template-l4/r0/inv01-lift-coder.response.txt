#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Clone)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Expression {
    operator: Option<u8>,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let mut operator = None;
    let mut offset = 0;

    if let Some(&b) = bytes.first() {
        if b"+#./;?&".contains(&b) {
            operator = Some(b);
            offset = 1;
        }
    }

    let body = &content[offset..];
    let body_pos = content_pos + offset;
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = 0;

    loop {
        let end = body.as_bytes()[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| start + n)
            .unwrap_or(body.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: body_pos + start,
            });
        }

        let raw = &body[start..end];
        let raw_bytes = raw.as_bytes();
        let modifier_at = raw_bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_at.unwrap_or(raw.len());
        let name = &raw[..name_end];

        if name.is_empty() {
            return Err(Error::InvalidName {
                pos: body_pos + start,
            });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName {
                pos: body_pos + start,
            });
        }
        for (i, &b) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(b) {
                return Err(Error::InvalidName {
                    pos: body_pos + start + i,
                });
            }
        }

        let modifier = if let Some(m) = modifier_at {
            let modifier_pos = body_pos + start + m;
            let suffix = &raw[m..];
            if suffix == "*" {
                Modifier::Explode
            } else if let Some(digits) = suffix.strip_prefix(':') {
                let d = digits.as_bytes();
                if !(1..=4).contains(&d.len())
                    || d[0] == b'0'
                    || !d.iter().all(u8::is_ascii_digit)
                {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                let n = digits.parse::<usize>().unwrap_or(0);
                if !(1..=9999).contains(&n) {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Modifier::Prefix(n)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        } else {
            Modifier::None
        };

        if seen.iter().any(|s| s == name) {
            return Err(Error::DuplicateVariable {
                pos: body_pos + start,
            });
        }
        seen.push(name.to_owned());
        specs.push(Specifier {
            name: name.to_owned(),
            start: body_pos + start,
            modifier,
        });

        if end == body.len() {
            break;
        }
        start = end + 1;
        if start == body.len() {
            return Err(Error::EmptyVariable {
                pos: body_pos + start,
            });
        }
    }

    Ok(Expression { operator, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || b"-._~".contains(&b);
        let reserved_byte = b":/?#[]@!$&'()*+,;=".contains(&b);
        if unreserved || (reserved && reserved_byte) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(b"0123456789ABCDEF"[(b >> 4) as usize] as char);
            out.push(b"0123456789ABCDEF"[(b & 15) as usize] as char);
        }
        i += 1;
    }
    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(n, _)| *n == name).map(|(_, v)| v)
}

fn named_field(name: &str, value: String, semicolon: bool) -> String {
    if semicolon && value.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + value.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&value);
        field
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.operator;
    let reserved = matches!(op, Some(b'+') | Some(b'#'));
    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        let scalar_prefix = match spec.modifier {
            Modifier::Prefix(n) => Some(n),
            _ => None,
        };
        if scalar_prefix.is_some() && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(s) => {
                let selected = if let Some(n) = scalar_prefix {
                    s.chars().take(n).collect::<String>()
                } else {
                    s.clone()
                };
                fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                    named_field(&spec.name, encode(&selected, reserved), op == Some(b';'))
                } else {
                    encode(&selected, reserved)
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                            named_field(&spec.name, encoded, op == Some(b';'))
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                        named_field(&spec.name, joined, op == Some(b';'))
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }

                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let field = if op == Some(b';') && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(op, Some(b';') | Some(b'?') | Some(b'&')) {
                        named_field(&spec.name, joined, op == Some(b';'))
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match op {
        None | Some(b'+') => (",", ""),
        Some(b'#') => (",", "#"),
        Some(b'.') => (".", "."),
        Some(b'/') => ("/", "/"),
        Some(b';') => (";", ";"),
        Some(b'?') => ("&", "?"),
        Some(b'&') => ("&", "&"),
        _ => (",", ""),
    };

    let mut result = String::from(prefix);
    result.push_str(&fields.join(separator));
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<(usize, usize, Expression)>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                if let Some(relative_nested) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_nested,
                    });
                }
                let content = &template[open + 1..close];
                let expr = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expr));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (start, end, expr) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expr, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
