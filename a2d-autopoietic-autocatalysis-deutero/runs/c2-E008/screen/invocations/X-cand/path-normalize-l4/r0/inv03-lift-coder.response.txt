use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|part| !part.is_empty())
            .is_some_and(|part| part == "." || part == "..")
}

fn components_with_offsets(path: &str, offset_base: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        out.push((path[start..i].to_owned(), offset_base + start));
    }

    out
}

fn components(path: &str) -> Vec<String> {
    components_with_offsets(path, 0)
        .into_iter()
        .map(|(part, _)| part)
        .collect()
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let joined = parts.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce(
    stack: &mut Vec<String>,
    items: &[(String, usize)],
    absolute: bool,
) -> Result<(), PathError> {
    for (component, at) in items {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|part| part != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: *at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(component.clone()),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut parts = Vec::new();
    reduce(&mut parts, &components_with_offsets(path, 0), absolute)?;
    Ok(render(&parts, absolute, intent))
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }

    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized)),
        Err(_) => Err(error),
    }
}

fn request_start<'a>(
    path: &'a str,
    cwd: &[String],
    home: Option<&[String]>,
) -> Result<(Vec<String>, usize, bool), PathError> {
    if path.starts_with('/') {
        return Ok((Vec::new(), 0, false));
    }

    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }

        let home = home.ok_or(PathError::MissingHome)?;
        let skip = if bytes.len() == 1 { 1 } else { 2 };
        return Ok((home.to_vec(), skip, true));
    }

    Ok((cwd.to_vec(), 0, false))
}

fn resolve_core(path: &str, cwd: &[String], home: Option<&[String]>) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (mut stack, skip, _) = request_start(path, cwd, home)?;

    if path.starts_with('/') {
        stack.clear();
    }

    reduce(
        &mut stack,
        &components_with_offsets(&path[skip..], skip),
        true,
    )?;
    Ok(render(&stack, true, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };
    resolve_core(path, &cwd_parts, home_parts.as_deref())
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut indices = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && link.path != "/"
            && !link.path.ends_with('/')
            && normalize(&link.path).is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

    Ok(indices)
}

#[derive(Clone)]
struct Frame {
    items: Vec<(String, usize)>,
    next: usize,
    completes_link: Option<usize>,
}

fn path_key(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn resolve_link_items(
    stack: &mut Vec<String>,
    root: &[(String, usize)],
    links: &[Link],
    indices: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<(), PathError> {
    let mut frames = vec![Frame {
        items: root.to_vec(),
        next: 0,
        completes_link: None,
    }];
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while !frames.is_empty() {
        let frame_index = frames.len() - 1;

        if frames[frame_index].next == frames[frame_index].items.len() {
            let frame = frames.pop().unwrap();
            if let Some(index) = frame.completes_link {
                debug_assert_eq!(active.pop(), Some(index));
            }
            continue;
        }

        let (component, at) = {
            let frame = &mut frames[frame_index];
            let item = frame.items[frame.next].clone();
            frame.next += 1;
            item
        };

        match component.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(component);
                let key = path_key(stack);
                let Some(&index) = indices.get(&key) else {
                    continue;
                };

                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if expansions >= max_expansions {
                    return Err(PathError::Limit { index });
                }
                expansions += 1;

                // The target starts at the link's parent unless it is absolute.
                stack.pop();
                let target = &links[index].target;
                if target.starts_with('/') {
                    stack.clear();
                }

                active.push(index);
                frames.push(Frame {
                    items: components_with_offsets(target, 0),
                    next: 0,
                    completes_link: Some(index),
                });
            }
        }
    }

    Ok(())
}

fn resolve_links_core(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
    links: &[Link],
    indices: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (mut stack, skip, _) = request_start(path, cwd, home)?;
    if path.starts_with('/') {
        stack.clear();
    }

    let items = components_with_offsets(&path[skip..], skip);
    resolve_link_items(&mut stack, &items, links, indices, max_expansions)?;
    Ok(render(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };
    let indices = validate_links(links)?;
    resolve_links_core(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        &indices,
        max_expansions,
    )
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();

    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

fn candidate_with_suffix(mut route: String, suffix: &[String]) -> String {
    if suffix.is_empty() {
        route
    } else if route == "." {
        suffix.join("/")
    } else {
        route.push('/');
        route.push_str(&suffix.join("/"));
        route
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = components(&resolved_base);
    let target_parts = components(&resolved_target);
    let target_is_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) else {
            continue;
        };
        let anchor = components(&probe);
        if anchor.len() > target_parts.len()
            || anchor
                .iter()
                .zip(&target_parts)
                .any(|(a, b)| a != b)
        {
            continue;
        }

        let key_parts = components(&link.path);
        let candidate = candidate_with_suffix(
            route(&base_parts, &key_parts),
            &target_parts[anchor.len()..],
        );
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_is_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) else {
            continue;
        };
        if components(&resolved) == target_parts {
            valid.push(candidate);
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });

    valid.into_iter().next().unwrap_or_else(|| ".".to_owned())
}
