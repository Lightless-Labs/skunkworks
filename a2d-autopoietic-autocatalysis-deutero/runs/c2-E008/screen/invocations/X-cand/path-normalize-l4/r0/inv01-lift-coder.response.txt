use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components_with_offsets(path: &str) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'/' {
            i += 1;
            continue;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_owned(), start));
    }
    result
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    components_with_offsets(path)
        .last()
        .map(|(part, _)| part == "." || part == "..")
        .unwrap_or(false)
}

fn join_absolute(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn join_normalized(stack: &[String], absolute: bool) -> String {
    if absolute {
        if stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    }
}

fn reduce_unlinked(
    path: &str,
    absolute: bool,
) -> Result<Vec<String>, PathError> {
    let mut stack: Vec<String> = Vec::new();

    for (part, at) in components_with_offsets(path) {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|p| p != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }

    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let stack = reduce_unlinked(path, absolute)?;
    let mut result = join_normalized(&stack, absolute);

    if directory_intent(path) && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

fn validate_context(path: &str) -> Result<String, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    normalize(path).map_err(|_| ())
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut indices = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .map(|normalized| normalized == link.path)
                .unwrap_or(false);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

    Ok(indices)
}

#[derive(Clone)]
struct Frame {
    parts: Vec<(String, usize)>,
    next: usize,
    completes_link: Option<usize>,
}

fn push_component(
    part: String,
    at: usize,
    stack: &mut Vec<String>,
) -> Result<(), PathError> {
    match part.as_str() {
        "." => {}
        ".." => {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        }
        _ => stack.push(part),
    }
    Ok(())
}

fn run_linked(
    initial_stack: Vec<String>,
    request: &str,
    request_start: usize,
    links: &[Link],
    link_indices: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial_stack;
    let mut frames = vec![Frame {
        parts: components_with_offsets(&request[request_start..])
            .into_iter()
            .map(|(part, at)| (part, at + request_start))
            .collect(),
        next: 0,
        completes_link: None,
    }];
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while !frames.is_empty() {
        let frame_index = frames.len() - 1;

        if frames[frame_index].next >= frames[frame_index].parts.len() {
            let frame = frames.pop().unwrap();
            if let Some(link_index) = frame.completes_link {
                if active.last() == Some(&link_index) {
                    active.pop();
                } else if let Some(position) = active.iter().rposition(|&i| i == link_index) {
                    active.remove(position);
                }
            }
            continue;
        }

        let (part, at) = {
            let frame = &mut frames[frame_index];
            let item = frame.parts[frame.next].clone();
            frame.next += 1;
            item
        };

        match part.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(part);
                let current = join_absolute(&stack);
                if let Some(&link_index) = link_indices.get(&current) {
                    if active.contains(&link_index) {
                        return Err(PathError::Loop { index: link_index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index: link_index });
                    }
                    expansions += 1;
                    active.push(link_index);

                    // The link name is replaced by its target. Relative targets
                    // therefore begin in the link's parent directory.
                    stack.pop();
                    let target = &links[link_index].target;
                    if target.starts_with('/') {
                        stack.clear();
                    }

                    frames.push(Frame {
                        parts: components_with_offsets(target),
                        next: 0,
                        completes_link: Some(link_index),
                    });
                }
            }
        }
    }

    Ok(stack)
}

fn expand_request_tilde<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(Option<&'a str>, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((None, 0));
    }

    let bytes = path.as_bytes();
    if bytes.len() == 1 {
        return if home.is_some() {
            Ok((home, 1))
        } else {
            Err(PathError::MissingHome)
        };
    }

    if bytes[1] == b'/' {
        return if home.is_some() {
            Ok((home, 2))
        } else {
            Err(PathError::MissingHome)
        };
    }

    Err(PathError::UnsupportedTilde { at: 1 })
}

fn resolve_impl(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let normalized_cwd = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let normalized_home = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    let link_indices = validate_links(links)?;
    validate_raw(path)?;

    let (expanded_home, request_start) = expand_request_tilde(path, normalized_home.as_deref())?;
    let absolute_request = path.as_bytes().first() == Some(&b'/');
    let initial_stack = if absolute_request {
        Vec::new()
    } else if let Some(home_path) = expanded_home {
        components_with_offsets(home_path)
            .into_iter()
            .map(|(part, _)| part)
            .collect()
    } else {
        components_with_offsets(&normalized_cwd)
            .into_iter()
            .map(|(part, _)| part)
            .collect()
    };

    // A bare tilde has no directory intent; "~/..." retains the original path's
    // intent, while the consumed prefix contributes no components.
    let stack = run_linked(
        initial_stack,
        path,
        request_start,
        links,
        &link_indices,
        max_expansions,
    )?;

    let mut result = join_absolute(&stack);
    let intent = if path == "~" {
        false
    } else {
        directory_intent(path)
    };
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    resolve_impl(path, cwd, home, &[], 0)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_impl(path, cwd, home, links, max_expansions)
}

fn absolute_components(path: &str) -> Vec<String> {
    components_with_offsets(path)
        .into_iter()
        .map(|(part, _)| part)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();

    let mut parts = Vec::new();
    for _ in common..base.len() {
        parts.push("..".to_owned());
    }
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;

    let base_components = absolute_components(&resolved_base);
    let target_components = absolute_components(&resolved_target);
    let target_is_directory = resolved_target.ends_with('/');
    let link_indices = validate_links(links)?;

    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };

        let probed_components = absolute_components(&probe);
        if probed_components.len() <= target_components.len()
            && probed_components
                .iter()
                .zip(target_components.iter())
                .all(|(a, b)| a == b)
        {
            let key_components = absolute_components(&link.path);
            let mut candidate = route(&base_components, &key_components);
            let suffix = &target_components[probed_components.len()..];

            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }

        if target_is_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if absolute_components(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    let _ = link_indices;
    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
