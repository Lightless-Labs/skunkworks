#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> impl Iterator<Item = (usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, component| {
            let start = *offset;
            *offset += component.len() + 1;
            Some((start, component))
        })
        .filter(|(_, component)| !component.is_empty())
}

fn reduce(
    initial: Vec<String>,
    path: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    for (at, component) in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: offset_base + at,
                    });
                } else {
                    stack.push("..".to_owned());
                }
            }
            name => stack.push(name.to_owned()),
        }
    }
    Ok(stack)
}

fn render(parts: &[String], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if absolute && parts.is_empty() {
        return "/".to_owned();
    }
    if directory && result != "." && result != "/" {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let directory = directory_intent(path);
    let parts = reduce(Vec::new(), path, 0, absolute)?;
    Ok(render(&parts, absolute, directory))
}

fn validated_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return Err(());
    }
    let parts = reduce(Vec::new(), path, 0, true).map_err(|_| ())?;
    Ok(parts)
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let expanded = if tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        Some(home_parts.ok_or(PathError::MissingHome)?)
    } else {
        None
    };

    let directory = directory_intent(path);
    let (initial, remainder, offset_base) = if let Some(parts) = expanded {
        let skip = if bytes.len() > 1 { 2 } else { 1 };
        (parts, &path[skip..], skip)
    } else if bytes[0] == b'/' {
        (Vec::new(), path, 0)
    } else {
        (cwd_parts, path, 0)
    };

    let parts = reduce(initial, remainder, offset_base, true)?;
    Ok(render(&parts, true, directory))
}
