#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }

    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn reduce_absolute<'a>(
    stack: &mut Vec<&'a str>,
    parts: &[(&'a str, usize)],
) -> Result<(), PathError> {
    for &(part, at) in parts {
        match part {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack: Vec<&str> = Vec::new();

    for (at, part) in path.split('/').scan(0usize, |offset, part| {
        let current = *offset;
        *offset += part.len() + 1;
        Some((current, part))
    }) {
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.last().is_some_and(|previous| *previous != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push(part);
            }
        } else {
            stack.push(part);
        }
    }

    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let normalized_cwd = if cwd.starts_with('/') {
        normalize(cwd).map_err(|_| PathError::InvalidCwd)?
    } else {
        return Err(PathError::InvalidCwd);
    };

    let normalized_home = match home {
        Some(value) => {
            if !value.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(normalize(value).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate_raw(path)?;

    let intent = directory_intent(path);
    let first_tilde = path.as_bytes()[0] == b'~';
    let (base, remainder, remainder_offset) = if first_tilde {
        if path.len() > 1 && !path[1..].starts_with('/') {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = normalized_home.as_deref().ok_or(PathError::MissingHome)?;
        let remainder_offset = if path.len() > 1 { 2 } else { 1 };
        (home, &path[remainder_offset..], remainder_offset)
    } else if path.starts_with('/') {
        ("/", path, 0)
    } else {
        (normalized_cwd.as_str(), path, 0)
    };

    let mut stack: Vec<&str> = if base == "/" {
        Vec::new()
    } else {
        base.trim_start_matches('/')
            .split('/')
            .filter(|part| !part.is_empty())
            .collect()
    };

    let parts: Vec<(&str, usize)> = remainder
        .split('/')
        .scan(remainder_offset, |offset, part| {
            let current = *offset;
            *offset += part.len() + 1;
            Some((part, current))
        })
        .filter(|(part, _)| !part.is_empty())
        .collect();

    reduce_absolute(&mut stack, &parts)?;

    let mut result = format!("/{}", stack.join("/"));
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}
