use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let mut n = num;
        let mut d = den;
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        Self::from_i128(a + b, self.den as i128 * rhs.den as i128)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        Self::from_i128(a - b, self.den as i128 * rhs.den as i128)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        Self::from_i128(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| Err(RationalError::Parse { at, kind });
        if bytes.is_empty() { return fail(0, ParseKind::Empty); }

        let mut i = 0;
        if bytes[0] == b'-' { i = 1; }
        let ns = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
        if i == ns {
            if i == bytes.len() { return fail(i, ParseKind::InvalidByte); }
            return fail(i, ParseKind::InvalidByte);
        }
        let ne = i;
        let mut ds = None;
        let mut de = None;
        if i < bytes.len() {
            if bytes[i] != b'/' { return fail(i, ParseKind::InvalidByte); }
            i += 1;
            let start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
            if i == start {
                if i == bytes.len() { return fail(i, ParseKind::InvalidByte); }
                return fail(i, ParseKind::InvalidByte);
            }
            ds = Some(start);
            de = Some(i);
            if i < bytes.len() { return fail(i, ParseKind::InvalidByte); }
        }

        let negative = bytes[0] == b'-';
        let nlen = ne - ns;
        if nlen >= 2 && bytes[ns] == b'0' { return fail(ns + 1, ParseKind::NonCanonical); }
        if negative && nlen == 1 && bytes[ns] == b'0' { return fail(0, ParseKind::NonCanonical); }
        if let (Some(s), Some(e)) = (ds, de) {
            if e - s >= 2 && bytes[s] == b'0' { return fail(s + 1, ParseKind::NonCanonical); }
        }

        let mut nmag: u128 = 0;
        for &b in &bytes[ns..ne] {
            nmag = nmag.saturating_mul(10).saturating_add((b - b'0') as u128);
        }
        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag > nlimit { return fail(0, ParseKind::OutOfRange); }

        let mut dmag: u128 = 1;
        if let (Some(s), Some(e)) = (ds, de) {
            dmag = 0;
            for &b in &bytes[s..e] {
                dmag = dmag.saturating_mul(10).saturating_add((b - b'0') as u128);
            }
            if dmag > i64::MAX as u128 { return fail(s, ParseKind::OutOfRange); }
            if dmag == 0 { return fail(s, ParseKind::ZeroDenominator); }
            if dmag == 1 { return fail(s - 1, ParseKind::NonCanonical); }
            if nmag == 0 { return fail(s - 1, ParseKind::NonCanonical); }
            if gcd_u128(nmag, dmag) > 1 { return fail(s - 1, ParseKind::NonCanonical); }
        }

        let n = if negative {
            if nmag == 1u128 << 63 { i64::MIN } else { -(nmag as i64) }
        } else { nmag as i64 };
        Ok(Self { num: n, den: dmag as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) } else { write!(f, "{}/{}", self.num, self.den) }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected { return Err(MatrixError::DataLength { expected, actual: data.len() }); }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols { return Err(MatrixError::Index { row, col }); }
        Ok(self.data[row * self.cols + col])
    }
    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows { data.push(self.data[r * self.cols + c]); }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch { left: (self.rows, self.cols), right: (rhs.rows, rhs.cols) });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k].mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&a).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(mut data: Vec<Rational>, rows: usize, cols: usize, candidates: usize)
        -> Result<Reduction, MatrixError>
    {
        let mut p = 0;
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        for c in 0..candidates {
            if p == rows { break; }
            let mut best: Option<usize> = None;
            for r in p..rows {
                let x = data[r * cols + c];
                if x.num == 0 { continue; }
                let key = (x.den as u64, x.num.unsigned_abs(), r);
                if best.map_or(true, |b| {
                    let y = data[b * cols + c];
                    key < (y.den as u64, y.num.unsigned_abs(), b)
                }) { best = Some(r); }
            }
            let q = match best { Some(q) => q, None => continue };
            if q != p {
                for j in 0..cols { data.swap(p * cols + j, q * cols + j); }
                swaps.push((p, q));
            }
            let pivot = data[p * cols + c];
            for j in 0..cols {
                data[p * cols + j] = data[p * cols + j].div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: j, step: p, source })?;
            }
            for r in 0..rows {
                if r == p { continue; }
                let factor = data[r * cols + c];
                if factor.num == 0 { continue; }
                for j in 0..cols {
                    let product = factor.mul(&data[p * cols + j])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: j, step: p, source })?;
                    data[r * cols + j] = data[r * cols + j].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: j, step: p, source })?;
                }
            }
            pivots.push(c);
            p += 1;
        }
        Ok(Reduction { matrix: Matrix { rows, cols, data }, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        Self::eliminate(self.data.clone(), self.rows, self.cols, self.cols)
    }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols { return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols }); }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let size = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(size);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n { data.push(if r == c { one } else { zero }); }
        }
        let red = Self::eliminate(data, n, width, n)?;
        if red.pivots.len() < n { return Err(MatrixError::Singular { rank: red.pivots.len() }); }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n { out.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]); }
        Ok(Self { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let width = self.cols.checked_add(rhs.cols).ok_or(SolveError::AugmentedShapeOverflow)?;
        let size = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(size);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * self.cols..(r + 1) * self.cols]);
            data.extend_from_slice(&rhs.data[r * rhs.cols..(r + 1) * rhs.cols]);
        }
        let red = Self::eliminate(data, self.rows, width, self.cols).map_err(SolveError::Matrix)?;
        let rank = red.pivots.len();
        for r in 0..self.rows {
            if red.matrix.data[r * width..r * width + self.cols].iter().all(|x| x.num == 0) {
                if let Some(j) = (0..rhs.cols).find(|&j| red.matrix.data[r * width + self.cols + j].num != 0) {
                    return Ok(Solution::None {
                        row: r, rhs_col: j,
                        witness: red.matrix.data[r * width..(r + 1) * width].to_vec(),
                    });
                }
            }
        }
        let n = self.cols;
        let k = rhs.cols;
        let plen = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; plen];
        for (i, &pc) in red.pivots.iter().enumerate() {
            for j in 0..k {
                particular[pc * k + j] = red.matrix.data[i * width + n + j];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular };
        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: red.pivots });
        }
        let free: Vec<usize> = (0..n).filter(|c| !red.pivots.contains(c)).collect();
        let blen = n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; blen];
        let one = Rational { num: 1, den: 1 };
        for (j, &fc) in free.iter().enumerate() {
            basis[fc * free.len() + j] = one;
            for (i, &pc) in red.pivots.iter().enumerate() {
                let x = red.matrix.data[i * width + fc].neg().map_err(|source|
                    SolveError::Matrix(MatrixError::Arithmetic { row: pc, col: j, step: rank, source }))?;
                basis[pc * free.len() + j] = x;
            }
        }
        Ok(Solution::Infinite {
            particular, free, basis: Matrix { rows: n, cols: blen.checked_div(n.max(1)).unwrap_or(0), data: basis },
            pivots: red.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}
impl Reduction { pub fn rank(&self) -> usize { self.pivots.len() } }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}
