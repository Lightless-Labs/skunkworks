use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn abs_i64_u128(n: i64) -> u128 {
    if n < 0 {
        (-(n as i128)) as u128
    } else {
        n as u128
    }
}

fn rational_from_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = self.num as i128 * a + rhs.num as i128 * b;
        let d = self.den as i128 * a;
        rational_from_parts(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = self.num as i128 * a - rhs.num as i128 * b;
        let d = self.den as i128 * a;
        rational_from_parts(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(abs_i64_u128(self.num), rhs.den as u128) as i128;
        let g2 = gcd_u128(abs_i64_u128(rhs.num), self.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.num as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.den as i128 / g1);
        rational_from_parts(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(abs_i64_u128(self.num), abs_i64_u128(rhs.num)) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.den as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.num as i128 / g1);
        rational_from_parts(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        rational_from_parts(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(err(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if b.is_ascii_digit() {
                continue;
            } else if b == b'-' && i == 0 {
                continue;
            } else {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let s = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');
        if s == num_start {
            return Err(err(s, ParseKind::InvalidByte));
        }
        if let Some(pos) = slash {
            if pos + 1 == bytes.len() {
                return Err(err(bytes.len(), ParseKind::InvalidByte));
            }
        }

        let numerator_digits = &bytes[num_start..s];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && numerator_digits == b"0" {
            return Err(err(0, ParseKind::NonCanonical));
        }

        let denominator_range = slash.map(|p| p + 1..bytes.len());
        if let Some(ref range) = denominator_range {
            let digits = &bytes[range.clone()];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(err(range.start + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut num_mag = 0u128;
        for &b in numerator_digits {
            num_mag = num_mag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
            if num_mag > num_limit {
                return Err(err(0, ParseKind::OutOfRange));
            }
        }

        let mut den_mag = 1u128;
        if let Some(ref range) = denominator_range {
            for &b in &bytes[range.clone()] {
                den_mag = den_mag
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
                if den_mag > i64::MAX as u128 {
                    return Err(err(range.start, ParseKind::OutOfRange));
                }
            }
        }

        if let Some(ref range) = denominator_range {
            if den_mag == 0 {
                return Err(err(range.start, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(err(s, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(err(s, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(err(s, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        Ok(Rational {
            num: signed_num,
            den: den_mag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let len = self.data.len();
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[c * self.rows + r] = self.data[r * self.cols + c];
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        matrix: &mut Matrix,
        candidate_cols: usize,
    ) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == matrix.rows {
                break;
            }
            let mut selected: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, abs_i64_u128(value.num), row);
                if selected
                    .map(|old| {
                        let v = matrix.data[old * matrix.cols + col];
                        key < (v.den, abs_i64_u128(v.num), old)
                    })
                    .unwrap_or(true)
                {
                    selected = Some(row);
                }
            }
            let Some(q) = selected else {
                continue;
            };

            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..matrix.cols {
                let idx = p * matrix.cols + c;
                matrix.data[idx] = matrix.data[idx]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let idx = row * matrix.cols + c;
                    let product = factor
                        .mul(&matrix.data[p * matrix.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    matrix.data[idx] = matrix.data[idx]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                }
            }

            pivots.push(col);
            p += 1;
        }
        Ok((pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = Self::eliminate(&mut matrix, self.cols)?;
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        self.rref().map(|r| r.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = Rational { num: 1, den: 1 };
        }
        let mut augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let (pivots, _) = Self::eliminate(&mut augmented, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut inverse = Vec::with_capacity(len / 2);
        for r in 0..n {
            for c in 0..n {
                inverse.push(augmented.data[r * width + n + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..m {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            for c in 0..k {
                data[r * width + n + c] = rhs.data[r * k + c];
            }
        }

        let mut augmented = Matrix {
            rows: m,
            cols: width,
            data,
        };
        let (pivots, _) =
            Self::eliminate(&mut augmented, n).map_err(SolveError::Matrix)?;

        for row in 0..m {
            let coeff_zero = (0..n).all(|c| augmented.data[row * width + c].num == 0);
            if coeff_zero {
                for rhs_col in 0..k {
                    if augmented.data[row * width + n + rhs_col].num != 0 {
                        let witness = augmented.data[row * width..(row + 1) * width].to_vec();
                        return Ok(Solution::None {
                            row,
                            rhs_col,
                            witness,
                        });
                    }
                }
            }
        }

        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] =
                    augmented.data[pivot_row * width + n + c];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        if pivots.len() == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let mut free = Vec::new();
        let mut pivot_mask = vec![false; n];
        for &col in &pivots {
            pivot_mask[col] = true;
        }
        for col in 0..n {
            if !pivot_mask[col] {
                free.push(col);
            }
        }

        let basis_len = n
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + basis_col] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
                let coefficient = augmented.data[pivot_row * width + free_col];
                let negated = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: basis_col,
                        step: pivots.len(),
                        source,
                    })
                })?;
                basis_data[pivot_col * free.len() + basis_col] = negated;
            }
        }
        let basis = Matrix {
            rows: n,
            cols: free.len(),
            data: basis_data,
        };
        Ok(Solution::Infinite {
            particular,
            free,
            basis,
            pivots,
        })
    }
}
