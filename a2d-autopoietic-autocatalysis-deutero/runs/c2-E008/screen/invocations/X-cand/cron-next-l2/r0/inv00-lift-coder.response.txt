#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = bounds(field);
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        let (base, step_bytes, step_offset) = match slash_positions.as_slice() {
            [] => (item, None, 0),
            [slash] => (&item[..*slash], Some(&item[*slash + 1..]), *slash + 1),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                })
            }
        };

        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            (None, None, true)
        } else {
            match hyphens.as_slice() {
                [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                    (Some((0, base)), None, false)
                }
                [hyphen]
                    if *hyphen > 0
                        && *hyphen + 1 < base.len()
                        && base[..*hyphen].iter().all(u8::is_ascii_digit)
                        && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
                {
                    (Some((0, &base[..*hyphen])), Some((*hyphen + 1, &base[*hyphen + 1..])), false)
                }
                _ => {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    })
                }
            }
        };

        if step_bytes.is_some_and(|s| s.is_empty() || !s.iter().all(u8::is_ascii_digit)) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (mut start, mut end, is_star) = if is_star_kind(&base_kind) {
            (minimum, maximum, true)
        } else {
            let (start_info, end_info, _) = base_kind;
            let (start_relative, start_bytes) = start_info.unwrap();
            let start_value = parse_number(start_bytes, field, item_start + start_relative)?;
            if start_value < minimum || start_value > maximum {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + start_relative,
                });
            }

            let end_value = if let Some((relative, number_bytes)) = end_info {
                let value = parse_number(number_bytes, field, item_start + relative)?;
                if value < minimum || value > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start + relative,
                    });
                }
                if start_value > value {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + relative,
                    });
                }
                value
            } else if step_bytes.is_some() {
                maximum
            } else {
                start_value
            };
            (start_value, end_value, false)
        };

        let step = if let Some(number_bytes) = step_bytes {
            let value = parse_number(number_bytes, field, item_start + step_offset)?;
            let width = maximum - minimum + 1;
            if value == 0 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + step_offset,
                });
            }
            value
        } else {
            1
        };

        if is_star && step_bytes.is_none() {
            start = minimum;
            end = maximum;
        }

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

fn is_star_kind(kind: &(Option<(usize, &[u8])>, Option<(usize, &[u8])>, bool)) -> bool {
    kind.2
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let parsed: Vec<Vec<bool>> = pieces
        .iter()
        .zip(FIELDS)
        .map(|(&piece, field)| parse_field(piece, field))
        .collect::<Result<_, _>>()?;

    Ok(Cron {
        selected: parsed.try_into().unwrap(),
        day_of_month_is_star: pieces[2] == "*",
        day_of_week_is_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if let Some(next) = increment_minute(current) {
                current = next;
                if self.matches(current)? {
                    return Ok(Some(current));
                }
            } else {
                return Ok(None);
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        at.minute += 1;
        return Some(at);
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return Some(at);
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return Some(at);
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return Some(at);
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return Some(at);
    }
    None
}
