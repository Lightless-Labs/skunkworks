use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [BTreeSet<u8>; 5],
    day_stars: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn parse_number(s: &str, field: Field, offset: usize, step: bool) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(err(field, offset, 1));
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| err(field, offset, 2))?;
    }
    if step {
        let (_, max) = domain(field);
        if n == 0 || n > max + 1 {
            return Err(err(field, offset, 5));
        }
    } else {
        let (min, max) = domain(field);
        if n < min || n > max {
            return Err(err(field, offset, 3));
        }
    }
    Ok(n)
}

fn parse_field(text: &str, field: Field) -> Result<BTreeSet<u8>, ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !matches!(b, b'0'..=b'9' | b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = domain(field);
    let mut result = BTreeSet::new();
    let mut item_start = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let bytes = item.as_bytes();
        let slashes: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, b)| (*b == b'/').then_some(i))
            .collect();

        let (base, step_text, step_offset) = match slashes.as_slice() {
            [] => (item, None, 0),
            [i] => (&item[..*i], Some(&item[*i + 1..]), item_start + *i + 1),
            _ => return Err(err(field, item_start, 0)),
        };

        let (kind, first_text, second_text, second_offset) = if base == "*" {
            (0u8, "", None, 0usize)
        } else if base.contains('*') {
            return Err(err(field, item_start, 0));
        } else {
            let hyphens: Vec<usize> = base
                .bytes()
                .enumerate()
                .filter_map(|(i, b)| (b == b'-').then_some(i))
                .collect();
            match hyphens.as_slice() {
                [] => (1, base, None, 0),
                [i] => (
                    2,
                    &base[..*i],
                    Some(&base[*i + 1..]),
                    item_start + *i + 1,
                ),
                _ => return Err(err(field, item_start, 0)),
            }
        };

        if (kind == 0 && base != "*")
            || (kind == 1 && first_text.is_empty())
            || (kind == 2
                && (first_text.is_empty() || second_text.unwrap_or("").is_empty()))
            || step_text.is_some_and(str::is_empty)
        {
            return Err(err(field, item_start, 0));
        }

        let (start, end) = if kind == 0 {
            (min, max)
        } else {
            let start = parse_number(first_text, field, item_start, false)?;
            if kind == 1 {
                (start, if step_text.is_some() { max } else { start })
            } else {
                let end = parse_number(second_text.unwrap(), field, second_offset, false)?;
                if start > end {
                    return Err(err(field, second_offset, 4));
                }
                (start, end)
            }
        };

        let step = match step_text {
            Some(s) => parse_number(s, field, step_offset, true)?,
            None => 1,
        };

        let mut value = start;
        while value <= end {
            result.insert(value as u8);
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }

    Ok(result)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut values: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    for i in 0..5 {
        values[i] = parse_field(parts[i], FIELDS[i])?;
    }

    Ok(Cron {
        values,
        day_stars: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.values[index].iter().copied().collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.values[0].contains(&at.minute)
            && self.values[1].contains(&at.hour)
            && self.values[2].contains(&at.day)
            && self.values[3].contains(&at.month)
            && self.values[4].contains(&weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        if after == maximum_datetime() {
            return Ok(None);
        }

        let mut current = after;
        loop {
            current = next_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
            if current == maximum_datetime() {
                return Ok(None);
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn maximum_datetime() -> CivilDateTime {
    CivilDateTime { year: 2399, month: 12, day: 31, hour: 23, minute: 59 }
}

fn next_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
