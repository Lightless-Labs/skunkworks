#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn validate_domain(value: u32, field: Field, offset: usize) -> Result<(), ParseError> {
    let (min, max) = domain(field);
    if value < min || value > max {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(())
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let (base_end, step_range) = if let Some(&slash) = slash_positions.first() {
        if slash == 0 || slash + 1 == item.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        (slash, Some((item_start + slash + 1, item_end)))
    } else {
        (item.len(), None)
    };

    let base = &item[..base_end];
    let (min, max) = domain(field);
    let (start, end, is_star) = if base == b"*" {
        (min, max, true)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        if let Some(&hyphen) = hyphens.first() {
            if hyphen == 0 || hyphen + 1 == base.len() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            let start_offset = item_start;
            let end_offset = item_start + hyphen + 1;
            let first = parse_number(&bytes, start_offset, start_offset + hyphen, field)?;
            validate_domain(first, field, start_offset)?;
            let last = parse_number(&bytes, end_offset, item_end - step_range.map_or(0, |_| 0), field)?;
            validate_domain(last, field, end_offset)?;
            if first > last {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (first, last, false)
        } else {
            let value = parse_number(&bytes, item_start, item_start + base.len(), field)?;
            validate_domain(value, field, item_start)?;
            let end = if step_range.is_some() { max } else { value };
            (value, end, false)
        }
    };

    let step = if let Some((step_start, step_end)) = step_range {
        let value = parse_number(&bytes, step_start, step_end, field)?;
        if value == 0 || value > max - min + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: step_start,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    loop {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) if next <= end => value = next,
            _ => break,
        }
    }

    let _ = is_star;
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (_, max) = domain(FIELDS[i]);
        vec![false; max as usize + 1]
    });

    for (index, piece) in pieces.iter().enumerate() {
        let field = FIELDS[index];
        let bytes = piece.as_bytes();

        for (offset, &byte) in bytes.iter().enumerate() {
            if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut item_start = 0;
        for (i, &byte) in bytes.iter().enumerate() {
            if byte == b',' {
                parse_item(bytes, item_start, i, field, &mut selected[index])?;
                item_start = i + 1;
            }
        }
        parse_item(bytes, item_start, bytes.len(), field, &mut selected[index])?;
    }

    Ok(Cron {
        selected,
        day_of_month_is_star: pieces[2] == "*",
        day_of_week_is_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };

        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| {
                (is_selected && domain(field).0 <= value as u32).then_some(value as u8)
            })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.selected[index]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);

        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.contains(Field::DayOfMonth, at.day)
            && self.contains(Field::DayOfWeek, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return true;
    }

    false
}
