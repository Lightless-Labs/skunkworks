pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open = pos;
                let close = match bytes[pos + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => pos + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[pos + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: pos + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let first = bytes[content_start];
                if matches!(first, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                }

                let mut names = Vec::new();
                let mut seen: Vec<&str> = Vec::new();

                loop {
                    let entry_start = content_start;
                    let entry_end = bytes[entry_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| entry_start + offset)
                        .unwrap_or(close);

                    if entry_start == entry_end {
                        return Err(Error::EmptyVariable { pos: entry_start });
                    }

                    let name_bytes = &bytes[entry_start..entry_end];
                    for (i, &b) in name_bytes.iter().enumerate() {
                        let valid = if i == 0 {
                            b.is_ascii_alphabetic() || b == b'_'
                        } else {
                            b.is_ascii_alphanumeric() || b == b'_'
                        };
                        if !valid {
                            return Err(Error::InvalidName {
                                pos: entry_start + i,
                            });
                        }
                    }

                    let name = &template[entry_start..entry_end];
                    if seen.contains(&name) {
                        return Err(Error::DuplicateVariable { pos: entry_start });
                    }
                    seen.push(name);
                    names.push(name);

                    if entry_end == close {
                        break;
                    }
                    content_start = entry_end + 1;
                }

                expressions.push((open, close, names));
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for (open, close, names) in expressions {
        output.push_str(&template[literal_start..open]);

        let mut fields = Vec::new();
        for name in names {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(encode(value));
            }
        }
        output.push_str(&fields.join(","));
        literal_start = close + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                result.push('%');
                result.push(hex_digit(high));
                result.push(hex_digit(low));
                i += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(hex_digit(byte >> 4));
            result.push(hex_digit(byte & 0x0f));
        }
        i += 1;
    }

    result
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

struct Expression<'a> {
    start: usize,
    end: usize,
    names: Vec<&'a str>,
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression<'_>, Error> {
    let bytes = template.as_bytes();
    let content_start = open + 1;

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(
        bytes[content_start],
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
    ) {
        return Err(Error::UnsupportedOperator { pos: content_start });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |offset| spec_start + offset);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        for (offset, &byte) in name_bytes.iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        names.push(name);

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        names,
    })
}

fn hex_value(byte: u8) -> Option<usize> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as usize),
        b'a'..=b'f' => Some((byte - b'a' + 10) as usize),
        b'A'..=b'F' => Some((byte - b'A' + 10) as usize),
        _ => None,
    }
}

fn encode_value(value: &str, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(HEX[high] as char);
                output.push(HEX[low] as char);
                index += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open = index;
                let Some(offset) = bytes[index + 1..].iter().position(|&byte| byte == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = index + 1 + offset;

                if let Some(offset) = bytes[index + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + offset,
                    });
                }

                expressions.push(parse_expression(template, open, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut first_field = true;
        for name in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                if !first_field {
                    output.push(',');
                }
                first_field = false;
                encode_value(value, &mut output);
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, String)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: content_start });
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = match bytes[spec_start..end].iter().position(|&b| b == b',') {
            Some(offset) => spec_start + offset,
            None => end,
        };

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_byte(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push((spec_start, name));

        if spec_end == end {
            break;
        }
        spec_start = spec_end + 1;
    }

    content_start = start + 1;
    let _ = content_start;

    Ok(Expression { start, end, names })
}

fn encode_value(value: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let Some(relative_end) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = i + 1 + relative_end;

                if let Some(relative_open) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + relative_open,
                    });
                }

                expressions.push(parse_expression(template, start, end)?);
                i = end + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);
        let mut fields = Vec::new();

        for (_, name) in expression.names {
            if let Some(value) = lookup(&name, vars) {
                fields.push(encode_value(value));
            }
        }

        output.push_str(&fields.join(","));
        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
