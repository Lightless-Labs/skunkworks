        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },

fn name_value(field: Field, s: &str) -> Option<u32> {
    let upper = s.to_ascii_uppercase();
    let names: &[&str] = match field {
        Field::Month => &["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    names.iter().position(|&n| n == upper).map(|i| {
        if field == Field::DayOfWeek { i as u32 } else { i as u32 + 1 }
    })
}

fn parse_number(s: &str, field: Field, offset: usize, is_step: bool) -> Result<u32, ParseError> {
    let b = s.as_bytes();
    if b.is_empty() || !b.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if b.len() > 1 && b[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut n = 0u32;
    for &c in b {
        n = n.checked_mul(10)
            .and_then(|v| v.checked_add((c - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    if is_step {
        let width = bounds(field).1 - bounds(field).0 + 1;
        if n == 0 || n > width {
            return Err(parse_error(field, offset, 5));
        }
    } else {
        let (lo, hi) = bounds(field);
        if n < lo || n > hi {
            return Err(parse_error(field, offset, 3));
        }
    }
    Ok(n)
}

fn parse_endpoint(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if s.as_bytes().iter().all(u8::is_ascii_digit) {
        parse_number(s, field, offset, false)
    } else if let Some(v) = name_value(field, s) {
        Ok(v)
    } else {
        Err(parse_error(field, offset, 0))
    }
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    for (i, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit() || b.is_ascii_alphabetic() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: item_start });
        }
        let item_bytes = item.as_bytes();
        let slash_positions: Vec<usize> = item_bytes.iter().enumerate().filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None }).collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }
        let (base, step_part, step_offset) = if let Some(&p) = slash_positions.first() {
            (&item[..p], Some(&item[p + 1..]), item_start + p + 1)
        } else {
            (item, None, item_start)
        };
        if base.is_empty() || step_part == Some("") {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }

        let hyphens: Vec<usize> = base.as_bytes().iter().enumerate().filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None }).collect();
        if base == "*" {
            // Valid wildcard.
        } else if hyphens.len() == 1 {
            let p = hyphens[0];
            if p == 0 || p + 1 == base.len() {
                return Err(ParseError::InvalidSyntax { field, offset: item_start });
            }
        } else if !hyphens.is_empty() {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }

        let step = if let Some(s) = step_part {
            if !s.as_bytes().iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax { field, offset: item_start });
            }
            Some(parse_number(s, field, step_offset, true)?)
        } else {
            None
        };

        let (start, end, end_offset) = if base == "*" {
            (min, max, item_start)
        } else if let Some(&p) = hyphens.first() {
            let a = &base[..p];
            let b = &base[p + 1..];
            let av = parse_endpoint(a, field, item_start)?;
            let bv = parse_endpoint(b, field, item_start + p + 1)?;
            if av > bv {
                return Err(ParseError::ReversedRange { field, offset: item_start + p + 1 });
            }
            (av, bv, item_start + p + 1)
        } else {
            let v = parse_endpoint(base, field, item_start)?;
            (v, if step.is_some() { max } else { v }, item_start)
        };

        let stride = step.unwrap_or(1);
        let mut v = start;
        while v <= end {
            selected[v as usize] = true;
            match v.checked_add(stride) {
                Some(next) if next > v => v = next,
                _ => break,
            }
        }
        let _ = end_offset;
        item_start += item.len() + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }
    let fields = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut parsed = Vec::with_capacity(5);
    for (i, field) in fields.iter().enumerate() {
        parsed.push(parse_field(pieces[i], *field)?);

    Ok(Cron {
        selected: parsed.try_into().ok().unwrap(),
        day_of_month_unrestricted: pieces[2] == "*",
        day_of_week_unrestricted: pieces[4] == "*",
    })
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days: u32 = 0;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

            .position(|&name| name == upper)
            .map(|i| i as u32),
        _ => None,
    }
}

fn parse_base_value(
    field: Field,
    text: &str,
    offset: usize,
) -> Result<u32, ParseError> {
    if text.bytes().all(|b| b.is_ascii_digit()) {
        if text.len() > 1 && text.as_bytes()[0] == b'0' {
            return Err(err(field, offset, 1));
        }
        let mut value = 0u32;
        for b in text.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or_else(|| err(field, offset, 2))?;
        }
        let (min, max) = domain(field);
        if value < min || value > max {
            return Err(err(field, offset, 3));
        }
        Ok(value)
    } else {
        name_value(field, text).ok_or_else(|| err(field, offset, 0))
    }
}

fn parse_step(field: Field, text: &str, offset: usize) -> Result<u32, ParseError> {
    if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) {
        return Err(err(field, offset, 5));
    }
    if text.len() > 1 && text.as_bytes()[0] == b'0' {
        return Err(err(field, offset, 1));
    }
    let mut value = 0u32;
    for b in text.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| err(field, offset, 2))?;
    }
    let (min, max) = domain(field);
    let width = max - min + 1;
    if value == 0 || value > width {
        return Err(err(field, offset, 5));
    }
    Ok(value)
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    for (offset, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0;
    for item in text.split(',') {
        let offset = item_start;
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset });
        }

        let slash_parts: Vec<&str> = item.split('/').collect();
        if slash_parts.len() > 2 || slash_parts[0].is_empty()
            || (slash_parts.len() == 2 && slash_parts[1].is_empty())
        {
            return Err(err(field, offset, 0));
        }

        let base = slash_parts[0];
        let step = if slash_parts.len() == 2 {
            Some(parse_step(field, slash_parts[1], offset + base.len() + 1)?)
        } else {
            None
        };

        let (start, end, start_offset, end_offset) = if base == "*" {
            (min, max, offset, offset)
        } else {
            let hyphens: Vec<usize> = base
                .bytes()
                .enumerate()
                .filter_map(|(i, b)| if b == b'-' { Some(i) } else { None })
                .collect();
            if hyphens.len() > 1 {
                return Err(err(field, offset, 0));
            }
            if let Some(&hyphen) = hyphens.first() {
                if hyphen == 0 || hyphen + 1 == base.len() {
                    return Err(err(field, offset, 0));
                }
                let left = &base[..hyphen];
                let right = &base[hyphen + 1..];
                if left == "*" || right == "*" {
                    return Err(err(field, offset, 0));
                }
                let start = parse_base_value(field, left, offset)?;
                let end_offset = offset + hyphen + 1;
                let end = parse_base_value(field, right, end_offset)?;
                if start > end {
                    return Err(err(field, end_offset, 4));
                }
                (start, end, offset, end_offset)
            } else {
                let value = parse_base_value(field, base, offset)?;
                let end = if step.is_some() { max } else { value };
                (value, end, offset, offset)
            }
        };

        let increment = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        let _ = (start_offset, end_offset);
        item_start += item.len() + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, field) in FIELDS.into_iter().enumerate() {
        selected[i] = parse_field(field, parts[i])?;
    }

    Ok(Cron {
        selected,
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_allowed)| {
                if is_allowed {
                    Some(value as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    fn has(&self, field: Field, value: u8) -> bool {
        self.selected[index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn date_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.has(Field::Minute, at.minute)
            && self.has(Field::Hour, at.hour)
            && self.has(Field::Month, at.month)
            && self.date_matches(at)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            if !advance_minute(&mut current) {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) -> bool {
    if at.year == 2399 && at.month == 12 && at.day == 31 && at.hour == 23 && at.minute == 59 {
        return false;
    }
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;
    at.year += 1;
    true
}

fn parse_number(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {

fn parse_endpoint(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {

fn parse_item(
    item: &[u8],
    field: Field,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset: item_offset });
    }

    let slash_positions: Vec<usize> = item.iter().enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();
    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }
    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, item_offset)
    };

    let hyphens: Vec<usize> = base.iter().enumerate()
        .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
        .collect();

    let (start_bytes, end_bytes, range) = if base == b"*" {
        (&[][..], None, false)
    } else if hyphens.len() == 1 {
        let h = hyphens[0];
        (&base[..h], Some(&base[h + 1..]), true)
    } else if hyphens.is_empty() {
        (base, None, false)
    } else {
        return Err(invalid_syntax(field, item_offset));
    };

    if base != b"*" {
        if start_bytes.is_empty() || start_bytes == b"*" {
            return Err(invalid_syntax(field, item_offset));
        }
        if let Some(end) = end_bytes {
            if end.is_empty() || end == b"*" {
                return Err(invalid_syntax(field, item_offset));
            }
        }
        let valid_atom = |bytes: &[u8]| {
            (!bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit))
                || name_value(field, bytes).is_some()
        };
        if !valid_atom(start_bytes) || end_bytes.is_some_and(|b| !valid_atom(b)) {
            return Err(invalid_syntax(field, item_offset));
        }
    }

    let (min, max) = bounds(field);
    let start = if base == b"*" {
        min
    } else {
        parse_endpoint(start_bytes, field, item_offset)?
    };
    let end = if let Some(end_bytes) = end_bytes {
        let end_offset = item_offset + (end_bytes.as_ptr() as usize - item.as_ptr() as usize);
        let value = parse_endpoint(end_bytes, field, end_offset)?;
        if start > value {
            return Err(ParseError::ReversedRange { field, offset: end_offset });
        }
        value
    } else if base == b"*" || step_bytes.is_some() {
        max
    } else {
        start
    };

    let step = if let Some(bytes) = step_bytes {
        if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
            return Err(invalid_syntax(field, item_offset));
        }
        let n = parse_number(bytes, field, step_offset)?;
        if n == 0 || n > width(field) {
            return Err(ParseError::BadStep { field, offset: step_offset });
        }
        n
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}
