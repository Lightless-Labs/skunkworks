```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy: bool,
    idle_since: u64,
    finish: Option<(u64, u32)>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    let mut server_ids = Vec::with_capacity(servers.len());
    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if server_ids.contains(&server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
        server_ids.push(server.id);
    }

    let mut customer_ids = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customer_ids.contains(&customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
        customer_ids.push(customer.id);
    }

    let mut ordered_customers: Vec<usize> = (0..customers.len()).collect();
    ordered_customers.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            busy: false,
            idle_since: server.opens,
            finish: None,
        })
        .collect();

    let mut trace = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut customer_pos = 0;
    let mut time = 0u64;

    loop {
        // Finish events, ordered by ascending server ID.
        let mut finishing: Vec<(usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                state
                    .finish
                    .filter(|(finish_time, _)| *finish_time == time)
                    .map(|(_, customer_id)| (i, customer_id))
            })
            .collect();
        finishing.sort_by_key(|(i, _)| states[*i].id);
        for (i, customer_id) in finishing {
            let state = &mut states[i];
            trace.push(Event::Finish {
                time,
                customer: customer_id,
                server: state.id,
            });
            state.finish = None;
            state.busy = false;
            state.idle_since = time;
        }

        // Open events, ordered by descending server ID.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.opens == time).then_some(i)
            })
            .collect();
        opening.sort_by_key(|i| std::cmp::Reverse(states[*i].id));
        for i in opening {
            let state = &mut states[i];
            state.opened = true;
            state.idle_since = time;
            trace.push(Event::Open {
                time,
                server: state.id,
            });
        }

        // Arrivals at this tick, in the precomputed cohort order.
        while customer_pos < ordered_customers.len()
            && customers[ordered_customers[customer_pos]].arrival == time
        {
            let index = ordered_customers[customer_pos];
            waiting.push(index);
            trace.push(Event::Arrive {
                time,
                customer: customers[index].id,
            });
            customer_pos += 1;
        }

        // Dispatch FIFO customers using the current server ranking.
        while !waiting.is_empty() {
            let best = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && !state.busy)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = best else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index];

            let after_service = time.checked_add(customer.service).ok_or(
                Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                },
            )?;
            let finish = after_service.checked_add(server.overhead).ok_or(
                Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                },
            )?;

            let server_id = server.id;
            trace.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                trace.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
            } else {
                states[server_index].busy = true;
                states[server_index].finish = Some((finish, customer.id));
            }
        }

        if customer_pos == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.finish.is_none())
            && states.iter().all(|state| state.opened)
        {
            break;
        }

        let next_arrival = ordered_customers
            .get(customer_pos)
            .map(|&index| customers[index].arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states
            .iter()
            .filter_map(|state| state.finish.map(|(finish, _)| finish))
            .min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&candidate| candidate > time)
            .min();

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(trace)
}
```
