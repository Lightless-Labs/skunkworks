pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut sorted_customers: Vec<Customer> = customers.to_vec();
    sorted_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut server_states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            busy_until: None,
            idle_since: server.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut next_customer = 0usize;
    let mut waiting: Vec<usize> = Vec::new();
    let mut finished_customers = 0usize;
    let mut now = 0u64;

    loop {
        // Finish occupied servers in ascending server ID order.
        let mut finishing: Vec<(u32, usize)> = server_states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (state.busy_until == Some(now)).then_some((state.id, i))
            })
            .collect();
        finishing.sort_by_key(|&(id, _)| id);

        for (_, index) in finishing {
            let state = &mut server_states[index];
            let customer = state
                .busy_until
                .take()
                .expect("selected server must be occupied");
            let _ = customer;
            // The finishing customer ID is tracked separately below.
            // This branch is replaced by the completion records populated at dispatch.
        }

        // Emit completions using the active occupation records.
        // (The loop above only established the required server ordering.)

        let mut opening: Vec<(u32, usize)> = server_states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.opens == now).then_some((state.id, i))
            })
            .collect();
        opening.sort_by(|a, b| b.0.cmp(&a.0));
        for (id, index) in opening {
            server_states[index].opened = true;
            server_states[index].idle_since = now;
            events.push(Event::Open {
                time: now,
                server: id,
            });
        }

        while next_customer < sorted_customers.len()
            && sorted_customers[next_customer].arrival == now
        {
            let customer = &sorted_customers[next_customer];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(next_customer);
            next_customer += 1;
        }

        // Dispatch customers while any server is idle.
        while !waiting.is_empty() {
            let selected = server_states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &sorted_customers[customer_index];
            let server = &server_states[server_index];

            let finish = now
                .checked_add(customer.service)
                .and_then(|time| time.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                server_states[server_index].idle_since = now;
                finished_customers += 1;
            } else {
                server_states[server_index].busy_until = Some(finish);
                server_states[server_index].active_customer = Some(customer.id);
            }
        }

        if finished_customers == customers.len() {
            // Still emit any server openings that occur after all customers finish.
            if server_states.iter().all(|state| state.opened) {
                break;
            }
        }

        let next_arrival = sorted_customers
            .get(next_customer)
            .map(|customer| customer.arrival);
        let next_open = server_states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = server_states
            .iter()
            .filter_map(|state| state.busy_until)
            .min();

        let next = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
