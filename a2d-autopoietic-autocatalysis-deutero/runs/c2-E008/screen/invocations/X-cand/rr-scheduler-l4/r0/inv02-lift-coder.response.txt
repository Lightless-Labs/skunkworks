```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

#[derive(Clone, Debug)]
struct JobState {
    burst_position: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
        });
    }
    events.sort_by_key(|event| {
        let kind_order = match event.kind {
            EventKind::Arrival => 0,
            EventKind::Wake { .. } => 1,
        };
        (event.time, kind_order, event.index)
    });

    let mut states: Vec<JobState> = jobs
        .iter()
        .map(|job| JobState {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut event_cursor = 0usize;
    let mut now = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    while states.iter().any(|state| state.completion.is_none()) {
        while event_cursor < events.len() && events[event_cursor].time <= now {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    let level = states[event.index].level;
                    queues[level].push_back(event.index);
                }
                EventKind::Wake { level } => {
                    states[event.index].level = level;
                    queues[level].push_back(event.index);
                }
            }
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = jobs[index].id;

        if let Some(last_id) = last_dispatch_id {
            if last_id != id {
                now += switch_cost as u64;
            }
        }

        while event_cursor < events.len() && events[event_cursor].time <= now {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    let event_level = states[event.index].level;
                    queues[event_level].push_back(event.index);
                }
                EventKind::Wake { level: wake_level } => {
                    states[event.index].level = wake_level;
                    queues[wake_level].push_back(event.index);
                }
            }
            event_cursor += 1;
        }

        let quantum = quanta[level];
        let consumed = states[index].remaining.min(quantum);
        let start = now;
        now += consumed as u64;
        let end = now;

        let burst_finished = consumed == states[index].remaining;
        if burst_finished {
            states[index].remaining = 0;
        } else {
            states[index].remaining -= consumed;
        }

        let job_finished = burst_finished && states[index].burst_position + 1 == jobs[index].bursts.len();
        runs.push(Run {
            id,
            start,
            end,
            finished: job_finished,
            level,
        });

        while event_cursor < events.len() && events[event_cursor].time <= end {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    let event_level = states[event.index].level;
                    queues[event_level].push_back(event.index);
                }
                EventKind::Wake { level: wake_level } => {
                    states[event.index].level = wake_level;
                    queues[wake_level].push_back(event.index);
                }
            }
            event_cursor += 1;
        }

        if job_finished {
            states[index].completion = Some(end);
        } else if burst_finished {
            let io_position = states[index].burst_position + 1;
            let io_delay = jobs[index].bursts[io_position] as u64;
            let next_cpu_position = io_position + 1;
            states[index].burst_position = next_cpu_position;
            states[index].remaining = jobs[index].bursts[next_cpu_position];
            let wake_level = level.saturating_sub(1);
            let wake = Event {
                time: end + io_delay,
                kind: EventKind::Wake { level: wake_level },
                index,
            };
            let insert_at = events[event_cursor..]
                .partition_point(|existing| {
                    let existing_kind = match existing.kind {
                        EventKind::Arrival => 0,
                        EventKind::Wake { .. } => 1,
                    };
                    (existing.time, existing_kind, existing.index)
                        <= (wake.time, 1, wake.index)
                })
                + event_cursor;
            events.insert(insert_at, wake);
        } else {
            let demoted_level = (level + 1).min(quanta.len() - 1);
            states[index].level = demoted_level;
            queues[demoted_level].push_back(index);
        }

        last_dispatch_id = Some(id);
    }

    let end = states
        .iter()
        .filter_map(|state| state.completion)
        .max()
        .unwrap_or(0);

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap_or(0);
        let turnaround = completion - job.arrival as u64;
        let work: u64 = job.bursts.iter().map(|&burst| burst as u64).sum();
        let waiting = turnaround - work;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}

use std::collections::VecDeque;
```
