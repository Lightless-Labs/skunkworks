```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

#[derive(Clone, Debug)]
struct JobState {
    burst_position: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

fn round_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        0
    } else {
        let denominator = count as u64;
        (total * 100 + denominator / 2) / denominator
    }
}

fn add_idle(idle: &mut Vec<(u64, u64)>, start: u64, end: u64) {
    if start >= end {
        return;
    }
    if let Some(last) = idle.last_mut() {
        if last.1 == start {
            last.1 = end;
            return;
        }
    }
    idle.push((start, end));
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut states: Vec<JobState> = jobs
        .iter()
        .map(|job| JobState {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        })
        .collect();
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut event_cursor = 0usize;
    let mut now = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completed = 0usize;

    while completed < jobs.len() {
        while event_cursor < events.len() && events[event_cursor].time <= now {
            let event = events[event_cursor];
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > now {
                add_idle(&mut idle, now, next_time);
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = jobs[index].id;

        if last_dispatch_id.is_some_and(|previous| previous != id) {
            now += u64::from(switch_cost);
        }

        while event_cursor < events.len() && events[event_cursor].time <= now {
            let event = events[event_cursor];
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        let consumed = states[index].remaining.min(quanta[level]);
        let start = now;
        now += u64::from(consumed);
        let end = now;

        states[index].remaining -= consumed;
        let burst_finished = states[index].remaining == 0;
        let final_burst = burst_finished
            && states[index].burst_position + 1 == jobs[index].bursts.len();
        let finished = burst_finished && final_burst;

        runs.push(Run {
            id,
            start,
            end,
            finished,
            level,
        });

        while event_cursor < events.len() && events[event_cursor].time <= end {
            let event = events[event_cursor];
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        if finished {
            states[index].completion = Some(end);
            completed += 1;
        } else if burst_finished {
            let io_position = states[index].burst_position + 1;
            let io_delay = jobs[index].bursts[io_position];
            states[index].burst_position += 2;
            states[index].remaining = jobs[index].bursts[states[index].burst_position];

            let wake = Event {
                time: end + u64::from(io_delay),
                kind: 1,
                index,
                level: level.saturating_sub(1),
            };

            let insertion = events[event_cursor..]
                .partition_point(|existing| {
                    (existing.time, existing.kind, existing.index)
                        <= (wake.time, wake.kind, wake.index)
                })
                + event_cursor;
            events.insert(insertion, wake);

            while event_cursor < events.len() && events[event_cursor].time <= end {
                let event = events[event_cursor];
                states[event.index].level = event.level;
                queues[event.level].push_back(event.index);
                event_cursor += 1;
            }
        } else {
            let demoted_level = (level + 1).min(quanta.len() - 1);
            states[index].level = demoted_level;
            queues[demoted_level].push_back(index);
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap_or(0);
        let turnaround = completion - u64::from(job.arrival);
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&delay| u64::from(delay))
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: round_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: round_mean_x100(total_turnaround, jobs.len()),
        end: now,
    })
}
```
