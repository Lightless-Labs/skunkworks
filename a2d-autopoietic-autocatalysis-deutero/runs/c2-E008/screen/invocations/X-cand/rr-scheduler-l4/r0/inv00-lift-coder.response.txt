#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
    level: usize,
}

struct RuntimeJob {
    level: usize,
    burst_index: usize,
    remaining: u32,
    completion: Option<u64>,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        0
    } else {
        let numerator = total * 100;
        let denominator = count as u64;
        (numerator + denominator / 2) / denominator
    }
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut runtimes: Vec<RuntimeJob> = jobs
        .iter()
        .map(|job| RuntimeJob {
            level: job.level,
            burst_index: 0,
            remaining: job.bursts[0],
            completion: None,
        })
        .collect();

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            kind: EventKind::Arrival,
            index,
            level: job.level,
        });
    }

    let mut ready: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut last_id: Option<u32> = None;
    let mut completed_count = 0usize;

    fn event_order(kind: EventKind) -> u8 {
        match kind {
            EventKind::Arrival => 0,
            EventKind::Wake => 1,
        }
    }

    fn admit_through(
        events: &mut Vec<Event>,
        ready: &mut [VecDeque<usize>],
        through: u64,
    ) {
        events.sort_by_key(|event| (event.time, event_order(event.kind), event.index));
        let count = events.partition_point(|event| event.time <= through);
        let admitted: Vec<Event> = events.drain(..count).collect();
        for event in admitted {
            ready[event.level].push_back(event.index);
        }
    }

    while completed_count < jobs.len() {
        admit_through(&mut events, &mut ready, now);

        let selected_level = ready.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if events.is_empty() {
                break;
            }
            events.sort_by_key(|event| (event.time, event_order(event.kind), event.index));
            let next_time = events[0].time;
            if next_time > now {
                if let Some((_, end)) = idle.last_mut() {
                    if *end == now {
                        *end = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_id = None;
            }
            continue;
        };

        let job_index = ready[level].pop_front().unwrap();
        let id = jobs[job_index].id;

        if last_id.is_some_and(|previous| previous != id) {
            now += u64::from(switch_cost);
        }
        admit_through(&mut events, &mut ready, now);

        let quantum = quanta[level];
        let consumed = runtimes[job_index].remaining.min(quantum);
        let start = now;
        now += u64::from(consumed);
        let end = now;

        runtimes[job_index].remaining -= consumed;
        let burst_finished = runtimes[job_index].remaining == 0;
        let final_burst = runtimes[job_index].burst_index + 1 == jobs[job_index].bursts.len();
        let finished = burst_finished && final_burst;

        runs.push(Run {
            id,
            start,
            end,
            finished,
            level,
        });

        admit_through(&mut events, &mut ready, end);

        if finished {
            runtimes[job_index].completion = Some(end);
            completed_count += 1;
        } else if !burst_finished {
            let next_level = (level + 1).min(quanta.len() - 1);
            runtimes[job_index].level = next_level;
            ready[next_level].push_back(job_index);
        } else {
            let io_position = runtimes[job_index].burst_index + 1;
            let io_delay = jobs[job_index].bursts[io_position];
            runtimes[job_index].burst_index += 2;
            runtimes[job_index].remaining = jobs[job_index].bursts[runtimes[job_index].burst_index];

            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: end + u64::from(io_delay),
                kind: EventKind::Wake,
                index: job_index,
                level: wake_level,
            });
        }

        last_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = runtimes[index].completion.unwrap_or(now);
        let turnaround = completion - u64::from(job.arrival);
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&delay| u64::from(delay))
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end: now,
    })
}

use std::collections::VecDeque;
