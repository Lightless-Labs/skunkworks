    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },

#[derive(Debug, Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Question),
            b'&' => Some(Self::Ampersand),
            _ => None,
        }
    }

    fn separator(self) -> Option<char> {
        match self {
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon | Self::Question | Self::Ampersand => Some('&'),
            Self::Simple | Self::Reserved | Self::Fragment => None,
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Question => Some('?'),
            Self::Ampersand => Some('&'),
            Self::Simple | Self::Reserved => None,
        }
    }

    fn is_named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Question | Self::Ampersand)
    }

    fn reserved_encoding(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

fn validate_expression<'a>(
    content: &'a str,
    content_start: usize,
    closing_pos: usize,
) -> Result<(Operator, Vec<Specifier<'a>>), Error> {
    let bytes = content.as_bytes();
    let mut offset = 0;
    let operator = if let Some(&first) = bytes.first() {
        if let Some(op) = Operator::from_byte(first) {
            offset = 1;
            op
        } else if matches!(first, b'=' | b',' | b'!' | b'@' | b'|' | b'$') {
            return Err(Error::UnsupportedOperator {
                pos: content_start,
            });
        } else {
            Operator::Simple
        }
    } else {
        return Err(Error::EmptyExpression { pos: closing_pos });
    };

    if offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: closing_pos });
    }

    let mut specifiers = Vec::new();
    let mut used = std::collections::HashSet::new();
    let mut spec_start = offset;

    for end in (offset..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if end == spec_start {
            return Err(Error::EmptyVariable {
                pos: content_start + spec_start,
            });
        }

        let name_bytes = &bytes[spec_start..end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName {
                pos: content_start + spec_start,
            });
        }
        for (i, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: content_start + spec_start + i,
                });
            }
        }

        let name = &content[spec_start..end];
        if !used.insert(name) {
            return Err(Error::DuplicateVariable {
                pos: content_start + spec_start,
            });
        }
        specifiers.push(Specifier {
            name,
            start: content_start + spec_start,
        });
        spec_start = end + 1;
    }

    Ok((operator, specifiers))
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(char::from(if high < 10 {
                    b'0' + high
                } else {
                    b'A' + high - 10
                }));
                output.push(char::from(if low < 10 {
                    b'0' + low
                } else {
                    b'A' + low - 10
                }));
                i += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || (reserved && is_reserved(byte))
        {
            output.push(char::from(byte));
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(char::from(HEX[(byte >> 4) as usize]));
            output.push(char::from(HEX[(byte & 0x0f) as usize]));
        }
        i += 1;
    }

    output
}

    )

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                if let Some(relative_open) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_open,
                    });
                }

                let content_start = open + 1;
                let content = &template[content_start..close];
                let (operator, specifiers) =
                    validate_expression(content, content_start, close)?;
                expressions.push((open, close + 1, operator, specifiers));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for (start, end, operator, specifiers) in expressions {
        output.push_str(&template[literal_start..start]);

        let mut fields = Vec::new();
        for specifier in specifiers {
            if let Some((_, value)) = vars.iter().rev().find(|(name, _)| *name == specifier.name) {
                let encoded = encode(value, operator.reserved_encoding());
                let field = if operator.is_named() {
                    if matches!(operator, Operator::Semicolon) && encoded.is_empty() {
                        specifier.name.to_owned()
                    } else {
                        format!("{}={}", specifier.name, encoded)
                    }
                } else {
                    encoded
                };
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            if let Some(prefix) = operator.prefix() {
                output.push(prefix);
            }
            if let Some(separator) = operator.separator() {
                output.push_str(&fields.join(&separator.to_string()));
            } else {
                output.push_str(&fields.join(","));
            }
        }

        literal_start = end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn parse_expression(content: &str, content_start: usize, close_pos: usize) -> Result<Expression<'_>, Error> {
    let bytes = content.as_bytes();
    let (operator, body_start) = if let Some(&first) = bytes.first() {
        match first {
            b'+' => (Operator::Plus, 1),
            b'#' => (Operator::Hash, 1),
            b'.' => (Operator::Dot, 1),
            b'/' => (Operator::Slash, 1),
            b';' => (Operator::Semicolon, 1),
            b'?' => (Operator::Question, 1),
            b'&' => (Operator::Ampersand, 1),
            b'=' | b',' | b'!' | b'@' | b'|' => {
                return Err(Error::InvalidName { pos: content_start });
            }
            _ => (Operator::Simple, 0),
        }
    } else {
        return Err(Error::EmptyExpression { pos: close_pos });
    };

    if body_start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let body = &content[body_start..];
    let body_bytes = body.as_bytes();
    let mut specifiers = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut entry_start = 0;

    loop {
        let entry_end = body_bytes[entry_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| entry_start + offset)
            .unwrap_or(body_bytes.len());

        if entry_start == entry_end {
            return Err(Error::EmptyVariable {
                pos: content_start + body_start + entry_start,
            });
        }

        let entry = &body[entry_start..entry_end];
        let entry_bytes = entry.as_bytes();
        let mut invalid_at = None;

        for (index, &byte) in entry_bytes.iter().enumerate() {
            let valid = if index == 0 {
                is_name_start(byte)
            } else {
                is_name_continue(byte)
            };
            if !valid {
                invalid_at = Some(index);
                break;
            }
        }

        if let Some(index) = invalid_at {
            return Err(Error::InvalidName {
                pos: content_start + body_start + entry_start + index,
            });
        }

        if seen.iter().any(|name| *name == entry) {
            return Err(Error::DuplicateVariable {
                pos: content_start + body_start + entry_start,
            });
        }

        seen.push(entry);
        specifiers.push(Specifier {
            name: entry,
            start: content_start + body_start + entry_start,
        });

        if entry_end == body_bytes.len() {
            break;
        }
        entry_start = entry_end + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn encode(value: &str, reserved_allowed: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved_allowed && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let (separator, prefix, reserved_allowed, named, semicolon_empty) =
        match expression.operator {
            Operator::Simple => (",", "", false, false, false),
            Operator::Plus => (",", "", true, false, false),
            Operator::Hash => (",", "#", true, false, false),
            Operator::Dot => (".", ".", false, false, false),
            Operator::Slash => ("/", "/", false, false, false),
            Operator::Semicolon => (";", ";", false, true, true),
            Operator::Question => ("&", "?", false, true, false),
            Operator::Ampersand => ("&", "&", false, true, false),
        };

    let mut fields = Vec::new();
    for specifier in &expression.specifiers {
        if let Some(value) = lookup(specifier.name, vars) {
            let encoded = encode(value, reserved_allowed);
            if named {
                if semicolon_empty && encoded.is_empty() {
                    fields.push(specifier.name.to_owned());
                } else {
                    fields.push(format!("{}={}", specifier.name, encoded));
                }
            } else {
                fields.push(encoded);
            }
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open_pos = cursor;
                let close_pos = match bytes[cursor + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(offset) => cursor + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open_pos }),
                };

                if let Some(offset) = bytes[cursor + 1..close_pos]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + offset,
                    });
                }

                let content_start = cursor + 1;
                let content = &template[content_start..close_pos];
                let expression = parse_expression(content, content_start, close_pos)?;
                expressions.push((open_pos, close_pos + 1, expression));
                cursor = close_pos + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut copied_until = 0;

    for (start, end, expression) in &expressions {
        output.push_str(&template[copied_until..*start]);
        output.push_str(&expand_expression(expression, vars));
        copied_until = *end;
    }
    output.push_str(&template[copied_until..]);

    Ok(output)
}

        Some(b'/') => (Operator::Slash, 1),
        Some(b';') => (Operator::Semicolon, 1),
        Some(b'?') => (Operator::Question, 1),
        Some(b'&') => (Operator::Ampersand, 1),
        Some(b'=') => {
            return Err(Error::UnsupportedOperator { pos: content_pos });
        }
        _ => (Operator::Simple, 0),
    };

    if body_start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut names = Vec::new();
    let mut start = body_start;

    loop {
        let end = (start..bytes.len())
            .find(|&index| bytes[index] == b',')
            .unwrap_or(bytes.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: content_pos + start,
            });
        }

        let specifier = &bytes[start..end];
        if !name_start(specifier[0]) {
            return Err(Error::InvalidName {
                pos: content_pos + start,
            });
        }
        for (offset, &byte) in specifier.iter().enumerate().skip(1) {
            if !name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: content_pos + start + offset,
                });
            }
        }

        let name = &content[start..end];
        if names.contains(&name) {
            return Err(Error::DuplicateVariable {
                pos: content_pos + start,
            });
        }
        names.push(name);

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(Expression { operator, names })
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')

        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let (separator, prefix, allow_reserved, named, semicolon_empty) =
        match expression.operator {
            Operator::Simple => (",", "", false, false, false),
            Operator::Reserved => (",", "", true, false, false),
            Operator::Fragment => (",", "#", true, false, false),
            Operator::Dot => (".", ".", false, false, false),
            Operator::Slash => ("/", "/", false, false, false),
            Operator::Semicolon => (";", ";", false, true, true),
            Operator::Question => ("&", "?", false, true, false),
            Operator::Ampersand => ("&", "&", false, true, false),
        };

    let mut fields = Vec::new();
    for &name in &expression.names {
        if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
            let encoded = encode(value, allow_reserved);
            let field = if named && !(semicolon_empty && encoded.is_empty()) {
                format!("{name}={encoded}")
            } else if named {
                name.to_owned()
            } else {
                encoded
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[cursor + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = cursor + 1 + relative_close;

                if let Some(relative_open) = bytes[cursor + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + relative_open,
                    });
                }

                let content_start = open + 1;
                let expression =
                    parse_expression(&template[content_start..close], content_start, close)?;
                expressions.push((open, close + 1, expression));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[literal_start..start]);
        output.push_str(&expand_expression(&expression, vars));
        literal_start = end;
    }
    output.push_str(&template[literal_start..]);

    Ok(output)
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, start) = match bytes.first().copied() {
        Some(b'+') => (Operator::Reserved, 1),
        Some(b'#') => (Operator::Fragment, 1),
        Some(b'.') => (Operator::Dot, 1),
        Some(b'/') => (Operator::Slash, 1),
        Some(b';') => (Operator::Semicolon, 1),
        Some(b'?') => (Operator::Query, 1),
        Some(b'&') => (Operator::Continuation, 1),
        Some(b'=' | b'!' | b'@' | b'|' | b'$' | b',' | b':') => {
            return Err(Error::UnsupportedOperator { pos: content_pos });
        }
        _ => (Operator::Simple, 0),
    };

    if start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut part_start = start;

    loop {
        let part_end = bytes[part_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| part_start + offset)
            .unwrap_or(bytes.len());

        let absolute_start = content_pos + part_start;
        if part_start == part_end {
            return Err(Error::EmptyVariable { pos: absolute_start });
        }

        let name_bytes = &bytes[part_start..part_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: absolute_start });
        }
        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_byte(byte) {
                return Err(Error::InvalidName {
                    pos: absolute_start + offset,
                });
            }
        }

        let name = content[part_start..part_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: absolute_start });
        }
        seen.push(name.clone());
        variables.push((name, absolute_start));

        if part_end == bytes.len() {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}
