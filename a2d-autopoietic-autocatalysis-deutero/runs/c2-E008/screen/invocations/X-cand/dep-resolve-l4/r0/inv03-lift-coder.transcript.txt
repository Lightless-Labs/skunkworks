fn satisfies(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {

                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                pct(package),
                version_text(*selected),
                cs
            );
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            pct(package),
            version_text(*selected),
            pct(feature),
            pct(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]

    selected: BTreeMap<String, Version>,
}

#[derive(Clone)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn sorted_unique_paths(paths: &mut Vec<Vec<String>>) {
    paths.sort();
    paths.dedup();
}

fn root_token(root: &Requirement, index: usize) -> String {
    let features = root
        .features
        .iter()
        .cloned()
        .collect::<BTreeSet<_>>()
        .into_iter()
        .collect::<Vec<_>>()
        .join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        features
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let features = dep
        .features
        .iter()
        .cloned()
        .collect::<BTreeSet<_>>()
        .into_iter()
        .collect::<Vec<_>>()
        .join(",");
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features,
        gate,
        platform
    )
}

fn build_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    state: &State,
    target: Target,
) -> Closure {
    let mut closure = Closure {
        constraints: BTreeMap::new(),
        origins: BTreeMap::new(),
        requests: BTreeMap::new(),
        paths: BTreeMap::new(),
    };

    for (i, root) in roots.iter().enumerate() {
        closure
            .constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        closure
            .requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        closure
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![root_token(root, i)],
            });
    }

    // Feature and dependency propagation is monotone; repeatedly scan selected
    // releases until no new package, feature, constraint, or provenance is found.
    loop {
        let mut changed = false;

        let selected_names = state.selected.keys().cloned().collect::<Vec<_>>();
        for source in selected_names {
            let version = state.selected[&source];
            let Some(pkg) = catalog.get(&source) else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            let active_features = closure.requests.get(&source).cloned().unwrap_or_default();

            for feature in &release.features {
                if active_features.contains(&feature.name) {
                    let reqs = closure.requests.entry(source.clone()).or_default();
                    for enabled in &feature.enables {
                        changed |= reqs.insert(enabled.clone());
                    }
                }
            }

            let source_paths = closure.paths.get(&source).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !active_features.contains(g))
                {
                    continue;
                }
                let token = dependency_token(&source, version, j, dep);
                let dest = dep.package.clone();

                let cs = closure.constraints.entry(dest.clone()).or_default();
                if !cs.contains(&dep.constraint) {
                    cs.push(dep.constraint);
                    changed = true;
                }

                let reqs = closure.requests.entry(dest.clone()).or_default();
                for f in &dep.features {
                    changed |= reqs.insert(f.clone());
                }

                let paths = closure.paths.entry(dest.clone()).or_default();
                for p in &source_paths {
                    let mut next = p.clone();
                    next.push(token.clone());
                    if !paths.contains(&next) {
                        paths.push(next);
                        changed = true;
                    }
                }

                let origins = closure.origins.entry(dest).or_default();
                for p in &source_paths {
                    let mut next = p.clone();
                    next.push(token.clone());
                    let origin = Origin {
                        constraint: dep.constraint,
                        path: next,
                    };
                    if !origins.iter().any(|o| {
                        o.constraint == origin.constraint && o.path == origin.path
                    }) {
                        origins.push(origin);
                        changed = true;
                    }
                }
            }
        }

        // Add root paths as canonical incoming paths for root packages.
        for (i, root) in roots.iter().enumerate() {
            let p = vec![root_token(root, i)];
            let paths = closure.paths.entry(root.package.clone()).or_default();
            if !paths.contains(&p) {
                paths.push(p);
                changed = true;
            }
        }

        // Keep the canonical shortest, lexicographically smallest path.
        for paths in closure.paths.values_mut() {
            sorted_unique_paths(paths);
            if paths.len() > 1 {
                let best = paths
                    .iter()
                    .min_by(|a, b| a.len().cmp(&b.len()).then(a.cmp(b)))
                    .cloned()
                    .unwrap();
                if paths.as_slice() != [best.clone()] {
                    *paths = vec![best];
                    changed = true;
                }
            }
        }

        if !changed {
            break;
        }
    }

    for constraints in closure.constraints.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    for origins in closure.origins.values_mut() {
        origins.sort_by(|a, b| {
            a.constraint
                .cmp(&b.constraint)
                .then(a.path.cmp(&b.path))
        });
    }
    closure
}

fn evidence_for(closure: &Closure, package: &str) -> Vec<Evidence> {
    let mut e = closure
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    e.sort_by(|a, b| a.constraint.cmp(&b.constraint).then(a.path.cmp(&b.path)));
    e
}

fn active_constraints<'a>(closure: &'a Closure, package: &str) -> &'a [Constraint] {
    closure
        .constraints
        .get(package)
        .map(Vec::as_slice)
        .unwrap_or(&[])
}

enum SearchOutcome {
    Solution(State, Closure),
    Error(ResolveError),
    Limit,
}

struct Search<'a> {
    catalog: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, State, Closure)>,
    first_error: Option<ResolveError>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, state: State) -> SearchOutcome {
        let closure = build_closure(&self.catalog, self.roots, &state, self.target);
        let active = closure.constraints.keys().cloned().collect::<Vec<_>>();

        if let Some(package) = active
            .iter()
            .filter(|p| !self.catalog.contains_key(*p))
            .min()
            .cloned()
        {
            return SearchOutcome::Error(ResolveError::Missing {
                evidence: evidence_for(&closure, &package),
                package,
            });
        }

        for package in state.selected.keys() {
            let version = state.selected[package];
            if active_constraints(&closure, package)
                .iter()
                .any(|c| !satisfies(version, *c))
            {
                return SearchOutcome::Error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: version,
                    requirements: active_constraints(&closure, package).to_vec(),
                    evidence: evidence_for(&closure, package),
                });
            }
        }

        for package in &active {
            if state.selected.contains_key(package) {
                continue;
            }
            let constraints = active_constraints(&closure, package);
            let has = self.catalog[*package].releases.iter().any(|r| {
                constraints.iter().all(|c| satisfies(r.version, *c))
            });
            if !has {
                return SearchOutcome::Error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.to_vec(),
                    evidence: evidence_for(&closure, package),
                });
            }
        }

        // Selected package features are validated after Missing and before
        // constraints, as required by the feature error precedence.
        for package in state.selected.keys() {
            let version = state.selected[package];
            let release = self.catalog[*package]
                .releases
                .iter()
                .find(|r| r.version == version)
                .unwrap();
            let declared = release
                .features
                .iter()
                .map(|f| f.name.as_str())
                .collect::<BTreeSet<_>>();
            if let Some(feature) = closure
                .requests
                .get(package)
                .into_iter()
                .flatten()
                .find(|f| !declared.contains(f.as_str()))
            {
                let path = closure
                    .paths
                    .get(package)
                    .and_then(|p| p.first())
                    .cloned()
                    .unwrap_or_default();
                return SearchOutcome::Error(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: version,
                    feature: feature.clone(),
                    path,
                });
            }
        }

        let next = active
            .iter()
            .find(|p| !state.selected.contains_key(*p))
            .cloned();
        let Some(package) = next else {
            return self.record_solution(state, closure);
        };

        let constraints = active_constraints(&closure, &package);
        let mut candidates = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| satisfies(r.version, *c)))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(&package) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        let mut first_local_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return SearchOutcome::Limit;
            }
            self.attempts += 1;

            let mut child = state.clone();
            child.selected.insert(package.clone(), candidate);
            match self.inspect(child) {
                SearchOutcome::Solution(_, _) => {}
                SearchOutcome::Limit => return SearchOutcome::Limit,
                SearchOutcome::Error(e) => {
                    if first_local_error.is_none() {
                        first_local_error = Some(e);
                    }
                }
            }
        }

        if let Some(error) = first_local_error {
            if self.first_error.is_none() {
                self.first_error = Some(error.clone());
            }
            SearchOutcome::Error(error)
        } else {
            SearchOutcome::Error(ResolveError::NoMatch {
                package,
                requirements: constraints.to_vec(),
                evidence: evidence_for(&closure, active.last().map(String::as_str).unwrap_or("")),
            })
        }
    }

    fn record_solution(&mut self, state: State, closure: Closure) -> SearchOutcome {
        let cost = self
            .locks
            .iter()
            .filter(|(p, v)| state.selected.get(*p) != Some(*v))
            .count();

        let vector = self
            .catalog_names
            .iter()
            .map(|name| state.selected.get(name).copied())
            .collect::<Vec<_>>();

        let replace = match &self.best {
            None => true,
            Some((best_cost, best_vector, _, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            self.best = Some((cost, vector, state, closure));
        }
        SearchOutcome::Solution(State::default(), Closure {
            constraints: BTreeMap::new(),
            origins: BTreeMap::new(),
            requests: BTreeMap::new(),
            paths: BTreeMap::new(),
        })
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();
    let lock_map = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();
    let mut search = Search {
        catalog: catalog_map,
        roots,
        locks: lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        catalog_names: catalog.iter().map(|p| p.name.clone()).collect(),
    };
    search.catalog_names.sort();

    match search.inspect(State::default()) {
        SearchOutcome::Limit => Err(ResolveError::Limit {
            attempts: search.max_attempts,
        }),
        _ => {
            if let Some((_, _, state, closure)) = search.best {
                let selected = state
                    .selected
                    .iter()
                    .map(|(package, version)| Selection {
                        package: package.clone(),
                        version: *version,
                    })
                    .collect::<Vec<_>>();

                let mut all_names = BTreeSet::new();
                all_names.extend(locks.iter().map(|l| l.package.clone()));
                all_names.extend(selected.iter().map(|s| s.package.clone()));

                let mut changes = Vec::new();
                for package in all_names {
                    let before = lock_map.get(&package).copied();
                    let after = state.selected.get(&package).copied();
                    if before == after {
                        continue;
                    }
                    let reason = match (before, after) {
                        (None, Some(_)) => ChangeReason::Added,
                        (Some(_), None) => ChangeReason::Removed,
                        (Some(old), Some(_)) => {
                            let exists = search
                                .catalog
                                .get(&package)
                                .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                            if !exists {
                                ChangeReason::Unavailable
                            } else if active_constraints(&closure, &package)
                                .iter()
                                .any(|c| !satisfies(old, *c))
                            {
                                ChangeReason::Rejected
                            } else {
                                ChangeReason::Rebalanced
                            }
                        }
                        (None, None) => unreachable!(),
                    };
                    changes.push(Change {
                        package,
                        before,
                        after,
                        reason,
                    });
                }

                Ok(Resolution { selected, changes })
            } else {
                Err(search.first_error.unwrap_or(ResolveError::NoMatch {
                    package: String::new(),
                    requirements: Vec::new(),
                    evidence: Vec::new(),
                }))
            }
        }
    }
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn constraint_key(c: &Constraint) -> (u8, Version) {
    match *c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(items: &[Constraint]) -> Vec<Constraint> {
    let mut v = items.to_vec();
    v.sort_by_key(constraint_key);
    v.dedup();
    v
}

fn constraint_text(c: Constraint) -> String {
    let (prefix, v) = match c {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{prefix}{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn platform_active(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let fs = {
        let mut f = dep.features.clone();
        f.sort();
        f.dedup();
        f.join(",")
    };
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs,
        gate,
        platform
    )
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut f = r.features.clone();
    f.sort();
    f.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        f.join(",")
    )
}

fn chosen_release<'a>(
    catalog: &'a BTreeMap<String, &'a Package>,
    selected: &BTreeMap<String, Version>,
    name: &str,
) -> Option<&'a Release> {
    let version = selected.get(name)?;
    catalog
        .get(name)?
        .releases
        .iter()
        .find(|r| r.version == *version)
}

fn compute_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut out = Closure::default();
    let mut queue: Vec<(String, String, Constraint, Vec<String>, Vec<String>)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        let features = root.features.clone();
        queue.push((
            root.package.clone(),
            root.package.clone(),
            root.constraint,
            path,
            features,
        ));
    }

    let mut processed_feature: BTreeSet<(String, String)> = BTreeSet::new();
    let mut processed_dep: BTreeSet<(String, Version, usize)> = BTreeSet::new();

    while !queue.is_empty() {
        let (package, _source, constraint, path, requested_features) = queue.remove(0);
        out.constraints
            .entry(package.clone())
            .or_default()
            .push(constraint);
        out.origins
            .entry(package.clone())
            .or_default()
            .push(Origin {
                constraint,
                path: path.clone(),
            });

        let ppath = out.paths.entry(package.clone()).or_insert_with(|| path.clone());
        if path.len() < ppath.len() || (path.len() == ppath.len() && path < *ppath) {
            *ppath = path.clone();
        }

        let fs = out.features.entry(package.clone()).or_default();
        for f in requested_features {
            fs.insert(f);
        }
    }

    loop {
        let mut changed = false;
        let names: Vec<String> = out.features.keys().cloned().collect();

        for package in names {
            let Some(version) = selected.get(&package).copied() else {
                continue;
            };
            let Some(release) = chosen_release(catalog, selected, &package) else {
                continue;
            };
            let active_features: Vec<String> = out.features[&package].iter().cloned().collect();

            for feature_name in active_features {
                if !processed_feature.insert((package.clone(), feature_name.clone())) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        let was_new = out
                            .features
                            .entry(package.clone())
                            .or_default()
                            .insert(enabled.clone());
                        if was_new {
                            changed = true;
                        }
                    }
                }
            }

            let feature_set = out.features.get(&package).cloned().unwrap_or_default();
            for (i, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !feature_set.contains(g))
                    || !processed_dep.insert((package.clone(), version, i))
                {
                    continue;
                }

                let source_path = out.paths.get(&package).cloned().unwrap_or_default();
                let mut path = source_path;
                path.push(dep_token(&package, version, i, dep));

                out.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                out.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });

                let dest_path = out.paths.entry(dep.package.clone()).or_insert_with(|| path.clone());
                if path.len() < dest_path.len()
                    || (path.len() == dest_path.len() && path < *dest_path)
                {
                    *dest_path = path.clone();
                }

                let dest_features = out.features.entry(dep.package.clone()).or_default();
                for feature in &dep.features {
                    if dest_features.insert(feature.clone()) {
                        changed = true;
                    }
                }
                // A dependency may introduce a package with no feature requests.
                out.constraints.entry(dep.package.clone()).or_default();
            }
        }

        if !changed {
            break;
        }
    }

    for cs in out.constraints.values_mut() {
        *cs = canonical_constraints(cs);
    }
    out
}

fn evidence_for(closure: &Closure, package: &str) -> Vec<Evidence> {
    let mut v = closure
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    v.sort_by(|a, b| {
        constraint_key(&a.constraint)
            .cmp(&constraint_key(&b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    v
}

fn dynamic_error(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> (Closure, Option<ResolveError>) {
    let closure = compute_closure(catalog, roots, selected, target);

    if let Some(package) = closure
        .constraints
        .keys()
        .filter(|p| !catalog.contains_key(*p))
        .next()
    {
        return (
            closure.clone(),
            Some(ResolveError::Missing {
                package: package.clone(),
                evidence: evidence_for(&closure, package),
            }),
        );
    }

    for (package, version) in selected {
        let Some(cs) = closure.constraints.get(package) else {
            continue;
        };
        if cs.iter().any(|c| !matches(*version, *c)) {
            return (
                closure.clone(),
                Some(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: cs.clone(),
                    evidence: evidence_for(&closure, package),
                }),
            );
        }
    }

    for (package, features) in &closure.features {
        let Some(version) = selected.get(package).copied() else {
            continue;
        };
        let Some(release) = chosen_release(catalog, selected, package) else {
            continue;
        };
        let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
        for feature in features {
            if !declared.contains(feature.as_str()) {
                return (
                    closure.clone(),
                    Some(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: version,
                        feature: feature.clone(),
                        path: closure.paths.get(package).cloned().unwrap_or_default(),
                    }),
                );
            }
        }
    }

    for (package, cs) in &closure.constraints {
        if selected.contains_key(package) {
            continue;
        }
        let pkg = catalog[package];
        if !pkg
            .releases
            .iter()
            .any(|r| cs.iter().all(|c| matches(r.version, *c)))
        {
            return (
                closure.clone(),
                Some(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: cs.clone(),
                    evidence: evidence_for(&closure, package),
                }),
            );
        }
    }

    (closure, None)
}

struct Search<'a> {
    catalog: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
    all_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (closure, error) = dynamic_error(&self.catalog, self.roots, selected, self.target);
        if let Some(error) = error {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = closure
            .constraints
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(package) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| selected.get(*name) != Some(*version))
                .count();
            let vector = self
                .all_names
                .iter()
                .map(|name| selected.get(name).copied().unwrap_or(Version {
                    major: 0,
                    minor: 0,
                    patch: 0,
                }))
                .collect::<Vec<_>>();
            let should_replace = match &self.best {
                None => true,
                Some((best_cost, best_vector, _, _)) => {
                    cost < *best_cost || (cost == *best_cost && vector > *best_vector)
                }
            };
            if should_replace {
                self.best = Some((cost, vector, selected.clone(), closure));
            }
            return Ok(());
        };

        let cs = closure.constraints.get(&package).cloned().unwrap_or_default();
        let mut candidates = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| cs.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(&package).copied() {
            if let Some(pos) = candidates.iter().position(|v| *v == lock) {
                candidates.remove(pos);
                candidates.insert(0, lock);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), version);
            let result = self.visit(selected);
            selected.remove(&package);
            result?;
        }
        Ok(())
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        validate_name(format!("root[{i}]"), &root.package)?;
        for (f, feature) in root.features.iter().enumerate() {
            validate_name(format!("root[{i}].feature[{f}]"), feature)?;
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{i}]"), &package.name)?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                validate_name(
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                    &dep.package,
                )?;
                if let Some(gate) = &dep.gate {
                    validate_name(
                        format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                        gate,
                    )?;
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    validate_name(
                        format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                        feature,
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                validate_name(
                    format!("catalog[{i}].release[{j}].feature[{f}]"),
                    &feature.name,
                )?;
                for (e, enabled) in feature.enables.iter().enumerate() {
                    validate_name(
                        format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"),
                        enabled,
                    )?;
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for p in catalog {
        if !package_names.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = BTreeSet::new();
        for release in &p.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {

                        package: p.name.clone(),
                        version: release.version,

        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        all_names,

    search.visit(&mut selected)?;

        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {

                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut change_names = BTreeSet::new();
    for name in lock_map.keys() {
        change_names.insert(name.clone());
    }
    for name in solution.keys() {
        change_names.insert(name.clone());
    }

    let mut changes = Vec::new();
    for package in change_names {
        let before = lock_map.get(&package).copied();
        let after = solution.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {

                    .get(&package)

                    ChangeReason::Rejected

                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {

            } else if x.minor > 0 {

fn canonical_path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)

                constraint: root.constraint,
                path: vec![root_token(i, root)],

            .or_default()

        let path = vec![root_token(i, root)];

            .entry(root.package.clone())
            .and_modify(|old| {
                if canonical_path_better(&path, old) {
                    *old = path.clone();
                }
            })
            .or_insert(path);
    }

                continue;
            };

            };

                .activated
                .get(pkg_name)

                .unwrap_or_default();

                        changed = true;
                    }
                }

                    continue;

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
    limit: bool,
}

impl<'a> Search<'a> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn lock_cost(&self, selection: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|l| selection.get(&l.package) != Some(&l.version))
            .count()
    }

    fn is_better_tie(
        &self,
        candidate: &BTreeMap<String, Version>,
        current: &BTreeMap<String, Version>,
    ) -> bool {
        let mut names = self.catalog.keys().cloned().collect::<Vec<_>>();
        names.sort();
        for name in names {
            match (candidate.get(&name), current.get(&name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn inspect(&mut self, selected: &mut BTreeMap<String, Version>) {
        if self.limit {
            return;
        }

        let cl = compute_closure(self.catalog, self.roots, selected, self.target);

        let mut missing = cl
            .constraints
            .keys()
            .filter(|p| !self.catalog.contains_key(*p))
            .cloned()
            .collect::<Vec<_>>();
        missing.sort();
        if let Some(package) = missing.first() {
            self.record_error(ResolveError::Missing {
                package: package.clone(),
                evidence: build_evidence(&cl, package),
            });
            return;
        }

        for (package, version) in selected.iter() {
            let active = cl.constraints.get(package).cloned().unwrap_or_default();
            if !satisfies(&active, *version) {
                self.record_error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: active,
                    evidence: build_evidence(&cl, package),
                });
                return;
            }
        }

        let mut selected_names = selected.keys().cloned().collect::<Vec<_>>();
        selected_names.sort();
        for package in selected_names {
            let version = selected[&package];
            let Some(catalog_package) = self.catalog.get(&package) else {
                continue;
            };
            let Some(release) = catalog_package
                .releases
                .iter()
                .find(|r| r.version == version)
            else {
                continue;
            };
            let activated = cl.activated.get(&package).cloned().unwrap_or_default();
            for feature in activated {
                if !release.features.iter().any(|f| f.name == feature) {
                    self.record_error(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: version,
                        feature,
                        path: cl.paths.get(&package).cloned().unwrap_or_default(),
                    });
                    return;
                }
            }
        }

        let mut active_packages = cl.constraints.keys().cloned().collect::<Vec<_>>();
        active_packages.sort();
        for package in active_packages {
            if selected.contains_key(&package) {
                continue;
            }
            let constraints = cl.constraints.get(&package).cloned().unwrap_or_default();
            let catalog_package = self.catalog[&package];
            let any = catalog_package
                .releases
                .iter()
                .any(|r| satisfies(&constraints, r.version));
            if !any {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints,
                    evidence: build_evidence(&cl, &package),
                });
                return;
            }
        }

        let next = cl
            .constraints
            .keys()
            .filter(|p| !selected.contains_key(*p))
            .min()
            .cloned();

        let Some(package) = next else {
            let cost = self.lock_cost(selected);
            let replace = match &self.best {
                None => true,
                Some(current) if cost < self.best_cost => true,
                Some(current) if cost == self.best_cost => self.is_better_tie(selected, current),
                _ => false,
            };
            if replace {
                self.best = Some(selected.clone());
                self.best_cost = cost;
            }
            return;
        };

        let constraints = cl.constraints.get(&package).cloned().unwrap_or_default();
        let mut candidates = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| satisfies(&constraints, r.version))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self
            .locks
            .iter()
            .find(|l| l.package == package)
            .map(|l| l.version)
        {
            if let Some(pos) = candidates.iter().position(|v| *v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            selected.insert(package.clone(), version);
            self.inspect(selected);
            selected.remove(&package);
            if self.limit {
                return;
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enable) in feature.enables.iter().enumerate() {
                    if !valid_name(enable) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enable.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
        limit: false,
    };
    let mut selected = BTreeMap::new();
    search.inspect(&mut selected);

    if search.limit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some(best) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let final_closure = compute_closure(&catalog_map, roots, &best, target);
    let lock_map = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();
    let all_names = lock_map
        .keys()
        .chain(best.keys())
        .cloned()
        .collect::<BTreeSet<_>>();

    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if !satisfies(
                    final_closure
                        .constraints
                        .get(&package)
                        .map(Vec::as_slice)
                        .unwrap_or(&[]),
                    old,
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_rank(constraint: Constraint) -> (u8, Version) {
    match constraint {
        Constraint::Exact(version) => (0, version),
        Constraint::AtLeast(version) => (1, version),
        Constraint::Before(version) => (2, version),
        Constraint::Caret(version) => (3, version),
    }
}

fn canonical_constraints(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut result = constraints.to_vec();
    result.sort_by_key(|constraint| constraint_rank(*constraint));
    result.dedup();
    result
}

fn constraint_text(constraint: Constraint) -> String {
    let (prefix, version) = match constraint {
        Constraint::Exact(version) => ("=", version),
        Constraint::AtLeast(version) => (">=", version),
        Constraint::Before(version) => ("<", version),
        Constraint::Caret(version) => ("^", version),
    };
    format!("{prefix}{}", version_text(version))
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints)
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn satisfies(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(required) => version == required,
        Constraint::AtLeast(required) => version >= required,
        Constraint::Before(required) => version < required,
        Constraint::Caret(required) => {
            version >= required
                && if required.major > 0 {
                    version.major == required.major
                } else if required.minor > 0 {
                    version.major == 0 && version.minor == required.minor
                } else {
                    version == required
                }
        }
    }
}

fn satisfies_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| satisfies(version, *constraint))

fn encode(text: &str) -> String {

            result.push(*byte as char);
        } else {
            result.push_str(&format!("%{byte:02X}"));

    }
    result

#[derive(Clone)]
struct Origin {

    path: Vec<String>,

#[derive(Clone, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn path_is_better(candidate: &[String], current: &[String]) -> bool {
    candidate.len() < current.len()
        || (candidate.len() == current.len() && candidate < current)
}

fn compute_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut closure = Closure::default();

    for (index, root) in roots.iter().enumerate() {
        let path = vec![root_token(index, root)];
        closure
            .constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        closure
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: path.clone(),
            });
        closure
            .requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        closure
            .paths
            .entry(root.package.clone())
            .and_modify(|existing| {
                if path_is_better(&path, existing) {
                    *existing = path.clone();
                }
            })
            .or_insert(path);
    }

    loop {
        let mut changed = false;

        for (package_name, version) in selected {
            let Some(package) = catalog.get(package_name) else {
                continue;
            };
            let Some(release) = package
                .releases
                .iter()
                .find(|release| release.version == *version)
            else {
                continue;
            };

            let requested = closure
                .requests
                .get(package_name)
                .cloned()
                .unwrap_or_default();
            let activated = closure
                .activated
                .entry(package_name.clone())
                .or_default();
            let old_len = activated.len();
            activated.extend(requested);
            if activated.len() != old_len {
                changed = true;
            }

            let active_features = closure
                .activated
                .get(package_name)
                .cloned()
                .unwrap_or_default();

            for feature_name in active_features {
                if let Some(feature) = release
                    .features
                    .iter()
                    .find(|feature| feature.name == feature_name)
                {
                    let requests = closure.requests.entry(package_name.clone()).or_default();
                    let old_len = requests.len();
                    requests.extend(feature.enables.iter().cloned());
                    if requests.len() != old_len {
                        changed = true;
                    }
                }
            }

            let active_features = closure
                .activated
                .get(package_name)
                .cloned()
                .unwrap_or_default();

            for (index, dependency) in release.deps.iter().enumerate() {
                if !platform_matches(dependency.platform, target)
                    || dependency
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !active_features.contains(gate))
                {
                    continue;
                }

                let token = dependency_token(package_name, *version, index, dependency);
                let source_path = closure
                    .paths
                    .get(package_name)
                    .cloned()
                    .unwrap_or_default();
                let mut path = source_path;
                path.push(token);

                closure
                    .constraints
                    .entry(dependency.package.clone())
                    .or_default()
                    .push(dependency.constraint);

                let origin = Origin {
                    constraint: dependency.constraint,
                    path: path.clone(),
                };
                let origins = closure
                    .origins
                    .entry(dependency.package.clone())
                    .or_default();
                if !origins.iter().any(|existing| {
                    existing.constraint == origin.constraint && existing.path == origin.path
                }) {
                    origins.push(origin);
                    changed = true;
                }

                let requests = closure
                    .requests
                    .entry(dependency.package.clone())
                    .or_default();
                let old_len = requests.len();
                requests.extend(dependency.features.iter().cloned());
                if requests.len() != old_len {
                    changed = true;
                }

                let destination_path = closure
                    .paths
                    .entry(dependency.package.clone())
                    .or_insert(path.clone());
                if path_is_better(&path, destination_path) {
                    *destination_path = path;
                    changed = true;
                }
            }
        }

        if !changed {
            break;
        }
    }

    for constraints in closure.constraints.values_mut() {
        *constraints = canonical_constraints(constraints);
    }
    closure
}

fn evidence_for(closure: &Closure, package: &str) -> Vec<Evidence> {
    let mut evidence = closure
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|origin| Evidence {
            constraint: origin.constraint,
            path: origin.path.clone(),
        })
        .collect::<Vec<_>>();
    evidence.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    evidence
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, BTreeMap<String, Version>)>,
    limit: bool,
}

impl<'a> Search<'a> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn lock_cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn tie_is_better(
        &self,
        candidate: &BTreeMap<String, Version>,
        current: &BTreeMap<String, Version>,
    ) -> bool {
        for package in self.catalog.keys() {
            match (candidate.get(package), current.get(package)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn inspect(&mut self, selected: &mut BTreeMap<String, Version>) {
        if self.limit {
            return;
        }

        let closure = compute_closure(self.catalog, self.roots, selected, self.target);

        if let Some(package) = closure
            .constraints
            .keys()
            .find(|package| !self.catalog.contains_key(*package))
        {
            self.record_error(ResolveError::Missing {
                package: package.clone(),
                evidence: evidence_for(&closure, package),
            });
            return;
        }

        let mut selected_names = selected.keys().cloned().collect::<Vec<_>>();
        selected_names.sort();
        for package in &selected_names {
            let version = selected[package];
            let constraints = closure
                .constraints
                .get(package)
                .map(Vec::as_slice)
                .unwrap_or(&[]);
            if !satisfies_all(version, constraints) {
                self.record_error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: version,
                    requirements: canonical_constraints(constraints),
                    evidence: evidence_for(&closure, package),
                });
                return;
            }
        }

        for package in &selected_names {
            let version = selected[package];
            let Some(release) = self.catalog[package]
                .releases
                .iter()
                .find(|release| release.version == version)
            else {
                continue;
            };
            if let Some(feature) = closure
                .activated
                .get(package)
                .into_iter()
                .flatten()
                .find(|feature| !release.features.iter().any(|declared| declared.name == **feature))
            {
                self.record_error(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: version,
                    feature: feature.clone(),
                    path: closure.paths.get(package).cloned().unwrap_or_default(),
                });
                return;
            }
        }

        for package in closure.constraints.keys() {
            if selected.contains_key(package) {
                continue;
            }
            let constraints = closure.constraints.get(package).map(Vec::as_slice).unwrap_or(&[]);
            if !self.catalog[package]
                .releases
                .iter()
                .any(|release| satisfies_all(release.version, constraints))
            {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical_constraints(constraints),
                    evidence: evidence_for(&closure, package),
                });
                return;
            }
        }

        let next = closure
            .constraints
            .keys()
            .find(|package| !selected.contains_key(*package))
            .cloned();

        let Some(package) = next else {
            let cost = self.lock_cost(selected);
            let should_replace = match &self.best {
                None => true,
                Some((best_cost, best_selection)) => {
                    cost < *best_cost
                        || (cost == *best_cost && self.tie_is_better(selected, best_selection))
                }
            };
            if should_replace {
                self.best = Some((cost, selected.clone()));
            }
            return;
        };

        let constraints = closure
            .constraints
            .get(&package)
            .cloned()
            .unwrap_or_default();
        let mut candidates = self.catalog[&package]
            .releases
            .iter()
            .filter(|release| satisfies_all(release.version, &constraints))
            .map(|release| release.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock_version) = self
            .locks
            .iter()
            .find(|lock| lock.package == package)
            .map(|lock| lock.version)
        {
            if let Some(index) = candidates.iter().position(|version| *version == lock_version) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            selected.insert(package.clone(), version);
            self.inspect(selected);
            selected.remove(&package);
            if self.limit {
                return;
            }
        }
    }
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        validate_name(format!("root[{index}]"), &root.package)?;
        for (feature_index, feature) in root.features.iter().enumerate() {
            validate_name(format!("root[{index}].feature[{feature_index}]"), feature)?;
        }
    }

    for (package_index, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{package_index}]"), &package.name)?;
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                validate_name(
                    format!(
                        "catalog[{package_index}].release[{release_index}].dep[{dependency_index}]"
                    ),
                    &dependency.package,
                )?;
                if let Some(gate) = &dependency.gate {
                    validate_name(
                        format!(
                            "catalog[{package_index}].release[{release_index}].dep[{dependency_index}].gate"
                        ),
                        gate,
                    )?;
                }
                for (feature_index, feature) in dependency.features.iter().enumerate() {
                    validate_name(
                        format!(
                            "catalog[{package_index}].release[{release_index}].dep[{dependency_index}].feature[{feature_index}]"
                        ),
                        feature,
                    )?;
                }
            }
            for (feature_index, feature) in release.features.iter().enumerate() {
                validate_name(
                    format!(
                        "catalog[{package_index}].release[{release_index}].feature[{feature_index}]"
                    ),
                    &feature.name,
                )?;
                for (enable_index, enable) in feature.enables.iter().enumerate() {
                    validate_name(
                        format!(
                            "catalog[{package_index}].release[{release_index}].feature[{feature_index}].enable[{enable_index}]"
                        ),
                        enable,
                    )?;
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (index, lock) in locks.iter().enumerate() {
        validate_name(format!("lock[{index}]"), &lock.package)?;
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let catalog_map = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect::<BTreeMap<_, _>>();
    let lock_map = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect::<BTreeMap<_, _>>();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        limit: false,
    };

    let mut selected = BTreeMap::new();
    search.inspect(&mut selected);

    if search.limit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, best_selection)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_result = best_selection
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let final_closure = compute_closure(&catalog_map, roots, &best_selection, target);
    let names = lock_map
        .keys()
        .chain(best_selection.keys())
        .cloned()
        .collect::<BTreeSet<_>>();

    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = best_selection.get(&package).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|catalog_package| {
                        catalog_package
                            .releases
                            .iter()
                            .any(|release| release.version == old)
                    });
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure
                    .constraints
                    .get(&package)
                    .is_some_and(|constraints| {
                        constraints
                            .iter()
                            .any(|constraint| !satisfies(old, *constraint))
                    })
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };

        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_result,
        changes,
    })
}
