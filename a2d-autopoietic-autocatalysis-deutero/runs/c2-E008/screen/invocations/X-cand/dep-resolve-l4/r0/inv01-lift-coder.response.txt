    InvalidName {
        site: String,

    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn constraint_key(c: &Constraint) -> (u8, Version) {
    match *c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(items: &[Constraint]) -> Vec<Constraint> {
    let mut v = items.to_vec();
    v.sort_by_key(constraint_key);
    v.dedup();
    v
}

fn constraint_text(c: Constraint) -> String {
    let (prefix, v) = match c {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{prefix}{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut e = evidence.to_vec();
    e.sort_by(|a, b| {
        constraint_key(&a.constraint)
            .cmp(&constraint_key(&b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    e
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn platform_active(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let fs = {
        let mut f = dep.features.clone();
        f.sort();
        f.dedup();
        f.join(",")
    };
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs,
        gate,
        platform
    )
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut f = r.features.clone();
    f.sort();
    f.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        f.join(",")
    )
}

fn chosen_release<'a>(
    catalog: &'a BTreeMap<String, &'a Package>,
    selected: &BTreeMap<String, Version>,
    name: &str,
) -> Option<&'a Release> {
    let version = selected.get(name)?;
    catalog
        .get(name)?
        .releases
        .iter()
        .find(|r| r.version == *version)
}

fn compute_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut out = Closure::default();
    let mut queue: Vec<(String, String, Constraint, Vec<String>, Vec<String>)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        let features = root.features.clone();
        queue.push((
            root.package.clone(),
            root.package.clone(),
            root.constraint,
            path,
            features,
        ));
    }

    let mut processed_feature: BTreeSet<(String, String)> = BTreeSet::new();
    let mut processed_dep: BTreeSet<(String, Version, usize)> = BTreeSet::new();

    while !queue.is_empty() {
        let (package, _source, constraint, path, requested_features) = queue.remove(0);
        out.constraints
            .entry(package.clone())
            .or_default()
            .push(constraint);
        out.origins
            .entry(package.clone())
            .or_default()
            .push(Origin {
                constraint,
                path: path.clone(),
            });

        let ppath = out.paths.entry(package.clone()).or_insert_with(|| path.clone());
        if path.len() < ppath.len() || (path.len() == ppath.len() && path < *ppath) {
            *ppath = path.clone();
        }

        let fs = out.features.entry(package.clone()).or_default();
        for f in requested_features {
            fs.insert(f);
        }
    }

    loop {
        let mut changed = false;
        let names: Vec<String> = out.features.keys().cloned().collect();

        for package in names {
            let Some(version) = selected.get(&package).copied() else {
                continue;
            };
            let Some(release) = chosen_release(catalog, selected, &package) else {
                continue;
            };
            let active_features: Vec<String> = out.features[&package].iter().cloned().collect();

            for feature_name in active_features {
                if !processed_feature.insert((package.clone(), feature_name.clone())) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        let was_new = out
                            .features
                            .entry(package.clone())
                            .or_default()
                            .insert(enabled.clone());
                        if was_new {
                            changed = true;
                        }
                    }
                }
            }

            let feature_set = out.features.get(&package).cloned().unwrap_or_default();
            for (i, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !feature_set.contains(g))
                    || !processed_dep.insert((package.clone(), version, i))
                {
                    continue;
                }

                let source_path = out.paths.get(&package).cloned().unwrap_or_default();
                let mut path = source_path;
                path.push(dep_token(&package, version, i, dep));

                out.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                out.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });

                let dest_path = out.paths.entry(dep.package.clone()).or_insert_with(|| path.clone());
                if path.len() < dest_path.len()
                    || (path.len() == dest_path.len() && path < *dest_path)
                {
                    *dest_path = path.clone();
                }

                let dest_features = out.features.entry(dep.package.clone()).or_default();
                for feature in &dep.features {
                    if dest_features.insert(feature.clone()) {
                        changed = true;
                    }
                }
                // A dependency may introduce a package with no feature requests.
                out.constraints.entry(dep.package.clone()).or_default();
            }
        }

        if !changed {
            break;
        }
    }

    for cs in out.constraints.values_mut() {
        *cs = canonical_constraints(cs);
    }
    out
}

fn evidence_for(closure: &Closure, package: &str) -> Vec<Evidence> {
    let mut v = closure
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    v.sort_by(|a, b| {
        constraint_key(&a.constraint)
            .cmp(&constraint_key(&b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    v
}

fn dynamic_error(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> (Closure, Option<ResolveError>) {
    let closure = compute_closure(catalog, roots, selected, target);

    if let Some(package) = closure
        .constraints
        .keys()
        .filter(|p| !catalog.contains_key(*p))
        .next()
    {
        return (
            closure.clone(),
            Some(ResolveError::Missing {
                package: package.clone(),
                evidence: evidence_for(&closure, package),
            }),
        );
    }

    for (package, version) in selected {
        let Some(cs) = closure.constraints.get(package) else {
            continue;
        };
        if cs.iter().any(|c| !matches(*version, *c)) {
            return (
                closure.clone(),
                Some(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: cs.clone(),
                    evidence: evidence_for(&closure, package),
                }),
            );
        }
    }

    for (package, features) in &closure.features {
        let Some(version) = selected.get(package).copied() else {
            continue;
        };
        let Some(release) = chosen_release(catalog, selected, package) else {
            continue;
        };
        let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
        for feature in features {
            if !declared.contains(feature.as_str()) {
                return (
                    closure.clone(),
                    Some(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: version,
                        feature: feature.clone(),
                        path: closure.paths.get(package).cloned().unwrap_or_default(),
                    }),
                );
            }
        }
    }

    for (package, cs) in &closure.constraints {
        if selected.contains_key(package) {
            continue;
        }
        let pkg = catalog[package];
        if !pkg
            .releases
            .iter()
            .any(|r| cs.iter().all(|c| matches(r.version, *c)))
        {
            return (
                closure.clone(),
                Some(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: cs.clone(),
                    evidence: evidence_for(&closure, package),
                }),
            );
        }
    }

    (closure, None)
}

struct Search<'a> {
    catalog: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
    all_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (closure, error) = dynamic_error(&self.catalog, self.roots, selected, self.target);
        if let Some(error) = error {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = closure
            .constraints
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(package) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| selected.get(*name) != Some(*version))
                .count();
            let vector = self
                .all_names
                .iter()
                .map(|name| selected.get(name).copied().unwrap_or(Version {
                    major: 0,
                    minor: 0,
                    patch: 0,
                }))
                .collect::<Vec<_>>();
            let should_replace = match &self.best {
                None => true,
                Some((best_cost, best_vector, _, _)) => {
                    cost < *best_cost || (cost == *best_cost && vector > *best_vector)
                }
            };
            if should_replace {
                self.best = Some((cost, vector, selected.clone(), closure));
            }
            return Ok(());
        };

        let cs = closure.constraints.get(&package).cloned().unwrap_or_default();
        let mut candidates = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| cs.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(&package).copied() {
            if let Some(pos) = candidates.iter().position(|v| *v == lock) {
                candidates.remove(pos);
                candidates.insert(0, lock);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), version);
            let result = self.visit(selected);
            selected.remove(&package);
            result?;
        }
        Ok(())
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        validate_name(format!("root[{i}]"), &root.package)?;
        for (f, feature) in root.features.iter().enumerate() {
            validate_name(format!("root[{i}].feature[{f}]"), feature)?;
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{i}]"), &package.name)?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                validate_name(
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                    &dep.package,
                )?;
                if let Some(gate) = &dep.gate {
                    validate_name(
                        format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                        gate,
                    )?;
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    validate_name(
                        format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                        feature,
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                validate_name(
                    format!("catalog[{i}].release[{j}].feature[{f}]"),
                    &feature.name,
                )?;
                for (e, enabled) in feature.enables.iter().enumerate() {
                    validate_name(
                        format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"),
                        enabled,
                    )?;
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for p in catalog {
        if !package_names.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = BTreeSet::new();
        for release in &p.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for p in catalog {
        for release in &p.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        validate_name(format!("lock[{i}]"), &lock.package)?;
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let catalog_map = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();
    let lock_map = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();

    let mut all_names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    all_names.sort();
    all_names.dedup();

    let mut search = Search {
        catalog: catalog_map,
        roots,
        locks: lock_map.clone(),
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        all_names,
    };

    let mut selected = BTreeMap::new();
    search.visit(&mut selected)?;

    let Some((_, _, solution, final_closure)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut change_names = BTreeSet::new();
    for name in lock_map.keys() {
        change_names.insert(name.clone());
    }
    for name in solution.keys() {
        change_names.insert(name.clone());
    }

    let mut changes = Vec::new();
    for package in change_names {
        let before = lock_map.get(&package).copied();
        let after = solution.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| cs.iter().any(|c| !matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
