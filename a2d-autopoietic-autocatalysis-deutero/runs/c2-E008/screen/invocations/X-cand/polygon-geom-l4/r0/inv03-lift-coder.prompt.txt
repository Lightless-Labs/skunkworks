const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective.last() == effective.first() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)

fn cross_vec(ax: i128, ay: i128, bx: i128, by: i128) -> i128 {
    ax * by - ay * bx
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    cross_vec(
        b.x as i128 - a.x as i128,
        b.y as i128 - a.y as i128,
        q.x as i128 - a.x as i128,
        q.y as i128 - a.y as i128,
    )
}

fn signed_area2(points: &[Point]) -> i128 {
    let n = points.len();
    let mut sum = 0i128;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_sqrt(n: u128) -> u128 {
    if n == 0 {
        return 0;
    }
    let mut lo = 0u128;
    let mut hi = 1u128;
    while hi.saturating_mul(hi) < n {
        hi *= 2;
    }
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid * mid < n {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let floor = if hi * hi == n { hi } else { lo };
    let next = floor + 1;
    let lower_diff = n - floor * floor;
    let upper_diff = next * next - n;
    if lower_diff >= upper_diff {
        next
    } else {
        floor
    }
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = a.x as i128 - b.x as i128;
    let dy = a.y as i128 - b.y as i128;
    let squared = (dx * dx + dy * dy) as u128;
    rounded_sqrt(squared * 1_000_000) as u64
}

fn endpoint_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b {
        (a, b)
    } else {
        (b, a)
    }
}

fn measures_for(points: &[Point]) -> Measures {
    let area_signed = signed_area2(points);
    let orientation = if area_signed > 0 {
        Orientation::CounterClockwise
    } else if area_signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let n = points.len();
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..n {
        let length = edge_length_milli(points[i], points[(i + 1) % n]);
        edge_milli.push(length);
        perimeter_milli += length;

                            i)

                    longest_edge = Some(i);
                }
            }
        }
    }

        signed_area2: area_signed,
        area2: area_signed.unsigned_abs(),

    }
}

fn point_on_segment(q: Point, a: Point, b: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let n = polygon.len();
    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..n {
            if point_on_segment(q, polygon[i], polygon[(i + 1) % n]) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut odd = false;
        for i in 0..n {
            let a = polygon[i];
            let b = polygon[(i + 1) % n];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    odd = !odd;
                }
            }
        }
        results.push(if odd {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

fn between_collinear(a: Point, b: Point, p: Point) -> bool {
    cross(a, b, p) == 0
        && p.x >= a.x.min(b.x)
        && p.x <= a.x.max(b.x)
        && p.y >= a.y.min(b.y)
        && p.y <= a.y.max(b.y)
}

fn monotone_chain_indices(points: &[(Point, usize)], include_collinear: bool) -> Vec<usize> {
    if points.len() <= 1 {
        return (0..points.len()).collect();
    }

    let should_pop = |a: Point, b: Point, c: Point| {
        let turn = cross(a, b, c);
        if include_collinear {
            turn < 0
        } else {
            turn <= 0
        }
    };

    let mut lower: Vec<usize> = Vec::new();
    for i in 0..points.len() {
        while lower.len() >= 2 {
            let a = points[lower[lower.len() - 2]].0;
            let b = points[lower[lower.len() - 1]].0;
            if should_pop(a, b, points[i].0) {
                lower.pop();
            } else {
                break;
            }
        }
        lower.push(i);
    }

    let mut upper: Vec<usize> = Vec::new();
    for i in (0..points.len()).rev() {
        while upper.len() >= 2 {
            let a = points[upper[upper.len() - 2]].0;
            let b = points[upper[upper.len() - 1]].0;
            if should_pop(a, b, points[i].0) {
                upper.pop();
            } else {
                break;
            }
        }
        upper.push(i);
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);
    lower
}

fn hull_metrics(vertices: &[Point]) -> (u128, u64) {
    if vertices.is_empty() {
        return (0, 0);
    }
    if vertices.len() == 1 {
        return (0, 0);
    }
    let area = signed_area2(vertices).unsigned_abs();
    let mut perimeter = 0u64;
    for i in 0..vertices.len() {
        perimeter += edge_length_milli(vertices[i], vertices[(i + 1) % vertices.len()]);
    }
    (area, perimeter)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = unique.iter_mut().find(|(p, _)| *p == point) {
            existing.1 = existing.1.min(index);
        } else {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|x| x.0).collect();
        let source_indices: Vec<usize> = unique.iter().map(|x| x.1).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|&(p, _)| cross(unique[0].0, unique[1].0, p) == 0);

    let selected: Vec<usize> = if all_collinear {
        match mode {
            Collinear::Include => (0..unique.len()).collect(),
            Collinear::Exclude => vec![0, unique.len() - 1],
        }
    } else {
        monotone_chain_indices(&unique, mode == Collinear::Include)
    };

    let vertices: Vec<Point> = selected.iter().map(|&i| unique[i].0).collect();
    let source_indices: Vec<usize> = selected.iter().map(|&i| unique[i].1).collect();
    let (area2, perimeter_milli) = hull_metrics(&vertices);
    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

fn orientation_sign(a: Point, b: Point, c: Point) -> i8 {
    match cross(a, b, c).cmp(&0) {
        std::cmp::Ordering::Less => -1,
        std::cmp::Ordering::Equal => 0,
        std::cmp::Ordering::Greater => 1,
    }
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let o1 = orientation_sign(a, b, c);
    let o2 = orientation_sign(a, b, d);
    let o3 = orientation_sign(c, d, a);
    let o4 = orientation_sign(c, d, b);

    if o1 != o2 && o3 != o4 {
        return true;
    }
    (o1 == 0 && point_on_segment(c, a, b))
        || (o2 == 0 && point_on_segment(d, a, b))
        || (o3 == 0 && point_on_segment(a, c, d))
        || (o4 == 0 && point_on_segment(b, c, d))
}

fn dot_at(prev: Point, current: Point, next: Point) -> i128 {
    (prev.x as i128 - current.x as i128) * (next.x as i128 - current.x as i128)
        + (prev.y as i128 - current.y as i128) * (next.y as i128 - current.y as i128)
}

                    duplicate_pair = Some(pair);

        return Err(GeomError::Duplicate { first, second });

        return Err(GeomError::ZeroArea);

        let prev = effective[(i + n - 1) % n];
        let current = effective[i];
        let next = effective[(i + 1) % n];
        if cross(prev, current, next) == 0 && dot_at(prev, current, next) > 0 {
            return Err(GeomError::AdjacentOverlap { vertex: i });
        }

            if segments_intersect(
                effective[i],
                effective[(i + 1) % n],

                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,

            }
        }

    let mut source_indices = Vec::with_capacity(n);

            (min_index + offset) % n
        } else {
            (min_index + n - offset) % n
        };
        vertices.push(effective[index]);
        source_indices.push(index);
    }

    let measures = measures_for(&vertices);
    Ok(Validated {
        vertices,
        source_indices,
        measures,
    })

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn shoelace(points: &[Point]) -> i128 {
    if points.is_empty() {
        return 0;
    }
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = a.x as i128 - b.x as i128;
    let dy = a.y as i128 - b.y as i128;
    let d2 = (dx * dx + dy * dy) as u128;
    let scaled = d2 * 1_000_000;
    let root = isqrt(scaled);
    let remainder = scaled - root * root;
    let rounded = if remainder * 4 >= root * 4 + 1 {
        root + 1
    } else {
        root
    };
    rounded as u64
}

    if a <= b {
        (a, b)
    } else {
        (b, a)

fn make_measures(points: &[Point]) -> Measures {
    let signed_area2 = shoelace(points);
    let area2 = signed_area2.unsigned_abs();
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter_milli = 0u64;
    for i in 0..points.len() {
        let len = rounded_edge_milli(points[i], points[(i + 1) % points.len()]);
        edge_milli.push(len);
        perimeter_milli += len;
    }

    let mut longest_edge = None;
    for i in 0..edge_milli.len() {
        match longest_edge {
            None => longest_edge = Some(i),
            Some(best) => {
                if edge_milli[i] > edge_milli[best]
                    || (edge_milli[i] == edge_milli[best]
                        && (endpoint_pair(points[i], points[(i + 1) % points.len()]),
                            i)

                                endpoint_pair(
                                    points[best],
                                    points[(best + 1) % points.len()],
                                ),
                                best,
                            ))

                    longest_edge = Some(i);
                }
            }
        }
    }

    Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let (vertices, _) = prepare(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &q) in queries.iter().enumerate() {
        if !in_range(q) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..vertices.len() {
            if on_segment(vertices[i], vertices[(i + 1) % vertices.len()], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            result.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..vertices.len() {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let side = cross(a, b, q);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }
        result.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(result)
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);

    if ab_c == 0 && on_segment(a, b, c) {
        return true;
    }
    if ab_d == 0 && on_segment(a, b, d) {
        return true;
    }
    if cd_a == 0 && on_segment(c, d, a) {
        return true;
    }
    if cd_b == 0 && on_segment(c, d, b) {
        return true;
    }
    (ab_c < 0) != (ab_d < 0) && (cd_a < 0) != (cd_b < 0)
}

    let px = prev.x as i128 - current.x as i128;
    let py = prev.y as i128 - current.y as i128;
    let nx = next.x as i128 - current.x as i128;
    let ny = next.y as i128 - current.y as i128;
    px * nx + py * ny
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {

    let n = vertices.len();

    for i in 0..n {
        for j in (i + 1)..n {
            if vertices[i] == vertices[j] {
                return Err(GeomError::Duplicate {
                    first: i,
                    second: j,
                });
            }
        }
    }

    let area = shoelace(&vertices);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let current = vertices[i];
        let next = vertices[(i + 1) % n];
        if cross(prev, current, next) == 0 && dot_from_vertex(prev, current, next) > 0 {
            return Err(GeomError::AdjacentOverlap { vertex: i });
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                vertices[i],
                vertices[(i + 1) % n],
                vertices[j],
                vertices[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let min_index = (0..n).min_by_key(|&i| vertices[i]).unwrap();
    let mut order = Vec::with_capacity(n);
    if area > 0 {
        for offset in 0..n {
            order.push((min_index + offset) % n);
        }
    } else {
        for offset in 0..n {
            order.push((min_index + n - offset) % n);
        }
    }
    let canonical: Vec<Point> = order.iter().map(|&i| vertices[i]).collect();
    let measures = make_measures(&canonical);

    Ok(Validated {
        vertices: canonical,
        source_indices: order,
        measures,
    })
}

        vertices,

    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx

    (a.x as i128 - p.x as i128) * (b.x as i128 - p.x as i128)
        + (a.y as i128 - p.y as i128) * (b.y as i128 - p.y as i128)

fn rounded_sqrt(n: u128) -> u128 {
    if n == 0 {
        return 0;
    }
    let mut lo = 0u128;
    let mut hi = 1u128;
    while hi <= n / hi {
        hi *= 2;
    }
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let lower_sq = lo * lo;
    let upper = lo + 1;
    let upper_sq = upper * upper;
    if n - lower_sq >= upper_sq - n {
        upper
    } else {
        lo
    }
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128).unsigned_abs();
    let dy = (a.y as i128 - b.y as i128).unsigned_abs();
    rounded_sqrt((dx * dx + dy * dy) * 1_000_000) as u64
}

fn measures_for(points: &[Point]) -> Measures {
    let signed = signed_area2(points);
    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    let mut longest_edge = None;

    for i in 0..points.len() {
        let length = edge_length_milli(points[i], points[(i + 1) % points.len()]);
        edge_milli.push(length);
        perimeter += length;

        let replace = match longest_edge {
            None => true,
            Some(best) if length > edge_milli[best] => true,
            Some(best) if length < edge_milli[best] => false,
            Some(best) => {
                let pair = {
                    let (a, b) = (points[i], points[(i + 1) % points.len()]);
                    if a <= b { (a, b) } else { (b, a) }
                };
                let best_pair = {
                    let (a, b) = (points[best], points[(best + 1) % points.len()]);
                    if a <= b { (a, b) } else { (b, a) }
                };
                pair < best_pair || (pair == best_pair && i < best)
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: if signed > 0 {
            Orientation::CounterClockwise
        } else if signed < 0 {
            Orientation::Clockwise
        } else {
            Orientation::Degenerate
        },

    Ok(measures_for(&vertices))
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let (vertices, _) = prepare_polygon(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &q) in queries.iter().enumerate() {
        if !in_range(q) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..vertices.len() {
            if on_segment(vertices[i], vertices[(i + 1) % vertices.len()], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            result.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..vertices.len() {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        result.push(if inside { Location::Inside } else { Location::Outside });
    }

    Ok(result)
}

fn point_on_hull_boundary(points: &[Point], q: Point) -> bool {
    if points.len() < 2 {
        return points.first().copied() == Some(q);
    }
    if points.len() == 2 {
        return on_segment(points[0], points[1], q);
    }
    (0..points.len()).any(|i| on_segment(points[i], points[(i + 1) % points.len()], q))
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (i, &p) in points.iter().enumerate() {
        if !unique.iter().any(|(q, _)| *q == p) {
            unique.push((p, i));
        }
    }
    unique.sort_by_key(|&(p, _)| p);

    let mut hull_pairs: Vec<(Point, usize)> = if unique.len() <= 1 {
        unique.clone()
    } else {
        let all_collinear = unique
            .iter()
            .skip(2)
            .all(|(p, _)| cross(unique[0].0, unique[1].0, *p) == 0);

        if all_collinear {
            match mode {
                Collinear::Include => unique.clone(),
                Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
            }
        } else {
            let mut lower: Vec<(Point, usize)> = Vec::new();
            for &item in &unique {
                while lower.len() >= 2 {
                    let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, item.0);
                    if turn > 0 || (mode == Collinear::Include && turn == 0) {
                        break;
                    }
                    lower.pop();
                }
                lower.push(item);
            }

            let mut upper: Vec<(Point, usize)> = Vec::new();
            for &item in unique.iter().rev() {
                while upper.len() >= 2 {
                    let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, item.0);
                    if turn > 0 || (mode == Collinear::Include && turn == 0) {
                        break;
                    }
                    upper.pop();
                }
                upper.push(item);
            }

            lower.pop();
            upper.pop();
            lower.extend(upper);
            lower
        }
    };

    if hull_pairs.len() >= 3 {
        let start = hull_pairs
            .iter()
            .enumerate()
            .min_by_key(|(_, (p, _))| *p)
            .map(|(i, _)| i)
            .unwrap();
        hull_pairs.rotate_left(start);
    }

    let vertices: Vec<Point> = hull_pairs.iter().map(|&(p, _)| p).collect();
    let source_indices: Vec<usize> = hull_pairs.iter().map(|&(_, i)| i).collect();
    let area2 = if vertices.len() < 3 {
        0
    } else {
        signed_area2(&vertices).unsigned_abs()
    };
    let perimeter_milli = if vertices.is_empty() {
        0
    } else {
        (0..vertices.len())
            .map(|i| edge_length_milli(vertices[i], vertices[(i + 1) % vertices.len()]))
            .sum()
    };

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let o1 = cross(a, b, c);
    let o2 = cross(a, b, d);
    let o3 = cross(c, d, a);
    let o4 = cross(c, d, b);

    if o1 == 0 && on_segment(a, b, c) {
        return true;
    }
    if o2 == 0 && on_segment(a, b, d) {
        return true;
    }
    if o3 == 0 && on_segment(c, d, a) {
        return true;
    }
    if o4 == 0 && on_segment(c, d, b) {
        return true;
    }
    (o1 > 0) != (o2 > 0) && (o3 > 0) != (o4 > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let (vertices, _) = prepare_polygon(points)?;
    let n = vertices.len();

    for i in 0..n {
        for j in (i + 1)..n {
            if vertices[i] == vertices[j] {
                return Err(GeomError::Duplicate {
                    first: i,
                    second: j,
                });
            }
        }
    }

    let signed = signed_area2(&vertices);
    if signed == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let cur = vertices[i];
        let next = vertices[(i + 1) % n];
        if cross(prev, cur, next) == 0 && dot_from(cur, prev, next) > 0 {
            return Err(GeomError::AdjacentOverlap { vertex: i });
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                vertices[i],
                vertices[(i + 1) % n],
                vertices[j],
                vertices[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let min_index = (0..n).min_by_key(|&i| vertices[i]).unwrap();
    let mut order: Vec<usize> = if signed > 0 {
        (0..n).map(|k| (min_index + k) % n).collect()
    } else {
        (0..n).map(|k| (min_index + n - k) % n).collect()
    };

    let canonical_vertices: Vec<Point> = order.iter().map(|&i| vertices[i]).collect();
    let measures = measures_for(&canonical_vertices);
    order.shrink_to_fit();

    Ok(Validated {
        vertices: canonical_vertices,
        source_indices: order,
        measures,
    })
}
