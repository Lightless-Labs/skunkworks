impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {

        let id = self.routes.len();
        self.routes.push(Route { id, segments });
        Ok(id)

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let segments = parse_path(head)?;

        let mut best: Option<(&Route, Vec<u8>)> = None;
        for route in &self.routes {
            if route.segments.len() != segments.len() {
                continue;
            }

            let mut rank = Vec::with_capacity(segments.len());
            let mut params = Vec::new();
            let mut matched = true;

            for (pattern_segment, value) in route.segments.iter().zip(&segments) {
                match pattern_segment {
                    Segment::Static(s) if s == value => rank.push(2),
                    Segment::Static(_) => {
                        matched = false;
                        break;
                    }
                    Segment::Param(name) => {
                        rank.push(1);
                        params.push((name.clone(), value.clone()));
                    }
                }
            }

            if matched {
                let replace = match &best {
                    None => true,
                    Some((best_route, best_rank)) => {
                        rank > *best_rank || (rank == *best_rank && route.id < best_route.id)
                    }
                };
                if replace {
                    best = Some((route, rank));
                    // Parameters are reconstructed below for the winning route.
                }
            }
        }

        let (route, _) = best.ok_or(Error::NotFound)?;
        let params = route
            .segments
            .iter()
            .zip(&segments)
            .filter_map(|(segment, value)| match segment {
                Segment::Param(name) => Some((name.clone(), value.clone())),
                Segment::Static(_) => None,
            })
            .collect();

        Ok(Match {
            route: route.id,
            params,
        })
    }

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    let raw_segments = split_segments(input).map_err(|at| Error::InvalidPattern { at })?;
    let mut result = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();

    for (start, raw) in raw_segments {
        if raw.starts_with(':') {
            let name = &raw[1..];
            let valid = {
                let bytes = name.as_bytes();
                !bytes.is_empty()
                    && bytes[0].is_ascii_lowercase()
                    && bytes[1..]
                        .iter()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
            };
            if !valid {
                return Err(Error::InvalidPattern { at: start });
            }
            if !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(Segment::Param(name.to_owned()));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            result.push(Segment::Static(
                decode_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?,
            ));
        }
    }

    Ok(result)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    let raw_segments = split_segments(input).map_err(|at| Error::InvalidPath { at })?;
    raw_segments
        .into_iter()
        .map(|(start, raw)| {
            decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })
        })
        .collect()
}

fn split_segments(input: &str) -> Result<Vec<(usize, &str)>, usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    if bytes.last() == Some(&b'/') {
        return Err(input.len() - 1);
    }

    let mut segments = Vec::new();
    let mut start = 1;
    for i in 1..input.len() {
        if bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            segments.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if start == input.len() {
        return Err(start);
    }
    segments.push((start, &input[start..]));
    Ok(segments)
}

fn decode_segment(raw: &str, start: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(start + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(start + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(start + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(start + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(start + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}
