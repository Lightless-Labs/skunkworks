#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const ONES: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000_000, "quintillion"),
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
];

fn append_small(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(ONES[n as usize]);
    } else {
        let tens = n / 10;
        let unit = n % 10;
        out.push_str(TENS[(tens - 2) as usize]);
        if unit != 0 {
            out.push('-');
            out.push_str(ONES[unit as usize]);
        }
    }
}

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(ONES[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_small(rem, out);
        }
    } else {
        append_small(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut remaining = magnitude;
    let mut higher_present = false;

    for &(scale, name) in &SCALES {
        let group = remaining / scale;
        remaining %= scale;
        if group != 0 {
            if !out.is_empty() && !(value < 0 && out == "minus ") {
                out.push(' ');
            }
            append_group(group as u16, &mut out);
            out.push(' ');
            out.push_str(name);
            higher_present = true;
        }
    }

    if remaining != 0 {
        if higher_present {
            out.push(' ');
            if remaining <= 99 {
                out.push_str("and ");
            }
        }
        append_group(remaining as u16, &mut out);
    }

    Ok(out)
}

#[derive(Clone, Copy)]
enum TokenKind {
    Minus,
    Zero,
    Add(u128),
    Hundred,
    Scale(u128),
    And,
}

fn classify(word: &str) -> Option<TokenKind> {
    if word == "minus" {
        return Some(TokenKind::Minus);
    }
    if word == "zero" {
        return Some(TokenKind::Zero);
    }
    for (i, spelling) in ONES.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(TokenKind::Add(i as u128));
        }
    }
    for (i, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(TokenKind::Add((i as u128 + 2) * 10));
        }
    }
    if word == "hundred" {
        return Some(TokenKind::Hundred);
    }
    if word == "and" {
        return Some(TokenKind::And);
    }
    for &(value, spelling) in &SCALES {
        if word == spelling {
            return Some(TokenKind::Scale(value));
        }
    }
    None
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut kinds = Vec::with_capacity(tokens.len());
    for &(word, index) in &tokens {
        let kind = classify(word).ok_or(WordsError::InvalidWord { index })?;
        kinds.push(kind);
    }

    let negative = matches!(kinds.first(), Some(TokenKind::Minus));
    for (i, kind) in kinds.iter().enumerate() {
        if matches!(kind, TokenKind::Minus) && i != 0 {
            return Err(WordsError::InvalidWord {
                index: tokens[i].1,
            });
        }
    }

    let arithmetic_start = usize::from(negative);
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for kind in kinds.iter().skip(arithmetic_start) {
        match *kind {
            TokenKind::Minus => {}
            TokenKind::Zero | TokenKind::And => {}
            TokenKind::Add(value) => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != input.as_bytes() {
        let differing = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(value)
}
