#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000_000, "quintillion"),
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
];

fn append_under_hundred(out: &mut String, n: u16) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

fn append_group(out: &mut String, n: u16) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_under_hundred(out, remainder);
        }
    } else {
        append_under_hundred(out, n);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut remaining = magnitude;
    let mut higher_present = false;

    for &(scale, name) in &SCALES {
        let group = remaining / scale;
        remaining %= scale;
        if group != 0 {
            if higher_present {
                out.push(' ');
            }
            append_group(&mut out, group as u16);
            out.push(' ');
            out.push_str(name);
            higher_present = true;
        }
    }

    if remaining != 0 {
        if higher_present {
            out.push(' ');
            if remaining <= 99 {
                out.push_str("and ");
            }
        }
        append_group(&mut out, remaining as u16);
    }

    Ok(out)
}

#[derive(Clone, Copy)]
enum TokenKind {
    Minus,
    Zero,
    Add(u128),
    Hundred,
    Scale(u128),
    And,
}

fn classify(word: &str) -> Option<TokenKind> {
    if word == "minus" {
        return Some(TokenKind::Minus);
    }
    if word == "zero" {
        return Some(TokenKind::Zero);
    }
    for (value, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(TokenKind::Add(value as u128));
        }
    }
    for (index, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(TokenKind::Add((index as u128 + 2) * 10));
        }
    }
    if word == "hundred" {
        return Some(TokenKind::Hundred);
    }
    if word == "and" {
        return Some(TokenKind::And);
    }
    for &(value, spelling) in &SCALES {
        if word == spelling {
            return Some(TokenKind::Scale(value));
        }
    }
    None
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut kinds = Vec::with_capacity(tokens.len());
    for &(word, index) in &tokens {
        kinds.push(classify(word).ok_or(WordsError::InvalidWord { index })?);
    }

    let negative = matches!(kinds.first(), Some(TokenKind::Minus));
    for (index, kind) in kinds.iter().enumerate() {
        if matches!(kind, TokenKind::Minus) && index != 0 {
            return Err(WordsError::InvalidWord {
                index: tokens[index].1,
            });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;

    for kind in kinds.iter().skip(usize::from(negative)) {
        match *kind {
            TokenKind::Minus | TokenKind::Zero | TokenKind::And => {}
            TokenKind::Add(value) => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
        }

        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let index = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index });
    }

    Ok(value)
}
