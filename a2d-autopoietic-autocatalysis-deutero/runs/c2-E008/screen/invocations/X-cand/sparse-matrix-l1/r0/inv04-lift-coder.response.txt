pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        if rows > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Row,
                size: rows,
            });
        }
        if cols > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Column,
                size: cols,
            });
        }

        // Validate the complete input before any coordinate sums are attempted.
        for (entry_index, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Row,
                    index: entry.row,
                    limit: rows,
                });
            }
            if entry.col >= cols {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Column,
                    index: entry.col,
                    limit: cols,
                });
            }
            if !entry.value.is_finite() {
                return Err(Error::NonFinite { entry: entry_index });
            }
        }

        // Use a dense slot map over the bounded shape to keep accumulation
        // linear in the input length while preserving its original order.
        let slot_count = rows * cols;
        let mut sums = vec![0.0_f64; slot_count];
        let mut seen = vec![false; slot_count];

        for (entry_index, entry) in entries.iter().enumerate() {
            let slot = entry.row * cols + entry.col;
            let sum = sums[slot] + entry.value;
            if !sum.is_finite() {
                return Err(Error::SumOverflow { entry: entry_index });
            }
            sums[slot] = sum;
            seen[slot] = true;
        }

        let mut canonical = Vec::new();
        for row in 0..rows {
            for col in 0..cols {
                let slot = row * cols + col;
                if seen[slot] && sums[slot] != 0.0 {
                    canonical.push(Entry {
                        row,
                        col,
                        value: sums[slot],
                    });
                }
            }
        }

        Ok(Self {
            rows,
            cols,
            entries: canonical,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}
