#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[0] == *effective.last().unwrap() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let aqx = q.x as i128 - a.x as i128;
    let aqy = q.y as i128 - a.y as i128;
    abx * aqy - aby * aqx
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128 - a.x as i128) as u128;
    let dy = (b.y as i128 - a.y as i128) as u128;
    let n = (dx * dx + dy * dy) * 1_000_000;

    let mut lo = 0u128;
    let mut hi = 3_000_000_000u128;
    while lo < hi {
        let mid = lo + (hi - lo + 1) / 2;
        if mid * mid <= n {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    let lower = lo;
    let midpoint_test = 4 * n;
    let threshold = 4 * lower * lower + 4 * lower + 1;
    if midpoint_test >= threshold {
        (lower + 1) as u64
    } else {
        lower as u64
    }
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn perimeter_and_edges(points: &[Point]) -> (Vec<u64>, u64) {
    let mut edges = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    if points.is_empty() {
        return (edges, perimeter);
    }
    for i in 0..points.len() {
        let length = rounded_edge_milli(points[i], points[(i + 1) % points.len()]);
        edges.push(length);
        perimeter += length;
    }
    (edges, perimeter)
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed = signed_area2(&polygon);
    let orientation = if signed > 0 {
        Orientation::CounterClockwise
    } else if signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };
    let (edge_milli, perimeter_milli) = perimeter_and_edges(&polygon);

    let mut longest_edge = None;
    for i in 0..polygon.len() {
        let replace = match longest_edge {
            None => true,
            Some(best) => {
                if edge_milli[i] != edge_milli[best] {
                    edge_milli[i] > edge_milli[best]
                } else {
                    let pair = {
                        let a = polygon[i];
                        let b = polygon[(i + 1) % polygon.len()];
                        if a <= b { (a, b) } else { (b, a) }
                    };
                    let best_pair = {
                        let a = polygon[best];
                        let b = polygon[(best + 1) % polygon.len()];
                        if a <= b { (a, b) } else { (b, a) }
                    };
                    pair < best_pair || (pair == best_pair && i < best)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        for i in 0..polygon.len() {
            if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], query) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > query.y) != (b.y > query.y) {
                let c = cross(a, b, query);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

fn monotone_hull(points: &[(Point, usize)], include_collinear: bool) -> Vec<(Point, usize)> {
    if points.len() <= 1 {
        return points.to_vec();
    }

    let all_collinear = points
        .iter()
        .skip(2)
        .all(|(p, _)| cross(points[0].0, points[1].0, *p) == 0);

    if all_collinear {
        if include_collinear {
            return points.to_vec();
        }
        return vec![points[0], *points.last().unwrap()];
    }

    let should_pop = |turn: i128| {
        if include_collinear {
            turn < 0
        } else {
            turn <= 0
        }
    };

    let mut lower = Vec::new();
    for &point in points {
        while lower.len() >= 2
            && should_pop(cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, point.0))
        {
            lower.pop();
        }
        lower.push(point);
    }

    let mut upper = Vec::new();
    for &point in points.iter().rev() {
        while upper.len() >= 2
            && should_pop(cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, point.0))
        {
            upper.pop();
        }
        upper.push(point);
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);
    lower
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut indexed: Vec<(Point, usize)> = Vec::with_capacity(points.len());
    for (index, &point) in points.iter().enumerate() {
        if !indexed.iter().any(|(existing, _)| *existing == point) {
            indexed.push((point, index));
        }
    }
    indexed.sort_by_key(|(point, _)| *point);

    let hull = monotone_hull(&indexed, mode == Collinear::Include);
    let vertices: Vec<Point> = hull.iter().map(|(point, _)| *point).collect();
    let source_indices: Vec<usize> = hull.iter().map(|(_, index)| *index).collect();

    let signed = signed_area2(&vertices);
    let (_, perimeter_milli) = perimeter_and_edges(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: signed.unsigned_abs(),
        perimeter_milli,
    })
}
