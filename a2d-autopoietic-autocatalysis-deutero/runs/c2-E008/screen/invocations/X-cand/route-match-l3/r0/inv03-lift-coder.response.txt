use std::collections::BTreeSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (segments, trailing_slash) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, ch)| if ch == '?' || ch == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (path_segments, trailing_slash) = parse_path(head)?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (i, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != trailing_slash
            {
                continue;
            }
            if let Some(params) = match_route(&route.segments, &path_segments) {
                candidates.push((i, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<(usize, Vec<(String, String)>)> = if method == "HEAD" {
            let has_head = candidates
                .iter()
                .any(|(i, _)| self.routes[*i].method == "HEAD");
            let wanted = if has_head { "HEAD" } else { "GET" };
            candidates
                .into_iter()
                .filter(|(i, _)| self.routes[*i].method == wanted)
                .collect()
        } else {
            candidates
                .into_iter()
                .filter(|(i, _)| self.routes[*i].method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (i, _) in &candidates {
                let route_method = &self.routes[*i].method;
                allowed.insert(route_method.clone());
                if route_method == "GET" {
                    allowed.insert("HEAD".to_owned());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let mut winner = eligible[0].clone();
        for candidate in eligible.into_iter().skip(1) {
            if self.routes[candidate.0].rank > self.routes[winner.0].rank {
                winner = candidate;
            }
        }

        let route = &self.routes[winner.0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != trailing_slash
            && !path_segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing_slash {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.0,
            params: winner.1,
            redirect,
        })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let mut raw_segments = split_segments(pattern);
    let trailing_slash = raw_segments
        .last()
        .is_some_and(|(segment, _)| segment.is_empty());
    if trailing_slash {
        raw_segments.pop();
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = BTreeSet::new();

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        let bytes = raw.as_bytes();
        if bytes[0] == b':' || bytes[0] == b'*' {
            let wildcard = bytes[0] == b'*';
            let name = &raw[1..];

            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard && index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }

            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded =
                decode_segment(raw, *start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok((segments, trailing_slash))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let raw_segments = split_segments(path);
    let trailing_slash = raw_segments
        .last()
        .is_some_and(|(segment, _)| segment.is_empty());
    let count = raw_segments.len() - usize::from(trailing_slash);
    let mut segments = Vec::with_capacity(count);

    for (raw, start) in raw_segments.iter().take(count) {
        if raw.is_empty() {
            return Err(Error::InvalidPath { at: *start });
        }
        segments.push(
            decode_segment(raw, *start).map_err(|at| Error::InvalidPath { at })?,
        );
    }

    Ok((segments, trailing_slash))
}

fn split_segments(input: &str) -> Vec<(&str, usize)> {
    let mut result = Vec::new();
    let mut start = 1;

    for (relative, _) in input[1..].match_indices('/') {
        let absolute = relative + 1;
        result.push((&input[start..absolute], start));
        start = absolute + 1;
    }
    result.push((&input[start..], start));
    result
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(base + i);
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

fn match_route(
    pattern: &[Segment],
    path: &[String],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), path[path_index..].join("/")));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
