fn valid_method(method: &str) -> bool {
    !method.is_empty()
        && method
            .bytes()

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let mut raw_segments = split_segments(pattern, true)?;
    let trailing_slash = raw_segments.last().map_or(false, |s| s.is_empty());
    if trailing_slash {
        raw_segments.pop();
    }

    let mut parsed = Vec::with_capacity(raw_segments.len());
    let mut names = BTreeSet::new();
    for (index, (part, start)) in raw_segments.iter().enumerate() {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }
        let bytes = part.as_bytes();
        if bytes[0] == b':' || bytes[0] == b'*' {
            let is_wildcard = bytes[0] == b'*';
            let name = &part[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if is_wildcard && index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            parsed.push(if is_wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded =
                decode_segment(part, *start).map_err(|at| Error::InvalidPattern { at })?;
            parsed.push(Segment::Static(decoded));
        }
    }

    Ok((parsed, trailing_slash))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let raw_segments = split_segments(path, false)?;
    let trailing_slash = raw_segments.last().map_or(false, |s| s.0.is_empty());
    let mut segments = Vec::with_capacity(raw_segments.len());

    for (part, start) in raw_segments {
        if part.is_empty() {
            if trailing_slash && start == path.len() {
                continue;
            }
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(part, start).map_err(|at| Error::InvalidPath { at })?);
    }

    Ok((segments, trailing_slash))
}

// Splits a slash-prefixed non-root path while preserving each segment's byte offset.
fn split_segments<'a>(
    path: &'a str,
    pattern: bool,
) -> Result<Vec<(&'a str, usize)>, Error> {
    let mut result = Vec::new();
    let mut start = 1;
    for (i, byte) in path.bytes().enumerate().skip(1) {
        if byte == b'/' {
            if i == path.len() - 1 {
                result.push((&path[start..i], start));
                result.push(("", path.len()));
            } else {
                result.push((&path[start..i], start));
            }
            start = i + 1;
        }
    }
    if start < path.len() {
        result.push((&path[start..], start));
    }

    if result.is_empty() {
        return Ok(result);
    }

    for (i, (part, offset)) in result.iter().enumerate() {
        if part.is_empty() && !(i + 1 == result.len() && *offset == path.len()) {
            return Err(if pattern {
                Error::InvalidPattern { at: *offset }
            } else {
                Error::InvalidPath { at: *offset }
            });
        }
    }

    Ok(result)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(base + i);
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn match_route(
    pattern: &[Segment],
    path: &[String],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let mut raw_segments = split_segments(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let trailing = raw_segments.last().is_some_and(|(s, _)| s.is_empty());
    if trailing {
        raw_segments.pop();
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        if let Some(name) = raw.strip_prefix(':') {
            let valid = valid_name(name);
            if !valid || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            let valid = valid_name(name);
            if !valid || !names.insert(name.to_owned()) || index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            let decoded = decode_segment(raw, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok((segments, trailing))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let raw_segments = split_segments(path).map_err(|at| Error::InvalidPath { at })?;
    let trailing = raw_segments.last().is_some_and(|(s, _)| s.is_empty());
    let count = raw_segments.len() - usize::from(trailing);
    let mut segments = Vec::with_capacity(count);

    for (raw, start) in raw_segments.iter().take(count) {
        if raw.is_empty() {
            return Err(Error::InvalidPath { at: *start });
        }
        segments.push(
            decode_segment(raw, *start).map_err(|at| Error::InvalidPath { at })?,
        );
    }

    Ok((segments, trailing))
}

fn split_segments(input: &str) -> Result<Vec<(&str, usize)>, usize> {
    let mut result = Vec::new();
    let mut start = 1;

    for (relative, _) in input[1..].match_indices('/') {
        let absolute = relative + 1;
        result.push((&input[start..absolute], start));
        start = absolute + 1;
    }
    result.push((&input[start..], start));

    // A slash immediately after the leading slash is an empty interior segment.
    if input.starts_with("//") {
        return Err(1);
    }

    Ok(result)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, start: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(start + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(start + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(start + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(start + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(start + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut ri = 0;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if request.get(ri)? != value {
                    return None;
                }
                ri += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(ri)?.clone()));
                ri += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), request[ri..].join("/")));
                ri = request.len();
            }
        }
    }

    if ri == request.len() {
        Some(params)
    } else {
        None
    }
}

fn rank(pattern: &[Segment]) -> Vec<u8> {
    let mut result = Vec::new();

        match segment {
            Segment::Static(_) => result.push(2),
            Segment::Param(_) => result.push(1),
            Segment::Wildcard(_) => {
                result.push(0);
                return result;
            }
        }
    }
    result.push(3);
    result
}

    input: &'a str,
    pattern: bool,

            Error::InvalidPattern { at }
        } else {

        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut result = Vec::new();
    let mut start = 1;

        if bytes[i] == b'/' {
            if i == start {
                return Err(invalid(start));
            }
            result.push((&input[start..i], start));
            start = i + 1;
        }
    }

    if start == end {
        return Err(invalid(start));
    }
    result.push((&input[start..end], start));
    Ok((result, trailing))

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        let bytes = raw.as_bytes();
        if bytes[0] == b':' || bytes[0] == b'*' {
            let wildcard = bytes[0] == b'*';
            let name = &raw[1..];

            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard && index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }

            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded =
                decode_segment(raw, *start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok((segments, trailing_slash))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let raw_segments = split_segments(path);
    let trailing_slash = raw_segments
        .last()
        .is_some_and(|(segment, _)| segment.is_empty());
    let count = raw_segments.len() - usize::from(trailing_slash);
    let mut segments = Vec::with_capacity(count);

    for (raw, start) in raw_segments.iter().take(count) {
        if raw.is_empty() {
            return Err(Error::InvalidPath { at: *start });
        }
        segments.push(
            decode_segment(raw, *start).map_err(|at| Error::InvalidPath { at })?,
        );
    }

    Ok((segments, trailing_slash))
}

fn split_segments(input: &str) -> Vec<(&str, usize)> {
    let mut result = Vec::new();
    let mut start = 1;

    for (relative, _) in input[1..].match_indices('/') {
        let absolute = relative + 1;
        result.push((&input[start..absolute], start));
        start = absolute + 1;
    }
    result.push((&input[start..], start));
    result
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(base + i);
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

fn match_route(
    pattern: &[Segment],
    path: &[String],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), path[path_index..].join("/")));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
