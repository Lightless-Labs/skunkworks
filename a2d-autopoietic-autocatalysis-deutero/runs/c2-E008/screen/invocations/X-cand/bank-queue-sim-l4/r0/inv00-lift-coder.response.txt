use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

#[derive(Clone)]
struct State {
    index: usize,
    remaining: u64,
    first: bool,
    paused: bool,
    pause_time: u64,
    entry: u64,
    budget: Option<u64>,
    deadline: Option<u64>,
    waited: u64,
}

#[derive(Clone)]
struct Active {
    state: State,
    dispatch: u64,
    work_at: u64,
    finish: u64,
}

fn event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiter_cmp(a: &State, b: &State, customers: &[Customer]) -> Ordering {
    let ca = &customers[a.index];
    let cb = &customers[b.index];
    ca.priority
        .cmp(&cb.priority)
        .then_with(|| match (a.paused, b.paused) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (true, true) => a.pause_time.cmp(&b.pause_time).then(ca.id.cmp(&cb.id)),
            (false, false) => ca
                .arrival
                .cmp(&cb.arrival)
                .then_with(|| cb.id.cmp(&ca.id)),
        })
}

fn add_waiting_segment(state: &mut State, now: u64) {
    let segment = now - state.entry;
    state.waited += segment;
    state.budget = state.budget.map(|b| b - segment);
}

fn expire_waiters(
    queue: &mut Vec<State>,
    now: u64,
    customers: &[Customer],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<State> = Vec::new();
    let mut keep: Vec<State> = Vec::with_capacity(queue.len());
    for state in queue.drain(..) {
        if state.deadline.is_some_and(|d| d <= now) {
            expired.push(state);
        } else {
            keep.push(state);
        }
    }
    expired.sort_by(|a, b| {
        a.deadline
            .cmp(&b.deadline)
            .then(customers[a.index].id.cmp(&customers[b.index].id))
    });
    for mut state in expired {
        add_waiting_segment(&mut state, now);
        let id = customers[state.index].id;
        event(events, now, id, Kind::Renege { waited: state.waited });
        stats.reneged += 1;
        stats.wait_ticks += state.waited as u128;
    }
    *queue = keep;
}

fn best_index(queue: &[State], customers: &[Customer]) -> Option<usize> {
    queue
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiter_cmp(a, b, customers))
        .map(|(i, _)| i)
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index,
                id: c.id,
            })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut queue: Vec<State> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut arrival_pos = 0usize;
    let mut now = customers
        .iter()
        .map(|c| c.arrival)
        .min()
        .unwrap_or(0);

    while stats.completed + stats.reneged + stats.balked < customers.len() {
        if let Some(a) = &active {
            if a.finish == now {
                let state = active.take().unwrap().state;
                let c = &customers[state.index];
                event(&mut events, now, c.id, Kind::Finish);
                stats.completed += 1;
                stats.completion_order.push(c.id);
                stats.turnaround_ticks += (now - c.arrival) as u128;
            }
        }

        expire_waiters(&mut queue, now, customers, &mut events, &mut stats);

        while arrival_pos < arrivals.len() && customers[arrivals[arrival_pos]].arrival == now {
            let index = arrivals[arrival_pos];
            arrival_pos += 1;
            let c = &customers[index];
            event(&mut events, now, c.id, Kind::Arrive);
            let active_slot = usize::from(active.is_none());
            if queue.len() < config.waiting_capacity + active_slot {
                stats.accepted += 1;
                queue.push(State {
                    index,
                    remaining: c.service,
                    first: true,
                    paused: false,
                    pause_time: 0,
                    entry: now,
                    budget: c.patience,
                    deadline: initial_deadlines[index],
                    waited: 0,
                });
            } else {
                stats.balked += 1;
                event(&mut events, now, c.id, Kind::Balk);
            }
        }

        if let Some(a) = &active {
            if now >= a.work_at {
                if let Some(wi) = best_index(&queue, customers) {
                    let running_priority = customers[a.state.index].priority;
                    let waiter_priority = customers[queue[wi].index].priority;
                    if waiter_priority < running_priority {
                        let by = customers[queue[wi].index].id;
                        let mut paused = active.take().unwrap();
                        let remaining = paused.finish - now;
                        let id = customers[paused.state.index].id;
                        event(
                            &mut events,
                            now,
                            id,
                            Kind::Preempt { remaining, by },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - paused.dispatch) as u128;
                        paused.state.remaining = remaining;
                        paused.state.paused = true;
                        paused.state.pause_time = now;
                        paused.state.entry = now;
                        paused.state.deadline = match paused.state.budget {
                            Some(b) => Some(now.checked_add(b).ok_or(Error::TimeOverflow {
                                customer: id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?),
                            None => None,
                        };
                        queue.push(paused.state);
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_index(&queue, customers) else {
                break;
            };
            let mut state = queue.remove(wi);
            add_waiting_segment(&mut state, now);
            let c = &customers[state.index];
            stats.wait_ticks += (now - state.entry) as u128;
            let switching = last_dispatched.is_some_and(|id| id != c.id);
            let cost = if switching { config.switch_cost } else { 0 };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: c.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(state.remaining).ok_or(Error::TimeOverflow {
                customer: c.id,
                time: now,
                stage: OverflowStage::Finish,
            })?;
            let kind = if state.first {
                stats.started += 1;
                Kind::Start {
                    remaining: state.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Resume {
                    remaining: state.remaining,
                    work_at,
                    finish,
                }
            };
            event(&mut events, now, c.id, kind);
            state.first = false;
            last_dispatched = Some(c.id);
            active = Some(Active {
                state,
                dispatch: now,
                work_at,
                finish,
            });

            if finish == now {
                let completed = active.take().unwrap().state;
                event(&mut events, now, c.id, Kind::Finish);
                stats.completed += 1;
                stats.completion_order.push(c.id);
                stats.turnaround_ticks += (now - c.arrival) as u128;
                // Count the zero-length active interval.
                let _ = completed;
                continue;
            }
            break;
        }

        expire_waiters(&mut queue, now, customers, &mut events, &mut stats);

        if stats.completed + stats.reneged + stats.balked == customers.len() {
            break;
        }

        let mut next: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now {
                next = Some(next.map_or(t, |n| n.min(t)));
            }
        };
        if arrival_pos < arrivals.len() {
            consider(customers[arrivals[arrival_pos]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if now < a.work_at {
                consider(a.work_at);
            }
        }
        for state in &queue {
            if let Some(d) = state.deadline {
                consider(d);
            }
        }
        if let Some(t) = next {
            now = t;
        } else {
            // Only zero-duration work can leave no future event; Phase E handles it.
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks.saturating_mul(1000), stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(
            stats.turnaround_ticks.saturating_mul(1000),
            stats.completed as u128,
        )
        .unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks.saturating_mul(10000), stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
