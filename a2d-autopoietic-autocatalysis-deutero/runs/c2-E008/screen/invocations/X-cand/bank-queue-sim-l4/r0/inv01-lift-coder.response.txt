use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    waited: u64,
    queue_since: u64,
    pause_time: Option<u64>,
    started: bool,
    accepted: bool,
    terminal: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch: u64,
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

fn waiting_cmp(a: &Job, b: &Job) -> Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(x), Some(y)) => x.cmp(&y).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter(jobs: &[Job], waiting: &[usize]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| waiting_cmp(&jobs[a], &jobs[b]))
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut seen = Vec::<u32>::new();
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        seen.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        let deadline = match c.patience {
            Some(p) => Some(
                c.arrival
                    .checked_add(p)
                    .ok_or(Error::DeadlineOverflow { index, id: c.id })?,
            ),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut jobs: Vec<Job> = customers
        .iter()
        .enumerate()
        .map(|(i, c)| Job {
            customer: c.clone(),
            deadline: initial_deadlines[i],
            remaining: c.service,
            waited: 0,
            queue_since: c.arrival,
            pause_time: None,
            started: false,
            accepted: false,
            terminal: false,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        jobs[a]
            .customer
            .arrival
            .cmp(&jobs[b].customer.arrival)
            .then_with(|| jobs[a].customer.priority.cmp(&jobs[b].customer.priority))
            .then_with(|| jobs[b].customer.id.cmp(&jobs[a].customer.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0;
    let mut last_dispatch_id: Option<u32> = None;
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut now = arrivals
        .first()
        .map(|&i| jobs[i].customer.arrival)
        .unwrap_or(0);

    loop {
        // Phase A: finish at this boundary.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let id = jobs[a.index].customer.id;
            emit(&mut events, now, id, Kind::Finish);
            jobs[a.index].terminal = true;
            stats.completed += 1;
            stats.turnaround_ticks += (now - jobs[a.index].customer.arrival) as u128;
            stats.completion_order.push(id);
        }

        // Phase B: expire waiters present at tick start.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            jobs[a]
                .deadline
                .cmp(&jobs[b].deadline)
                .then_with(|| jobs[a].customer.id.cmp(&jobs[b].customer.id))
        });
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_since;
                jobs[i].waited = jobs[i].waited.saturating_add(segment);
                stats.wait_ticks += segment as u128;
                emit(
                    &mut events,
                    now,
                    jobs[i].customer.id,
                    Kind::Renege { waited: jobs[i].waited },
                );
                jobs[i].terminal = true;
                stats.reneged += 1;
            }
        }

        // Phase C: full arrival cohort, with no dispatch between arrivals.
        while arrival_pos < arrivals.len()
            && jobs[arrivals[arrival_pos]].customer.arrival == now
        {
            let i = arrivals[arrival_pos];
            arrival_pos += 1;
            let id = jobs[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() < gate {
                jobs[i].accepted = true;
                jobs[i].queue_since = now;
                waiting.push(i);
                stats.accepted += 1;
            } else {
                emit(&mut events, now, id, Kind::Balk);
                stats.balked += 1;
                jobs[i].terminal = true;
            }
        }

        // Phase D: priority preemption after switching protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&jobs, &waiting) {
                    if jobs[w].customer.priority < jobs[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let remaining = a.finish - now;
                        let by = jobs[w].customer.id;
                        let paused_id = jobs[a.index].customer.id;
                        emit(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt { remaining, by },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch) as u128;
                        let old_deadline = match jobs[a.index].deadline {
                            Some(d) => {
                                let spent = now - jobs[a.index].queue_since;
                                let budget = d - jobs[a.index].queue_since;
                                let left = budget.saturating_sub(spent);
                                Some(now.checked_add(left).ok_or_else(|| {
                                    time_overflow(
                                        paused_id,
                                        now,
                                        OverflowStage::Patience,
                                    )
                                })?)
                            }
                            None => None,
                        };
                        jobs[a.index].remaining = remaining;
                        jobs[a.index].queue_since = now;
                        jobs[a.index].pause_time = Some(now);
                        jobs[a.index].deadline = old_deadline;
                        waiting.push(a.index);
                    }
                }
            }
        }

        // Phase E: dispatch as many inline completions permit.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&jobs, &waiting) else {
                break;
            };
            waiting.retain(|&x| x != i);

            let segment = now - jobs[i].queue_since;
            jobs[i].waited = jobs[i].waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;
            if let Some(d) = jobs[i].deadline {
                let remaining_budget = d.saturating_sub(now);
                jobs[i].deadline = Some(now.saturating_add(remaining_budget));
            }

            let id = jobs[i].customer.id;
            let switch = if last_dispatch_id.is_some_and(|prev| prev != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now
                .checked_add(switch)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::WorkAt))?;
            let finish = work_at
                .checked_add(jobs[i].remaining)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::Finish))?;

            let first = !jobs[i].started;
            if first {
                jobs[i].started = true;
                stats.started += 1;
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Start {
                        remaining: jobs[i].remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Resume {
                        remaining: jobs[i].remaining,
                        work_at,
                        finish,
                    },
                );
            }
            last_dispatch_id = Some(id);
            active = Some(Active {
                index: i,
                work_at,
                finish,
                dispatch: now,
            });

            if finish == now {
                let a = active.take().unwrap();
                emit(&mut events, now, id, Kind::Finish);
                jobs[i].terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - jobs[i].customer.arrival) as u128;
                stats.completion_order.push(id);
                // This interval has zero duration.
                let _ = a;
            } else {
                break;
            }
        }

        // Phase F: expire newly queued waiters after all inline dispatches.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            jobs[a]
                .deadline
                .cmp(&jobs[b].deadline)
                .then_with(|| jobs[a].customer.id.cmp(&jobs[b].customer.id))
        });
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_since;
                jobs[i].waited = jobs[i].waited.saturating_add(segment);
                stats.wait_ticks += segment as u128;
                emit(
                    &mut events,
                    now,
                    jobs[i].customer.id,
                    Kind::Renege { waited: jobs[i].waited },
                );
                jobs[i].terminal = true;
                stats.reneged += 1;
            }
        }

        if arrival_pos == arrivals.len() && waiting.is_empty() && active.is_none() {
            break;
        }

        let mut next: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next.map_or(true, |v| t < v) {
                next = Some(t);
            }
        };
        if arrival_pos < arrivals.len() {
            consider(jobs[arrivals[arrival_pos]].customer.arrival);
        }
        if let Some(a) = active.as_ref() {
            consider(a.finish);
            if now < a.work_at {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].deadline {
                consider(d);
            }
        }
        let Some(t) = next else {
            break;
        };
        now = t;
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(
            stats.turnaround_ticks * 1000,
            stats.completed as u128,
        )
        .unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
