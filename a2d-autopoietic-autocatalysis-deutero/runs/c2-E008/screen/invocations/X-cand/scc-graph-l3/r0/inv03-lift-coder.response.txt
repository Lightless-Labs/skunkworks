pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }

    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative Kosaraju traversal avoids recursion on long paths.
    let mut seen = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }

        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.pop() {
            if next_index < adjacency[vertex].len() {
                stack.push((vertex, next_index + 1));
                let next = adjacency[vertex][next_index];
                if !seen[next] {
                    seen[next] = true;
                    stack.push((next, 0));
                }
            } else {
                finish_order.push(vertex);
            }
        }
    }

    let mut assigned = vec![false; n];
    let mut components = Vec::new();
    for &start in finish_order.iter().rev() {
        if assigned[start] {
            continue;
        }

        assigned[start] = true;
        let mut members = Vec::new();
        let mut stack = vec![start];
        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &next in &reverse[vertex] {
                if !assigned[next] {
                    assigned[next] = true;
                    stack.push(next);
                }
            }
        }
        members.sort_unstable();
        components.push(members);
    }

    components.sort_by(|a, b| {
        b.len()
            .cmp(&a.len())
            .then_with(|| a[0].cmp(&b[0]))
    });

    let component_count = components.len();
    let mut component_of = vec![0usize; n];
    for (id, members) in components.iter().enumerate() {
        for &vertex in members {
            component_of[vertex] = id;
        }
    }

    let mut cyclic: Vec<bool> = components.iter().map(|c| c.len() > 1).collect();
    let mut internal_edges = vec![0usize; component_count];
    let mut grouped_arcs =
        std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let source = component_of[from];
        let target = component_of[to];

        if source == target {
            internal_edges[source] += 1;
            if from == to {
                cyclic[source] = true;
            }
        } else {
            let entry = grouped_arcs
                .entry((source, target))
                .or_insert((0, edge_index));
            entry.0 += 1;
        }
    }

    let mut arcs: Vec<Arc> = grouped_arcs
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); component_count];
    let mut indegree = vec![0usize; component_count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    // Component minima are distinct, so the third tuple field is immaterial
    // after applying the required priority keys.
    let mut ready = std::collections::BinaryHeap::new();
    for id in 0..component_count {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], id));
        }
    }

    let mut topo = Vec::with_capacity(component_count);
    let mut depth = vec![0usize; component_count];
    while let Some((_, _, id)) = ready.pop() {
        topo.push(id);
        for &next in &outgoing[id] {
            depth[next] = depth[next].max(depth[id] + 1);
            indegree[next] -= 1;
            if indegree[next] == 0 {
                ready.push((internal_edges[next], components[next][0], next));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let vertex_count = self.component_of.len();
        if from >= vertex_count {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= vertex_count {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let component_count = self.components.len();
        let mut outgoing = vec![Vec::new(); component_count];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }

        if let Some(route) = shortest_lex_route(a, b, &outgoing) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_route(b, a, &outgoing) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }

        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn shortest_lex_route(
    start: usize,
    goal: usize,
    outgoing: &[Vec<usize>],
) -> Option<Vec<usize>> {
    let count = outgoing.len();
    let mut reverse = vec![Vec::new(); count];
    for (from, neighbors) in outgoing.iter().enumerate() {
        for &to in neighbors {
            reverse[to].push(from);
        }
    }

    let mut distance = vec![usize::MAX; count];
    let mut queue = std::collections::VecDeque::new();
    distance[goal] = 0;
    queue.push_back(goal);

    while let Some(vertex) = queue.pop_front() {
        for &previous in &reverse[vertex] {
            if distance[previous] == usize::MAX {
                distance[previous] = distance[vertex] + 1;
                queue.push_back(previous);
            }
        }
    }

    if distance[start] == usize::MAX {
        return None;
    }

    let mut route = vec![start];
    let mut current = start;
    while current != goal {
        let next = outgoing[current]
            .iter()
            .copied()
            .filter(|&vertex| {
                distance[vertex] != usize::MAX
                    && distance[vertex] + 1 == distance[current]
            })
            .min()?;
        route.push(next);
        current = next;
    }

    Some(route)
}
