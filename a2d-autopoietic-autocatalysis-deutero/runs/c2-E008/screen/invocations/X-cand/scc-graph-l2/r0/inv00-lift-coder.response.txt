pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }

    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (index, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::<usize>::new(); n];
    let mut reverse = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative DFS finishing order avoids recursion on long directed paths.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);
    for start in 0..n {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.last_mut() {
            if *next_index < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_index];
                *next_index += 1;
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                finish_order.push(finished);
            }
        }
    }

    let mut raw_components = Vec::<Vec<usize>>::new();
    let mut raw_component_of = vec![usize::MAX; n];
    for &start in finish_order.iter().rev() {
        if raw_component_of[start] != usize::MAX {
            continue;
        }

        let component_id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        raw_component_of[start] = component_id;

        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &neighbor in &reverse[vertex] {
                if raw_component_of[neighbor] == usize::MAX {
                    raw_component_of[neighbor] = component_id;
                    stack.push(neighbor);
                }
            }
        }

        members.sort_unstable();
        raw_components.push(members);
    }

    let mut component_order: Vec<usize> = (0..raw_components.len()).collect();
    component_order.sort_unstable_by(|&a, &b| {
        raw_components[b]
            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
    });

    let mut raw_to_canonical = vec![0usize; raw_components.len()];
    let mut components = Vec::with_capacity(raw_components.len());
    for (canonical_id, &raw_id) in component_order.iter().enumerate() {
        raw_to_canonical[raw_id] = canonical_id;
        components.push(raw_components[raw_id].clone());
    }

    let component_of: Vec<usize> = raw_component_of
        .iter()
        .map(|&raw_id| raw_to_canonical[raw_id])
        .collect();

    let component_count = components.len();
    let mut cyclic: Vec<bool> = components
        .iter()
        .map(|members| members.len() > 1)
        .collect();
    let mut internal_edges = vec![0usize; component_count];
    let mut arc_counts = vec![0usize; component_count * component_count];
    let mut arc_first = vec![usize::MAX; component_count * component_count];

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let from_component = component_of[from];
        let to_component = component_of[to];

        if from_component == to_component {
            internal_edges[from_component] += 1;
            if from == to {
                cyclic[from_component] = true;
            }
        } else {
            let key = from_component * component_count + to_component;
            arc_counts[key] += 1;
            if edge_index < arc_first[key] {
                arc_first[key] = edge_index;
            }
        }
    }

    let mut arcs = Vec::new();
    for from in 0..component_count {
        for to in (0..component_count).rev() {
            let key = from * component_count + to;
            if arc_counts[key] != 0 {
                arcs.push(Arc {
                    from,
                    to,
                    multiplicity: arc_counts[key],
                    first_edge: arc_first[key],
                });
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
