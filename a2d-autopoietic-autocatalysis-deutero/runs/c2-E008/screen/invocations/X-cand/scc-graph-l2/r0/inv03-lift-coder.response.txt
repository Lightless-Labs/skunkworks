pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::<usize>::new(); n];
    let mut reverse = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative Kosaraju first pass.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);
    for start in 0..n {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.pop() {
            if next_index < adjacency[vertex].len() {
                stack.push((vertex, next_index + 1));
                let next = adjacency[vertex][next_index];
                if !visited[next] {
                    visited[next] = true;
                    stack.push((next, 0));
                }
            } else {
                finish_order.push(vertex);
            }
        }
    }

    // Second pass assigns preliminary component IDs.
    let mut preliminary_of = vec![usize::MAX; n];
    let mut preliminary_components = Vec::<Vec<usize>>::new();
    for &start in finish_order.iter().rev() {
        if preliminary_of[start] != usize::MAX {
            continue;
        }

        let id = preliminary_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        preliminary_of[start] = id;

        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &next in &reverse[vertex] {
                if preliminary_of[next] == usize::MAX {
                    preliminary_of[next] = id;
                    stack.push(next);
                }
            }
        }

        members.sort_unstable();
        preliminary_components.push(members);
    }

    let mut order: Vec<usize> = (0..preliminary_components.len()).collect();
    order.sort_unstable_by(|&a, &b| {
        preliminary_components[b]
            .len()
            .cmp(&preliminary_components[a].len())
            .then_with(|| preliminary_components[a][0].cmp(&preliminary_components[b][0]))
    });

    let mut remap = vec![0usize; order.len()];
    let mut components = Vec::with_capacity(order.len());
    for (canonical_id, &preliminary_id) in order.iter().enumerate() {
        remap[preliminary_id] = canonical_id;
        components.push(preliminary_components[preliminary_id].clone());
    }

    let component_of: Vec<usize> = preliminary_of
        .iter()
        .map(|&id| remap[id])
        .collect();

    let component_count = components.len();
    let mut cyclic: Vec<bool> = components
        .iter()
        .map(|members| members.len() >= 2)
        .collect();
    let mut internal_edges = vec![0usize; component_count];

    // Key each cross-component pair directly; this is bounded by the input edge count.
    let mut pair_data = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let from_component = component_of[from];
        let to_component = component_of[to];

        if from_component == to_component {
            internal_edges[from_component] += 1;
            if from == to {
                cyclic[from_component] = true;
            }
        } else {
            let entry = pair_data
                .entry((from_component, to_component))
                .or_insert((0, edge_index));
            entry.0 += 1;
            entry.1 = entry.1.min(edge_index);
        }
    }

    let mut arcs: Vec<Arc> = pair_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_unstable_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
