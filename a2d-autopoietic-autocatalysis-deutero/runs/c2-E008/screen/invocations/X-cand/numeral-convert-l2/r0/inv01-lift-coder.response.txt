#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input[..hash] != expected {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.is_empty() {
        return Err(BaseError::Separator {
            operand,
            index: payload_start,
        });
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn accumulate(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        let mut carry = digit as u16;
        for limb in &mut value {
            let current = *limb as u16 * radix as u16 + carry;
            *limb = (current % 1_000_000_000) as u8;
            carry = current / 1_000_000_000;
        }
        while carry != 0 {
            value.push((carry % 1_000_000_000) as u8);
            carry /= 1_000_000_000;
        }
    }
    value
}

// Big integers are stored little-endian in base 1,000,000,000.
// This multiplication routine is only used with small radix values.
fn accumulate_big(digits: &[u8], radix: u8) -> Vec<u32> {
    let mut value = vec![0u32];
    for &digit in digits {
        let mut carry = digit as u64;
        for limb in &mut value {
            let current = *limb as u64 * radix as u64 + carry;
            *limb = (current % 1_000_000_000) as u32;
            carry = current / 1_000_000_000;
        }
        while carry != 0 {
            value.push((carry % 1_000_000_000) as u32);
            carry /= 1_000_000_000;
        }
    }
    value
}

fn add_big(left: &[u32], right: &[u32]) -> Vec<u32> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for i in 0..left.len().max(right.len()) {
        let sum = left.get(i).copied().unwrap_or(0) as u64
            + right.get(i).copied().unwrap_or(0) as u64
            + carry;
        result.push((sum % 1_000_000_000) as u32);
        carry = sum / 1_000_000_000;
    }
    if carry != 0 {
        result.push(carry as u32);
    }
    result
}

fn big_to_radix(mut value: Vec<u32>, radix: u8) -> Vec<u8> {
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    if value.iter().all(|&x| x == 0) {
        return vec![0];
    }

    let mut digits = Vec::new();
    while value.iter().any(|&x| x != 0) {
        let mut remainder = 0u64;
        for limb in value.iter_mut().rev() {
            let current = remainder * 1_000_000_000 + *limb as u64;
            *limb = (current / radix as u64) as u32;
            remainder = current % radix as u64;
        }
        digits.push(remainder as u8);
        while value.len() > 1 && value.last() == Some(&0) {
            value.pop();
        }
    }
    digits.reverse();
    digits
}

fn format_digits(mut digits: Vec<u8>, style: BaseStyle) -> String {
    if digits.len() < style.min_width {
        let mut padded = vec![0; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::new();
    for (i, digit) in digits.iter().enumerate() {
        if style.group != 0 && i != 0 && (digits.len() - i) % style.group == 0 {
            output.push('_');
        }
        let ch = match *digit {
            0..=9 => (b'0' + *digit) as char,
            value => {
                let base = if style.upper { b'A' } else { b'a' };
                (base + value - 10) as char
            }
        };
        output.push(ch);
    }
    output
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    let value = accumulate_big(&digits, from);
    Ok(format_digits(big_to_radix(value, to), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = accumulate_big(&left_digits, base);
    let right_value = accumulate_big(&right_digits, base);
    Ok(format_digits(big_to_radix(add_big(&left_value, &right_value), base), style))
}
