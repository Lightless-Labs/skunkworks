#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

const DECIMAL_LIMB_BASE: u32 = 1_000_000_000;

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let payload_start = if let Some(hash) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    // Check every payload byte for invalid digits before checking separators.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }

        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.is_empty() {
        let offset = payload.iter().position(|&byte| byte == b'_').unwrap_or(0);
        return Err(BaseError::Separator {
            operand,
            index: payload_start + offset,
        });
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Store the value as little-endian base-1,000,000,000 limbs.
fn parse_value(digits: &[u8], radix: u8) -> Vec<u32> {
    let mut value = vec![0u32];

    for &digit in digits {
        let mut carry = digit as u64;

        for limb in &mut value {
            let current = *limb as u64 * radix as u64 + carry;
            *limb = (current % DECIMAL_LIMB_BASE as u64) as u32;
            carry = current / DECIMAL_LIMB_BASE as u64;
        }

        while carry != 0 {
            value.push((carry % DECIMAL_LIMB_BASE as u64) as u32);
            carry /= DECIMAL_LIMB_BASE as u64;
        }
    }

    value
}

fn add_values(left: &[u32], right: &[u32]) -> Vec<u32> {
    let length = left.len().max(right.len());
    let mut result = Vec::with_capacity(length + 1);
    let mut carry = 0u64;

    for index in 0..length {
        let sum = left.get(index).copied().unwrap_or(0) as u64
            + right.get(index).copied().unwrap_or(0) as u64
            + carry;
        result.push((sum % DECIMAL_LIMB_BASE as u64) as u32);
        carry = sum / DECIMAL_LIMB_BASE as u64;
    }

    if carry != 0 {
        result.push(carry as u32);
    }

    result
}

fn divide_value(value: &mut Vec<u32>, divisor: u8) -> u8 {
    let mut remainder = 0u64;

    for limb in value.iter_mut().rev() {
        let current = remainder * DECIMAL_LIMB_BASE as u64 + *limb as u64;
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }

    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }

    remainder as u8
}

fn format_value(mut value: Vec<u32>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();

    if value.iter().all(|&limb| limb == 0) {
        digits.push(0);
    } else {
        while value.iter().any(|&limb| limb != 0) {
            digits.push(divide_value(&mut value, radix));
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let padding = style.min_width - digits.len();
        let mut padded = vec![0u8; padding];
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::new();
    for (index, &digit) in digits.iter().enumerate() {
        if style.group != 0
            && index != 0
            && (digits.len() - index) % style.group == 0
        {
            output.push('_');
        }

        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(byte as char);
    }

    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(format_value(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let left_value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    Ok(format_value(add_values(&left_value, &right_value), base, style))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        if &bytes[..hash] != base.to_string().as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < base => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.is_empty() {
        let offset = payload.iter().position(|&b| b == b'_').unwrap_or(0);
        return Err(BaseError::Separator {
            operand,
            index: payload_start + offset,
        });
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u32;
    for digit in value.iter_mut().rev() {
        let n = *digit as u32 * multiplier as u32 + carry;
        *digit = (n % 10) as u8;
        carry = n / 10;
    }
    while carry > 0 {
        value.insert(0, (carry % 10) as u8);
        carry /= 10;
    }
}

fn add_decimal(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry > 0 {
        let a = if i > 0 {
            i -= 1;
            left[i] as u16
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            right[j] as u16
        } else {
            0
        };
        let sum = a + b + carry;
        result.push((sum % 10) as u8);
        carry = sum / 10;
    }

    result.reverse();
    result
}

fn to_decimal(digits: &[u8], base: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_add(&mut value, base, digit);
    }
    value
}

fn from_decimal(mut value: Vec<u8>, base: u8) -> Vec<u8> {
    if value.iter().all(|&d| d == 0) {
        return vec![0];
    }

    let mut output = Vec::new();
    while value.iter().any(|&d| d != 0) {
        let mut quotient = Vec::with_capacity(value.len());
        let mut remainder = 0u16;
        for digit in value {
            let n = remainder * 10 + digit as u16;
            let q = n / base as u16;
            remainder = n % base as u16;
            if !quotient.is_empty() || q != 0 {
                quotient.push(q as u8);
            }
        }
        output.push(remainder as u8);
        value = quotient;
    }
    output.reverse();
    output
}

fn format_digits(mut digits: Vec<u8>, style: BaseStyle) -> String {
    if digits.is_empty() {
        digits.push(0);
    }
    if digits.len() < style.min_width {
        let mut padded = vec![0; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::new();
    for (index, digit) in digits.iter().enumerate() {
        if style.group != 0 && index > 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        let ch = if *digit < 10 {
            (b'0' + *digit) as char
        } else if style.upper {
            (b'A' + (*digit - 10)) as char
        } else {
            (b'a' + (*digit - 10)) as char
        };
        output.push(ch);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let decimal = to_decimal(&digits, from);
    let output = from_decimal(decimal, to);
    Ok(format_digits(output, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_decimal = to_decimal(&left_digits, base);
    let right_decimal = to_decimal(&right_digits, base);
    let sum = add_decimal(&left_decimal, &right_decimal);
    let output = from_decimal(sum, base);
    Ok(format_digits(output, style))
}
