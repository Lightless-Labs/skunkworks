#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = base.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = tag_end + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < base => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, base: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in value.iter_mut() {
        let n = (*limb as u16) * (base as u16) + carry;
        *limb = (n % 1_000_000_000) as u8;
        carry = n / 1_000_000_000;
    }
    while carry != 0 {
        value.push((carry % 1_000_000_000) as u8);
        carry /= 1_000_000_000;
    }
}

// Limbs are little-endian base 1e9, stored as u32 values.
fn parse_value(digits: &[u8], base: u8) -> Vec<u32> {
    let mut value: Vec<u32> = vec![0];
    for &digit in digits {
        let mut carry = digit as u64;
        for limb in value.iter_mut() {
            let n = (*limb as u64) * (base as u64) + carry;
            *limb = (n % 1_000_000_000) as u32;
            carry = n / 1_000_000_000;
        }
        while carry != 0 {
            value.push((carry % 1_000_000_000) as u32);
            carry /= 1_000_000_000;
        }
    }
    value
}

fn add_values(left: &[u32], right: &[u32]) -> Vec<u32> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for index in 0..left.len().max(right.len()) {
        let sum = left.get(index).copied().unwrap_or(0) as u64
            + right.get(index).copied().unwrap_or(0) as u64
            + carry;
        result.push((sum % 1_000_000_000) as u32);
        carry = sum / 1_000_000_000;
    }
    if carry != 0 {
        result.push(carry as u32);
    }
    result
}

fn divide_value(value: &mut Vec<u32>, divisor: u8) -> u8 {
    let mut remainder = 0u64;
    for limb in value.iter_mut().rev() {
        let n = remainder * 1_000_000_000 + *limb as u64;
        *limb = (n / divisor as u64) as u32;
        remainder = n % divisor as u64;
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn format_value(mut value: Vec<u32>, base: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if value.iter().all(|&limb| limb == 0) {
        digits.push(0u8);
    } else {
        while value.iter().any(|&limb| limb != 0) {
            digits.push(divide_value(&mut value, base));
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let padding = style.min_width - digits.len();
        let mut padded = vec![0u8; padding];
        padded.extend(digits);
        digits = padded;
    }

    let mut rendered = Vec::with_capacity(digits.len());
    for digit in digits {
        rendered.push(match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        });
    }

    let mut output = String::new();
    for (index, &byte) in rendered.iter().enumerate() {
        if style.group != 0
            && index != 0
            && (rendered.len() - index) % style.group == 0
        {
            output.push('_');
        }
        output.push(byte as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(format_value(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    Ok(format_value(add_values(&left_value, &right_value), base, style))
}
