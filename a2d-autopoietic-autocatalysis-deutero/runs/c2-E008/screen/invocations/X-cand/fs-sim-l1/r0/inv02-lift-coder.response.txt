```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

#[derive(Debug)]
pub struct Fs {
    entries: std::collections::HashMap<String, Entry>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::HashMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self { entries }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let bytes = path.as_bytes();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                let component = &path[start..i];
                if component.is_empty() || component == "." || component == ".." {
                    let offset = if component.is_empty() {
                        if i == bytes.len() { bytes.len() - 1 } else { i }
                    } else {
                        start
                    };
                    return Err(Error::InvalidPath(offset));
                }
                components.push(component);
                start = i + 1;
            }
        }
        Ok(components)
    }

    fn canonical(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn lookup_parent(
        &self,
        components: &[&str],
    ) -> Result<(String, String), Error> {
        let parent_components = &components[..components.len() - 1];
        let mut prefix = String::from("/");
        for (i, component) in parent_components.iter().enumerate() {
            if prefix != "/" {
                prefix.push('/');
            }
            prefix.push_str(component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
                Some(Entry::Dir) => {}
            }
            let _ = i;
        }
        let parent = Self::canonical(parent_components);
        let mut final_path = parent.clone();
        if final_path != "/" {
            final_path.push('/');
        }
        final_path.push_str(components[components.len() - 1]);
        Ok((parent, final_path))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }
        let (_, final_path) = self.lookup_parent(&components)?;
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }
        let (_, final_path) = self.lookup_parent(&components)?;
        match self.entries.get_mut(&final_path) {
            Some(Entry::Dir) => Err(Error::IsDir(final_path)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(final_path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }
        let (_, final_path) = self.lookup_parent(&components)?;
        match self.entries.get(&final_path) {
            None => Err(Error::Missing(final_path)),
            Some(Entry::Dir) => Err(Error::IsDir(final_path)),
            Some(Entry::File(contents)) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let directory = Self::canonical(&components);

        if components.is_empty() {
            // The root is created by new and is always a directory.
        } else {
            let (_, final_path) = self.lookup_parent(&components)?;
            match self.entries.get(&final_path) {
                None => return Err(Error::Missing(final_path)),
                Some(Entry::File(_)) => return Err(Error::NotDir(final_path)),
                Some(Entry::Dir) => {}
            }
        }

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }
            let Some(rest) = entry_path.strip_prefix(&directory) else {
                continue;
            };
            let Some(basename) = rest.strip_prefix('/') else {
                continue;
            };
            if basename.is_empty() || basename.contains('/') {
                continue;
            }
            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
            };
            children.push((basename.to_owned(), kind));
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }
}
```
