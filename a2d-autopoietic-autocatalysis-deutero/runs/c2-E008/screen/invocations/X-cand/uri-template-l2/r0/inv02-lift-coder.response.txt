#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression<'a> {
    operator: Option<u8>,
    variables: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    // Validate the complete template before looking up any binding.
    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open = pos;
                let close = match bytes[pos + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => pos + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[pos + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: pos + 1 + offset });
                }

                let mut content_start = pos + 1;
                let operator = match bytes.get(content_start).copied() {
                    Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
                        let op = bytes[content_start];
                        content_start += 1;
                        if op != b'+' && op != b'#' {
                            return Err(Error::UnsupportedOperator { pos: pos + 1 });
                        }
                        Some(op)
                    }
                    _ => None,
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut variables = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut item_start = content_start;

                loop {
                    let item_end = bytes[item_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| item_start + offset)
                        .unwrap_or(close);

                    if item_start == item_end {
                        return Err(Error::EmptyVariable { pos: item_start });
                    }

                    let name_bytes = &bytes[item_start..item_end];
                    if !is_name_start(name_bytes[0]) {
                        return Err(Error::InvalidName { pos: item_start });
                    }
                    for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
                        if !is_name_continue(byte) {
                            return Err(Error::InvalidName { pos: item_start + offset });
                        }
                    }

                    let name = &template[item_start..item_end];
                    if seen.iter().any(|&existing| existing == name) {
                        return Err(Error::DuplicateVariable { pos: item_start });
                    }
                    seen.push(name);
                    variables.push(name);

                    if item_end == close {
                        break;
                    }
                    item_start = item_end + 1;
                }

                expressions.push(Expression { operator, variables });
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    let mut expression_index = 0;

    while cursor < bytes.len() {
        if bytes[cursor] == b'{' {
            let close = cursor + 1 + bytes[cursor + 1..]
                .iter()
                .position(|&b| b == b'}')
                .expect("template was validated");
            output.push_str(&template[cursor..cursor + 1]);

            let expression = &expressions[expression_index];
            expression_index += 1;

            let mut fields = Vec::new();
            for &name in &expression.variables {
                if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                    fields.push(encode_value(value, expression.operator.is_some()));
                }
            }

            if !fields.is_empty() {
                if expression.operator == Some(b'#') {
                    output.push('#');
                }
                for (index, field) in fields.iter().enumerate() {
                    if index > 0 {
                        output.push(',');
                    }
                    output.push_str(field);
                }
            }

            cursor = close + 1;
        } else {
            let next_open = bytes[cursor..]
                .iter()
                .position(|&b| b == b'{')
                .map(|offset| cursor + offset)
                .unwrap_or(bytes.len());
            output.push_str(&template[cursor..next_open]);
            cursor = next_open;
        }
    }

    Ok(output)
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut encoded = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == b'%'
            && pos + 2 < bytes.len()
            && bytes[pos + 1].is_ascii_hexdigit()
            && bytes[pos + 2].is_ascii_hexdigit()
        {
            encoded.push('%');
            encoded.push((bytes[pos + 1] as char).to_ascii_uppercase());
            encoded.push((bytes[pos + 2] as char).to_ascii_uppercase());
            pos += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            encoded.push(byte as char);
        } else {
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
        pos += 1;
    }

    encoded
}

const HEX: &[u8; 16] = b"0123456789ABCDEF";

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
