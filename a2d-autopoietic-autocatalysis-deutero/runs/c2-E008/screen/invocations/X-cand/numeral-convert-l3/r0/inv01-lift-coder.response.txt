#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1_i128 << 120) - 1;

fn base(system: &SignedSystem) -> i128 {
    match system {
        SignedSystem::BalancedTernary => 3,
        SignedSystem::Negabinary => -2,
    }
}

fn checksum_digit(payload: &[u8]) -> u8 {
    let sum = payload.iter().enumerate().fold(0_u64, |sum, (index, &byte)| {
        let code = match byte {
            b'T' => 2_u64,
            b'1' => 1_u64,
            _ => 0_u64,
        };
        (sum + ((index as u64 + 1) * code) % 7) % 7
    });
    sum as u8
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    if !(-LIMIT..=LIMIT).contains(&value) {
        return Err(SignedError::OutOfRange);
    }

    let mut digits = Vec::new();
    if value == 0 {
        digits.push(b'0');
    } else {
        let mut current = value;
        match system {
            SignedSystem::BalancedTernary => {
                while current != 0 {
                    let mut remainder = current % 3;
                    current /= 3;
                    if remainder == 2 {
                        remainder = -1;
                        current += 1;
                    } else if remainder == -2 {
                        remainder = 1;
                        current -= 1;
                    }
                    digits.push(match remainder {
                        -1 => b'T',
                        0 => b'0',
                        _ => b'1',
                    });
                }
            }
            SignedSystem::Negabinary => {
                while current != 0 {
                    let remainder = current.rem_euclid(2);
                    digits.push(if remainder == 0 { b'0' } else { b'1' });
                    current = (current - remainder) / -2;
                }
            }
        }
        digits.reverse();
    }

    let mut result = String::from_utf8(digits.clone()).expect("digit alphabet is ASCII");
    if checksum {
        result.push(':');
        result.push(char::from(b'0' + checksum_digit(&digits)));
    }
    Ok(result)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    if input.is_empty() {
        return Err(SignedError::Empty);
    }

    let bytes = input.as_bytes();
    let colon = bytes.iter().position(|&byte| byte == b':');
    let payload_end = colon.unwrap_or(bytes.len());

    if payload_end == 0 {
        return Err(SignedError::Empty);
    }

    for (index, &byte) in bytes[..payload_end].iter().enumerate() {
        let valid = match system {
            SignedSystem::BalancedTernary => matches!(byte, b'T' | b'0' | b'1'),
            SignedSystem::Negabinary => matches!(byte, b'0' | b'1'),
        };
        if !valid {
            return Err(SignedError::InvalidByte { index });
        }
    }

    if let Some(colon_index) = colon {
        if bytes.len() != colon_index + 2 || !matches!(bytes[colon_index + 1], b'0'..=b'6') {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
    }

    if payload_end > 1 && bytes[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let radix = base(&system);
    let mut value = 0_i128;

    for (index, &byte) in bytes[..payload_end].iter().enumerate() {
        let digit = match byte {
            b'T' => -1_i128,
            b'1' => 1_i128,
            _ => 0_i128,
        };

        value = value
            .checked_mul(radix)
            .and_then(|value| value.checked_add(digit))
            .ok_or(SignedError::Range { index })?;

        if !(-LIMIT..=LIMIT).contains(&value) {
            return Err(SignedError::Range { index });
        }
    }

    if let Some(colon_index) = colon {
        let expected = checksum_digit(&bytes[..payload_end]);
        let found = bytes[colon_index + 1] - b'0';
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(value)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, checksum)
}
