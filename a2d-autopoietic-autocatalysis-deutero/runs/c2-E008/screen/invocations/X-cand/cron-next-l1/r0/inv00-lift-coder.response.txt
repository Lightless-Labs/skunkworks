#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn error(field: Field, offset: usize, make: fn(Field, usize) -> ParseError) -> ParseError {
    make(field, offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    let number = &bytes[start..end];

    if number.len() > 1 && number[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in number {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    if is_step {
        let (min, max) = domain(field);
        let width = (max - min + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else {
        let (min, max) = domain(field);
        if value < min as u32 || value > max as u32 {
            return Err(ParseError::OutOfRange { field, offset: start });
        }
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let exact_star = text == "*";
    let (minimum, maximum) = domain(field);
    let mut selected = vec![false; 256];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base_end, step_range) = if let Some(&slash) = slash_positions.first() {
            (slash, Some((slash + 1, item.len())))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            0
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen == 0
                || hyphen + 1 == base.len()
                || base[..hyphen].contains(&b'*')
                || base[hyphen + 1..].contains(&b'*')
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            2
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            1
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        if let Some((step_start, step_end)) = step_range {
            if step_start == step_end
                || !item[step_start..step_end]
                    .iter()
                    .all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let (start, end) = match base_kind {
            0 => (minimum as u32, maximum as u32),
            1 => {
                let value = parse_number(
                    base,
                    item_start,
                    item_start + base.len(),
                    field,
                    false,
                )?;
                (value, if step_range.is_some() { maximum as u32 } else { value })
            }
            _ => {
                let hyphen = hyphens[0];
                let start = parse_number(
                    base,
                    0,
                    hyphen,
                    field,
                    false,
                )
                .map_err(|e| shift_error(e, item_start))?;
                let end = parse_number(
                    base,
                    hyphen + 1,
                    base.len(),
                    field,
                    false,
                )
                .map_err(|e| shift_error(e, item_start))?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
                (start, end)
            }
        };

        let step = if let Some((step_start, step_end)) = step_range {
            parse_number(
                item,
                step_start,
                step_end,
                field,
                true,
            )
            .map_err(|e| shift_error(e, item_start))?
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok((
        selected
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect(),
        exact_star,
    ))
}

fn shift_error(error: ParseError, base: usize) -> ParseError {
    match error {
        ParseError::LeadingZero { field, offset } => ParseError::LeadingZero {
            field,
            offset: base + offset,
        },
        ParseError::NumberOverflow { field, offset } => ParseError::NumberOverflow {
            field,
            offset: base + offset,
        },
        ParseError::OutOfRange { field, offset } => ParseError::OutOfRange {
            field,
            offset: base + offset,
        },
        ParseError::BadStep { field, offset } => ParseError::BadStep {
            field,
            offset: base + offset,
        },
        other => other,
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_stars = [false; 2];

    for i in 0..5 {
        let (values, exact_star) = parse_field(pieces[i], fields[i])?;
        selected[i] = values;
        if i == 2 {
            day_stars[0] = exact_star;
        } else if i == 4 {
            day_stars[1] = exact_star;
        }
    }

    Ok(Cron {
        selected,
        day_stars,
    })
}
