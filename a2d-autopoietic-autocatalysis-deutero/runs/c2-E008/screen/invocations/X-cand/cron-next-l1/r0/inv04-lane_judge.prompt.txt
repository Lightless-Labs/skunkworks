fn parse_field(input: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    for (offset, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_domain(field);
    let width = (maximum - minimum + 1) as u32;
    let mut selected = vec![false; 256];
    let mut item_start = 0;

    for item in input.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        parse_item(item, item_start, field, minimum, maximum, width, &mut selected)?;
        item_start += item.len() + 1;
    }

    Ok((0u16..=255)
        .filter(|&value| selected[value as usize])
        .map(|value| value as u8)
        .collect())
}

fn parse_item(
    item: &str,
    item_offset: usize,
    field: Field,
    minimum: u8,
    maximum: u8,
    width: u32,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &byte)| (byte == b'/').then_some(i))
        .collect();

    let (base_end, step_start) = match slash_positions.as_slice() {
        [] => (bytes.len(), None),
        [slash] => (*slash, Some(*slash + 1)),
        _ => {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    };

    let base = &bytes[..base_end];
    if base.is_empty() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base_kind, start_bytes, end_bytes, end_offset) = if base == b"*" {
        (0u8, None, None, 0usize)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &byte)| (byte == b'-').then_some(i))
            .collect();

        match hyphens.as_slice() {
            [] if is_digits(base) => (1, Some(base), None, 0),
            [hyphen]
                if *hyphen > 0
                    && *hyphen + 1 < base.len()
                    && is_digits(&base[..*hyphen])
                    && is_digits(&base[*hyphen + 1..]) =>
            {
                (2, Some(&base[..*hyphen]), Some(&base[*hyphen + 1..]), *hyphen + 1)
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_offset,
                });
            }
        }
    };

    if step_start.is_some() && step_start == Some(bytes.len()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (start, end) = match base_kind {
        0 => (minimum, maximum),
        1 => {
            let number = parse_number(start_bytes.unwrap(), item_offset, field, false, width)?;
            check_domain(number.0, number.1, field)?;
            (number.0 as u8, number.0 as u8)
        }
        _ => {
            let first = parse_number(start_bytes.unwrap(), item_offset, field, false, width)?;
            check_domain(first.0, first.1, field)?;
            let end_bytes = end_bytes.unwrap();
            let end_offset = item_offset + end_offset;
            let second = parse_number(end_bytes, end_offset, field, false, width)?;
            check_domain(second.0, second.1, field)?;
            if first.0 > second.0 {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (first.0 as u8, second.0 as u8)
        }
    };

    let step = if let Some(step_offset) = step_start {
        let raw = &bytes[step_offset..];
        if !is_digits(raw) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        let (value, offset) = parse_number(raw, item_offset + step_offset, field, true, width)?;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset });
        }
        value
    } else {
        0
    };

    let selection_end = if step != 0 && base_kind == 1 {
        maximum
    } else {
        end
    };

    if step == 0 {
        for value in start..=end {
            selected[value as usize] = true;
        }
    } else {
        let mut value = start as u32;
        let limit = selection_end as u32;
        while value <= limit {
            selected[value as usize] = true;
            value += step;
        }
    }

    Ok(())
}

fn is_digits(bytes: &[u8]) -> bool {
    !bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit)
}

fn parse_number(
    bytes: &[u8],
    offset: usize,
    field: Field,
    is_step: bool,
    _width: u32,
) -> Result<(u32, usize), ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if is_step {
        Ok((value, offset))
    } else {
        Ok((value, offset))
    }
}

fn check_domain(value: u32, offset: usize, field: Field) -> Result<(), ParseError> {
    let (minimum, maximum) = field_domain(field);
    if value < minimum as u32 || value > maximum as u32 {
        Err(ParseError::OutOfRange { field, offset })
    } else {
        Ok(())
    }
}

    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,

fn parse_field(text: &str, field: Field, selected: &mut [bool]) -> Result<(), ParseError> {
    for (offset, byte) in text.bytes().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(error(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let bytes = text.as_bytes();
    let mut item_start = 0;
    for item_end in (0..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if item_start == item_end {
            return Err(error(field, item_start, ErrorKind::EmptyItem));
        }
        parse_item(&bytes[item_start..item_end], item_start, field, selected)?;
        item_start = item_end + 1;
    }
    Ok(())
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    let hyphen_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    if slash_positions.len() > 1 || hyphen_positions.len() > 1 {
        return Err(error(field, item_offset, ErrorKind::InvalidSyntax));
    }

    let base_end = slash_positions.first().copied().unwrap_or(item.len());
    let base = &item[..base_end];
    let step_bytes = slash_positions.first().map(|&i| &item[i + 1..]);

    let is_star = base == b"*";
    let is_number = !base.is_empty() && base.iter().all(u8::is_ascii_digit);
    let is_range = if let Some(&hyphen) = hyphen_positions.first() {
        hyphen < base_end
            && hyphen > 0
            && hyphen + 1 < base_end
            && base[..hyphen].iter().all(u8::is_ascii_digit)
            && base[hyphen + 1..].iter().all(u8::is_ascii_digit)
    } else {
        false
    };

    if !(is_star || is_number || is_range) {
        return Err(error(field, item_offset, ErrorKind::InvalidSyntax));
    }
    if step_bytes.is_some_and(|s| s.is_empty()) {
        return Err(error(field, item_offset, ErrorKind::InvalidSyntax));
    }

    let (start, end, end_offset) = if is_star {
        (field_min(field) as u32, field_max(field) as u32, item_offset)
    } else if is_number {
        let value = parse_number(base, item_offset, field, false)?;
        check_domain(value, item_offset, field)?;
        let end = if step_bytes.is_some() {
            field_max(field) as u32
        } else {
            value
        };
        (value, end, item_offset)
    } else {
        let hyphen = hyphen_positions[0];
        let start = parse_number(&base[..hyphen], item_offset, field, false)?;
        check_domain(start, item_offset, field)?;
        let end_pos = item_offset + hyphen + 1;
        let end = parse_number(&base[hyphen + 1..], end_pos, field, false)?;
        check_domain(end, end_pos, field)?;
        if start > end {
            return Err(error(field, end_pos, ErrorKind::ReversedRange));
        }
        (start, end, end_pos)
    };

    let step = if let Some(step_bytes) = step_bytes {
        let step_offset = item_offset + base_end + 1;
        let value = parse_number(step_bytes, step_offset, field, true)?;
        if value < 1 || value > field_width(field) {
            return Err(error(field, step_offset, ErrorKind::BadStep));
        }
        value
    } else {
        1
    };

    let _ = end_offset;
    let mut value = start;
    while value <= end {
        let index = (value - field_min(field) as u32) as usize;
        selected[index] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_number(
    bytes: &[u8],
    offset: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(error(field, offset, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| error(field, offset, ErrorKind::NumberOverflow))?;
    }

    if is_step && (value < 1 || value > field_width(field)) {
        return Err(error(field, offset, ErrorKind::BadStep));
    }
    Ok(value)
}

fn check_domain(value: u32, offset: usize, field: Field) -> Result<(), ParseError> {
    if value < field_min(field) as u32 || value > field_max(field) as u32 {
        return Err(error(field, offset, ErrorKind::OutOfRange));
    }
    Ok(())
}

        vec![false; field_width(field) as usize]
    });

    for (index, (text, field)) in parts.iter().zip(fields).enumerate() {
        parse_field(text, field, &mut selected[index])?;
    }

    Ok(Cron {
        selected,
        day_star_literal: [parts[2] == "*", parts[4] == "*"],
    })
}

fn parse_field(
    text: &str,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut item_start = 0;
    for item_end in (0..=bytes.len()).filter(|&index| {
        index == bytes.len() || bytes[index] == b','
    }) {
        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        parse_item(&bytes[item_start..item_end], item_start, field, selected)?;
        item_start = item_end + 1;
    }

    Ok(())
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'/').then_some(index))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let base_end = slash_positions.first().copied().unwrap_or(item.len());
    let base = &item[..base_end];
    let step_bytes = slash_positions.first().map(|&index| &item[index + 1..]);

    let hyphen_positions: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'-').then_some(index))
        .collect();

    let is_star = base == b"*";
    let is_number = !base.is_empty() && base.iter().all(u8::is_ascii_digit);
    let range_hyphen = match hyphen_positions.as_slice() {
        [index] => Some(*index),
        _ => None,
    };
    let is_range = range_hyphen.is_some_and(|hyphen| {
        hyphen > 0
            && hyphen + 1 < base.len()
            && base[..hyphen].iter().all(u8::is_ascii_digit)
            && base[hyphen + 1..].iter().all(u8::is_ascii_digit)
    });

    if !(is_star || is_number || (hyphen_positions.len() == 1 && is_range)) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    if step_bytes.is_some_and(|bytes| bytes.is_empty()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (start, end) = if is_star {
        (field_min(field) as u32, field_max(field) as u32)
    } else if is_number {
        let value = parse_number(base, item_offset, field)?;
        check_domain(value, item_offset, field)?;
        let end = if step_bytes.is_some() {
            field_max(field) as u32
        } else {
            value
        };
        (value, end)
    } else {
        let hyphen = range_hyphen.expect("validated range has one hyphen");
        let start = parse_number(&base[..hyphen], item_offset, field)?;
        check_domain(start, item_offset, field)?;

        let end_offset = item_offset + hyphen + 1;
        let end = parse_number(&base[hyphen + 1..], end_offset, field)?;
        check_domain(end, end_offset, field)?;

        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: end_offset,
            });
        }

        (start, end)
    };

    let step = if let Some(step_bytes) = step_bytes {
        let step_offset = item_offset + base_end + 1;
        let value = parse_number(step_bytes, step_offset, field)?;
        if value == 0 || value > field_width(field) {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - field_min(field) as u32) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_number(bytes: &[u8], offset: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|value| value.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    Ok(value)
}

fn check_domain(value: u32, offset: usize, field: Field) -> Result<(), ParseError> {
    if value < field_min(field) as u32 || value > field_max(field) as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(())
}
