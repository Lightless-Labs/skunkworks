```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_star_literal: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(offset, &is_selected)| {
                is_selected.then_some(field_min(field) + offset as u8)
            })
            .collect()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_min(field: Field) -> u8 {
    match field {
        Field::Minute | Field::Hour | Field::DayOfWeek => 0,
        Field::DayOfMonth | Field::Month => 1,
    }
}

fn field_max(field: Field) -> u8 {
    match field {
        Field::Minute => 59,
        Field::Hour => 23,
        Field::DayOfMonth => 31,
        Field::Month => 12,
        Field::DayOfWeek => 6,
    }
}

fn field_width(field: Field) -> u32 {
    (field_max(field) - field_min(field) + 1) as u32
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|index| {
        let field = fields[index];
        vec![false; field_width(field) as usize]
    });

    for (index, (text, field)) in parts.iter().zip(fields).enumerate() {
        parse_field(text, field, &mut selected[index])?;
    }

    Ok(Cron {
        selected,
        day_star_literal: [parts[2] == "*", parts[4] == "*"],
    })
}

fn parse_field(
    text: &str,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut item_start = 0;
    for item_end in (0..=bytes.len()).filter(|&index| {
        index == bytes.len() || bytes[index] == b','
    }) {
        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        parse_item(&bytes[item_start..item_end], item_start, field, selected)?;
        item_start = item_end + 1;
    }

    Ok(())
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'/').then_some(index))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let base_end = slash_positions.first().copied().unwrap_or(item.len());
    let base = &item[..base_end];
    let step_bytes = slash_positions.first().map(|&index| &item[index + 1..]);

    let hyphen_positions: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'-').then_some(index))
        .collect();

    let is_star = base == b"*";
    let is_number = !base.is_empty() && base.iter().all(u8::is_ascii_digit);
    let range_hyphen = match hyphen_positions.as_slice() {
        [index] => Some(*index),
        _ => None,
    };
    let is_range = range_hyphen.is_some_and(|hyphen| {
        hyphen > 0
            && hyphen + 1 < base.len()
            && base[..hyphen].iter().all(u8::is_ascii_digit)
            && base[hyphen + 1..].iter().all(u8::is_ascii_digit)
    });

    if !(is_star || is_number || (hyphen_positions.len() == 1 && is_range)) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    if step_bytes.is_some_and(|bytes| bytes.is_empty()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (start, end) = if is_star {
        (field_min(field) as u32, field_max(field) as u32)
    } else if is_number {
        let value = parse_number(base, item_offset, field)?;
        check_domain(value, item_offset, field)?;
        let end = if step_bytes.is_some() {
            field_max(field) as u32
        } else {
            value
        };
        (value, end)
    } else {
        let hyphen = range_hyphen.expect("validated range has one hyphen");
        let start = parse_number(&base[..hyphen], item_offset, field)?;
        check_domain(start, item_offset, field)?;

        let end_offset = item_offset + hyphen + 1;
        let end = parse_number(&base[hyphen + 1..], end_offset, field)?;
        check_domain(end, end_offset, field)?;

        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: end_offset,
            });
        }

        (start, end)
    };

    let step = if let Some(step_bytes) = step_bytes {
        let step_offset = item_offset + base_end + 1;
        let value = parse_number(step_bytes, step_offset, field)?;
        if value == 0 || value > field_width(field) {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - field_min(field) as u32) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_number(bytes: &[u8], offset: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|value| value.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    Ok(value)
}

fn check_domain(value: u32, offset: usize, field: Field) -> Result<(), ParseError> {
    if value < field_min(field) as u32 || value > field_max(field) as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(())
}
```
