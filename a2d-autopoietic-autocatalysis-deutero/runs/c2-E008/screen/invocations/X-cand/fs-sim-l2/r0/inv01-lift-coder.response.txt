```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
            cwd: Vec::new(),
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other.to_owned()),
            }
        }
        Ok(components)
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn lookup<'a>(root: &'a Entry, components: &[String]) -> Result<&'a Entry, Error> {
        let mut current = root;
        let mut prefix = Vec::new();

        for component in components {
            let path = Self::canonical(&prefix);
            let children = match current {
                Entry::Dir(children) => children,
                Entry::File(_) => return Err(Error::NotDir(path)),
            };

            prefix.push(component.clone());
            current = children
                .get(component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
        }

        Ok(current)
    }

    fn lookup_mut<'a>(
        root: &'a mut Entry,
        components: &[String],
    ) -> Result<&'a mut Entry, Error> {
        let mut current = root;
        let mut prefix = Vec::new();

        for component in components {
            let path = Self::canonical(&prefix);
            let children = match current {
                Entry::Dir(children) => children,
                Entry::File(_) => return Err(Error::NotDir(path)),
            };

            prefix.push(component.clone());
            current = children
                .get_mut(component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
        }

        Ok(current)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let (name, parent_components) = components.split_last().unwrap();
        let parent_path = Self::canonical(parent_components);

        let parent = Self::lookup_mut(&mut self.root, parent_components)?;
        let children = match parent {
            Entry::Dir(children) => children,
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
        };

        let final_path = Self::canonical(&components);
        if children.contains_key(name) {
            return Err(Error::Exists(final_path));
        }

        children.insert(name.clone(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let (name, parent_components) = components.split_last().unwrap();
        let parent_path = Self::canonical(parent_components);
        let final_path = Self::canonical(&components);

        let parent = Self::lookup_mut(&mut self.root, parent_components)?;
        let children = match parent {
            Entry::Dir(children) => children,
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
        };

        match children.get_mut(name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(final_path)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                children.insert(name.clone(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        match Self::lookup(&self.root, &components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical(&components))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        match Self::lookup(&self.root, &components)? {
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        match Self::lookup(&self.root, &components)? {
            Entry::Dir(_) => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
