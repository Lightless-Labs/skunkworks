#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes.last() != Some(&b'-')
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(required) => version == required,
        Constraint::AtLeast(required) => version >= required,
        Constraint::Before(required) => version < required,
        Constraint::Caret(required) => {
            version >= required
                && if required.major > 0 {
                    version.major == required.major
                } else if required.minor > 0 {
                    version.major == 0 && version.minor == required.minor
                } else {
                    version == required
                }
        }
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|&constraint| matches_constraint(version, constraint))
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(version) => format!("={}", version_text(version)),
        Constraint::AtLeast(version) => format!(">={}", version_text(version)),
        Constraint::Before(version) => format!("<{}", version_text(version)),
        Constraint::Caret(version) => format!("^{}", version_text(version)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|&constraint| constraint_text(constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut encoded = String::new();
    for byte in text.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            encoded.push(byte as char);
        } else {
            encoded.push('%');
            encoded.push_str(&format!("{:02X}", byte));
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_names(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{index}]"),
                name: root.package.clone(),
            });
        }
    }

    for (package_index, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{package_index}]"),
                name: package.name.clone(),
            });
        }
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!(
                            "catalog[{package_index}].release[{release_index}].dep[{dependency_index}]"
                        ),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }

    for (lock_index, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{lock_index}]"),
                name: lock.package.clone(),
            });
        }
    }

    Ok(())
}

type ConstraintMap = std::collections::BTreeMap<String, Vec<Constraint>>;
type AssignmentMap = std::collections::BTreeMap<String, Version>;
type CatalogMap<'a> = std::collections::BTreeMap<String, &'a Package>;
type LockMap = std::collections::BTreeMap<String, Version>;

fn active_constraints(
    roots: &[Requirement],
    catalog: &CatalogMap<'_>,
    assignments: &AssignmentMap,
) -> ConstraintMap {
    let mut active = ConstraintMap::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (package_name, version) in assignments {
        if let Some(package) = catalog.get(package_name) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for dependency in &release.deps {
                    active
                        .entry(dependency.package.clone())
                        .or_default()
                        .push(dependency.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }

    active
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: CatalogMap<'a>,
    locks: LockMap,
    attempts: usize,
    max_attempts: usize,
    best: Option<AssignmentMap>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn lock_cost(&self, assignments: &AssignmentMap) -> usize {
        self.locks
            .iter()
            .filter(|(package, version)| assignments.get(*package) != Some(*version))
            .count()
    }

    fn vector_is_better(&self, left: &AssignmentMap, right: &AssignmentMap) -> bool {
        for name in self.catalog.keys() {
            match (left.get(name), right.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(left_version), Some(right_version)) => {
                    match left_version.cmp(right_version) {
                        std::cmp::Ordering::Less => return false,
                        std::cmp::Ordering::Greater => return true,
                        std::cmp::Ordering::Equal => {}
                    }
                }
            }
        }
        false
    }

    fn consider_solution(&mut self, assignments: &AssignmentMap) {
        let cost = self.lock_cost(assignments);
        let replace = match &self.best {
            None => true,
            Some(best) if cost < self.best_cost => true,
            Some(best) if cost == self.best_cost => self.vector_is_better(assignments, best),
            _ => false,
        };

        if replace {
            self.best = Some(assignments.clone());
            self.best_cost = cost;
        }
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn visit(&mut self, assignments: &mut AssignmentMap) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, &self.catalog, assignments);

        if let Some(package) = active
            .keys()
            .find(|package| !self.catalog.contains_key(*package))
        {
            self.record_error(ResolveError::Missing {
                package: package.clone(),
            });
            return Ok(());
        }

        for (package, version) in assignments.iter() {
            if let Some(constraints) = active.get(package) {
                if !satisfies(*version, constraints) {
                    self.record_error(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                    return Ok(());
                }
            }
        }

        for (package, constraints) in &active {
            if assignments.contains_key(package) {
                continue;
            }
            let catalog_package = self.catalog[package];
            if !catalog_package
                .releases
                .iter()
                .any(|release| satisfies(release.version, constraints))
            {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
                return Ok(());
            }
        }

        let next_package = active
            .keys()
            .find(|package| !assignments.contains_key(*package))
            .cloned();

        let Some(package_name) = next_package else {
            self.consider_solution(assignments);
            return Ok(());
        };

        let constraints = &active[&package_name];
        let catalog_package = self.catalog[&package_name];
        let mut candidates: Vec<Version> = catalog_package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| satisfies(*version, constraints))
            .collect();
        candidates.sort_by(|left, right| right.cmp(left));

        if let Some(locked_version) = self.locks.get(&package_name) {
            if let Some(index) = candidates
                .iter()
                .position(|version| version == locked_version)
            {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }

            self.attempts += 1;
            assignments.insert(package_name.clone(), candidate);
            let child_result = self.visit(assignments);
            assignments.remove(&package_name);
            child_result?;
        }

        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: CatalogMap<'_> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let lock_map: LockMap = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: catalog_map,
        locks: lock_map,
        attempts: 0,
        max_attempts,
        best: None,
        best_cost: usize::MAX,
        first_error: None,
    };

    search.visit(&mut AssignmentMap::new())?;

    let Some(assignments) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
        }));
    };

    let final_constraints = active_constraints(roots, &search.catalog, &assignments);
    let selected = assignments
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_packages = std::collections::BTreeSet::new();
    all_packages.extend(search.locks.keys().cloned());
    all_packages.extend(assignments.keys().cloned());

    let mut changes = Vec::new();
    for package in all_packages {
        let before = search.locks.get(&package).copied();
        let after = assignments.get(&package).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = search.catalog.get(&package).is_some_and(|catalog_package| {
                    catalog_package
                        .releases
                        .iter()
                        .any(|release| release.version == locked)
                });

                if !exists {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(&package)
                    .is_some_and(|constraints| !satisfies(locked, constraints))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };

        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
