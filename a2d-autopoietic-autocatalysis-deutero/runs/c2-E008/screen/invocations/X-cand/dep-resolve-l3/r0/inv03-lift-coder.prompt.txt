fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_match(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|&c| matches(version, c))
}

    format!("{}.{}.{}", v.major, v.minor, v.patch)

fn constraint_text(c: Constraint) -> String {

fn active_constraints(
    roots: &[Requirement],
    selected: &[(String, Version)],
    catalog: &[Package],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in selected {
        if let Some(pkg) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<Vec<(String, Version)>>,
    best_cost: usize,
    best_vector: Vec<Option<Version>>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn lock_cost(&self, selection: &[(String, Version)]) -> usize {
        self.locks
            .iter()
            .filter(|lock| {
                !selection
                    .iter()
                    .any(|(name, version)| *name == lock.package && *version == lock.version)
            })
            .count()
    }

    fn vector(&self, selection: &[(String, Version)]) -> Vec<Option<Version>> {
        self.catalog
            .iter()
            .map(|pkg| {
                selection
                    .iter()
                    .find(|(name, _)| *name == pkg.name)
                    .map(|(_, version)| *version)
            })
            .collect()
    }

    fn record_solution(&mut self, selection: &[(String, Version)]) {
        let cost = self.lock_cost(selection);
        let vector = self.vector(selection);
        let better = self.best.is_none()
            || cost < self.best_cost
            || (cost == self.best_cost && vector > self.best_vector);
        if better {
            self.best = Some(selection.to_vec());
            self.best_cost = cost;
            self.best_vector = vector;
        }
    }

    fn remember_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn visit(&mut self, selected: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, selected, self.catalog);
        if active.is_empty() {
            self.record_solution(selected);
            return Ok(());
        }

        if let Some(name) = active
            .keys()
            .find(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .cloned()
        {
            let error = ResolveError::Missing { package: name };
            self.remember_error(error);
            return Ok(());
        }

        let mut ordered_selected = selected.clone();
        ordered_selected.sort_by(|a, b| a.0.cmp(&b.0));
        for (name, version) in &ordered_selected {
            if let Some(constraints) = active.get(name) {
                if !constraints_match(*version, constraints) {
                    let error = ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    };
                    self.remember_error(error);
                    return Ok(());
                }
            }
        }

        let mut branch_name: Option<String> = None;
        for (name, constraints) in &active {
            if selected.iter().any(|(selected_name, _)| selected_name == name) {
                continue;
            }
            let pkg = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !pkg
                .releases
                .iter()
                .any(|release| constraints_match(release.version, constraints))
            {
                let error = ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                };
                self.remember_error(error);
                return Ok(());
            }
            if branch_name.is_none() {
                branch_name = Some(name.clone());
            }
        }

        let Some(name) = branch_name else {
            self.record_solution(selected);
            return Ok(());
        };
        let constraints = &active[&name];
        let pkg = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<&Release> = pkg
            .releases
            .iter()
            .filter(|r| constraints_match(r.version, constraints))
            .collect();
        candidates.sort_by(|a, b| b.version.cmp(&a.version));

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(index) = candidates.iter().position(|r| r.version == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for release in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.push((name.clone(), release.version));
            let result = self.visit(selected);
            selected.pop();
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, pkg) in catalog.iter().enumerate() {
        if !valid_name(&pkg.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: pkg.name.clone(),
            });
        }
        for (j, release) in pkg.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for pkg in catalog {
        if !seen_packages.insert(pkg.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: pkg.name.clone(),
            });
        }
    }

    for pkg in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &pkg.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: pkg.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],

impl Search<'_> {
    fn consider_solution(&mut self, assignments: &AssignmentMap) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| assignments.get(*name) != Some(*version))
            .count();

        let replace = match &self.result.best {
            None => true,
            Some(best) if cost < self.result.best_cost => true,
            Some(best) if cost == self.result.best_cost => {
                self.compare_version_vectors(assignments, best).is_gt()
            }
            _ => false,
        };

        if replace {
            self.result.best = Some(assignments.clone());
            self.result.best_cost = cost;
        }
    }

    fn compare_version_vectors(
        &self,
        left: &AssignmentMap,
        right: &AssignmentMap,
    ) -> std::cmp::Ordering {
        for name in self.catalog.keys() {
            match (left.get(name), right.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (Some(a), Some(b)) => {
                    let ordering = a.cmp(b);
                    if !ordering.is_eq() {
                        return ordering;
                    }
                }
            }
        }
        std::cmp::Ordering::Equal
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.result.first_error.is_none() {
            self.result.first_error = Some(error);
        }
    }

    fn visit(&mut self, assignments: &mut AssignmentMap) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, self.catalog, assignments);

        if let Some(name) = active.keys().find(|name| !self.catalog.contains_key(*name)) {
            self.record_error(ResolveError::Missing {
                package: name.clone(),
            });
            return Ok(());
        }

        for (name, version) in assignments.iter() {
            if let Some(constraints) = active.get(name) {
                if !satisfies(*version, constraints) {
                    self.record_error(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                    return Ok(());
                }
            }
        }

        for (name, constraints) in &active {
            if assignments.contains_key(name) {
                continue;
            }
            let package = self.catalog[name];
            if !package
                .releases
                .iter()
                .any(|release| satisfies(release.version, constraints))
            {
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
                return Ok(());
            }
        }

        let next = active.keys().find(|name| !assignments.contains_key(*name));
        let Some(name) = next.cloned() else {
            self.consider_solution(assignments);
            return Ok(());
        };

        let constraints = &active[&name];
        let package = self.catalog[&name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| satisfies(*version, constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(&name) {
            if let Some(index) = candidates.iter().position(|version| version == locked) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assignments.insert(name.clone(), version);
            let child = self.visit(assignments);
            assignments.remove(&name);
            child?;
        }

        Ok(())
    }
}

        let active = active_constraints(self.roots, assigned, &self.catalog);

        if let Some(name) = active.keys().find(|name| !self.catalog.contains_key(*name)) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assigned.iter() {
            if let Some(cs) = active.get(name) {
                if !matches_all(*version, cs) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for (name, cs) in &active {
            if !assigned.contains_key(name) {
                let pkg = self.catalog[name];
                if !pkg.releases.iter().any(|r| matches_all(r.version, cs)) {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: cs.clone(),
                    });
                }
            }
        }

        let next = active.keys().find(|name| !assigned.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| assigned.get(*p) != Some(*v))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best)) => {
                    cost < *best_cost
                        || (cost == *best_cost
                            && self.vector_better(assigned, best))
                }
            };
            if replace {
                self.best = Some((cost, assigned.clone()));
            }
            return Ok(());
        };

        let cs = &active[&name];
        let mut candidates: Vec<Version> = self.catalog[&name]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches_all(*v, cs))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                candidates.remove(pos);
                candidates.insert(0, *lock);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), candidate);
            match self.inspect(assigned) {
                Ok(()) => {}
                Err(e @ ResolveError::Limit { .. }) => {
                    assigned.remove(&name);
                    return Err(e);
                }
                Err(e) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(e);
                    }
                }
            }
            assigned.remove(&name);
        }
        Ok(())
    }

    fn vector_better(&self, a: &Assigned, b: &Assigned) -> bool {
        let names: std::collections::BTreeSet<&String> = self
            .catalog
            .keys()
            .chain(a.keys())
            .chain(b.keys())
            .collect();
        for name in names {
            match (a.get(name), b.get(name)) {
                (Some(x), Some(y)) if x != y => return x > y,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, pkg) in catalog.iter().enumerate() {
        if !valid_name(&pkg.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: pkg.name.clone(),
            });
        }
        for (j, release) in pkg.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for pkg in catalog {
        if !seen.insert(pkg.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: pkg.name.clone(),
            });
        }
    }

    for pkg in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &pkg.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: pkg.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_map = std::collections::BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: std::collections::BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let mut search = Search {
        roots,
        catalog: catalog_map,
        locks: lock_map.clone(),
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };
    let mut assigned = Assigned::new();

    if let Err(e) = search.inspect(&mut assigned) {
        if matches!(e, ResolveError::Limit { .. }) {
            return Err(e);
        }
        if search.best.is_none() {
            return Err(search.first_error.unwrap_or(e));
        }
    }

    let Some((_, solution)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
        }));
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = active_constraints(roots, &solution, &search.catalog);
    let mut all_names: std::collections::BTreeSet<String> =
        lock_map.keys().cloned().collect();
    all_names.extend(solution.keys().cloned());

    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = solution.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .catalog
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_active
                    .get(&package)
                    .is_some_and(|cs| matches_all(old, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
