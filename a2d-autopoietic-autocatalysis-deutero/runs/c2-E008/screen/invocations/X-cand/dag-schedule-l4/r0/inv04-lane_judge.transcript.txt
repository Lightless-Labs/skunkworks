pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    use std::collections::BTreeSet;

    let n = tasks.len();

    // Dependency validation takes precedence over every other phase.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shape and bounds one task at a time.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &limit)) in task.needs.iter().zip(capacity).enumerate() {
            if need > limit {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: limit,
                });
            }
        }
    }

    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest currently available ID.
    let mut available = BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut remaining_indegree = indegree.clone();
    let mut order = Vec::with_capacity(n);
    while let Some(task_id) = available.pop_first() {
        order.push(task_id);
        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &task_id in &order {
            removed[task_id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors = indegree;
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut worker_available = vec![0u64; worker_count];
    let mut used = vec![0u64; capacity.len()];
    let mut running = Vec::<Running>::new();
    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Finish all tasks at this time in ascending task ID order.
        let mut ending: Vec<Running> = running
            .iter()
            .copied()
            .filter(|item| item.finish == now)
            .collect();
        ending.sort_by_key(|item| item.task);
        if !ending.is_empty() {
            running.retain(|item| item.finish != now);
            for item in ending {
                completed[item.task] = true;
                completed_count += 1;
                for (resource, &need) in tasks[item.task].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }

                trace.push_str(&format!("{now}|end|{}|{}|", item.task, item.worker));
                append_resources(&mut trace, &used);
                trace.push('\n');

                for &child in &successors[item.task] {
                    remaining_predecessors[child] -= 1;
                }
            }
        }

        // Start one task at a time, reevaluating after every admission.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > now
                    || remaining_predecessors[task_id] != 0
                {
                    continue;
                }

                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(resource, &need)| {
                        u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                    });
                if !fits {
                    continue;
                }

                let has_worker = worker_available.iter().any(|&at| at <= now);
                if !has_worker {
                    continue;
                }

                let outranks = match best {
                    None => true,
                    Some(other) => {
                        tails[task_id] > tails[other]
                            || (tails[task_id] == tails[other]
                                && (tasks[task_id].priority > tasks[other].priority
                                    || (tasks[task_id].priority == tasks[other].priority
                                        && task_id < other)))
                    }
                };
                if outranks {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now)
                .expect("an available worker was established");

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }

            trace.push_str(&format!("{now}|start|{task_id}|{worker}|"));
            append_resources(&mut trace, &used);
            trace.push('\n');

            if finish == now {
                // Zero-duration tasks complete before the next task is selected.
                completed[task_id] = true;
                completed_count += 1;
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }

                trace.push_str(&format!("{now}|end|{task_id}|{worker}|"));
                append_resources(&mut trace, &used);
                trace.push('\n');

                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }

            makespan = makespan.max(finish);
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |event: u64| {
            if event > now {
                next_time = Some(next_time.map_or(event, |current| current.min(event)));
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for (task_id, task) in tasks.iter().enumerate() {
            if !started[task_id] {
                consider(task.release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        if let Some(event) = next_time {
            now = event;
        } else {
            // A validated acyclic graph with valid workers always has a future event
            // whenever unfinished tasks remain.
            unreachable!("unfinished valid schedule has no future event");
        }
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task is started and completed"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Validate resource shape and bounds task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &limit)) in task.needs.iter().zip(capacity).enumerate() {
            if need > limit {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: limit,
                });
            }
        }
    }

    // Build predecessor-to-successor edges and perform deterministic Kahn traversal.
    let mut children = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut removed = vec![false; n];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        let blocked = (0..n).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let mut worker_available_at = vec![0u64; worker_count];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();
    let mut used = vec![0u64; capacity.len()];
    let mut running: Vec<Running> = Vec::new();
    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        n
    ];
    let mut makespan = 0u64;
    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut now = 0u64;

    while completed.iter().any(|&done| !done) {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, task)| (task.finish == now).then_some(index))
            .collect();
        finishing.sort_by_key(|&index| running[index].task);

        for index in finishing.into_iter().rev() {
            let task_run = running.remove(index);
            let task_id = task_run.task;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            completed[task_id] = true;
            makespan = makespan.max(now);

            trace.push_str(&format!(
                "{now}|end|{task_id}|{}|",
                task_run.worker
            ));
            append_resource_snapshot(&mut trace, &used);
            trace.push('\n');

            for &child in &children[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= now
                        && remaining_predecessors[task_id] == 0
                        && tasks[task_id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(resource, &need)| {
                                u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                            })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tail[b]
                    .cmp(&tail[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            // Select the first available worker in the task's cyclic preference order.
            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available_at[worker] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at =
                finish
                    .checked_add(cooldowns[worker])
                    .ok_or(Error::Overflow {
                        task: task_id,
                        stage: Stage::Turnaround,
                    })?;

            started[task_id] = true;
            runs[task_id] = Run {
                start: now,
                finish,
                worker,
            };
            worker_available_at[worker] = available_at;

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }

            trace.push_str(&format!("{now}|start|{task_id}|{worker}|"));
            append_resource_snapshot(&mut trace, &used);
            trace.push('\n');

            if finish == now {
                // Zero-duration tasks complete immediately before another admission.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                makespan = makespan.max(now);

                trace.push_str(&format!("{now}|end|{task_id}|{worker}|"));
                append_resource_snapshot(&mut trace, &used);
                trace.push('\n');

                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed.iter().all(|&done| done) {
            break;
        }

        let mut next_event: Option<u64> = None;
        let mut consider = |event: u64| {
            if event > now {
                next_event = Some(next_event.map_or(event, |current| current.min(event)));
            }
        };

        for task in &running {
            consider(task.finish);
        }
        for (task_id, task) in tasks.iter().enumerate() {
            if !started[task_id] {
                consider(task.release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        if let Some(event) = next_event {
            now = event;
        } else {
            // Every valid acyclic input with a valid worker count has a future
            // event whenever unfinished work remains.
            break;
        }
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::fmt::Write;

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            successors[dependency].push(task_id);
        }
    }

    let mut ready = BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            ready.push(Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree.clone();
    while let Some(Reverse(task_id)) = ready.pop() {
        order.push(task_id);
        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                ready.push(Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &task_id in &order {
            removed[task_id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_successor_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u128; resource_count];
    let mut remaining_predecessors = indegree;
    let mut started = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut finish_times = vec![0u64; n];
    let mut running = vec![false; n];
    let mut worker_available = vec![0u64; worker_count];
    let mut current_time = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    let mut trace = String::new();
    writeln!(
        trace,
        "workers={};resources={}",
        worker_count, resource_count
    )
    .unwrap();

    while completed_count < n {
        let mut finishing: Vec<usize> = (0..n)
            .filter(|&task_id| running[task_id] && finish_times[task_id] == current_time)
            .collect();
        finishing.sort_unstable();

        for task_id in finishing {
            running[task_id] = false;
            completed_count += 1;
            makespan = makespan.max(current_time);

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] -= u128::from(need);
            }

            let worker = runs[task_id].as_ref().unwrap().worker;
            append_trace_line(
                &mut trace,
                current_time,
                "end",
                task_id,
                worker,
                &used,
            );

            for &child in &successors[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= current_time
                        && remaining_predecessors[task_id] == 0
                        && tasks[task_id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(resource, &need)| {
                                u128::from(need)
                                    <= u128::from(capacity[resource]) - used[resource]
                            })
                        && worker_available
                            .iter()
                            .any(|&available| available <= current_time)
                })
                .collect();

            if candidates.is_empty() {
                break;
            }

            candidates.sort_by(|&left, &right| {
                tails[right]
                    .cmp(&tails[left])
                    .then_with(|| tasks[right].priority.cmp(&tasks[left].priority))
                    .then_with(|| left.cmp(&right))
            });
            let task_id = candidates[0];

            let start = current_time;
            let finish = start
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= current_time)
                .unwrap();

            let available_again = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_again;
            runs[task_id] = Some(Run {
                start,
                finish,
                worker,
            });
            finish_times[task_id] = finish;

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u128::from(need);
            }
            append_trace_line(&mut trace, start, "start", task_id, worker, &used);

            if tasks[task_id].duration == 0 {
                running[task_id] = true;
                running[task_id] = false;
                completed_count += 1;
                makespan = makespan.max(current_time);

                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u128::from(need);
                }
                append_trace_line(&mut trace, current_time, "end", task_id, worker, &used);

                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running[task_id] = true;
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for task_id in 0..n {
            if running[task_id] && finish_times[task_id] > current_time {
                next_time = Some(next_time.map_or(finish_times[task_id], |t| {
                    t.min(finish_times[task_id])
                }));
            }
            if !started[task_id] && tasks[task_id].release > current_time {
                next_time = Some(next_time.map_or(tasks[task_id].release, |t| {
                    t.min(tasks[task_id].release)
                }));
            }
        }
        for &available in &worker_available {
            if available > current_time {
                next_time = Some(next_time.map_or(available, |t| t.min(available)));
            }
        }

        current_time = next_time.expect("unfinished tasks must have a future event");
    }

    for (task_id, run) in runs.iter().enumerate() {
        if run.is_none() {
            // The only possible missing run would indicate an internal scheduling bug.
            unreachable!("every task in an acyclic graph must be scheduled");
        }
        let _ = task_id;
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    writeln!(trace, "makespan={}", makespan).unwrap();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn append_trace_line(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u128],
) {
    write!(trace, "{}|{}|{}|{}|", time, kind, task, worker).unwrap();
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            write!(trace, "{value}").unwrap();
        }
    }
    trace.push('\n');
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    use std::collections::BTreeSet;

    let n = tasks.len();

    // Validate dependencies before all other checks.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = BTreeSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn order, always removing the smallest currently available ID.
    let mut available = BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(task_id) = available.pop_first() {
        order.push(task_id);
        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed: Vec<bool> = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        finish: u64,
        worker: usize,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut time = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    fn append_resources(trace: &mut String, used: &[u64]) {
        if used.is_empty() {
            trace.push('-');
        } else {
            for (index, value) in used.iter().enumerate() {
                if index != 0 {
                    trace.push(',');
                }
                trace.push_str(&value.to_string());
            }
        }
    }

    loop {
        // Complete all tasks finishing now, in ascending task ID.
        let mut ending: Vec<Running> = running
            .iter()
            .copied()
            .filter(|r| r.finish == time)
            .collect();
        ending.sort_by_key(|r| r.task);

        if !ending.is_empty() {
            running.retain(|r| r.finish != time);
            for item in ending {
                completed[item.task] = true;
                completed_count += 1;
                for (resource, &need) in tasks[item.task].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|",
                    time, item.task, item.worker
                ));
                append_resources(&mut trace, &used);
                trace.push('\n');
            }
        }

        // Start tasks one at a time, reevaluating eligibility and priority each time.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > time
                    || !tasks[task_id]
                        .deps
                        .iter()
                        .all(|&dep| completed[dep])
                {
                    continue;
                }

                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(resource, &need)| {
                        u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                    });
                if !fits {
                    continue;
                }

                let worker = (0..worker_count)
                    .map(|offset| (task_id % worker_count + offset) % worker_count)
                    .find(|&w| worker_available[w] <= time);
                if worker.is_none() {
                    continue;
                }

                let outranks = match best {
                    None => true,
                    Some(other) => {
                        tails[task_id] > tails[other]
                            || (tails[task_id] == tails[other]
                                && (tasks[task_id].priority > tasks[other].priority
                                    || (tasks[task_id].priority == tasks[other].priority
                                        && task_id < other)))
                    }
                };
                if outranks {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= time)
                .expect("worker availability was checked");

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let turnaround = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = turnaround;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });

            trace.push_str(&format!(
                "{}|start|{}|{}|",
                time, task_id, worker
            ));
            append_resources(&mut trace, &used);
            trace.push('\n');

            makespan = makespan.max(finish);
            running.push(Running {
                task: task_id,
                finish,
                worker,
            });

            // A newly started zero-duration task completes immediately.
            if finish == time {
                running.retain(|r| r.task != task_id);
                completed[task_id] = true;
                completed_count += 1;
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|",
                    time, task_id, worker
                ));
                append_resources(&mut trace, &used);
                trace.push('\n');
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest event that can change eligibility or availability.
        let mut next_time: Option<u64> = None;
        for item in &running {
            if item.finish > time {
                next_time = Some(next_time.map_or(item.finish, |t| t.min(item.finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > time {
                next_time = Some(
                    next_time.map_or(tasks[task_id].release, |t| t.min(tasks[task_id].release)),
                );
            }
        }
        for &available_at in &worker_available {
            if available_at > time {
                next_time = Some(next_time.map_or(available_at, |t| t.min(available_at)));
            }
        }

        match next_time {
            Some(next) => time = next,
            None => {
                // For an acyclic graph, unfinished work always has a future event.
                // This guard keeps the function total in case of an invariant violation.
                let blocked: Vec<usize> = (0..n).filter(|&id| !completed[id]).collect();
                return Err(Error::Cycle { blocked });
            }
        }
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task completed and therefore started"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}
