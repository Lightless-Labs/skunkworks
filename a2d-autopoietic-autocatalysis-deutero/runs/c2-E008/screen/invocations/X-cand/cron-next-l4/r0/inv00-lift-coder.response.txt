#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32, usize) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn names(field: Field) -> &'static [(&'static str, u32)] {
    match field {
        Field::Month => &[
            ("JAN", 1), ("FEB", 2), ("MAR", 3), ("APR", 4),
            ("MAY", 5), ("JUN", 6), ("JUL", 7), ("AUG", 8),
            ("SEP", 9), ("OCT", 10), ("NOV", 11), ("DEC", 12),
        ],
        Field::DayOfWeek => &[
            ("SUN", 0), ("MON", 1), ("TUE", 2), ("WED", 3),
            ("THU", 4), ("FRI", 5), ("SAT", 6),
        ],
        _ => &[],
    }
}

fn parse_atom(field: Field, bytes: &[u8], base: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset: base });
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        let offset = base;
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut value = 0u32;
        for &byte in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((byte - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let (min, max, _) = domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok(value);
    }
    if bytes.iter().all(u8::is_ascii_alphabetic) {
        let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
        for &(name, value) in names(field) {
            if upper.as_slice() == name.as_bytes() {
                return Ok(value);
            }
        }
    }
    Err(ParseError::InvalidSyntax { field, offset: base })
}

fn parse_step(field: Field, bytes: &[u8], offset: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (_, _, width) = domain(field);
    if value == 0 || value > width as u32 {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_item(
    field: Field,
    item: &[u8],
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset: item_offset });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax { field, offset: item_offset });
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, 0)
    };
    if base.is_empty() || step_bytes.is_some_and(|s| s.is_empty()) {
        return Err(ParseError::InvalidSyntax { field, offset: item_offset });
    }

    let (min, max, _) = domain(field);
    let star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let mut start = min;
    let mut end = max;
    let mut end_offset = item_offset;

    if star {
        if !hyphens.is_empty() {
            return Err(ParseError::InvalidSyntax { field, offset: item_offset });
        }
    } else if hyphens.is_empty() {
        start = parse_atom(field, base, item_offset)?;
        end = if step_bytes.is_some() { max } else { start };
    } else {
        if hyphens.len() != 1 {
            return Err(ParseError::InvalidSyntax { field, offset: item_offset });
        }
        let dash = hyphens[0];
        if dash == 0 || dash + 1 == base.len() {
            return Err(ParseError::InvalidSyntax { field, offset: item_offset });
        }
        start = parse_atom(field, &base[..dash], item_offset)?;
        end_offset = item_offset + dash + 1;
        end = parse_atom(field, &base[dash + 1..], end_offset)?;
        if start > end {
            return Err(ParseError::ReversedRange { field, offset: end_offset });
        }
    }

    let step = match step_bytes {
        Some(bytes) => parse_step(field, bytes, step_offset)?,
        None => 1,
    };

    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, _, width) = domain(field);
    let mut selected = vec![false; width];
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(field, &bytes[start..index], start, &mut selected)?;
            start = index + 1;
        }
    }
    parse_item(field, &bytes[start..], start, &mut selected)?;
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    if input.as_bytes().first() == Some(&b'@') {
        let expansion = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        return parse_cron(expansion);
    }

    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut parsed = Vec::with_capacity(5);
    for (index, &field) in FIELDS.iter().enumerate() {
        parsed.push(parse_field(field, parts[index])?);
    }

    Ok(Cron {
        sets: parsed.try_into().ok().unwrap(),
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        let (min, _, _) = domain(field);
        self.sets[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some((min as usize + i) as u8))
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let (min, max, _) = domain(field);
        if (value as u32) < min || (value as u32) > max {
            return false;
        }
        self.sets[field_index(field)][value as usize - min as usize]
    }

    fn day_matches(&self, date: CivilDateTime) -> bool {
        let dom_free = self.unrestricted_days[0];
        let dow_free = self.unrestricted_days[1];
        if dom_free && dow_free {
            return true;
        }
        let dom = self.contains(Field::DayOfMonth, date.day);
        let dow = self.contains(Field::DayOfWeek, weekday(date.year, date.month, date.day));
        if dom_free {
            return dow;
        }
        if dow_free {
            return dom;
        }
        if date.day == days_in_month(date.year, date.month) {
            dom && dow
        } else {
            dom || dow
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.day_matches(at)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut date = (after.year, after.month, after.day);
        loop {
            let (year, month, day) = date;
            let weekday = weekday(year, month, day);
            let day_matches = self.day_matches(CivilDateTime {
                year, month, day, hour: 0, minute: 0,
            }) && self.contains(Field::DayOfWeek, weekday);
            let date_matches = self.contains(Field::Month, month)
                && self.day_matches(CivilDateTime {
                    year, month, day, hour: 0, minute: 0,
                });
            if date_matches {
                for hour in self.allowed(Field::Hour) {
                    for minute in self.allowed(Field::Minute) {
                        let candidate = CivilDateTime { year, month, day, hour, minute };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            let _ = day_matches;
            match next_date(year, month, day) {
                Some(next) => date = next,
                None => return Ok(None),
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut date = (before.year, before.month, before.day);
        loop {
            let (year, month, day) = date;
            if self.contains(Field::Month, month)
                && self.day_matches(CivilDateTime {
                    year, month, day, hour: 0, minute: 0,
                })
            {
                let hours = self.allowed(Field::Hour);
                let minutes = self.allowed(Field::Minute);
                for &hour in hours.iter().rev() {
                    for &minute in minutes.iter().rev() {
                        let candidate = CivilDateTime { year, month, day, hour, minute };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            match previous_date(year, month, day) {
                Some(prev) => date = prev,
                None => return Ok(None),
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while cursor < end {
            match self.next_fire(cursor) {
                Ok(Some(fire)) if fire < end => {
                    if fires.len() == cap {
                        return Ok(FireWindow { fires, truncated: true });
                    }
                    fires.push(fire);
                    cursor = fire;
                }
                _ => break,
            }
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        FIELDS
            .iter()
            .map(|&field| {
                let values = self.allowed(field);
                let (_, _, width) = domain(field);
                let is_day = matches!(field, Field::DayOfMonth | Field::DayOfWeek);
                let unrestricted = match field {
                    Field::DayOfMonth => self.unrestricted_days[0],
                    Field::DayOfWeek => self.unrestricted_days[1],
                    _ => false,
                };
                if unrestricted {
                    return "*".to_string();
                }
                if values.len() == width {
                    return if is_day { "*/1".to_string() } else { "*".to_string() };
                }
                values.iter().map(u8::to_string).collect::<Vec<_>>().join(",")
            })
            .collect::<Vec<_>>()
            .join(" ")
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn next_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day < days_in_month(year, month) {
        Some((year, month, day + 1))
    } else if month < 12 {
        Some((year, month + 1, 1))
    } else if year < 2399 {
        Some((year + 1, 1, 1))
    } else {
        None
    }
}

fn previous_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day > 1 {
        Some((year, month, day - 1))
    } else if month > 1 {
        let prev_month = month - 1;
        Some((year, prev_month, days_in_month(year, prev_month)))
    } else if year > 2000 {
        Some((year - 1, 12, 31))
    } else {
        None
    }
}
