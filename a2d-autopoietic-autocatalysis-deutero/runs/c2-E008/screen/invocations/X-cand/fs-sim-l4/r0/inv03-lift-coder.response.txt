```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn normalize(path: &str, cwd: &str) -> String {
    let mut parts: Vec<String> = if path.starts_with('/') {
        Vec::new()
    } else {
        cwd.split('/')
            .filter(|s| !s.is_empty())
            .map(str::to_owned)
            .collect()
    };

    for part in path.split('/') {
        match part {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            _ => parts.push(part.to_owned()),
        }
    }
    canonical(&parts)
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|s| !s.is_empty())
        .map(str::to_owned)
        .collect()
}

fn parent(path: &str) -> String {
    if path == "/" {
        return "/".to_owned();
    }
    match path.rfind('/') {
        Some(0) => "/".to_owned(),
        Some(i) => path[..i].to_owned(),
        _ => "/".to_owned(),
    }
}

fn basename(path: &str) -> String {
    if path == "/" {
        "/".to_owned()
    } else {
        path.rsplit('/').next().unwrap_or("").to_owned()
    }
}

fn child_path(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn within(path: &str, dir: &str) -> bool {
    if dir == "/" {
        path.starts_with('/')
    } else {
        path == dir
            || path
                .strip_prefix(dir)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_missing_final: bool,
    ) -> Result<String, Error> {
        let mut todo = components(path);
        let mut physical: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if todo.is_empty() {
                return Ok(canonical(&physical));
            }

            let component = todo.remove(0);
            let mut candidate_parts = physical.clone();
            candidate_parts.push(component.clone());
            let candidate = canonical(&candidate_parts);
            let is_final = todo.is_empty();

            let Some(entry) = self.entries.get(&candidate) else {
                if create_missing_final && is_final {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            };

            let should_follow =
                matches!(entry, Entry::Link(_)) && (follow_final || !is_final);

            if should_follow {
                if !visited.insert(candidate.clone()) {
                    return Err(Error::Loop(candidate));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(candidate));
                }
                expansions += 1;

                let Entry::Link(target) = entry else {
                    unreachable!()
                };

                let target_base = if target.starts_with('/') {
                    "/".to_owned()
                } else {
                    canonical(&physical)
                };
                let expanded_target = normalize(target, &target_base);
                let mut new_todo = components(&expanded_target);
                new_todo.extend(todo);
                todo = new_todo;
                physical.clear();
                continue;
            }

            if !is_final && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
            physical.push(component);
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let name = basename(path);
        let resolved_parent = self.resolve(&parent(path), true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| path.as_str() != dir && parent(path) == dir)
            .cloned()
            .collect()
    }

    fn remove_subtree(&mut self, path: &str) {
        let doomed: Vec<String> = self
            .entries
            .keys()
            .filter(|entry_path| within(entry_path, path))
            .cloned()
            .collect();
        for path in doomed {
            self.entries.remove(&path);
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        let (resolved_parent, name) = self.resolve_parent(&normalized)?;
        let target = child_path(&resolved_parent, &name);

        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        let target = self.resolve(&normalized, true, true)?;

        match self.entries.get(&target) {
            Some(Entry::Dir) => Err(Error::IsDir(target)),
            Some(Entry::File(_)) => {
                self.entries.insert(target, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let target_parent = parent(&target);
                match self.entries.get(&target_parent) {
                    Some(Entry::Dir) => {
                        self.entries.insert(target, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(target_parent)),
                    None => Err(Error::Missing(target_parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;

        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => unreachable!(),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;

        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }

        let mut result: Vec<(String, Kind)> = self
            .children(&resolved)
            .into_iter()
            .map(|child| {
                let name = basename(&child);
                let kind = match self.entries.get(&child).unwrap() {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                    Entry::Link(_) => Kind::Link,
                };
                (name, kind)
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;

        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        validate_path(from)?;
        validate_path(to)?;
        let from_normalized = normalize(from, &self.cwd);
        let to_normalized = normalize(to, &self.cwd);

        if from_normalized == "/" || to_normalized == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&from_normalized)?;
        let source = child_path(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&to_normalized)?;
        let destination = child_path(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(existing) = &destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| within(path, &source))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if destination_entry.is_some() {
            self.entries.remove(&destination);
        }

        for (old_path, _) in &moving {
            self.entries.remove(old_path);
        }
        for (old_path, entry) in moving {
            let suffix = &old_path[source.len()..];
            self.entries.insert(format!("{destination}{suffix}"), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        if normalized == "/" {
            return Err(Error::Root);
        }

        let (resolved_parent, name) = self.resolve_parent(&normalized)?;
        let target = child_path(&resolved_parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        self.remove_subtree(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        validate_path(link)?;
        let normalized_link = normalize(link, &self.cwd);

        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (resolved_parent, name) = self.resolve_parent(&normalized_link)?;
        let resolved_link = child_path(&resolved_parent, &name);
        if self.entries.contains_key(&resolved_link) {
            return Err(Error::Exists(resolved_link));
        }
        self.entries
            .insert(resolved_link, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, false, false)?;

        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        validate_path(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;

        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|entry_path| within(entry_path, &resolved))
            .cloned()
            .collect();
        paths.sort();

        let mut rows = Vec::with_capacity(paths.len());
        for current in paths {
            let size = match self.entries.get(&current).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => self
                    .entries
                    .iter()
                    .filter(|(entry_path, _)| {
                        entry_path.as_str() != current && within(entry_path, &current)
                    })
                    .map(|(_, entry)| match entry {
                        Entry::File(data) => data.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
            };
            rows.push((current, size));
        }
        Ok(rows)
    }
}
```
