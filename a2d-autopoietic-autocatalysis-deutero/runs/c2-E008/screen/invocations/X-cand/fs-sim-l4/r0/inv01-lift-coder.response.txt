```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|c| !c.is_empty() && *c != ".")
            .map(str::to_string)
            .collect()
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let absolute = path.starts_with('/');
        let mut parts = if absolute {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for c in Self::components(path) {
            if c == ".." {
                parts.pop();
            } else {
                parts.push(c);
            }
        }
        Self::canonical(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            _ => "/".to_string(),
        }
    }

    fn basename(path: &str) -> String {
        path.rsplit('/').next().unwrap_or("").to_string()
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || path
                .strip_prefix(dir)
                .is_some_and(|rest| rest.starts_with('/'))
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let mut todo: Vec<String> = Self::components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        while !todo.is_empty() {
            let comp = todo.remove(0);
            let candidate_parts = {
                let mut p = resolved.clone();
                p.push(comp);
                p
            };
            let candidate = Self::canonical(&candidate_parts);
            let entry = self.entries.get(&candidate);

            let is_final = todo.is_empty();
            let should_follow = matches!(entry, Some(Entry::Link(_)))
                && (follow_final || !is_final);

            if should_follow {
                if !visited.insert(candidate.clone()) {
                    return Err(Error::Loop(candidate));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(candidate));
                }
                expansions += 1;

                let target = match entry.unwrap() {
                    Entry::Link(t) => t.clone(),
                    _ => unreachable!(),
                };
                let mut target_parts = if target.starts_with('/') {
                    Vec::new()
                } else {
                    resolved.clone()
                };
                for part in Self::components(&target) {
                    if part == ".." {
                        target_parts.pop();
                    } else {
                        target_parts.push(part);
                    }
                }
                let mut combined = target_parts;
                combined.extend(todo.into_iter());
                todo = combined;
                resolved.clear();
                continue;
            }

            if entry.is_none() {
                if create_final && is_final {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            }

            if !is_final && !matches!(entry, Some(Entry::Dir)) {
                return Err(Error::NotDir(candidate));
            }

            resolved.push(comp);
        }

        Ok(Self::canonical(&resolved))
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let name = Self::basename(path);
        let physical_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&physical_parent) {
            Some(Entry::Dir) => Ok((physical_parent, name)),
            Some(_) => Err(Error::NotDir(physical_parent)),
            None => Err(Error::Missing(physical_parent)),
        }
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| *p != dir && Self::parent(p) == dir)
            .cloned()
            .collect()
    }

    fn remove_subtree(&mut self, path: &str) {
        let doomed: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, path))
            .cloned()
            .collect();
        for p in doomed {
            self.entries.remove(&p);
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let (parent, name) = self.resolve_parent(&normalized)?;
        let full = Self::child_path(&parent, &name);
        if self.entries.contains_key(&full) {
            return Err(Error::Exists(full));
        }
        self.entries.insert(full, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, true)?;
        if resolved == "/" {
            return Err(Error::IsDir("/".to_string()));
        }
        if let Some(entry) = self.entries.get(&resolved) {
            match entry {
                Entry::Dir => return Err(Error::IsDir(resolved)),
                Entry::File(_) => {}
                Entry::Link(_) => unreachable!(),
            }
        } else {
            let (parent, _) = self.resolve_parent(&resolved)?;
            if parent != Self::parent(&resolved) {
                return Err(Error::Missing(Self::parent(&resolved)));
            }
        }
        self.entries
            .insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut rows: Vec<(String, Kind)> = self
            .entries
            .iter()
            .filter(|(p, _)| *p != &resolved && Self::parent(p) == resolved)
            .map(|(p, e)| (Self::basename(p), e.kind()))
            .collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        Self::validate(to)?;
        let src_norm = Self::normalize(from, &self.cwd);
        let dst_norm = Self::normalize(to, &self.cwd);
        if src_norm == "/" || dst_norm == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&src_norm)?;
        let src = Self::child_path(&src_parent, &src_name);
        let src_entry = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(src_entry, Entry::Dir) && Self::is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&dst_norm)?;
        let dst = Self::child_path(&dst_parent, &dst_name);

        if matches!(self.entries.get(&dst), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }

        if matches!(src_entry, Entry::Dir) && Self::is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        if let Some(dst_entry) = self.entries.get(&dst) {
            if !overwrite {
                return Err(Error::Exists(dst));
            }
            match (&src_entry, dst_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&dst).is_empty() {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| Self::is_within(p, &src))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();
        if self.entries.contains_key(&dst) {
            self.remove_subtree(&dst);
        }
        self.remove_subtree(&src);
        for (old_path, entry) in moving {
            let suffix = &old_path[src.len()..];
            self.entries.insert(format!("{}{}", dst, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let full = Self::child_path(&parent, &name);
        let entry = self
            .entries
            .get(&full)
            .cloned()
            .ok_or_else(|| Error::Missing(full.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &full) {
            return Err(Error::Busy(full));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&full).is_empty() {
            return Err(Error::NotEmpty(full));
        }
        self.remove_subtree(&full);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let normalized = Self::normalize(link, &self.cwd);
        let (parent, name) = self.resolve_parent(&normalized)?;
        let full = Self::child_path(&parent, &name);
        if self.entries.contains_key(&full) {
            return Err(Error::Exists(full));
        }
        self.entries.insert(full, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        let root_entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;
        if !matches!(root_entry, Entry::Dir | Entry::File(_)) {
            return Err(Error::NotDir(resolved));
        }

        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &resolved))
            .cloned()
            .collect();
        paths.sort();

        fn size(fs: &Fs, path: &str) -> u64 {
            match fs.entries.get(path) {
                Some(Entry::File(data)) => data.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                Some(Entry::Dir) => fs
                    .children(path)
                    .iter()
                    .map(|child| size(fs, child))
                    .sum(),
                None => 0,
            }
        }

        Ok(paths
            .into_iter()
            .map(|p| {
                let n = size(self, &p);
                (p, n)
            })
            .collect())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
