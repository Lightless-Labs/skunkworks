    File,

pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

        entries.insert("/".to_owned(), Entry::Dir);

            cwd: "/".to_owned(),

            components.extend(self.cwd.split('/').filter(|part| !part.is_empty()));

                    components.pop();
                }
                _ => components.push(part),

        }

            Ok("/".to_owned())
        } else {
            Ok(format!("/{}", components.join("/")))
        }

        path.split('/').filter(|part| !part.is_empty()).collect()

        self.cwd.clone()

            Entry::Dir => {
                self.cwd = path;
                Ok(())

            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

        let from = self.validate_and_normalize(from)?;
        let to = self.validate_and_normalize(to)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);

            return Err(Error::Busy(from));

            return Err(Error::Busy(to));

            return Ok(());

            return Err(Error::Cycle(to));
        }

        let destination = self.entries.get(&to).cloned();
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if let Some(dest) = destination {
            match (&source, &dest) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&to).is_empty() {
                        return Err(Error::NotEmpty(to));
                    }
                }
                (Entry::File(_), Entry::File(_)) => {}
            }
        }

        if self.entries.contains_key(&to) {
            self.entries.remove(&to);
        }
        self.move_subtree(&from, &to);
        Ok(())
    }

        if path == "/" {
            return Err(Error::Root);

            .get(&path)

            return Err(Error::Busy(path));

        if matches!(entry, Entry::Dir) && !recursive && !self.children(&path).is_empty() {
            return Err(Error::NotEmpty(path));
        }

        let prefix = format!("{}/", path);
        self.entries
            .retain(|candidate, _| candidate != &path && (!recursive || !candidate.starts_with(&prefix)));
        Ok(())
    }
}

    IsDir(String),

enum Entry {

    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            parts.extend(self.cwd.split('/').filter(|s| !s.is_empty()));
        }

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }

        if parts.is_empty() {
            Ok("/".to_owned())
        } else {
            let mut result = String::new();
            for part in parts {
                result.push('/');
                result.push_str(part);
            }
            Ok(result)
        }
    }

    fn parent(path: &str) -> (&str, &str) {
        if path == "/" {
            return ("/", "");
        }
        match path.rfind('/') {
            Some(0) => ("/", &path[1..]),
            Some(i) => (&path[..i], &path[i + 1..]),
            None => ("/", path),
        }
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn is_within(path: &str, ancestor: &str) -> bool {
        if ancestor == "/" {
            return path.starts_with('/');
        }
        path == ancestor
            || path
                .strip_prefix(ancestor)
                .is_some_and(|rest| rest.starts_with('/'))
    }

    fn resolve_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent(path);
        self.lookup(parent)?;
        match self.entries.get(parent) {
            Some(Entry::Dir) => Ok(()),
            _ => Err(Error::NotDir(parent.to_owned())),
        }
    }

    fn lookup(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self
                .entries
                .get("/")
                .ok_or_else(|| Error::Missing("/".to_owned()));
        }

        let mut prefix = String::new();
        let parts: Vec<&str> = path.split('/').filter(|s| !s.is_empty()).collect();

        for (i, part) in parts.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(part);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(entry) if i + 1 < parts.len() && !matches!(entry, Entry::Dir) => {
                    return Err(Error::NotDir(prefix));
                }
                Some(_) => {}
            }
        }

        self.entries
            .get(path)
            .ok_or_else(|| Error::Missing(path.to_owned()))
    }

    fn direct_children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                *candidate != path
                    && Self::parent(candidate).0 == path
            })
            .cloned()
            .collect()
    }

    fn subtree_paths(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, path))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        self.resolve_parent(&resolved)?;
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {

        if resolved == "/" {

        match self.entries.get_mut(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())

        }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let resolved = self.normalize(path)?;
        match self.lookup(&resolved)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir => Err(Error::IsDir(resolved)),

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate_and_normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_string()),
            }
        }

        Ok(Self::canonical(&components))
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn parent_and_name(path: &str) -> Option<(String, String)> {
        let components = Self::components(path);
        let name = components.last()?.clone();
        let parent = Self::canonical(&components[..components.len() - 1]);
        Some((parent, name))
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn resolve_directory(&self, path: &str) -> Result<(), Error> {
        if path == "/" {
            return Ok(());
        }
        let mut prefix = String::new();
        for component in Self::components(path) {
            prefix.push('/');
            prefix.push_str(&component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_entry(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self.entries.get("/").ok_or_else(|| Error::Missing("/".into()));
        }
        let (parent, _) = Self::parent_and_name(path).expect("non-root canonical path");
        self.resolve_directory(&parent)?;
        self.entries
            .get(path)
            .ok_or_else(|| Error::Missing(path.to_string()))
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        directory == "/"
            || path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn direct_children(&self, path: &str) -> Vec<String> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{path}/")
        };
        self.entries
            .keys()
            .filter_map(|candidate| {
                if candidate == path || !candidate.starts_with(&prefix) {
                    return None;
                }
                let remainder = &candidate[prefix.len()..];
                if remainder.is_empty() || remainder.contains('/') {
                    None
                } else {
                    Some(candidate.clone())
                }
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.validate_and_normalize(path)?;
        if resolved == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, _) = Self::parent_and_name(&resolved).expect("non-root canonical path");
        self.resolve_directory(&parent)?;
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let resolved = self.validate_and_normalize(path)?;
        if resolved == "/" {
            return Err(Error::IsDir("/".to_string()));
        }
        let (parent, _) = Self::parent_and_name(&resolved).expect("non-root canonical path");
        self.resolve_directory(&parent)?;
        match self.entries.get_mut(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(contents)) => {
                *contents = data.to_vec();
                Ok(())
            }
            None => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let resolved = self.validate_and_normalize(path)?;
        match self.resolve_entry(&resolved)? {
            Entry::Dir => Err(Error::IsDir(resolved)),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let resolved = self.validate_and_normalize(path)?;
        match self.resolve_entry(&resolved)? {
            Entry::File(_) => return Err(Error::NotDir(resolved)),
            Entry::Dir => {}
        }

        let mut children: Vec<(String, Kind)> = self
            .direct_children(&resolved)
            .into_iter()
            .filter_map(|child| {
                let name = child.rsplit('/').next()?.to_string();
                let kind = match self.entries.get(&child)? {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                Some((name, kind))
            })
            .collect();
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.validate_and_normalize(path)?;
        match self.resolve_entry(&resolved)? {
            Entry::Dir => {
                self.cwd = resolved;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.validate_and_normalize(from)?;
        let destination = self.validate_and_normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, _) =
            Self::parent_and_name(&source).expect("non-root canonical path");

        }

            return Ok(());
        }

            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(target) = &destination_entry {
            match (&source_entry, target) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) => {

            .collect();
        for path in old_paths {
            self.entries.remove(&path);
        }
        for (path, entry) in moving {

        }
        Ok(())
    }

        if resolved == "/" {
            return Err(Error::Root);
        }

        if matches!(entry, Entry::Dir) && Self::contains_path(&resolved, &self.cwd) {

        }

            && !self.direct_children(&resolved).is_empty()

        }

            .keys()
            .filter(|candidate| {

            })
            .cloned()
            .collect();
        for candidate in paths {
            self.entries.remove(&candidate);
        }
        Ok(())
    }
