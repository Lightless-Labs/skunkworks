```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate_and_normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_string()),
            }
        }

        Ok(Self::canonical(&components))
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn parent_and_name(path: &str) -> Option<(String, String)> {
        let components = Self::components(path);
        let name = components.last()?.clone();
        let parent = Self::canonical(&components[..components.len() - 1]);
        Some((parent, name))
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn resolve_directory(&self, path: &str) -> Result<(), Error> {
        if path == "/" {
            return Ok(());
        }
        let mut prefix = String::new();
        for component in Self::components(path) {
            prefix.push('/');
            prefix.push_str(&component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_entry(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self.entries.get("/").ok_or_else(|| Error::Missing("/".into()));
        }
        let (parent, _) = Self::parent_and_name(path).expect("non-root canonical path");
        self.resolve_directory(&parent)?;
        self.entries
            .get(path)
            .ok_or_else(|| Error::Missing(path.to_string()))
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        directory == "/"
            || path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn direct_children(&self, path: &str) -> Vec<String> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{path}/")
        };
        self.entries
            .keys()
            .filter_map(|candidate| {
                if candidate == path || !candidate.starts_with(&prefix) {
                    return None;
                }
                let remainder = &candidate[prefix.len()..];
                if remainder.is_empty() || remainder.contains('/') {
                    None
                } else {
                    Some(candidate.clone())
                }
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.validate_and_normalize(path)?;
        if resolved == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, _) = Self::parent_and_name(&resolved).expect("non-root canonical path");
        self.resolve_directory(&parent)?;
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let resolved = self.validate_and_normalize(path)?;
        if resolved == "/" {
            return Err(Error::IsDir("/".to_string()));
        }
        let (parent, _) = Self::parent_and_name(&resolved).expect("non-root canonical path");
        self.resolve_directory(&parent)?;
        match self.entries.get_mut(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(contents)) => {
                *contents = data.to_vec();
                Ok(())
            }
            None => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let resolved = self.validate_and_normalize(path)?;
        match self.resolve_entry(&resolved)? {
            Entry::Dir => Err(Error::IsDir(resolved)),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let resolved = self.validate_and_normalize(path)?;
        match self.resolve_entry(&resolved)? {
            Entry::File(_) => return Err(Error::NotDir(resolved)),
            Entry::Dir => {}
        }

        let mut children: Vec<(String, Kind)> = self
            .direct_children(&resolved)
            .into_iter()
            .filter_map(|child| {
                let name = child.rsplit('/').next()?.to_string();
                let kind = match self.entries.get(&child)? {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                Some((name, kind))
            })
            .collect();
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.validate_and_normalize(path)?;
        match self.resolve_entry(&resolved)? {
            Entry::Dir => {
                self.cwd = resolved;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.validate_and_normalize(from)?;
        let destination = self.validate_and_normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, _) =
            Self::parent_and_name(&source).expect("non-root canonical path");
        self.resolve_directory(&source_parent)?;
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (destination_parent, _) =
            Self::parent_and_name(&destination).expect("non-root canonical path");
        self.resolve_directory(&destination_parent)?;

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(target) = &destination_entry {
            match (&source_entry, target) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) => {
                    if !self.direct_children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::File(_), Entry::File(_)) => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter_map(|(path, entry)| {
                if path == &source || Self::contains_path(&source, path) {
                    let suffix = &path[source.len()..];
                    Some((format!("{destination}{suffix}"), entry.clone()))
                } else {
                    None
                }
            })
            .collect();

        if destination_entry.is_some() {
            self.entries.remove(&destination);
        }
        let old_paths: Vec<String> = self
            .entries
            .keys()
            .filter(|path| *path == &source || Self::contains_path(&source, path))
            .cloned()
            .collect();
        for path in old_paths {
            self.entries.remove(&path);
        }
        for (path, entry) in moving {
            self.entries.insert(path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let resolved = self.validate_and_normalize(path)?;
        if resolved == "/" {
            return Err(Error::Root);
        }

        let (parent, _) = Self::parent_and_name(&resolved).expect("non-root canonical path");
        self.resolve_directory(&parent)?;
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains_path(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir)
            && !recursive
            && !self.direct_children(&resolved).is_empty()
        {
            return Err(Error::NotEmpty(resolved));
        }

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| {
                *candidate == &resolved
                    || (recursive && Self::contains_path(&resolved, candidate))
            })
            .cloned()
            .collect();
        for candidate in paths {
            self.entries.remove(&candidate);
        }
        Ok(())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
