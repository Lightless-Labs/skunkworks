```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_wide(mut num: i128, mut den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if den < 0 {
        num = -num;
        den = -den;
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let divisor = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    num /= divisor;
    den /= divisor;

    if num < i64::MIN as i128
        || num > i64::MAX as i128
        || den < 1
        || den > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: num as i64,
        den: den as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_wide(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let left = self.num as i128 * rhs.den as i128;
        let right = rhs.num as i128 * self.den as i128;
        let den = self.den as i128 * rhs.den as i128;
        from_wide(left + right, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let left = self.num as i128 * rhs.den as i128;
        let right = rhs.num as i128 * self.den as i128;
        let den = self.den as i128 * rhs.den as i128;
        from_wide(left - right, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_wide(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_wide(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let num = -(self.num as i128);
        if num < i64::MIN as i128 || num > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        // Validate the complete byte grammar before checking canonicality or range.
        let mut i = usize::from(bytes[0] == b'-');
        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }
        let numerator_end = i;

        let slash_at = if i < bytes.len() && bytes[i] == b'/' {
            let at = i;
            i += 1;
            let denominator_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            Some(at)
        } else {
            None
        };

        if i < bytes.len() {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let negative = bytes[0] == b'-';
        let numerator_digits = &bytes[numerator_start..numerator_end];

        // Canonicality checks, in the required order.
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && numerator_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        let denominator_digits = slash_at.map(|slash| &bytes[slash + 1..]);
        if let Some(digits) = denominator_digits {
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_error(slash_at.unwrap() + 2, ParseKind::NonCanonical));
            }
        }

        // Accumulate with a bound check at each digit, so arbitrarily long input
        // cannot overflow the accumulator.
        let numerator_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0u128;
        for &byte in numerator_digits {
            let digit = (byte - b'0') as u128;
            if numerator_magnitude > (numerator_limit - digit) / 10 {
                return Err(parse_error(0, ParseKind::OutOfRange));
            }
            numerator_magnitude = numerator_magnitude * 10 + digit;
        }

        let mut denominator = 1u128;
        if let Some(digits) = denominator_digits {
            for &byte in digits {
                let digit = (byte - b'0') as u128;
                if denominator > (i64::MAX as u128 - digit) / 10 {
                    return Err(parse_error(slash_at.unwrap() + 1, ParseKind::OutOfRange));
                }
                denominator = denominator * 10 + digit;
            }

            let denominator_start = slash_at.unwrap() + 1;
            if denominator == 0 {
                return Err(parse_error(denominator_start, ParseKind::ZeroDenominator));
            }
            if denominator == 1 {
                return Err(parse_error(slash_at.unwrap(), ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_error(slash_at.unwrap(), ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, denominator) > 1 {
                return Err(parse_error(slash_at.unwrap(), ParseKind::NonCanonical));
            }
        }

        let num = if negative {
            if numerator_magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        Ok(Rational {
            num,
            den: denominator as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
