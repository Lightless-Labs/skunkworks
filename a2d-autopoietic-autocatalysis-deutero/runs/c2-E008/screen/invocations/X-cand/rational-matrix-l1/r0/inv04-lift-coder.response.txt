```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }

    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_i128_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_parts(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_parts(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_parts(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        from_i128_parts(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        from_i128_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| Err(RationalError::Parse { at, kind });

        if bytes.is_empty() {
            return fail(0, ParseKind::Empty);
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }

        if i == numerator_start {
            return fail(i, ParseKind::InvalidByte);
        }

        let numerator_end = i;
        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let offset = i;
            i += 1;
            Some(offset)
        } else {
            None
        };

        let denominator_start = i;
        if slash.is_some() {
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start {
                return fail(i, ParseKind::InvalidByte);
            }
        }

        if i < bytes.len() {
            return fail(i, ParseKind::InvalidByte);
        }

        let numerator_digits = &bytes[numerator_start..numerator_end];

        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return fail(numerator_start + 1, ParseKind::NonCanonical);
        }
        if negative && numerator_digits == b"0" {
            return fail(0, ParseKind::NonCanonical);
        }

        let denominator_digits = slash.map(|_| &bytes[denominator_start..]);

        if let Some(digits) = denominator_digits {
            if digits.len() >= 2 && digits[0] == b'0' {
                return fail(denominator_start + 1, ParseKind::NonCanonical);
            }
        }

        let numerator_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut magnitude = 0u128;
        for &byte in numerator_digits {
            magnitude = magnitude * 10 + (byte - b'0') as u128;
            if magnitude > numerator_limit {
                return fail(0, ParseKind::OutOfRange);
            }
        }

        let mut denominator = 1u128;
        if let Some(digits) = denominator_digits {
            for &byte in digits {
                denominator = denominator * 10 + (byte - b'0') as u128;
                if denominator > i64::MAX as u128 {
                    return fail(denominator_start, ParseKind::OutOfRange);
                }
            }

            if digits == b"0" {
                return fail(denominator_start, ParseKind::ZeroDenominator);
            }

            let slash_offset = slash.unwrap();
            if denominator == 1 {
                return fail(slash_offset, ParseKind::NonCanonical);
            }
            if magnitude == 0 {
                return fail(slash_offset, ParseKind::NonCanonical);
            }
            if gcd(magnitude, denominator) > 1 {
                return fail(slash_offset, ParseKind::NonCanonical);
            }
        }

        let numerator = if negative {
            if magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(magnitude as i64)
            }
        } else {
            magnitude as i64
        };

        Ok(Rational {
            num: numerator,
            den: denominator as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
