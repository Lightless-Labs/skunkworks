struct Traversal<'a> {
    stack: Vec<String>,
    links: &'a [Link],
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl<'a> Traversal<'a> {
    fn render_stack(&self) -> String {
        format!("/{}", self.stack.join("/"))
    }

    fn find_link(&self) -> Option<usize> {
        let current = self.render_stack();
        self.links.iter().position(|link| link.path == current)
    }

    fn expand_link(&mut self, index: usize) -> Result<(), PathError> {
        if self.active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if self.expansions >= self.max_expansions {
            return Err(PathError::Limit { index });
        }

        self.expansions += 1;
        self.active.push(index);

        let target = self.links[index].target.as_str();
        let target_absolute = target.starts_with('/');
        if target_absolute {
            self.stack.clear();
        } else {
            self.stack.pop();
        }

        let result = self.process_string(target, true, 0, true);
        self.active.pop();
        result
    }

    fn process_component(
        &mut self,
        component: &str,
        at: usize,
        absolute: bool,
        lookup: bool,
    ) -> Result<(), PathError> {
        push_component(&mut self.stack, component, absolute, at)?;
        if lookup && component != "." && component != ".." && !component.is_empty() {
            if let Some(index) = self.find_link() {
                self.expand_link(index)?;
            }
        }
        Ok(())
    }

    fn process_string(
        &mut self,
        raw: &str,
        absolute: bool,
        base_offset: usize,
        lookup: bool,
    ) -> Result<(), PathError> {
        let mut offset = base_offset;
        for component in raw.split('/') {
            if !component.is_empty() {
                self.process_component(component, offset, absolute, lookup)?;
            }
            offset += component.len() + 1;
        }
        Ok(())
    }
}

pub fn resolve_links(

    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {

    let home_components = match home {
        Some(home) => Some(validate_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,

    validate_links(links)?;
    validate_raw(path)?;
    let (expanded, request) = tilde_request(path, home)?;

    let absolute = path.starts_with('/');
    let intent = directory_intent(path);

    let mut traversal = Traversal {
        stack: Vec::new(),
        links,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    if !absolute {
        let starting = if expanded {
            home_components.unwrap_or_default()
        } else {
            cwd_components
        };
        for component in starting {
            traversal.process_component(&component, 0, true, true)?;
        }
    }

    let base_offset = if expanded {
        if path.len() > 1 { 2 } else { 1 }
    } else {
        0
    };
    traversal.process_string(request, true, base_offset, true)?;

    Ok(render(&traversal.stack, true, intent))
}

    pub target: String,

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn append_directory_slash(mut result: String, directory: bool) -> String {
    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn render_absolute(components: &[String], directory: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&components.join("/"));
    append_directory_slash(result, directory)
}

fn render_relative(components: &[String], directory: bool) -> String {
    let mut result = if components.is_empty() {
        String::from(".")
    } else {
        components.join("/")
    };
    append_directory_slash(result, directory)
}

fn components(path: &str) -> impl Iterator<Item = (usize, &str)> {
    path.split('/').scan(0usize, |offset, component| {
        let start = *offset;
        *offset += component.len() + 1;
        Some((start, component))
    })
}

fn normalize_parts(path: &str) -> Result<(bool, Vec<String>), PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let mut stack: Vec<String> = Vec::new();

    for (at, component) in components(path) {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|previous| previous != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push(String::from(".."));
            }
        } else {
            stack.push(component.to_owned());
        }
    }

    Ok((absolute, stack))
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    let directory = directory_intent(path);
    let (absolute, stack) = normalize_parts(path)?;

    if absolute {
        Ok(render_absolute(&stack, directory))
    } else {
        Ok(render_relative(&stack, directory))
    }
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize_parts(path) {
        Ok((true, components)) => Ok(components),
        _ => Err(error),
    }
}

fn requested_start<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, &'a str, usize, Vec<String>), PathError> {
    if path.starts_with('~') {
        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_components = match home {
            Some(value) => match normalize_parts(value) {
                Ok((true, parts)) => parts,
                _ => return Err(PathError::InvalidHome),
            },
            None => return Err(PathError::MissingHome),
        };
        let remainder_start = if path.len() >= 2 { 2 } else { 1 };
        Ok((
            false,
            &path[remainder_start..],
            remainder_start,
            home_components,
        ))
    } else if path.starts_with('/') {
        Ok((true, path, 0, Vec::new()))
    } else {
        Ok((false, path, 0, Vec::new()))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    if path.starts_with('~') && path.len() > 1 && path.as_bytes()[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    let directory = directory_intent(path);
    let (absolute, remainder, offset, home_for_tilde) =
        requested_start(path, home)?;
    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        home_for_tilde
    } else {
        cwd_components
    };

    for (local_at, component) in components(remainder) {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape {
                    at: offset + local_at,
                });
            }
        } else {
            stack.push(component.to_owned());
        }
    }

    Ok(render_absolute(&stack, directory))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut table: HashMap<String, (usize, String)> = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let canonical = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !canonical || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((first, _)) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        table.insert(link.path.clone(), (index, link.target.clone()));
    }

    Ok(table)
}

    while let Some(component) = pending.pop() {
        if component.value.is_empty() || component.value == "." {
            continue;
        }
        if component.value == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at: component.at });
            }
            continue;
        }

        stack.push(component.value);
        let key = absolute_key(stack);
        if let Some((index, target)) = table.get(&key) {
            if active.contains(index) {
                return Err(PathError::Loop { index: *index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index: *index });
            }
            *expansions += 1;
            active.push(*index);

            let absolute_target = target.starts_with('/');
            if absolute_target {
                stack.clear();
            } else {
                stack.pop();
            }

            let mut target_work: Vec<WorkComponent> = components(target)
                .filter(|(_, value)| !value.is_empty())
                .map(|(at, value)| WorkComponent {
                    value: value.to_owned(),
                    at,
                })
                .collect();
            target_work.reverse();

            // Preserve the not-yet-processed request components while the target runs.
            let remaining = std::mem::take(&mut pending);
            process_work(
                target_work,
                stack,
                table,
                active,
                expansions,
                max_expansions,
            )?;
            active.pop();
            pending = remaining;
        }
    }
    Ok(())
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components_with_offsets(path: &str) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let mut start = None;

    for (i, byte) in path.bytes().enumerate() {
        if byte == b'/' {
            if let Some(begin) = start.take() {
                result.push((begin, &path[begin..i]));
            }
        } else if start.is_none() {
            start = Some(i);
        }
    }

    if let Some(begin) = start {
        result.push((begin, &path[begin..]));
    }
    result
}

fn reduce_components(
    stack: &mut Vec<String>,
    path: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, component) in components_with_offsets(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: offset_base + at,
                    });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(component.to_string()),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        result = "/".to_string();
    }

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce_components(&mut stack, path, 0, absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    match normalize(path) {
        Ok(normalized) => Ok(normalized
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()),
        Err(_) => Err(PathError::InvalidCwd),
    }
}

fn normalized_home(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidHome);
    }
    match normalize(path) {
        Ok(normalized) => Ok(normalized
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()),
        Err(_) => Err(PathError::InvalidHome),
    }
}

fn tilde_remainder<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<Option<&'a str>, PathError> {
    if !path.starts_with('~') {
        return Ok(None);
    }

    let bytes = path.as_bytes();
    if bytes.len() > 1 && bytes[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if home.is_none() {
        return Err(PathError::MissingHome);
    }

    Ok(Some(if bytes.len() > 1 { &path[2..] } else { "" }))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd)?;
    let home_components = match home {
        Some(path) => Some(normalized_home(path)?),
        None => None,
    };

    validate_raw(path)?;

    let intent = directory_intent(path);
    let mut stack;
    let mut remainder = path;
    let absolute;

    if let Some(rest) = tilde_remainder(path, home_components.as_deref())? {
        stack = home_components.unwrap_or_default();
        remainder = rest;
        absolute = true;
    } else if path.starts_with('/') {
        stack = Vec::new();
        absolute = true;
    } else {
        stack = cwd_components;
        absolute = true;
    }

    reduce_components(&mut stack, remainder, 0, absolute)?;
    Ok(render(&stack, true, intent))
}

enum Source<'a> {
    Request(&'a str),
    Target(&'a str),
    Context,
}

#[derive(Clone, Copy)]
struct Component<'a> {
    value: &'a str,
    at: usize,
    source: Source<'a>,
}

fn components_for<'a>(path: &'a str, source: Source<'a>) -> Vec<Component<'a>> {
    components_with_offsets(path)
        .into_iter()
        .map(|(at, value)| Component { value, at, source })
        .collect()

fn link_index(path: &str, links: &[Link]) -> Option<usize> {
    links.iter().position(|link| link.path == path)
}

fn apply_component(
    stack: &mut Vec<String>,
    component: &Component<'_>,
) -> Result<(), PathError> {
    match component.value {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else {
                let at = match component.source {
                    Source::Context => 0,
                    Source::Request(_) | Source::Target(_) => component.at,
                };
                return Err(PathError::Escape { at });
            }
        }
        _ => stack.push(component.value.to_string()),
    }
    Ok(())
}

fn traverse_components(
    stack: &mut Vec<String>,
    components: &[Component<'_>],
    links: &[Link],
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    for (position, component) in components.iter().enumerate() {
        match component.value {
            "." => continue,
            ".." => {
                apply_component(stack, component)?;
                continue;
            }
            _ => stack.push(component.value.to_string()),
        }

        let candidate = format!("/{}", stack.join("/"));
        if let Some(index) = link_index(&candidate, links) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            *expansions += 1;
            active.push(index);

            stack.pop();
            let target = &links[index].target;
            let target_absolute = target.starts_with('/');
            if target_absolute {
                stack.clear();
            }
            let target_components =
                components_for(target, Source::Target(target));
            traverse_components(
                stack,
                &target_components,
                links,
                active,
                expansions,
                max_expansions,
            )?;
            active.pop();

            // The target is complete before the remainder of this source resumes.
            // The loop naturally processes the next original component.
        }

        let _ = position;
    }
    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd)?;
    let home_components = match home {
        Some(path) => Some(normalized_home(path)?),
        None => None,
    };

    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path)
            && link.path != "/";
        if !path_valid || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }

    validate_raw(path)?;
    let intent = directory_intent(path);

    let mut stack;
    let mut request_remainder = path;
    if let Some(rest) = tilde_remainder(path, home_components.as_deref())? {
        stack = home_components.unwrap_or_default();
        request_remainder = rest;
    } else if path.starts_with('/') {
        stack = Vec::new();
    } else {
        stack = cwd_components;
    }

    let mut expansions = 0usize;
    let mut active = Vec::new();

    // Starting-directory components are already normalized and cannot contain
    // dot or parent components. Each has context offset zero.
    let context_components: Vec<Component<'_>> = stack
        .iter()
        .map(|value| Component {
            value,
            at: 0,
            source: Source::Context,
        })
        .collect();

    if !path.starts_with('/') && !path.starts_with('~') {
        stack.clear();
        traverse_components(
            &mut stack,
            &context_components,
            links,
            &mut active,
            &mut expansions,
            max_expansions,
        )?;
    }

    let request_components = components_for(request_remainder, Source::Request(path));
    traverse_components(
        &mut stack,
        &request_components,
        links,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

    Ok(render(&stack, true, intent))
