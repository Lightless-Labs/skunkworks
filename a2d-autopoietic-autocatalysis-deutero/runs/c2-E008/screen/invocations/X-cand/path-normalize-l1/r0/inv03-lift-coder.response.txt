#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();

    if let Some(at) = bytes.iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = bytes.iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = bytes
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = bytes[0] == b'/';
    let directory_intent = path.ends_with('/');
    let mut last_nonempty: Option<&str> = None;
    let mut components: Vec<&str> = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        let component_start = offset;
        offset += component.len() + 1;

        if component.is_empty() {
            continue;
        }

        last_nonempty = Some(component);

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|name| *name != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component_start });
                } else {
                    components.push(component);
                }
            }
            _ => components.push(component),
        }
    }

    let directory_intent =
        directory_intent || matches!(last_nonempty, Some(".") | Some(".."));

    let joined = components.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
