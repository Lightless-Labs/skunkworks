```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    pause_time: Option<u64>,
    remaining: u64,
}

struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn better(a: &Job, b: &Job) -> bool {
    let pa = a.customer.priority;
    let pb = b.customer.priority;
    if pa != pb {
        return pa < pb;
    }

    match (a.pause_time, b.pause_time) {
        (Some(ta), Some(tb)) => {
            ta < tb || (ta == tb && a.customer.id < b.customer.id)
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            a.customer.arrival < b.customer.arrival
                || (a.customer.arrival == b.customer.arrival
                    && a.customer.id > b.customer.id)
        }
    }
}

fn best_waiter_index(waiting: &[Job]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.map_or(true, |j| better(&waiting[i], &waiting[j])) {
            best = Some(i);
        }
    }
    best
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut ids = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;
    let mut now = 0u64;

    while finished_count < customers.len() {
        // Complete a scheduled finish before processing anything else at this tick.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let r = running.take().unwrap();
            events.push(Event {
                time: now,
                customer: r.job.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        // Admit the whole cohort in sorted order before making dispatch decisions.
        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == now
        {
            let customer = customers[arrivals[next_arrival]].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        // Protection expires at work_at; reconsider preemption after this tick's arrivals.
        if let Some(r) = running.as_ref() {
            if now >= r.work_at && now < r.finish {
                if let Some(index) = best_waiter_index(&waiting) {
                    if waiting[index].customer.priority < r.job.customer.priority {
                        let waiter_id = waiting[index].customer.id;
                        let mut r = running.take().unwrap();
                        let remaining = r.finish - now;
                        events.push(Event {
                            time: now,
                            customer: r.job.customer.id,
                            kind: Kind::Preempt {
                                remaining,
                                by: waiter_id,
                            },
                        });
                        r.job.remaining = remaining;
                        r.job.pause_time = Some(now);
                        waiting.push(r.job);
                    }
                }
            }
        }

        // Dispatch until the server is occupied or there are no waiters.
        while running.is_none() {
            let Some(index) = best_waiter_index(&waiting) else {
                break;
            };
            let mut job = waiting.remove(index);
            let id = job.customer.id;
            let cost = if last_dispatched.is_some_and(|previous| previous != id) {
                switch_cost
            } else {
                0
            };
            let work_at = now
                .checked_add(cost)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                })?;
            let finish = work_at
                .checked_add(job.remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                })?;

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            job.started = true;
            job.pause_time = None;
            last_dispatched = Some(id);
            events.push(Event {
                time: now,
                customer: id,
                kind,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            } else {
                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time = None;
        if next_arrival < arrivals.len() {
            next_time = Some(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(r) = running.as_ref() {
            for t in [r.finish, r.work_at] {
                if t > now && next_time.map_or(true, |n| t < n) {
                    next_time = Some(t);
                }
            }
        }

        // If no future event remains, the queue and server must also be empty.
        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    Ok(events)
}
```
