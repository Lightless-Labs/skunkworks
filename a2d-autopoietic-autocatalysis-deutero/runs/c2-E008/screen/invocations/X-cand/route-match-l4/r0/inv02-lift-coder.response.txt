use std::collections::{BTreeSet, HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(bytes: &[u8], base: usize, pattern: bool) -> Result<String, usize> {
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }
    if !pattern && out.is_empty() {
        return Err(base);
    }
    String::from_utf8(out).map_err(|_| base)
}

fn split_segments<'a>(input: &'a str, pattern: bool) -> Result<(Vec<(usize, &'a str)>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut pieces = Vec::new();
    let mut start = 1;
    let mut trailing = false;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(start);
                }
                trailing = true;
                break;
            }
            if i == start {
                return Err(start);
            }
            pieces.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if !trailing {
        if start == bytes.len() {
            return Err(start);
        }
        pieces.push((start, &input[start..]));
    }

    // A final slash on a non-root path is kept separately.
    let _ = pattern;
    Ok((pieces, trailing))
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), usize> {
    let (pieces, trailing) = split_segments(pattern, true)?;
    let mut segments = Vec::with_capacity(pieces.len());
    let mut names = HashSet::new();

    for (index, (at, raw)) in pieces.iter().enumerate() {
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_capture_name(name) {
                return Err(*at);
            }
            if !names.insert(name.to_string()) {
                return Err(*at);
            }
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_capture_name(name) {
                return Err(*at);
            }
            if index + 1 != pieces.len() {
                return Err(*at);
            }
            if !names.insert(name.to_string()) {
                return Err(*at);
            }
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            let decoded = decode_segment(raw.as_bytes(), *at, true)?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok((segments, trailing))
}

fn parse_request(path: &str) -> Result<(Vec<String>, bool), usize> {
    let end = path
        .bytes()
        .position(|b| b == b'?' || b == b'#')
        .unwrap_or(path.len());
    let head = &path[..end];
    let (pieces, trailing) = split_segments(head, false)?;
    let mut decoded = Vec::with_capacity(pieces.len());
    for (at, raw) in pieces {
        decoded.push(decode_segment(raw.as_bytes(), at, false)?);
    }
    Ok((decoded, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn rank_for(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));

    if let Some(wi) = wildcard_index {
        if request.len() < wi {
            return None;
        }
    } else if request.len() != route.segments.len() {
        return None;
    }

    let mut captures = Vec::new();
    for (i, segment) in route.segments.iter().enumerate() {
        match segment {
            Segment::Static(expected) => {
                if request.get(i)? != expected {
                    return None;
                }
            }
            Segment::Param(name) => {
                let value = request.get(i)?;
                captures.push((name.clone(), value.clone()));
            }
            Segment::Wildcard(name) => {
                let value = request[i..].join("/");
                captures.push((name.clone(), value));
                break;
            }
        }
    }
    Some(captures)
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)
            .map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let candidate = Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            rank: rank_for(&segments),
            segments,
            trailing,
        };

        if let Some(existing) = self
            .routes
            .iter()
            .filter(|route| shape_conflicts(route, &candidate, self.policy))
            .map(|route| route.id)
            .min()
        {
            return Err(Error::Conflict { with: existing });
        }

        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (request, trailing) =
            parse_request(path).map_err(|at| Error::InvalidPath { at })?;

        let candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| match_route(route, &request).map(|params| (route, params)))
            .filter(|(route, _)| {
                self.policy != SlashPolicy::Strict || route.trailing == trailing
            })
            .collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_string());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id)))
            .unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect
            && !request.is_empty()
            && winner.trailing != trailing
        {
            let head_end = path
                .bytes()
                .position(|b| b == b'?' || b == b'#')
                .unwrap_or(path.len());
            let mut target = path[..head_end].to_string();
            if winner.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|route| route.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = HashMap::new();
        let mut seen = HashSet::new();
        for &(key, value) in params {
            if !seen.insert(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.insert(key, value);
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => capture_names.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &capture_names {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let capture_set: HashSet<&str> = capture_names.iter().copied().collect();
        if let Some(extra) = supplied
            .keys()
            .filter(|key| !capture_set.contains(**key))
            .min()
        {
            return Err(Error::ExtraParam {
                name: (*extra).to_string(),
            });
        }

        for segment in &route.segments {
            let (name, wildcard) = match segment {
                Segment::Param(name) => (name.as_str(), false),
                Segment::Wildcard(name) => (name.as_str(), true),
                Segment::Static(_) => continue,
            };
            let value = supplied[name];
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid = if wildcard {
                valid_ascii
                    && (value.is_empty()
                        || bytes
                            .split(|b| *b == b'/')
                            .all(|part| !part.is_empty()))
            } else {
                valid_ascii && !value.is_empty() && !value.contains('/')
            };
            if !valid {
                return Err(Error::InvalidValue {
                    name: name.to_string(),
                });
            }
        }

        fn encode_component(input: &str) -> String {
            let mut out = String::new();
            for b in input.bytes() {
                if unreserved(b) {
                    out.push(b as char);
                } else {
                    const HEX: &[u8; 16] = b"0123456789ABCDEF";
                    out.push('%');
                    out.push(HEX[(b >> 4) as usize] as char);
                    out.push(HEX[(b & 0x0f) as usize] as char);
                }
            }
            out
        }

        let mut output = String::from("/");
        let mut emitted = 0usize;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(value));
                    emitted += 1;
                }
                Segment::Param(name) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(supplied[name]));
                    emitted += 1;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name];
                    if !value.is_empty() {
                        if emitted > 0 {
                            output.push('/');
                        }
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part));
                        }
                        emitted += value.split('/').count();
                    }
                }
            }
        }

        if output != "/" && route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}
