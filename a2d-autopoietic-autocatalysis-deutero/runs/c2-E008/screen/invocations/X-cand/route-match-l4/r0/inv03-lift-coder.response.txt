#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-'
        })
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(bytes: &[u8], base: usize) -> Result<String, usize> {
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !unreserved(b) {
                return Err(base + i);
            }
            decoded.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segment is ASCII"))
}

fn split_path(input: &str) -> Result<(Vec<(usize, &str)>, bool), usize> {
    if input.is_empty() || !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut pieces = Vec::new();
    let mut start = 1;

    for i in 1..end {
        if bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            pieces.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if start == end {
        return Err(start);
    }
    pieces.push((start, &input[start..end]));

    Ok((pieces, trailing))
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), usize> {
    let (pieces, trailing) = split_path(input)?;
    let mut segments = Vec::with_capacity(pieces.len());
    let mut names = std::collections::HashSet::new();
    let mut rank = Vec::new();

    for (index, (at, raw)) in pieces.iter().enumerate() {
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(*at);
            }
            segments.push(Segment::Param(name.to_string()));
            rank.push(1);
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(*at);
            }
            if index + 1 != pieces.len() {
                return Err(*at);
            }
            segments.push(Segment::Wildcard(name.to_string()));
            rank.push(0);
            break;
        } else {
            let value = decode_segment(raw.as_bytes(), *at)?;
            segments.push(Segment::Static(value));
            rank.push(2);
        }
    }

    if !segments.iter().any(|segment| matches!(segment, Segment::Wildcard(_))) {
        rank.push(3);
    }

    Ok((segments, trailing, rank))
}

fn parse_request(input: &str) -> Result<(Vec<String>, bool), usize> {
    let end = input
        .bytes()
        .position(|b| b == b'?' || b == b'#')
        .unwrap_or(input.len());
    let head = &input[..end];
    let (pieces, trailing) = split_path(head)?;
    let mut segments = Vec::with_capacity(pieces.len());

    for (at, raw) in pieces {
        segments.push(decode_segment(raw.as_bytes(), at)?);
    }

    Ok((segments, trailing))
}

fn same_shape(a: &Route, method: &str, segments: &[Segment], trailing: bool, policy: SlashPolicy) -> bool {
    if a.method != method || a.segments.len() != segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != trailing {
        return false;
    }
    a.segments.iter().zip(segments).all(|(left, right)| match (left, right) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_at = route
        .segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));

    match wildcard_at {
        Some(index) if request.len() < index => return None,
        None if request.len() != route.segments.len() => return None,
        _ => {}
    }

    let mut params = Vec::new();
    for (index, segment) in route.segments.iter().enumerate() {
        match segment {
            Segment::Static(expected) => {
                if request.get(index)? != expected {
                    return None;
                }
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(index)?.clone()));
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), request[index..].join("/")));
                break;
            }
        }
    }
    Some(params)
}

fn encode_component(input: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut output = String::new();

    for byte in input.bytes() {
        if unreserved(byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }

    output
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing, rank) =
            parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;

        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        if let Some(id) = self
            .routes
            .iter()
            .filter(|route| same_shape(route, method, &segments, trailing, self.policy))
            .map(|route| route.id)
            .min()
        {
            return Err(Error::Conflict { with: id });
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (request, request_trailing) =
            parse_request(path).map_err(|at| Error::InvalidPath { at })?;

        let candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| {
                if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                    return None;
                }
                match_route(route, &request).map(|params| (route, params))
            })
            .collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates.iter().any(|(route, _)| route.method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_string());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id))
        });
        let (winner, params) = eligible[0];

        let redirect = if self.policy == SlashPolicy::Redirect
            && !request.is_empty()
            && winner.trailing != request_trailing
        {
            let mut target = path[..path
                .bytes()
                .position(|b| b == b'?' || b == b'#')
                .unwrap_or(path.len())]
                .to_string();
            if winner.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for &(key, value) in params {
            if supplied.contains_key(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.insert(key, value);
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &required {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let required_set: std::collections::HashSet<&str> =
            required.iter().copied().collect();
        if let Some(extra) = supplied
            .keys()
            .filter(|key| !required_set.contains(**key))
            .min()
        {
            return Err(Error::ExtraParam {
                name: (*extra).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = supplied[name.as_str()];
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || (!value.is_empty()
                            && value.split('/').any(|part| part.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut output = String::from("/");
        let mut emitted = false;

        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_component(value));
                    emitted = true;
                }
                Segment::Param(name) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_component(supplied[name.as_str()]));
                    emitted = true;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        if emitted {
                            output.push('/');
                        }
                        for (index, component) in value.split('/').enumerate() {
                            if index > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(component));
                        }
                        emitted = true;
                    }
                }
            }
        }

        if route.trailing && emitted {
            output.push('/');
        }

        Ok(output)
    }
}
