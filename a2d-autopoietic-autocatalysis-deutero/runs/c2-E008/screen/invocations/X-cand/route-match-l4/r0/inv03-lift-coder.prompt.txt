#[derive(Clone)]
enum Segment {

struct Route {

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(s: &str) -> bool {
    let mut it = s.bytes();
    matches!(it.next(), Some(b'a'..=b'z'))
        && it.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(s: &str) -> bool {
    let mut it = s.bytes();
    matches!(it.next(), Some(b'a'..=b'z'))
        && it.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_segment(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex(bytes[i + 2]).ok_or(base + i)?;
            let d = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&d) || d == b'/' || unreserved(d) {
                return Err(base + i);
            }
            out.push(d);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn parse_path(path: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let mut body_start = 1;
    let trailing = bytes.last() == Some(&b'/');
    let body_end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();

    while body_start < body_end {
        let end = bytes[body_start..body_end]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| body_start + p)
            .unwrap_or(body_end);
        if end == body_start {
            return Err(body_start);
        }
        let raw = &path[body_start..end];

        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(body_start);
            }
            if marker == b'*' && end != body_end {
                return Err(body_start);
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            });
        } else {
            let decoded = decode_segment(raw, body_start)?;
            segments.push(Segment::Static(decoded));
        }
        body_start = end + 1;
    }

    Ok((segments, trailing))
}

fn encode_segment(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method != method {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if route.segments.len() != segments.len() {
                continue;
            }
            let same = route.segments.iter().zip(&segments).all(|(a, b)| match (a, b) {
                (Segment::Static(x), Segment::Static(y)) => x == y,
                (Segment::Param(_), Segment::Param(_)) => true,
                (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                _ => false,
            });
            if same {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let mut rank = Vec::new();
        for seg in &segments {
            match seg {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => rank.push(0),
            }
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) =
            parse_path(head, false).map_err(|at| Error::InvalidPath { at })?;

        let decoded: Vec<String> = request_segments
            .into_iter()
            .map(|s| match s {
                Segment::Static(v) => v,
                _ => unreachable!(),
            })
            .collect();

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            let mut params = Vec::new();
            let mut i = 0;
            let mut matched = true;
            for seg in &route.segments {
                match seg {
                    Segment::Static(value) => {
                        if decoded.get(i) != Some(value) {
                            matched = false;
                            break;
                        }
                        i += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = decoded.get(i) {
                            params.push((name.clone(), value.clone()));
                            i += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = decoded[i..].join("/");
                        params.push((name.clone(), value));
                        i = decoded.len();
                    }
                }
            }
            if matched && i == decoded.len() {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            let has_head = candidates.iter().any(|(r, _)| r.method == "HEAD");
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| {
                    if r.method == if has_head { "HEAD" } else { "GET" } {
                        Some(i)
                    } else {
                        None
                    }
                })
                .collect()
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| (r.method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates.iter().map(|(r, _)| r.method.clone()).collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let winner_index = *eligible
            .iter()
            .max_by(|&&a, &&b| {
                candidates[a]
                    .0
                    .rank
                    .cmp(&candidates[b].0.rank)
                    .then_with(|| candidates[b].0.id.cmp(&candidates[a].0.id))
            })
            .unwrap();
        let (route, params) = &candidates[winner_index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !decoded.is_empty()
        {
            let mut target = head.to_string();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut required = Vec::new();
        for seg in &route.segments {
            match seg {
                Segment::Param(n) | Segment::Wildcard(n) => required.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }
        let extra = supplied
            .keys()
            .filter(|k| !required.contains(k))
            .min()
            .copied();
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: key.to_string(),
            });
        }

        for seg in &route.segments {
            match seg {
                Segment::Param(n) => {
                    let value = supplied[n];
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Wildcard(n) => {
                    let value = supplied[n];
                    let valid = value.is_empty()
                        || (value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                            && value.split('/').all(|part| !part.is_empty()));
                    if !valid {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut out = String::from("/");
        let mut first = true;
        for seg in &route.segments {
            match seg {
                Segment::Static(v) => {
                    if !first {
                        out.push('/');
                    }
                    out.push_str(&encode_segment(v));
                    first = false;
                }
                Segment::Param(n) => {
                    if !first {
                        out.push('/');
                    }
                    out.push_str(&encode_segment(supplied[n]));
                    first = false;
                }
                Segment::Wildcard(n) => {
                    let value = supplied[n];
                    if !value.is_empty() {
                        if !first {
                            out.push('/');
                        }
                        out.push_str(
                            &value
                                .split('/')
                                .map(encode_segment)
                                .collect::<Vec<_>>()
                                .join("/"),
                        );
                        first = false;
                    }
                }
            }
        }
        if route.trailing && out != "/" {
            out.push('/');
        }
        Ok(out)
    }

fn split_segments<'a>(input: &'a str, pattern: bool) -> Result<(Vec<(usize, &'a str)>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut pieces = Vec::new();
    let mut start = 1;
    let mut trailing = false;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {

                if start == i {
                    return Err(start);
                }
                trailing = true;
                break;

            if i == start {
                return Err(start);

    for (i, segment) in route.segments.iter().enumerate() {
        match segment {
            Segment::Static(expected) => {
                if request.get(i)? != expected {
                    return None;
                }
            }
            Segment::Param(name) => {
                let value = request.get(i)?;
                captures.push((name.clone(), value.clone()));
            }
            Segment::Wildcard(name) => {
                let value = request[i..].join("/");
                captures.push((name.clone(), value));
                break;
            }
        }
    }
    Some(captures)
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)
            .map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let candidate = Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            rank: rank_for(&segments),
            segments,
            trailing,
        };

        if let Some(existing) = self
            .routes
            .iter()
            .filter(|route| shape_conflicts(route, &candidate, self.policy))
            .map(|route| route.id)
            .min()
        {
            return Err(Error::Conflict { with: existing });
        }

        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (request, trailing) =
            parse_request(path).map_err(|at| Error::InvalidPath { at })?;

        let candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| match_route(route, &request).map(|params| (route, params)))
            .filter(|(route, _)| {
                self.policy != SlashPolicy::Strict || route.trailing == trailing
            })
            .collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_string());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id)))
            .unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect
            && !request.is_empty()
            && winner.trailing != trailing
        {
            let head_end = path
                .bytes()
                .position(|b| b == b'?' || b == b'#')
                .unwrap_or(path.len());
            let mut target = path[..head_end].to_string();
            if winner.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|route| route.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = HashMap::new();
        let mut seen = HashSet::new();
        for &(key, value) in params {
            if !seen.insert(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.insert(key, value);
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => capture_names.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &capture_names {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let capture_set: HashSet<&str> = capture_names.iter().copied().collect();
        if let Some(extra) = supplied
            .keys()
            .filter(|key| !capture_set.contains(**key))
            .min()
        {
            return Err(Error::ExtraParam {
                name: (*extra).to_string(),
            });
        }

        for segment in &route.segments {
            let (name, wildcard) = match segment {
                Segment::Param(name) => (name.as_str(), false),
                Segment::Wildcard(name) => (name.as_str(), true),
                Segment::Static(_) => continue,
            };
            let value = supplied[name];
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid = if wildcard {
                valid_ascii
                    && (value.is_empty()
                        || bytes
                            .split(|b| *b == b'/')
                            .all(|part| !part.is_empty()))
            } else {
                valid_ascii && !value.is_empty() && !value.contains('/')
            };
            if !valid {
                return Err(Error::InvalidValue {
                    name: name.to_string(),
                });
            }
        }

        fn encode_component(input: &str) -> String {
            let mut out = String::new();
            for b in input.bytes() {
                if unreserved(b) {
                    out.push(b as char);
                } else {
                    const HEX: &[u8; 16] = b"0123456789ABCDEF";
                    out.push('%');
                    out.push(HEX[(b >> 4) as usize] as char);
                    out.push(HEX[(b & 0x0f) as usize] as char);
                }
            }
            out
        }

        let mut output = String::from("/");
        let mut emitted = 0usize;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(value));
                    emitted += 1;
                }
                Segment::Param(name) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(supplied[name]));
                    emitted += 1;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name];
                    if !value.is_empty() {
                        if emitted > 0 {
                            output.push('/');
                        }
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part));
                        }
                        emitted += value.split('/').count();
                    }
                }
            }
        }

        if output != "/" && route.trailing {
            output.push('/');
        }
        Ok(output)
    }
