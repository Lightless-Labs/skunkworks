#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();

    let mut now = 0u64;
    let mut event_pos = 0usize;
    let mut last_dispatched: Option<u32> = None;

    fn admit(
        jobs: &[Job],
        events: &[usize],
        event_pos: &mut usize,
        through: u64,
        remaining: &[u32],
        completion: &mut [Option<u64>],
        ready: &mut std::collections::VecDeque<usize>,
    ) {
        while *event_pos < events.len() {
            let index = events[*event_pos];
            let job = &jobs[index];
            if u64::from(job.arrival) > through {
                break;
            }
            *event_pos += 1;
            if remaining[index] == 0 {
                completion[index] = Some(u64::from(job.arrival));
            } else {
                ready.push_back(index);
            }
        }
    }

    loop {
        admit(
            jobs,
            &events,
            &mut event_pos,
            now,
            &remaining,
            &mut completion,
            &mut ready,
        );

        if let Some(index) = ready.pop_front() {
            let job_id = jobs[index].id;

            if last_dispatched.is_some_and(|last| last != job_id) {
                now += u64::from(switch_cost);
            }

            admit(
                jobs,
                &events,
                &mut event_pos,
                now,
                &remaining,
                &mut completion,
                &mut ready,
            );

            let consumed = remaining[index].min(quantum);
            let start = now;
            now += u64::from(consumed);
            remaining[index] -= consumed;
            let finished = remaining[index] == 0;

            runs.push(Run {
                id: job_id,
                start,
                end: now,
                finished,
            });

            admit(
                jobs,
                &events,
                &mut event_pos,
                now,
                &remaining,
                &mut completion,
                &mut ready,
            );

            if finished {
                completion[index] = Some(now);
            } else {
                ready.push_back(index);
            }
            last_dispatched = Some(job_id);
        } else if event_pos < events.len() {
            let next_time = u64::from(jobs[events[event_pos]].arrival);
            if next_time > now {
                if let Some(last_idle) = idle.last_mut() {
                    if last_idle.1 == now {
                        last_idle.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatched = None;
            }
        } else {
            break;
        }
    }

    let completion: Vec<(u32, u64)> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}
