```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();

    let mut event_pos = 0usize;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;

    loop {
        while event_pos < events.len() {
            let index = events[event_pos];
            let job = &jobs[index];
            let arrival = u64::from(job.arrival);
            if arrival > time {
                break;
            }

            event_pos += 1;
            if remaining[index] == 0 {
                completion[index] = Some(arrival);
            } else {
                ready.push_back(index);
            }
        }

        if let Some(index) = ready.pop_front() {
            let job_id = jobs[index].id;

            if last_dispatched_id.is_some_and(|previous| previous != job_id) {
                time += u64::from(switch_cost);
            }

            while event_pos < events.len() {
                let arriving_index = events[event_pos];
                let arriving_job = &jobs[arriving_index];
                let arrival = u64::from(arriving_job.arrival);
                if arrival > time {
                    break;
                }

                event_pos += 1;
                if remaining[arriving_index] == 0 {
                    completion[arriving_index] = Some(arrival);
                } else {
                    ready.push_back(arriving_index);
                }
            }

            let consumed = remaining[index].min(quantum);
            let start = time;
            time += u64::from(consumed);
            remaining[index] -= consumed;
            let finished = remaining[index] == 0;

            runs.push(Run {
                id: job_id,
                start,
                end: time,
                finished,
            });

            while event_pos < events.len() {
                let arriving_index = events[event_pos];
                let arriving_job = &jobs[arriving_index];
                let arrival = u64::from(arriving_job.arrival);
                if arrival > time {
                    break;
                }

                event_pos += 1;
                if remaining[arriving_index] == 0 {
                    completion[arriving_index] = Some(arrival);
                } else {
                    ready.push_back(arriving_index);
                }
            }

            if finished {
                completion[index] = Some(time);
            } else {
                ready.push_back(index);
            }
            last_dispatched_id = Some(job_id);
        } else if event_pos < events.len() {
            let next_time = u64::from(jobs[events[event_pos]].arrival);
            if next_time > time {
                if let Some(last_idle) = idle.last_mut() {
                    if last_idle.1 == time {
                        last_idle.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
        } else {
            break;
        }
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            (
                job.id,
                completion[index].expect("every job completes by the end of the schedule"),
            )
        })
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
```
