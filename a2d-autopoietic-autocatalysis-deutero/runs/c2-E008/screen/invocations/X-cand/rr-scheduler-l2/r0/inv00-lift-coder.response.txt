#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion_times = vec![0u64; jobs.len()];
    let mut completed = vec![false; jobs.len()];
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut event_pos = 0usize;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut completed_count = 0usize;

    let admit_through = |limit: u64,
                         event_pos: &mut usize,
                         ready: &mut std::collections::VecDeque<usize>,
                         completed: &mut [bool],
                         completion_times: &mut [u64],
                         completed_count: &mut usize| {
        while *event_pos < events.len() {
            let index = events[*event_pos];
            let job = &jobs[index];
            let arrival = u64::from(job.arrival);
            if arrival > limit {
                break;
            }
            *event_pos += 1;
            if job.ticks == 0 {
                completed[index] = true;
                completion_times[index] = arrival;
                *completed_count += 1;
            } else {
                ready.push_back(index);
            }
        }
    };

    while completed_count < jobs.len() {
        admit_through(
            time,
            &mut event_pos,
            &mut ready,
            &mut completed,
            &mut completion_times,
            &mut completed_count,
        );

        if ready.is_empty() {
            if event_pos >= events.len() {
                break;
            }
            let next_time = u64::from(jobs[events[event_pos]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        }

        let index = ready.pop_front().expect("ready queue was nonempty");
        let job = &jobs[index];

        if last_dispatched_id.is_some_and(|previous| previous != job.id) {
            time += u64::from(switch_cost);
        }

        admit_through(
            time,
            &mut event_pos,
            &mut ready,
            &mut completed,
            &mut completion_times,
            &mut completed_count,
        );

        let start = time;
        let remaining = u64::from(job.ticks);
        let consumed = remaining.min(u64::from(quantum));
        time += consumed;
        let finished = consumed == remaining;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished,
        });

        admit_through(
            time,
            &mut event_pos,
            &mut ready,
            &mut completed,
            &mut completion_times,
            &mut completed_count,
        );

        if finished {
            completed[index] = true;
            completion_times[index] = time;
            completed_count += 1;
        } else {
            ready.push_back(index);
        }
        last_dispatched_id = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion_times[index]))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
