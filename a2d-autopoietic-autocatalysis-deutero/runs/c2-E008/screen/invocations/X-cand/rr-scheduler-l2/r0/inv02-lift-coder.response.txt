#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrivals: Vec<usize> = (0..jobs.len()).collect();
    arrivals.sort_by_key(|&i| (jobs[i].arrival, i));

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completion = vec![(0, 0); jobs.len()];
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut ready = std::collections::VecDeque::new();
    let mut next_arrival = 0;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut completed = 0usize;

    let admit_through = |time: u64,
                         next_arrival: &mut usize,
                         arrivals: &[usize],
                         jobs: &[Job],
                         remaining: &[u32],
                         ready: &mut std::collections::VecDeque<usize>,
                         completion: &mut [(u32, u64)],
                         completed: &mut usize| {
        while *next_arrival < arrivals.len() {
            let index = arrivals[*next_arrival];
            let job = &jobs[index];
            if u64::from(job.arrival) > time {
                break;
            }
            *next_arrival += 1;
            if remaining[index] == 0 {
                completion[index] = (job.id, u64::from(job.arrival));
                *completed += 1;
            } else {
                ready.push_back(index);
            }
        }
    };

    while completed < jobs.len() {
        admit_through(
            time,
            &mut next_arrival,
            &arrivals,
            jobs,
            &remaining,
            &mut ready,
            &mut completion,
            &mut completed,
        );

        if ready.is_empty() {
            if next_arrival >= arrivals.len() {
                break;
            }
            let next_time = u64::from(jobs[arrivals[next_arrival]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
            continue;
        }

        let index = ready.pop_front().unwrap();
        let job = &jobs[index];

        if last_dispatched.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }

        admit_through(
            time,
            &mut next_arrival,
            &arrivals,
            jobs,
            &remaining,
            &mut ready,
            &mut completion,
            &mut completed,
        );

        let consumed = remaining[index].min(quantum);
        let start = time;
        time += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished,
        });

        admit_through(
            time,
            &mut next_arrival,
            &arrivals,
            jobs,
            &remaining,
            &mut ready,
            &mut completion,
            &mut completed,
        );

        if finished {
            completion[index] = (job.id, time);
            completed += 1;
        } else {
            ready.push_back(index);
        }

        last_dispatched = Some(job.id);
    }

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
