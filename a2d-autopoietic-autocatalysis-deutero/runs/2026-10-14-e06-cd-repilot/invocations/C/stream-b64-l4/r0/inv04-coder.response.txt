use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{self:?}")
    }
}

impl std::error::Error for Error {}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_byte(options: Options, value: u8) -> u8 {
    const TABLE: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    match (options.alphabet, value) {
        (Alphabet::UrlSafe, 62) => b'-',
        (Alphabet::UrlSafe, 63) => b'_',
        _ => TABLE[value as usize],
    }
}

fn alphabet_value(options: Options, byte: u8, offset: usize) -> Result<u8, Error> {
    let value = match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if options.alphabet == Alphabet::Standard => Some(62),
        b'/' if options.alphabet == Alphabet::Standard => Some(63),
        b'-' if options.alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if options.alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    };
    if let Some(v) = value {
        return Ok(v);
    }

    if options.strict
        && matches!(
            (options.alphabet, byte),
            (Alphabet::Standard, b'-' | b'_') | (Alphabet::UrlSafe, b'+' | b'/')
        )
    {
        Err(Error::MixedAlphabet { offset })
    } else {
        Err(Error::InvalidByte { offset, byte })
    }
}

fn is_ignored(options: Options, byte: u8) -> bool {
    matches!(options.whitespace, Whitespace::IgnoreAscii)
        && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

fn symbols_for_bytes(input: &[u8], options: Options) -> Vec<u8> {
    let mut result = Vec::with_capacity(input.len().div_ceil(3) * 4);
    for chunk in input.chunks(3) {
        let a = chunk[0];
        let b = *chunk.get(1).unwrap_or(&0);
        let c = *chunk.get(2).unwrap_or(&0);
        result.push(alphabet_byte(options, a >> 2));
        result.push(alphabet_byte(options, ((a & 3) << 4) | (b >> 4)));
        if chunk.len() >= 2 {
            result.push(alphabet_byte(options, ((b & 15) << 2) | (c >> 6)));
        } else if options.padding != Padding::Forbidden {
            result.push(b'=');
        }
        if chunk.len() >= 3 {
            result.push(alphabet_byte(options, c & 63));
        } else if options.padding != Padding::Forbidden {
            result.push(b'=');
        }
    }
    result
}

fn wrap_symbols(symbols: &[u8], whitespace: Whitespace) -> String {
    match whitespace {
        Whitespace::Wrapped { width, ending } => {
            let mut out = Vec::new();
            for (i, b) in symbols.iter().enumerate() {
                if i > 0 && i % width == 0 {
                    match ending {
                        LineEnding::Lf => out.push(b'\n'),
                        LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
                    }
                }
                out.push(*b);
            }
            String::from_utf8(out).expect("base64 output is ASCII")
        }
        _ => String::from_utf8(symbols.to_vec()).expect("base64 output is ASCII"),
    }
}

/// Encodes bytes using the selected alphabet, padding, and line wrapping.
pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    Ok(wrap_symbols(&symbols_for_bytes(input, options), options.whitespace))
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
    value: Option<u8>,
}

#[derive(Clone, Copy)]
enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

#[derive(Clone, Copy)]
enum PendingFrame {
    None,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    closed: bool,
    symbols_in_row: usize,
    emitted_any: bool,
}

impl Encoder {
    /// Creates an open incremental encoder.
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            closed: false,
            symbols_in_row: 0,
            emitted_any: false,
        })
    }

    fn emit_symbols(&mut self, symbols: &[u8], out: &mut String) {
        for &byte in symbols {
            if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
                if self.symbols_in_row == width {
                    match ending {
                        LineEnding::Lf => out.push('\n'),
                        LineEnding::CrLf => out.push_str("\r\n"),
                    }
                    self.symbols_in_row = 0;
                }
            }
            out.push(byte as char);
            self.symbols_in_row += 1;
            self.emitted_any = true;
        }
    }

    /// Encodes all newly available complete three-byte groups.
    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut combined = std::mem::take(&mut self.tail);
        combined.extend_from_slice(input);
        let complete_len = combined.len() / 3 * 3;
        let mut out = String::new();
        if complete_len > 0 {
            let symbols = symbols_for_bytes(&combined[..complete_len], self.options);
            self.emit_symbols(&symbols, &mut out);
        }
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(out)
    }

    /// Encodes any buffered tail and closes the encoder.
    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        if !self.tail.is_empty() {
            let symbols = symbols_for_bytes(&self.tail, self.options);
            self.emit_symbols(&symbols, &mut out);
        }
        self.tail.clear();
        self.closed = true;
        Ok(out)
    }
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbols_seen: usize,
    group: Vec<Symbol>,
    terminal_padding: bool,
    frame: FrameState,
    wrapped_count: usize,
    closed: bool,
    sticky: Option<Error>,
}

impl Decoder {
    /// Creates an open incremental decoder.
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols_seen: 0,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            frame: FrameState::Normal,
            wrapped_count: 0,
            closed: false,
            sticky: None,
        })
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn decode_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        let g = &self.group;
        let a = g[0].value.unwrap();
        let b = g[1].value.unwrap();

        if g[2].byte == b'=' {
            if g[3].byte != b'=' {
                return Err(Error::BadPadding { offset: g[3].offset });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: g[2].offset });
            }
            if self.options.strict && (b & 0x0f) != 0 {
                return Err(Error::NonCanonical { offset: g[1].offset });
            }
            out.push((a << 2) | (b >> 4));
            self.terminal_padding = true;
        } else {
            let c = g[2].value.unwrap();
            if g[3].byte == b'=' {
                if self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset: g[3].offset });
                }
                if self.options.strict && (c & 0x03) != 0 {
                    return Err(Error::NonCanonical { offset: g[2].offset });
                }
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                self.terminal_padding = true;
            } else {
                let d = g[3].value.unwrap();
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                out.push((c << 6) | d);
            }
        }
        self.group.clear();
        Ok(())
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols_seen >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbols_seen += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group.len() == 2 && self.group[2].byte == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        let position = self.group.len();
        if byte == b'=' {
            if position < 2 {
                return Err(Error::BadPadding { offset });
            }
            if position == 2 {
                self.group.push(Symbol {
                    byte,
                    offset,
                    value: None,
                });
                return Ok(());
            }
            self.group.push(Symbol {
                byte,
                offset,
                value: None,
            });
            self.decode_group(out)?;
            return Ok(());
        }

        let value = alphabet_value(self.options, byte, offset)?;
        self.group.push(Symbol {
            byte,
            offset,
            value: Some(value),
        });
        if self.group.len() == 4 {
            self.decode_group(out)?;
        }
        Ok(())
    }

    fn process_frame(&mut self, byte: u8, offset: usize) -> Result<Option<bool>, Error> {
        match self.options.whitespace {
            Whitespace::None => Ok(Some(true)),
            Whitespace::IgnoreAscii => Ok(if is_ignored(self.options, byte) {
                Some(false)
            } else {
                Some(true)
            }),
            Whitespace::Wrapped { width, ending } => {
                match self.frame {
                    FrameState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.frame = FrameState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        return Ok(Some(false));
                    }
                    FrameState::NeedSymbol { .. } => {
                        self.frame = FrameState::Normal;
                    }
                    FrameState::Normal => {}
                }

                if self.wrapped_count == width {
                    let first = offset;
                    match ending {
                        LineEnding::Lf => {
                            if byte != b'\n' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.frame = FrameState::NeedSymbol {
                                separator_offset: first,
                            };
                            return Ok(Some(false));
                        }
                        LineEnding::CrLf => {
                            if byte != b'\r' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.frame = FrameState::NeedLf { cr_offset: offset };
                            return Ok(Some(false));
                        }
                    }
                }

                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset });
                }
                self.wrapped_count += 1;
                Ok(Some(true))
            }
        }
    }

    /// Decodes newly completed groups from the supplied bytes.
    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            let framed = match self.process_frame(byte, offset) {
                Ok(v) => v,
                Err(e) => return self.fail(e),
            };
            if framed == Some(false) {
                continue;
            }
            if let Err(e) = self.process_symbol(byte, offset, &mut out) {
                return self.fail(e);
            }
        }
        Ok(out)
    }

    /// Completes decoding, checks residual input, and closes the decoder.
    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }

        if let FrameState::NeedLf { cr_offset } = self.frame {
            return self.fail(Error::BadWrap { offset: cr_offset });
        }
        if let FrameState::NeedSymbol { separator_offset } = self.frame {
            return self.fail(Error::BadWrap {
                offset: separator_offset,
            });
        }

        let mut out = Vec::new();
        if self.group.len() == 3 && self.group[2].byte == b'=' {
            return self.fail(Error::BadPadding {
                offset: self.group[2].offset,
            });
        }

        if !self.group.is_empty() {
            match self.options.padding {
                Padding::Required => {
                    return self.fail(Error::Truncated {
                        offset: self.group[0].offset,
                    });
                }
                Padding::Forbidden | Padding::Optional => match self.group.len() {
                    1 => {
                        return self.fail(Error::Truncated {
                            offset: self.group[0].offset,
                        });
                    }
                    2 => {
                        let a = self.group[0].value.unwrap();
                        let b = self.group[1].value.unwrap();
                        if self.options.strict && (b & 0x0f) != 0 {
                            return self.fail(Error::NonCanonical {
                                offset: self.group[1].offset,
                            });
                        }
                        out.push((a << 2) | (b >> 4));
                    }
                    3 => {
                        let a = self.group[0].value.unwrap();
                        let b = self.group[1].value.unwrap();
                        let c = self.group[2].value.unwrap();
                        if self.options.strict && (c & 0x03) != 0 {
                            return self.fail(Error::NonCanonical {
                                offset: self.group[2].offset,
                            });
                        }
                        out.push((a << 2) | (b >> 4));
                        out.push((b << 4) | (c >> 2));
                    }
                    _ => {}
                },
            }
        }
        self.closed = true;
        Ok(out)
    }
}

/// Decodes bytes using the selected alphabet, padding, whitespace, and validation rules.
pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

fn encode_incremental(options: Options, chunks: &[&[u8]]) -> Result<String, Error> {
    let mut encoder = Encoder::new(options)?;
    let mut result = String::new();
    for chunk in chunks {
        result.push_str(&encoder.push(chunk)?);
    }
    result.push_str(&encoder.finish()?);
    Ok(result)
}

fn _frame_state_is_copy(state: FrameState) -> FrameState {
    match state {
        FrameState::Normal => FrameState::Normal,
        FrameState::NeedLf { cr_offset } => FrameState::NeedLf { cr_offset },
        FrameState::NeedSymbol { separator_offset } => {
            FrameState::NeedSymbol { separator_offset }
        }
    }
}

fn _pending_frame_placeholder(_: PendingFrame) {}

fn _encoder_state(_: bool, _: bool) {}

#[cfg(test)]
mod tests {
    use super::*;

    fn opts() -> Options {
        Options {
            alphabet: Alphabet::Standard,
            padding: Padding::Required,
            whitespace: Whitespace::None,
            strict: false,
            max_symbols: None,
        }
    }

    #[test]
    fn encode_normal_and_empty() {
        assert_eq!(encode(b"Cat", opts()).unwrap(), "Q2F0");
        assert_eq!(encode(b"", opts()).unwrap(), "");
    }

    #[test]
    fn decode_invalid_and_padding() {
        assert_eq!(decode(b"T2s=", opts()).unwrap(), b"Ok");
        assert_eq!(
            decode(b"AA=A", opts()),
            Err(Error::BadPadding { offset: 3 })
        );
    }

    #[test]
    fn optional_unpadded_and_strict_bits() {
        let mut options = opts();
        options.padding = Padding::Optional;
        assert_eq!(decode(b"Zg", options).unwrap(), b"f");
        options.strict = true;
        assert_eq!(
            decode(b"Zh==", options),
            Err(Error::NonCanonical { offset: 1 })
        );
    }

    #[test]
    fn streaming_wrapped_round_trip() {
        let mut options = opts();
        options.whitespace = Whitespace::Wrapped {
            width: 4,
            ending: LineEnding::Lf,
        };
        let encoded = encode(b"QrsT", options).unwrap();
        assert_eq!(encoded, "UXJz\nVA==");
        assert_eq!(decode(encoded.as_bytes(), options).unwrap(), b"QrsT");
    }
}
