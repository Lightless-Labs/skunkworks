use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Status {
    Queued,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u64,
    status: Status,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    entries: HashMap<String, Entry>,
}

impl Scheduler {
    /// Creates an independent scheduler with the given aging interval.
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            entries: HashMap::new(),
        }
    }

    /// Inserts a job, permanently reserving its ID.
    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.entries.contains_key(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.wrapping_add(1);
        self.entries.insert(
            job.id.clone(),
            Entry {
                job,
                sequence,
                status: Status::Queued,
            },
        );
        self.last_time = now;
        Ok(())
    }

    /// Cancels a queued job identified by its exact ID.
    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let status = self.entries.get(id).map(|entry| entry.status);
        match status {
            None => return Err(Error::Unknown),
            Some(Status::Cancelled) => return Err(Error::Cancelled),
            Some(Status::Completed) => return Err(Error::Completed),
            Some(Status::Queued) => {}
        }

        if let Some(entry) = self.entries.get_mut(id) {
            entry.status = Status::Cancelled;
        }
        self.last_time = now;
        Ok(())
    }

    /// Dispatches the highest-priority ready job, if one is available.
    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let selected = self
            .entries
            .iter()
            .filter(|(_, entry)| {
                entry.status == Status::Queued && entry.job.ready_at <= now
            })
            .map(|(id, entry)| {
                let effective_priority =
                    effective_priority(&entry.job, now, self.aging_ms);
                (id.clone(), effective_priority, entry.sequence)
            })
            .max_by(|a, b| {
                a.1.cmp(&b.1).then_with(|| b.2.cmp(&a.2))
            });

        self.last_time = now;
        let Some((id, effective_priority, _)) = selected else {
            return Ok(None);
        };

        if let Some(entry) = self.entries.get_mut(&id) {
            entry.status = Status::Completed;
        }
        Ok(Some(Dispatch {
            id,
            effective_priority,
        }))
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn effective_priority(job: &Job, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return job.priority;
    }
    let age_steps = (now - job.ready_at) / aging_ms;
    (job.priority as u64 + age_steps).min(9) as u8
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    fn job(id: &str, ready_at: u64, priority: u8) -> Job {
        Job {
            id: id.to_owned(),
            ready_at,
            priority,
        }
    }

    #[test]
    fn dispatches_ready_job() {
        let mut scheduler = Scheduler::new(0);
        scheduler.insert(0, job("job-a", 0, 3)).unwrap();
        assert_eq!(
            scheduler.next_ready(0).unwrap(),
            Some(Dispatch {
                id: "job-a".to_owned(),
                effective_priority: 3
            })
        );
    }

    #[test]
    fn rejects_bad_id_without_advancing_time() {
        let mut scheduler = Scheduler::new(0);
        assert_eq!(scheduler.insert(5, job("Bad", 0, 1)), Err(Error::BadId));
        assert_eq!(scheduler.insert(4, job("good", 0, 1)), Ok(()));
    }

    #[test]
    fn ages_and_breaks_ties_by_insertion_order() {
        let mut scheduler = Scheduler::new(10);
        scheduler.insert(0, job("early", 0, 1)).unwrap();
        scheduler.insert(0, job("later", 10, 2)).unwrap();
        let dispatched = scheduler.next_ready(20).unwrap().unwrap();
        assert_eq!(dispatched.id, "early");
        assert_eq!(dispatched.effective_priority, 3);
    }

    #[test]
    fn cancellation_reserves_id_and_unknown_is_reported() {
        let mut scheduler = Scheduler::new(10);
        scheduler.insert(0, job("reserved", 100, 1)).unwrap();
        scheduler.cancel(1, "reserved").unwrap();
        assert_eq!(scheduler.cancel(2, "reserved"), Err(Error::Cancelled));
        assert_eq!(
            scheduler.insert(2, job("reserved", 0, 1)),
            Err(Error::Duplicate)
        );
        assert_eq!(scheduler.cancel(2, "missing"), Err(Error::Unknown));
    }

    #[test]
    fn ready_jobs_dispatch_end_to_end() {
        let mut scheduler = Scheduler::new(5);
        scheduler.insert(0, job("future", 9, 4)).unwrap();
        assert_eq!(scheduler.next_ready(8).unwrap(), None);
        assert_eq!(
            scheduler.next_ready(14).unwrap(),
            Some(Dispatch {
                id: "future".to_owned(),
                effective_priority: 5
            })
        );
        assert_eq!(scheduler.cancel(14, "future"), Err(Error::Completed));
    }
}
