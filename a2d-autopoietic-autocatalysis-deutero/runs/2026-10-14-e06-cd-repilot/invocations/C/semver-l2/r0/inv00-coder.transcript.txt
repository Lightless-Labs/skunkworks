fn byte(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while byte(s, *pos).is_some_and(|b| b.is_ascii_digit()) {
        *pos += 1;
    }
    if *pos == start {
        return match byte(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut pos = start;
    loop {
        let id_start = pos;
        while pos < end {
            let b = s.as_bytes()[pos];
            if b == b'.' {
                break;
            }
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
        if pos == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..pos];
        if prerelease
            && id.len() > 1
            && id.starts_with('0')
            && id.bytes().all(|b| b.is_ascii_digit())

            return Err(VersionError::LeadingZero { at: id_start });

            break;
        }
        pos += 1;

            _ => return Err(VersionError::InvalidChar { at: pos }),

        }
    }

    match byte(s, pos) {
        Some(b'.') => pos += 1,
        None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
        _ => return Err(VersionError::InvalidChar { at: pos }),

    let mut build = Vec::new();

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())

fn compare_pre_id(a: &str, b: &str) -> Ordering {
    let an = is_numeric_id(a);
    let bn = is_numeric_id(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

/// Compares versions using semantic-version precedence, ignoring build metadata.
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {

                    }
                }

            }
        })
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let full_candidate = s.contains('-') || s.contains('+') ||
        (s.split('.').count() >= 3 && !s.split('.').any(|x| matches!(x, "*" | "x" | "X")));

        return parse_version(s)
            .map(|v| Partial { nums: vec![v.major, v.minor, v.patch], full: Some(v) })
            .map_err(|error| RangeError::Version { at, error });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            wildcard = true;
        } else {

                || (field.len() > 1 && field.starts_with('0'))

                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;

        }

}

fn upper_for(part: &Partial, kind: char, at: usize) -> Result<Version, RangeError> {
    let n = part.count();

    v.pre = vec!["0".to_string()];
    v.build.clear();

        '^' if v.major > 0 || n == 1 => 0,
        '^' if v.minor > 0 || n == 2 => 1,
        '^' => 2,
        '~' if n == 1 => 0,
        '~' => 1,
        _ if n == 1 => 0,
        _ => 1,

        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,

    for (token, at) in ts {
        if token == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            last_or_end = Some(at + 2);
        } else {
            current.push((token, at));
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: last_or_end.unwrap_or(s.len()) });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn check(v: &Version, c: &Check) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Eq => ord == Ordering::Equal,
        Op::Ge => ord != Ordering::Less,
        Op::Gt => ord == Ordering::Greater,
    }
}

/// Returns whether a version satisfies the range, including prerelease admission rules.
pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.contains(&(v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.checks.iter().all(|c| check(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates.iter().enumerate().map(|(index, s)| {
        parse_version(s).map_err(|error| CandidateError { index, error })
    }).collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

/// Returns the greatest matching candidate by semantic-version precedence.
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

/// Returns the least matching candidate by semantic-version precedence.
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
