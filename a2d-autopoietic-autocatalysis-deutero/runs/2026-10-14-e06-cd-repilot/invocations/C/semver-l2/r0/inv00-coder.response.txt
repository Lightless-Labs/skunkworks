use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while byte(s, *pos).is_some_and(|b| b.is_ascii_digit()) {
        *pos += 1;
    }
    if *pos == start {
        return match byte(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut pos = start;
    loop {
        let id_start = pos;
        while pos < end {
            let b = s.as_bytes()[pos];
            if b == b'.' {
                break;
            }
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
        if pos == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..pos];
        if prerelease
            && id.len() > 1
            && id.starts_with('0')
            && id.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_string());
        if pos == end {
            break;
        }
        pos += 1;
        if pos == end {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
    }
    Ok(result)
}

/// Parses a strict semantic version string.
pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        match byte(s, pos) {
            Some(b'.') => pos += 1,
            None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        if _ == 0 {
            break;
        }
    }
    let minor = parse_core(s, &mut pos)?;
    match byte(s, pos) {
        Some(b'.') => pos += 1,
        None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }
    let patch = parse_core(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    match byte(s, pos) {
        None => {}
        Some(b'-') => {
            let pre_start = pos + 1;
            let plus = s[pre_start..].find('+').map(|i| pre_start + i);
            let pre_end = plus.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(plus_pos) = plus {
                build = parse_identifiers(s, plus_pos + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, pos + 1, s.len(), false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre_id(a: &str, b: &str) -> Ordering {
    let an = is_numeric_id(a);
    let bn = is_numeric_id(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

/// Compares versions using semantic-version precedence, ignoring build metadata.
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_pre_id(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Lt,
    Le,
    Eq,
    Ge,
    Gt,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, base: usize) -> Vec<(&str, usize)> {
    let mut out = Vec::new();
    let mut i = 0;
    while i < s.len() {
        while i < s.len() && ws(s.as_bytes()[i]) {
            i += 1;
        }
        if i == s.len() {
            break;
        }
        let start = i;
        while i < s.len() && !ws(s.as_bytes()[i]) {
            i += 1;
        }
        out.push((&s[start..i], base + start));
    }
    out
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}

impl Partial {
    fn padded(&self) -> Version {
        let major = *self.nums.first().unwrap_or(&0);
        let minor = *self.nums.get(1).unwrap_or(&0);
        let patch = *self.nums.get(2).unwrap_or(&0);
        self.full.clone().unwrap_or(Version {
            major, minor, patch, pre: Vec::new(), build: Vec::new(),
        })
    }
    fn count(&self) -> usize { self.nums.len() }
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let full_candidate = s.contains('-') || s.contains('+') ||
        (s.split('.').count() >= 3 && !s.split('.').any(|x| matches!(x, "*" | "x" | "X")));
    if full_candidate {
        return parse_version(s)
            .map(|v| Partial { nums: vec![v.major, v.minor, v.patch], full: Some(v) })
            .map_err(|error| RangeError::Version { at, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial { nums, full: None })
}

fn upper_for(part: &Partial, kind: char, at: usize) -> Result<Version, RangeError> {
    let n = part.count();
    let mut v = part.padded();
    v.pre = vec!["0".to_string()];
    v.build.clear();
    let field = match kind {
        '^' if v.major > 0 || n == 1 => 0,
        '^' if v.minor > 0 || n == 2 => 1,
        '^' => 2,
        '~' if n == 1 => 0,
        '~' => 1,
        _ if n == 1 => 0,
        _ => 1,
    };
    let value = match field {
        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,
    };
    *value = value.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field {
        0 => { v.minor = 0; v.patch = 0; }
        1 => v.patch = 0,
        _ => {}
    }
    Ok(v)
}

fn add_check(arm: &mut Arm, op: Op, v: Version) {
    if !v.pre.is_empty() {
        arm.anchors.push((v.major, v.minor, v.patch));
    }
    arm.checks.push(Check { op, version: v });
}

fn parse_arm(ts: &[(&str, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { checks: Vec::new(), anchors: Vec::new() };
    if ts.is_empty() {
        return Ok(arm);
    }
    if let Some((idx, (_, at))) = ts.iter().enumerate().find(|(_, (t, _))| *t == "-") {
        if idx != 1 || ts.len() != 3 {
            return Err(RangeError::Syntax { at: *at });
        }
        let (lo, lo_at) = ts[0];
        let (hi, hi_at) = ts[2];
        let lower = parse_partial(lo, lo_at)?;
        let upper = parse_partial(hi, hi_at)?;
        if lower.count() == 0 { return Err(RangeError::Syntax { at: lo_at }); }
        if upper.count() == 0 { return Err(RangeError::Syntax { at: hi_at }); }
        add_check(&mut arm, Op::Ge, lower.padded());
        if let Some(full) = upper.full {
            add_check(&mut arm, Op::Le, full);
        } else {
            let end = if upper.count() == 1 || upper.count() == 2 {
                upper_for(&upper, '~', hi_at)?
            } else {
                upper.padded()
            };
            add_check(&mut arm, if upper.count() < 3 { Op::Lt } else { Op::Le }, end);
        }
        return Ok(arm);
    }

    for &(token, at) in ts {
        if token.is_empty() { return Err(RangeError::Syntax { at }); }
        let b = token.as_bytes()[0];
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'=') {
            return Err(RangeError::Syntax { at });
        }
        let (op, operand, operand_at) = if token.starts_with(">=") {
            (Some(Op::Ge), &token[2..], at + 2)
        } else if token.starts_with("<=") {
            (Some(Op::Le), &token[2..], at + 2)
        } else if token.starts_with('>') {
            (Some(Op::Gt), &token[1..], at + 1)
        } else if token.starts_with('<') {
            (Some(Op::Lt), &token[1..], at + 1)
        } else if token.starts_with('=') {
            (Some(Op::Eq), &token[1..], at + 1)
        } else {
            (None, token, at)
        };
        if let Some(op) = op {
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = parse_version(operand)
                .map_err(|error| RangeError::Version { at: operand_at, error })?;
            add_check(&mut arm, op, v);
            continue;
        }
        if let Some(kind @ ('^' | '~')) = token.chars().next() {
            let operand = &token[1..];
            let operand_at = at + 1;
            let p = parse_partial(operand, operand_at)?;
            if p.count() == 0 { return Err(RangeError::Syntax { at }); }
            let lower = p.padded();
            let upper = upper_for(&p, kind, operand_at)?;
            add_check(&mut arm, Op::Ge, lower);
            add_check(&mut arm, Op::Lt, upper);
            continue;
        }
        let p = parse_partial(token, at)?;
        if let Some(full) = p.full {
            add_check(&mut arm, Op::Eq, full);
        } else if p.count() == 0 {
            continue;
        } else if p.count() == 1 || p.count() == 2 {
            let lower = p.padded();
            let upper = upper_for(&p, '~', at)?;
            add_check(&mut arm, Op::Ge, lower);
            add_check(&mut arm, Op::Lt, upper);
        } else {
            add_check(&mut arm, Op::Eq, p.padded());
        }
    }
    Ok(arm)
}

/// Parses a range expression into conjunctive arms with optional disjunctions.
pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let ts = tokens(s, 0);
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut last_or_end = None;
    for (token, at) in ts {
        if token == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            last_or_end = Some(at + 2);
        } else {
            current.push((token, at));
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: last_or_end.unwrap_or(s.len()) });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn check(v: &Version, c: &Check) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Eq => ord == Ordering::Equal,
        Op::Ge => ord != Ordering::Less,
        Op::Gt => ord == Ordering::Greater,
    }
}

/// Returns whether a version satisfies the range, including prerelease admission rules.
pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.contains(&(v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.checks.iter().all(|c| check(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates.iter().enumerate().map(|(index, s)| {
        parse_version(s).map_err(|error| CandidateError { index, error })
    }).collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

/// Returns the greatest matching candidate by semantic-version precedence.
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

/// Returns the least matching candidate by semantic-version precedence.
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version { parse_version(s).unwrap() }
    fn pr(s: &str) -> Range { parse_range(s).unwrap() }

    #[test]
    fn version_normal_and_edge() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(parse_version("01.0.0"), Err(VersionError::LeadingZero { at: 0 }));
    }

    #[test]
    fn range_expansion_and_prerelease() {
        assert!(matches(&pv("2.5.1"), &pr("^2.1.0")));
        assert!(matches(&pv("2.5.1"), &pr("2.5.x")));
        assert!(matches(&pv("2.1.3-rc"), &pr(">=2.1.3-beta <2.1.3")));
        assert!(!matches(&pv("2.1.3-rc"), &pr(">=2.1.3")));
    }

    #[test]
    fn disjunction_and_extrema_end_to_end() {
        assert!(matches(&pv("4.0.0"), &pr("2.x || 4.0.0")));
        let cs = vec!["2.2.0".into(), "2.8.0".into(), "3.0.0".into()];
        assert_eq!(max_satisfying(&cs, &pr("2.x")), Ok(Some("2.8.0".into())));
        assert_eq!(min_satisfying(&cs, &pr("2.x")), Ok(Some("2.2.0".into())));
    }
}
