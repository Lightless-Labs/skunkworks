use std::collections::{BTreeMap, BTreeSet};

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(want) => version == want,
        Constraint::AtLeast(want) => version >= want,
        Constraint::Before(want) => version < want,
        Constraint::Caret(want) => {
            version >= want
                && if want.major > 0 {
                    version.major == want.major
                } else if want.minor > 0 {
                    version.major == 0 && version.minor == want.minor
                } else {
                    version == want
                }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| matches(version, *c))
}

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &BTreeMap<String, Version>,
) -> BTreeMap<String, Vec<Constraint>> {
    let mut active: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in selected {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical_constraints(std::mem::take(constraints));
    }
    active
}

fn constraints_for(
    active: &BTreeMap<String, Vec<Constraint>>,
    package: &str,
) -> Vec<Constraint> {
    active.get(package).cloned().unwrap_or_default()
}

fn inspect_node(
    catalog: &[Package],
    active: &BTreeMap<String, Vec<Constraint>>,
    selected: &BTreeMap<String, Version>,
) -> Result<Option<String>, ResolveError> {
    for name in active.keys() {
        if !catalog.iter().any(|p| p.name == *name) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }
    }

    for (name, version) in selected {
        if let Some(constraints) = active.get(name) {
            if !satisfies(*version, constraints) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in active {
        if selected.contains_key(name) {
            continue;
        }
        let package = catalog.iter().find(|p| p.name == *name).unwrap();
        if !package
            .releases
            .iter()
            .any(|r| satisfies(r.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    Ok(active
        .keys()
        .find(|name| !selected.contains_key(*name))
        .cloned())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn record_failure(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn consider(&mut self, selected: &BTreeMap<String, Version>) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count();
        let names: BTreeSet<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        let vector: Vec<Option<Version>> = names
            .into_iter()
            .map(|name| selected.get(name).copied())
            .collect();

        let replace = match &self.best {
            None => true,
            Some((best_cost, best_vector, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            self.best = Some((cost, vector, selected.clone()));
        }
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, self.catalog, selected);
        let branch = match inspect_node(self.catalog, &active, selected) {
            Ok(branch) => branch,
            Err(error) => {
                self.record_failure(error);
                return Ok(());
            }
        };

        let Some(package_name) = branch else {
            self.consider(selected);
            return Ok(());
        };

        let package = self
            .catalog
            .iter()
            .find(|p| p.name == package_name)
            .expect("active package was checked");
        let constraints = constraints_for(&active, &package_name);
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| satisfies(*v, &constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(&package_name) {
            if let Some(index) = candidates.iter().position(|v| v == locked) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), version);
            let result = self.visit(selected);
            selected.remove(&package_name);
            result?;
        }
        Ok(())
    }
}

fn build_resolution(
    catalog: &[Package],
    locks: &[Lock],
    selected: BTreeMap<String, Version>,
    roots: &[Requirement],
) -> Resolution {
    let selected_vec: Vec<Selection> = selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    let final_active = active_constraints(roots, catalog, &selected);
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let names: BTreeSet<String> = lock_map
        .keys()
        .chain(selected.keys())
        .cloned()
        .collect();

    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !satisfies(old, &constraints_for(&final_active, &package)) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Resolution {
        selected: selected_vec,
        changes,
    }
}

/// Resolves package requirements and reconciles the result with the supplied locks.
pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks: locks
            .iter()
            .map(|lock| (lock.package.clone(), lock.version))
            .collect(),
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;

    match search.best {
        Some((_, _, selected)) => Ok(build_resolution(catalog, locks, selected, roots)),
        None => Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{byte:02X}"));
        }
    }
    out
}

/// Formats a resolution error as its deterministic machine-readable representation.
pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => {
            let constraints = canonical_constraints(requirements.clone())
                .iter()
                .copied()
                .map(constraint_text)
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "no-match|package={}|constraints={constraints}",
                encode(package)
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => {
            let constraints = canonical_constraints(requirements.clone())
                .iter()
                .copied()
                .map(constraint_text)
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={constraints}",
                encode(package),
                version_text(*selected)
            )
        }
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn v(major: u32, minor: u32, patch: u32) -> Version {
        Version { major, minor, patch }
    }
    fn x(n: u32) -> Version {
        v(n, 0, 0)
    }
    fn req(package: &str, constraint: Constraint) -> Requirement {
        Requirement {
            package: package.into(),
            constraint,
        }
    }
    fn release(n: u32, deps: Vec<Requirement>) -> Release {
        Release { version: x(n), deps }
    }
    fn package(name: &str, releases: Vec<Release>) -> Package {
        Package {
            name: name.into(),
            releases,
        }
    }

    #[test]
    fn resolves_transitive_dependencies() {
        let catalog = vec![
            package("engine", vec![release(1, vec![req("util", Constraint::AtLeast(x(1)))])]),
            package("util", vec![release(1, vec![])]),
        ];
        let result = resolve(&catalog, &[req("engine", Constraint::Exact(x(1)))], &[], 10)
            .unwrap();
        assert_eq!(result.selected.len(), 2);
        assert_eq!(result.selected[0].package, "engine");
        assert_eq!(result.selected[1].package, "util");
    }

    #[test]
    fn rejects_invalid_names() {
        let error = resolve(
            &[],
            &[req("Bad", Constraint::Exact(x(1)))],
            &[],
            0,
        )
        .unwrap_err();
        assert_eq!(
            error_text(&error),
            "invalid-name|site=root%5B0%5D|name=Bad"
        );
    }

    #[test]
    fn locks_are_retained_when_possible() {
        let catalog = vec![package("app", vec![release(1, vec![]), release(2, vec![])])];
        let result = resolve(
            &catalog,
            &[req("app", Constraint::AtLeast(x(1)))],
            &[Lock {
                package: "app".into(),
                version: x(1),
            }],
            10,
        )
        .unwrap();
        assert_eq!(result.selected[0].version, x(1));
        assert!(result.changes.is_empty());
    }

    #[test]
    fn empty_roots_remove_locks_without_attempts() {
        let result = resolve(
            &[],
            &[],
            &[Lock {
                package: "orphan".into(),
                version: x(4),
            }],
            0,
        )
        .unwrap();
        assert_eq!(result.changes.len(), 1);
        assert_eq!(result.changes[0].reason, ChangeReason::Removed);
    }
}
