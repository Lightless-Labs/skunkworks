use std::collections::{BTreeMap, BTreeSet};

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

/// Resolves root requirements to the first complete solution in deterministic candidate order.
pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;
    validate_duplicates(catalog)?;

    let packages: BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut root_constraints: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    for requirement in roots {
        root_constraints
            .entry(requirement.package.clone())
            .or_default()
            .push(requirement.constraint);
    }

    let mut attempts = 0;
    let mut selected = BTreeMap::new();
    search(
        &packages,
        &root_constraints,
        &mut selected,
        &mut attempts,
        max_attempts,
    )
}

/// Formats a resolution error as a deterministic, percent-encoded string.
pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }

    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut names = BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(wanted) => version == wanted,
        Constraint::AtLeast(wanted) => version >= wanted,
        Constraint::Before(wanted) => version < wanted,
        Constraint::Caret(wanted) => {
            version >= wanted
                && if wanted.major > 0 {
                    version.major == wanted.major
                } else if wanted.minor > 0 {
                    version.major == 0 && version.minor == wanted.minor
                } else {
                    version == wanted
                }
        }
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn active_constraints(
    roots: &BTreeMap<String, Vec<Constraint>>,
    selected: &BTreeMap<String, Version>,
    packages: &BTreeMap<&str, &Package>,
) -> BTreeMap<String, Vec<Constraint>> {
    let mut active = roots.clone();
    for (name, version) in selected {
        if let Some(package) = packages.get(name.as_str()) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for dependency in &release.deps {
                    active
                        .entry(dependency.package.clone())
                        .or_default()
                        .push(dependency.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    active
}

fn search(
    packages: &BTreeMap<&str, &Package>,
    roots: &BTreeMap<String, Vec<Constraint>>,
    selected: &mut BTreeMap<String, Version>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let active = active_constraints(roots, selected, packages);

    if let Some(name) = active.keys().find(|name| !packages.contains_key(name.as_str())) {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in selected.iter() {
        if let Some(constraints) = active.get(name) {
            if constraints.iter().any(|constraint| !matches(*version, *constraint)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let mut candidates_by_name = BTreeMap::new();
    for (name, constraints) in &active {
        if selected.contains_key(name) {
            continue;
        }
        let package = packages[name.as_str()];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| constraints.iter().all(|constraint| matches(*version, *constraint)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if candidates.is_empty() {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
        candidates_by_name.insert(name.clone(), candidates);
    }

    let Some((name, candidates)) = candidates_by_name.into_iter().next() else {
        return Ok(selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect());
    };

    let mut first_error = None;
    for candidate in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.insert(name.clone(), candidate);
        match search(packages, roots, selected, attempts, max_attempts) {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => {
                selected.remove(&name);
                return Err(error);
            }
            Err(error) => {
                selected.remove(&name);
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or(ResolveError::NoMatch {
        package: name.clone(),
        requirements: active.get(&name).cloned().unwrap_or_default(),
    }))
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(version) => format!("={}", version_text(version)),
        Constraint::AtLeast(version) => format!(">={}", version_text(version)),
        Constraint::Before(version) => format!("<{}", version_text(version)),
        Constraint::Caret(version) => format!("^{}", version_text(version)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut result = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            result.push(*byte as char);
        } else {
            result.push_str(&format!("%{byte:02X}"));
        }
    }
    result
}

#[cfg(test)]
mod tests {
    use super::*;

    fn v(major: u32, minor: u32, patch: u32) -> Version {
        Version { major, minor, patch }
    }
    fn x(n: u32) -> Version {
        v(n, 0, 0)
    }
    fn e(n: u32) -> Constraint {
        Constraint::Exact(x(n))
    }
    fn ge(n: u32) -> Constraint {
        Constraint::AtLeast(x(n))
    }
    fn q(package: &str, constraint: Constraint) -> Requirement {
        Requirement {
            package: package.into(),
            constraint,
        }
    }
    fn r(n: u32, deps: Vec<Requirement>) -> Release {
        Release { version: x(n), deps }
    }
    fn p(name: &str, releases: Vec<Release>) -> Package {
        Package { name: name.into(), releases }
    }
    fn sel(package: &str, n: u32) -> Selection {
        Selection {
            package: package.into(),
            version: x(n),
        }
    }

    #[test]
    fn resolves_transitive_dependencies_and_sorted_selection() {
        let catalog = vec![
            p("engine", vec![r(1, vec![q("util", ge(1))])]),
            p("util", vec![r(1, vec![])]),
        ];
        assert_eq!(
            resolve(&catalog, &[q("engine", e(1))], 20).unwrap(),
            vec![sel("engine", 1), sel("util", 1)]
        );
    }

    #[test]
    fn ranges_and_caret_boundaries() {
        let catalog = vec![p("codec", vec![r(3, vec![]), r(1, vec![]), r(2, vec![])])];
        assert_eq!(
            resolve(
                &catalog,
                &[q("codec", ge(1)), q("codec", Constraint::Before(x(3)))],
                20
            )
            .unwrap(),
            vec![sel("codec", 2)]
        );
        assert!(matches!(
            resolve(
                &[p("lib", vec![Release { version: v(0, 3, 0), deps: vec![] }])],
                &[q("lib", Constraint::Caret(v(0, 2, 0)))],
                10
            ),
            Err(ResolveError::NoMatch { .. })
        ));
    }

    #[test]
    fn validates_names_before_duplicate_checks() {
        let catalog = vec![
            p("same", vec![r(1, vec![])]),
            p("same", vec![r(1, vec![q("Bad Name", e(1))])]),
        ];
        assert_eq!(
            resolve(&catalog, &[], 0),
            Err(ResolveError::InvalidName {
                site: "catalog[1].release[0].dep[0]".into(),
                name: "Bad Name".into(),
            })
        );
    }

    #[test]
    fn backtracks_and_formats_errors() {
        let catalog = vec![p(
            "app",
            vec![r(2, vec![q("missing", e(1))]), r(1, vec![])],
        )];
        assert_eq!(
            resolve(&catalog, &[q("app", ge(1))], 10).unwrap(),
            vec![sel("app", 1)]
        );
        assert_eq!(
            error_text(&ResolveError::InvalidName {
                site: "root[0]".into(),
                name: "Bad Name".into()
            }),
            "invalid-name|site=root%5B0%5D|name=Bad%20Name"
        );
    }

    #[test]
    fn zero_budget_only_limits_when_assignment_is_needed() {
        assert_eq!(resolve(&[], &[], 0), Ok(vec![]));
        assert_eq!(
            resolve(&[p("thing", vec![r(1, vec![])])], &[q("thing", e(1))], 0),
            Err(ResolveError::Limit { attempts: 0 })
        );
    }
}
