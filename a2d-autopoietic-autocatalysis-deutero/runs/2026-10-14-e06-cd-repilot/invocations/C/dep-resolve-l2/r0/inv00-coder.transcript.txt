fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{:02X}", byte));
        }
    }
    out
}

/// Formats a resolution error as a deterministic, percent-encoded string.
pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,

            "no-match|package={}|constraints={}",

            constraints_text(requirements)
        ),
        ResolveError::Conflict {

            requirements,

            version_text(*selected),
            constraints_text(requirements)

        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

type Active = BTreeMap<String, Vec<Constraint>>;
type Assigned = BTreeMap<String, Version>;

fn active_constraints(roots: &[Requirement], assigned: &Assigned, catalog: &BTreeMap<String, &Package>) -> Active {
    let mut active: Active = BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assigned {
        if let Some(package) = catalog.get(name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        canonicalize(constraints);
    }
    active
}

enum SearchResult {
    Success(Assigned),
    Failure(ResolveError),
    Limit,
}

fn search(
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    assigned: &mut Assigned,
    attempts: &mut usize,
    max_attempts: usize,
) -> SearchResult {
    let active = active_constraints(roots, assigned, catalog);

    if let Some(name) = active.keys().find(|name| !catalog.contains_key(*name)) {
        return SearchResult::Failure(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in assigned.iter() {
        if let Some(constraints) = active.get(name) {
            if !constraints.iter().all(|c| matches(*version, *c)) {
                return SearchResult::Failure(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in &active {
        if assigned.contains_key(name) {
            continue;
        }
        let package = catalog[name];
        if !package
            .releases
            .iter()
            .any(|release| constraints.iter().all(|c| matches(release.version, *c)))
        {
            return SearchResult::Failure(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let Some(name) = active.keys().find(|name| !assigned.contains_key(*name)) else {
        return SearchResult::Success(assigned.clone());
    };

    let constraints = &active[name];
    let mut candidates: Vec<Version> = catalog[name]
        .releases
        .iter()
        .map(|release| release.version)
        .filter(|version| constraints.iter().all(|c| matches(*version, *c)))
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    let mut first_failure = None;
    for version in candidates {
        if *attempts == max_attempts {
            return SearchResult::Limit;
        }
        *attempts += 1;
        assigned.insert(name.clone(), version);
        match search(roots, catalog, assigned, attempts, max_attempts) {
            success @ SearchResult::Success(_) => return success,
            SearchResult::Limit => return SearchResult::Limit,
            SearchResult::Failure(error) => {
                if first_failure.is_none() {
                    first_failure = Some(error);
                }
                assigned.remove(name);
            }
        }
    }

    SearchResult::Failure(first_failure.unwrap_or_else(|| ResolveError::NoMatch {
        package: name.clone(),
        requirements: constraints.clone(),
    }))
}

/// Resolves roots against a package catalog using bounded depth-first search.
pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let mut assigned = BTreeMap::new();
    let mut attempts = 0;

    match search(
        roots,
        &catalog_map,
        &mut assigned,
        &mut attempts,
        max_attempts,
    ) {
        SearchResult::Success(versions) => Ok(versions
            .into_iter()
            .map(|(package, version)| Selection { package, version })
            .collect()),
        SearchResult::Failure(error) => Err(error),
        SearchResult::Limit => Err(ResolveError::Limit { attempts }),
    }
}
