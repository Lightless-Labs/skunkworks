use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyShards,
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    Blocked { shard: usize },
    Missing { key: u64 },
    Pinned { key: u64 },
    BadShard { shard: usize },
    PinnedCapacity { minimum: u128, target: u128 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub pinned: bool,
    pub shard: usize,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacities: Vec<u64>,
    usage: Vec<u64>,
    entries: BTreeMap<u64, Entry>,
    order: VecDeque<u64>,
}

impl Cache {
    /// Creates a cache with the supplied per-shard capacities.
    pub fn new(capacities: Vec<u64>) -> Result<Self, Error> {
        if capacities.is_empty() {
            return Err(Error::EmptyShards);
        }
        Ok(Self {
            usage: vec![0; capacities.len()],
            capacities,
            entries: BTreeMap::new(),
            order: VecDeque::new(),
        })
    }

    /// Returns a copy of capacities in shard order.
    pub fn capacities(&self) -> Vec<u64> {
        self.capacities.clone()
    }

    /// Returns a copy of resident weights in shard order.
    pub fn usage(&self) -> Vec<u64> {
        self.usage.clone()
    }

    /// Returns the exact aggregate resident weight.
    pub fn used(&self) -> u128 {
        self.usage.iter().map(|&n| n as u128).sum()
    }

    /// Returns cloned entries from globally oldest to newest.
    pub fn snapshot(&self) -> Vec<Entry> {
        self.order
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect()
    }

    fn shard(&self, key: u64) -> usize {
        (key % self.capacities.len() as u64) as usize
    }

    fn touch(&mut self, key: u64) {
        self.order.retain(|&k| k != key);
        self.order.push_back(key);
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.usage[entry.shard] -= entry.weight;
        self.order.retain(|&k| k != key);
        Some(entry)
    }

    fn local_victim(&self, shard: usize) -> Option<u64> {
        self.order
            .iter()
            .find(|&&key| {
                self.entries
                    .get(&key)
                    .is_some_and(|entry| entry.shard == shard && !entry.pinned)
            })
            .copied()
    }

    fn global_victim(&self) -> Option<u64> {
        self.order
            .iter()
            .find(|&&key| self.entries.get(&key).is_some_and(|entry| !entry.pinned))
            .copied()
    }

    fn pinned_weight(&self, shard: Option<usize>, except: Option<u64>) -> u128 {
        self.entries
            .values()
            .filter(|entry| {
                entry.pinned
                    && Some(entry.key) != except
                    && shard.is_none_or(|s| entry.shard == s)
            })
            .map(|entry| entry.weight as u128)
            .sum()
    }

    /// Looks up a key and moves a hit to newest global recency.
    pub fn get(&mut self, key: u64) -> Option<i64> {
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.touch(key);
        }
        value
    }

    /// Changes the pin bit of an existing key without changing recency.
    pub fn set_pin(&mut self, key: u64, pinned: bool) -> Result<(), Error> {
        let entry = self.entries.get_mut(&key).ok_or(Error::Missing { key })?;
        entry.pinned = pinned;
        Ok(())
    }

    /// Removes an unpinned key and returns its value.
    pub fn remove(&mut self, key: u64) -> Result<i64, Error> {
        match self.entries.get(&key) {
            None => Err(Error::Missing { key }),
            Some(entry) if entry.pinned => Err(Error::Pinned { key }),
            Some(_) => Ok(self.erase(key).expect("entry was checked").value),
        }
    }

    /// Inserts or replaces a key, evicting oldest eligible entries from its shard.
    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        pinned: bool,
    ) -> Result<Vec<u64>, Error> {
        let shard = self.shard(key);
        let capacity = self.capacities[shard];

        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > capacity {
            return Err(Error::TooHeavy { weight, capacity });
        }

        let other_pinned = self.pinned_weight(Some(shard), Some(key));
        if other_pinned > (capacity - weight) as u128 {
            return Err(Error::Blocked { shard });
        }

        self.erase(key);
        let mut removed = Vec::new();
        while self.usage[shard] > capacity - weight {
            let victim = self
                .local_victim(shard)
                .expect("pinned feasibility guarantees an unpinned victim");
            self.erase(victim);
            removed.push(victim);
        }

        self.usage[shard] += weight;
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                weight,
                pinned,
                shard,
            },
        );
        self.touch(key);
        Ok(removed)
    }

    /// Sets a shard capacity and evicts its oldest unpinned entries as needed.
    pub fn resize(&mut self, shard: usize, capacity: u64) -> Result<Vec<u64>, Error> {
        if shard >= self.capacities.len() {
            return Err(Error::BadShard { shard });
        }

        let minimum = self.pinned_weight(Some(shard), None);
        if minimum > capacity as u128 {
            return Err(Error::PinnedCapacity {
                minimum,
                target: capacity as u128,
            });
        }

        self.capacities[shard] = capacity;
        let mut removed = Vec::new();
        while self.usage[shard] > capacity {
            let victim = self
                .local_victim(shard)
                .expect("pinned weight fits within the requested capacity");
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }

    /// Evicts globally oldest unpinned entries until usage is at most target.
    pub fn trim(&mut self, target: u128) -> Result<Vec<u64>, Error> {
        let minimum = self.pinned_weight(None, None);
        if minimum > target {
            return Err(Error::PinnedCapacity { minimum, target });
        }

        let mut removed = Vec::new();
        while self.used() > target {
            let victim = self
                .global_victim()
                .expect("pinned weight fits within the target");
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn local_eviction_and_global_order() {
        let mut cache = Cache::new(vec![3, 4]).unwrap();
        cache.insert(2, 20, 2, false).unwrap();
        cache.insert(1, 10, 2, false).unwrap();
        assert_eq!(cache.insert(4, 40, 2, false), Ok(vec![2]));
        assert_eq!(
            cache.snapshot().iter().map(|entry| entry.key).collect::<Vec<_>>(),
            vec![1, 4]
        );
        assert_eq!(cache.usage(), vec![2, 2]);
    }

    #[test]
    fn invalid_input_and_failed_replacement_preserve_state() {
        assert_eq!(Cache::new(vec![]), Err(Error::EmptyShards));
        let mut cache = Cache::new(vec![5]).unwrap();
        cache.insert(1, 10, 2, true).unwrap();
        let before = cache.snapshot();
        assert_eq!(
            cache.insert(1, 11, 0, false),
            Err(Error::ZeroWeight)
        );
        assert_eq!(cache.snapshot(), before);
        assert_eq!(cache.get(1), Some(10));
    }

    #[test]
    fn pins_trim_and_resize_end_to_end() {
        let mut cache = Cache::new(vec![9, 9]).unwrap();
        cache.insert(2, 20, 2, false).unwrap();
        cache.insert(1, 10, 3, true).unwrap();
        assert_eq!(cache.trim(4), Err(Error::PinnedCapacity { minimum: 3, target: 4 }).map(|_| vec![]));
        cache.set_pin(1, false).unwrap();
        assert_eq!(cache.trim(4), Ok(vec![2]));
        assert_eq!(cache.resize(1, 0), Ok(vec![1]));
        assert_eq!(cache.used(), 0);
    }

    #[test]
    fn pinned_get_touches_and_unpin_preserves_position() {
        let mut cache = Cache::new(vec![4]).unwrap();
        cache.insert(1, 10, 1, true).unwrap();
        cache.insert(2, 20, 1, false).unwrap();
        assert_eq!(cache.get(1), Some(10));
        cache.set_pin(1, false).unwrap();
        assert_eq!(
            cache.insert(3, 30, 3, false),
            Ok(vec![2, 1])
        );
    }
}
