#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn constraint_rank(constraint: &Constraint) -> (u8, Version) {
    match *constraint {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort_by_key(constraint_rank);
    constraints.dedup();
}

fn permits(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(expected) => version == expected,
        Constraint::AtLeast(minimum) => version >= minimum,
        Constraint::Before(limit) => version < limit,
        Constraint::Caret(base) => {
            version >= base
                && if base.major > 0 {
                    version.major == base.major
                } else if base.minor > 0 {
                    version.major == 0 && version.minor == base.minor
                } else {
                    version == base
                }
        }
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(value: &str) -> String {
    let mut encoded = String::new();
    for byte in value.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            encoded.push(byte as char);
        } else {
            encoded.push_str(&format!("%{byte:02X}"));
        }
    }
    encoded
}

/// Formats a resolution error as its deterministic, percent-encoded text representation.
pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
    }
}

fn active_requirements(
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    catalog_by_name: &std::collections::BTreeMap<String, &Package>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in selected {
        if let Some(package) = catalog_by_name.get(name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        canonicalize(constraints);
    }
    active
}

/// Resolves roots and their transitive dependencies using deterministic highest-version selection.
pub fn resolve(catalog: &[Package], roots: &[Requirement]) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let catalog_by_name: std::collections::BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let mut selected = std::collections::BTreeMap::<String, Version>::new();

    loop {
        let active = active_requirements(roots, &selected, &catalog_by_name);

        if let Some(name) = active
            .keys()
            .find(|name| !catalog_by_name.contains_key(*name))
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in &selected {
            if let Some(constraints) = active.get(name) {
                if !constraints
                    .iter()
                    .all(|constraint| permits(*constraint, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let mut next = None;
        for (name, constraints) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = catalog_by_name[name];
            let candidate = package
                .releases
                .iter()
                .filter(|release| {
                    constraints
                        .iter()
                        .all(|constraint| permits(*constraint, release.version))
                })
                .map(|release| release.version)
                .max();

            match candidate {
                None => {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
                Some(version) if next.is_none() => next = Some((name.clone(), version)),
                Some(_) => {}
            }
        }

        match next {
            Some((name, version)) => {
                selected.insert(name, version);
            }
            None => {
                return Ok(selected
                    .into_iter()
                    .map(|(package, version)| Selection { package, version })
                    .collect());
            }
        }
    }
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    fn v(major: u32, minor: u32, patch: u32) -> Version {
        Version {
            major,
            minor,
            patch,
        }
    }

    fn x(n: u32) -> Version {
        v(n, 0, 0)
    }

    fn q(package: &str, constraint: Constraint) -> Requirement {
        Requirement {
            package: package.into(),
            constraint,
        }
    }

    fn r(n: u32, deps: Vec<Requirement>) -> Release {
        Release { version: x(n), deps }
    }

    fn p(name: &str, releases: Vec<Release>) -> Package {
        Package {
            name: name.into(),
            releases,
        }
    }

    #[test]
    fn transitive_selection_and_order() {
        let catalog = vec![
            p("engine", vec![r(1, vec![q("util", Constraint::AtLeast(x(1)))])]),
            p("util", vec![r(1, vec![])]),
        ];
        let result = resolve(&catalog, &[q("engine", Constraint::Exact(x(1)))]).unwrap();
        assert_eq!(
            result,
            vec![
                Selection {
                    package: "engine".into(),
                    version: x(1)
                },
                Selection {
                    package: "util".into(),
                    version: x(1)
                }
            ]
        );
    }

    #[test]
    fn empty_roots_and_invalid_unreachable_name() {
        assert_eq!(resolve(&[], &[]).unwrap(), Vec::<Selection>::new());

        let catalog = vec![p("valid", vec![r(1, vec![q("Bad", Constraint::Exact(x(1)))])])];
        assert!(matches!(
            resolve(&catalog, &[]),
            Err(ResolveError::InvalidName { .. })
        ));
    }

    #[test]
    fn ranges_choose_highest_matching_version() {
        let catalog = vec![p("codec", vec![r(3, vec![]), r(1, vec![]), r(2, vec![])])];
        let result = resolve(
            &catalog,
            &[
                q("codec", Constraint::AtLeast(x(1))),
                q("codec", Constraint::Before(x(3))),
            ],
        )
        .unwrap();
        assert_eq!(result[0].version, x(2));
    }

    #[test]
    fn error_format_encodes_and_canonicalizes() {
        let error = ResolveError::NoMatch {
            package: "Bad Name".into(),
            requirements: vec![
                Constraint::Before(x(2)),
                Constraint::Exact(x(1)),
                Constraint::Exact(x(1)),
            ],
        };
        assert_eq!(
            error_text(&error),
            "no-match|package=Bad%20Name|constraints==1.0.0,<2.0.0"
        );
    }
}
