use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while matches!(byte(s, *pos), Some(b'0'..=b'9')) {
        *pos += 1;
    }
    if *pos == start {
        return match byte(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_ids(s: &str, start: usize, end: usize, prerelease: bool) -> Result<Vec<String>, VersionError> {
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end {
            let c = s.as_bytes()[p];
            if c.is_ascii_alphanumeric() || c == b'-' {
                p += 1;
            } else if c == b'.' {
                break;
            } else {
                return Err(VersionError::InvalidChar { at: p });
            }
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if prerelease && id.bytes().all(|c| c.is_ascii_digit()) && id.len() > 1 && id.starts_with('0') {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_owned());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

/// Parses a strict semantic version string.
pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    for _ in 0..2 {
        match byte(s, p) {
            Some(b'.') => p += 1,
            None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: p }),
            _ => return Err(VersionError::InvalidChar { at: p }),
        }
        let n = parse_core(s, &mut p)?;
        if p == 0 {
            unreachable!();
        }
        if _ == 0 {
            // The first post-major field is minor; retain it for the next iteration.
        }
        if _ == 0 {
            // Parsed below through the explicit field sequence instead.
        }
        if _ == 1 {
            let patch = n;
            let mut pre = Vec::new();
            let mut build = Vec::new();
            if byte(s, p) == Some(b'-') {
                let pre_start = p + 1;
                let plus = s[pre_start..].find('+').map(|i| pre_start + i);
                let pre_end = plus.unwrap_or(s.len());
                pre = parse_ids(s, pre_start, pre_end, true)?;
                p = pre_end;
            }
            if byte(s, p) == Some(b'+') {
                let build_start = p + 1;
                build = parse_ids(s, build_start, s.len(), false)?;
                p = s.len();
            }
            if p != s.len() {
                return Err(VersionError::InvalidChar { at: p });
            }
            return Ok(Version { major, minor: 0, patch, pre, build });
        }
    }
    unreachable!()
}

fn parse_version_fixed(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() { return Err(VersionError::Empty); }
    let mut p=0;
    let major=parse_core(s,&mut p)?;
    match byte(s,p) {
        Some(b'.')=>p+=1,
        None|Some(b'-'|b'+')=>return Err(VersionError::MissingCore{at:p}),
        _=>return Err(VersionError::InvalidChar{at:p})
    }
    let minor=parse_core(s,&mut p)?;
    match byte(s,p) {
        Some(b'.')=>p+=1,
        None|Some(b'-'|b'+')=>return Err(VersionError::MissingCore{at:p}),
        _=>return Err(VersionError::InvalidChar{at:p})
    }
    let patch=parse_core(s,&mut p)?;
    match byte(s,p) {
        None=>return Ok(Version{major,minor,patch,pre:vec![],build:vec![]}),
        Some(b'-'|b'+')=>{},
        _=>return Err(VersionError::InvalidChar{at:p})
    }
    let mut pre=Vec::new(); let mut build=Vec::new();
    if byte(s,p)==Some(b'-') {
        let st=p+1;
        let plus=s[st..].find('+').map(|i|st+i);
        let end=plus.unwrap_or(s.len());
        pre=parse_ids(s,st,end,true)?;
        p=end;
    }
    if byte(s,p)==Some(b'+') {
        let st=p+1;
        build=parse_ids(s,st,s.len(),false)?;
        p=s.len();
    }
    if p!=s.len(){return Err(VersionError::InvalidChar{at:p});}
    Ok(Version{major,minor,patch,pre,build})
}

/// Compares semantic-version precedence, ignoring build metadata.
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major,a.minor,a.patch).cmp(&(b.major,b.minor,b.patch)).then_with(|| {
        match (a.pre.is_empty(), b.pre.is_empty()) {
            (true,true)=>Ordering::Equal, (true,false)=>Ordering::Greater, (false,true)=>Ordering::Less,
            (false,false)=> {
                for (x,y) in a.pre.iter().zip(&b.pre) {
                    let xn=x.bytes().all(|c|c.is_ascii_digit());
                    let yn=y.bytes().all(|c|c.is_ascii_digit());
                    let o=match (xn,yn) {
                        (true,false)=>Ordering::Less, (false,true)=>Ordering::Greater,
                        (true,true)=>x.len().cmp(&y.len()).then_with(||x.as_bytes().cmp(y.as_bytes())),
                        _=>x.as_bytes().cmp(y.as_bytes())
                    };
                    if o!=Ordering::Equal{return o;}
                }
                a.pre.len().cmp(&b.pre.len())
            }
        }
    })
}

fn ws(b:u8)->bool { matches!(b,b' '|b'\t'|b'\r'|b'\n') }
fn tokens(s:&str, base:usize)->Vec<(String,usize)> {
    let mut out=Vec::new(); let mut i=0; let b=s.as_bytes();
    while i<b.len() {
        while i<b.len()&&ws(b[i]){i+=1;}
        if i==b.len(){break;}
        let st=i; while i<b.len()&&!ws(b[i]){i+=1;}
        out.push((s[st..i].to_owned(),base+st));
    }
    out
}

fn full_operand(s:&str,at:usize)->Result<Version,RangeError>{

    parse_version_fixed(s).map_err(|e|verror(at,e))

fn partial(s:&str,at:usize)->Result<(Version,usize,bool),RangeError>{
    let full=s.contains('-')||s.contains('+') || (s.split('.').count()>=3 && !s.split('.').any(|x|matches!(x,"*"|"x"|"X")));
    if full {return Ok((full_operand(s,at)?,3,true));}
    let f:Vec<&str>=s.split('.').collect();
    if f.is_empty()||f.len()>3||f.iter().any(|x|x.is_empty()){return Err(RangeError::Syntax{at});}
    let mut nums=Vec::new(); let mut wild=false;
    for x in &f {
        if matches!(*x,"*"|"x"|"X"){wild=true;}
        else if wild || !x.bytes().all(|c|c.is_ascii_digit()) || (x.len()>1&&x.starts_with('0')) {
            return Err(RangeError::Syntax{at});
        } else {
            nums.push(x.parse::<u64>().map_err(|_|RangeError::Syntax{at})?);
        }
    }
    let k=nums.len();
    Ok((Version{major:*nums.get(0).unwrap_or(&0),minor:*nums.get(1).unwrap_or(&0),patch:*nums.get(2).unwrap_or(&0),pre:vec![],build:vec![]},k,false))
}
fn endpoint_upper(v:&Version, k:usize, mode:&str, at:usize)->Result<Version,RangeError>{
    let mut u=Version{major:v.major,minor:v.minor,patch:v.patch,pre:vec!["0".into()],build:vec![]};
    let field=match mode {
        "caret"=>if v.major>0||k==1{0}else if v.minor>0||k==2{1}else{2},
        "tilde"=>if k==1{0}else{1},
        _=>if k==1{0}else{1}
    };
    let n=match field {0=>&mut u.major,1=>&mut u.minor,_=>&mut u.patch};
    *n=n.checked_add(1).ok_or(RangeError::Syntax{at})?;
    if field<=1{u.minor=if field==0{0}else{u.minor};}
    if field<=2{u.patch=if field<=1{0}else{u.patch};}
    Ok(u)
}
fn push_bound(arm:&mut Arm,v:Version,inclusive:bool,lower:bool){
    if !v.pre.is_empty(){arm.anchors.push((v.major,v.minor,v.patch));}
    arm.bounds.push(Bound{version:v,inclusive,lower});
}
fn parse_arm(ts:&[(String,usize)])->Result<Arm,RangeError>{
    let mut arm=Arm{bounds:vec![],anchors:vec![]};
    if ts.is_empty(){return Ok(arm);}
    let minuses:Vec<usize>=ts.iter().enumerate().filter(|(_,t)|t.0=="-").map(|(i,_)|i).collect();
    if !minuses.is_empty(){
        if ts.len()!=3||minuses!=vec![1]{return Err(RangeError::Syntax{at:ts[minuses[0]].1});}
        let (lv,lk,lf)=partial(&ts[0].0,ts[0].1)?;
        let (uv,uk,uf)=partial(&ts[2].0,ts[2].1)?;
        if lk==0{return Err(RangeError::Syntax{at:ts[0].1});}
        if uk==0{return Err(RangeError::Syntax{at:ts[2].1});}
        push_bound(&mut arm,lv,true,true);
        if uf {push_bound(&mut arm,uv,true,false);}
        else if uk<3 {let up=endpoint_upper(&uv,uk,"hyphen",ts[2].1)?;push_bound(&mut arm,up,false,false);}
        else {push_bound(&mut arm,uv,true,false);}
        return Ok(arm);
    }
    for (tok,at) in ts {
        let b=tok.as_bytes();
        if b.is_empty(){return Err(RangeError::Syntax{at:*at});}
        let first=b[0];
        if matches!(first,b'<'|b'>'|b'=') {
            let (op,n)=if tok.starts_with(">="){(">=",2)}else if tok.starts_with("<="){("<=",2)}else if tok.starts_with(">"){(">",1)}else if tok.starts_with("<"){("<",1)}else if tok.starts_with("="){("=",1)}else{("",0)};
            let v=full_operand(&tok[n..],at+n)?;
            if op=="=" {push_bound(&mut arm,v.clone(),true,true);push_bound(&mut arm,v,true,false);}
            else {push_bound(&mut arm,v,op==">=",op.starts_with('>'));}
        } else if first==b'^'||first==b'~' {
            let op=first; let sub=&tok[1..]; let subat=at+1;
            let (v,k,full)=partial(sub,subat)?;
            if k==0{return Err(RangeError::Syntax{at:*at});}
            push_bound(&mut arm,v.clone(),true,true);
            let up=endpoint_upper(&v,k,if op==b'^'{"caret"}else{"tilde"},subat)?;
            push_bound(&mut arm,up,false,false);
            let _=full;
        } else if first.is_ascii_digit()||matches!(first,b'*'|b'x'|b'X') {
            let (v,k,full)=partial(tok, *at)?;
            if full {push_bound(&mut arm,v.clone(),true,true);push_bound(&mut arm,v,true,false);}
            else if k==1||k==2 {
                push_bound(&mut arm,v.clone(),true,true);
                let up=endpoint_upper(&v,k,"partial",*at)?;
                push_bound(&mut arm,up,false,false);
            }
        } else {return Err(RangeError::Syntax{at:*at});}
    }
    Ok(arm)
}

/// Parses a range expression.
pub fn parse_range(s:&str)->Result<Range,RangeError>{
    let all=tokens(s,0);
    if all.is_empty(){return Err(RangeError::Empty{at:0});}
    let mut groups:Vec<Vec<(String,usize)>>=vec![vec![]];
    let mut prev_sep=false;
    for t in all {
        if t.0=="||" {
            if groups.last().unwrap().is_empty(){return Err(RangeError::Empty{at:t.1});}
            groups.push(vec![]);prev_sep=true;
        } else {groups.last_mut().unwrap().push(t);prev_sep=false;}
    }
    if groups.last().unwrap().is_empty(){
        let at=if prev_sep{s.len()}else{s.len()};
        return Err(RangeError::Empty{at});
    }
    let mut arms=Vec::new();
    for g in groups { arms.push(parse_arm(&g)?); }
    Ok(Range{arms})
}
fn arm_match(v:&Version,a:&Arm,include:bool)->bool {
    if !v.pre.is_empty()&&!include&&!a.anchors.contains(&(v.major,v.minor,v.patch)){return false;}
    a.bounds.iter().all(|b|{
        let o=compare_versions(v,&b.version);
        if b.lower {o==Ordering::Greater||(b.inclusive&&o==Ordering::Equal)}
        else{o==Ordering::Less||(b.inclusive&&o==Ordering::Equal)}
    })
}
/// Checks whether a version matches a range using default prerelease admission.
pub fn matches(v:&Version,r:&Range)->bool{matches_with(v,r,false)}
/// Checks a version against a range, optionally admitting all prereleases.
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool{
    r.arms.iter().any(|a|arm_match(v,a,include_prerelease))
}

pub struct CandidateError{pub index:usize,pub error:VersionError}
fn extrema(c:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in c.iter().enumerate(){parsed.push(parse_version_fixed(s).map_err(|error|CandidateError{index:i,error})?);}
    let mut best=None;
    for (i,v) in parsed.iter().enumerate(){
        if matches(v,r) {
            if best.map_or(true,|j|if max{compare_versions(v,&parsed[j])==Ordering::Greater}else{compare_versions(v,&parsed[j])==Ordering::Less}){best=Some(i);}
        }
    }
    Ok(best.map(|i|c[i].clone()))
}

/// Determines whether the two ranges have a common version.
pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool{
    if include_prerelease {
        // Bounds provide a practical witness for the supported finite comparator set.
        for x in &a.arms { for y in &b.arms {
            let mut candidates=vec![Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]}];
            candidates.extend(x.bounds.iter().map(|b|b.version.clone()));
            candidates.extend(y.bounds.iter().map(|b|b.version.clone()));
            if candidates.iter().any(|v|arm_match(v,x,true)&&arm_match(v,y,true)){return true;}
        }} return false;
    }
    for x in &a.arms { for y in &b.arms {
        let anchors:Vec<_>=x.anchors.iter().filter(|t|y.anchors.contains(t)).collect();
        for t in anchors {
            let v=Version{major:t.0,minor:t.1,patch:t.2,pre:vec!["0".into()],build:vec![]};
            if arm_match(&v,x,false)&&arm_match(&v,y,false){return true;}
        }
        let candidates=vec![
            Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]},
            Version{major:u64::MAX,minor:u64::MAX,patch:u64::MAX,pre:vec![],build:vec![]},
            x.bounds.first().map(|b|b.version.clone()).unwrap_or(Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]}),
            y.bounds.first().map(|b|b.version.clone()).unwrap_or(Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]}),
        ];
        if candidates.iter().any(|v|arm_match(v,x,false)&&arm_match(v,y,false)){return true;}
    }}
    false
}
fn version_display(v:&Version)->String{
    let mut s=format!("{}.{}.{}",v.major,v.minor,v.patch);
    if !v.pre.is_empty(){s.push('-');s.push_str(&v.pre.join("."));}
    s
}
/// Serializes a range into its canonical display form.
pub fn canonical(r:&Range)->String{
    let mut out=Vec::new();
    for a in &r.arms {
        let mut lo:Option<&Bound>=None;let mut hi:Option<&Bound>=None;
        for b in &a.bounds {
            let slot=if b.lower{&mut lo}else{&mut hi};
            if slot.map_or(true,|old|if b.lower{compare_versions(&b.version,&old.version)==Ordering::Greater||(compare_versions(&b.version,&old.version)==Ordering::Equal&&!b.inclusive)}else{compare_versions(&b.version,&old.version)==Ordering::Less||(compare_versions(&b.version,&old.version)==Ordering::Equal&&!b.inclusive)}) {*slot=Some(b);}
        }
        let l=lo.map(|b|format!("{}{}",if b.inclusive{">="}else{">"},version_display(&b.version))).unwrap_or("*".into());
        let h=hi.map(|b|format!("{}{}",if b.inclusive{"<="}else{"<"},version_display(&b.version))).unwrap_or("*".into());
        let mut anchors=a.anchors.clone();anchors.sort();anchors.dedup();
        let p=if anchors.is_empty(){"-".into()}else{anchors.iter().map(|t|format!("{}.{}.{}",t.0,t.1,t.2)).collect::<Vec<_>>().join(",")};
        out.push(format!("[{};{};pre={}]",l,h,p));
    }
    out.sort();out.dedup();if out.is_empty(){"EMPTY".into()}else{out.join(" || ")}
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage{name:String}, DuplicateLabel{package:String,label:String},
    InvalidCandidate{package:String,index:usize,error:VersionError},
    InvalidRange{package:String,label:String,path:Vec<usize>,error:RangeError},
    NoSolution{package:String,labels:Vec<String>}
}
fn validate_expr(e:&SetExpr,path:&mut Vec<usize>,pkg:&str,label:&str,out:&mut Vec<Vec<Option<Range>>>)->Result<(),SelectError>{
    match e {
        SetExpr::Range(s)=>{let r=parse_range(s).map_err(|error|SelectError::InvalidRange{package:pkg.into(),label:label.into(),path:path.clone(),error})?;out.push(vec![Some(r)]);Ok(())}
        SetExpr::Union(xs)|SetExpr::Intersection(xs)=>{for(i,x)in xs.iter().enumerate(){path.push(i);validate_expr(x,path,pkg,label,out)?;path.pop();}Ok(())}
    }
}
fn eval_expr(e:&SetExpr,v:&Version,ranges:&mut std::slice::Iter<Option<Range>>,inc:bool)->bool{
    match e {
        SetExpr::Range(_)=>ranges.next().and_then(|x|x.as_ref()).map_or(false,|r|matches_with(v,r,inc)),
        SetExpr::Union(xs)=>xs.iter().any(|x|eval_expr(x,v,ranges,inc)),
        SetExpr::Intersection(xs)=>xs.iter().all(|x|eval_expr(x,v,ranges,inc))
    }
}
/// Selects one satisfying candidate for every independent package.
pub fn select_packages(packages:&[Package],preference:Preference,include_prerelease:bool)->Result<Vec<Selection>,SelectError>{
    let mut names=std::collections::HashSet::new();
    for p in packages {if !names.insert(p.name.clone()){return Err(SelectError::DuplicatePackage{name:p.name.clone()});}}
    for p in packages {
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints {if !labels.insert(c.label.clone()){return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()});}}
    }
    let mut parsed=Vec::new();
    for p in packages {
        let mut vs=Vec::new();
        for(i,s)in p.candidates.iter().enumerate(){vs.push(parse_version_fixed(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);}
        parsed.push(vs);
    }
    let mut compiled:Vec<Vec<Vec<Option<Range>>>>=Vec::new();
    for p in packages {
        let mut cs=Vec::new();
        for c in &p.constraints {
            let mut leaves=Vec::new();validate_expr(&c.expr,&mut vec![],&p.name,&c.label,&mut leaves)?;
            cs.push(leaves.into_iter().flatten().collect());
        }
        compiled.push(cs);
    }
    let mut result=Vec::new();
    for (pi,p) in packages.iter().enumerate() {
        let mut good=Vec::new();
        'cand: for (i,v) in parsed[pi].iter().enumerate() {
            for (ci,c) in p.constraints.iter().enumerate() {
                let mut it=compiled[pi][ci].iter();
                if !eval_expr(&c.expr,v,&mut it,include_prerelease){continue 'cand;}
            }
            good.push(i);
        }
        if good.is_empty(){
            let mut diag=Vec::new();
            if !p.candidates.is_empty() {
                let n=p.constraints.len();
                'search: for size in 1..=n {
                    let mut comb=Vec::new();
                    fn gen(start:usize,n:usize,k:usize,cur:&mut Vec<usize>,out:&mut Vec<Vec<usize>>){
                        if cur.len()==k{out.push(cur.clone());return;}
                        for i in start..n{cur.push(i);gen(i+1,n,k,cur,out);cur.pop();}
                    }
                    let mut all=Vec::new();gen(0,n,size,&mut comb,&mut all);
                    let mut choices=Vec::new();
                    for set in all {
                        let any=parsed[pi].iter().any(|v|set.iter().all(|&j|{
                            let c=&p.constraints[j];let mut it=compiled[pi][j].iter();eval_expr(&c.expr,v,&mut it,include_prerelease)
                        }));
                        if !any {let mut labs=set.iter().map(|&j|p.constraints[j].label.clone()).collect::<Vec<_>>();labs.sort();choices.push(labs);}
                    }
                    if !choices.is_empty(){choices.sort();diag=choices.remove(0);break 'search;}
                }
            }
            return Err(SelectError::NoSolution{package:p.name.clone(),labels:diag});
        }
        let mut best=good[0];
        for &i in good.iter().skip(1){
            let o=compare_versions(&parsed[pi][i],&parsed[pi][best]);
            let choose=match preference {
                Preference::Highest=>o==Ordering::Greater,
                Preference::Lowest=>o==Ordering::Less,
                Preference::StableFirst=>{
                    let si=parsed[pi][i].pre.is_empty();let sb=parsed[pi][best].pre.is_empty();
                    if si!=sb{si}else{o==Ordering::Greater}
                }
            };
            if choose{best=i;}
        }
        result.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }
    Ok(result)
}
