fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;

        let candidates=vec![
            Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]},
            Version{major:u64::MAX,minor:u64::MAX,patch:u64::MAX,pre:vec![],build:vec![]},
            x.bounds.first().map(|b|b.version.clone()).unwrap_or(Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]}),
            y.bounds.first().map(|b|b.version.clone()).unwrap_or(Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]}),
        ];
        if candidates.iter().any(|v|arm_match(v,x,false)&&arm_match(v,y,false)){return true;}
    }}
    false
}
fn version_display(v:&Version)->String{
    let mut s=format!("{}.{}.{}",v.major,v.minor,v.patch);
    if !v.pre.is_empty(){s.push('-');s.push_str(&v.pre.join("."));}
    s
}
/// Serializes a range into its canonical display form.
pub fn canonical(r:&Range)->String{
    let mut out=Vec::new();
    for a in &r.arms {
        let mut lo:Option<&Bound>=None;let mut hi:Option<&Bound>=None;
        for b in &a.bounds {
            let slot=if b.lower{&mut lo}else{&mut hi};
            if slot.map_or(true,|old|if b.lower{compare_versions(&b.version,&old.version)==Ordering::Greater||(compare_versions(&b.version,&old.version)==Ordering::Equal&&!b.inclusive)}else{compare_versions(&b.version,&old.version)==Ordering::Less||(compare_versions(&b.version,&old.version)==Ordering::Equal&&!b.inclusive)}) {*slot=Some(b);}
        }
        let l=lo.map(|b|format!("{}{}",if b.inclusive{">="}else{">"},version_display(&b.version))).unwrap_or("*".into());
        let h=hi.map(|b|format!("{}{}",if b.inclusive{"<="}else{"<"},version_display(&b.version))).unwrap_or("*".into());
        let mut anchors=a.anchors.clone();anchors.sort();anchors.dedup();
        let p=if anchors.is_empty(){"-".into()}else{anchors.iter().map(|t|format!("{}.{}.{}",t.0,t.1,t.2)).collect::<Vec<_>>().join(",")};
        out.push(format!("[{};{};pre={}]",l,h,p));
    }
    out.sort();out.dedup();if out.is_empty(){"EMPTY".into()}else{out.join(" || ")}
}

pub enum SelectError {
    DuplicatePackage{name:String}, DuplicateLabel{package:String,label:String},
    InvalidCandidate{package:String,index:usize,error:VersionError},
    InvalidRange{package:String,label:String,path:Vec<usize>,error:RangeError},

fn validate_expr(e:&SetExpr,path:&mut Vec<usize>,pkg:&str,label:&str,out:&mut Vec<Vec<Option<Range>>>)->Result<(),SelectError>{
    match e {
        SetExpr::Range(s)=>{let r=parse_range(s).map_err(|error|SelectError::InvalidRange{package:pkg.into(),label:label.into(),path:path.clone(),error})?;out.push(vec![Some(r)]);Ok(())}
        SetExpr::Union(xs)|SetExpr::Intersection(xs)=>{for(i,x)in xs.iter().enumerate(){path.push(i);validate_expr(x,path,pkg,label,out)?;path.pop();}Ok(())}
    }
}
fn eval_expr(e:&SetExpr,v:&Version,ranges:&mut std::slice::Iter<Option<Range>>,inc:bool)->bool{
    match e {
        SetExpr::Range(_)=>ranges.next().and_then(|x|x.as_ref()).map_or(false,|r|matches_with(v,r,inc)),
        SetExpr::Union(xs)=>xs.iter().any(|x|eval_expr(x,v,ranges,inc)),
        SetExpr::Intersection(xs)=>xs.iter().all(|x|eval_expr(x,v,ranges,inc))
    }
}
/// Selects one satisfying candidate for every independent package.
pub fn select_packages(packages:&[Package],preference:Preference,include_prerelease:bool)->Result<Vec<Selection>,SelectError>{
    let mut names=std::collections::HashSet::new();
    for p in packages {if !names.insert(p.name.clone()){return Err(SelectError::DuplicatePackage{name:p.name.clone()});}}
    for p in packages {
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints {if !labels.insert(c.label.clone()){return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()});}}
    }
    let mut parsed=Vec::new();
    for p in packages {
        let mut vs=Vec::new();
        for(i,s)in p.candidates.iter().enumerate(){vs.push(parse_version_fixed(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);}
        parsed.push(vs);
    }
    let mut compiled:Vec<Vec<Vec<Option<Range>>>>=Vec::new();
    for p in packages {
        let mut cs=Vec::new();
        for c in &p.constraints {
            let mut leaves=Vec::new();validate_expr(&c.expr,&mut vec![],&p.name,&c.label,&mut leaves)?;
            cs.push(leaves.into_iter().flatten().collect());
        }
        compiled.push(cs);
    }
    let mut result=Vec::new();
    for (pi,p) in packages.iter().enumerate() {
        let mut good=Vec::new();
        'cand: for (i,v) in parsed[pi].iter().enumerate() {
            for (ci,c) in p.constraints.iter().enumerate() {
                let mut it=compiled[pi][ci].iter();
                if !eval_expr(&c.expr,v,&mut it,include_prerelease){continue 'cand;}
            }
            good.push(i);
        }
        if good.is_empty(){
            let mut diag=Vec::new();
            if !p.candidates.is_empty() {
                let n=p.constraints.len();
                'search: for size in 1..=n {
                    let mut comb=Vec::new();
                    fn gen(start:usize,n:usize,k:usize,cur:&mut Vec<usize>,out:&mut Vec<Vec<usize>>){
                        if cur.len()==k{out.push(cur.clone());return;}
                        for i in start..n{cur.push(i);gen(i+1,n,k,cur,out);cur.pop();}
                    }
                    let mut all=Vec::new();gen(0,n,size,&mut comb,&mut all);
                    let mut choices=Vec::new();
                    for set in all {
                        let any=parsed[pi].iter().any(|v|set.iter().all(|&j|{
                            let c=&p.constraints[j];let mut it=compiled[pi][j].iter();eval_expr(&c.expr,v,&mut it,include_prerelease)
                        }));
                        if !any {let mut labs=set.iter().map(|&j|p.constraints[j].label.clone()).collect::<Vec<_>>();labs.sort();choices.push(labs);}
                    }
                    if !choices.is_empty(){choices.sort();diag=choices.remove(0);break 'search;}
                }
            }
            return Err(SelectError::NoSolution{package:p.name.clone(),labels:diag});
        }
        let mut best=good[0];
        for &i in good.iter().skip(1){
            let o=compare_versions(&parsed[pi][i],&parsed[pi][best]);
            let choose=match preference {
                Preference::Highest=>o==Ordering::Greater,
                Preference::Lowest=>o==Ordering::Less,
                Preference::StableFirst=>{
                    let si=parsed[pi][i].pre.is_empty();let sb=parsed[pi][best].pre.is_empty();
                    if si!=sb{si}else{o==Ordering::Greater}
                }
            };
            if choose{best=i;}
        }
        result.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }
    Ok(result)
}
