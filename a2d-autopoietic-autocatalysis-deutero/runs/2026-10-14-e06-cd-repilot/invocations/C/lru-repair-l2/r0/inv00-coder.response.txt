use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroPeriod,
    Disabled,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub frequency: u8,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacity: usize,
    period: u32,
    pending: u32,
    entries: BTreeMap<u64, Entry>,
    recency: VecDeque<u64>,
}

impl Cache {
    /// Creates an empty cache with the given capacity and aging period.
    pub fn new(capacity: usize, period: u32) -> Result<Self, Error> {
        if period == 0 {
            return Err(Error::ZeroPeriod);
        }
        Ok(Self {
            capacity,
            period,
            pending: 0,
            entries: BTreeMap::new(),
            recency: VecDeque::new(),
        })
    }

    /// Returns the cache's entry capacity.
    pub fn capacity(&self) -> usize {
        self.capacity
    }

    /// Returns the number of counted operations since the last automatic aging round.
    pub fn pending(&self) -> u32 {
        self.pending
    }

    /// Returns cloned entries in eviction priority order.
    pub fn snapshot(&self) -> Vec<Entry> {
        let mut result = self.entries_by_recency();
        result.sort_by_key(|entry| {
            let recency_index = self
                .recency
                .iter()
                .position(|key| *key == entry.key)
                .unwrap_or(usize::MAX);
            (entry.frequency, recency_index)
        });
        result
    }

    fn entries_by_recency(&self) -> Vec<Entry> {
        self.recency
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect()
    }

    fn halve(&mut self) {
        for entry in self.entries.values_mut() {
            entry.frequency = (entry.frequency + 1) / 2;
        }
    }

    fn begin(&mut self) {
        self.pending += 1;
        if self.pending == self.period {
            self.halve();
            self.pending = 0;
        }
    }

    fn touch(&mut self, key: u64) {
        self.recency.retain(|existing| *existing != key);
        self.recency.push_back(key);
    }

    fn credit(&mut self, key: u64) {
        if let Some(entry) = self.entries.get_mut(&key) {
            entry.frequency = entry.frequency.saturating_add(1).min(7);
        }
        self.touch(key);
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.recency.retain(|existing| *existing != key);
        Some(entry)
    }

    fn victim(&self) -> Option<u64> {
        self.recency.iter().copied().min_by_key(|key| {
            let frequency = self.entries.get(key).map_or(u8::MAX, |entry| entry.frequency);
            (frequency, self.recency.iter().position(|item| item == key).unwrap_or(usize::MAX))
        })
    }

    fn take_victim(&mut self) -> Option<u64> {
        let key = self.victim()?;
        self.erase(key);
        Some(key)
    }

    /// Looks up a key, crediting and touching it on a hit.
    pub fn get(&mut self, key: u64) -> Option<i64> {
        self.begin();
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.credit(key);
        }
        value
    }

    /// Inserts or updates a key, returning any eviction.
    pub fn put(&mut self, key: u64, value: i64) -> Result<Vec<u64>, Error> {
        if self.capacity == 0 {
            return Err(Error::Disabled);
        }

        self.begin();
        if let Some(entry) = self.entries.get_mut(&key) {
            entry.value = value;
            entry.frequency = entry.frequency.saturating_add(1).min(7);
            self.touch(key);
            return Ok(Vec::new());
        }

        let mut evicted = Vec::new();
        if self.entries.len() >= self.capacity {
            if let Some(victim) = self.take_victim() {
                evicted.push(victim);
            }
        }

        self.entries.insert(
            key,
            Entry {
                key,
                value,
                frequency: 1,
            },
        );
        self.touch(key);
        Ok(evicted)
    }

    /// Removes a key and returns its value, if present.
    pub fn remove(&mut self, key: u64) -> Option<i64> {
        self.begin();
        self.erase(key).map(|entry| entry.value)
    }

    /// Applies the requested number of aging rounds.
    pub fn age(&mut self, rounds: u8) {
        for _ in 0..rounds {
            self.halve();
        }
    }

    /// Changes capacity and returns evicted keys in priority order.
    pub fn resize(&mut self, capacity: usize) -> Vec<u64> {
        self.capacity = capacity;
        let mut evicted = Vec::new();
        while self.entries.len() > capacity {
            if let Some(victim) = self.take_victim() {
                evicted.push(victim);
            } else {
                break;
            }
        }
        evicted
    }
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn lfu_basic_path() {
        let mut c = Cache::new(2, 100).unwrap();
        c.put(1, 10).unwrap();
        c.put(2, 20).unwrap();
        assert_eq!(c.get(1), Some(10));
        assert_eq!(c.put(3, 30), Ok(vec![2]));
        assert_eq!(
            c.snapshot().iter().map(|entry| entry.key).collect::<Vec<_>>(),
            vec![3, 1]
        );
        assert_eq!(c.capacity(), 2);
    }

    #[test]
    fn aging_and_pending() {
        let mut c = Cache::new(3, 10).unwrap();
        c.put(4, 40).unwrap();
        c.get(4);
        c.get(4);
        c.get(4);
        assert_eq!(c.pending(), 4);
        c.age(1);
        assert_eq!(c.snapshot()[0].frequency, 2);
        assert_eq!(c.pending(), 4);
    }

    #[test]
    fn disabled_put_does_not_count() {
        let mut c = Cache::new(0, 1).unwrap();
        assert_eq!(c.put(1, 0), Err(Error::Disabled));
        assert_eq!(c.pending(), 0);
        assert!(c.get(1).is_none());
        assert_eq!(c.pending(), 0);
    }

    #[test]
    fn update_remove_resize_end_to_end() {
        let mut c = Cache::new(4, 100).unwrap();
        c.put(8, 80).unwrap();
        assert_eq!(c.put(8, 81), Ok(vec![]));
        assert_eq!(c.remove(8), Some(81));
        c.put(9, 90).unwrap();
        c.put(10, 100).unwrap();
        assert_eq!(c.resize(1), vec![9]);
        assert_eq!(c.get(10), Some(100));
    }
}
