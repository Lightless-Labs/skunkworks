            });
        }

        let expires_at = match ttl {
            Some(duration) => Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };

        let mut removed = self.expire(now);
        self.erase(key);

        if ttl == Some(0) {
            return Ok(removed);
        }

        // Subtraction is safe after validation; this avoids overflowing used + weight.
        let target_used = self.capacity - weight;
        while self.used > target_used {
            if let Some(victim) = self.oldest() {
                self.erase(victim);
                removed.push(victim);
            } else {
                break;
            }
        }

        self.used += weight;
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                weight,
                expires_at,
            },
        );
        self.touch(key);
        Ok(removed)
    }

    /// Sets a new capacity, purging expired entries and evicting oldest entries as needed.
    pub fn resize(&mut self, capacity: u64, now: u64) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        let mut removed = self.expire(now);
        self.capacity = capacity;

        while self.used > capacity {
            if let Some(key) = self.oldest() {
                self.erase(key);
                removed.push(key);
            } else {
                break;
            }
        }
        Ok(removed)
    }
