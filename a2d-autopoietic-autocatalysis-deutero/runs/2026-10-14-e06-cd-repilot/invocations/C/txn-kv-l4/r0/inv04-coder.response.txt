use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Entry {
    value: Option<String>,
    deadline: Option<u64>,
}

impl Entry {
    fn live_at(&self, now: u64) -> bool {
        self.value.is_some() && self.deadline.is_none_or(|d| d > now)
    }
}

#[derive(Debug, Clone)]
struct Version {
    revision: u64,
    entry: Entry,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Entry>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    /// Creates an empty store with clock and revision set to zero.
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn selected_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return Some(entry.clone());
            }
            let begin_revision = transaction.begin_revision;
            return self.data.get(key)?.iter()
                .rev()
                .find(|v| v.revision <= begin_revision)
                .map(|v| v.entry.clone());
        }
        self.data.get(key)?.last().map(|v| v.entry.clone())
    }

    fn visible(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, Entry> {
        let mut result = BTreeMap::new();
        for key in self.data.keys() {
            if let Some(entry) = self.selected_entry(key, tx) {
                if entry.live_at(now) {
                    result.insert(key.clone(), entry);
                }
            }
        }
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                for (key, entry) in &transaction.writes {
                    if entry.live_at(now) {
                        result.insert(key.clone(), entry.clone());
                    } else {
                        result.remove(key);
                    }
                }
            }
        }
        result
    }

    fn append(&mut self, key: String, entry: Entry) {
        self.revision += 1;
        self.data.entry(key).or_default().push(Version {
            revision: self.revision,
            entry,
        });
    }

    /// Begins a transaction at the current committed revision.
    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(id, Transaction {
            begin_revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    /// Commits a transaction atomically, or returns its first write conflict.
    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self.transactions.get(&id).ok_or(Error::UnknownTransaction(id))?;
        let mut conflicts = Vec::new();
        for key in transaction.writes.keys() {
            if self.data.get(key).and_then(|v| v.last())
                .is_some_and(|v| v.revision > transaction.begin_revision)
            {
                conflicts.push(key.clone());
            }
        }
        if let Some(key) = conflicts.into_iter().next() {
            return Err(Error::Conflict { tx: id, key });
        }
        let transaction = self.transactions.remove(&id).unwrap();
        if transaction.writes.is_empty() {
            return Ok(self.revision);
        }
        self.revision += 1;
        for (key, entry) in transaction.writes {
            self.data.entry(key).or_default().push(Version {
                revision: self.revision,
                entry,
            });
        }
        Ok(self.revision)
    }

    /// Rolls back and removes an active transaction.
    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        self.transactions.remove(&id).ok_or(Error::UnknownTransaction(id))?;
        Ok(())
    }

    /// Sets a value, optionally with a relative time-to-live.
    pub fn set(&mut self, now: u64, tx: Option<u64>, key: &str, value: &str, ttl: Option<u64>) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let entry = match ttl {
            Some(0) => Entry { value: None, deadline: None },
            Some(d) => Entry {
                value: Some(value.to_string()),
                deadline: Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            },
            None => Entry { value: Some(value.to_string()), deadline: None },
        };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), entry);
        } else {
            self.append(key.to_string(), entry);
        }
        Ok(())
    }

    /// Deletes a key and reports whether it was live before deletion.
    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let was_live = self.selected_entry(key, tx).is_some_and(|e| e.live_at(now));
        let tombstone = Entry { value: None, deadline: None };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), tombstone);
        } else {
            self.append(key.to_string(), tombstone);
        }
        Ok(was_live)
    }

    /// Returns a live value for a key, if present in the requested view.
    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.selected_entry(key, tx)
            .filter(|e| e.live_at(now))
            .and_then(|e| e.value))
    }

    /// Counts live keys whose values exactly match the supplied string.
    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self.visible(tx, now).values()
            .filter(|e| e.value.as_deref() == Some(value))
            .count())
    }

    /// Returns a filtered page of live key/value pairs.
    pub fn scan(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>, offset: usize, limit: usize, cursor: Option<&Cursor>) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(c) = cursor {
            if c.start != start || c.end.as_deref() != end || c.prefix.as_deref() != prefix
                || c.offset != offset || c.last.is_empty()
                || c.last.as_str() < start
                || end.is_some_and(|e| c.last.as_str() >= e)
                || prefix.is_some_and(|p| !c.last.starts_with(p))
            {
                return Err(Error::CursorMismatch);
            }
        }
        let visible = self.visible(tx, now);
        let mut keys: Vec<_> = visible.keys().filter(|k| {
            k.as_str() >= start
                && end.is_none_or(|e| k.as_str() < e)
                && prefix.is_none_or(|p| k.starts_with(p))
                && cursor.is_none_or(|c| k.as_str() > c.last)
        }).cloned().collect();
        if cursor.is_none() {
            keys = keys.into_iter().skip(offset).collect();
        }
        let take = limit.min(keys.len());
        let has_more = take < keys.len();
        let selected: Vec<String> = keys.into_iter().take(take).collect();
        let items = selected.iter().map(|k| {
            (k.clone(), visible.get(k).and_then(|e| e.value.clone()).unwrap())
        }).collect();
        let next = if has_more && !selected.is_empty() {
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: selected.last().unwrap().clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    /// Counts and sums canonical signed integer values in the filtered view.
    pub fn aggregate(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        let visible = self.visible(tx, now);
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, entry) in visible {
            if key.as_str() < start
                || end.is_some_and(|e| key.as_str() >= e)
                || prefix.is_some_and(|p| !key.starts_with(p))
            {
                continue;
            }
            let value = entry.value.unwrap();
            let parsed = parse_canonical_i64(&value)
                .ok_or_else(|| Error::InvalidNumber { key: key.clone(), value: value.clone() })?;
            sum = sum.checked_add(parsed).ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    /// Saves a materialized copy of the current committed live data.
    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let copy = self.visible(None, now);
        self.snapshots.insert(name.to_string(), copy);
        Ok(())
    }

    /// Restores a saved snapshot and returns the resulting committed revision.
    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        validate_name(name)?;
        let saved = self.snapshots.get(name).ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }
        let saved_live: BTreeMap<String, Entry> = saved.iter()
            .filter(|(_, e)| e.live_at(now))
            .map(|(k, e)| (k.clone(), e.clone()))
            .collect();
        let current = self.visible(None, now);
        let keys: BTreeSet<String> = current.keys().chain(saved_live.keys()).cloned().collect();
        if keys.is_empty() {
            return Ok(self.revision);
        }
        self.revision += 1;
        for key in keys {
            let entry = saved_live.get(&key).cloned()
                .unwrap_or(Entry { value: None, deadline: None });
            self.data.entry(key).or_default().push(Version {
                revision: self.revision,
                entry,
            });
        }
        Ok(self.revision)
    }

    /// Removes a named snapshot.
    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        validate_name(name)?;
        self.snapshots.remove(name)
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        Ok(())
    }

    /// Collects obsolete committed versions while retaining versions pinned by transactions.
    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        let begin_revisions: Vec<u64> = self.transactions.values().map(|t| t.begin_revision).collect();
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for begin in &begin_revisions {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= *begin) {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain { removed += 1; }
                retain
            });
        }
        Ok(removed)
    }

    /// Returns counts for the current store state.
    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values().flat_map(|v| v.iter())
            .filter(|v| v.entry.value.is_none()).count();
        let live_keys = self.visible(None, now).len();
        Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.values().filter(|v| !v.is_empty()).count(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        }.pipe(Ok)
    }
}

fn validate_name(name: &str) -> Result<(), Error> {
    for (index, ch) in name.char_indices() {
        let valid = if index == 0 {
            ch.is_ascii_lowercase()
        } else {
            ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
        };
        if !valid {
            return Err(Error::NameInvalid { at: index });
        }
    }
    Ok(())
}

fn parse_canonical_i64(value: &str) -> Option<i64> {
    if value == "0" {
        return Some(0);
    }
    let digits = value.strip_prefix('-').unwrap_or(value);
    if digits.is_empty()
        || digits.starts_with('0')
        || !digits.bytes().all(|b| b.is_ascii_digit())
        || (value.starts_with('-') && digits == "0")
    {
        return None;
    }
    value.parse::<i64>().ok()
}

trait Pipe: Sized {
    fn pipe<T>(self, f: impl FnOnce(Self) -> T) -> T { f(self) }
}
impl<T> Pipe for T {}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn transaction_isolation_and_commit() {
        let mut s = Store::new();
        s.set(1, None, "key", "base", None).unwrap();
        s.begin(1, 7, false).unwrap();
        s.set(2, None, "key", "later", None).unwrap();
        assert_eq!(s.get(2, Some(7), "key"), Ok(Some("base".into())));
        s.set(2, Some(7), "other", "own", None).unwrap();
        assert_eq!(s.commit(2, 7), Ok(3));
    }

    #[test]
    fn invalid_input_and_clock_admission() {
        let mut s = Store::new();
        assert_eq!(s.get(3, None, ""), Err(Error::KeyEmpty));
        assert_eq!(s.stats(2), Err(Error::TimeBackwards { previous: 3, now: 2 }));
    }

    #[test]
    fn scan_aggregate_and_snapshot_round_trip() {
        let mut s = Store::new();
        s.set(0, None, "a", "2", None).unwrap();
        s.set(0, None, "b", "-1", None).unwrap();
        assert_eq!(s.aggregate(0, None, "", None, None), Ok(Aggregate { count: 2, sum: 1 }));
        let page = s.scan(0, None, "", None, None, 0, 1, None).unwrap();
        assert_eq!(page.items, vec![("a".into(), "2".into())]);
        s.snapshot(0, "saved").unwrap();
        s.delete(0, None, "a").unwrap();
        assert_eq!(s.restore(0, "saved"), Ok(4));
        assert_eq!(s.get(0, None, "a"), Ok(Some("2".into())));
    }
}
