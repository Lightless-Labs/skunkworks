use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, at: usize) -> Option<u8> {
    s.as_bytes().get(at).copied()
}

fn core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while matches!(byte_at(s, *pos), Some(b'0'..=b'9')) {
        *pos += 1;
    }

    if *pos == start {
        return match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut pos = ident_start;
        while pos < end {
            let b = s.as_bytes()[pos];
            if b.is_ascii_alphanumeric() || b == b'-' {
                pos += 1;
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }

        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &s[ident_start..pos];
        if prerelease
            && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.len() > 1
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        result.push(ident.to_owned());
        if pos == end {
            break;
        }

        ident_start = pos + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

/// Parses a strict semantic version string.
pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = core_field(s, &mut pos)?;
    for _ in 0..2 {
        match byte_at(s, pos) {
            Some(b'.') => pos += 1,
            None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        let value = core_field(s, &mut pos)?;
        if _ == 0 {
            // The second core number is the minor field.
            // Stored below after parsing the patch.
        }
        if _ == 0 {
            // Keep the field values explicit without changing validation order.
            let minor = value;
            match byte_at(s, pos) {
                Some(b'.') => {
                    pos += 1;
                    let patch = core_field(s, &mut pos)?;
                    return finish_version(s, pos, major, minor, patch);
                }
                None | Some(b'-' | b'+') => {
                    return Err(VersionError::MissingCore { at: pos });
                }
                _ => return Err(VersionError::InvalidChar { at: pos }),
            }
        }
    }

    unreachable!()
}

fn finish_version(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if byte_at(s, pos) == Some(b'-') {
        pos += 1;
        let pre_start = pos;
        let pre_end = s[pos..]
            .find('+')
            .map(|n| pos + n)
            .unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }

    if byte_at(s, pos) == Some(b'+') {
        pos += 1;
        build = parse_identifiers(s, pos, s.len(), false)?;
        pos = s.len();
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = !a.is_empty() && a.bytes().all(|c| c.is_ascii_digit());
    let b_numeric = !b.is_empty() && b.bytes().all(|c| c.is_ascii_digit());

    match (a_numeric, b_numeric) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

/// Compares versions by semantic precedence, ignoring build metadata.
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_identifiers(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

/// Parses a conjunction of full-version comparators.
pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_space(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_space(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, operand_offset) = if token.starts_with(">=") {
            (Op::Ge, 2)
        } else if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with('>') {
            (Op::Gt, 1)
        } else if token.starts_with('<') {
            (Op::Lt, 1)
        } else if token.starts_with('=') {
            (Op::Eq, 1)
        } else {
            (Op::Eq, 0)
        };

        if operand_offset == token.len()
            || !token.as_bytes()[operand_offset].is_ascii_digit()
        {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &token[operand_offset..];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: token_start + operand_offset,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { comparators })
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

/// Returns whether a version satisfies every comparator in a range.
pub fn matches(v: &Version, r: &Range) -> bool {
    let prerelease_admitted = r.comparators.iter().any(|c| {
        !c.version.pre.is_empty()
            && (c.version.major, c.version.minor, c.version.patch)
                == (v.major, v.minor, v.patch)
    });

    if !v.pre.is_empty() && !prerelease_admitted {
        return false;
    }

    r.comparators.iter().all(|c| {
        let ord = compare_versions(v, &c.version);
        match c.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    #[test]
    fn parses_and_compares_versions() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(
            compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")),
            Ordering::Less
        );
    }

    #[test]
    fn rejects_invalid_core_and_identifier() {
        assert_eq!(
            parse_version("01.2.3"),
            Err(VersionError::LeadingZero { at: 0 })
        );
        assert_eq!(
            parse_version("1.2.3-01"),
            Err(VersionError::LeadingZero { at: 6 })
        );
        assert_eq!(
            parse_version("1.2.3+"),
            Err(VersionError::EmptyIdentifier { at: 6 })
        );
    }

    #[test]
    fn range_conjunction_and_prerelease_admission() {
        let range = parse_range(">=2.1.0 <3.0.0").unwrap();
        assert!(matches(&pv("2.5.0"), &range));
        assert!(!matches(&pv("3.0.0"), &range));

        let prerelease_range = parse_range(">=2.1.3-beta <2.1.3").unwrap();
        assert!(matches(&pv("2.1.3-rc"), &prerelease_range));
        assert!(!matches(&pv("2.1.4-rc"), &prerelease_range));
    }
}
