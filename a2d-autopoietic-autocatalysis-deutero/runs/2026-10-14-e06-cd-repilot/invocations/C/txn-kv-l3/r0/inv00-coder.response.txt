use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    NoTransaction,
    DepthLimit,
    SnapshotMissing(String),
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Entry {
    value: String,
    deadline: Option<u64>,
}

type Data = BTreeMap<String, Entry>;

pub struct Store {
    committed: Data,
    scopes: Vec<Data>,
    snapshots: BTreeMap<String, Data>,
    clock: u64,
}

impl Store {
    /// Creates an empty store with its logical clock set to zero.
    pub fn new() -> Self {
        Self {
            committed: BTreeMap::new(),
            scopes: Vec::new(),
            snapshots: BTreeMap::new(),
            clock: 0,
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        self.purge_visible(now);
        Ok(())
    }

    fn visible(&self) -> &Data {
        self.scopes.last().unwrap_or(&self.committed)
    }

    fn visible_mut(&mut self) -> &mut Data {
        self.scopes.last_mut().unwrap_or(&mut self.committed)
    }

    fn purge(data: &mut Data, now: u64) {
        data.retain(|_, entry| entry.deadline.is_none_or(|deadline| deadline > now));
    }

    fn purge_visible(&mut self, now: u64) {
        Self::purge(self.visible_mut(), now);
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (i, b) in name.bytes().enumerate() {
            let valid = if i == 0 {
                b.is_ascii_lowercase()
            } else {
                b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'
            };
            if !valid {
                return Err(Error::NameInvalid { at: i });
            }
        }
        Ok(())
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.is_none_or(|e| key < e)
            && prefix.is_none_or(|p| key.starts_with(p))
    }

    /// Opens a working scope copied from the current visible map.
    pub fn begin(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.scopes.len() >= 4 {
            return Err(Error::DepthLimit);
        }
        self.scopes.push(self.visible().clone());
        Ok(())
    }

    /// Commits the innermost scope into its parent map.
    pub fn commit(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        let data = self.scopes.pop().ok_or(Error::NoTransaction)?;
        if let Some(parent) = self.scopes.last_mut() {
            *parent = data;
        } else {
            self.committed = data;
        }
        Ok(())
    }

    /// Discards the innermost scope and restores its parent map.
    pub fn rollback(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.scopes.pop().is_none() {
            return Err(Error::NoTransaction);
        }
        self.purge_visible(now);
        Ok(())
    }

    /// Sets a key to a value, optionally expiring it after the given duration.
    pub fn set(
        &mut self,
        now: u64,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        let deadline = match ttl {
            Some(0) => {
                self.visible_mut().remove(key);
                return Ok(());
            }
            Some(d) => Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };
        self.visible_mut().insert(
            key.to_owned(),
            Entry {
                value: value.to_owned(),
                deadline,
            },
        );
        Ok(())
    }

    /// Deletes a key and returns whether it was present.
    pub fn delete(&mut self, now: u64, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.visible_mut().remove(key).is_some())
    }

    /// Returns an owned copy of the value for a nonempty key.
    pub fn get(&mut self, now: u64, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.visible().get(key).map(|entry| entry.value.clone()))
    }

    /// Counts visible keys whose complete value equals the supplied value.
    pub fn count(&mut self, now: u64, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        Ok(self
            .visible()
            .values()
            .filter(|entry| entry.value == value)
            .count())
    }

    /// Returns the number of open working scopes.
    pub fn depth(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        Ok(self.scopes.len())
    }

    /// Saves a copy of the current visible map under a validated name.
    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        self.snapshots.insert(name.to_owned(), self.visible().clone());
        Ok(())
    }

    /// Replaces the visible map with a copy of a named snapshot.
    pub fn restore(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        let mut restored = saved;
        Self::purge(&mut restored, now);
        *self.visible_mut() = restored;
        Ok(())
    }

    /// Removes a named snapshot.
    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_owned()));
        }
        Ok(())
    }

    /// Returns a sorted, filtered page of visible key/value pairs.
    pub fn scan(
        &mut self,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let data = self.visible();
        let mut selected = data.iter().filter(|(key, _)| {
            Self::matches(key, start, end, prefix)
                && cursor.is_none_or(|c| key.as_str() > c.last.as_str())
        });

        if cursor.is_none() {
            for _ in 0..offset {
                if selected.next().is_none() {
                    return Ok(Page {
                        items: Vec::new(),
                        next: None,
                    });
                }
            }
        }

        let mut items = Vec::new();
        for _ in 0..limit {
            match selected.next() {
                Some((key, entry)) => items.push((key.clone(), entry.value.clone())),
                None => break,
            }
        }

        let has_more = !items.is_empty() && selected.next().is_some();
        let next = if has_more {
            Some(Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: items.last().unwrap().0.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    /// Sums canonical signed integer values for keys matching the filters.
    pub fn aggregate(
        &mut self,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }

        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, entry) in self.visible().iter() {
            if !Self::matches(key, start, end, prefix) {
                continue;
            }
            let Some(value) = parse_canonical_i64(&entry.value) else {
                return Err(Error::InvalidNumber {
                    key: key.clone(),
                    value: entry.value.clone(),
                });
            };
            sum = sum.checked_add(value).ok_or_else(|| Error::SumOverflow {
                key: key.clone(),
            })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }
}

fn parse_canonical_i64(s: &str) -> Option<i64> {
    if s == "0" {
        return Some(0);
    }
    let digits = s.strip_prefix('-').unwrap_or(s);
    if digits.is_empty()
        || digits.starts_with('0')
        || !digits.bytes().all(|b| b.is_ascii_digit())
        || (s.starts_with('-') && digits == "0")
    {
        return None;
    }
    s.parse::<i64>().ok()
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn ttl_and_empty_key() {
        let mut s = Store::new();
        s.set(2, "x", "v", Some(2)).unwrap();
        assert_eq!(s.get(3, "x"), Ok(Some("v".into())));
        assert_eq!(s.get(4, "x"), Ok(None));
        assert_eq!(s.get(4, ""), Err(Error::KeyEmpty));
    }

    #[test]
    fn nested_scope_and_snapshot() {
        let mut s = Store::new();
        s.set(0, "a", "1", None).unwrap();
        s.begin(1).unwrap();
        s.set(1, "a", "2", None).unwrap();
        s.snapshot(1, "saved").unwrap();
        s.rollback(2).unwrap();
        assert_eq!(s.get(2, "a"), Ok(Some("1".into())));
        s.restore(2, "saved").unwrap();
        assert_eq!(s.get(2, "a"), Ok(Some("2".into())));
    }

    #[test]
    fn scan_aggregate_and_invalid_number() {
        let mut s = Store::new();
        s.set(0, "aa", "2", None).unwrap();
        s.set(0, "ab", "-1", None).unwrap();
        let p = s.scan(0, "", None, Some("a"), 0, 1, None).unwrap();
        assert_eq!(p.items, vec![("aa".into(), "2".into())]);
        assert!(p.next.is_some());
        assert_eq!(
            s.aggregate(0, "", None, None),
            Ok(Aggregate { count: 2, sum: 1 })
        );
        s.set(0, "zz", "01", None).unwrap();
        assert_eq!(
            s.aggregate(0, "", None, None),
            Err(Error::InvalidNumber {
                key: "zz".into(),
                value: "01".into()
            })
        );
    }

    #[test]
    fn public_pages_end_to_end() {
        let mut s = Store::new();
        for (k, v) in [("aa", "1"), ("ab", "2"), ("ac", "3")] {
            s.set(0, k, v, None).unwrap();
        }
        let p = s.scan(0, "", None, Some("a"), 0, 2, None).unwrap();
        let q = s
            .scan(0, "", None, Some("a"), 0, 2, p.next.as_ref())
            .unwrap();
        assert_eq!(q.items, vec![("ac".into(), "3".into())]);
        assert_eq!(q.next, None);
    }
}
