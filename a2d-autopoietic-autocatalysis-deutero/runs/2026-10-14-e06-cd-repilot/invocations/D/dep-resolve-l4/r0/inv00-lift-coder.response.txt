fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(values: &[Constraint]) -> Vec<Constraint> {
    let mut out = values.to_vec();
    out.sort();
    out.dedup();
    out
}

fn percent_encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    out
}

fn error_evidence_lines(evidence: &[Evidence]) -> String {
    let mut out = String::new();
    for ev in sorted_evidence(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(ev.constraint));
        out.push_str(" via ");
        out.push_str(&ev.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            percent_encode(package),
            version_text(*version),
            percent_encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", percent_encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            percent_encode(package),
            error_evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let constraints = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "no-match|package={}|constraints={}{}",
                percent_encode(package),
                constraints,
                error_evidence_lines(evidence)
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let constraints = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={}{}",
                percent_encode(package),
                version_text(*selected),
                constraints,
                error_evidence_lines(evidence)
            )
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\
             \n  F:{} via {}",
            percent_encode(package),
            version_text(*selected),
            percent_encode(feature),
            percent_encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|c| matches_constraint(version, *c))
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn sorted_features(features: &[String]) -> Vec<String> {
    let mut out = features.to_vec();
    out.sort();
    out.dedup();
    out
}

#[derive(Clone, Default)]
struct Active {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requested: BTreeMap<String, BTreeSet<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

use std::collections::{BTreeMap, BTreeSet};

fn root_token(index: usize, root: &Requirement) -> String {
    let features = sorted_features(&root.features).join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        features
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let features = sorted_features(&dep.features).join(",");
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        gate,
        platform
    )
    .replace("+{}", &format!("+{{{}}}", features))
}

fn compute_active(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let packages: BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut active = Active::default();
    let mut queue = Vec::<(String, String)>::new();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        active
            .constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        active
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: path.clone(),
            });
        active.requested.entry(root.package.clone()).or_default().extend(
            root.features.iter().cloned(),
        );
        choose_path(&mut active.paths, &root.package, path);
        queue.push((root.package.clone(), format!("\0root:{}", i)));
    }

    let mut processed_features = BTreeSet::<(String, String)>::new();
    let mut processed_deps = BTreeSet::<(String, Version, usize)>::new();

    loop {
        let mut progress = false;

        let package_names: Vec<String> = active.requested.keys().cloned().collect();
        for package_name in package_names {
            let Some(version) = selected.get(&package_name).copied() else {
                continue;
            };
            let Some(package) = packages.get(package_name.as_str()) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested: Vec<String> = active
                .requested
                .get(&package_name)
                .map(|s| s.iter().cloned().collect())
                .unwrap_or_default();

            for feature_name in requested {
                if !processed_features.insert((package_name.clone(), feature_name.clone())) {
                    continue;
                }
                active
                    .activated
                    .entry(package_name.clone())
                    .or_default()
                    .insert(feature_name.clone());
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        active
                            .requested
                            .entry(package_name.clone())
                            .or_default()
                            .insert(enabled.clone());
                    }
                }
                progress = true;
            }

            for (dep_index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|gate| {
                        !active
                            .activated
                            .get(&package_name)
                            .is_some_and(|features| features.contains(gate))
                    })
                {
                    continue;
                }
                if !processed_deps.insert((package_name.clone(), version, dep_index)) {
                    continue;
                }

                let source_path = active
                    .paths
                    .get(&package_name)
                    .cloned()
                    .unwrap_or_default();
                let token = dep_token(&package_name, version, dep_index, dep);
                let mut incoming_path = source_path;
                incoming_path.push(token);

                active
                    .constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                active
                    .origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: incoming_path.clone(),
                    });
                active
                    .requested
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                choose_path(&mut active.paths, &dep.package, incoming_path);
                progress = true;
            }
        }

        if !progress {
            break;
        }
    }

    active
}

fn choose_path(paths: &mut BTreeMap<String, Vec<String>>, package: &str, path: Vec<String>) {
    match paths.get(package) {
        Some(old) if old.len() < path.len() || (old.len() == path.len() && old <= &path) => {}
        _ => {
            paths.insert(package.to_string(), path);
        }
    }
}

fn canonical_active_constraints(active: &Active, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        active
            .constraints
            .get(package)
            .map(Vec::as_slice)
            .unwrap_or(&[]),
    )
}

fn active_evidence(active: &Active, package: &str) -> Vec<Evidence> {
    let mut evidence: Vec<Evidence> = active
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    evidence.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    evidence
}

fn dynamic_error(
    catalog: &[Package],
    selected: &BTreeMap<String, Version>,
    active: &Active,
) -> Option<ResolveError> {
    for name in active.constraints.keys() {
        if !catalog.iter().any(|p| &p.name == name) {
            return Some(ResolveError::Missing {
                package: name.clone(),
                evidence: active_evidence(active, name),
            });
        }
    }

    for (name, version) in selected {
        if !active.constraints.contains_key(name) {
            continue;
        }
        let constraints = canonical_active_constraints(active, name);
        if !matches_all(*version, &constraints) {
            return Some(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: constraints,
                evidence: active_evidence(active, name),
            });
        }
    }

    for (name, constraints) in &active.constraints {
        if selected.contains_key(name) {
            continue;
        }
        let Some(package) = catalog.iter().find(|p| &p.name == name) else {
            continue;
        };
        if !package
            .releases
            .iter()
            .any(|r| matches_all(r.version, constraints))
        {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: canonical_constraints(constraints),
                evidence: active_evidence(active, name),
            });
        }
    }

    for (name, version) in selected {
        if !active.constraints.contains_key(name) {
            continue;
        }
        let Some(package) = catalog.iter().find(|p| &p.name == name) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        if let Some(features) = active.activated.get(name) {
            for feature in features {
                if !release.features.iter().any(|f| &f.name == feature) {
                    return Some(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: active.paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }
    }

    None
}

struct SearchResult {
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

fn lock_cost(selected: &BTreeMap<String, Version>, locks: &[Lock]) -> usize {
    locks
        .iter()
        .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
        .count()
}

fn better_solution(
    candidate: &(usize, BTreeMap<String, Version>),
    current: &(usize, BTreeMap<String, Version>),
    package_names: &[String],
) -> bool {
    if candidate.0 != current.0 {
        return candidate.0 < current.0;
    }
    for name in package_names {
        let a = candidate.1.get(name);
        let b = current.1.get(name);
        match (a, b) {
            (Some(av), Some(bv)) if av != bv => return av > bv,
            (Some(_), None) => return true,
            (None, Some(_)) => return false,
            _ => {}
        }
    }
    false
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
    attempts: &mut usize,
    selected: &mut BTreeMap<String, Version>,
    package_names: &[String],
) -> Result<SearchResult, ResolveError> {
    let active = compute_active(catalog, roots, selected, target);

    if let Some(error) = dynamic_error(catalog, selected, &active) {
        return Ok(SearchResult {
            best: None,
            first_error: Some(error),
        });
    }

    let next = active
        .constraints
        .keys()
        .find(|name| !selected.contains_key(*name))
        .cloned();

    let Some(package_name) = next else {
        return Ok(SearchResult {
            best: Some((lock_cost(selected, locks), selected.clone())),
            first_error: None,
        });
    };

    let package = catalog
        .iter()
        .find(|p| p.name == package_name)
        .expect("active packages were checked for absence");
    let constraints = active
        .constraints
        .get(&package_name)
        .map(Vec::as_slice)
        .unwrap_or(&[]);
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|r| matches_all(r.version, constraints))
        .map(|r| r.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(lock) = locks.iter().find(|l| l.package == package_name) {
        if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
            let locked = candidates.remove(pos);
            candidates.insert(0, locked);
        }
    }

    let mut result = SearchResult {
        best: None,
        first_error: None,
    };

    for candidate in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.insert(package_name.clone(), candidate);

        let child = search(
            catalog,
            roots,
            locks,
            target,
            max_attempts,
            attempts,
            selected,
            package_names,
        );

        selected.remove(&package_name);

        let child = child?;
        if let Some(solution) = child.best {
            if result
                .best
                .as_ref()
                .is_none_or(|current| better_solution(&solution, current, package_names))
            {
                result.best = Some(solution);
            }
        }
        if result.first_error.is_none() {
            result.first_error = child.first_error;
        }
    }

    Ok(result)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    let mut attempts = 0;
    let mut selected = BTreeMap::new();
    let result = search(
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        &mut attempts,
        &mut selected,
        &package_names,
    )?;

    let Some((_, best)) = result.best else {
        return Err(result.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_active = compute_active(catalog, roots, &best, target);
    let selected_vec: Vec<Selection> = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let before: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let after = best;
    let mut all_names = BTreeSet::new();
    all_names.extend(before.keys().cloned());
    all_names.extend(after.keys().cloned());
    let mut changes = Vec::new();

    for package in all_names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog.iter().any(|p| {
                    p.name == package && p.releases.iter().any(|r| r.version == old)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else if !matches_all(
                    old,
                    &canonical_active_constraints(&final_active, &package),
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
