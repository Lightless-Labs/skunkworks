#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    text_pieces: Vec<String>,
    consumed: usize,
    marker_ended: bool,
    raw: bool,
    pieces: usize,
    gap: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn line_text(line: &Line) -> String {
    let mut s = " ".repeat(line.indent);
    for (i, piece) in line.text_pieces.iter().enumerate() {
        if i != 0 {
            s.push(' ');
        }
        s.push_str(piece);
    }
    s
}

fn consumed_vector(solution: &Solution) -> Vec<usize> {
    solution.lines.iter().map(|l| l.consumed).collect()
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    let av = consumed_vector(a);
    let bv = consumed_vector(b);
    if av != bv {
        return av > bv;
    }
    let at: Vec<String> = a.lines.iter().map(line_text).collect();
    let bt: Vec<String> = b.lines.iter().map(line_text).collect();
    at < bt
}

fn word_piece(word: &Word, start: usize, end: usize) -> String {
    word.chars[start..end].iter().collect()
}

fn solve_paragraph(words: &[Word], opts: &WrapOptions) -> Vec<Line> {
    use std::collections::HashMap;

    type Key = (usize, usize, bool);
    fn rec(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        opts: &WrapOptions,
        memo: &mut HashMap<Key, Option<Solution>>,
    ) -> Option<Solution> {
        if wi == words.len() {
            return Some(Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            });
        }
        let key = (wi, off, first);
        if let Some(cached) = memo.get(&key) {
            return cached.clone();
        }

        let indent = if first { opts.first_indent } else { opts.rest_indent };
        let cap = opts.width - indent;
        let mut candidates: Vec<(usize, usize, bool, bool, Vec<String>, usize)> = Vec::new();
        let mut used = 0usize;
        let mut pieces = Vec::new();
        let mut idx = wi;
        let mut start = off;
        let mut piece_count = 0usize;

        loop {
            let word = &words[idx];
            let remaining = word.chars.len() - start;
            let room = cap.saturating_sub(used + if piece_count > 0 { 1 } else { 0 });

            // Complete this word, and optionally continue with following words.
            if remaining <= room {
                let add = if piece_count > 0 { 1 } else { 0 };
                used += add;
                pieces.push(word_piece(word, start, word.chars.len()));
                used += remaining;
                piece_count += 1;
                candidates.push((idx + 1, 0, false, false, pieces.clone(), piece_count));
                if idx + 1 == words.len() {
                    break;
                }
                idx += 1;
                start = 0;
                continue;
            }

            // Marker-ended prefixes.
            for &m in &word.markers {
                if m > start && m < word.chars.len() {
                    let prefix_len = m - start;
                    let add = if piece_count > 0 { 1 } else { 0 };
                    if used + add + prefix_len + 1 <= cap {
                        let mut p = pieces.clone();
                        if piece_count > 0 {
                            // Space is represented by the normal join.
                        }
                        p.push(format!("{}-", word_piece(word, start, m)));
                        candidates.push((idx, m, true, false, p, piece_count + 1));
                    }
                }
            }

            // Raw split is allowed only when this line has no prior pieces.
            if piece_count == 0 && remaining > cap {
                let marker_fits = word
                    .markers
                    .iter()
                    .any(|&m| m > start && m < word.chars.len() && m - start + 1 <= cap);
                if !marker_fits {
                    let end = start + cap;
                    pieces.push(word_piece(word, start, end));
                    candidates.push((idx, end, false, true, pieces.clone(), 1));
                }
            }
            break;
        }

        let mut best: Option<Solution> = None;
        for (nwi, noff, marker, raw, pieces, pc) in candidates {
            let displayed: usize = pieces.iter().map(|p| p.chars().count()).sum::<usize>()
                + pieces.len().saturating_sub(1);
            let gap = opts.width - indent - displayed;
            let next = rec(
                nwi,
                noff,
                false,
                words,
                opts,
                memo,
            );
            let Some(mut tail) = next else { continue };
            let nonfinal = !tail.lines.is_empty();
            let mut cost = (gap * gap) as u64;
            if !nonfinal {
                cost /= 4;
            }
            if marker {
                cost += opts.hyphen_cost as u64;
            }
            if raw {
                cost += (2 * opts.width + 1) as u64;
            }
            if nonfinal && pc == 1 {
                cost += 3;
            }
            let line = Line {
                indent,
                text_pieces: pieces,
                consumed: if marker || raw {
                    if nwi == wi { noff - off } else { 0 }
                } else {
                    0
                },
                marker_ended: marker,
                raw,
                pieces: pc,
                gap,
            };
            tail.cost += cost;
            tail.marker_lines += usize::from(marker);
            tail.lines.insert(0, line);
            if best.as_ref().is_none_or(|b| better(&tail, b)) {
                best = Some(tail);
            }
        }

        memo.insert(key, best.clone());
        best
    }

    let mut memo = HashMap::new();
    let mut solution = rec(0, 0, true, words, opts, &mut memo)
        .unwrap_or(Solution { cost: 0, marker_lines: 0, lines: Vec::new() });
    let _ = &mut solution;
    solution.lines
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word { chars: Vec::new(), markers: Vec::new() };
    for &c in chars {
        if separators(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word { chars: Vec::new(), markers: Vec::new() };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    words
}

fn render(lines: &[Line], opts: &WrapOptions) -> Vec<String> {
    let mut out = Vec::new();
    for (li, line) in lines.iter().enumerate() {
        let mut s = " ".repeat(line.indent);
        let gaps = line.text_pieces.len().saturating_sub(1);
        let justify = opts.justify && li + 1 < lines.len() && gaps > 0;
        let extra = if justify { opts.width - (line.indent + line.text_pieces.iter().map(|p| p.chars().count()).sum::<usize>() + gaps) } else { 0 };
        for (i, piece) in line.text_pieces.iter().enumerate() {
            if i > 0 {
                s.push(' ');
                if justify {
                    let base = extra / gaps;
                    let rem = extra % gaps;
                    let start = li % gaps;
                    let pos = i - 1;
                    let cyclic = (pos + gaps - start) % gaps;
                    s.push_str(&" ".repeat(base + usize::from(cyclic < rem)));
                }
            }
            s.push_str(piece);
        }
        out.push(s);
    }
    out
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let count = input.chars().count();
    if count > 512 {
        return Err(WrapError::InputTooLong { chars: count });
    }

    let mut original = Vec::new();
    let mut byte_indices = Vec::new();
    for (i, c) in input.char_indices() {
        original.push(c);
        byte_indices.push(i);
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index: i });
        }
    }

    let mut marker_bad = None;
    for i in 0..original.len() {
        if original[i] == '~' {
            let good = i > 0
                && i + 1 < original.len()
                && !separators(original[i - 1])
                && !separators(original[i + 1])
                && original[i - 1] != '~'
                && original[i + 1] != '~';
            if !good {
                marker_bad = Some(byte_indices[i]);
                break;
            }
        }
    }
    if let Some(index) = marker_bad {
        return Err(WrapError::InvalidMarker { index });
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < original.len() {
        if original[i] == '\r' {
            if i + 1 < original.len() && original[i + 1] == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(original[i]);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut seg_start = 0;
    let mut pos = 0;
    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let mut end = pos;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - pos >= 2 {
                let words = parse_words(&normalized[seg_start..pos]);
                let lines = solve_paragraph(&words, options);
                output.extend(render(&lines, options));
                for _ in 1..(end - pos) {
                    output.push(String::new());
                }
                seg_start = end;
                pos = end;
                continue;
            }
        }
        pos += 1;
    }
    let words = parse_words(&normalized[seg_start..]);
    let lines = solve_paragraph(&words, options);
    output.extend(render(&lines, options));
    output
}
