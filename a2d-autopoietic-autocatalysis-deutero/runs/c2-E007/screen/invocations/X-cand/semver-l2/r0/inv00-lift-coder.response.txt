use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn core_number(s: &str, start: usize, end: usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    if start == end {
        return Err(VersionError::MissingCore { at: start });
    }
    let field = &s[start..end];
    if field.len() > 1 && field.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    field
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut pos = start;
    loop {
        let ident_start = pos;
        while pos < end && bytes[pos] != b'.' {
            let b = bytes[pos];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..pos];
        if is_pre
            && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.len() > 1
            && ident.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        ids.push(ident.to_string());
        if pos == end {
            break;
        }
        pos += 1;
        if pos == end {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
    }
    ids
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let b = s.as_bytes();
    let mut pos = 0;
    let mut nums = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < b.len() && b[pos].is_ascii_digit() {
            pos += 1;
        }
        nums[i] = core_number(s, start, pos)?;
        if i < 2 {
            if pos >= b.len() || b[pos] == b'-' || b[pos] == b'+' {
                return Err(VersionError::MissingCore { at: pos });
            }
            if b[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < b.len() {
        match b[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let plus = b[pre_start..]
                    .iter()
                    .position(|&c| c == b'+')
                    .map(|i| pre_start + i);
                let pre_end = plus.unwrap_or(b.len());
                pre = parse_identifiers(s, pre_start, pre_end, true)?;
                pos = pre_end;
                if pos < b.len() {
                    let build_start = pos + 1;
                    build = parse_identifiers(s, build_start, b.len(), false)?;
                }
            }
            b'+' => {
                build = parse_identifiers(s, pos + 1, b.len(), false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let o = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => o == Ordering::Equal,
        Op::Lt => o == Ordering::Less,
        Op::Le => o != Ordering::Greater,
        Op::Gt => o == Ordering::Greater,
        Op::Ge => o != Ordering::Less,
    }
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    while p < end {
        while p < end && whitespace(b[p]) {
            p += 1;
        }
        if p == end {
            break;
        }
        let st = p;
        while p < end && !whitespace(b[p]) {
            p += 1;
        }
        out.push((st, &s[st..p]));
    }
    out
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}

fn partial_or_full(s: &str, at: usize) -> Result<Partial, RangeError> {
    let bytes = s.as_bytes();
    let fullish = s.contains('-')
        || s.contains('+')
        || (s.split('.').count() >= 3
            && !s.split('.').any(|x| matches!(x, "*" | "x" | "X")));
    if fullish {
        let v = parse_version(s).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            nums: vec![v.major, v.minor, v.patch],
            full: Some(v),
        });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    let _ = bytes;
    Ok(Partial { nums, full: None })
}

fn padded(p: &Partial) -> (u64, u64, u64) {
    (
        *p.nums.first().unwrap_or(&0),
        *p.nums.get(1).unwrap_or(&0),
        *p.nums.get(2).unwrap_or(&0),
    )
}

fn version_tuple(v: Version) -> Version {
    v
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn add_cmp(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push((v.major, v.minor, v.patch));
    }
    arm.comparators.push(Comparator { op, version: v });
}

fn bump(value: u64, at: usize) -> Result<u64, RangeError> {
    value.checked_add(1).ok_or(RangeError::Syntax { at })
}

fn upper_bare(p: &Partial, at: usize) -> Result<Option<Version>, RangeError> {
    let k = p.nums.len();
    let (ma, mi, pa) = padded(p);
    match k {
        0 => Ok(None),
        1 => Ok(Some(endpoint(bump(ma, at)?, 0, 0))),
        2 => Ok(Some(endpoint(ma, bump(mi, at)?, 0))),
        _ => Ok(None),
    }
}

fn lower_version(p: &Partial) -> Version {
    if let Some(v) = &p.full {
        v.clone()
    } else {
        let (a, b, c) = padded(p);
        endpoint(a, b, c)
    }
}

fn parse_expanding(
    token: &str,
    token_at: usize,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let b = token.as_bytes();
    let first = b[0];
    if matches!(first, b'<' | b'>' | b'=') {
        let (op, n) = if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with(">=") {
            (Op::Ge, 2)
        } else {
            (match first {
                b'<' => Op::Lt,
                b'>' => Op::Gt,
                _ => Op::Eq,
            }, 1)
        };
        let operand_at = token_at + n;
        let operand = &token[n..];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_at });
        }
        let v = parse_version(operand)
            .map_err(|error| RangeError::Version { at: operand_at, error })?;
        add_cmp(arm, op, v, true);
        return Ok(());
    }

    let (prefix, operand, operand_at) = if first == b'^' || first == b'~' {
        (Some(first), &token[1..], token_at + 1)
    } else {
        (None, token, token_at)
    };

    if let Some(mark) = prefix {
        let p = partial_or_full(operand, operand_at)?;
        if p.nums.is_empty() {
            return Err(RangeError::Syntax { at: token_at });
        }
        let lower = lower_version(&p);
        add_cmp(arm, Op::Ge, lower, p.full.is_some());
        let (ma, mi, pa) = padded(&p);
        let k = p.nums.len();
        let up = if mark == b'^' {
            if ma > 0 || k == 1 {
                endpoint(bump(ma, token_at)?, 0, 0)
            } else if mi > 0 || k == 2 {
                endpoint(ma, bump(mi, token_at)?, 0)
            } else {
                endpoint(ma, mi, bump(pa, token_at)?)
            }
        } else if k == 1 {
            endpoint(bump(ma, token_at)?, 0, 0)
        } else {
            endpoint(ma, bump(mi, token_at)?, 0)
        };
        add_cmp(arm, Op::Lt, up, false);
        return Ok(());
    }

    let p = partial_or_full(token, token_at)?;
    if let Some(v) = p.full {
        add_cmp(arm, Op::Eq, v, true);
    } else if let Some(up) = upper_bare(&p, token_at)? {
        let low = lower_version(&p);
        add_cmp(arm, Op::Ge, low, false);
        add_cmp(arm, Op::Lt, up, false);
    }
    Ok(())
}

fn parse_hyphen_arm(ts: &[(usize, &str)], arm: &mut Arm) -> Result<bool, RangeError> {
    let minus = ts.iter().find(|(_, t)| *t == "-");
    if minus.is_none() {
        return Ok(false);
    }
    if ts.len() != 3 || ts[1].1 != "-" {
        return Err(RangeError::Syntax {
            at: minus.unwrap().0,
        });
    }
    let (lo_at, lo_s) = ts[0];
    let (hi_at, hi_s) = ts[2];
    let lo = partial_or_full(lo_s, lo_at)?;
    let hi = partial_or_full(hi_s, hi_at)?;
    if lo.nums.is_empty() {
        return Err(RangeError::Syntax { at: lo_at });
    }
    if hi.nums.is_empty() {
        return Err(RangeError::Syntax { at: hi_at });
    }
    let low = lower_version(&lo);
    add_cmp(arm, Op::Ge, low, lo.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
    if let Some(v) = hi.full {
        add_cmp(arm, Op::Le, v, true);
    } else {
        let k = hi.nums.len();
        let (ma, mi, _) = padded(&hi);
        let up = if k == 1 {
            endpoint(bump(ma, hi_at)?, 0, 0)
        } else if k == 2 {
            endpoint(ma, bump(mi, hi_at)?, 0)
        } else {
            endpoint(ma, mi, hi.nums[2])
        };
        if k < 3 {
            add_cmp(arm, Op::Lt, up, false);
        } else {
            add_cmp(arm, Op::Le, up, false);
        }
    }
    Ok(true)
}

fn parse_arm(s: &str, ts: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    if parse_hyphen_arm(ts, &mut arm)? {
        return Ok(arm);
    }
    for &(at, token) in ts {
        if token.is_empty() {
            continue;
        }
        let first = token.as_bytes()[0];
        if !(first.is_ascii_digit()
            || matches!(first, b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'='))
        {
            return Err(RangeError::Syntax { at });
        }
        parse_expanding(token, at, &mut arm)?;
    }
    let _ = s;
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut arms = Vec::new();
    let mut pos = 0;
    let mut saw_any = false;
    let mut current_start = 0;
    let mut current_end = 0;
    let mut found_or = false;

    while pos < b.len() {
        while pos < b.len() && whitespace(b[pos]) {
            pos += 1;
        }
        if pos == b.len() {
            break;
        }
        if pos + 1 < b.len() && &b[pos..pos + 2] == b"||" {
            if !saw_any || current_end == 0 {
                return Err(RangeError::Empty { at: pos });
            }
            let ts = tokens(s, current_start, current_end);
            arms.push(parse_arm(s, &ts)?);
            pos += 2;
            found_or = true;
            saw_any = false;
            current_start = pos;
            current_end = 0;
            continue;
        }
        if !saw_any {
            current_start = pos;
            saw_any = true;
        }
        let st = pos;
        while pos < b.len() && !whitespace(b[pos]) {
            pos += 1;
        }
        current_end = pos;
        if pos < b.len() {
            let mut q = pos;
            while q < b.len() && whitespace(b[q]) {
                q += 1;
            }
            if q + 1 < b.len() && &b[q..q + 2] == b"||" {
                let ts = tokens(s, current_start, current_end);
                arms.push(parse_arm(s, &ts)?);
                pos = q + 2;
                found_or = true;
                saw_any = false;
                current_start = pos;
                current_end = 0;
            } else {
                let _ = st;
            }
        }
    }

    if found_or {
        if !saw_any || current_end == 0 {
            return Err(RangeError::Empty { at: s.len() });
        }
    }
    if saw_any {
        let ts = tokens(s, current_start, current_end);
        arms.push(parse_arm(s, &ts)?);
    } else if arms.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        arm.comparators.iter().all(|c| cmp_matches(v, c))
            && (v.pre.is_empty()
                || arm
                    .anchors
                    .iter()
                    .any(|&(a, b, c)| (v.major, v.minor, v.patch) == (a, b, c)))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (maximum && ord == Ordering::Greater)
                    || (!maximum && ord == Ordering::Less)
                {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}
