#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(32..=126).contains(&byte) && byte != b'\t' && byte != b'\n' && byte != b'\r' {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c| c == ' ' || c == '\t' || c == '\n' || c == '\r')
        .filter(|word| !word.is_empty())
    {
        let mut suffix = word;

        loop {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if suffix.len() <= width {
                let needed = suffix.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(suffix);
                    break;
                }

                if !line.is_empty() {
                    lines.push(std::mem::take(&mut line));
                    continue;
                }
            } else if line.is_empty() {
                lines.push(suffix[..width].to_string());
                suffix = &suffix[width..];
                if suffix.is_empty() {
                    break;
                }
                continue;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&suffix[..available]);
                    lines.push(std::mem::take(&mut line));
                    suffix = &suffix[available..];
                    if suffix.is_empty() {
                        break;
                    }
                    continue;
                }

                lines.push(std::mem::take(&mut line));
                continue;
            }

            break;
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
