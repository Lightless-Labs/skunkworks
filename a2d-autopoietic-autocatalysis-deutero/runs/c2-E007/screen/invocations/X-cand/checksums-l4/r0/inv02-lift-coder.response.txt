#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc_raw = crc32_byte_raw(self.crc_raw, byte);
            self.adler_a = (self.adler_a + byte as u32) % 65521;
            self.adler_b = (self.adler_b + self.adler_a) % 65521;
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        for offset in (1..9).chain(11..19) {
            if hex_value(bytes[offset]).is_none() {
                return Err(SnapshotError::Hex { offset });
            }
        }

        let crc = parse_hex_u32(bytes, 1, 9);
        let adler = parse_hex_u32(bytes, 11, 19);
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn crc32_byte_raw(mut crc: u32, byte: u8) -> u32 {
    crc ^= byte as u32;
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb8_8320;
        }
    }
    crc
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    for &byte in data {
        crc = crc32_byte_raw(crc, byte);
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_update_word(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65535;
    *b = (*b + *a) % 65535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        fletcher32_update_word(
            &mut a,
            &mut b,
            ((pair[0] as u16) << 8) | pair[1] as u16,
        );
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_update_word(&mut a, &mut b, (last as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                fletcher32_update_word(
                    &mut a,
                    &mut b,
                    ((high as u16) << 8) | byte as u16,
                );
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_update_word(&mut a, &mut b, (high as u16) << 8);
    }
    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc: u16 = if variant == Crc16::Xmodem {
                0
            } else {
                0xffff
            };
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000 != 0;
                    crc = crc.wrapping_shl(1);
                    if top {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc = 0xffffu16;
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1 != 0;
                    crc >>= 1;
                    if low {
                        crc ^= 0x8408;
                    }
                }
            }
            crc ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut stream = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_pending: Option<u8> = None;
    let mut crc16_state: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            stream.update(std::slice::from_ref(&byte));

            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_pending.take() {
                fletcher32_update_word(
                    &mut f32_a,
                    &mut f32_b,
                    ((high as u16) << 8) | byte as u16,
                );
            } else {
                f32_pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_state ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = crc16_state & 0x8000 != 0;
                        crc16_state = crc16_state.wrapping_shl(1);
                        if top {
                            crc16_state ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_state ^= byte as u16;
                    for _ in 0..8 {
                        let low = crc16_state & 1 != 0;
                        crc16_state >>= 1;
                        if low {
                            crc16_state ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(high) = f32_pending {
        fletcher32_update_word(&mut f32_a, &mut f32_b, (high as u16) << 8);
    }

    let crc16 = if variant == Crc16::X25 {
        crc16_state ^ 0xffff
    } else {
        crc16_state
    };

    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16,
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn crc32_zero_byte_power(mut value: u32, mut byte_count: u64) -> u32 {
    if byte_count == 0 {
        return value;
    }

    let mut operator = [0u32; 32];
    for bit in 0..32 {
        operator[bit] = crc32_byte_raw(1u32 << bit, 0);
    }

    while byte_count != 0 {
        if byte_count & 1 != 0 {
            value = gf2_matrix_times(&operator, value);
        }
        byte_count >>= 1;
        if byte_count != 0 {
            let mut squared = [0u32; 32];
            gf2_matrix_square(&mut squared, &operator);
            operator = squared;
        }
    }
    value
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    crc32_zero_byte_power(left, right_len) ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total_len = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total_len = total_len
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }

    Ok(Part {
        crc32: combined,
        len: total_len,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);
        len += selected.len() as u64;
    }
    let hashes = stream.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> u32 {
    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = (value << 4) | hex_value(byte).unwrap();
    }
    value
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> u64 {
    let mut value = 0u64;
    for &byte in &bytes[start..end] {
        value = (value << 4) | hex_value(byte).unwrap() as u64;
    }
    value
}

fn selected_len(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |total, span| {
        total + (span.end - span.start) as u64
    })
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for offset in (4..20).chain(25..33).chain(40..48) {
        if hex_value(bytes[offset]).is_none() {
            return Err(VerifyError::Hex { offset });
        }
    }

    let expected_len = parse_hex_u64(bytes, 4, 20);
    let expected_crc = parse_hex_u32(bytes, 25, 33);
    let expected_adler = parse_hex_u32(bytes, 40, 48);
    if expected_len != selected_len(spans) {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
