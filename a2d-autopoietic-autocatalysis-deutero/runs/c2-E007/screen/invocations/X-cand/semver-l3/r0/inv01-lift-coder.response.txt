use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, at: usize) -> Option<u8> {
    s.as_bytes().get(at).copied()
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return match byte(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let ident_start = p;
        while p < end && s.as_bytes()[p] != b'.' {
            let b = s.as_bytes()[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if ident_start == p {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..p];
        if pre && ident.len() > 1 && ident.as_bytes().iter().all(|b| is_digit(*b))
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        out.push(ident.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    for _ in 0..2 {
        match byte(s, p) {
            Some(b'.') => p += 1,
            None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: p }),
            _ => return Err(VersionError::InvalidChar { at: p }),
        }
        let n = parse_core(s, &mut p)?;
        if _ == 0 {}
        // Keep the two fields explicit below.
        if p > s.len() {
            unreachable!();
        }
        if _ == 0 {
            let _ = n;
        }
    }
    // Reparse core values in a straightforward pass (the first pass above validates offsets).
    let mut q = 0;
    let major = parse_core(s, &mut q)?;
    if byte(s, q) != Some(b'.') {
        unreachable!();
    }
    q += 1;
    let minor = parse_core(s, &mut q)?;
    if byte(s, q) != Some(b'.') {
        unreachable!();
    }
    q += 1;
    let patch = parse_core(s, &mut q)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    match byte(s, q) {
        None => {}
        Some(b'-') => {
            q += 1;
            let plus = s[q..].find('+').map(|i| q + i).unwrap_or(s.len());
            pre = parse_identifiers(s, q, plus, true)?;
            q = plus;
            if byte(s, q) == Some(b'+') {
                q += 1;
                build = parse_identifiers(s, q, s.len(), false)?;
                q = s.len();
            }
        }
        Some(b'+') => {
            q += 1;
            build = parse_identifiers(s, q, s.len(), false)?;
            q = s.len();
        }
        _ => return Err(VersionError::InvalidChar { at: q }),
    }
    if q != s.len() {
        return Err(VersionError::InvalidChar { at: q });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_identifier(x);
        let yn = numeric_identifier(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Bound {
    version: Version,
    inclusive: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    lower: Option<Bound>,
    upper: Option<Bound>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let mut p = start;
    while p < end {
        while p < end && whitespace(s.as_bytes()[p]) {
            p += 1;
        }
        if p == end {
            break;
        }
        let st = p;
        while p < end && !whitespace(s.as_bytes()[p]) {
            p += 1;
        }
        out.push((st, &s[st..p]));
    }
    out
}

fn full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn version(major: u64, minor: u64, patch: u64, pre: Vec<String>, build: Vec<String>) -> Version {
    Version { major, minor, patch, pre, build }
}

fn stable_tuple(v: &Version) -> Version {
    version(v.major, v.minor, v.patch, Vec::new(), Vec::new())
}

fn next_tuple(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (mut ma, mut mi, mut pa) = (v.major, v.minor, v.patch);
    let n = match field {
        0 => &mut ma,
        1 => &mut mi,
        _ => &mut pa,
    };
    *n = n.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field {
        0 => { mi = 0; pa = 0; }
        1 => pa = 0,
        _ => {}
    }
    Ok(version(ma, mi, pa, vec!["0".into()], Vec::new()))
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields { nums: Vec<u64>, k: usize },
}

fn parse_partial(raw: &str, at: usize) -> Result<Partial, RangeError> {
    if raw.contains('-') || raw.contains('+') {
        return full_operand(raw, at).map(Partial::Full);
    }
    let fields: Vec<&str> = raw.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if fields.len() >= 3 && !has_wild {
        return full_operand(raw, at).map(Partial::Full);
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.as_bytes().iter().all(|b| is_digit(*b))
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial::Fields { k: nums.len(), nums })
}

fn padded(p: &Partial) -> Version {
    match p {
        Partial::Full(v) => v.clone(),
        Partial::Fields { nums, .. } => version(
            *nums.get(0).unwrap_or(&0),
            *nums.get(1).unwrap_or(&0),
            *nums.get(2).unwrap_or(&0),
            Vec::new(),
            Vec::new(),
        ),
    }
}

fn arm_add(arm: &mut Arm, op: &str, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push((v.major, v.minor, v.patch));
    }
    let b = Bound { version: v, inclusive: matches!(op, ">=" | "<=" | "=") };
    if op == "=" {
        tighten_lower(arm, b.clone());
        tighten_upper(arm, b);
    } else if op.starts_with('>') {
        tighten_lower(arm, b);
    } else {
        tighten_upper(arm, b);
    }
}

fn tighten_lower(arm: &mut Arm, b: Bound) {
    let replace = arm.lower.as_ref().map_or(true, |old| {
        compare_versions(&b.version, &old.version) == Ordering::Greater
            || (compare_versions(&b.version, &old.version) == Ordering::Equal && !b.inclusive && old.inclusive)
    });
    if replace { arm.lower = Some(b); }
}

fn tighten_upper(arm: &mut Arm, b: Bound) {
    let replace = arm.upper.as_ref().map_or(true, |old| {
        compare_versions(&b.version, &old.version) == Ordering::Less
            || (compare_versions(&b.version, &old.version) == Ordering::Equal && !b.inclusive && old.inclusive)
    });
    if replace { arm.upper = Some(b); }
}

fn add_partial(arm: &mut Arm, raw: &str, at: usize, op: &str) -> Result<(), RangeError> {
    let p = parse_partial(raw, at)?;
    match p {
        Partial::Full(v) => {
            if op == "bare" {
                arm_add(arm, "=", v, true);
            } else {
                arm_add(arm, op, v, true);
            }
        }
        Partial::Fields { nums, k } => {
            if op == "bare" {
                if k == 0 { return Ok(()); }
                let lo = padded(&Partial::Fields { nums: nums.clone(), k });
                arm_add(arm, ">=", lo.clone(), false);
                if k < 3 {
                    let hi = next_tuple(&lo, k - 1, at)?;
                    arm_add(arm, "<", hi, false);
                }
            } else if op == "^" || op == "~" {
                if k == 0 { return Err(RangeError::Syntax { at }); }
                let lo = padded(&Partial::Fields { nums: nums.clone(), k });
                arm_add(arm, ">=", lo.clone(), false);
                let field = if op == "^" {
                    if lo.major > 0 || k == 1 { 0 }
                    else if lo.minor > 0 || k == 2 { 1 }
                    else { 2 }
                } else if k == 1 { 0 } else { 1 };
                arm_add(arm, "<", next_tuple(&lo, field, at)?, false);
            }
            _ => unreachable!(),
        }
    }
    Ok(())
}

fn parse_arm(s: &str, toks: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { lower: None, upper: None, anchors: Vec::new() };
    if toks.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let minuses: Vec<usize> = toks.iter().filter(|(_, t)| *t == &"-").map(|(p, _)| *p).collect();
    if !minuses.is_empty() {
        if toks.len() != 3 || toks[1].1 != "-" {
            return Err(RangeError::Syntax { at: minuses[0] });
        }
        let (la, lt) = toks[0];
        let (ua, ut) = toks[2];
        let lp = parse_partial(lt, la)?;
        let up = parse_partial(ut, ua)?;
        let (lk, lv) = match lp {
            Partial::Full(v) => (3, v),
            Partial::Fields { nums, k } => (k, padded(&Partial::Fields { nums, k })),
        };
        let (uk, uv, uf) = match up {
            Partial::Full(v) => (3, v, true),
            Partial::Fields { nums, k } => (k, padded(&Partial::Fields { nums, k }), false),
        };
        if lk == 0 { return Err(RangeError::Syntax { at: la }); }
        if uk == 0 { return Err(RangeError::Syntax { at: ua }); }
        arm_add(&mut arm, ">=", lv, true);
        if uf {
            arm_add(&mut arm, "<=", uv, true);
        } else if uk < 3 {
            let upper = next_tuple(&uv, uk - 1, ua)?;
            arm_add(&mut arm, "<", upper, false);
        } else {
            arm_add(&mut arm, "<=", uv, false);
        }
        return Ok(arm);
    }
    for &(at, tok) in toks {
        let (op, operand, operand_at) = if tok.starts_with("^") || tok.starts_with("~") {
            (&tok[..1], &tok[1..], at + 1)
        } else if tok.starts_with(">=") || tok.starts_with("<=") {
            (&tok[..2], &tok[2..], at + 2)
        } else if tok.starts_with(">") || tok.starts_with("<") || tok.starts_with("=") {
            (&tok[..1], &tok[1..], at + 1)
        } else {
            ("bare", tok, at)
        };
        if operand.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if op == "bare" || op == "^" || op == "~" {
            if !operand.as_bytes()[0].is_ascii_digit() && !matches!(operand.as_bytes()[0], b'*' | b'x' | b'X') {
                return Err(RangeError::Syntax { at });
            }
            add_partial(&mut arm, operand, operand_at, op)?;
        } else {
            if !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = full_operand(operand, operand_at)?;
            arm_add(&mut arm, op, v, true);
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut segments = Vec::new();
    let mut p = 0;
    let mut arm_start = 0;
    while p < bytes.len() {
        if p + 1 < bytes.len() && bytes[p] == b'|' && bytes[p + 1] == b'|' {
            let before = tokens(s, arm_start, p);
            if before.is_empty() {
                return Err(RangeError::Empty { at: p });
            }
            segments.push((arm_start, p));
            p += 2;
            while p < bytes.len() && whitespace(bytes[p]) { p += 1; }
            arm_start = p;
        } else {
            p += 1;
        }
    }
    let tail = tokens(s, arm_start, s.len());
    if tail.is_empty() {
        if segments.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    segments.push((arm_start, s.len()));
    let mut arms = Vec::new();
    for (a, b) in segments {
        let ts = tokens(s, a, b);
        arms.push(parse_arm(s, &ts)?);
    }
    Ok(Range { arms })
}

fn admitted(v: &Version, arm: &Arm) -> bool {
    v.pre.is_empty() || arm.anchors.contains(&(v.major, v.minor, v.patch))
}

fn in_arm(v: &Version, arm: &Arm, include_pre: bool) -> bool {
    if !include_pre && !admitted(v, arm) { return false; }
    if let Some(b) = &arm.lower {
        let c = compare_versions(v, &b.version);
        if c == Ordering::Less || (c == Ordering::Equal && !b.inclusive) { return false; }
    }
    if let Some(b) = &arm.upper {
        let c = compare_versions(v, &b.version);
        if c == Ordering::Greater || (c == Ordering::Equal && !b.inclusive) { return false; }
    }
    true
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| in_arm(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (i, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index: i, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let c = compare_versions(v, &parsed[j]);
            if (max && c == Ordering::Greater) || (!max && c == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn arm_has_witness(arm: &Arm, include_pre: bool) -> bool {
    if let (Some(l), Some(u)) = (&arm.lower, &arm.upper) {
        let c = compare_versions(&l.version, &u.version);
        if c == Ordering::Greater || (c == Ordering::Equal && !(l.inclusive && u.inclusive)) {
            return false;
        }
    }
    if include_pre {
        return true;
    }
    // Stable candidates exist at every finite lower bound unless the interval excludes
    // that release and has no room for the next core tuple.
    if let Some(l) = &arm.lower {
        if l.version.pre.is_empty() {
            if arm.upper.as_ref().map_or(true, |u| {
                let c = compare_versions(&l.version, &u.version);
                c == Ordering::Less || (c == Ordering::Equal && l.inclusive && u.inclusive)
            }) && l.inclusive {
                return true;
            }
        }
    }
    // Anchored prereleases are direct witnesses when they satisfy the interval.
    for &(ma, mi, pa) in &arm.anchors {
        let v = version(ma, mi, pa, vec!["0".into()], Vec::new());
        if in_arm(&v, arm, false) { return true; }
    }
    // Check a stable tuple from the finite endpoint and its immediate successor.
    let candidates = [arm.lower.as_ref(), arm.upper.as_ref()];
    for b in candidates.into_iter().flatten() {
        let v = stable_tuple(&b.version);
        if in_arm(&v, arm, false) { return true; }
        let mut next = v.clone();
        if next.patch < u64::MAX { next.patch += 1; }
        else if next.minor < u64::MAX { next.minor += 1; next.patch = 0; }
        else if next.major < u64::MAX { next.major += 1; next.minor = 0; next.patch = 0; }
        else { continue; }
        if in_arm(&next, arm, false) { return true; }
    }
    arm.lower.is_none() && arm.upper.is_none()
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut both = aa.clone();
            if let Some(l) = &bb.lower { tighten_lower(&mut both, l.clone()); }
            if let Some(u) = &bb.upper { tighten_upper(&mut both, u.clone()); }
            both.anchors = aa.anchors.iter().copied()
                .filter(|x| bb.anchors.contains(x)).collect();
            if arm_has_witness(&both, include_prerelease) { return true; }
        }
    }
    false
}

fn version_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn bound_text(b: &Option<Bound>, lower: bool) -> String {
    match b {
        None => "*".into(),
        Some(b) => format!("{}{}", if lower {
            if b.inclusive { ">=" } else { ">" }
        } else if b.inclusive { "<=" } else { "<" }, version_text(&b.version)),
    }
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        if !arm_has_witness(arm, false) { continue; }
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        let mut clean = arm.clone();
        for b in clean.lower.iter_mut().chain(clean.upper.iter_mut()) {
            b.version.build.clear();
        }
        out.push(format!(
            "[{};{};pre={}]",
            bound_text(&clean.lower, true),
            bound_text(&clean.upper, false),
            anchor_text
        ));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}
