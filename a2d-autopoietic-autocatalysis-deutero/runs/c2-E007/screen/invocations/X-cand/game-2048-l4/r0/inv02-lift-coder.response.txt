#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
    ScoreOverflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };
    let line_len = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };

    for line in 0..line_count {
        let indices: Vec<usize> = (0..line_len)
            .map(|offset| match direction {
                Direction::Left => line * board.width + offset,
                Direction::Right => line * board.width + (board.width - 1 - offset),
                Direction::Up => offset * board.width + line,
                Direction::Down => (board.height - 1 - offset) * board.width + line,
            })
            .collect();

        let occupied: Vec<usize> = indices
            .iter()
            .copied()
            .filter(|&index| board.cells[index] != 0)
            .collect();

        let mut outputs: Vec<(u32, Option<[usize; 2]>)> = Vec::new();
        let mut i = 0;
        while i < occupied.len() {
            let first = occupied[i];
            let value = board.cells[first];
            if i + 1 < occupied.len() && board.cells[occupied[i + 1]] == value {
                let second = occupied[i + 1];
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first, second });
                }
                let merged = value * 2;
                score = score.checked_add(merged as u64).ok_or(Error::ScoreOverflow)?;
                outputs.push((merged, Some([first, second])));
                i += 2;
            } else {
                outputs.push((value, None));
                i += 1;
            }
        }

        for (slot, (value, source)) in outputs.into_iter().enumerate() {
            let to = indices[slot];
            cells[to] = value;
            if let Some(from) = source {
                merges.push(Merge { from, to, value });
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(moved) if moved.changed => result.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(result)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Spawn {
    pub index: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

fn next_rng(state: u64) -> u64 {
    state
        .wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empty: Vec<usize> = board
        .cells
        .iter()
        .enumerate()
        .filter_map(|(index, &value)| (value == 0).then_some(index))
        .collect();

    if empty.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let a = next_rng(rng);
    let index = empty[(a % empty.len() as u64) as usize];
    let b = next_rng(a);
    let value = if b % 10 == 0 { 4 } else { 2 };

    let mut result_board = board.clone();
    result_board.cells[index] = value;

    Ok(SpawnResult {
        board: result_board,
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {
    pub board: Board,
    pub rng: u64,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
    pub spawn: Option<Spawn>,
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub board: Board,
    pub score: u64,
    pub rng: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Game {
    current: State,
    previous: Option<State>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Command {
    Move(Direction),
    Undo,
    Check,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Event {
    Moved(Turn),
    Undone(bool),
    Checked(bool),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Replay {
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReplayError {
    pub index: Option<usize>,
    pub error: Error,
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

pub fn new_game(initial: State) -> Result<Game, Error> {
    validate(&initial.board)?;
    Ok(Game {
        current: initial,
        previous: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(&game.current.board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let cumulative_score = game
        .current
        .score
        .checked_add(moved.score)
        .ok_or(Error::ScoreOverflow)?;

    let spawned = spawn_tile(&moved.board, game.current.rng)?;
    let old_state = game.current.clone();
    game.current = State {
        board: spawned.board.clone(),
        score: cumulative_score,
        rng: spawned.rng,
    };
    game.previous = Some(old_state);

    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

pub fn undo(game: &mut Game) -> bool {
    if let Some(previous) = game.previous.take() {
        game.current = previous;
        true
    } else {
        false
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.previous.is_some()
}

pub fn game_over(game: &Game) -> bool {
    legal_moves(&game.current.board)
        .map(|moves| moves.is_empty())
        .unwrap_or(true)
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };

    let mut events = Vec::new();
    for (index, command) in commands.iter().enumerate() {
        match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => events.push(Event::Moved(turn)),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => events.push(Event::Undone(undo(&mut game))),
            Command::Check => events.push(Event::Checked(game_over(&game))),
        }
    }

    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    let expected = board.width * board.height;
    if board.cells.len() != expected {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}
