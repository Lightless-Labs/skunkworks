```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u64,
}

pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    next_age: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let canonical = Self::canonicalize_key(key)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        let age = self.next_age;
        self.next_age = self.next_age.wrapping_add(1);
        self.entries.insert(
            canonical.clone(),
            Entry {
                item: Item {
                    key: canonical,
                    priority,
                    value: value.to_owned(),
                },
                age,
            },
        );
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_key()
            .and_then(|key| self.entries.get(key))
            .map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?.to_owned();
        self.entries.remove(&key).map(|entry| entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = Self::canonicalize_key(key)?;
        Ok(self.entries.get(&canonical).map(|entry| entry.item.clone()))
    }

    fn best_key(&self) -> Option<&str> {
        self.entries
            .values()
            .min_by(|a, b| Self::compare_entries(a, b))
            .map(|entry| entry.item.key.as_str())
    }

    fn compare_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn canonicalize_key(key: &str) -> Result<String, Error> {
        let bytes = key.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }

        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }

        Ok(key[start..end].to_ascii_lowercase())
    }
}

impl Default for IndexedHeap {
    fn default() -> Self {
        Self::new()
    }
}
```
