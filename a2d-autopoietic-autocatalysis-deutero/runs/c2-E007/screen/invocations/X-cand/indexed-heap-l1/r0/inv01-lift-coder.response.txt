#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

pub struct IndexedHeap {
    items: Vec<HeapEntry>,
    next_age: u128,
}

struct HeapEntry {
    item: Item,
    age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            items: Vec::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize_key(key)?;

        if self.items.iter().any(|entry| entry.item.key == key) {
            return Err(Error::Duplicate { key });
        }

        let entry = HeapEntry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };
        self.next_age = self.next_age.wrapping_add(1);

        self.items.push(entry);
        let mut index = self.items.len() - 1;
        while index > 0 {
            let parent = (index - 1) / 2;
            if !is_better(&self.items[index], &self.items[parent]) {
                break;
            }
            self.items.swap(index, parent);
            index = parent;
        }

        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.items.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.items.is_empty() {
            return None;
        }

        let last = self.items.len() - 1;
        self.items.swap(0, last);
        let result = self.items.pop().map(|entry| entry.item);

        if !self.items.is_empty() {
            let mut index = 0;
            loop {
                let left = index * 2 + 1;
                let right = left + 1;
                let mut best = index;

                if left < self.items.len() && is_better(&self.items[left], &self.items[best]) {
                    best = left;
                }
                if right < self.items.len() && is_better(&self.items[right], &self.items[best]) {
                    best = right;
                }
                if best == index {
                    break;
                }

                self.items.swap(index, best);
                index = best;
            }
        }

        result
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize_key(key)?;
        Ok(self
            .items
            .iter()
            .find(|entry| entry.item.key == key)
            .map(|entry| entry.item.clone()))
    }
}

fn canonicalize_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    for (relative, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if relative == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + relative });
        }
    }

    let canonical: Vec<u8> = bytes[start..end]
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect();

    Ok(String::from_utf8(canonical).expect("validated ASCII key"))
}

fn is_better(left: &HeapEntry, right: &HeapEntry) -> bool {
    left.item
        .priority
        .cmp(&right.item.priority)
        .then_with(|| left.item.value.len().cmp(&right.item.value.len()))
        .then_with(|| left.age.cmp(&right.age))
        .is_lt()
}
