use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u64,
}

pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    next_age: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let canonical = Self::canonicalize_key(key)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        let age = self.next_age;
        self.next_age = self.next_age.wrapping_add(1);
        self.entries.insert(
            canonical.clone(),
            Entry {
                item: Item {
                    key: canonical,
                    priority,
                    value: value.to_owned(),
                },
                age,
            },
        );
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_key()
            .and_then(|key| self.entries.get(key))
            .map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?.to_owned();
        self.entries.remove(&key).map(|entry| entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = Self::canonicalize_key(key)?;
        Ok(self.entries.get(&canonical).map(|entry| entry.item.clone()))
    }

    fn best_key(&self) -> Option<&str> {
        self.entries
            .values()
            .min_by(|a, b| Self::compare_entries(a, b))
            .map(|entry| entry.item.key.as_str())
    }

    fn compare_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn canonicalize_key(key: &str) -> Result<String, Error> {
        let bytes = key.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }

        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }

        Ok(key[start..end].to_ascii_lowercase())
    }
}

impl Default for IndexedHeap {
    fn default() -> Self {

    }

pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

pub struct IndexedHeap {
    items: Vec<HeapEntry>,
    next_age: u128,
}

struct HeapEntry {
    item: Item,
    age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            items: Vec::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize_key(key)?;

        if self.items.iter().any(|entry| entry.item.key == key) {
            return Err(Error::Duplicate { key });
        }

        let entry = HeapEntry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };
        self.next_age = self.next_age.wrapping_add(1);

        self.items.push(entry);
        let mut index = self.items.len() - 1;
        while index > 0 {
            let parent = (index - 1) / 2;
            if !is_better(&self.items[index], &self.items[parent]) {
                break;
            }
            self.items.swap(index, parent);
            index = parent;
        }

        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.items.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.items.is_empty() {
            return None;
        }

        let last = self.items.len() - 1;
        self.items.swap(0, last);
        let result = self.items.pop().map(|entry| entry.item);

        if !self.items.is_empty() {
            let mut index = 0;
            loop {
                let left = index * 2 + 1;
                let right = left + 1;
                let mut best = index;

                if left < self.items.len() && is_better(&self.items[left], &self.items[best]) {
                    best = left;
                }
                if right < self.items.len() && is_better(&self.items[right], &self.items[best]) {
                    best = right;
                }
                if best == index {
                    break;
                }

                self.items.swap(index, best);
                index = best;
            }
        }

        result
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize_key(key)?;
        Ok(self
            .items
            .iter()
            .find(|entry| entry.item.key == key)
            .map(|entry| entry.item.clone()))
    }
}

fn canonicalize_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    for (relative, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if relative == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + relative });
        }
    }

    let canonical: Vec<u8> = bytes[start..end]
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect();

    Ok(String::from_utf8(canonical).expect("validated ASCII key"))
}

        .then_with(|| left.item.value.len().cmp(&right.item.value.len()))

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

struct Entry {
    item: Item,
    age: u64,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let canonical = canonicalize_key(key)?;
        if self.indices.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        let age = self.next_age;
        self.next_age = self.next_age.wrapping_add(1);

        let index = self.entries.len();
        self.entries.push(Entry {
            item: Item {
                key: canonical.clone(),
                priority,
                value: value.to_owned(),
            },
            age,
        });
        self.indices.insert(canonical, index);
        self.sift_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            return None;
        }

        let last = self.entries.len() - 1;
        self.swap_entries(0, last);
        let removed = self.entries.pop().expect("heap was nonempty");
        self.indices.remove(&removed.item.key);

        if !self.entries.is_empty() {
            self.sift_down(0);
        }

        Some(removed.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = canonicalize_key(key)?;
        Ok(self
            .indices
            .get(&canonical)
            .map(|&index| self.entries[index].item.clone()))
    }

    fn less(&self, a: usize, b: usize) -> bool {
        let left = &self.entries[a];
        let right = &self.entries[b];

        left.item
            .priority
            .cmp(&right.item.priority)
            .then_with(|| left.item.value.len().cmp(&right.item.value.len()))
            .then_with(|| left.age.cmp(&right.age))
            .is_lt()
    }

    fn swap_entries(&mut self, a: usize, b: usize) {
        if a == b {
            return;
        }

        self.entries.swap(a, b);
        let key_a = self.entries[a].item.key.clone();
        let key_b = self.entries[b].item.key.clone();
        self.indices.insert(key_a, a);
        self.indices.insert(key_b, b);
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !self.less(index, parent) {
                break;
            }
            self.swap_entries(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        loop {
            let left = index * 2 + 1;
            if left >= self.entries.len() {
                break;
            }

            let right = left + 1;
            let best_child = if right < self.entries.len() && self.less(right, left) {
                right
            } else {
                left
            };

            if !self.less(best_child, index) {
                break;
            }

            self.swap_entries(index, best_child);
            index = best_child;
        }
    }
}
