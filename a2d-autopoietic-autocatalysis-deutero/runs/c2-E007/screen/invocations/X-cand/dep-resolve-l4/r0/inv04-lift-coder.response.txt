        site: String,

    },

    },

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut v = cs.to_vec();
    v.sort_by_key(|c| constraint_key(*c));
    v.dedup();
    v
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(first: String, evidence: &[Evidence]) -> String {
    let mut out = first;
    for e in evidence_sorted(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn satisfies(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches_constraint(v, *c))
}

fn target_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn sorted_features(features: &[String]) -> Vec<String> {
    features.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect()
}

fn token_features(features: &[String]) -> String {
    sorted_features(features).join(",")
}

fn root_token(i: usize, root: &Requirement) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        root.package,
        constraint_text(root.constraint),
        token_features(&root.features)
    )
}

fn dependency_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        token_features(&dep.features),
        gate,
        platform
    )
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    requested: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    active_deps: BTreeMap<String, BTreeSet<usize>>,
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn add_path(paths: &mut BTreeMap<String, Vec<String>>, package: &str, path: Vec<String>) -> bool {
    match paths.get(package) {
        None => {
            paths.insert(package.to_string(), path);
            true
        }
        Some(old) if path_better(&path, old) => {
            paths.insert(package.to_string(), path);
            true
        }
        _ => false,
    }
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut active = Active::default();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        active.origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: path.clone(),
        });
        active
            .requested
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        add_path(&mut active.paths, &root.package, path);
    }

    let mut processed_deps: BTreeSet<(String, usize)> = BTreeSet::new();
    let mut processed_features: BTreeSet<(String, String)> = BTreeSet::new();

    loop {
        let mut changed = false;

        let names: Vec<String> = active.requested.keys().cloned().collect();
        for package_name in names {
            let Some(version) = selected.get(&package_name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&package_name).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested: Vec<String> = active
                .requested
                .get(&package_name)
                .into_iter()
                .flat_map(|s| s.iter().cloned())
                .collect();

            for feature_name in requested {
                if !processed_features.insert((package_name.clone(), feature_name.clone())) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if active
                            .requested
                            .entry(package_name.clone())
                            .or_default()
                            .insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
            }

            let source_features = active
                .requested
                .get(&package_name)
                .cloned()
                .unwrap_or_default();

            for (index, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !source_features.contains(gate))
                {
                    continue;
                }
                if !processed_deps.insert((package_name.clone(), index)) {
                    continue;
                }
                active
                    .active_deps
                    .entry(package_name.clone())
                    .or_default()
                    .insert(index);

                let token = dependency_token(&package_name, version, index, dep);
                let mut path = active.paths.get(&package_name).cloned().unwrap_or_default();
                path.push(token);
                active.origins.entry(dep.package.clone()).or_default().push(Origin {
                    constraint: dep.constraint,
                    path: path.clone(),
                });

                let requested_set = active.requested.entry(dep.package.clone()).or_default();
                let old_len = requested_set.len();
                requested_set.extend(dep.features.iter().cloned());
                if requested_set.len() != old_len {
                    changed = true;
                }
                if add_path(&mut active.paths, &dep.package, path) {
                    changed = true;
                }
            }
        }

        if !changed {
            break;
        }
    }

    // Recompute canonical paths by relaxing active dependency edges.
    let mut paths = BTreeMap::new();
    for (i, root) in roots.iter().enumerate() {
        add_path(&mut paths, &root.package, vec![root_token(i, root)]);
    }
    loop {
        let mut changed = false;
        for (source, indices) in &active.active_deps {
            let Some(source_path) = paths.get(source).cloned() else {
                continue;
            };
            let Some(version) = selected.get(source).copied() else {
                continue;
            };
            let Some(package) = catalog.get(source).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            for index in indices {
                let dep = &release.deps[*index];
                let mut path = source_path.clone();
                path.push(dependency_token(source, version, *index, dep));
                if add_path(&mut paths, &dep.package, path) {
                    changed = true;
                }
            }
        }
        if !changed {
            break;
        }
    }
    active.paths = paths;

    // Rebuild evidence using each source package's canonical path.
    let mut origins: BTreeMap<String, Vec<Origin>> = BTreeMap::new();
    for (i, root) in roots.iter().enumerate() {
        origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: vec![root_token(i, root)],
        });
    }
    for (source, indices) in &active.active_deps {
        let Some(source_path) = active.paths.get(source) else {
            continue;
        };
        let Some(version) = selected.get(source).copied() else {
            continue;
        };
        let Some(package) = catalog.get(source).copied() else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        for index in indices {
            let dep = &release.deps[*index];
            let mut path = source_path.clone();
            path.push(dependency_token(source, version, *index, dep));
            origins.entry(dep.package.clone()).or_default().push(Origin {
                constraint: dep.constraint,
                path,
            });
        }
    }
    active.origins = origins;
    active
}

fn active_constraints(active: &Active, package: &str) -> Vec<Constraint> {
    let values = active
        .origins
        .get(package)
        .into_iter()
        .flat_map(|v| v.iter().map(|o| o.constraint))
        .collect::<Vec<_>>();
    canonical_constraints(&values)
}

fn active_evidence(active: &Active, package: &str) -> Vec<Evidence> {
    let values = active
        .origins
        .get(package)
        .into_iter()
        .flat_map(|v| {
            v.iter().map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
        })
        .collect::<Vec<_>>();
    evidence_sorted(&values)
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn lock_cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count()
    }

    fn better_tie(&self, a: &BTreeMap<String, Version>, b: &BTreeMap<String, Version>) -> bool {
        for name in self.catalog.keys() {
            match (a.get(name), b.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(x), Some(y)) if x != y => return x > y,
                _ => {}
            }
        }
        false
    }

    fn inspect(&mut self, selected: &mut BTreeMap<String, Version>) -> bool {
        let active = compute_active(self.catalog, self.roots, selected, self.target);
        let active_names: BTreeSet<String> = active.origins.keys().cloned().collect();

        for name in &active_names {
            if !self.catalog.contains_key(name) {
                self.record_error(ResolveError::Missing {
                    package: name.clone(),
                    evidence: active_evidence(&active, name),
                });
                return false;
            }
        }

        for name in selected.keys() {
            if !active_names.contains(name) {
                continue;
            }
            let version = selected[name];
            let package = self.catalog[name];
            let release = package.releases.iter().find(|r| r.version == version);
            let requested = active.requested.get(name).cloned().unwrap_or_default();
            if let Some(release) = release {
                for feature in requested {
                    if !release.features.iter().any(|f| f.name == feature) {
                        self.record_error(ResolveError::MissingFeature {
                            package: name.clone(),
                            selected: *version,
                            feature,
                            path: active.paths.get(name).cloned().unwrap_or_default(),
                        });
                        return false;
                    }
                }
            }
        }

        for name in selected.keys() {
            if !active_names.contains(name) {
                continue;
            }
            let version = selected[name];
            let constraints = active_constraints(&active, name);
            if !satisfies(version, &constraints) {
                self.record_error(ResolveError::Conflict {
                    package: name.clone(),
                    selected: version,
                    requirements: constraints,
                    evidence: active_evidence(&active, name),
                });
                return false;
            }
        }

        for name in &active_names {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.catalog[name];
            let constraints = active_constraints(&active, name);
            if !package
                .releases
                .iter()
                .any(|r| satisfies(r.version, &constraints))
            {
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints,
                    evidence: active_evidence(&active, name),
                });
                return false;
            }
        }

        let next = active_names
            .iter()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(name) = next else {
            let cost = self.lock_cost(selected);
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_sel)) => {
                    cost < *best_cost || (cost == *best_cost && self.better_tie(selected, best_sel))
                }
            };
            if replace {
                self.best = Some((cost, selected.clone()));
            }
            return true;
        };

        let package = self.catalog[&name];
        let constraints = active_constraints(&active, &name);
        let mut candidates = package
            .releases
            .iter()
            .filter(|r| satisfies(r.version, &constraints))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return false;
            }
            self.attempts += 1;
            selected.insert(name.clone(), candidate);
            self.inspect(selected);
            selected.remove(&name);

            if self.attempts == self.max_attempts
                && self.best.is_none()
                && self.first_error.is_none()
            {
                // Let the parent determine whether more work is required.
            }
        }
        true
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", site),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", site, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = BTreeMap::new();
    search.inspect(&mut selected);

    if search.attempts >= max_attempts && search.best.is_none() && search.first_error.is_none() {
        return Err(ResolveError::Limit { attempts: max_attempts });
    }

    let Some((_, best)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: vec![],
            evidence: vec![],
        }));
    };

    let final_active = compute_active(&catalog_map, roots, &best, target);
    let selected_vec = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(best.keys().cloned());

    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !satisfies(old, &active_constraints(&final_active, &package)) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
