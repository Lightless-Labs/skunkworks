#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian decimal digits; zero is represented as [0] and is never negative.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut first_separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                first_separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len())
            .find(|&index| bytes[index].is_ascii_digit())
            .expect("nonempty valid magnitude has a digit");

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let excess_index = (start..bytes.len())
                .filter(|&index| bytes[index].is_ascii_digit())
                .nth(excess_number)
                .expect("digit count exceeds limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: excess_index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }

        let is_zero = digits.iter().all(|&digit| digit == 0);
        Ok(Self {
            negative: negative && !is_zero,
            digits,
        })
    }

        let actual_len = self.digits.len().max(1);
        let width = actual_len.max(min_digits);

        for position in (0..width).rev() {
            if group > 0 && position != width - 1 && (position + 1) % group == 0 {
                magnitude.push('_');
            }
            let digit = if position < self.digits.len() {
                self.digits[position]
            } else {
                0
            };
            magnitude.push(char::from(b'0' + digit));
        }

        if self.negative {
            let mut result = String::with_capacity(magnitude.len() + 1);
            result.push('-');
            result.push_str(&magnitude);
            result
        } else {
            magnitude
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = add_magnitude(&self.digits, &rhs.digits);
            Self::canonical(self.negative, digits)
        } else {

                Ordering::Equal => Self::zero(),

        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {

        } else {

                    sub_magnitude(&rhs.digits, &self.digits),

                Ordering::Equal => Self::zero(),
            }
        }
    }

        if is_zero(&self.digits) {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

            return Self::zero();
        }

        let mut result = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &left) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &right) in rhs.digits.iter().enumerate() {
                let index = i + j;
                let product = result[index] as u16 + left as u16 * right as u16 + carry;
                result[index] = (product % 10) as u8;
                carry = product / 10;
            }

            let mut index = i + rhs.digits.len();
            while carry > 0 {
                if index == result.len() {
                    result.push(0);
                }
                let sum = result[index] as u16 + carry;
                result[index] = (sum % 10) as u8;
                carry = sum / 10;
                index += 1;
            }
        }

        Self::canonical(self.negative != rhs.negative, result)
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn canonical(negative: bool, mut digits: Vec<u8>) -> Self {

            digits.pop();

            digits.push(0);
        }
        let zero = is_zero(&digits);
        Self {
            negative: negative && !zero,
            digits,
        }
    }

        Ordering::Equal => {
            for (&a, &b) in left.iter().rev().zip(right.iter().rev()) {
                match a.cmp(&b) {
                    Ordering::Equal => {}
                    ordering => return ordering,
                }
            }
            Ordering::Equal
        }
        ordering => ordering,
    }

fn add_magnitude(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u8;

    for index in 0..left.len().max(right.len()) {
        let sum = left.get(index).copied().unwrap_or(0)
            + right.get(index).copied().unwrap_or(0)
            + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }

    if carry != 0 {
        result.push(carry);
    }
    result
}

// Requires left >= right.
fn sub_magnitude(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len());
    let mut borrow = 0i8;

    for (index, &left_digit) in left.iter().enumerate() {
        let mut difference =
            left_digit as i8 - right.get(index).copied().unwrap_or(0) as i8 - borrow;
        if difference < 0 {
            difference += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(difference as u8);
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                invalid = Some(index);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit_index = None;
        let mut first_digit = 0u8;
        let mut first_bad_separator = None;

        for index in start..bytes.len() {
            match bytes[index] {
                b'_' => {
                    let before_is_digit =
                        index > start && bytes[index - 1].is_ascii_digit();
                    let after_is_digit =
                        index + 1 < bytes.len() && bytes[index + 1].is_ascii_digit();
                    if (!before_is_digit || !after_is_digit)
                        && first_bad_separator.is_none()
                    {
                        first_bad_separator = Some(index);
                    }
                }
                byte => {
                    if first_digit_index.is_none() {
                        first_digit_index = Some(index);
                        first_digit = byte - b'0';
                    }
                    digit_count += 1;
                }
            }
        }

        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        if digit_count > 1 && first_digit == 0 {
            return Err(Error::LeadingZero {
                index: first_digit_index.expect("a nonempty magnitude has a digit"),
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for index in start..bytes.len() {
                if bytes[index].is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        let mut value = Self { negative, digits };
        value.normalize();
        Ok(value)
    }

        if self.negative && !self.is_zero() {
            output.push('-');

    }

    fn subtract_magnitudes(left: &[u8], right: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(left.len());
        let mut borrow = 0i8;

        for index in 0..left.len() {
            let mut difference =
                left[index] as i8 - right.get(index).copied().unwrap_or(0) as i8 - borrow;
            if difference < 0 {
                difference += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(difference as u8);
        }

        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result
    }
}

use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits; zero is represented as [0] with negative=false.
    digits: Vec<u8>,
    negative: bool,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("digits", &self.digits)
            .field("negative", &self.negative)
            .finish()
    }
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        // Check separators only after the full invalid-byte scan succeeds.
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let mut first_digit_index = None;
        let mut digit_count = 0usize;
        let mut first_digit = 0u8;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' {
                if first_digit_index.is_none() {
                    first_digit_index = Some(index);
                    first_digit = byte - b'0';
                }
                digit_count += 1;
            }
        }

        let first_index = first_digit_index.expect("a sign-only input was rejected");
        if digit_count > 1 && first_digit == 0 {
            return Err(Error::LeadingZero { index: first_index });
        }
        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        let is_zero = digits.iter().all(|&digit| digit == 0);
        Ok(Self {
            digits,
            negative: negative && !is_zero,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = min_digits.max(magnitude_len);
        let padding = width - magnitude_len;
        let mut out = String::with_capacity(width + width / 3 + usize::from(self.negative));
        if self.negative {
            out.push('-');
        }

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }
            let from_left = position;
            if from_left < padding {
                out.push('0');
            } else {
                let digit_index = width - 1 - from_left;
                out.push((b'0' + self.digits[digit_index]) as char);
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = add_magnitude(&self.digits, &rhs.digits);
            Self::from_parts(digits, self.negative)
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    Self::from_parts(sub_magnitude(&self.digits, &rhs.digits), self.negative)
                }
                Ordering::Less => {
                    Self::from_parts(sub_magnitude(&rhs.digits, &self.digits), rhs.negative)
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &left) in self.digits.iter().enumerate() {
            for (j, &right) in rhs.digits.iter().enumerate() {
                accum[i + j] += u32::from(left) * u32::from(right);
            }
        }

        for i in 0..accum.len() - 1 {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }

        let mut digits: Vec<u8> = accum.into_iter().map(|digit| digit as u8).collect();
        normalize(&mut digits);
        Self::from_parts(digits, self.negative ^ rhs.negative)
    }

    fn zero() -> Self {
        Self {
            digits: vec![0],
            negative: false,
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.iter().all(|&digit| digit == 0)
    }

    fn from_parts(mut digits: Vec<u8>, negative: bool) -> Self {
        normalize(&mut digits);
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            digits,
            negative: negative && !is_zero,
        }
    }
}

fn normalize(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}

fn cmp_magnitude(left: &[u8], right: &[u8]) -> Ordering {
    let left_len = significant_len(left);
    let right_len = significant_len(right);
    match left_len.cmp(&right_len) {
        Ordering::Equal => {
            for index in (0..left_len).rev() {
                match left[index].cmp(&right[index]) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn significant_len(digits: &[u8]) -> usize {
    let mut len = digits.len();
    while len > 1 && digits[len - 1] == 0 {
        len -= 1;
    }
    len
}

fn add_magnitude(left: &[u8], right: &[u8]) -> Vec<u8> {
    let max_len = left.len().max(right.len());
    let mut result = Vec::with_capacity(max_len + 1);
    let mut carry = 0u8;
    for index in 0..max_len {
        let sum = left.get(index).copied().unwrap_or(0)
            + right.get(index).copied().unwrap_or(0)
            + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry > 0 {
        result.push(carry);
    }
    result
}

// Requires left magnitude >= right magnitude.
fn sub_magnitude(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len());
    let mut borrow = 0i8;
    for index in 0..left.len() {
        let mut digit = i16::from(left[index]) - i16::from(borrow);
        let rhs_digit = i16::from(right.get(index).copied().unwrap_or(0));
        if digit < rhs_digit {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push((digit - rhs_digit) as u8);
    }
    normalize(&mut result);
    result
}
