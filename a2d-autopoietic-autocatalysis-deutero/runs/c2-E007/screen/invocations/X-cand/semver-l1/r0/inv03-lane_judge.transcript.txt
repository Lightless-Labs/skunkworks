#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn is_identifier_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;

    loop {
        let mut pos = id_start;
        while pos < end && bytes[pos] != b'.' {
            if !is_identifier_byte(bytes[pos]) {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        if is_pre {
            let id = &bytes[id_start..pos];
            if id.iter().all(u8::is_ascii_digit)
                && id.len() > 1
                && id[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: id_start });
            }
        }

        result.push(s[id_start..pos].to_owned());

        if pos == end {
            break;
        }
        id_start = pos + 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            let b = bytes.get(pos).copied();
            if is_core_terminator(b) {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        let digits = &bytes[start..pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in digits {
            value = match value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
            {
                Some(n) => n,
                None => return Err(VersionError::Overflow { at: start }),
            };
        }
        core[field] = value;

        if field < 2 {
            if pos >= bytes.len() || bytes[pos] != b'.' {
                let b = bytes.get(pos).copied();
                if matches!(b, None | Some(b'-' | b'+')) {
                    return Err(VersionError::MissingCore { at: pos });
                }
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let build_mark = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pre_start + offset);
        let pre_end = build_mark.unwrap_or(bytes.len());

        pre = parse_identifiers(s, pre_start, pre_end, true)?;

        if let Some(plus_pos) = build_mark {
            let build_start = plus_pos + 1;
            build = parse_identifiers(s, build_start, bytes.len(), false)?;
            pos = bytes.len();
        } else {
            pos = bytes.len();
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core_order = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core_order != Ordering::Equal {
        return core_order;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (a_id, b_id) in a.pre.iter().zip(&b.pre) {
                let order = compare_identifier(a_id, b_id);
                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let (operator, operand_start) = match bytes[token_start] {
            b'=' => (Operator::Equal, token_start + 1),
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                (Operator::LessEqual, token_start + 2)
            }
            b'<' => (Operator::Less, token_start + 1),
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                (Operator::GreaterEqual, token_start + 2)
            }
            b'>' => (Operator::Greater, token_start + 1),
            _ => (Operator::Equal, token_start),
        };

        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        comparators.push(Comparator { operator, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { comparators })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let admitted_pre = v.pre.is_empty()
        || r.comparators.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && (comparator.version.major, comparator.version.minor, comparator.version.patch)
                    == (v.major, v.minor, v.patch)
        });

    if !admitted_pre {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let order = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Equal => order == Ordering::Equal,
            Operator::Less => order == Ordering::Less,
            Operator::LessEqual => order != Ordering::Greater,
            Operator::Greater => order == Ordering::Greater,
            Operator::GreaterEqual => order != Ordering::Less,
        }
    })
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut core = [0u64; 3];

    for (field, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            return Err(if pos == bytes.len()
                || matches!(bytes[pos], b'.' | b'-' | b'+')
            {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut number = 0u64;
        for &digit in &bytes[start..pos] {
            number = number
                .checked_mul(10)
                .and_then(|n| n.checked_add(u64::from(digit - b'0')))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *value = number;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let (ids, next) = parse_identifiers(bytes, pos, true)?;
                pre = ids;
                pos = next;
                if pos < bytes.len() {
                    if bytes[pos] != b'+' {
                        return Err(VersionError::InvalidChar { at: pos });
                    }
                    pos += 1;
                    let (ids, next) = parse_identifiers(bytes, pos, false)?;
                    build = ids;
                    pos = next;
                }
            }
            b'+' => {
                pos += 1;
                let (ids, next) = parse_identifiers(bytes, pos, false)?;
                build = ids;
                pos = next;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    mut pos: usize,
    is_pre: bool,
) -> Result<(Vec<String>, usize), VersionError> {
    let mut ids = Vec::new();

    loop {
        let start = pos;
        while pos < bytes.len()
            && matches!(bytes[pos], b'.' | b'+')
        {
            break;
        }
        while pos < bytes.len()
            && bytes[pos] != b'.'
            && bytes[pos] != b'+'
        {
            let b = bytes[pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        if is_pre
            && pos - start > 1
            && bytes[start..pos].iter().all(u8::is_ascii_digit)
            && bytes[start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }

        ids.push(String::from_utf8(bytes[start..pos].to_vec()).unwrap());

        if pos == bytes.len() || bytes[pos] == b'+' {
            return Ok((ids, pos));
        }

        pos += 1; // dot
        if pos == bytes.len() || bytes[pos] == b'+' {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.iter().zip(b.iter()) {
        let left_numeric = left.bytes().all(|c| c.is_ascii_digit());
        let right_numeric = right.bytes().all(|c| c.is_ascii_digit());

        let order = match (left_numeric, right_numeric) {
            (true, true) => left
                .len()
                .cmp(&right.len())
                .then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if order != Ordering::Equal {
            return order;
        }
    }

    a.len().cmp(&b.len())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {

    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, operator_len) = if token.starts_with("<=") {
            (Op::LessEqual, 2)
        } else if token.starts_with(">=") {
            (Op::GreaterEqual, 2)
        } else if token.starts_with('<') {
            (Op::Less, 1)
        } else if token.starts_with('>') {
            (Op::Greater, 1)
        } else if token.starts_with('=') {
            (Op::Equal, 1)
        } else {
            (Op::Equal, 0)
        };

        let operand_start = token_start + operator_len;
        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });

            at: operand_start,
            error,

        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],

        prerelease_admitted
            && arm.iter().all(|comparator| {
                let order = compare_versions(v, &comparator.version);
                match comparator.op {
                    Op::Equal => order == Ordering::Equal,
                    Op::Less => order == Ordering::Less,
                    Op::LessEqual => order != Ordering::Greater,
                    Op::Greater => order == Ordering::Greater,
                    Op::GreaterEqual => order != Ordering::Less,
                }
            })
    })
}

fn same_core(a: &Version, b: &Version) -> bool {
    a.major == b.major && a.minor == b.minor && a.patch == b.patch

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut core = [0u64; 3];

    for (field, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            return Err(if pos == bytes.len()
                || matches!(bytes[pos], b'.' | b'-' | b'+')
            {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }

        let digits = &bytes[start..pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut n = 0u64;
        for &digit in digits {
            match n
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
            {
                Some(next) => n = next,
                None => return Err(VersionError::Overflow { at: start }),
            }
        }
        *value = n;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());
        parse_identifiers(bytes, pos, pre_end, true, &mut pre)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(bytes, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let mut id_start = start;
    loop {
        let id_end = bytes[id_start..end]
            .iter()
            .position(|&b| b == b'.')
            .map(|i| id_start + i)
            .unwrap_or(end);

        if id_start == id_end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let id = &bytes[id_start..id_end];
        for (i, &b) in id.iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: id_start + i });
            }
        }

        if is_pre
            && id.len() > 1
            && id[0] == b'0'
            && id.iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        out.push(String::from_utf8(id.to_vec()).expect("identifier is ASCII"));

        if id_end == end {
            break;
        }
        id_start = id_end + 1;
    }
    Ok(())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (left, right) in a.iter().zip(b) {
        let left_numeric = left.bytes().all(|b| b.is_ascii_digit());
        let right_numeric = right.bytes().all(|b| b.is_ascii_digit());

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => left
                .len()
                .cmp(&right.len())
                .then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = vec![Vec::new()];
    let mut pos = 0;
    let mut saw_token = false;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        saw_token = true;
        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, op_len) = if token.starts_with(">=") {
            (Op::GreaterEqual, 2)
        } else if token.starts_with("<=") {
            (Op::LessEqual, 2)
        } else if token.starts_with('>') {
            (Op::Greater, 1)
        } else if token.starts_with('<') {
            (Op::Less, 1)
        } else if token.starts_with('=') {
            (Op::Equal, 1)
        } else {
            (Op::Equal, 0)
        };

        let operand_start = token_start + op_len;
        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        arms[0].push(Comparator { op, version });
    }

    if !saw_token {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted = arm.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && core_tuple(&comparator.version) == core_tuple(v)
        });

        if !v.pre.is_empty() && !admitted {
            return false;
        }

        arm.iter().all(|comparator| {
            let ordering = compare_versions(v, &comparator.version);
            match comparator.op {
                Op::Equal => ordering == Ordering::Equal,
                Op::Less => ordering == Ordering::Less,
                Op::LessEqual => ordering != Ordering::Greater,
                Op::Greater => ordering == Ordering::Greater,
                Op::GreaterEqual => ordering != Ordering::Less,
            }
        })
    })
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}
