```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
    pub maintenance: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
    MaintenanceActive,
    MaintenanceRequired,
    AlreadyMode,
    ReconcileMismatch { coin: u16, expected: u16, found: u16 },
    ValueDrift { before: u64, after: u64 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
    Maintenance { enabled: bool },
    Reconciled { before: [u16; 4], after: [u16; 4] },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Audit {
    pub inserted: u64,
    pub refunded: u64,
    pub sales: u64,
    pub change: u64,
    pub adjustment: i64,
    pub cash: u64,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&d| d == coin)
}

fn push_event(events: &mut Vec<Event>, kind: EventKind) {
    events.push(Event {
        seq: events.len(),
        kind,
    });
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn tube_value(tubes: &[u16; 4]) -> u64 {
    tubes
        .iter()
        .zip(DENOMINATIONS.iter())
        .map(|(&count, &coin)| u64::from(count) * u64::from(coin))
        .sum()
}

fn exact_change(tubes: &[u16; 4], amount: u16) -> Option<Vec<u16>> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let denomination = DENOMINATIONS[i];
        let take = tubes[i].min(remaining / denomination);
        greedy[i] = take;
        remaining -= take * denomination;
    }
    if remaining == 0 {
        let mut result = Vec::new();
        for i in (0..4).rev() {
            result.extend(std::iter::repeat_n(DENOMINATIONS[i], usize::from(greedy[i])));
        }
        return Some(result);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;

    fn search(
        i: isize,
        remaining: u16,
        tubes: &[u16; 4],
        counts: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_count: &mut u32,
    ) {
        if i < 0 {
            if remaining == 0 {
                let total: u32 = counts.iter().map(|&x| u32::from(x)).sum();
                let is_better = match best {
                    None => true,
                    Some(current) => {
                        if total < *best_count {
                            true
                        } else if total > *best_count {
                            false
                        } else {
                            (0..4).rev().find_map(|j| {
                                if counts[j] != current[j] {
                                    Some(counts[j] > current[j])
                                } else {
                                    None
                                }
                            }).unwrap_or(false)
                        }
                    }
                };
                if is_better {
                    *best = Some(*counts);
                    *best_count = total;
                }
            }
            return;
        }

        let index = i as usize;
        let coin = DENOMINATIONS[index];
        let max_take = tubes[index].min(remaining / coin);
        for take in 0..=max_take {
            let used: u32 = counts.iter().map(|&x| u32::from(x)).sum();
            if used + u32::from(take) > *best_count {
                break;
            }
            counts[index] = take;
            search(
                i - 1,
                remaining - take * coin,
                tubes,
                counts,
                best,
                best_count,
            );
        }
        counts[index] = 0;
    }

    let mut counts = [0u16; 4];
    search(3, amount, tubes, &mut counts, &mut best, &mut best_count);
    best.map(|counts| {
        let mut result = Vec::new();
        for i in (0..4).rev() {
            result.extend(std::iter::repeat_n(DENOMINATIONS[i], usize::from(counts[i])));
        }
        result
    })
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::<String>::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let index = denomination_index(coin).ok_or(Error::BadCoin { coin })?;
        let new_credit = u32::from(self.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }
        let new_credit = new_credit as u16;
        self.escrow.push(coin);
        self.credit = new_credit;
        self.inserted = self.inserted.saturating_add(u64::from(coin));
        let _ = index;
        push_event(
            &mut self.events,
            EventKind::Inserted {
                coin,
                credit: new_credit,
            },
        );
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !self.products.iter().any(|p| p.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        push_event(
            &mut self.events,
            EventKind::Selected {
                id: id.to_owned(),
            },
        );
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self.products.iter().position(|p| p.id == id).unwrap();
        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change = exact_change(&self.tubes, amount).ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for coin in &change {
            let i = denomination_index(*coin).unwrap();
            proposed[i] -= 1;
        }
        for coin in &self.escrow {
            let i = denomination_index(*coin).unwrap();
            proposed[i] = proposed[i].saturating_add(1);
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let mut change_value = 0u64;
        for coin in &change {
            change_value = change_value.saturating_add(u64::from(*coin));
        }

        self.tubes = proposed;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        self.change = self.change.saturating_add(change_value);

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        push_event(
            &mut self.events,
            EventKind::Vended {
                id,
                price,
                change,
            },
        );
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let credit = self.credit;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.refunded = self.refunded.saturating_add(u64::from(credit));
        push_event(
            &mut self.events,
            EventKind::Refunded {
                coins: coins.clone(),
            },
        );
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = self
            .products
            .iter()
            .position(|p| p.id == id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        push_event(
            &mut self.events,
            EventKind::PriceChanged {
                id: id.to_owned(),
                old,
                new: price,
            },
        );
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = self
            .products
            .iter()
            .position(|p| p.id == id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let total = u32::from(self.products[index].stock) + u32::from(count);
        if total > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = total as u16;
        push_event(
            &mut self.events,
            EventKind::Restocked {
                id: id.to_owned(),
                added: count,
                stock: total as u16,
            },
        );
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        push_event(&mut self.events, EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value(&self.tubes).saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(
        &mut self,
        expected: [u16; 4],
        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMINATIONS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let before_value = tube_value(&self.tubes);
        let after_value = tube_value(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift {
                before: before_value,
                after: after_value,
            });
        }

        let delta = i128::from(after_value) - i128::from(before_value);
        self.adjustment = if delta >= 0 {
            self.adjustment.saturating_add(delta as i64)
        } else {
            self.adjustment.saturating_sub((-delta) as i64)
        };
        let before = self.tubes;
        self.tubes = actual;
        push_event(
            &mut self.events,
            EventKind::Reconciled {
                before,
                after: actual,
            },
        );
        Ok(())
    }
}
```
