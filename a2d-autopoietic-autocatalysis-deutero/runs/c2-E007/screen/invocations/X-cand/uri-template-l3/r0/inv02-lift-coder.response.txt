#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Operator,
    names: Vec<(&'a str, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let rest = &template[cursor..];
        let Some(relative_open) = rest.find(['{', '}']) else {
            break;
        };
        let brace_pos = cursor + relative_open;
        match template.as_bytes()[brace_pos] {
            b'}' => return Err(Error::UnexpectedClose { pos: brace_pos }),
            b'{' => {
                let content_start = brace_pos + 1;
                let Some(relative_close) = template[content_start..].find('}') else {
                    return Err(Error::Unclosed { pos: brace_pos });
                };
                let close_pos = content_start + relative_close;

                if let Some(relative_nested) = template[content_start..close_pos].find('{') {
                    return Err(Error::NestedOpen {
                        pos: content_start + relative_nested,
                    });
                }

                let expression = parse_expression(template, content_start, close_pos)?;
                expressions.push((brace_pos, close_pos + 1, expression));
                cursor = close_pos + 1;
            }
            _ => unreachable!(),
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    for (open_pos, after_close, expression) in expressions {
        output.push_str(&template[literal_start..open_pos]);
        expand_expression(&mut output, &expression, vars);
        literal_start = after_close;
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn parse_expression(
    template: &str,
    content_start: usize,
    close_pos: usize,
) -> Result<Expression<'_>, Error> {
    let bytes = template.as_bytes();
    let mut start = content_start;
    let operator = if start < close_pos {
        match bytes[start] {
            b'+' => {
                start += 1;
                Operator::Reserved
            }
            b'#' => {
                start += 1;
                Operator::Fragment
            }
            b'.' => {
                start += 1;
                Operator::Dot
            }
            b'/' => {
                start += 1;
                Operator::Slash
            }
            b';' => {
                start += 1;
                Operator::Semicolon
            }
            b'?' => {
                start += 1;
                Operator::Query
            }
            b'&' => {
                start += 1;
                Operator::Continuation
            }
            b'!' | b'=' | b',' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if start == close_pos {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut names = Vec::new();
    let mut spec_start = start;
    let mut seen: Vec<&str> = Vec::new();

    loop {
        let spec_end = template[spec_start..close_pos]
            .find(',')
            .map(|n| spec_start + n)
            .unwrap_or(close_pos);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name = &template[spec_start..spec_end];
        let name_bytes = name.as_bytes();
        if !name_bytes[0].is_ascii_alphabetic() && name_bytes[0] != b'_' {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (i, byte) in name_bytes.iter().enumerate().skip(1) {
            if !byte.is_ascii_alphanumeric() && *byte != b'_' {
                return Err(Error::InvalidName {
                    pos: spec_start + i,
                });
            }
        }

        if seen.iter().any(|previous| *previous == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        names.push((name, spec_start));

        if spec_end == close_pos {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, names })
}

fn expand_expression(output: &mut String, expression: &Expression<'_>, vars: &[(&str, &str)]) {
    let separator = match expression.operator {
        Operator::Simple | Operator::Reserved | Operator::Fragment => ",",
        Operator::Dot => ".",
        Operator::Slash => "/",
        Operator::Semicolon | Operator::Query | Operator::Continuation => "&",
    };

    let mut fields = Vec::new();
    for &(name, _) in &expression.names {
        if let Some(value) = vars.iter().rev().find(|(key, _)| *key == name).map(|(_, v)| *v) {
            let encoded = encode(value, matches!(expression.operator, Operator::Reserved | Operator::Fragment));
            let field = match expression.operator {
                Operator::Semicolon if encoded.is_empty() => name.to_string(),
                Operator::Semicolon | Operator::Query | Operator::Continuation => {
                    format!("{name}={encoded}")
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return;
    }

    match expression.operator {
        Operator::Fragment => output.push('#'),
        Operator::Dot => output.push('.'),
        Operator::Slash => output.push('/'),
        Operator::Semicolon => output.push(';'),
        Operator::Query => output.push('?'),
        Operator::Continuation => output.push('&'),
        Operator::Simple | Operator::Reserved => {}
    }

    for (index, field) in fields.iter().enumerate() {
        if index > 0 {
            output.push_str(separator);
        }
        output.push_str(field);
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                result.push('%');
                result.push(HEX[high] as char);
                result.push(HEX[low] as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    result
}

fn hex_value(byte: u8) -> Option<usize> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as usize),
        b'a'..=b'f' => Some((byte - b'a' + 10) as usize),
        b'A'..=b'F' => Some((byte - b'A' + 10) as usize),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'('
            | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
