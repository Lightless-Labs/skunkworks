#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<u8>,
    names: Vec<&'a str>,
    name_positions: Vec<usize>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;

                if let Some(relative_inner) =
                    bytes[open + 1..close].iter().position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_inner,
                    });
                }

                let mut content_start = open + 1;
                let mut operator = None;
                if content_start < close
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' | b'!' | b'@' | b'|'
                    )
                {
                    let op = bytes[content_start];
                    if !matches!(op, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    operator = Some(op);
                    content_start += 1;
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut name_positions = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &template[spec_start..spec_end];
                    let spec_bytes = spec.as_bytes();
                    if !is_name_start(spec_bytes[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    for (offset, &byte) in spec_bytes.iter().enumerate().skip(1) {
                        if !is_name_continue(byte) {
                            return Err(Error::InvalidName {
                                pos: spec_start + offset,
                            });
                        }
                    }

                    if seen.iter().any(|&name| name == spec) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(spec);
                    names.push(spec);
                    name_positions.push(spec_start);

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start: open,
                    end: close + 1,
                    operator,
                    names,
                    name_positions,
                });
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);
        append_expression(&mut output, expression, vars);
        literal_start = expression.end;
    }
    output.push_str(&template[literal_start..]);

    Ok(output)
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn append_expression(output: &mut String, expression: Expression<'_>, vars: &[(&str, &str)]) {
    let mut fields: Vec<(&str, String)> = Vec::new();

    for &name in &expression.names {
        if let Some(value) = vars.iter().rev().find_map(|&(binding_name, value)| {
            if binding_name == name {
                Some(value)
            } else {
                None
            }
        }) {
            fields.push((name, encode_value(value, expression.operator)));
        }
    }

    if fields.is_empty() {
        return;
    }

    let operator = expression.operator;
    let separator = match operator {
        Some(b'.') => '.',
        Some(b'/') => '/',
        Some(b';') => ';',
        Some(b'?') | Some(b'&') => '&',
        _ => ',',
    };

    match operator {
        Some(b'#') => output.push('#'),
        Some(b'.') => output.push('.'),
        Some(b'/') => output.push('/'),
        Some(b';') => output.push(';'),
        Some(b'?') => output.push('?'),
        Some(b'&') => output.push('&'),
        _ => {}
    }

    for (index, (name, value)) in fields.iter().enumerate() {
        if index > 0 {
            output.push(separator);
        }

        match operator {
            Some(b';') => {
                output.push_str(name);
                if !value.is_empty() {
                    output.push('=');
                    output.push_str(value);
                }
            }
            Some(b'?') | Some(b'&') => {
                output.push_str(name);
                output.push('=');
                output.push_str(value);
            }
            _ => output.push_str(value),
        }
    }
}

fn encode_value(value: &str, operator: Option<u8>) -> String {
    let bytes = value.as_bytes();
    let mut encoded = String::new();
    let reserved_operator = matches!(operator, Some(b'+') | Some(b'#'));
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                encoded.push('%');
                encoded.push(hex_digit(high));
                encoded.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved_operator && is_reserved(byte)) {
            encoded.push(byte as char);
        } else {
            encoded.push('%');
            encoded.push(hex_digit(byte >> 4));
            encoded.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    encoded
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
