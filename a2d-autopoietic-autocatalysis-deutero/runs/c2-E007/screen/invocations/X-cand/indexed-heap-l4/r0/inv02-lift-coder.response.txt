```rust
use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Debug, Clone)]
struct Node {
    item: Item,
    age: u128,
}

#[derive(Debug, Clone)]
pub struct IndexedHeap {
    nodes: Vec<Node>,
    capacity: Option<usize>,
    next_age: u128,
}

fn canonicalize_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

fn compare_nodes(a: &Node, b: &Node) -> Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            nodes: Vec::new(),
            capacity: None,
            next_age: 0,
        }
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self {
            nodes: Vec::new(),
            capacity: Some(capacity),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.nodes.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn find_index(&self, key: &str) -> Option<usize> {
        self.nodes.iter().position(|node| node.item.key == key)
    }

    fn sorted_indices(&self) -> Vec<usize> {
        let mut indices: Vec<usize> = (0..self.nodes.len()).collect();
        indices.sort_by(|&a, &b| compare_nodes(&self.nodes[a], &self.nodes[b]));
        indices
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut indices: Vec<usize> = (0..self.nodes.len()).collect();
            indices.sort_by_key(|&i| self.nodes[i].age);
            for (age, index) in indices.into_iter().enumerate() {
                self.nodes[index].age = age as u128;
            }
            self.next_age = self.nodes.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    fn trim_to_capacity(&mut self) -> Vec<Item> {
        let Some(capacity) = self.capacity else {
            return Vec::new();
        };

        let mut evicted = Vec::new();
        while self.nodes.len() > capacity {
            let worst = (0..self.nodes.len())
                .max_by(|&a, &b| compare_nodes(&self.nodes[a], &self.nodes[b]))
                .expect("nonempty while over capacity");
            evicted.push(self.nodes.swap_remove(worst).item);
        }
        evicted
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonicalize_key(key)?;
        if self.find_index(&key).is_some() {
            return Err(Error::Duplicate { key });
        }

        let proposed = Item {
            key,
            priority,
            value: value.to_owned(),
        };

        if let Some(capacity) = self.capacity {
            if capacity == 0 {
                return Ok(Admission::Discarded(proposed));
            }
            if self.nodes.len() >= capacity {
                let worst = (0..self.nodes.len())
                    .max_by(|&a, &b| compare_nodes(&self.nodes[a], &self.nodes[b]))
                    .expect("a full positive-capacity queue is nonempty");
                let proposed_node = Node {
                    item: proposed.clone(),
                    age: self.next_age,
                };
                if compare_nodes(&proposed_node, &self.nodes[worst]) != Ordering::Less {
                    return Ok(Admission::Discarded(proposed));
                }
                let evicted = self.nodes.swap_remove(worst).item;
                let age = self.fresh_age();
                self.nodes.push(Node { item: proposed, age });
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.fresh_age();
        self.nodes.push(Node { item: proposed, age });
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.sorted_indices()
            .first()
            .map(|&index| self.nodes[index].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let best = (0..self.nodes.len())
            .min_by(|&a, &b| compare_nodes(&self.nodes[a], &self.nodes[b]))?;
        Some(self.nodes.swap_remove(best).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize_key(key)?;
        Ok(self.find_index(&key).map(|index| self.nodes[index].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize_key(key)?;
        let index = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = self.nodes[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.nodes[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize_key(key)?;
        let index = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = self.nodes[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.nodes[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize_key(key)?;
        let index = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(std::mem::replace(
            &mut self.nodes[index].item.value,
            value.to_owned(),
        ))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize_key(key)?;
        let index = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(self.nodes.swap_remove(index).item)
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order = other.sorted_indices();
        if policy == CollisionPolicy::Reject {
            for &index in &source_order {
                let key = &other.nodes[index].item.key;
                if self.find_index(key).is_some() {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let source_items: Vec<Item> = source_order
            .into_iter()
            .map(|index| other.nodes[index].item.clone())
            .collect();

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for source_item in source_items {
            if let Some(index) = self.find_index(&source_item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.nodes[index].item.priority = source_item.priority;
                        self.nodes[index].item.value = source_item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                self.nodes.push(Node {
                    item: source_item,
                    age,
                });
                report.moved += 1;
            }
        }

        other.nodes.clear();
        report.evicted = self.trim_to_capacity();
        Ok(report)
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        self.trim_to_capacity()
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.sorted_indices()
            .into_iter()
            .map(|index| self.nodes[index].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }
}
```
