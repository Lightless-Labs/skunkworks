```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    capacity: Option<usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::with_capacity_internal(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_capacity_internal(Some(capacity))
    }

    fn with_capacity_internal(capacity: Option<usize>) -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            capacity,
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn canonical_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();
        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }
        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &b) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                b.is_ascii_alphabetic()
            } else {
                b.is_ascii_alphabetic() || b.is_ascii_digit() || b == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }
        Ok(input[start..end].to_ascii_lowercase())
    }

    fn compare(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (i, entry) in self.entries.iter().enumerate() {
            self.indices.insert(entry.item.key.clone(), i);
        }
    }

    fn ordered_indices(&self) -> Vec<usize> {
        let mut indices: Vec<usize> = (0..self.entries.len()).collect();
        indices.sort_by(|&a, &b| Self::compare(&self.entries[a], &self.entries[b]));
        indices
    }

    fn pop_worst(&mut self) -> Option<Item> {
        let worst = self
            .entries
            .iter()
            .enumerate()
            .max_by(|(_, a), (_, b)| Self::compare(a, b))
            .map(|(i, _)| i)?;
        Some(self.remove_index(worst).item)
    }

    fn remove_index(&mut self, index: usize) -> Entry {
        let removed = self.entries.swap_remove(index);
        self.indices.remove(&removed.item.key);
        if index < self.entries.len() {
            self.indices
                .insert(self.entries[index].item.key.clone(), index);
        }
        removed
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<Admission, Error> {
        let key = Self::canonical_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Item {
            key,
            priority,
            value: value.to_owned(),
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed));
        }

        let entry = Entry {
            item: proposed.clone(),
            age: self.next_age,
        };

        if let Some(limit) = self.capacity {
            if self.entries.len() >= limit {
                let worst = self
                    .entries
                    .iter()
                    .enumerate()
                    .max_by(|(_, a), (_, b)| Self::compare(a, b))
                    .map(|(i, e)| (i, e));
                if let Some((worst_index, worst_entry)) = worst {
                    if Self::compare(&entry, worst_entry) != std::cmp::Ordering::Less {
                        return Ok(Admission::Discarded(proposed));
                    }
                    let evicted = self.remove_index(worst_index).item;
                    self.entries.push(entry);
                    self.indices.insert(proposed.key.clone(), self.entries.len() - 1);
                    self.next_age = self.next_age.wrapping_add(1);
                    self.rebuild_indices();
                    return Ok(Admission::Evicted(evicted));
                }
            }
        }

        self.entries.push(entry);
        self.indices.insert(proposed.key, self.entries.len() - 1);
        self.next_age = self.next_age.wrapping_add(1);
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.ordered_indices()
            .first()
            .map(|&i| self.entries[i].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let best = self
            .entries
            .iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| Self::compare(a, b))
            .map(|(i, _)| i)?;
        Some(self.remove_index(best).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonical_key(key)?;
        Ok(self.indices.get(&key).map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(std::mem::replace(
            &mut self.entries[index].item.value,
            value.to_owned(),
        ))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.remove_index(index).item)
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = Vec::new();
        if let Some(limit) = capacity {
            while self.entries.len() > limit {
                if let Some(item) = self.pop_worst() {
                    evicted.push(item);
                }
            }
        }
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.ordered_indices()
            .into_iter()
            .map(|i| self.entries[i].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let ordered = other.ordered_indices();
        if policy == CollisionPolicy::Reject {
            for &i in &ordered {
                let key = &other.entries[i].item.key;
                if self.indices.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let source: Vec<Entry> = std::mem::take(&mut other.entries);
        other.indices.clear();

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for i in ordered {
            let source_entry = source[i].clone();
            let key = source_entry.item.key.clone();
            if let Some(&index) = self.indices.get(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = source_entry.item.priority;
                        self.entries[index].item.value = source_entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let entry = Entry {
                    item: source_entry.item,
                    age: self.next_age,
                };
                self.next_age = self.next_age.wrapping_add(1);
                self.indices.insert(key, self.entries.len());
                self.entries.push(entry);
                report.moved += 1;
            }
        }

        if let Some(limit) = self.capacity {
            while self.entries.len() > limit {
                if let Some(item) = self.pop_worst() {
                    report.evicted.push(item);
                }
            }
        }

        Ok(report)
    }
}
```
