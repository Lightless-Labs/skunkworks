use std::cmp::Reverse;
use std::collections::{BTreeMap, BTreeSet, BinaryHeap, HashSet, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EdgeStatus {
    Duplicate,
    Internal,
    Link,
    Merge,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Step {
    pub edge_index: usize,
    pub status: EdgeStatus,
    pub before: (usize, usize),
    pub after: (usize, usize),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub edge_index: usize,
    pub before: Vec<usize>,
    pub after: usize,
    pub members: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Batch {
    pub steps: Vec<Step>,
    pub merges: Vec<Merge>,
    pub graph: Graph,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn validate(n: usize, edges: &[(usize, usize)]) -> Result<(), GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }
    Ok(())
}

fn analyze_valid(n: usize, edges: &[(usize, usize)]) -> Graph {
    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next)) = stack.pop() {
            if next < adj[v].len() {
                stack.push((v, next + 1));
                let w = adj[v][next];
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                order.push(v);
            }
        }
    }

    let mut raw_of = vec![usize::MAX; n];
    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    for &root in order.iter().rev() {
        if raw_of[root] != usize::MAX {
            continue;
        }
        let id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![root];
        raw_of[root] = id;
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if raw_of[w] == usize::MAX {
                    raw_of[w] = id;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    let mut permutation: Vec<usize> = (0..raw_components.len()).collect();
    permutation.sort_by(|&a, &b| {
        raw_components[b]
            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
    });
    let mut remap = vec![0; raw_components.len()];
    let mut components = Vec::with_capacity(raw_components.len());
    for (new_id, old_id) in permutation.into_iter().enumerate() {
        remap[old_id] = new_id;
        components.push(raw_components[old_id].clone());
    }
    let component_of: Vec<usize> = raw_of.iter().map(|&id| remap[id]).collect();
    let c = components.len();

    let mut cyclic: Vec<bool> = components.iter().map(|x| x.len() > 1).collect();
    let mut internal_edges = vec![0; c];
    let mut arc_data: BTreeMap<(usize, usize), (usize, usize)> = BTreeMap::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }
    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); c];
    let mut indegree = vec![0usize; c];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = BinaryHeap::new();
    for id in 0..c {
        if indegree[id] == 0 {
            ready.push((
                internal_edges[id],
                components[id][0],
                Reverse(id),
            ));
        }
    }
    let mut topo = Vec::with_capacity(c);
    let mut depth = vec![0usize; c];
    while let Some((_, _, Reverse(v))) = ready.pop() {
        topo.push(v);
        for &w in &outgoing[v] {
            depth[w] = depth[w].max(depth[v] + 1);
            indegree[w] -= 1;
            if indegree[w] == 0 {
                ready.push((internal_edges[w], components[w][0], Reverse(w)));
            }
        }
    }

    Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    }
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;
    Ok(analyze_valid(n, edges))
}

fn shortest_route(graph: &Graph, start: usize, goal: usize) -> Option<Vec<usize>> {
    if start == goal {
        return Some(vec![start]);
    }
    let c = graph.components.len();
    let mut outgoing = vec![Vec::new(); c];
    for arc in &graph.arcs {
        outgoing[arc.from].push(arc.to);
    }
    for next in &mut outgoing {
        next.sort_unstable();
    }

    let mut distance = vec![usize::MAX; c];
    let mut queue = VecDeque::new();
    distance[start] = 0;
    queue.push_back(start);
    while let Some(v) = queue.pop_front() {
        for &w in &outgoing[v] {
            if distance[w] == usize::MAX {
                distance[w] = distance[v] + 1;
                queue.push_back(w);
            }
        }
    }
    if distance[goal] == usize::MAX {
        return None;
    }

    let mut route = vec![start];
    let mut current = start;
    while current != goal {
        let next = outgoing[current]
            .iter()
            .copied()
            .filter(|&w| distance[w] == distance[current] + 1 && distance[w] <= distance[goal])
            .find(|&w| {
                // Keep only vertices that can still reach the goal along shortest paths.
                let mut seen = vec![false; c];
                let mut q = VecDeque::from([w]);
                seen[w] = true;
                while let Some(x) = q.pop_front() {
                    if x == goal {
                        return true;
                    }
                    for &y in &outgoing[x] {
                        if !seen[y]
                            && distance[y] == distance[x] + 1
                            && distance[y] <= distance[goal]
                        {
                            seen[y] = true;
                            q.push_back(y);
                        }
                    }
                }
                false
            })?;
        current = next;
        route.push(current);
    }
    Some(route)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let n = self.component_of.len();
        if from >= n {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            return Ok(if self.cyclic[a] {
                Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                }
            } else {
                Query {
                    kind: Relation::Trivial,
                    route: vec![a],
                }
            });
        }
        if let Some(route) = shortest_route(self, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_route(self, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    graph: Graph,
    history: Vec<Merge>,
    pairs: HashSet<(usize, usize)>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        let graph = analyze(n, edges)?;
        Ok(Self {
            n,
            edges: edges.to_vec(),
            graph,
            history: Vec::new(),
            pairs: edges.iter().copied().collect(),
        })
    }

    pub fn snapshot(&self) -> Graph {
        self.graph.clone()
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(from, to)) in edges.iter().enumerate() {
            if from >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: from,
                });
            }
            if to >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: to,
                });
            }
        }

        let mut steps = Vec::with_capacity(edges.len());
        let mut merges = Vec::new();
        for &(from, to) in edges {
            let edge_index = self.edges.len();
            let before = (
                self.graph.component_of[from],
                self.graph.component_of[to],
            );
            let duplicate = self.pairs.contains(&(from, to));
            self.edges.push((from, to));
            self.pairs.insert((from, to));
            self.graph = analyze_valid(self.n, &self.edges);
            let after = (
                self.graph.component_of[from],
                self.graph.component_of[to],
            );

            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };
            if status == EdgeStatus::Merge {
                let mut before_ids = BTreeSet::new();
                for (vertex, &id) in self.graph.component_of.iter().enumerate() {
                    if self.graph.component_of[vertex] == after.0 {
                        // Recover the pre-insertion component ID using the previous graph below.
                        let _ = id;
                    }
                }
                // A pre-state is reconstructed from the graph before this occurrence.
                // The united members are exactly the vertices in the post-state component.
                let members = self.graph.components[after.0].clone();
                // Every distinct old component represented by these members is included.
                // The old graph is obtained by removing just the current edge.
                let mut prior_edges = self.edges.clone();
                prior_edges.pop();
                let prior = analyze_valid(self.n, &prior_edges);
                for &v in &members {
                    before_ids.insert(prior.component_of[v]);
                }
                let event = Merge {
                    edge_index,
                    before: before_ids.into_iter().collect(),
                    after: after.0,
                    members,
                };
                merges.push(event.clone());
                self.history.push(event);
            }
            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.graph.clone(),
        })
    }
}
