#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

const ADLER_MOD: u32 = 65_521;

fn crc32_update(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low_bit = crc & 1;
            crc >>= 1;
            if low_bit != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % ADLER_MOD;
        *b = (*b + *a) % ADLER_MOD;
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_update(0xffff_ffff, data) ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;

    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }

    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;

        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }

        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = crc32_update(self.crc, data);
        adler_update(&mut self.a, &mut self.b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;

        for span in spans {
            self.update(&data[span.start..span.end]);
        }

        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksum = self.checksum();
        format!("c{:08x}:a{:08x}", checksum.crc32, checksum.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();

        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        for offset in (1..9).chain(11..19) {
            let byte = bytes[offset];
            if !byte.is_ascii_digit() && !(b'a'..=b'f').contains(&byte) {
                return Err(SnapshotError::Hex { offset });
            }
        }

        let crc = parse_hex(&bytes[1..9]);
        let adler = parse_hex(&bytes[11..19]);
        Self::from_seed(crc, adler)
    }
}

fn parse_hex(bytes: &[u8]) -> u32 {
    let mut value = 0u32;
    for &byte in bytes {
        let digit = match byte {
            b'0'..=b'9' => (byte - b'0') as u32,
            b'a'..=b'f' => (byte - b'a' + 10) as u32,
            _ => unreachable!(),
        };
        value = (value << 4) | digit;
    }
    value
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;

    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }

    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut chunks = data.chunks_exact(2);

    for word in &mut chunks {
        let value = ((word[0] as u64) << 8) | word[1] as u64;
        a = (a + value) % 65_535;
        b = (b + a) % 65_535;
    }

    let remainder = chunks.remainder();
    if let Some(&byte) = remainder.first() {
        a = (a + ((byte as u64) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }

    ((b as u32) << 16) | a as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut a = 0u64;
    let mut b = 0u64;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                let word = ((high as u64) << 8) | byte as u64;
                a = (a + word) % 65_535;
                b = (b + a) % 65_535;
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        a = (a + ((high as u64) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }

    Ok(((b as u32) << 16) | a as u32)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc: u16 = match variant {
                Crc16::Xmodem => 0,
                Crc16::False => 0xffff,
                Crc16::X25 => unreachable!(),
            };

            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top_bit = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top_bit != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc: u16 = 0xffff;
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low_bit = crc & 1;
                    crc >>= 1;
                    if low_bit != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
            crc ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffff;
    let mut adler_a = 1;
    let mut adler_b = 0;
    let mut fletcher16_a = 0u32;
    let mut fletcher16_b = 0u32;
    let mut fletcher32_a = 0u64;
    let mut fletcher32_b = 0u64;
    let mut crc16_reg: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc ^= byte as u32;
            for _ in 0..8 {
                let low_bit = crc & 1;
                crc >>= 1;
                if low_bit != 0 {
                    crc ^= 0xedb8_8320;
                }
            }

            adler_a = (adler_a + byte as u32) % ADLER_MOD;
            adler_b = (adler_b + adler_a) % ADLER_MOD;

            fletcher16_a = (fletcher16_a + byte as u32) % 255;
            fletcher16_b = (fletcher16_b + fletcher16_a) % 255;

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_reg ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top_bit = crc16_reg & 0x8000;
                        crc16_reg = crc16_reg.wrapping_shl(1);
                        if top_bit != 0 {
                            crc16_reg ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_reg ^= byte as u16;
                    for _ in 0..8 {
                        let low_bit = crc16_reg & 1;
                        crc16_reg >>= 1;
                        if low_bit != 0 {
                            crc16_reg ^= 0x8408;
                        }
                    }
                }
            }

            if let Some(high) = pending_high.take() {
                let word = ((high as u64) << 8) | byte as u64;
                fletcher32_a = (fletcher32_a + word) % 65_535;
                fletcher32_b = (fletcher32_b + fletcher32_a) % 65_535;
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_a = (fletcher32_a + ((high as u64) << 8)) % 65_535;
        fletcher32_b = (fletcher32_b + fletcher32_a) % 65_535;
    }

    let crc16_result = match variant {
        Crc16::Xmodem | Crc16::False => crc16_reg,
        Crc16::X25 => crc16_reg ^ 0xffff,
    };

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((fletcher16_b as u16) << 8) | fletcher16_a as u16,
        fletcher32: ((fletcher32_b as u32) << 16) | fletcher32_a as u32,
        crc16: crc16_result,
    })
}
