```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
    level: usize,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        0
    } else {
        let numerator = total * 100;
        let denominator = count as u64;
        (numerator + denominator / 2) / denominator
    }
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let count = jobs.len();
    if count == 0 {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(count * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
            level: job.level,
        });
    }
    events.sort_by_key(|event| {
        (
            event.time,
            match event.kind {
                EventKind::Arrival => 0,
                EventKind::Wake => 1,
            },
            event.index,
        )
    });

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut burst_position = vec![0usize; count];
    let mut remaining_cpu: Vec<u32> = jobs.iter().map(|job| job.bursts[0]).collect();
    let mut completion = vec![0u64; count];
    let mut completed = vec![false; count];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut event_cursor = 0usize;
    let mut completed_count = 0usize;
    let mut last_dispatch_id: Option<u32> = None;

    while completed_count < count {
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let job_id = jobs[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != job_id) {
            time += switch_cost as u64;
        }
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        let start = time;
        let consumed = remaining_cpu[index].min(quanta[level]);
        time += consumed as u64;
        remaining_cpu[index] -= consumed;

        let current_burst_ends = remaining_cpu[index] == 0;
        let finished = current_burst_ends && burst_position[index] + 1 == jobs[index].bursts.len();
        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished,
            level,
        });

        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        last_dispatch_id = Some(job_id);

        if !current_burst_ends {
            let next_level = (level + 1).min(quanta.len() - 1);
            queues[next_level].push_back(index);
            continue;
        }

        if finished {
            completion[index] = time;
            completed[index] = true;
            completed_count += 1;
            continue;
        }

        let io_position = burst_position[index] + 1;
        let io_delay = jobs[index].bursts[io_position] as u64;
        let next_cpu_position = io_position + 1;
        burst_position[index] = next_cpu_position;
        remaining_cpu[index] = jobs[index].bursts[next_cpu_position];

        let wake_level = level.saturating_sub(1);
        let wake = Event {
            time: time + io_delay,
            kind: EventKind::Wake,
            index,
            level: wake_level,
        };
        let insert_at = events.partition_point(|event| {
            (
                event.time,
                match event.kind {
                    EventKind::Arrival => 0,
                    EventKind::Wake => 1,
                },
                event.index,
            ) <= (
                wake.time,
                1,
                wake.index,
            )
        });
        events.insert(insert_at, wake);
    }

    let mut stats = Vec::with_capacity(count);
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let turnaround = completion[index] - job.arrival as u64;
        let total_cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| burst as u64)
            .sum();
        let total_io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&delay| delay as u64)
            .sum();
        let waiting = turnaround - total_cpu - total_io;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion: completion[index],
            turnaround,
            waiting,
        });
    }

    let end = completion.iter().copied().max().unwrap_or(0);
    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, count),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, count),
        end,
    })
}

use std::collections::VecDeque;
```
