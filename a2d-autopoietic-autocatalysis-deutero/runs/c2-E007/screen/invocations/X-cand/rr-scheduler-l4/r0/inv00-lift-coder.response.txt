```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    burst_pos: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: job.arrival as u64,
            bursts: job.bursts.clone(),
            burst_pos: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut last_dispatch_id: Option<u32> = None;

    loop {
        events.sort_by_key(|event| (event.time, event.kind, event.index));

        let mut admitted = Vec::new();
        let mut remaining_events = Vec::new();
        for event in events.drain(..) {
            if event.time <= now {
                admitted.push(event);
            } else {
                remaining_events.push(event);
            }
        }
        events = remaining_events;
        for event in admitted {
            queues[event.level].push_back(event.index);
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if events.is_empty() {
                break;
            }
            events.sort_by_key(|event| (event.time, event.kind, event.index));
            let next_time = events[0].time;
            if next_time > now {
                if let Some((_, old_end)) = idle.last_mut() {
                    if *old_end == now {
                        *old_end = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = processes[index].id;
        if last_dispatch_id.is_some_and(|last| last != id) {
            now += switch_cost as u64;
        }

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        let mut admitted = Vec::new();
        let mut remaining_events = Vec::new();
        for event in events.drain(..) {
            if event.time <= now {
                admitted.push(event);
            } else {
                remaining_events.push(event);
            }
        }
        events = remaining_events;
        for event in admitted {
            queues[event.level].push_back(event.index);
        }

        let quantum = quanta[level];
        let consumed = processes[index].remaining.min(quantum);
        let start = now;
        now += consumed as u64;
        let finished_cpu = processes[index].remaining == consumed;
        processes[index].remaining -= consumed;

        let final_burst = finished_cpu && processes[index].burst_pos + 1 == processes[index].bursts.len();
        runs.push(Run {
            id,
            start,
            end: now,
            finished: final_burst,
            level,
        });

        // Events through the run endpoint are admitted before any requeue or wake.
        events.sort_by_key(|event| (event.time, event.kind, event.index));
        let mut admitted = Vec::new();
        let mut remaining_events = Vec::new();
        for event in events.drain(..) {
            if event.time <= now {
                admitted.push(event);
            } else {
                remaining_events.push(event);
            }
        }
        events = remaining_events;
        for event in admitted {
            queues[event.level].push_back(event.index);
        }

        if !finished_cpu {
            processes[index].level = (level + 1).min(quanta.len() - 1);
            queues[processes[index].level].push_back(index);
        } else if final_burst {
            processes[index].completion = Some(now);
        } else {
            let io_delay = processes[index].bursts[processes[index].burst_pos + 1] as u64;
            processes[index].burst_pos += 2;
            processes[index].remaining = processes[index].bursts[processes[index].burst_pos];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: now + io_delay,
                kind: 1,
                index,
                level: wake_level,
            });
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = processes[index].completion.unwrap_or(0);
        let turnaround = completion - job.arrival as u64;
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let waiting = turnaround - cpu_total - io_total;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = if count == 0 {
        0
    } else {
        (total_waiting * 100 + count / 2) / count
    };
    let mean_turnaround_x100 = if count == 0 {
        0
    } else {
        (total_turnaround * 100 + count / 2) / count
    };
    let end = stats.iter().map(|stat| stat.completion).max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}
```
