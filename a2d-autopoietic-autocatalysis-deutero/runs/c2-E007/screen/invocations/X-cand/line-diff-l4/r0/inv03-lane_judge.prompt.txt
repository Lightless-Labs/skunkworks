fn split_lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn canonical_changes(base: &[String], other: &[String], side: usize) -> Vec<Change> {
    let n = base.len();
    let m = other.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == other[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut operations: Vec<(bool, String)> = Vec::new(); // true = Keep
    let (mut i, mut j) = (0, 0);
    while i < n && j < m {
        if base[i] == other[j] {
            operations.push((true, base[i].clone()));
            i += 1;
            j += 1;
        } else {
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            if delete_len > insert_len || (delete_len == insert_len && (i + j) % 2 == 0) {
                operations.push((false, base[i].clone()));
                i += 1;
            } else {
                operations.push((false, other[j].clone()));
                j += 1;
            }
        }
    }
    while i < n {
        operations.push((false, base[i].clone()));
        i += 1;
    }
    while j < m {
        operations.push((false, other[j].clone()));
        j += 1;
    }

    let mut changes = Vec::new();
    let mut base_pos = 0;
    let mut active_start = 0;
    let mut active_deletes = 0;
    let mut active_replacement = Vec::new();
    let mut active = false;

    let flush = |changes: &mut Vec<Change>,
                 active: &mut bool,
                 start: usize,
                 deletes: &mut usize,
                 replacement: &mut Vec<String>| {
        if *active {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(replacement),
                side,
            });
            *deletes = 0;
            *active = false;
        }
    };

    for (keep, value) in operations {
        if keep {
            flush(
                &mut changes,
                &mut active,
                active_start,
                &mut active_deletes,
                &mut active_replacement,
            );
            base_pos += 1;
        } else {
            if !active {
                active = true;
                active_start = base_pos;
            }
            if base_pos < n && value == base[base_pos] {
                active_deletes += 1;
                base_pos += 1;
            } else {
                active_replacement.push(value);
            }
        }
    }
    flush(
        &mut changes,
        &mut active,
        active_start,
        &mut active_deletes,
        &mut active_replacement,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn project(base: &[String], start: usize, end: usize, changes: &[&Change]) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;
    for change in changes {
        if cursor < change.start {
            result.extend(base[cursor..change.start].iter().cloned());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (side, input) in [
        (MergeSide::Base, base),
        (MergeSide::Left, left),
        (MergeSide::Right, right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (side, input) in [
        (MergeSide::Base, base),
        (MergeSide::Left, left),
        (MergeSide::Right, right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = canonical_changes(&base_lines, &left_lines, 0);
    changes.extend(canonical_changes(&base_lines, &right_lines, 1));

    let count = changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = root(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..count {
        for j in (i + 1)..count {
            if interacts(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<Vec<usize>> = Vec::new();
    for i in 0..count {
        let ri = root(&mut parent, i);
        if let Some(group) = groups.iter_mut().find(|g| root(&mut parent, g[0]) == ri) {
            group.push(i);
        } else {
            groups.push(vec![i]);
        }
    }

    let mut components: Vec<Component> = groups
        .into_iter()
        .map(|indices| {
            let start = indices.iter().map(|&i| changes[i].start).min().unwrap();
            let end = indices.iter().map(|&i| changes[i].end).max().unwrap();
            Component {
                start,
                end,
                changes: indices,
            }
        })
        .collect();

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output_lines = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0;

    for component in components {
        if base_cursor < component.start {
            output_lines.extend(base_lines[base_cursor..component.start].iter().cloned());
        }

        let mut left_changes: Vec<&Change> = component
            .changes
            .iter()
            .map(|&i| &changes[i])
            .filter(|c| c.side == 0)
            .collect();
        let mut right_changes: Vec<&Change> = component
            .changes
            .iter()
            .map(|&i| &changes[i])
            .filter(|c| c.side == 1)
            .collect();
        left_changes.sort_by_key(|c| (c.start, c.end));
        right_changes.sort_by_key(|c| (c.start, c.end));

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
        );

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines.len();
                    let base_len = component.end - component.start;
                    let base_position = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output_lines.push(format!("<<<<<<< LEFT {ordinal}"));
                    output_lines.extend(left_projection.iter().cloned());
                    output_lines.push(format!("||||||| BASE {base_position},{base_len}"));
                    output_lines.extend(base_projection.iter().cloned());
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection.iter().cloned());
                    output_lines.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        output_lines.extend(base_lines[base_cursor..].iter().cloned());
    }

    let text = if output_lines.is_empty() {
        String::new()
    } else {
        let mut text = output_lines.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn canonical_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {

            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut changes = Vec::new();
    let (mut i, mut j, mut base_pos) = (0usize, 0usize, 0usize);
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let finish_run = |changes: &mut Vec<Change>,
                      run_start: &mut Option<usize>,
                      deletes: &mut usize,
                      inserts: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            finish_run(&mut changes, &mut run_start, &mut deletes, &mut inserts);
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            run_start.get_or_insert(base_pos);
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            run_start.get_or_insert(base_pos);
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0

            if delete {
                deletes += 1;
                i += 1;

                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    finish_run(&mut changes, &mut run_start, &mut deletes, &mut inserts);
    changes

    for (side, input) in &inputs {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side: side.clone() });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = canonical_changes(&base_lines, &left_lines);
    let right_changes = canonical_changes(&base_lines, &right_lines);

    let mut nodes: Vec<(bool, usize)> = Vec::new();
    nodes.extend((0..left_changes.len()).map(|i| (true, i)));
    nodes.extend((0..right_changes.len()).map(|i| (false, i)));

    let mut seen = vec![false; nodes.len()];
    let mut components = Vec::new();

    for seed in 0..nodes.len() {
        if seen[seed] {
            continue;
        }

        let mut stack = vec![seed];
        seen[seed] = true;
        let mut component_left = Vec::new();
        let mut component_right = Vec::new();

        while let Some(node_idx) = stack.pop() {
            let (is_left, change_idx) = nodes[node_idx];
            let change = if is_left {
                &left_changes[change_idx]
            } else {
                &right_changes[change_idx]
            };

            if is_left {
                component_left.push(change.clone());
            } else {
                component_right.push(change.clone());
            }

            for other_idx in 0..nodes.len() {
                if seen[other_idx] {
                    continue;
                }
                let (other_left, other_change_idx) = nodes[other_idx];
                let other = if other_left {
                    &left_changes[other_change_idx]
                } else {
                    &right_changes[other_change_idx]
                };
                if interacts(change, other) {
                    seen[other_idx] = true;
                    stack.push(other_idx);
                }
            }
        }

        component_left.sort_by_key(|c| (c.start, c.end));
        component_right.sort_by_key(|c| (c.start, c.end));
        let all_changes = component_left.iter().chain(component_right.iter());
        let start = all_changes.clone().map(|c| c.start).min().unwrap();
        let end = all_changes.map(|c| c.end).max().unwrap();
        components.push(Component {
            start,
            end,
            left: component_left,
            right: component_right,
        });
    }

    components.sort_by_key(|c| (c.start, c.start != c.end, c.end));

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        if base_cursor < component.start {
            output.extend(base_lines[base_cursor..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.left,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.right,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_line_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    push_line(&mut output, format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection);
                    push_line(&mut output, format!("||||||| BASE {base_line_start},{base_len}"));
                    output.extend(base_projection);
                    push_line(&mut output, "=======".to_owned());
                    output.extend(right_projection);

                    let output_len = 4
                        + component.left.iter().map(|_| 0usize).sum::<usize>()
                        + 0; // The payload lengths are recorded below.
                    let _ = output_len;
                    let block_len = output.len() - output_start + 1;
                    push_line(&mut output, format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: block_len,
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        output.extend(base_lines[base_cursor..].iter().cloned());
    }

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
}

#[derive(Debug, Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<Change>,
}

fn lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn canonical_changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut i = 0;
    let mut j = 0;
    let mut base_pos = 0;
    let mut changes = Vec::new();
    let mut active_start = None;
    let mut deleted = 0usize;
    let mut inserted = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active_start: &mut Option<usize>,
                 deleted: &mut usize,
                 inserted: &mut Vec<String>,
                 side: usize| {
        if let Some(start) = active_start.take() {
            changes.push(Change {
                start,
                end: start + *deleted,
                replacement: std::mem::take(inserted),
                side,
            });
            *deleted = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(
                &mut changes,
                &mut active_start,
                &mut deleted,
                &mut inserted,
                side,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            active_start.get_or_insert(base_pos);
            inserted.push(new[j].clone());
            j += 1;
        } else if j == m {
            active_start.get_or_insert(base_pos);
            deleted += 1;
            i += 1;
            base_pos += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            let choose_delete = if del > ins {
                true
            } else if ins > del {
                false
            } else {
                (i + j) % 2 == 0
            };

            active_start.get_or_insert(base_pos);
            if choose_delete {
                deleted += 1;
                i += 1;
                base_pos += 1;
            } else {
                inserted.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(
        &mut changes,
        &mut active_start,
        &mut deleted,
        &mut inserted,
        side,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;

    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(base: &[String], component: &Component, side: usize) -> Vec<String> {
    let mut changes: Vec<&Change> = component
        .changes
        .iter()
        .filter(|change| change.side == side)
        .collect();
    changes.sort_by_key(|change| (change.start, change.end));

    let mut out = Vec::new();
    let mut cursor = component.start;

    for change in changes {
        if change.start < cursor {
            continue;
        }
        out.extend(base[cursor..change.start].iter().cloned());
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    out.extend(base[cursor..component.end].iter().cloned());
    out
}

fn append_lines(out: &mut Vec<String>, lines: &[String]) {
    out.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = canonical_changes(&base_lines, &left_lines, 0);
    changes.extend(canonical_changes(&base_lines, &right_lines, 1));

    let mut components: Vec<Component> = Vec::new();
    let mut visited = vec![false; changes.len()];

    for seed in 0..changes.len() {
        if visited[seed] {
            continue;
        }
        visited[seed] = true;
        let mut members = vec![seed];
        let mut stack = vec![seed];

        while let Some(current) = stack.pop() {
            for other in 0..changes.len() {
                if !visited[other] && interacts(&changes[current], &changes[other]) {
                    visited[other] = true;
                    members.push(other);
                    stack.push(other);
                }
            }
        }

        let start = members
            .iter()
            .map(|&index| changes[index].start)
            .min()
            .unwrap();
        let end = members
            .iter()
            .map(|&index| changes[index].end)
            .max()
            .unwrap();
        let component_changes = members
            .into_iter()
            .map(|index| changes[index].clone())
            .collect();

        components.push(Component {
            start,
            end,
            changes: component_changes,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then_with(|| a.end.cmp(&b.end))
    });

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in &components {
        append_lines(&mut output, &base_lines[base_cursor..component.start]);

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(&base_lines, component, 0);
        let right_projection = projection(&base_lines, component, 1);

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);

                    let region_len = component.end - component.start;
                    let displayed_start = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {displayed_start},{region_len}"
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    append_lines(&mut output, &base_lines[base_cursor..]);
    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
