    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Debug, Clone)]
struct Row {
    prefix: u8,
    payload: String,
}

#[derive(Debug, Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn physical_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }
    let mut lines: Vec<&str> = text.split_terminator('\n').collect();
    // split_terminator already omits the final empty segment for LF-terminated text.
    // For non-LF-terminated input, it includes the final physical line.
    lines.shrink_to_fit();
    lines
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.starts_with('0') {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str, line_number: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_number };
    if !line.starts_with("@@ -") || !line.ends_with(" @@") {
        return Err(bad());
    }
    let inner = &line[4..line.len() - 3];
    let (old, new) = inner.split_once(" +").ok_or_else(bad)?;
    if new.contains(' ') || old.contains(' ') {
        return Err(bad());
    }
    let (a, b) = old.split_once(',').ok_or_else(bad)?;
    let (c, d) = new.split_once(',').ok_or_else(bad)?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return Err(bad());
    }
    let a = parse_number(a).ok_or_else(bad)?;
    let b = parse_number(b).ok_or_else(bad)?;
    let c = parse_number(c).ok_or_else(bad)?;
    let d = parse_number(d).ok_or_else(bad)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    Ok((a, b, c, d))
}

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 { start - 1 } else { start }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    if lines.first().copied() != Some("--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if lines.get(1).copied() != Some("+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_old_end = 0usize;
    let mut cumulative_delta: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }

        let (a, b, c, d) = parse_hunk_header(lines[i], header_line)?;
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        i += 1;

        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let bytes = lines[i].as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            if lines[i].contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let prefix = bytes[0];
            let payload = lines[i][1..].to_owned();
            if prefix == b' ' || prefix == b'-' {
                old_seen += 1;
            }
            if prefix == b' ' || prefix == b'+' {
                new_seen += 1;
            }
            rows.push(Row { prefix, payload });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i128 + cumulative_delta;
        if required_new < 0 || required_new != new_start as i128 {
            return Err(ApplyError::Misordered { line: header_line });
        }

        let hunk = Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        };

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }
        previous_old_end = old_start + b;
        cumulative_delta += d as i128 - b as i128;
        hunks.push(hunk);
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut lines: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source.split_terminator('\n').map(str::to_owned).collect()
    };
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(std::mem::take(&mut lines))
}

fn abs_distance(a: i128, b: i128) -> i128 {
    (a - b).abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.prefix == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.prefix == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_starts = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut expected_payloads = Vec::new();
                let mut old_index = 0usize;
                for row in &hunk.rows {
                    if row.prefix == b' ' || row.prefix == b'-' {
                        let drop = (row.prefix == b' ' && old_index < l)
                            || (row.prefix == b' '
                                && old_index >= hunk.old_count.saturating_sub(r));
                        if !drop {
                            expected_payloads.push(row.payload.as_str());
                        }
                        old_index += 1;
                    }
                }

                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_distance(s as i128, expected);
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let from = s + l;
                    let to = s + hunk.old_count - r;
                    if from > to || to > working.len() {
                        continue;
                    }
                    if expected_payloads.len() != to - from
                        || !expected_payloads
                            .iter()
                            .zip(&working[from..to])
                            .all(|(a, b)| *a == b)
                    {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(old_rank) if rank < old_rank => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(old_rank) if rank == old_rank => {
                            best_starts.push((s, l, r));
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_index + 1 });
        }

        best_starts.sort_by_key(|&(s, l, _)| (s, l));
        let mut distinct = Vec::new();
        for &(s, _, _) in &best_starts {
            if distinct.last().copied() != Some(s) {
                distinct.push(s);
            }
        }
        if distinct.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }

        let (s, l, r) = best_starts[0];
        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        let mut replacement = Vec::new();
        let mut old_index = 0usize;
        for row in &hunk.rows {
            match row.prefix {
                b' ' | b'-' => {
                    let dropped = row.prefix == b' '
                        && (old_index < l || old_index >= hunk.old_count.saturating_sub(r));
                    old_index += 1;
                    if dropped {
                        continue;
                    }
                    if row.prefix == b' ' || row.prefix == b'+' {
                        replacement.push(row.payload.clone());
                    }
                }
                b'+' => replacement.push(row.payload.clone()),
                _ => unreachable!(),
            }
        }

        working.splice(replace_start..replace_end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

            return Err(ApplyError::BadHeader { line: i + 1 });
        }

        if !lines[i].starts_with("@@") {

            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }
        if old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i128 + delta;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_end = old_start.saturating_add(b);
        delta += d as i128 - b as i128;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });

        if hunks.len() == 17 {
            return Err(ApplyError::TooLarge);
        }
    }

    Ok(hunks)
}

fn apply_hunk(
    source: &mut Vec<String>,
    hunk: &Hunk,
    number: usize,
    options: &PatchOptions,
    net: i128,
    drift: i128,
) -> Result<i128, ApplyError> {
    let lead = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
    let trail = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
    let max_l = options.fuzz.min(lead);
    let max_r = options.fuzz.min(trail);
    let expected = hunk.old_start as i128 + net + drift;

    let mut best_rank: Option<(usize, i128)> = None;
    let mut starts = Vec::<(usize, usize, usize)>::new();

    for l in 0..=max_l {
        for r in 0..=max_r {
            if l + r > hunk.old_count {
                continue;
            }
            let expected_rows: Vec<&str> = hunk.rows[l..hunk.rows.len() - r]
                .iter()
                .filter(|row| row.kind == b' ' || row.kind == b'-')
                .map(|row| row.payload.as_str())
                .collect();
            let max_start = source.len().saturating_sub(hunk.old_count);
            if hunk.old_count > source.len() {
                continue;
            }
            for s in 0..=max_start {
                let distance = (s as i128 - expected).abs();
                if distance > options.max_offset as i128 {
                    continue;
                }
                let actual_start = s + l;
                let actual_end = s + hunk.old_count - r;
                if actual_end > source.len() || actual_end - actual_start != expected_rows.len() {
                    continue;
                }
                if !expected_rows.iter().enumerate().all(|(j, p)| source[actual_start + j] == *p) {
                    continue;
                }
                let rank = (l + r, distance);
                match best_rank {
                    None => {
                        best_rank = Some(rank);
                        starts.clear();
                        starts.push((s, l, r));
                    }
                    Some(old) if rank < old => {
                        best_rank = Some(rank);
                        starts.clear();
                        starts.push((s, l, r));
                    }
                    Some(old) if rank == old => starts.push((s, l, r)),
                    _ => {}
                }
            }
        }
    }

    if starts.is_empty() {
        return Err(ApplyError::NoMatch { hunk: number });
    }
    starts.sort_unstable_by_key(|&(s, l, _)| (s, l));
    starts.dedup_by_key(|entry| entry.0);
    if starts.len() > 1 {
        return Err(ApplyError::Ambiguous {
            hunk: number,
            first: starts[0].0,
            second: starts[1].0,
        });
    }
    let (s, l, r) = starts[0];

    let replacement: Vec<String> = hunk.rows[l..hunk.rows.len() - r]
        .iter()
        .filter(|row| row.kind == b' ' || row.kind == b'+')
        .map(|row| row.payload.clone())
        .collect();
    let replace_start = s + l;
    let replace_end = s + hunk.old_count - r;
    source.splice(replace_start..replace_end, replacement);

    if source.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    Ok(s as i128 - (hunk.old_start as i128 + net))
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }

    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1].split('\n').map(str::to_owned).collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;
    for (index, hunk) in hunks.iter().enumerate() {
        let previous_net = net;
        drift = apply_hunk(&mut working, hunk, index + 1, &options, net, drift)?;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        let _ = previous_net;
    }

    let mut result = String::new();
    for line in working {
        result.push_str(&line);
        result.push('\n');
    }
    Ok(result)
}

    pub fuzz: usize,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Row {
    kind: u8,
    payload: String,
    line: usize,
}

#[derive(Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(text: &str) -> Option<usize> {
    if text.is_empty()
        || !text.bytes().all(|b| b.is_ascii_digit())
        || (text.len() > 1 && text.starts_with('0'))
    {
        return None;
    }
    let mut value = 0usize;
    for byte in text.bytes() {
        value = value.checked_mul(10)?.checked_add((byte - b'0') as usize)?;
        if value > 128 {
            return None;
        }
    }
    Some(value)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let inner = line
        .strip_prefix("@@ -")
        .and_then(|s| s.strip_suffix(" @@"))
        .ok_or_else(bad)?;
    let (old, new) = inner.split_once(" +").ok_or_else(bad)?;
    if new.contains(' ') || old.contains(' ') {
        return Err(bad());
    }
    let (a, b) = old.split_once(',').ok_or_else(bad)?;
    let (c, d) = new.split_once(',').ok_or_else(bad)?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return Err(bad());
    }
    let a = parse_number(a).ok_or_else(bad)?;
    let b = parse_number(b).ok_or_else(bad)?;
    let c = parse_number(c).ok_or_else(bad)?;
    let d = parse_number(d).ok_or_else(bad)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    let old_start = if b > 0 { a - 1 } else { a };
    let new_start = if d > 0 { c - 1 } else { c };
    Ok((old_start, b, new_start, d))
}

fn physical_lines(text: &str) -> Vec<&str> {
    text.split_terminator('\n').collect()
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for (idx, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(idx).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: idx + 1 });
        }
    }

    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_end = 0usize;
    let mut cumulative_delta: i64 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let (old_start, old_count, new_start, new_count) =
            parse_hunk_header(lines[i], header_line)?;
        i += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let line = lines[i];
            let bytes = line.as_bytes();
            if bytes.is_empty()
                || !matches!(bytes[0], b' ' | b'-' | b'+')
                || bytes[1..].contains(&b'\r')
            {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let kind = bytes[0];
            let payload = &line[1..];
            match kind {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                }
                b'-' => old_seen += 1,
                b'+' => new_seen += 1,
                _ => unreachable!(),
            }
            rows.push(Row {
                kind,
                payload: payload.to_owned(),
                line: row_line,
            });
            i += 1;
        }

        if old_seen != old_count || new_seen != new_count {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_end = old_start + old_count;
        let required_new = (old_start as i64)
            .checked_add(cumulative_delta)
            .unwrap_or(i64::MIN);
        if old_start < previous_end || required_new < 0 || new_start as i64 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_end = old_end;
        cumulative_delta += new_count as i64 - old_count as i64;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count,
            new_count,
            rows,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let lines: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source
            .strip_suffix('\n')
            .unwrap_or(source)
            .split('\n')
            .map(str::to_owned)
            .collect()
    };
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i64 = 0;
    let mut drift: i64 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let leading = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
        let trailing = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
        let max_l = options.fuzz.min(leading);
        let max_r = options.fuzz.min(trailing);

        let expected = hunk.old_start as i64 + net + drift;
        let mut best_rank: Option<(usize, i64)> = None;
        let mut best: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    if row.kind == b' ' || row.kind == b'-' {
                        old_payloads.push(row.payload.as_str());
                    }
                }
                let required_len = hunk.old_count - l - r;
                if old_payloads.len() != required_len {
                    continue;
                }

                for s in 0..=working.len() {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (s as i64 - expected).abs();
                    if distance > options.max_offset as i64 {
                        continue;
                    }
                    if old_payloads
                        .iter()
                        .enumerate()
                        .all(|(k, payload)| working[s + l + k] == *payload)
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(current) if rank < current => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(current) if rank == current => best.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
            }
        }

        if best.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }

        let mut starts: Vec<usize> = best.iter().map(|item| item.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        let start = starts[0];
        let (_, l, r) = best
            .iter()
            .filter(|item| item.0 == start)
            .min_by_key(|item| item.1)
            .copied()
            .unwrap();

        let mut replacement = Vec::new();
        for row in &hunk.rows[l..hunk.rows.len() - r] {
            if row.kind == b' ' || row.kind == b'+' {
                replacement.push(row.payload.clone());
            }
        }

        let replace_start = start + l;
        let replace_end = start + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i64 - hunk.old_count as i64;
        drift = start as i64 - (hunk.old_start as i64 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
