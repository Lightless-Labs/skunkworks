#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct SearchResult {
    best: Option<Vec<Selection>>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    attempts: usize,
    max_attempts: usize,
    best: Option<Vec<Selection>>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn package(&self, name: &str) -> Option<&Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn lock_version(&self, name: &str) -> Option<Version> {
        self.locks
            .iter()
            .find(|l| l.package == name)
            .map(|l| l.version)
    }

    fn inspect(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();

        for root in self.roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in selected.iter() {
            if let Some(package) = self.package(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }

        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }

        if let Some(name) = active.keys().find(|name| self.package(name).is_none()) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints
                    .iter()
                    .all(|c| constraint_matches(*c, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.package(name).expect("missing checked above");
            if !package.releases.iter().any(|r| {
                constraints
                    .iter()
                    .all(|c| constraint_matches(*c, r.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        if let Some(name) = next {
            let constraints = active.get(&name).cloned().unwrap_or_default();
            let package = self.package(&name).expect("missing checked above");
            let mut candidates: Vec<Version> = package
                .releases
                .iter()
                .map(|r| r.version)
                .filter(|v| constraints.iter().all(|c| constraint_matches(*c, *v)))
                .collect();
            candidates.sort_by(|a, b| b.cmp(a));

            if let Some(locked) = self.lock_version(&name) {
                if let Some(pos) = candidates.iter().position(|v| *v == locked) {
                    let value = candidates.remove(pos);
                    candidates.insert(0, value);
                }
            }

            for candidate in candidates {
                if self.attempts == self.max_attempts {
                    return Err(ResolveError::Limit {
                        attempts: self.max_attempts,
                    });
                }
                self.attempts += 1;
                selected.insert(name.clone(), candidate);
                let result = self.inspect(selected);
                selected.remove(&name);

                match result {
                    Ok(()) => {}
                    Err(ResolveError::Limit { attempts }) => {
                        return Err(ResolveError::Limit { attempts });
                    }
                    Err(error) => {
                        if self.first_error.is_none() {
                            self.first_error = Some(error);
                        }
                    }
                }
            }
            return Ok(());
        }

        let solution: Vec<Selection> = selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();

        let cost = self
            .locks
            .iter()
            .filter(|lock| {
                !solution
                    .iter()
                    .any(|s| s.package == lock.package && s.version == lock.version)
            })
            .count();

        let replace = match &self.best {
            None => true,
            Some(best) if cost < self.best_cost => true,
            Some(best) if cost > self.best_cost => false,
            Some(best) => {
                let names: std::collections::BTreeSet<&str> = self
                    .catalog
                    .iter()
                    .map(|p| p.name.as_str())
                    .collect();
                let current_map: std::collections::BTreeMap<&str, Version> = solution
                    .iter()
                    .map(|s| (s.package.as_str(), s.version))
                    .collect();
                let best_map: std::collections::BTreeMap<&str, Version> = best
                    .iter()
                    .map(|s| (s.package.as_str(), s.version))
                    .collect();
                let mut ordering = std::cmp::Ordering::Equal;
                for name in names {
                    ordering = current_map
                        .get(name)
                        .cmp(&best_map.get(name));
                    if ordering != std::cmp::Ordering::Equal {
                        break;
                    }
                }
                ordering == std::cmp::Ordering::Greater
            }
        };

        if replace {
            self.best = Some(solution);
            self.best_cost = cost;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks,
        attempts: 0,
        max_attempts,
        best: None,
        best_cost: usize::MAX,
        first_error: None,
    };

    let mut selected = std::collections::BTreeMap::new();
    if let Err(error) = search.inspect(&mut selected) {
        return Err(error);
    }

    let selected = match search.best {
        Some(solution) => solution,
        None => {
            return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
                package: String::new(),
                requirements: Vec::new(),
            }));
        }
    };

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for selection in &selected {
        if let Some(package) = catalog.iter().find(|p| p.name == selection.package) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|r| r.version == selection.version)
            {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }

    let mut changes = Vec::new();
    let names: std::collections::BTreeSet<String> = locks
        .iter()
        .map(|l| l.package.clone())
        .chain(selected.iter().map(|s| s.package.clone()))
        .collect();

    for name in names {
        let before = locks.iter().find(|l| l.package == name).map(|l| l.version);
        let after = selected
            .iter()
            .find(|s| s.package == name)
            .map(|s| s.version);
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == name)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if !active.get(&name).unwrap_or(&Vec::new()).iter().all(|c| {
                    constraint_matches(*c, old)
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
