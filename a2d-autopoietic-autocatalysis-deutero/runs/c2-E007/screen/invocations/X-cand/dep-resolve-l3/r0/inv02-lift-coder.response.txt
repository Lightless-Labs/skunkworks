#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_rank(*c));
    cs.dedup();
    cs
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical(cs.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn active_constraints(
    roots: &[Requirement],
    selected: &[(String, Release)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for r in roots {
        active
            .entry(r.package.clone())
            .or_default()
            .push(r.constraint);
    }
    for (_, release) in selected {
        for r in &release.deps {
            active
                .entry(r.package.clone())
                .or_default()
                .push(r.constraint);
        }
    }
    for cs in active.values_mut() {
        *cs = canonical(std::mem::take(cs));
    }
    active
}

fn release_for<'a>(pkg: &'a Package, version: Version) -> Option<&'a Release> {
    pkg.releases.iter().find(|r| r.version == version)
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<Vec<(String, Release)>>,
    best_cost: usize,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn record_failure(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn solution_better(&self, candidate: &[(String, Release)], cost: usize) -> bool {
        let Some(best) = &self.best else {
            return true;
        };
        if cost != self.best_cost {
            return cost < self.best_cost;
        }

        let left: std::collections::BTreeMap<_, _> = candidate
            .iter()
            .map(|(n, r)| (n.as_str(), r.version))
            .collect();
        let right: std::collections::BTreeMap<_, _> =
            best.iter().map(|(n, r)| (n.as_str(), r.version)).collect();

        for name in &self.catalog_names {
            match (left.get(name.as_str()), right.get(name.as_str())) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn visit(&mut self, selected: &mut Vec<(String, Release)>) -> bool {
        let active = active_constraints(self.roots, selected);

        for name in active.keys() {
            if !self.catalog.iter().any(|p| p.name == *name) {
                self.record_failure(ResolveError::Missing {
                    package: name.clone(),
                });
                return true;
            }
        }

        let assigned: std::collections::BTreeMap<_, _> = selected
            .iter()
            .map(|(n, r)| (n.as_str(), r.version))
            .collect();

        for (name, constraints) in &active {
            if let Some(version) = assigned.get(name.as_str()) {
                if !constraints.iter().all(|c| matches(**version, *c)) {
                    self.record_failure(ResolveError::Conflict {
                        package: name.clone(),
                        selected: **version,
                        requirements: constraints.clone(),
                    });
                    return true;
                }
            }
        }

        for (name, constraints) in &active {
            if assigned.contains_key(name.as_str()) {
                continue;
            }
            let pkg = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !pkg
                .releases
                .iter()
                .any(|r| constraints.iter().all(|c| matches(r.version, *c)))
            {
                self.record_failure(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
                return true;
            }
        }

        let next = active
            .keys()
            .find(|name| !assigned.contains_key(name.as_str()))
            .cloned();

        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|lock| {
                    !selected
                        .iter()
                        .any(|(n, r)| n == &lock.package && r.version == lock.version)
                })
                .count();
            if self.solution_better(selected, cost) {
                self.best = Some(selected.clone());
                self.best_cost = cost;
            }
            return false;
        };

        let constraints = &active[&name];
        let pkg = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<Release> = pkg
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
            .cloned()
            .collect();
        candidates.sort_by(|a, b| b.version.cmp(&a.version));

        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(index) = candidates.iter().position(|r| r.version == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return true;
            }
            self.attempts += 1;
            selected.push((name.clone(), candidate));
            let stop = self.visit(selected);
            selected.pop();

            if stop {
                if self.attempts == self.max_attempts && self.best.is_none() {
                    return true;
                }
                // A Limit is represented by the budget being exhausted when
                // an additional assignment would be required; visit reports it
                // through this condition below.
                if self.attempts == self.max_attempts {
                    let active_after = active_constraints(self.roots, selected);
                    let has_unassigned = active_after.keys().any(|n| {
                        !selected.iter().any(|(s, _)| s == n)
                    });
                    if has_unassigned {
                        return true;
                    }
                }
            }
        }
        false
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        validate_name(format!("root[{i}]"), &r.package)?;
    }
    for (i, p) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{i}]"), &p.name)?;
        for (j, release) in p.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                validate_name(
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                    &dep.package,
                )?;
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for p in catalog {
        if !seen_packages.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &p.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        validate_name(format!("lock[{i}]"), &lock.package)?;
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
        catalog_names,
    };
    let mut selected = Vec::new();
    let stopped = search.visit(&mut selected);

    if stopped {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    if let Some(best) = search.best {
        let mut selected_out: Vec<Selection> = best
            .iter()
            .map(|(package, release)| Selection {
                package: package.clone(),
                version: release.version,
            })
            .collect();
        selected_out.sort_by(|a, b| a.package.cmp(&b.package));

        let final_constraints = active_constraints(roots, &best);
        let chosen: std::collections::BTreeMap<_, _> = best
            .iter()
            .map(|(n, r)| (n.as_str(), r.version))
            .collect();
        let lock_map: std::collections::BTreeMap<_, _> = locks
            .iter()
            .map(|l| (l.package.as_str(), l.version))
            .collect();

        let mut names = std::collections::BTreeSet::new();
        for lock in locks {
            names.insert(lock.package.as_str());
        }
        for (name, _) in &best {
            names.insert(name.as_str());
        }

        let mut changes = Vec::new();
        for name in names {
            let before = lock_map.get(name).copied();
            let after = chosen.get(name).copied();
            if before == after {
                continue;
            }
            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let exists = catalog
                        .iter()
                        .find(|p| p.name == name)
                        .is_some_and(|p| release_for(p, old).is_some());
                    if !exists {
                        ChangeReason::Unavailable
                    } else if !final_constraints
                        .get(name)
                        .map(|cs| cs.iter().all(|c| matches(old, *c)))
                        .unwrap_or(true)
                    {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
                (None, None) => continue,
            };
            changes.push(Change {
                package: name.to_string(),
                before,
                after,
                reason,
            });
        }
        changes.sort_by(|a, b| a.package.cmp(&b.package));
        Ok(Resolution {
            selected: selected_out,
            changes,
        })
    } else {
        Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }))
    }
}
