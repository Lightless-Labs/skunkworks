use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<(Vec<Point>, Vec<usize>), GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let len = points.len()
        - usize::from(points.len() >= 2 && points[0] == points[points.len() - 1]);
    if len < 3 {
        return Err(GeomError::TooFewVertices { len });
    }

    Ok((points[..len].to_vec(), (0..len).collect()))
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0_i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn integer_sqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = 1_u128 << ((128 - n.leading_zeros() as usize + 1) / 2);
    loop {
        let y = (x + n / x) / 2;
        if y >= x {
            return x;
        }
        x = y;
    }
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128 - a.x as i128).unsigned_abs();
    let dy = (b.y as i128 - a.y as i128).unsigned_abs();
    let squared_scaled = (dx * dx + dy * dy) * 1_000_000;
    let root = integer_sqrt(squared_scaled);
    let rounded = if squared_scaled - root * root >= root + 1 {
        root + 1
    } else {
        root
    };
    rounded as u64
}

fn measures(points: &[Point]) -> Measures {
    let signed_area2 = shoelace(points);
    let edge_milli: Vec<u64> = (0..points.len())
        .map(|i| rounded_edge_milli(points[i], points[(i + 1) % points.len()]))
        .collect();
    let perimeter_milli = edge_milli.iter().map(|&x| x as u128).sum::<u128>() as u64;

    let mut longest: Option<(usize, (Point, Point))> = None;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        let endpoints = if a <= b { (a, b) } else { (b, a) };

        match longest {
            None => longest = Some((i, endpoints)),
            Some((best_index, best_endpoints)) => {
                if edge_milli[i] > edge_milli[best_index]
                    || (edge_milli[i] == edge_milli[best_index]
                        && (endpoints < best_endpoints
                            || (endpoints == best_endpoints && i < best_index)))
                {
                    longest = Some((i, endpoints));
                }
            }
        }
    }

    Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation: match signed_area2.cmp(&0) {
            Ordering::Less => Orientation::Clockwise,
            Ordering::Equal => Orientation::Degenerate,
            Ordering::Greater => Orientation::CounterClockwise,
        },
        edge_milli,
        perimeter_milli,
        longest_edge: longest.map(|entry| entry.0),
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let (effective, _) = prepare(points)?;
    Ok(measures(&effective))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let (polygon, _) = prepare(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    Ok(queries
        .iter()
        .map(|&query| {
            for i in 0..polygon.len() {
                if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], query) {
                    return Location::Boundary { edge: i };
                }
            }

            let mut odd = false;
            for i in 0..polygon.len() {
                let a = polygon[i];
                let b = polygon[(i + 1) % polygon.len()];
                if (a.y > query.y) != (b.y > query.y) {
                    let side = cross(a, b, query);
                    if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                        odd = !odd;
                    }
                }
            }
            if odd {
                Location::Inside
            } else {
                Location::Outside
            }
        })
        .collect())
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(p, _)| p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|entry| entry.0);

    let output = if unique.len() <= 2 {
        unique
    } else {
        let collinear = unique[2..].iter().all(|entry| {
            cross(unique[0].0, unique[1].0, entry.0) == 0
        });

        if collinear {
            match mode {
                Collinear::Include => unique,
                Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
            }
        } else {
            let mut lower = Vec::new();
            for &point in &unique {
                while lower.len() >= 2
                    && cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, point.0) <= 0
                {
                    lower.pop();
                }
                lower.push(point);
            }

            let mut upper = Vec::new();
            for &point in unique.iter().rev() {
                while upper.len() >= 2
                    && cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, point.0) <= 0
                {
                    upper.pop();
                }
                upper.push(point);
            }
            lower.pop();
            upper.pop();
            let mut corners = lower;
            corners.extend(upper);

            if mode == Collinear::Exclude {
                corners
            } else {
                let mut expanded = Vec::new();
                for i in 0..corners.len() {
                    let a = corners[i].0;
                    let b = corners[(i + 1) % corners.len()].0;
                    let mut edge_points: Vec<_> = unique
                        .iter()
                        .copied()
                        .filter(|entry| on_segment(a, b, entry.0))
                        .collect();
                    edge_points.sort_by_key(|entry| {
                        let dx = b.x as i128 - a.x as i128;
                        let dy = b.y as i128 - a.y as i128;
                        (entry.0.x as i128 - a.x as i128) * dx
                            + (entry.0.y as i128 - a.y as i128) * dy
                    });
                    for point in edge_points {
                        if expanded.last().map_or(true, |last: &(Point, usize)| last.0 != point.0) {
                            expanded.push(point);
                        }
                    }
                }
                expanded
            }
        }
    };

    let vertices: Vec<Point> = output.iter().map(|entry| entry.0).collect();
    let source_indices: Vec<usize> = output.iter().map(|entry| entry.1).collect();
    let (area2, perimeter_milli) = match vertices.len() {
        0 => (0, 0),
        1 => (0, 0),
        2 => {
            let length = rounded_edge_milli(vertices[0], vertices[1]);
            (0, length * 2)
        }
        _ => {
            let m = measures(&vertices);
            (m.area2, m.perimeter_milli)
        }
    };

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);

    if (ab_c == 0 && on_segment(a, b, c))
        || (ab_d == 0 && on_segment(a, b, d))
        || (cd_a == 0 && on_segment(c, d, a))
        || (cd_b == 0 && on_segment(c, d, b))
    {
        return true;
    }
    (ab_c > 0) != (ab_d > 0) && (cd_a > 0) != (cd_b > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let (polygon, _) = prepare(points)?;
    let n = polygon.len();

    for i in 0..n {
        for j in i + 1..n {
            if polygon[i] == polygon[j] {
                return Err(GeomError::Duplicate {
                    first: i,
                    second: j,
                });
            }
        }
    }

    let area = shoelace(&polygon);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let previous = polygon[(i + n - 1) % n];
        let current = polygon[i];
        let next = polygon[(i + 1) % n];
        if cross(previous, current, next) == 0 {
            let dot = (previous.x as i128 - current.x as i128)
                * (next.x as i128 - current.x as i128)
                + (previous.y as i128 - current.y as i128)
                    * (next.y as i128 - current.y as i128);
            if dot > 0 {
                return Err(GeomError::AdjacentOverlap { vertex: i });
            }
        }
    }

    for i in 0..n {
        for j in i + 1..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                polygon[i],
                polygon[(i + 1) % n],
                polygon[j],
                polygon[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let start = (0..n).min_by_key(|&i| polygon[i]).unwrap();
    let ccw = area > 0;
    let mut vertices = Vec::with_capacity(n);
    let mut source_indices = Vec::with_capacity(n);
    for step in 0..n {
        let index = if ccw {
            (start + step) % n
        } else {
            (start + n - step) % n
        };
        vertices.push(polygon[index]);
        source_indices.push(index);
    }

    Ok(Validated {
        measures: measures(&vertices),
        vertices,
        source_indices,
    })
}
