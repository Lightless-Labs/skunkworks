#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct DayRange {
    pub start: u32,
    pub end: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Rounding {
    Down,
    Up,
    HalfEven,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Proration {
    pub active_days: u32,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Meter {

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
    Month,
    Period,
    Credit { index: usize },
    CreditOrder { index: usize },
    InvoiceDiscount,
    InvalidName { index: usize },
    DuplicateName { index: usize },
    Discount,
    CombinedOverflow,
    Meter { index: usize, source: Box<Error> },
    InvoiceOverflow,
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().and_then(|tier| tier.up_to).is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn checked_band(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    let amount = (units as u128) * (unit as u128) + (start as u128);
    if amount > u64::MAX as u128 {
        return Err(Error::Overflow { index });
    }
    Ok(amount as u64)
}

fn quote_validated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                previous = bound;
                remaining.min(width)
            }
            None => remaining,
        };

        if units > 0 {
            let amount = checked_band(units, tier.unit, tier.start, index)?;
            let next_total = (total as u128) + (amount as u128);
            if next_total > u64::MAX as u128 {
                return Err(Error::Overflow { index });
            }
            total = next_total as u64;
            bands.push(Band {
                index,
                units,
                amount,
            });
            remaining -= units;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}

fn volume_validated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let index = tiers
        .iter()
        .position(|tier| tier.up_to.is_none_or(|bound| bound >= quantity))
        .expect("validated tier schedule has an unbounded final tier");
    let tier = &tiers[index];
    let amount = checked_band(quantity, tier.unit, tier.start, index)?;

    Ok(Charge {
        amount,
        bands: vec![Band {
            index,
            units: quantity,
            amount,
        }],
    })
}

fn flat_charge(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }
    let amount = checked_band(quantity, unit, start, 0)?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index: 0,
            units: quantity,
            amount,
        }],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => flat_charge(quantity, *unit, *start),
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            quote_validated(quantity, tiers)
        }
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match price {
        Price::Flat { unit, start } => flat_charge(quantity, *unit, *start)?,
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            match mode {
                Mode::Graduated => quote_validated(quantity, tiers)?,
                Mode::Volume => volume_validated(quantity, tiers)?,
            }
        }
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = (charge.amount as u128) + (minimum_adjustment as u128);
    if total > u64::MAX as u128 {
        return Err(Error::Overflow {
            index: charge.bands.last().map_or(0, |band| band.index),
        });
    }

    Ok(Bill {
        charge,
        minimum_adjustment,
        total: total as u64,
    })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut previous_end = start;
    let mut credited_days = 0u32;

    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        previous_end = credit.end;
        credited_days += credit.end - credit.start;
    }

    let active_days = end - start - credited_days;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice_remainder = remainder * 2;
            if twice_remainder > denominator
                || (twice_remainder == denominator && quotient % 2 == 1)
            {
                quotient + 1
            } else {
                quotient
            }
        }
    };

    Ok(Proration {
        active_days,
        amount: rounded as u64,
    })
}

fn discount_amount(base: u64, bps: u32) -> u64 {
    (((base as u128) * (bps as u128)) / 10_000u128) as u64
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 32
        && !bytes
            .iter()
            .any(|&byte| byte <= 0x1f || byte == 0x7f || byte == b'|' || byte == b':')
}

pub fn invoice(
    meters: &[Meter],
    invoice_discount_bps: u32,
    credit: u64,
) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 {
        return Err(Error::InvoiceDiscount);
    }

    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) {
            return Err(Error::InvalidName { index });
        }
        if meters[..index]
            .iter()
            .any(|previous| previous.name == meter.name)
        {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;

    for (index, meter) in meters.iter().enumerate() {
        let result = (|| {
            if meter.discount_bps > 10_000 {
                return Err(Error::Discount);
            }

            let bill = bill(
                meter.quantity,
                &meter.price,
                meter.mode.clone(),
                meter.minimum,
            )?;
            let monthly = prorate(
                meter.monthly_fee,
                meter.days,
                meter.start,
                meter.end,
                &meter.credits,
                meter.rounding.clone(),
            )?;

            let base = (bill.total as u128) + (monthly.amount as u128);
            if base > u64::MAX as u128 {
                return Err(Error::CombinedOverflow);
            }
            let base = base as u64;
            let discount = discount_amount(base, meter.discount_bps);
            let total = base - discount;

            Ok(InvoiceLine {
                name: meter.name.clone(),
                usage: bill.charge.amount,
                minimum_adjustment: bill.minimum_adjustment,
                monthly: monthly.amount,
                discount,
                total,
            })
        })()
        .map_err(|source| Error::Meter {
            index,
            source: Box::new(source),
        })?;

        let next_subtotal = (subtotal as u128) + (result.total as u128);
        if next_subtotal > u64::MAX as u128 {
            return Err(Error::InvoiceOverflow);
        }
        subtotal = next_subtotal as u64;
        lines.push(result);
    }

    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));

    let discount = discount_amount(subtotal, invoice_discount_bps);
    let credit = credit.min(subtotal - discount);
    let total = subtotal - discount - credit;

    let mut rendered = String::new();
    for line in &lines {
        rendered.push_str(&line.name.len().to_string());
        rendered.push(':');
        rendered.push_str(&line.name);
        rendered.push('|');
        rendered.push_str(&line.usage.to_string());
        rendered.push('|');
        rendered.push_str(&line.minimum_adjustment.to_string());
        rendered.push('|');
        rendered.push_str(&line.monthly.to_string());
        rendered.push('|');
        rendered.push_str(&line.discount.to_string());
        rendered.push('|');
        rendered.push_str(&line.total.to_string());
        rendered.push('\n');
    }
    rendered.push_str("TOTAL|");
    rendered.push_str(&subtotal.to_string());
    rendered.push('|');
    rendered.push_str(&discount.to_string());
    rendered.push('|');
    rendered.push_str(&credit.to_string());
    rendered.push('|');
    rendered.push_str(&total.to_string());
    rendered.push('\n');

    Ok(Invoice {
        lines,
        subtotal,
        discount,
        credit,
        total,
        rendered,
    })
}
