#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
struct Specifier {
    name: String,
    modifier: Modifier,
    pos: usize,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Plus,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

fn name_char_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn name_char(b: u8) -> bool {
    name_char_start(b) || b.is_ascii_digit()
}

fn parse_specifier(s: &str, pos: usize) -> Result<Specifier, Error> {
    if s.is_empty() {
        return Err(Error::EmptyVariable { pos });
    }

    let bytes = s.as_bytes();
    let split = bytes
        .iter()
        .position(|&b| b == b'*' || b == b':')
        .unwrap_or(bytes.len());

    let name = &s[..split];
    if name.is_empty() {
        return Err(Error::InvalidName { pos });
    }

    let name_bytes = name.as_bytes();
    if !name_char_start(name_bytes[0]) {
        return Err(Error::InvalidName { pos });
    }
    for (i, &b) in name_bytes.iter().enumerate().skip(1) {
        if !name_char(b) {
            return Err(Error::InvalidName { pos: pos + i });
        }
    }

    let modifier = if split == bytes.len() {
        Modifier::None
    } else {
        let suffix = &s[split..];
        let modifier_pos = pos + split;
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let db = digits.as_bytes();
            if db.is_empty()
                || db.len() > 4
                || !db[0].is_ascii_digit()
                || db[0] == b'0'
                || db.iter().any(|b| !b.is_ascii_digit())
            {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            let mut n = 0usize;
            for &b in db {
                n = n * 10 + (b - b'0') as usize;
            }
            if n > 9999 {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Prefix(n)
        } else {
            return Err(Error::InvalidModifier { pos: modifier_pos });
        }
    };

    Ok(Specifier {
        name: name.to_string(),
        modifier,
        pos,
    })
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<(Operator, Vec<Specifier>), Error> {
    let mut operator = Operator::Simple;
    let mut start = 0;

    if let Some(&b) = content.as_bytes().first() {
        operator = match b {
            b'+' => {
                start = 1;
                Operator::Plus
            }
            b'#' => {
                start = 1;
                Operator::Fragment
            }
            b'.' => {
                start = 1;
                Operator::Dot
            }
            b'/' => {
                start = 1;
                Operator::Slash
            }
            b';' => {
                start = 1;
                Operator::Semicolon
            }
            b'?' => {
                start = 1;
                Operator::Query
            }
            b'&' => {
                start = 1;
                Operator::Continuation
            }
            b'!' | b'=' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator {
                    pos: content_pos,
                });
            }
            _ => Operator::Simple,
        };
    }

    if start == content.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let rest = &content[start..];
    let rest_pos = content_pos + start;
    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut offset = 0;

    for part in rest.split(',') {
        let spec = parse_specifier(part, rest_pos + offset)?;
        if seen.iter().any(|name| name == &spec.name) {
            return Err(Error::DuplicateVariable { pos: spec.pos });
        }
        seen.push(spec.name.clone());
        specs.push(spec);
        offset += part.len() + 1;
    }

    Ok((operator, specs))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(n: u8) -> u8 {
    if n < 10 { b'0' + n } else { b'A' + (n - 10) }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h1), Some(h2)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(hex_digit(h1) as char);
                out.push(hex_digit(h2) as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4) as char);
            out.push(hex_digit(b & 0x0f) as char);
        }
        i += 1;
    }

    out
}

fn prefix_chars(value: &str, n: usize) -> String {
    value.chars().take(n).collect()
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn expand_expression(operator: Operator, specs: &[Specifier], vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(operator, Operator::Plus | Operator::Fragment);
    let mut fields = Vec::<String>::new();

    for spec in specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(s) => {
                let source = match spec.modifier {
                    Modifier::Prefix(n) => prefix_chars(s, n),
                    _ => s.clone(),
                };
                let encoded = encode(&source, reserved);
                fields.push(format_scalar_field(operator, &spec.name, encoded));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(format_scalar_field(operator, &spec.name, encoded));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(format_scalar_field(operator, &spec.name, joined));
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                for i in 0..entries.len() {
                    for j in 0..i {
                        if entries[i].0 == entries[j].0 {
                            return Err(Error::DuplicateKey { pos: spec.pos });
                        }
                    }
                }

                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, val) in entries {
                        let k = encode(key, reserved);
                        let v = encode(val, reserved);
                        if matches!(operator, Operator::Semicolon) && v.is_empty() {
                            fields.push(k);
                        } else {
                            fields.push(format!("{k}={v}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(val, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(format_scalar_field(operator, &spec.name, joined));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

fn format_scalar_field(operator: Operator, name: &str, encoded: String) -> String {
    match operator {
        Operator::Semicolon => {
            if encoded.is_empty() {
                name.to_string()
            } else {
                format!("{name}={encoded}")
            }
        }
        Operator::Query | Operator::Continuation => format!("{name}={encoded}"),
        _ => encoded,
    }
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;
    let mut literal_start = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                output.push_str(&template[literal_start..cursor]);
                let open = cursor;
                let mut close = cursor + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }
                if let Some(relative) = bytes[cursor + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + relative,
                    });
                }

                let content = &template[cursor + 1..close];
                let (operator, specs) = parse_expression(content, cursor + 1, close)?;
                output.push_str(&expand_expression(operator, &specs, vars)?);
                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
