#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Query),
            b'&' => Some(Self::Continuation),
            _ => None,
        }
    }

    fn separator(self) -> char {
        match self {
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Query | Self::Continuation => '&',
            Self::Simple | Self::Reserved | Self::Fragment => ',',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continuation => Some('&'),
            Self::Simple | Self::Reserved => None,
        }
    }

    fn is_named(self) -> bool {
        matches!(
            self,
            Self::Semicolon | Self::Query | Self::Continuation
        )
    }

    fn is_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

#[derive(Debug, Clone)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone)]
struct Specifier<'a> {
    name: &'a str,
    pos: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone)]
struct Expression<'a> {
    operator: Operator,
    specifiers: Vec<Specifier<'a>>,
}

#[derive(Debug, Clone)]
enum Part<'a> {
    Literal(&'a str),
    Expression(Expression<'a>),
}

fn valid_name_byte(byte: u8, first: bool) -> bool {
    if first {
        byte.is_ascii_alphabetic() || byte == b'_'
    } else {
        byte.is_ascii_alphanumeric() || byte == b'_'
    }
}

fn parse_specifier<'a>(text: &'a str, pos: usize) -> Result<Specifier<'a>, Error> {
    let bytes = text.as_bytes();

    let modifier_index = bytes
        .iter()
        .position(|&byte| byte == b'*' || byte == b':')
        .unwrap_or(bytes.len());

    if modifier_index == 0 {
        return Err(Error::InvalidName { pos });
    }

    for (index, &byte) in bytes[..modifier_index].iter().enumerate() {
        if !valid_name_byte(byte, index == 0) {
            return Err(Error::InvalidName { pos: pos + index });
        }
    }

    let modifier = if modifier_index == bytes.len() {
        Modifier::None
    } else {
        let modifier_pos = pos + modifier_index;
        let suffix = &bytes[modifier_index..];

        if suffix == b"*" {
            Modifier::Explode
        } else if suffix.first() == Some(&b':') {
            let digits = &suffix[1..];
            if digits.is_empty()
                || digits.len() > 4
                || digits[0] == b'0'
                || !digits.iter().all(u8::is_ascii_digit)
            {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }

            let mut value = 0usize;
            for &digit in digits {
                value = value * 10 + usize::from(digit - b'0');
            }

            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: modifier_pos });
        }
    };

    Ok(Specifier {
        name: &text[..modifier_index],
        pos,
        modifier,
    })
}

fn parse_expression<'a>(
    content: &'a str,
    content_pos: usize,
    closing_pos: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = content.as_bytes();
    let mut operator = Operator::Simple;
    let mut body_start = 0;

    if let Some(&first) = bytes.first() {
        if let Some(found) = Operator::from_byte(first) {
            operator = found;
            body_start = 1;
        } else if matches!(first, b'=' | b',' | b'!') {
            return Err(Error::UnsupportedOperator {
                pos: content_pos,
            });
        }
    }

    if body_start == bytes.len() {
        return Err(Error::EmptyExpression { pos: closing_pos });
    }

    let body = &content[body_start..];
    let body_pos = content_pos + body_start;
    let body_bytes = body.as_bytes();
    let mut specifiers = Vec::new();
    let mut start = 0usize;
    let mut seen = Vec::<&str>::new();

    for index in 0..=body_bytes.len() {
        if index == body_bytes.len() || body_bytes[index] == b',' {
            if index == start {
                return Err(Error::EmptyVariable {
                    pos: body_pos + start,
                });
            }

            let spec = parse_specifier(&body[start..index], body_pos + start)?;
            if seen.iter().any(|name| *name == spec.name) {
                return Err(Error::DuplicateVariable { pos: spec.pos });
            }
            seen.push(spec.name);
            specifiers.push(spec);
            start = index + 1;
        }
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn parse_template(template: &str) -> Result<Vec<Part<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut cursor = 0usize;
    let mut literal_start = 0usize;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                if literal_start < cursor {
                    parts.push(Part::Literal(&template[literal_start..cursor]));
                }

                let opening = cursor;
                let mut closing = None;
                let mut inner_opening = None;
                let mut scan = cursor + 1;

                while scan < bytes.len() {
                    match bytes[scan] {
                        b'{' if inner_opening.is_none() => inner_opening = Some(scan),
                        b'}' => {
                            closing = Some(scan);
                            break;
                        }
                        _ => {}
                    }
                    scan += 1;
                }

                let Some(closing_pos) = closing else {
                    return Err(Error::Unclosed { pos: opening });
                };

                if let Some(pos) = inner_opening {
                    return Err(Error::NestedOpen { pos });
                }

                let content = &template[opening + 1..closing_pos];
                let expression =
                    parse_expression(content, opening + 1, closing_pos)?;
                parts.push(Part::Expression(expression));

                cursor = closing_pos + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    if literal_start < template.len() {
        parts.push(Part::Literal(&template[literal_start..]));
    }

    Ok(parts)
}

fn lookup<'a>(vars: &'a [(&str, Value)], name: &str) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0usize;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(char::from(byte));
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => char::from(b'0' + value),
        _ => char::from(b'A' + value - 10),
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode_named(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + value.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&value);
        field
    }
}

fn expand_expression(
    expression: &Expression<'_>,
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let operator = expression.operator;
    let reserved = operator.is_reserved();
    let mut fields = Vec::<String>::new();

    for specifier in &expression.specifiers {
        let Some(value) = lookup(vars, specifier.name) else {
            continue;
        };

        if matches!(specifier.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: specifier.pos });
        }

        let explode = matches!(specifier.modifier, Modifier::Explode);

        match value {
            Value::Scalar(scalar) => {
                let selected = match specifier.modifier {
                    Modifier::Prefix(count) => scalar.chars().take(count).collect::<String>(),
                    Modifier::None | Modifier::Explode => scalar.clone(),
                };
                let encoded = encode(&selected, reserved);
                if operator.is_named() {
                    fields.push(encode_named(specifier.name, encoded, operator));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }

                if explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        if operator.is_named() {
                            fields.push(encode_named(specifier.name, encoded, operator));
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if operator.is_named() {
                        fields.push(encode_named(specifier.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|seen| *seen == key) {
                        return Err(Error::DuplicateKey { pos: specifier.pos });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let mut field = encoded_key;
                        if operator == Operator::Semicolon && encoded_value.is_empty() {
                            // Semicolon exploded maps still omit equals for empty values.
                        } else {
                            field.push('=');
                            field.push_str(&encoded_value);
                        }
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    if operator.is_named() {
                        fields.push(encode_named(specifier.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let parts = parse_template(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(literal) => output.push_str(literal),
            Part::Expression(expression) => {
                output.push_str(&expand_expression(&expression, vars)?);
            }
        }
    }

    Ok(output)
}
