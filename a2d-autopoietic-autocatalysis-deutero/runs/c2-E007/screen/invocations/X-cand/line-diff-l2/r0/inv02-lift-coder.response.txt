#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }

    text.strip_suffix('\n')
        .unwrap()
        .split('\n')
        .collect()
}

fn validate<'a>(
    old: &'a str,
    new: &'a str,
    context: usize,
) -> Result<(Vec<&'a str>, Vec<&'a str>), DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }
    if let Some(byte) = old.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::Old,
            byte,
        });
    }
    if let Some(byte) = new.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::New,
            byte,
        });
    }
    if !old.is_empty() && !old.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);
    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    Ok((old_lines, new_lines))
}

fn make_script(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let rows = old.len() + 1;
    let cols = new.len() + 1;
    let mut lcs = vec![vec![0usize; cols]; rows];

    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut script = Vec::new();
    let (mut i, mut j) = (0, 0);

    while i < old.len() && j < new.len() {
        if old[i] == new[j] {
            script.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let delete = if delete_score > insert_score {
                true
            } else if delete_score < insert_score {
                false
            } else {
                (i + j) % 2 == 0
            };

            if delete {
                script.push(Edit::Delete {
                    old_index: i,
                    line: old[i].to_owned(),
                });
                i += 1;
            } else {
                script.push(Edit::Insert {
                    new_index: j,
                    line: new[j].to_owned(),
                });
                j += 1;
            }
        }
    }

    while i < old.len() {
        script.push(Edit::Delete {
            old_index: i,
            line: old[i].to_owned(),
        });
        i += 1;
    }
    while j < new.len() {
        script.push(Edit::Insert {
            new_index: j,
            line: new[j].to_owned(),
        });
        j += 1;
    }

    script
}

fn is_change(edit: &Edit) -> bool {
    matches!(edit, Edit::Delete { .. } | Edit::Insert { .. })
}

fn hunk_ranges(script: &[Edit], context: usize) -> Vec<(usize, usize)> {
    let mut groups = Vec::<(usize, usize)>::new();
    let mut i = 0;

    while i < script.len() {
        if !is_change(&script[i]) {
            i += 1;
            continue;
        }

        let start = i;
        while i + 1 < script.len() && is_change(&script[i + 1]) {
            i += 1;
        }
        groups.push((start, i + 1));
        i += 1;
    }

    let merge_limit = 2 * context;
    let mut merged = Vec::<(usize, usize)>::new();
    for group in groups {
        if let Some(last) = merged.last_mut() {
            let keeps_between = group.0 - last.1;
            if keeps_between <= merge_limit {
                last.1 = group.1;
                continue;
            }
        }
        merged.push(group);
    }

    merged
        .into_iter()
        .map(|(start, end)| {
            let mut expanded_start = start;
            let mut preceding = 0;
            while expanded_start > 0
                && preceding < context
                && matches!(script[expanded_start - 1], Edit::Keep { .. })
            {
                expanded_start -= 1;
                preceding += 1;
            }

            let mut expanded_end = end;
            let mut following = 0;
            while expanded_end < script.len()
                && following < context
                && matches!(script[expanded_end], Edit::Keep { .. })
            {
                expanded_end += 1;
                following += 1;
            }

            (expanded_start, expanded_end)
        })
        .collect()
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    let (old_lines, new_lines) = validate(old, new, context)?;
    let script = make_script(&old_lines, &new_lines);

    if !script.iter().any(is_change) {
        return Ok(String::new());
    }

    let ranges = hunk_ranges(&script, context);
    let mut output = String::from("--- before\n+++ after\n");

    for (start, end) in ranges {
        let mut old_count = 0usize;
        let mut new_count = 0usize;

        for edit in &script[start..end] {
            match edit {
                Edit::Keep { .. } => {
                    old_count += 1;
                    new_count += 1;
                }
                Edit::Delete { .. } => old_count += 1,
                Edit::Insert { .. } => new_count += 1,
            }
        }

        let old_before = script[..start]
            .iter()
            .filter(|edit| !matches!(edit, Edit::Insert { .. }))
            .count();
        let new_before = script[..start]
            .iter()
            .filter(|edit| !matches!(edit, Edit::Delete { .. }))
            .count();

        let old_start = if old_count > 0 {
            old_before + 1
        } else {
            old_before
        };
        let new_start = if new_count > 0 {
            new_before + 1
        } else {
            new_before
        };

        output.push_str(&format!(
            "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
        ));

        for edit in &script[start..end] {
            match edit {
                Edit::Keep { line, .. } => {
                    output.push(' ');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Delete { line, .. } => {
                    output.push('-');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Insert { line, .. } => {
                    output.push('+');
                    output.push_str(line);
                    output.push('\n');
                }
            }
        }
    }

    Ok(output)
}
