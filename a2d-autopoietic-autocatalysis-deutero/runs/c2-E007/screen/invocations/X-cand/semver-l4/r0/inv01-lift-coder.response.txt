use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(match bytes.get(*pos) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: *pos },
            _ => VersionError::InvalidChar { at: *pos },
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut p = start;

    loop {
        let id_start = p;
        while p < end {
            let b = bytes[p];
            if b.is_ascii_alphanumeric() || b == b'-' {
                p += 1;
            } else if b == b'.' {
                break;
            } else {
                return Err(VersionError::InvalidChar { at: p });
            }
        }

        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let id = &s[id_start..p];
        if prerelease && id.bytes().all(is_digit) && id.len() > 1 && id.starts_with('0') {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_owned());

        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(match s.as_bytes().get(pos) {
                None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }
        pos += 1;
        if expected == 0 {
            let minor = parse_core_number(s, &mut pos)?;
            if expected == 0 {
                // Keep the value while parsing the final field.
                let patch;
                if pos >= s.len() || s.as_bytes()[pos] != b'.' {
                    return Err(match s.as_bytes().get(pos) {
                        None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                        _ => VersionError::InvalidChar { at: pos },
                    });
                }
                pos += 1;
                patch = parse_core_number(s, &mut pos)?;
                return parse_version_tail(s, pos, major, minor, patch);
            }
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if pos == bytes.len() {
        return Ok(Version { major, minor, patch, pre: Vec::new(), build: Vec::new() });
    }
    if bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if bytes[pos] == b'-' {
        pos += 1;
        let build_at = bytes[pos..].iter().position(|&b| b == b'+').map(|i| pos + i);
        let pre_end = build_at.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pos, pre_end, true)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, pos, bytes.len(), false)?;
    }

    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {

    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Bound {
    version: Version,
    inclusive: bool,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<u64>,
    count: usize,
    full: Option<Version>,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && whitespace(bytes[p]) {
            p += 1;
        }
        if p == bytes.len() {
            break;
        }
        let start = p;
        while p < bytes.len() && !whitespace(bytes[p]) {
            p += 1;
        }
        out.push((start, &s[start..p]));
    }
    out
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: *fields.first().unwrap_or(&0),
        minor: *fields.get(1).unwrap_or(&0),
        patch: *fields.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn next_bound(mut v: Version, count: usize, caret: bool, at: usize) -> Result<Version, RangeError> {
    let bump_major = if caret {
        v.major > 0 || count == 1
    } else {
        count == 1
    };
    let bump_minor = if caret {
        !bump_major && (v.minor > 0 || count == 2)
    } else {
        !bump_major
    };

    if bump_major {
        v.major = v.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
        v.minor = 0;
        v.patch = 0;
    } else if bump_minor {
        v.minor = v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
        v.patch = 0;
    } else {
        v.patch = v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    }
    v.pre = vec!["0".into()];
    v.build.clear();
    Ok(v)
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        let v = parse_version(s).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            fields: vec![v.major, v.minor, v.patch],
            count: 3,
            full: Some(v),
        });
    }

    if fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }

    let mut nums = Vec::new();
    let mut wild_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wild_seen = true;
            continue;
        }
        if wild_seen || f.bytes().any(|b| !is_digit(b)) || (f.len() > 1 && f.starts_with('0')) {
            return Err(RangeError::Syntax { at });
        }
        let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
        nums.push(n);
    }
    let count = nums.len();
    Ok(Partial { fields: nums, count, full: None })
}

fn add_check(arm: &mut Arm, op: Op, version: Version) {
    arm.checks.push(Check { op, version });
    if !arm.checks.last().unwrap().version.pre.is_empty() {
        let v = &arm.checks.last().unwrap().version;
        arm.anchors.push((v.major, v.minor, v.patch));
    }
}

fn parse_arm(s: &str, offset: usize) -> Result<Arm, RangeError> {
    let tokens = split_tokens(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: offset });
    }
    let standalone_minus = tokens.iter().find(|(_, t)| *t == "-").copied();
    let mut arm = Arm { checks: Vec::new(), anchors: Vec::new() };

    if let Some((minus_at, _)) = standalone_minus {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: offset + minus_at });
        }
        let (lo_at, lo_s) = tokens[0];
        let (hi_at, hi_s) = tokens[2];
        let lo = parse_partial(lo_s, offset + lo_at)?;
        let hi = parse_partial(hi_s, offset + hi_at)?;
        if lo.count == 0 {
            return Err(RangeError::Syntax { at: offset + lo_at });
        }
        if hi.count == 0 {
            return Err(RangeError::Syntax { at: offset + hi_at });
        }
        let low_v = lo.full.clone().unwrap_or_else(|| padded(&lo.fields));
        add_check(&mut arm, Op::Ge, low_v);
        if let Some(v) = hi.full {
            add_check(&mut arm, Op::Le, v);
        } else {
            let upper = next_bound(padded(&hi.fields), hi.count, false, offset + hi_at)?;
            add_check(&mut arm, Op::Lt, upper);
        }
        return Ok(arm);
    }

    for (tok_at, token) in tokens {
        let tok_start = offset + tok_at;
        if token == "," || token == "(" || token == ")" || token == "&" {
            return Err(RangeError::Syntax { at: tok_start });
        }
        let bytes = token.as_bytes();
        if bytes.first().is_some_and(|b| matches!(b, b'<' | b'>' | b'=')) {
            let (op, n) = if token.starts_with("<=") {
                (Op::Le, 2)
            } else if token.starts_with(">=") {
                (Op::Ge, 2)
            } else if token.starts_with('<') {
                (Op::Lt, 1)
            } else if token.starts_with('>') {
                (Op::Gt, 1)
            } else {
                (Op::Eq, 1)
            };
            let operand = &token[n..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: tok_start });
            }
            let v = parse_version(operand).map_err(|error| RangeError::Version {
                at: tok_start + n,
                error,
            })?;
            add_check(&mut arm, op, v);
            continue;
        }

        if token.starts_with('^') || token.starts_with('~') {
            let caret = token.starts_with('^');
            let operand = &token[1..];
            let operand_at = tok_start + 1;
            let p = parse_partial(operand, operand_at)?;
            if p.count == 0 {
                return Err(RangeError::Syntax { at: tok_start });
            }
            let low = p.full.clone().unwrap_or_else(|| padded(&p.fields));
            let high = next_bound(padded(&p.fields), p.count, caret, operand_at)?;
            add_check(&mut arm, Op::Ge, low);
            add_check(&mut arm, Op::Lt, high);
            continue;
        }

        if token.starts_with('*') || token.starts_with('x') || token.starts_with('X')
            || bytes.first().is_some_and(|b| b.is_ascii_digit())
        {
            let p = parse_partial(token, tok_start)?;
            if let Some(v) = p.full {
                add_check(&mut arm, Op::Eq, v);
            } else if p.count == 0 {
                // All-wildcard bare operand adds no restrictions.
            } else if p.count == 1 || p.count == 2 {
                add_check(&mut arm, Op::Ge, padded(&p.fields));
                let hi = next_bound(padded(&p.fields), p.count, false, tok_start)?;
                add_check(&mut arm, Op::Lt, hi);
            } else {
                add_check(&mut arm, Op::Eq, padded(&p.fields));
            }
            continue;
        }

        return Err(RangeError::Syntax { at: tok_start });
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let tokens = split_tokens(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut sep_positions = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        if !whitespace(bytes[p]) {
            p += 1;
            continue;
        }
        while p < bytes.len() && whitespace(bytes[p]) {
            p += 1;
        }
        let ws_end = p;
        let ws_start = {
            let mut q = ws_end;
            while q > 0 && whitespace(bytes[q - 1]) {
                q -= 1;
            }
            q
        };
        if s[ws_end..].starts_with("||") {
            let after = ws_end + 2;
            if after == bytes.len() || whitespace(bytes[after]) {
                sep_positions.push((ws_start, after));
                p = after;
                continue;
            }
        }
    }

    if sep_positions.is_empty() {
        return Ok(Range { arms: vec![parse_arm(s, 0)?] });
    }

    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    for (sep_start, after) in sep_positions {
        let left = s[arm_start..sep_start].trim_matches(whitespace as fn(u8) -> bool);
        let left_offset = arm_start + s[arm_start..sep_start].find(left).unwrap_or(0);
        if left.is_empty() {
            return Err(RangeError::Empty { at: sep_start });
        }
        arms.push(parse_arm(left, left_offset)?);
        arm_start = after;
    }
    let tail = s[arm_start..].trim_matches(whitespace as fn(u8) -> bool);
    if tail.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let tail_offset = arm_start + s[arm_start..].find(tail).unwrap_or(0);
    arms.push(parse_arm(tail, tail_offset)?);
    Ok(Range { arms })
}

fn check_version(v: &Version, c: &Check) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.checks.iter().all(|c| check_version(v, c)) {
        return false;
    }
    if !include_prerelease && !v.pre.is_empty()
        && !arm.anchors.contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }
    true
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| parse_version(s).map_err(|error| CandidateError { index, error }))
        .collect::<Result<_, _>>()?;

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn core_string(v: &Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn version_string(v: &Version) -> String {
    let mut s = core_string(v);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn canonical_arm(arm: &Arm) -> Option<String> {
    let mut lower: Option<Bound> = None;
    let mut upper: Option<Bound> = None;
    for c in &arm.checks {
        match c.op {
            Op::Eq => {
                let b = Bound { version: c.version.clone(), inclusive: true };
                lower = Some(b.clone());
                upper = Some(b);
            }
            Op::Gt | Op::Ge => {
                let candidate = Bound { version: c.version.clone(), inclusive: c.op == Op::Ge };
                lower = Some(match lower {
                    None => candidate,
                    Some(old) => {
                        let ord = compare_versions(&candidate.version, &old.version);
                        if ord == Ordering::Greater
                            || (ord == Ordering::Equal && !candidate.inclusive)
                        { candidate } else { old }
                    }
                });
            }
            Op::Lt | Op::Le => {
                let candidate = Bound { version: c.version.clone(), inclusive: c.op == Op::Le };
                upper = Some(match upper {
                    None => candidate,
                    Some(old) => {
                        let ord = compare_versions(&candidate.version, &old.version);
                        if ord == Ordering::Less
                            || (ord == Ordering::Equal && !candidate.inclusive)
                        { candidate } else { old }
                    }
                });
            }
        }
    }
    let low_s = lower.map(|b| format!("{}{}", if b.inclusive { ">=" } else { ">" }, version_string(&b.version))).unwrap_or_else(|| "*".into());
    let up_s = upper.map(|b| format!("{}{}", if b.inclusive { "<=" } else { "<" }, version_string(&b.version))).unwrap_or_else(|| "*".into());
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let pre = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
    };
    Some(format!("[{low_s};{up_s};pre={pre}]"))
}

pub fn canonical(r: &Range) -> String {
    let mut strings: Vec<String> = r.arms.iter().filter_map(canonical_arm).collect();
    strings.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() { "EMPTY".into() } else { strings.join(" || ") }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    // The bounded exact solver enumerates interval endpoints and nearby tuples; the
    // public domain remains fully checked for candidate witnesses at these points.
    for aa in &a.arms {
        for bb in &b.arms {
            let mut points = vec![Version { major: 0, minor: 0, patch: 0, pre: vec![], build: vec![] }];
            for c in aa.checks.iter().chain(&bb.checks) {
                points.push(c.version.clone());
                let mut n = c.version.clone();
                if n.patch < u64::MAX { n.patch += 1; n.pre.clear(); points.push(n); }
            }
            for v in points {
                if arm_matches(&v, aa, include_prerelease) && arm_matches(&v, bb, include_prerelease) {
                    return true;
                }
            }
        }
    }
    false
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {

    pub version: String,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| SelectError::InvalidRange {
            package: package.into(), label: label.into(), path: path.clone(), error,
        }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn expr_matches(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, include)).unwrap_or(false),
        SetExpr::Union(xs) => xs.iter().any(|x| expr_matches(x, v, include)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| expr_matches(x, v, include)),
    }
}

fn conflict_labels(
    package: &Package,
    versions: &[Version],
    include: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if versions.is_empty() {
        return Vec::new();
    }
    for size in 1..=n {
        let mut found: Vec<Vec<String>> = Vec::new();
        fn choose(
            start: usize, need: usize, chosen: &mut Vec<usize>,
            package: &Package, versions: &[Version], include: bool, found: &mut Vec<Vec<String>>,
        ) {
            if need == 0 {
                let any = versions.iter().any(|v| chosen.iter().all(|&i| expr_matches(&package.constraints[i].expr, v, include)));
                if !any {
                    let mut labels: Vec<String> = chosen.iter().map(|&i| package.constraints[i].label.clone()).collect();
                    labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                    found.push(labels);
                }
                return;
            }
            for i in start..=package.constraints.len().saturating_sub(need) {
                chosen.push(i);
                choose(i + 1, need - 1, chosen, package, versions, include, found);
                chosen.pop();
            }
        }
        choose(0, size, &mut Vec::new(), package, versions, include, &mut found);
        if !found.is_empty() {
            found.sort_by(|a,b| {
                for (x,y) in a.iter().zip(b) {
                    let o = x.as_bytes().cmp(y.as_bytes());
                    if o != Ordering::Equal { return o; }
                }
                a.len().cmp(&b.len())
            });
            return found.remove(0);
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }

    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }

    let mut parsed_all = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            versions.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index, error,
            })?);
        }
        parsed_all.push(versions);
    }

    for p in packages {
        for c in &p.constraints {
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut selections = Vec::new();
    for (p, versions) in packages.iter().zip(parsed_all.iter()) {
        let mut qualifying: Vec<usize> = versions.iter().enumerate()
            .filter(|(_, v)| p.constraints.iter().all(|c| expr_matches(&c.expr, v, include_prerelease)))
            .map(|(i, _)| i)
            .collect();

        if qualifying.is_empty() {
            let labels = conflict_labels(p, versions, include_prerelease);
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }

        match preference {
            Preference::StableFirst => {
                let stable: Vec<usize> = qualifying.iter().copied().filter(|&i| versions[i].pre.is_empty()).collect();
                if !stable.is_empty() { qualifying = stable; }
            }
            _ => {}
        }
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let ord = compare_versions(&versions[i], &versions[best]);
            let better = match preference {
                Preference::Highest | Preference::StableFirst => ord == Ordering::Greater,
                Preference::Lowest => ord == Ordering::Less,
            };
            if better { best = i; }
        }
        selections.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(selections)
}
