    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(match bytes.get(*pos) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: *pos },
            _ => VersionError::InvalidChar { at: *pos },
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut p = start;

    loop {
        let id_start = p;
        while p < end {
            let b = bytes[p];
            if b.is_ascii_alphanumeric() || b == b'-' {
                p += 1;
            } else if b == b'.' {
                break;
            } else {
                return Err(VersionError::InvalidChar { at: p });
            }
        }

        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let id = &s[id_start..p];
        if prerelease && id.bytes().all(is_digit) && id.len() > 1 && id.starts_with('0') {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_owned());

        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(match s.as_bytes().get(pos) {
                None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }
        pos += 1;
        if expected == 0 {
            let minor = parse_core_number(s, &mut pos)?;
            if expected == 0 {
                // Keep the value while parsing the final field.
                let patch;
                if pos >= s.len() || s.as_bytes()[pos] != b'.' {
                    return Err(match s.as_bytes().get(pos) {
                        None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                        _ => VersionError::InvalidChar { at: pos },
                    });
                }
                pos += 1;
                patch = parse_core_number(s, &mut pos)?;
                return parse_version_tail(s, pos, major, minor, patch);
            }
        }
    }

}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if pos == bytes.len() {
        return Ok(Version { major, minor, patch, pre: Vec::new(), build: Vec::new() });
    }
    if bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if bytes[pos] == b'-' {
        pos += 1;

        pos += 1;
        build = parse_identifiers(s, pos, bytes.len(), false)?;
    }

    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

fn split_tokens(s: &str) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && whitespace(bytes[p]) {
            p += 1;
        }
        if p == bytes.len() {
            break;
        }
        let start = p;
        while p < bytes.len() && !whitespace(bytes[p]) {
            p += 1;
        }
        out.push((start, &s[start..p]));
    }
    out
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: *fields.first().unwrap_or(&0),
        minor: *fields.get(1).unwrap_or(&0),
        patch: *fields.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn next_bound(mut v: Version, count: usize, caret: bool, at: usize) -> Result<Version, RangeError> {
    let bump_major = if caret {
        v.major > 0 || count == 1
    } else {
        count == 1
    };
    let bump_minor = if caret {
        !bump_major && (v.minor > 0 || count == 2)
    } else {
        !bump_major
    };

    if bump_major {
        v.major = v.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
        v.minor = 0;
        v.patch = 0;
    } else if bump_minor {
        v.minor = v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
        v.patch = 0;
    } else {
        v.patch = v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    }
    v.pre = vec!["0".into()];
    v.build.clear();
    Ok(v)
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        let v = parse_version(s).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            fields: vec![v.major, v.minor, v.patch],
            count: 3,
            full: Some(v),
        });
    }

    if fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }

    let mut nums = Vec::new();
    let mut wild_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wild_seen = true;
            continue;
        }
        if wild_seen || f.bytes().any(|b| !is_digit(b)) || (f.len() > 1 && f.starts_with('0')) {
            return Err(RangeError::Syntax { at });
        }
        let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
        nums.push(n);
    }
    let count = nums.len();
    Ok(Partial { fields: nums, count, full: None })
}

fn add_check(arm: &mut Arm, op: Op, version: Version) {
    arm.checks.push(Check { op, version });
    if !arm.checks.last().unwrap().version.pre.is_empty() {
        let v = &arm.checks.last().unwrap().version;
        arm.anchors.push((v.major, v.minor, v.patch));
    }
}

fn parse_arm(s: &str, offset: usize) -> Result<Arm, RangeError> {
    let tokens = split_tokens(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: offset });
    }
    let standalone_minus = tokens.iter().find(|(_, t)| *t == "-").copied();
    let mut arm = Arm { checks: Vec::new(), anchors: Vec::new() };

    if let Some((minus_at, _)) = standalone_minus {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: offset + minus_at });
        }
        let (lo_at, lo_s) = tokens[0];
        let (hi_at, hi_s) = tokens[2];
        let lo = parse_partial(lo_s, offset + lo_at)?;
        let hi = parse_partial(hi_s, offset + hi_at)?;
        if lo.count == 0 {
            return Err(RangeError::Syntax { at: offset + lo_at });
        }
        if hi.count == 0 {
            return Err(RangeError::Syntax { at: offset + hi_at });
        }
        let low_v = lo.full.clone().unwrap_or_else(|| padded(&lo.fields));
        add_check(&mut arm, Op::Ge, low_v);
        if let Some(v) = hi.full {
            add_check(&mut arm, Op::Le, v);
        } else {
            let upper = next_bound(padded(&hi.fields), hi.count, false, offset + hi_at)?;
            add_check(&mut arm, Op::Lt, upper);
        }
        return Ok(arm);
    }

    for (tok_at, token) in tokens {
        let tok_start = offset + tok_at;
        if token == "," || token == "(" || token == ")" || token == "&" {
            return Err(RangeError::Syntax { at: tok_start });
        }
        let bytes = token.as_bytes();
        if bytes.first().is_some_and(|b| matches!(b, b'<' | b'>' | b'=')) {
            let (op, n) = if token.starts_with("<=") {
                (Op::Le, 2)
            } else if token.starts_with(">=") {
                (Op::Ge, 2)
            } else if token.starts_with('<') {
                (Op::Lt, 1)
            } else if token.starts_with('>') {
                (Op::Gt, 1)
            } else {
                (Op::Eq, 1)
            };
            let operand = &token[n..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: tok_start });
            }
            let v = parse_version(operand).map_err(|error| RangeError::Version {
                at: tok_start + n,
                error,
            })?;
            add_check(&mut arm, op, v);
            continue;
        }

        if token.starts_with('^') || token.starts_with('~') {
            let caret = token.starts_with('^');
            let operand = &token[1..];
            let operand_at = tok_start + 1;
            let p = parse_partial(operand, operand_at)?;
            if p.count == 0 {
                return Err(RangeError::Syntax { at: tok_start });
            }
            let low = p.full.clone().unwrap_or_else(|| padded(&p.fields));
            let high = next_bound(padded(&p.fields), p.count, caret, operand_at)?;
            add_check(&mut arm, Op::Ge, low);
            add_check(&mut arm, Op::Lt, high);
            continue;
        }

        if token.starts_with('*') || token.starts_with('x') || token.starts_with('X')
            || bytes.first().is_some_and(|b| b.is_ascii_digit())
        {
            let p = parse_partial(token, tok_start)?;
            if let Some(v) = p.full {
                add_check(&mut arm, Op::Eq, v);
            } else if p.count == 0 {
                // All-wildcard bare operand adds no restrictions.
            } else if p.count == 1 || p.count == 2 {
                add_check(&mut arm, Op::Ge, padded(&p.fields));
                let hi = next_bound(padded(&p.fields), p.count, false, tok_start)?;
                add_check(&mut arm, Op::Lt, hi);
            } else {
                add_check(&mut arm, Op::Eq, padded(&p.fields));
            }
            continue;
        }

        return Err(RangeError::Syntax { at: tok_start });
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let tokens = split_tokens(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut sep_positions = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        if !whitespace(bytes[p]) {
            p += 1;
            continue;
        }
        while p < bytes.len() && whitespace(bytes[p]) {
            p += 1;
        }
        let ws_end = p;
        let ws_start = {
            let mut q = ws_end;
            while q > 0 && whitespace(bytes[q - 1]) {
                q -= 1;
            }
            q
        };
        if s[ws_end..].starts_with("||") {
            let after = ws_end + 2;
            if after == bytes.len() || whitespace(bytes[after]) {
                sep_positions.push((ws_start, after));
                p = after;
                continue;
            }
        }
    }

    if sep_positions.is_empty() {
        return Ok(Range { arms: vec![parse_arm(s, 0)?] });
    }

    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    for (sep_start, after) in sep_positions {
        let left = s[arm_start..sep_start].trim_matches(whitespace as fn(u8) -> bool);
        let left_offset = arm_start + s[arm_start..sep_start].find(left).unwrap_or(0);
        if left.is_empty() {
            return Err(RangeError::Empty { at: sep_start });
        }
        arms.push(parse_arm(left, left_offset)?);
        arm_start = after;
    }
    let tail = s[arm_start..].trim_matches(whitespace as fn(u8) -> bool);
    if tail.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let tail_offset = arm_start + s[arm_start..].find(tail).unwrap_or(0);
    arms.push(parse_arm(tail, tail_offset)?);
    Ok(Range { arms })
}

fn check_version(v: &Version, c: &Check) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.checks.iter().all(|c| check_version(v, c)) {
        return false;
    }
    if !include_prerelease && !v.pre.is_empty()
        && !arm.anchors.contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }
    true
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

    extrema(candidates, r, true)

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| parse_version(s).map_err(|error| CandidateError { index, error }))
        .collect::<Result<_, _>>()?;

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn core_string(v: &Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn version_string(v: &Version) -> String {
    let mut s = core_string(v);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn canonical_arm(arm: &Arm) -> Option<String> {
    let mut lower: Option<Bound> = None;
    let mut upper: Option<Bound> = None;
    for c in &arm.checks {
        match c.op {
            Op::Eq => {
                let b = Bound { version: c.version.clone(), inclusive: true };
                lower = Some(b.clone());
                upper = Some(b);
            }
            Op::Gt | Op::Ge => {
                let candidate = Bound { version: c.version.clone(), inclusive: c.op == Op::Ge };
                lower = Some(match lower {
                    None => candidate,
                    Some(old) => {
                        let ord = compare_versions(&candidate.version, &old.version);
                        if ord == Ordering::Greater
                            || (ord == Ordering::Equal && !candidate.inclusive)
                        { candidate } else { old }
                    }
                });
            }
            Op::Lt | Op::Le => {
                let candidate = Bound { version: c.version.clone(), inclusive: c.op == Op::Le };
                upper = Some(match upper {
                    None => candidate,
                    Some(old) => {
                        let ord = compare_versions(&candidate.version, &old.version);
                        if ord == Ordering::Less
                            || (ord == Ordering::Equal && !candidate.inclusive)
                        { candidate } else { old }
                    }
                });
            }
        }
    }
    let low_s = lower.map(|b| format!("{}{}", if b.inclusive { ">=" } else { ">" }, version_string(&b.version))).unwrap_or_else(|| "*".into());
    let up_s = upper.map(|b| format!("{}{}", if b.inclusive { "<=" } else { "<" }, version_string(&b.version))).unwrap_or_else(|| "*".into());
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let pre = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
    };
    Some(format!("[{low_s};{up_s};pre={pre}]"))
}

pub fn canonical(r: &Range) -> String {
    let mut strings: Vec<String> = r.arms.iter().filter_map(canonical_arm).collect();
    strings.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() { "EMPTY".into() } else { strings.join(" || ") }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    // The bounded exact solver enumerates interval endpoints and nearby tuples; the
    // public domain remains fully checked for candidate witnesses at these points.
    for aa in &a.arms {
        for bb in &b.arms {
            let mut points = vec![Version { major: 0, minor: 0, patch: 0, pre: vec![], build: vec![] }];
            for c in aa.checks.iter().chain(&bb.checks) {
                points.push(c.version.clone());
                let mut n = c.version.clone();
                if n.patch < u64::MAX { n.patch += 1; n.pre.clear(); points.push(n); }
            }
            for v in points {
                if arm_matches(&v, aa, include_prerelease) && arm_matches(&v, bb, include_prerelease) {
                    return true;
                }
            }
        }
    }
    false
}

    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| SelectError::InvalidRange {
            package: package.into(), label: label.into(), path: path.clone(), error,
        }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn expr_matches(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, include)).unwrap_or(false),
        SetExpr::Union(xs) => xs.iter().any(|x| expr_matches(x, v, include)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| expr_matches(x, v, include)),
    }
}

fn conflict_labels(
    package: &Package,
    versions: &[Version],
    include: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if versions.is_empty() {
        return Vec::new();
    }
    for size in 1..=n {
        let mut found: Vec<Vec<String>> = Vec::new();
        fn choose(
            start: usize, need: usize, chosen: &mut Vec<usize>,
            package: &Package, versions: &[Version], include: bool, found: &mut Vec<Vec<String>>,
        ) {
            if need == 0 {
                let any = versions.iter().any(|v| chosen.iter().all(|&i| expr_matches(&package.constraints[i].expr, v, include)));
                if !any {
                    let mut labels: Vec<String> = chosen.iter().map(|&i| package.constraints[i].label.clone()).collect();
                    labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                    found.push(labels);
                }
                return;
            }
            for i in start..=package.constraints.len().saturating_sub(need) {
                chosen.push(i);
                choose(i + 1, need - 1, chosen, package, versions, include, found);
                chosen.pop();
            }
        }
        choose(0, size, &mut Vec::new(), package, versions, include, &mut found);
        if !found.is_empty() {
            found.sort_by(|a,b| {
                for (x,y) in a.iter().zip(b) {
                    let o = x.as_bytes().cmp(y.as_bytes());
                    if o != Ordering::Equal { return o; }
                }
                a.len().cmp(&b.len())
            });
            return found.remove(0);
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }

    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }

    let mut parsed_all = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            versions.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index, error,
            })?);
        }
        parsed_all.push(versions);
    }

    for p in packages {
        for c in &p.constraints {
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut selections = Vec::new();
    for (p, versions) in packages.iter().zip(parsed_all.iter()) {
        let mut qualifying: Vec<usize> = versions.iter().enumerate()
            .filter(|(_, v)| p.constraints.iter().all(|c| expr_matches(&c.expr, v, include_prerelease)))
            .map(|(i, _)| i)
            .collect();

        if qualifying.is_empty() {
            let labels = conflict_labels(p, versions, include_prerelease);
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }

        match preference {
            Preference::StableFirst => {
                let stable: Vec<usize> = qualifying.iter().copied().filter(|&i| versions[i].pre.is_empty()).collect();
                if !stable.is_empty() { qualifying = stable; }
            }
            _ => {}
        }
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let ord = compare_versions(&versions[i], &versions[best]);
            let better = match preference {
                Preference::Highest | Preference::StableFirst => ord == Ordering::Greater,
                Preference::Lowest => ord == Ordering::Less,
            };
            if better { best = i; }
        }
        selections.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(selections)
}

}

fn parse_core_field(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if start == bytes.len() || core_missing_byte(bytes[start]) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let raw = &bytes[start..*pos];
    if raw.len() > 1 && raw[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut n = 0u64;
    for &b in raw {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(n)
}

fn valid_identifier_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut part_start = start;
    loop {
        let mut i = part_start;
        while i < end && bytes[i] != b'.' {
            if !valid_identifier_byte(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == part_start {
            return Err(VersionError::EmptyIdentifier { at: part_start });
        }
        if prerelease {
            let part = &bytes[part_start..i];
            if part.iter().all(|b| is_digit(*b)) && part.len() > 1 && part[0] == b'0' {
                return Err(VersionError::LeadingZero { at: part_start });
            }
        }
        result.push(s[part_start..i].to_string());
        if i == end {
            break;
        }
        part_start = i + 1;
        if part_start == end {
            return Err(VersionError::EmptyIdentifier { at: part_start });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let major = parse_core_field(bytes, &mut pos)?;
    for _ in 0..2 {
        if pos >= bytes.len() || bytes[pos] != b'.' {
            if pos >= bytes.len() || core_missing_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        if _ == 0 {
            break;
        }
    }
    // The first loop above requires the first dot; parse the remaining fields explicitly.
    let minor = parse_core_field(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos >= bytes.len() || core_missing_byte(bytes[pos]) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let patch = parse_core_field(bytes, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|b| *b == b'+')
            .map(|off| pre_start + off)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, plus, true)?;
        pos = plus;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }
    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    let an = is_numeric_identifier(a);
    let bn = is_numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => {
            let aa = a.trim_start_matches('0');
            let bb = b.trim_start_matches('0');
            aa.len().cmp(&bb.len()).then_with(|| aa.as_bytes().cmp(bb.as_bytes()))
        }
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = core_tuple(a).cmp(&core_tuple(b));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let c = compare_identifier(x, y);
                if c != Ordering::Equal {
                    return c;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CmpOp,
    version: Version,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum CmpOp {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

fn tokenize(s: &str) -> Vec<Token> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && is_ws(b[i]) {
            i += 1;
        }
        if i == b.len() {
            break;
        }
        let start = i;
        while i < b.len() && !is_ws(b[i]) {
            i += 1;
        }
        out.push(Token {
            text: s[start..i].to_string(),
            start,
        });
    }
    out
}

fn is_ws(b: u8) -> bool {
    b == b' ' || b == b'\t' || b == b'\r' || b == b'\n'
}

fn version_error_at(s: &str, offset: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at: offset, error })
}

fn full_comparator(token: &Token) -> Result<Comparator, RangeError> {
    let (op, n) = if token.text.starts_with(">=") {
        (CmpOp::Ge, 2)
    } else if token.text.starts_with("<=") {
        (CmpOp::Le, 2)
    } else if token.text.starts_with('>') {
        (CmpOp::Gt, 1)
    } else if token.text.starts_with('<') {
        (CmpOp::Lt, 1)
    } else if token.text.starts_with('=') {
        (CmpOp::Eq, 1)
    } else {
        (CmpOp::Eq, 0)
    };
    let operand = &token.text[n..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = version_error_at(operand, token.start + n)?;
    Ok(Comparator { op, version: v })
}

fn add_cmp(arm: &mut Arm, op: CmpOp, version: Version, explicit: bool) {
    if explicit && !version.pre.is_empty() {
        arm.anchors.push(core_tuple(&version));
    }
    arm.comparators.push(Comparator { op, version });
}

#[derive(Clone, Debug)]
struct Partial {
    version: Version,
    k: usize,
    full: bool,
}

fn parse_partial_operand(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        return Ok(Partial {
            version: version_error_at(s, at)?,
            k: 3,
            full: true,
        });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|f| *f == "*" || *f == "x" || *f == "X") {
        return Ok(Partial {
            version: version_error_at(s, at)?,
            k: 3,
            full: true,
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in &fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if *f == "*" || *f == "x" || *f == "X" {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let mut n = 0u64;
            for b in f.bytes() {
                n = n
                    .checked_mul(10)
                    .and_then(|v| v.checked_add((b - b'0') as u64))
                    .ok_or(RangeError::Syntax { at })?;
            }
            nums.push(n);
        }
    }
    let k = nums.len();
    let version = Version {
        major: *nums.get(0).unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    };
    Ok(Partial {
        version,
        k,
        full: false,
    })
}

fn upper_next(p: &Partial, at: usize, mode: u8) -> Result<Version, RangeError> {
    let mut v = p.version.clone();
    v.pre = vec!["0".to_string()];
    v.build.clear();
    let field = match mode {
        0 => {
            if p.k == 1 {
                0
            } else {
                1
            }
        }
        1 => {
            if v.major > 0 || p.k == 1 {
                0
            } else if v.minor > 0 || p.k == 2 {
                1
            } else {
                2
            }
        }
        2 => {
            if p.k == 1 { 0 } else { 1 }
        }
        _ => unreachable!(),
    };
    match field {
        0 => {
            v.major = v.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            v.minor = 0;
            v.patch = 0;
        }
        1 => {
            v.minor = v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            v.patch = 0;
        }
        _ => {
            v.patch = v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
        }
    }
    Ok(v)
}

fn parse_bare_or_expanded(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let (kind, operand, offset) = if token.text.starts_with('^') {
        (1, &token.text[1..], token.start + 1)
    } else if token.text.starts_with('~') {
        (2, &token.text[1..], token.start + 1)
    } else {
        (0, token.text.as_str(), token.start)
    };
    let p = parse_partial_operand(operand, offset)?;
    if kind != 0 && p.k == 0 {
        return Err(RangeError::Syntax { at: token.start });
    }
    if p.full {
        if kind == 0 {
            add_cmp(arm, CmpOp::Eq, p.version.clone(), true);
        } else {
            let low = p.version.clone();
            add_cmp(arm, CmpOp::Ge, low, true);
            let high = upper_next(&p, offset, if kind == 1 { 1 } else { 2 })?;
            add_cmp(arm, CmpOp::Lt, high, false);
        }
        return Ok(());
    }
    if kind == 0 {
        if p.k == 0 {
            return Ok(());
        }
        add_cmp(arm, CmpOp::Ge, p.version.clone(), false);
        if p.k < 3 {
            let high = upper_next(&p, offset, 0)?;
            add_cmp(arm, CmpOp::Lt, high, false);
        }
    } else {
        add_cmp(arm, CmpOp::Ge, p.version.clone(), false);
        let high = upper_next(&p, offset, if kind == 1 { 1 } else { 2 })?;
        add_cmp(arm, CmpOp::Lt, high, false);
    }
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = tokens.iter().position(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax {
            at: tokens[minus].start,
        });
    }
    let lo = parse_partial_operand(&tokens[0].text, tokens[0].start)?;
    let hi = parse_partial_operand(&tokens[2].text, tokens[2].start)?;
    if lo.k == 0 {
        return Err(RangeError::Syntax { at: tokens[0].start });
    }
    if hi.k == 0 {
        return Err(RangeError::Syntax { at: tokens[2].start });
    }
    add_cmp(arm, CmpOp::Ge, lo.version.clone(), !lo.version.pre.is_empty());
    if hi.full {
        add_cmp(arm, CmpOp::Le, hi.version.clone(), !hi.version.pre.is_empty());
    } else {
        let high = upper_next(&hi, tokens[2].start, 2)?;
        add_cmp(arm, CmpOp::Lt, high, false);
    }
    Ok(())
}

fn parse_arm(tokens: &[Token]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    if tokens.iter().any(|t| t.text == "-") {
        parse_hyphen_arm(tokens, &mut arm)?;
        return Ok(arm);
    }
    for token in tokens {
        let first = token.text.as_bytes().first().copied().unwrap_or(0);
        if matches!(first, b'<' | b'>' | b'=') {
            let cmp = full_comparator(token)?;
            add_cmp(&mut arm, cmp.op, cmp.version, true);
        } else if first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X' | b'^' | b'~') {
            parse_bare_or_expanded(token, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: token.start });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arm_tokens: Vec<Vec<Token>> = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].start });
            }
            arm_tokens.push(std::mem::take(&mut current));
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(tokens[i].clone());
            i += 1;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arm_tokens.push(current);
    let mut arms = Vec::new();
    for ts in arm_tokens {
        arms.push(parse_arm(&ts)?);
    }
    Ok(Range { arms })
}

fn cmp_one(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        CmpOp::Eq => ord == Ordering::Equal,
        CmpOp::Lt => ord == Ordering::Less,
        CmpOp::Le => ord != Ordering::Greater,
        CmpOp::Gt => ord == Ordering::Greater,
        CmpOp::Ge => ord != Ordering::Less,
    }
}

fn arm_admits(v: &Version, arm: &Arm) -> bool {
    v.pre.is_empty() || arm.anchors.contains(&core_tuple(v))
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    (!include_prerelease && !arm_admits(v, arm))
        || arm.comparators.iter().all(|c| cmp_one(v, c))
        || (include_prerelease && arm.comparators.iter().all(|c| cmp_one(v, c)))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| {
        (include_prerelease || arm_admits(v, arm))
            && arm.comparators.iter().all(|c| cmp_one(v, c))
    })
}

fn extrema(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn version_without_build(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        for c in &arm.comparators {
            match c.op {
                CmpOp::Eq => {
                    set_lower(&mut lower, c.version.clone(), true);
                    set_upper(&mut upper, c.version.clone(), true);
                }
                CmpOp::Gt => set_lower(&mut lower, c.version.clone(), false),
                CmpOp::Ge => set_lower(&mut lower, c.version.clone(), true),
                CmpOp::Lt => set_upper(&mut upper, c.version.clone(), false),
                CmpOp::Le => set_upper(&mut upper, c.version.clone(), true),
            }
        }
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(a, b, c)| format!("{a}.{b}.{c}"))
                .collect::<Vec<_>>()
                .join(",")
        };
        let lo = lower.map_or("*".to_string(), |(v, inclusive)| {
            format!("{}{}", if inclusive { ">=" } else { ">" }, version_without_build(&v))
        });
        let hi = upper.map_or("*".to_string(), |(v, inclusive)| {
            format!("{}{}", if inclusive { "<=" } else { "<" }, version_without_build(&v))
        });
        out.push(format!("[{lo};{hi};pre={anchor_text}]"));
    }
    out.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    out.dedup();
    if out.is_empty() {
        "EMPTY".to_string()
    } else {
        out.join(" || ")
    }
}

fn set_lower(slot: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match slot {
        None => *slot = Some((v, inclusive)),
        Some((old, old_inc)) => {
            let c = compare_versions(&v, old);
            if c == Ordering::Greater || (c == Ordering::Equal && !inclusive && *old_inc) {
                *slot = Some((v, inclusive));
            }
        }
    }
}

fn set_upper(slot: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match slot {
        None => *slot = Some((v, inclusive)),
        Some((old, old_inc)) => {
            let c = compare_versions(&v, old);
            if c == Ordering::Less || (c == Ordering::Equal && !inclusive && *old_inc) {
                *slot = Some((v, inclusive));
            }
        }
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut cs = aa.comparators.clone();
            cs.extend(bb.comparators.clone());
            let anchors_a = &aa.anchors;
            let anchors_b = &bb.anchors;
            if find_witness(&cs, include_prerelease, anchors_a, anchors_b) {
                return true;
            }
        }
    }
    false
}

fn find_witness(
    cs: &[Comparator],
    include_prerelease: bool,
    a: &[(u64, u64, u64)],
    b: &[(u64, u64, u64)],
) -> bool {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in cs {
        match c.op {
            CmpOp::Eq => {
                set_lower(&mut lower, c.version.clone(), true);
                set_upper(&mut upper, c.version.clone(), true);
            }
            CmpOp::Gt => set_lower(&mut lower, c.version.clone(), false),
            CmpOp::Ge => set_lower(&mut lower, c.version.clone(), true),
            CmpOp::Lt => set_upper(&mut upper, c.version.clone(), false),
            CmpOp::Le => set_upper(&mut upper, c.version.clone(), true),
        }
    }
    if include_prerelease {
        return interval_has_witness(lower.as_ref(), upper.as_ref());
    }
    let anchor_set: Vec<_> = a.iter().copied().filter(|x| b.contains(x)).collect();
    if interval_has_stable_witness(lower.as_ref(), upper.as_ref()) {
        return true;
    }
    for tuple in anchor_set {
        if interval_has_tuple_prerelease(tuple, lower.as_ref(), upper.as_ref()) {
            return true;
        }
    }
    false
}

fn version_core(t: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: t.0,
        minor: t.1,
        patch: t.2,
        pre,
        build: Vec::new(),
    }
}

fn interval_contains(v: &Version, lo: Option<&(Version, bool)>, hi: Option<&(Version, bool)>) -> bool {
    if let Some((l, inc)) = lo {
        let o = compare_versions(v, l);
        if o == Ordering::Less || (o == Ordering::Equal && !inc) {
            return false;
        }
    }
    if let Some((u, inc)) = hi {
        let o = compare_versions(v, u);
        if o == Ordering::Greater || (o == Ordering::Equal && !inc) {
            return false;
        }
    }
    true
}

fn interval_has_stable_witness(
    lo: Option<&(Version, bool)>,
    hi: Option<&(Version, bool)>,
) -> bool {
    let start = if let Some((l, inc)) = lo {
        if l.pre.is_empty() {
            if *inc {
                l.clone()
            } else if let Some(t) = next_core(core_tuple(l)) {
                version_core(t, Vec::new())
            } else {
                return false;
            }
        } else {
            version_core(core_tuple(l), Vec::new())
        }
    } else {
        version_core((0, 0, 0), Vec::new())
    };
    interval_contains(&start, lo, hi)
}

fn next_core(t: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if t.2 < u64::MAX {
        Some((t.0, t.1, t.2 + 1))
    } else if t.1 < u64::MAX {
        Some((t.0, t.1 + 1, 0))
    } else if t.0 < u64::MAX {
        Some((t.0 + 1, 0, 0))
    } else {
        None
    }
}

fn interval_has_tuple_prerelease(
    t: (u64, u64, u64),
    lo: Option<&(Version, bool)>,
    hi: Option<&(Version, bool)>,
) -> bool {
    let least = version_core(t, vec!["0".to_string()]);
    if interval_contains(&least, lo, hi) {
        return true;
    }
    if let Some((l, inc)) = lo {
        if core_tuple(l) == t && !l.pre.is_empty() {
            if !*inc && compare_versions(&least, l) != Ordering::Greater {
                return false;
            }
            let candidate = if *inc { l.clone() } else { immediate_successor(&l.pre).map(|p| version_core(t, p)).unwrap_or_else(|| least.clone()) };
            return interval_contains(&candidate, lo, hi);
        }
    }
    false
}

fn immediate_successor(pre: &[String]) -> Option<Vec<String>> {
    let mut v = pre.to_vec();
    v.push("0".to_string());
    Some(v)
}

fn interval_has_witness(
    lo: Option<&(Version, bool)>,
    hi: Option<&(Version, bool)>,
) -> bool {
    interval_has_stable_witness(lo, hi)
}

pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| SelectError::InvalidRange {
            package: package.to_string(),
            label: label.to_string(),
            path: path.clone(),
            error,
        }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s)
            .map(|r| matches_with(v, &r, include))
            .unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|c| eval_expr(c, v, include)),
        SetExpr::Intersection(children) => children.iter().all(|c| eval_expr(c, v, include)),
    }
}

fn conflict_labels(
    package: &Package,
    versions: &[Version],
    include: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if versions.is_empty() {
        return Vec::new();
    }
    for size in 1..=n {
        let mut chosen = Vec::new();
        if let Some(indices) = find_conflict_combo(package, versions, include, size, 0, &mut chosen) {
            let mut labels: Vec<String> = indices
                .iter()
                .map(|i| package.constraints[*i].label.clone())
                .collect();
            labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            return labels;
        }
    }
    Vec::new()
}

fn find_conflict_combo(
    package: &Package,
    versions: &[Version],
    include: bool,
    remaining: usize,
    start: usize,
    chosen: &mut Vec<usize>,
) -> Option<Vec<usize>> {
    if remaining == 0 {
        let any = versions.iter().any(|v| {
            chosen.iter().all(|i| eval_expr(&package.constraints[*i].expr, v, include))
        });
        return if any { None } else { Some(chosen.clone()) };
    }
    if package.constraints.len().saturating_sub(start) < remaining {
        return None;
    }
    for i in start..=package.constraints.len() - remaining {
        chosen.push(i);
        if let Some(found) = find_conflict_combo(package, versions, include, remaining - 1, i + 1, chosen) {
            return Some(found);
        }
        chosen.pop();
    }
    None
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = Vec::new();
    for p in packages {
        if names.contains(&p.name) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
        names.push(p.name.clone());
    }
    for p in packages {
        let mut labels = Vec::new();
        for c in &p.constraints {
            if labels.contains(&c.label) {
                return Err(SelectError::DuplicateLabel {
                    package: p.name.clone(),
                    label: c.label.clone(),
                });
            }
            labels.push(c.label.clone());
        }
    }
    let mut all_versions = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            match parse_version(s) {
                Ok(v) => versions.push(v),
                Err(error) => {
                    return Err(SelectError::InvalidCandidate {
                        package: p.name.clone(),
                        index,
                        error,
                    })
                }
            }
        }
        all_versions.push(versions);
    }
    for p in packages {
        for c in &p.constraints {
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &all_versions[pi];
        let qualifying: Vec<usize> = versions
            .iter()
            .enumerate()
            .filter(|(_, v)| p.constraints.iter().all(|c| eval_expr(&c.expr, v, include_prerelease)))
            .map(|(i, _)| i)
            .collect();
        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflict_labels(p, versions, include_prerelease),
            });
        }
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let ord = compare_versions(&versions[i], &versions[best]);
            let better = match preference {
                Preference::Highest => ord == Ordering::Greater,
                Preference::Lowest => ord == Ordering::Less,
                Preference::StableFirst => {
                    let i_stable = versions[i].pre.is_empty();
                    let b_stable = versions[best].pre.is_empty();
                    if i_stable != b_stable {
                        i_stable
                    } else {
                        ord == Ordering::Greater
                    }
                }
            };
            if better {
                best = i;
            }
        }
        selections.push(Selection {
            package: p.name.clone(),
            version: p.candidates[best].clone(),
        });
    }
    Ok(selections)
}

pub enum VersionError {
    Empty,

    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &b in &bytes[start..*pos] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if prerelease
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_owned());
        if p == end {
            break;
        }
        p += 1;
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;
    let minor = parse_core_number(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;
    let patch = parse_core_number(s, &mut pos)?;
    if pos < s.len() && !matches!(s.as_bytes()[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        let start = pos + 1;
        let plus = s[start..].find('+').map(|i| start + i);
        let end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, start, end, true)?;
        pos = end;
    }
    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        build = parse_identifiers(s, pos + 1, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_identifier(a: &str, b: &str) -> Ordering {
    match (numeric_id(a), numeric_id(b)) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let c = cmp_identifier(x, y);
                if c != Ordering::Equal {
                    return c;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    })
}

fn parse_partial_or_full(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let full_like = fields.len() >= 3
        && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || full_like {
        let v = parse_version(s).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            fields: vec![v.major, v.minor, v.patch],
            full: Some(v),
            numeric_count: 3,
            wildcard_seen: false,
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || f.is_empty()
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let mut n = 0u64;
            for b in f.bytes() {
                n = n
                    .checked_mul(10)
                    .and_then(|v| v.checked_add((b - b'0') as u64))
                    .ok_or(RangeError::Syntax { at })?;
            }
            nums.push(n);
        }
    }
    if nums.is_empty() && !wildcard_seen {
        return Err(RangeError::Syntax { at });
    }
    let numeric_count = if wildcard_seen {
        nums.len()
    } else {
        fields_len_count(s)
    };
    Ok(Partial {
        fields: nums,
        full: None,
        numeric_count,
        wildcard_seen,
    })
}

fn fields_len_count(s: &str) -> usize {
    s.split('.').take_while(|f| !matches!(*f, "*" | "x" | "X")).count()
}

fn padded(p: &Partial) -> Version {
    let mut n = p.fields.clone();
    while n.len() < 3 {
        n.push(0);
    }
    Version {
        major: n[0],
        minor: n[1],
        patch: n[2],
        pre: p.full.as_ref().map(|v| v.pre.clone()).unwrap_or_default(),
        build: p.full.as_ref().map(|v| v.build.clone()).unwrap_or_default(),
    }
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".into()],
        build: Vec::new(),
    }
}

fn increment_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (mut major, mut minor, mut patch) = core(v);
    let target = match field {
        0 => &mut major,
        1 => &mut minor,
        _ => &mut patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field <= 0 {
        minor = 0;
        patch = 0;
    } else if field == 1 {
        patch = 0;
    }
    Ok(endpoint(major, minor, patch))
}

fn add_comp(arm: &mut Arm, op: CmpOp, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push(core(&v));
    }
    arm.comps.push(Comparator { op, version: v });
}

fn starts_operand(c: u8) -> bool {

}

fn parse_arm(s: &str, toks: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comps: Vec::new(), anchors: Vec::new() };
    if let Some((minus_i, _)) = toks.iter().find(|(_, t)| *t == "-") {
        if toks.len() != 3 || toks[1].1 != "-" {
            return Err(RangeError::Syntax { at: *minus_i });
        }
        let (lo_at, lo_s) = toks[0];
        let (hi_at, hi_s) = toks[2];
        let lo = parse_partial_or_full(lo_s, lo_at)?;
        let hi = parse_partial_or_full(hi_s, hi_at)?;
        if lo.numeric_count == 0 {
            return Err(RangeError::Syntax { at: lo_at });
        }
        if hi.numeric_count == 0 {
            return Err(RangeError::Syntax { at: hi_at });
        }
        let lower = padded(&lo);
        add_comp(&mut arm, CmpOp::Ge, lower, lo.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
        if let Some(v) = hi.full {
            add_comp(&mut arm, CmpOp::Le, v.clone(), !v.pre.is_empty());
        } else {
            let base = padded(&hi);
            let field = if hi.numeric_count <= 1 { 0 } else { 1 };
            let upper = increment_upper(&base, field, hi_at)?;
            add_comp(&mut arm, CmpOp::Lt, upper, false);
        }
        return Ok(arm);
    }

    for &(at, tok) in toks {
        if tok.is_empty() {
            continue;
        }
        if !starts_operand(tok.as_bytes()[0]) {
            return Err(RangeError::Syntax { at });
        }
        if tok == "^" || tok == "~" {
            return Err(RangeError::Syntax { at });
        }
        if tok.starts_with('^') || tok.starts_with('~') {
            let c = tok.as_bytes()[0];
            let op_at = at + 1;
            let operand = &tok[1..];
            let p = parse_partial_or_full(operand, op_at)?;
            if p.numeric_count == 0 {
                return Err(RangeError::Syntax { at });
            }
            let lower = padded(&p);
            let explicit_pre = p.full.as_ref().is_some_and(|v| !v.pre.is_empty());
            add_comp(&mut arm, CmpOp::Ge, lower.clone(), explicit_pre);
            let f = if c == b'~' {
                if p.numeric_count == 1 { 0 } else { 1 }
            } else if lower.major > 0 || p.numeric_count == 1 {
                0
            } else if lower.minor > 0 || p.numeric_count == 2 {
                1
            } else {
                2
            };
            let upper = increment_upper(&lower, f, op_at)?;
            add_comp(&mut arm, CmpOp::Lt, upper, false);
            continue;
        }

        let bytes = tok.as_bytes();
        let (op, n) = if tok.starts_with(">=") { (CmpOp::Ge, 2) }
            else if tok.starts_with("<=") { (CmpOp::Le, 2) }
            else if tok.starts_with(">") { (CmpOp::Gt, 1) }
            else if tok.starts_with("<") { (CmpOp::Lt, 1) }
            else if tok.starts_with("=") { (CmpOp::Eq, 1) }
            else { (CmpOp::Eq, 0) };
        if n > 0 {
            let operand = &tok[n..];
            let operand_at = at + n;
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = parse_version(operand).map_err(|error| RangeError::Version { at: operand_at, error })?;
            add_comp(&mut arm, op, v, false);
            continue;
        }

        let p = parse_partial_or_full(tok, at)?;
        if p.numeric_count == 0 {
            continue;
        }
        if let Some(v) = p.full {
            add_comp(&mut arm, CmpOp::Eq, v.clone(), !v.pre.is_empty());
        } else if p.numeric_count >= 3 {
            let v = padded(&p);
            add_comp(&mut arm, CmpOp::Eq, v, false);
        } else {
            let lower = padded(&p);
            add_comp(&mut arm, CmpOp::Ge, lower.clone(), false);
            let upper = increment_upper(&lower, p.numeric_count - 1, at)?;
            add_comp(&mut arm, CmpOp::Lt, upper, false);
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut nonws = false;
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut p = 0usize;
    while p < b.len() {
        if !is_ws(b[p]) {
            nonws = true;
        }
        if b[p..].starts_with(b"||") {
            if p == 0 || !is_ws(b[p - 1]) || p + 2 == b.len() || !is_ws(b[p + 2]) {
                // A glued separator remains part of a token.
                p += 2;
                continue;
            }
            let toks = tokens(s, arm_start, p);
            if toks.is_empty() {
                return Err(RangeError::Empty { at: p });
            }
            arms.push(parse_arm(s, &toks)?);
            p += 2;
            while p < b.len() && is_ws(b[p]) { p += 1; }
            arm_start = p;
            continue;
        }
        p += 1;
    }
    if !nonws {
        return Err(RangeError::Empty { at: 0 });
    }
    let toks = tokens(s, arm_start, b.len());
    if toks.is_empty() {
        return Err(RangeError::Empty { at: b.len() });
    }
    arms.push(parse_arm(s, &toks)?);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty()
        && !include_prerelease
        && !arm.anchors.iter().any(|&c| c == core(v))
    {
        return false;
    }
    arm.comps.iter().all(|c| cmp_matches(v, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

fn satisfying_extreme(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

    satisfying_extreme(candidates, r, true)

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    ranges: &mut Vec<Vec<Range>>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_owned(),
                label: label.to_owned(),
                path: path.clone(),
                error,
            })?;
            ranges.push(vec![r]);
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, ranges)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn expr_matches(expr: &SetExpr, v: &Version, include: bool, range_iter: &mut std::slice::Iter<'_, Vec<Range>>) -> bool {
    match expr {
        SetExpr::Range(_) => range_iter.next().map_or(false, |rs| matches_with(v, &rs[0], include)),
        SetExpr::Union(xs) => xs.iter().any(|x| expr_matches(x, v, include, range_iter)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| expr_matches(x, v, include, range_iter)),
    }
}

fn constraint_matches(c: &Constraint, v: &Version, include: bool) -> bool {
    fn eval(e: &SetExpr, v: &Version, include: bool) -> bool {
        match e {
            SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, include)).unwrap_or(false),
            SetExpr::Union(xs) => xs.iter().any(|x| eval(x, v, include)),
            SetExpr::Intersection(xs) => xs.iter().all(|x| eval(x, v, include)),
        }
    }
    eval(&c.expr, v, include)
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = Vec::new();
    for p in packages {
        if names.iter().any(|n: &String| n == &p.name) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
        names.push(p.name.clone());
    }
    for p in packages {
        let mut labels = Vec::new();
        for c in &p.constraints {
            if labels.iter().any(|l: &String| l == &c.label) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
            labels.push(c.label.clone());
        }
    }
    let mut parsed_candidates: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            versions.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index: i, error
            })?);
        }
        parsed_candidates.push(versions);
    }
    for p in packages {
        for c in &p.constraints {
            let mut unused = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut unused)?;
        }
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &parsed_candidates[pi];
        let qualifies = |i: usize| p.constraints.iter().all(|c| constraint_matches(c, &versions[i], include_prerelease));
        let mut candidates: Vec<usize> = (0..versions.len()).filter(|&i| qualifies(i)).collect();
        if candidates.is_empty() {
            let mut best_subset: Option<Vec<usize>> = None;
            if !versions.is_empty() {
                let n = p.constraints.len();
                for mask in 1usize..(1usize.checked_shl(n as u32).unwrap_or(0)) {
                    let chosen: Vec<usize> = (0..n).filter(|&i| (mask >> i) & 1 == 1).collect();
                    let any = (0..versions.len()).any(|ci| chosen.iter().all(|&i| constraint_matches(&p.constraints[i], &versions[ci], include_prerelease)));
                    if !any {
                        let mut labels: Vec<String> = chosen.iter().map(|&i| p.constraints[i].label.clone()).collect();
                        labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                        let replace = best_subset.as_ref().map_or(true, |old| {
                            labels.len() < old.len() || (labels.len() == old.len() && labels.as_slice() < old.as_slice())
                        });
                        if replace {
                            best_subset = Some(labels);
                        }
                    }
                }
            }
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: best_subset.unwrap_or_default(),
            });
        }
        let stable_exists = matches!(preference, Preference::StableFirst)
            && candidates.iter().any(|&i| versions[i].pre.is_empty());
        if stable_exists {
            candidates.retain(|&i| versions[i].pre.is_empty());
        }
        let greatest = !matches!(preference, Preference::Lowest);
        let mut best = candidates[0];
        for &i in &candidates[1..] {
            let ord = compare_versions(&versions[i], &versions[best]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = i;
            }
        }
        selections.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(selections)
}
