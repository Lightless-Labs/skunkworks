#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        match template.as_bytes()[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let opening = cursor;
                let Some(relative_close) = template[opening + 1..].find('}') else {
                    return Err(Error::Unclosed { pos: opening });
                };
                let closing = opening + 1 + relative_close;

                if let Some(relative_inner) = template[opening + 1..closing].find('{') {
                    return Err(Error::NestedOpen {
                        pos: opening + 1 + relative_inner,
                    });
                }

                let (operator, content_start) =
                    match template.as_bytes().get(opening + 1).copied() {
                        Some(b'+') => (Operator::Reserved, opening + 2),
                        Some(b'#') => (Operator::Fragment, opening + 2),
                        Some(b'.' | b'/' | b';' | b'?' | b'&') => {
                            return Err(Error::UnsupportedOperator {
                                pos: opening + 1,
                            });
                        }
                        _ => (Operator::Simple, opening + 1),
                    };

                if content_start == closing {
                    return Err(Error::EmptyExpression { pos: closing });
                }

                let content = &template[content_start..closing];
                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut part_start = content_start;

                for part in content.split(',') {
                    if part.is_empty() {
                        return Err(Error::EmptyVariable { pos: part_start });
                    }

                    let mut bytes = part.bytes();
                    if !matches!(bytes.next(), Some(b'A'..=b'Z' | b'a'..=b'z' | b'_')) {
                        return Err(Error::InvalidName { pos: part_start });
                    }

                    let mut valid_len = 1;
                    for byte in bytes {
                        if matches!(byte, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'_') {
                            valid_len += 1;
                        } else {
                            break;
                        }
                    }
                    if valid_len != part.len() {
                        return Err(Error::InvalidName {
                            pos: part_start + valid_len,
                        });
                    }

                    if seen.iter().any(|name| name == part) {
                        return Err(Error::DuplicateVariable { pos: part_start });
                    }

                    let name = part.to_owned();
                    seen.push(name.clone());
                    names.push((name, part_start));
                    part_start += part.len() + 1;
                }

                expressions.push(Expression {
                    start: opening,
                    end: closing + 1,
                    operator,
                    names,
                });
                cursor = closing + 1;
            }
            _ => {
                cursor += template[cursor..]
                    .chars()
                    .next()
                    .expect("cursor is within the template")
                    .len_utf8();
            }
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields = Vec::new();
        for (name, _) in &expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(encode(value, expression.operator != Operator::Simple));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Operator::Fragment {
                output.push('#');
            }
            let separator = ",";
            for (index, field) in fields.iter().enumerate() {
                if index > 0 {
                    output.push_str(separator);
                }
                output.push_str(field);
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    matches!(
        byte,
        b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~'
    )
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'('
            | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    b"0123456789ABCDEF"[value as usize] as char
}
