#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    pos: usize,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    specifiers: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let mut start = 0;
    let mut operator = None;

    if let Some(&first) = bytes.first() {
        let ch = first as char;
        if matches!(ch, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            if ch != '+' && ch != '#' {
                return Err(Error::UnsupportedOperator {
                    pos: content_pos,
                });
            }
            operator = Some(ch);
            start = 1;
        }
    }

    if start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specifiers = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut item_start = start;

    for item_end in (start..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if item_end == item_start {
            return Err(Error::EmptyVariable {
                pos: content_pos + item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        if !is_name_start(item[0]) {
            return Err(Error::InvalidName {
                pos: content_pos + item_start,
            });
        }

        for (offset, &byte) in item.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: content_pos + item_start + offset,
                });
            }
        }

        let name = content[item_start..item_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable {
                pos: content_pos + item_start,
            });
        }
        seen.push(name.clone());
        specifiers.push(Specifier {
            name,
            pos: content_pos + item_start,
        });

        item_start = item_end + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let mut output = String::new();
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit() {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved = allow_reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
                    | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }

        i += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();
    let allow_reserved = matches!(expression.operator, Some('+') | Some('#'));

    for specifier in &expression.specifiers {
        let _ = specifier.pos;
        if let Some(value) = lookup(&specifier.name, vars) {
            fields.push(encode_value(value, allow_reserved));
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let joined = fields.join(",");
    if expression.operator == Some('#') {
        format!("#{joined}")
    } else {
        joined
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut parsed = Vec::<(usize, usize, Expression)>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = i + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(relative) = bytes[open + 1..close].iter().position(|&byte| byte == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative,
                    });
                }

                let content = &template[open + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                parsed.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for (open, end, expression) in parsed {
        output.push_str(&template[cursor..open]);
        output.push_str(&expand_expression(&expression, vars));
        cursor = end;
    }
    output.push_str(&template[cursor..]);

    Ok(output)
}
