#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }

    let expected = board.width * board.height;
    if board.cells.len() != expected {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

fn move_line(
    board: &Board,
    output: &mut [u32],
    merges: &mut Vec<Merge>,
    score: &mut u64,
    direction: Direction,
    line: usize,
) -> Result<(), Error> {
    let length = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };

    let index_at = |position: usize| -> usize {
        match direction {
            Direction::Left => line * board.width + position,
            Direction::Right => line * board.width + (board.width - 1 - position),
            Direction::Up => position * board.width + line,
            Direction::Down => (board.height - 1 - position) * board.width + line,
        }
    };

    let mut tiles: Vec<(u32, usize)> = Vec::new();
    for position in 0..length {
        let index = index_at(position);
        let value = board.cells[index];
        if value != 0 {
            tiles.push((value, index));
        }
    }

    let mut packed: Vec<u32> = Vec::with_capacity(length);
    let mut cursor = 0;
    while cursor < tiles.len() {
        let (value, index) = tiles[cursor];
        if cursor + 1 < tiles.len() && tiles[cursor + 1].0 == value {
            let second_index = tiles[cursor + 1].1;
            let Some(merged_value) = value.checked_mul(2) else {
                return Err(Error::MergeOverflow {
                    first: index,
                    second: second_index,
                });
            };
            let destination = index_at(packed.len());
            packed.push(merged_value);
            merges.push(Merge {
                from: [index, second_index],
                to: destination,
                value: merged_value,
            });
            *score += u64::from(merged_value);
            cursor += 2;
        } else {
            packed.push(value);
            cursor += 1;
        }
    }

    for position in 0..length {
        output[index_at(position)] = packed.get(position).copied().unwrap_or(0);
    }

    Ok(())
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        move_line(
            board,
            &mut cells,
            &mut merges,
            &mut score,
            direction,
            line,
        )?;
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let directions = [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ];
    let mut legal = Vec::new();

    for direction in directions {
        if let Ok(result) = move_board(board, direction) {
            if result.changed {
                legal.push(direction);
            }
        }
    }

    Ok(legal)
}
