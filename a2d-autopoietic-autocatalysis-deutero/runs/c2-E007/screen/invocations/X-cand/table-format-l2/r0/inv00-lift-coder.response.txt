#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let column_count = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != column_count {
            return Err(TableError::Ragged {
                row: row_index,
                expected: column_count,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != column_count {
        return Err(TableError::AlignmentCount {
            expected: column_count,
            actual: options.alignments.len(),
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            if let Some((byte, _)) = cell
                .char_indices()
                .find(|(_, ch)| !(' '..='~').contains(ch))
            {
                return Err(TableError::InvalidChar {
                    row: row_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; column_count];
    for row in &trimmed {
        for (column_index, cell) in row.iter().enumerate() {
            widths[column_index] = widths[column_index].max(cell.len());
        }
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let alignment_for = |column: usize| {
        options
            .alignments
            .get(column)
            .unwrap_or(&Alignment::Left)
    };

    let mut output = String::new();

    let write_border = |output: &mut String| {
        output.push('+');
        for &width in &widths {
            output.extend(std::iter::repeat_n('-', width + 2));
            output.push('+');
        }
        output.push('\n');
    };

    let write_row = |output: &mut String, row_index: usize| {
        output.push('|');
        for (column, cell) in trimmed[row_index].iter().enumerate() {
            let extra = widths[column] - cell.len();
            let (left, right) = match alignment_for(column) {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            output.push(' ');
            output.extend(std::iter::repeat_n(' ', left));
            output.push_str(cell);
            output.extend(std::iter::repeat_n(' ', right));
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    };

    write_border(&mut output);
    for row_index in 0..rows.len() {
        write_row(&mut output, row_index);
        if options.header && row_index == 0 && rows.len() > 1 {
            write_border(&mut output);
        }
    }
    write_border(&mut output);

    Ok(output)
}
