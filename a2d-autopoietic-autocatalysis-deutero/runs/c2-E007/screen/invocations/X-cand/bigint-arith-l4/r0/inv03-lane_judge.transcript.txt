    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const LIMB_BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.limbs.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs(&self) -> Self {
        let mut out = self.clone();
        out.negative = false;
        out
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    fn mul_small(&mut self, factor: u32) {
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::LIMB_BASE) as u32;
            carry = value / Self::LIMB_BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::LIMB_BASE) as u32);
            carry /= Self::LIMB_BASE;
        }
        self.normalize();
    }

    fn add_small(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::LIMB_BASE) as u32;
            carry = sum / Self::LIMB_BASE;
            i += 1;
        }
        self.normalize();
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        ord => return ord,
                    }
                }
                Ordering::Equal
            }
            ord => ord,
        }
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.limbs.len().max(b.limbs.len());
        let mut limbs = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let av = *a.limbs.get(i).unwrap_or(&0) as u64;
            let bv = *b.limbs.get(i).unwrap_or(&0) as u64;
            let sum = av + bv + carry;
            limbs.push((sum % Self::LIMB_BASE) as u32);
            carry = sum / Self::LIMB_BASE;
        }
        if carry != 0 {
            limbs.push(carry as u32);
        }
        let mut out = Self {
            negative: false,
            limbs,
        };
        out.normalize();
        out
    }

    // Assumes |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut limbs = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let av = a.limbs[i] as i64;
            let bv = *b.limbs.get(i).unwrap_or(&0) as i64;
            let mut value = av - bv - borrow;
            if value < 0 {
                value += Self::LIMB_BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            limbs.push(value as u32);
        }
        let mut out = Self {
            negative: false,
            limbs,
        };
        out.normalize();
        out
    }

    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.cmp_abs(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }

        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut remainder = Self::zero();

        for &limb in a.limbs.iter().rev() {
            remainder.limbs.insert(0, limb);
            remainder.normalize();

            let mut low = 0u64;
            let mut high = Self::LIMB_BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = b.mul_small_copy(mid as u32);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }

            quotient_be.push(best as u32);
            if best != 0 {
                let product = b.mul_small_copy(best as u32);
                remainder = Self::sub_abs(&remainder, &product);
            }
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn mul_small_copy(&self, factor: u32) -> Self {
        let mut out = self.abs();
        out.mul_small(factor);
        out
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut remainder = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let current = remainder * Self::LIMB_BASE + *limb as u64;
            *limb = (current / divisor as u64) as u32;
            remainder = current % divisor as u64;
        }
        self.normalize();
        remainder as u32
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        let mut digits = Vec::new();
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            match Self::digit_value(bytes[i]) {
                Some(v) if v < radix => digits.push((i, v)),
                _ => {
                    if invalid.is_none() {
                        invalid = Some(i);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > start
                    && bytes[i - 1] != b'_'
                    && Self::digit_value(bytes[i - 1]).is_some_and(|v| v < radix);
                let right_ok = i + 1 < bytes.len()
                    && bytes[i + 1] != b'_'
                    && Self::digit_value(bytes[i + 1]).is_some_and(|v| v < radix);
                if !(left_ok && right_ok) && malformed_separator.is_none() {
                    malformed_separator = Some(i);
                }
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        if digits.len() > 1 && digits[0].1 == 0 {
            return Err(Error::LeadingZero { index: digits[0].0 });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digits[max_digits].0,
            });
        }

        let mut out = Self::zero();
        for &(_, digit) in &digits {
            out.mul_small(radix);
            out.add_small(digit);
        }
        out.negative = negative && !out.is_zero();
        Ok(out)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.abs();
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push('0');
        } else {
            while !value.is_zero() {
                let digit = value.div_small(radix);
                digits.push(char::from_digit(digit, radix).unwrap());
            }
            digits.reverse();
        }

        while digits.len() < min_digits {
            digits.insert(0, '0');
        }

        let mut result = String::new();
        if self.negative && !self.is_zero() {
            result.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                result.push('_');
            }
            result.push(*digit);
        }
        Ok(result)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self::add_abs(self, rhs);
            out.negative = self.negative && !out.is_zero();
            out
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self::sub_abs(self, rhs);
                    out.negative = self.negative;
                    out
                }
                Ordering::Less => {
                    let mut out = Self::sub_abs(rhs, self);
                    out.negative = rhs.negative;
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = limbs[index] as u64 + a as u64 * b as u64 + carry;
                limbs[index] = (value % Self::LIMB_BASE) as u32;
                carry = value / Self::LIMB_BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == limbs.len() {
                    limbs.push(0);
                }
                let value = limbs[index] as u64 + carry;
                limbs[index] = (value % Self::LIMB_BASE) as u32;
                carry = value / Self::LIMB_BASE;
                index += 1;
            }
        }
        let mut out = Self {
            negative: self.negative != rhs.negative,
            limbs,
        };
        out.normalize();
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::div_rem_abs(self, rhs);
        quotient.negative =
            (self.negative != rhs.negative) && !quotient.is_zero();
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {

            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });

            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let remainder = a.div_rem(&b).expect("nonzero divisor").1;
            a = b;
            b = remainder.abs();
        }
        a.negative = false;
        a
    }

const LIMB_BASE: u64 = 1_000_000_000;

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs
            .len()
            .cmp(&rhs.limbs.len())
            .then_with(|| self.limbs.iter().rev().cmp(rhs.limbs.iter().rev()))
    }

    fn add_abs(&self, rhs: &Self) -> Self {
        let n = self.limbs.len().max(rhs.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let a = self.limbs.get(i).copied().unwrap_or(0) as u64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = a + b + carry;
            out.push((sum % LIMB_BASE) as u32);
            carry = sum / LIMB_BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self {
            negative: false,
            limbs: out,
        }
    }

    // Requires |self| >= |rhs|.
    fn sub_abs(&self, rhs: &Self) -> Self {
        let mut out = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;
        for i in 0..self.limbs.len() {
            let a = self.limbs[i] as i64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut value = a - b - borrow;
            if value < 0 {
                value += LIMB_BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u32);
        }
        let mut result = Self {
            negative: false,
            limbs: out,
        };
        result.normalize();
        result
    }

    fn mul_small_abs(&self, factor: u32) -> Self {
        if factor == 0 || self.is_zero() {
            return Self::zero();
        }
        let mut out = Vec::with_capacity(self.limbs.len() + 1);
        let mut carry = 0u64;
        for &limb in &self.limbs {
            let product = limb as u64 * factor as u64 + carry;
            out.push((product % LIMB_BASE) as u32);
            carry = product / LIMB_BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self {
            negative: false,
            limbs: out,
        }
    }

    fn add_small_abs(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % LIMB_BASE) as u32;
            carry = sum / LIMB_BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut out = vec![0; self.limbs.len()];
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let current = remainder * LIMB_BASE + self.limbs[i] as u64;
            out[i] = (current / divisor as u64) as u32;
            remainder = current % divisor as u64;
        }
        let mut quotient = Self {
            negative: false,
            limbs: out,
        };
        quotient.normalize();
        (quotient, remainder as u32)
    }

    fn bit_len(&self) -> usize {
        if self.is_zero() {
            return 0;
        }
        let mut n = self.clone();
        let mut bits = 0;
        while !n.is_zero() {
            n = n.div_rem_small_abs(2).0;
            bits += 1;
        }
        bits
    }

    fn bit(&self, bit: usize) -> bool {
        let mut n = self.clone();
        for _ in 0..bit {
            if n.is_zero() {
                return false;
            }
            n = n.div_rem_small_abs(2).0;
        }
        !n.is_zero() && n.div_rem_small_abs(2).1 == 1
    }

    fn shl1_add_bit(&mut self, bit: bool) {
        let mut carry = bit as u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * 2 + carry;
            *limb = (value % LIMB_BASE) as u32;
            carry = value / LIMB_BASE;
        }
        if carry != 0 {
            self.limbs.push(carry as u32);
        }
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && Self::digit_value(byte).is_none_or(|d| d >= radix) {
                invalid = Some(index);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_' {
                let valid_neighbor = |i: usize| {
                    i >= start
                        && i < bytes.len()
                        && bytes[i] != b'_'
                        && Self::digit_value(bytes[i]).is_some_and(|d| d < radix)
                };
                if !valid_neighbor(index.wrapping_sub(1))
                    || !valid_neighbor(index + 1)
                {
                    separator = Some(index);
                    break;
                }
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let mut seen = 0usize;
            let index = (start..bytes.len())
                .find(|&i| {
                    if bytes[i] == b'_' {
                        false
                    } else {
                        let is_excess = seen == excess_number;
                        seen += 1;
                        is_excess
                    }
                })
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut result = Self::zero();
        for &byte in &bytes[start..] {
            if byte != b'_' {
                result = result.mul_small_abs(radix);
                result.add_small_abs(Self::digit_value(byte).unwrap());
            }
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0u8);
        } else {
            while !n.is_zero() {
                let (q, r) = n.div_rem_small_abs(radix);
                digits.push(r as u8);
                n = q;
            }
            digits.reverse();
        }
        if digits.len() < min_digits {
            let mut padded = vec![0u8; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut output = String::new();
        if self.negative && !self.is_zero() {
            output.push('-');
        }
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                output.push('_');
            }
            output.push(if digit < 10 {
                (b'0' + digit) as char
            } else {
                (b'a' + digit - 10) as char
            });
        }
        Ok(output)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = self.add_abs(rhs);
            result.negative = self.negative;
            result.normalize();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Greater => {
                    let mut result = self.sub_abs(rhs);
                    result.negative = self.negative;
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = rhs.sub_abs(self);
                    result.negative = rhs.negative;
                    result.normalize();
                    result
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = out[index] + a as u64 * b as u64 + carry;
                out[index] = value % LIMB_BASE;
                carry = value / LIMB_BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == out.len() {
                    out.push(0);
                }
                let value = out[index] + carry;
                out[index] = value % LIMB_BASE;
                carry = value / LIMB_BASE;
                index += 1;
            }
        }
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            limbs: out.into_iter().map(|x| x as u32).collect(),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let dividend = self.abs();
        let divisor = rhs.abs();
        if dividend.cmp_abs(&divisor) == Ordering::Less {
            return Ok((Self::zero(), self.clone()));
        }

        let mut quotient = Self::zero();
        let mut remainder = Self::zero();
        let bit_len = dividend.bit_len();
        let mut quotient_bits = vec![false; bit_len];

        for bit in (0..bit_len).rev() {
            remainder.shl1_add_bit(dividend.bit(bit));
            if remainder.cmp_abs(&divisor) != Ordering::Less {
                remainder = remainder.sub_abs(&divisor);
                quotient_bits[bit] = true;
            }
        }

        for bit in (0..bit_len).rev() {
            if quotient_bits[bit] {
                quotient.shl1_add_bit(true);
            } else if !quotient.is_zero() {
                quotient.shl1_add_bit(false);
            }
        }
        quotient.negative = !quotient.is_zero() && (self.negative ^ rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power != 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let remainder = a.div_rem(&b).unwrap().1.abs();
            a = b;
            b = remainder;
        }
        a.negative = false;
        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

const BASE: u64 = 1_000_000_000;
const POW_LIMIT: u32 = 1024;

impl Big {
    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_values = Vec::with_capacity(bytes.len() - start);
        let mut digit_indices = Vec::with_capacity(bytes.len() - start);
        let mut invalid = None;

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            match digit_value(byte) {
                Some(value) if value < radix => {
                    digit_values.push(value);
                    digit_indices.push(index);
                }
                _ => {
                    if invalid.is_none() {
                        invalid = Some(index);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                return Err(Error::Separator { index });
            }
        }

        if digit_values.len() > 1 && digit_values[0] == 0 {
            return Err(Error::LeadingZero {
                index: digit_indices[0],
            });
        }
        if digit_values.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_indices[max_digits],
            });
        }

        let mut result = Self::zero();
        for value in digit_values {
            result.mul_small_assign(radix);
            result.add_small_assign(value);
        }
        if !result.is_zero() {
            result.negative = negative;
        }
        Ok(result)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut magnitude = self.abs();
        let mut digits = Vec::new();
        if magnitude.is_zero() {
            digits.push(0);
        } else {
            while !magnitude.is_zero() {
                digits.push(magnitude.div_small_assign(radix));
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut output = String::new();
        if self.negative && !self.is_zero() {
            output.push('-');
        }
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                output.push('_');
            }
            output.push(digit_char(digit));
        }
        Ok(output)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_abs(self, rhs),
            (true, true) => cmp_abs(rhs, self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        match (self.negative, rhs.negative) {
            (a, b) if a == b => {
                let mut result = add_abs(self, rhs);
                result.negative = a && !result.is_zero();
                result
            }
            _ => match cmp_abs(self, rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = sub_abs(self, rhs);
                    result.negative = self.negative && !result.is_zero();
                    result
                }
                Ordering::Less => {
                    let mut result = sub_abs(rhs, self);
                    result.negative = rhs.negative && !result.is_zero();
                    result
                }
            },
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = mul_abs(self, rhs);
        result.negative = self.negative ^ rhs.negative;
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = div_rem_abs(&self.abs(), &rhs.abs());
        if !quotient.is_zero() {
            quotient.negative = self.negative ^ rhs.negative;
        }
        if !remainder.is_zero() {
            remainder.negative = self.negative;
        }
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: POW_LIMIT,
            });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power > 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = remainder.abs();
        }
        a
    }

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn mul_small_assign(&mut self, multiplier: u32) {
        if multiplier == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * multiplier as u64 + carry;
            *limb = (value % BASE) as u32;
            carry = value / BASE;
        }
        while carry > 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
    }

    fn add_small_assign(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut index = 0;
        while carry > 0 {
            if index == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[index] as u64 + carry;
            self.limbs[index] = (value % BASE) as u32;
            carry = value / BASE;
            index += 1;
        }
    }

    fn div_small_assign(&mut self, divisor: u32) -> u32 {
        let mut remainder = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let value = remainder * BASE + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        self.normalize();
        remainder as u32
    }
}

    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
        b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
        _ => None,
    }
}

fn digit_char(value: u32) -> char {
    match value {
        0..=9 => (b'0' + value as u8) as char,
        _ => (b'a' + (value - 10) as u8) as char,
    }
}

fn cmp_abs(a: &Big, b: &Big) -> Ordering {
    a.limbs
        .len()
        .cmp(&b.limbs.len())
        .then_with(|| {
            a.limbs
                .iter()
                .rev()
                .cmp(b.limbs.iter().rev())
        })
}

fn add_abs(a: &Big, b: &Big) -> Big {
    let length = a.limbs.len().max(b.limbs.len());
    let mut limbs = Vec::with_capacity(length + 1);
    let mut carry = 0u64;
    for i in 0..length {
        let value = a.limbs.get(i).copied().unwrap_or(0) as u64
            + b.limbs.get(i).copied().unwrap_or(0) as u64
            + carry;
        limbs.push((value % BASE) as u32);
        carry = value / BASE;
    }
    if carry > 0 {
        limbs.push(carry as u32);
    }
    Big { negative: false, limbs }
}

fn sub_abs(a: &Big, b: &Big) -> Big {
    let mut limbs = Vec::with_capacity(a.limbs.len());
    let mut borrow = 0i64;
    for i in 0..a.limbs.len() {
        let mut value = a.limbs[i] as i64
            - b.limbs.get(i).copied().unwrap_or(0) as i64
            - borrow;
        if value < 0 {
            value += BASE as i64;
            borrow = 1;
        } else {
            borrow = 0;
        }
        limbs.push(value as u32);
    }
    let mut result = Big { negative: false, limbs };
    result.normalize();
    result
}

fn mul_abs(a: &Big, b: &Big) -> Big {
    let mut limbs = vec![0u32; a.limbs.len() + b.limbs.len()];
    for (i, &left) in a.limbs.iter().enumerate() {
        let mut carry = 0u64;
        for (j, &right) in b.limbs.iter().enumerate() {
            let index = i + j;
            let value = limbs[index] as u64 + left as u64 * right as u64 + carry;
            limbs[index] = (value % BASE) as u32;
            carry = value / BASE;
        }
        let mut index = i + b.limbs.len();
        while carry > 0 {
            if index == limbs.len() {
                limbs.push(0);
            }
            let value = limbs[index] as u64 + carry;
            limbs[index] = (value % BASE) as u32;
            carry = value / BASE;
            index += 1;
        }
    }
    let mut result = Big { negative: false, limbs };
    result.normalize();
    result
}

fn div_rem_abs(dividend: &Big, divisor: &Big) -> (Big, Big) {
    if cmp_abs(dividend, divisor) == Ordering::Less {
        return (Big::zero(), dividend.clone());
    }

    let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
    let mut remainder = Big::zero();

    for &limb in dividend.limbs.iter().rev() {
        remainder.limbs.insert(0, limb);
        remainder.normalize();

        let mut low = 0u64;
        let mut high = BASE - 1;
        let mut best = 0u64;
        while low <= high {
            let middle = (low + high) / 2;
            let product = mul_abs_small(divisor, middle);
            match cmp_abs(&product, &remainder) {
                Ordering::Greater => {
                    if middle == 0 {
                        break;
                    }
                    high = middle - 1;
                }
                _ => {
                    best = middle;
                    low = middle + 1;
                }
            }
        }

        if best > 0 {
            remainder = sub_abs(&remainder, &mul_abs_small(divisor, best));
        }
        quotient_be.push(best as u32);
    }

    quotient_be.reverse();
    let mut quotient = Big { negative: false, limbs: quotient_be };
    quotient.normalize();
    (quotient, remainder)
}

fn mul_abs_small(value: &Big, multiplier: u64) -> Big {
    if multiplier == 0 || value.is_zero() {
        return Big::zero();
    }
    let mut limbs = Vec::with_capacity(value.limbs.len() + 1);
    let mut carry = 0u64;
    for &limb in &value.limbs {
        let product = limb as u64 * multiplier + carry;
        limbs.push((product % BASE) as u32);
        carry = product / BASE;
    }
    while carry > 0 {
        limbs.push((carry % BASE) as u32);
        carry /= BASE;
    }
    Big { negative: false, limbs }
}
