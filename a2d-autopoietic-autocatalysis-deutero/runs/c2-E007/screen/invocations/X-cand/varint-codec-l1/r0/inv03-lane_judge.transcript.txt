pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut encoded = Vec::with_capacity(10);
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            encoded.push(payload);
            break;
        }
        encoded.push(payload | 0x80);
    }
    encoded
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;

    for index in 0..10 {
        let offset = match start.checked_add(index) {
            Some(offset) => offset,
            None => return Err(Error::Truncated { offset: input.len(), start }),
        };

        if index == 9 && offset >= input.len() {
            return Err(Error::Truncated { offset: input.len(), start });
        }

        let byte = match input.get(offset) {
            Some(&byte) => byte,
            None => return Err(Error::Truncated { offset: input.len(), start }),
        };
        let payload = byte & 0x7f;

        if index == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow { offset, start });
            }
            value |= (payload as u64) << 63;
            return Ok((value, offset + 1));
        }

        value |= (payload as u64) << (index * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });

        values.push(value);

}
