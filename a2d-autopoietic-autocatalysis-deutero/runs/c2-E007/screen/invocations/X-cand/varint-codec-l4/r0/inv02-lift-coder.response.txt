#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Message {
    pub fields: Vec<Field>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Field {
    UInt { tag: u32, value: u64 },
    Bytes { tag: u32, data: Vec<u8> },
    Nested { tag: u32, message: Message },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
    NonCanonical { offset: usize },
    DepthExceeded { offset: usize, depth: usize },
    InvalidDepthLimit { limit: usize },
}

fn put_varint(mut value: u64, out: &mut Vec<u8>) {
    while value >= 0x80 {
        out.push((value as u8) | 0x80);
        value >>= 7;
    }
    out.push(value as u8);
}

fn read_varint(input: &[u8], mut cursor: usize, end: usize) -> Result<(u64, usize), Error> {
    let start = cursor;
    let mut value = 0u64;

    for i in 0..10 {
        if cursor >= end {
            return Err(Error::Truncated { offset: end, start });
        }
        let byte = input[cursor];
        cursor += 1;
        let payload = (byte & 0x7f) as u64;

        if i == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow {
                offset: cursor - 1,
                start,
            });
        }

        value |= payload << (i * 7);
        if byte & 0x80 == 0 {
            if i > 0 && payload == 0 {
                return Err(Error::NonCanonical { offset: start });
            }
            return Ok((value, cursor));
        }
    }

    unreachable!()
}

fn usize_from_length(length: u64, offset: usize) -> Result<usize, Error> {
    usize::try_from(length).map_err(|_| Error::LengthOutOfBounds { offset, length })
}

fn encode_body(message: &Message, max_depth: usize, depth: usize) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous: Option<(u32, bool)> = None;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, is_bytes) = match field {
            Field::UInt { tag, .. } => (*tag, false),
            Field::Bytes { tag, .. } => (*tag, true),
            Field::Nested { tag, .. } => (*tag, false),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag {
                offset: index,
                tag: tag as u64,
            });
        }

        if let Some((prev_tag, prev_bytes)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder {
                    offset: index,
                    previous: prev_tag,
                    tag,
                });
            }
            if tag == prev_tag && !(prev_bytes && is_bytes) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        let (wire, payload) = match field {
            Field::UInt { value, .. } => {
                let mut p = Vec::new();
                put_varint(*value, &mut p);
                (0u64, p)
            }
            Field::Bytes { data, .. } => {
                let mut p = Vec::new();
                put_varint(data.len() as u64, &mut p);
                p.extend_from_slice(data);
                (2u64, p)
            }
            Field::Nested { message, .. } => {
                let child_depth = depth + 1;
                if child_depth > max_depth {
                    return Err(Error::DepthExceeded {
                        offset: index,
                        depth: child_depth,
                    });
                }
                let child = encode_body(message, max_depth, child_depth)?;
                let mut p = Vec::new();
                put_varint(child.len() as u64, &mut p);
                p.extend_from_slice(&child);
                (3u64, p)
            }
        };

        put_varint(((tag as u64) << 3) | wire, &mut body);
        body.extend_from_slice(&payload);
        previous = Some((tag, is_bytes));
    }

    Ok(body)
}

pub fn encode_message(message: &Message, max_depth: usize) -> Result<Vec<u8>, Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    let body = encode_body(message, max_depth, 0)?;
    let mut output = Vec::new();
    put_varint(body.len() as u64, &mut output);
    output.extend_from_slice(&body);
    Ok(output)
}

fn decode_body(
    input: &[u8],
    mut cursor: usize,
    end: usize,
    max_depth: usize,
    depth: usize,
) -> Result<Message, Error> {
    let mut fields: Vec<Field> = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    while cursor < end {
        let key_start = cursor;
        let (key, next) = read_varint(input, cursor, end)?;
        cursor = next;

        let tag64 = key >> 3;
        let wire = (key & 7) as u8;

        if !(1..=255).contains(&tag64) {
            return Err(Error::InvalidTag {
                offset: key_start,
                tag: tag64,
            });
        }
        let tag = tag64 as u32;

        if !matches!(wire, 0 | 1 | 2 | 3 | 5) {
            return Err(Error::InvalidWire {
                offset: key_start,
                wire,
            });
        }

        if let Some((prev_tag, prev_wire)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder {
                    offset: key_start,
                    previous: prev_tag,
                    tag,
                });
            }
            if tag == prev_tag && !(prev_wire == 2 && wire == 2) {
                return Err(Error::Duplicate {
                    offset: key_start,
                    tag,
                });
            }
        }
        previous = Some((tag, wire));

        let retained = tag <= 15;
        match wire {
            0 => {
                let (value, next) = read_varint(input, cursor, end)?;
                cursor = next;
                if retained {
                    fields.push(Field::UInt { tag, value });
                }
            }
            1 | 5 => {
                let count = if wire == 1 { 8usize } else { 4usize };
                let remaining = end - cursor;
                if remaining < count {
                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: count as u64,
                    });
                }
                cursor += count;
            }
            2 | 3 => {
                let length_start = cursor;
                let (length, next) = read_varint(input, cursor, end)?;
                cursor = next;
                let length_usize = usize_from_length(length, length_start)?;
                let remaining = end - cursor;
                if length_usize > remaining {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length,
                    });
                }
                let body_start = cursor;
                let body_end = cursor + length_usize;

                if wire == 2 {
                    if retained {
                        let data = input[body_start..body_end].to_vec();
                        if let Some(Field::Bytes { tag: prev_tag, data: prev_data }) =
                            fields.last_mut()
                        {
                            if *prev_tag == tag {
                                prev_data.extend_from_slice(&data);
                            } else {
                                fields.push(Field::Bytes { tag, data });
                            }
                        } else {
                            fields.push(Field::Bytes { tag, data });
                        }
                    }
                    cursor = body_end;
                } else {
                    let child_depth = depth + 1;
                    if child_depth > max_depth {
                        return Err(Error::DepthExceeded {
                            offset: key_start,
                            depth: child_depth,
                        });
                    }
                    let child = decode_body(input, body_start, body_end, max_depth, child_depth)?;
                    cursor = body_end;
                    if retained {
                        fields.push(Field::Nested {
                            tag,
                            message: child,
                        });
                    }
                }
            }
            _ => unreachable!(),
        }
    }

    Ok(Message { fields })
}

pub fn decode_message(
    input: &[u8],
    start: usize,
    max_depth: usize,
) -> Result<(Message, usize), Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let (length, body_start) = read_varint(input, start, input.len())?;
    let length_usize = usize_from_length(length, start)?;
    let remaining = input.len() - body_start;
    if length_usize > remaining {
        return Err(Error::LengthOutOfBounds {
            offset: start,
            length,
        });
    }

    let body_end = body_start + length_usize;
    let message = decode_body(input, body_start, body_end, max_depth, 0)?;
    Ok((message, body_end))
}
