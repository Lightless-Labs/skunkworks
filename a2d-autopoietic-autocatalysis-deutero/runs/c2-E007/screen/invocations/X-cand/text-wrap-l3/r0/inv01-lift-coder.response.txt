#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    opportunities: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    word_index: usize,
}

struct Line {
    indent: usize,
    pieces: Vec<Piece>,
}

impl Line {
    fn content_len(&self) -> usize {
        self.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn render(&self, width: usize, justify: bool, is_final: bool, line_index: usize) -> String {
        let mut out = " ".repeat(self.indent);
        let content_len = self.content_len();
        let gaps = self.pieces.len().saturating_sub(1);
        if justify && !is_final && gaps > 0 {
            let extra = width.saturating_sub(self.indent + content_len);
            let each = extra / gaps;
            let rem = extra % gaps;
            let start = line_index % gaps;
            for (i, piece) in self.pieces.iter().enumerate() {
                if i > 0 {
                    let gap_i = i - 1;
                    let bonus = usize::from((gap_i + gaps - start) % gaps < rem);
                    out.push_str(&" ".repeat(1 + each + bonus));
                }
                out.push_str(&piece.text);
            }
        } else {
            for (i, piece) in self.pieces.iter().enumerate() {
                if i > 0 {
                    out.push(' ');
                }
                out.push_str(&piece.text);
            }
        }
        out
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut opportunities = Vec::new();

    let finish = |words: &mut Vec<Word>,
                  chars: &mut Vec<char>,
                  opportunities: &mut Vec<usize>| {
        if !chars.is_empty() {
            words.push(Word {
                chars: std::mem::take(chars),
                opportunities: std::mem::take(opportunities),
            });
        } else {
            opportunities.clear();
        }
    };

    for &c in segment {
        if is_separator(c) {
            finish(&mut words, &mut chars, &mut opportunities);
        } else if c == '~' {
            opportunities.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    finish(&mut words, &mut chars, &mut opportunities);
    words
}

fn process_paragraph(words: &[Word], options: &WrapOptions, output: &mut Vec<String>) {
    if words.is_empty() {
        return;
    }

    let mut lines: Vec<Line> = Vec::new();
    let mut current = Line {
        indent: options.first_indent,
        pieces: Vec::new(),
    };

    for (word_index, word) in words.iter().enumerate() {
        let mut offset = 0usize;
        while offset < word.chars.len() {
            let cap = options.width - current.indent;
            let used = current.content_len();
            let room = cap.saturating_sub(used + usize::from(!current.pieces.is_empty()));
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room {
                if !current.pieces.is_empty() {
                    current.pieces.push(Piece {
                        text: String::new(),
                        word_index,
                    });
                }
                current.pieces.push(Piece {
                    text: word.chars[offset..].iter().collect(),
                    word_index,
                });
                offset = word.chars.len();
                continue;
            }

            let avail = if current.pieces.is_empty() {
                cap
            } else {
                cap.saturating_sub(used)
            };
            let candidate = word
                .opportunities
                .iter()
                .copied()
                .filter(|&p| p > offset && p < word.chars.len() && p - offset + 1 <= avail)
                .last();

            if let Some(pos) = candidate {
                let unused = avail - (pos - offset + 1);
                let accept = current.pieces.is_empty()
                    || unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                        < avail.saturating_mul(avail);
                if accept {
                    if !current.pieces.is_empty() {
                        current.pieces.push(Piece {
                            text: String::new(),
                            word_index,
                        });
                    }
                    let mut fragment: String = word.chars[offset..pos].iter().collect();
                    fragment.push('-');
                    current.pieces.push(Piece {
                        text: fragment,
                        word_index,
                    });
                    lines.push(current);
                    current = Line {
                        indent: options.rest_indent,
                        pieces: Vec::new(),
                    };
                    offset = pos;
                    continue;
                }
            }

            if !current.pieces.is_empty() {
                lines.push(current);
                current = Line {
                    indent: options.rest_indent,
                    pieces: Vec::new(),
                };
                continue;
            }

            if suffix_len > cap {
                let end = offset + cap;
                current.pieces.push(Piece {
                    text: word.chars[offset..end].iter().collect(),
                    word_index,
                });
                lines.push(current);
                current = Line {
                    indent: options.rest_indent,
                    pieces: Vec::new(),
                };
                offset = end;
            } else {
                current.pieces.push(Piece {
                    text: word.chars[offset..].iter().collect(),
                    word_index,
                });
                offset = word.chars.len();
            }
        }
    }

    if !current.pieces.is_empty() {
        lines.push(current);
    }

    let last = lines.len().saturating_sub(1);
    for (i, line) in lines.iter().enumerate() {
        output.push(line.render(options.width, options.justify, i == last, i));
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }
        let valid_side = |j: usize| {
            j < original.len() && original[j].1 != '~' && !is_separator(original[j].1)
        };
        if i == 0 || i + 1 >= original.len() || !valid_side(i - 1) || !valid_side(i + 1) {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut end = i + 1;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                let words = parse_words(&normalized[segment_start..i]);
                process_paragraph(&words, options, &mut output);
                for _ in 1..(end - i) {
                    output.push(String::new());
                }
                segment_start = end;
                i = end;
                continue;
            }
        }
        i += 1;
    }

    let words = parse_words(&normalized[segment_start..]);
    process_paragraph(&words, options, &mut output);
    Ok(output)
}
