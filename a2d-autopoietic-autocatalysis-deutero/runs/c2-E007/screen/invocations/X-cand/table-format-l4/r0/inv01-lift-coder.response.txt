#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Debug, Clone)]
struct Token {
    text: String,
    width: usize,
}

#[derive(Debug, Clone)]
struct Cell {
    lines: Vec<Vec<Token>>,
}

fn overhead(mode: &Mode, n: usize) -> usize {
    match mode {
        Mode::Grid | Mode::Markdown => 3 * n + 1,
        Mode::Compact => 2 * n.saturating_sub(1),
    }
}

fn minimum_column_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn tokenize_line(line: &str, markdown: bool) -> Vec<Token> {
    let trimmed = line.trim_end_matches(' ');
    let mut tokens = Vec::new();
    for ch in trimmed.chars() {
        let text = if markdown {
            match ch {
                '|' => "\\|".to_string(),
                '\\' => "\\\\".to_string(),
                _ => ch.to_string(),
            }
        } else {
            ch.to_string()
        };
        let width = text.chars().count();
        tokens.push(Token { text, width });
    }
    tokens
}

fn validate_and_normalize(rows: &[Vec<String>], markdown: bool) -> Result<Vec<Vec<Cell>>, TableError> {
    let mut result = Vec::with_capacity(rows.len());

    for (row_index, row) in rows.iter().enumerate() {
        let mut cells = Vec::with_capacity(row.len());
        for (column_index, input) in row.iter().enumerate() {
            let mut byte = 0;
            let chars: Vec<(usize, char)> = input.char_indices().collect();
            let mut i = 0;
            while i < chars.len() {
                let (offset, ch) = chars[i];
                if ch == '\r' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                        i += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte: offset,
                    });
                }
                if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte: offset,
                    });
                }
                byte = offset;
                let _ = byte;
                i += 1;
            }

            let normalized = input.replace("\r\n", "\n");
            let lines = normalized
                .split('\n')
                .map(|line| tokenize_line(line, markdown))
                .collect();
            cells.push(Cell { lines });
        }
        result.push(cells);
    }

    Ok(result)
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|token| token.width).sum()
}

fn split_fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.to_vec()];
            }
            let available = width.saturating_sub(1);
            let mut used = 0;
            let mut kept = Vec::new();
            for token in tokens {
                if used + token.width > available {
                    break;
                }
                used += token.width;
                kept.push(token.clone());
            }
            kept.push(Token {
                text: "…".to_string(),
                width: 1,
            });
            vec![kept]
        }
        Overflow::Wrap => {
            let mut fragments = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut used = 0;
                let mut end = start;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                if end == start {
                    end += 1;
                }
                fragments.push(tokens[start..end].to_vec());
                start = end;
            }
            fragments
        }
    }
}

fn render_aligned(
    fragment: &[Token],
    width: usize,
    alignment: &Alignment,
    omit_right: bool,
) -> String {
    let content: String = fragment.iter().map(|t| t.text.as_str()).collect();
    let content_width = token_width(fragment);
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!(
        "{}{}{}",
        " ".repeat(left),
        content,
        " ".repeat(if omit_right { 0 } else { right })
    )
}

fn physical_lines(
    row: &[Cell],
    widths: &[usize],
    alignments: &[Alignment],
    overflows: &[Overflow],
) -> Vec<Vec<Vec<Token>>> {
    let mut fragments_per_cell = Vec::with_capacity(row.len());
    let mut height = 1;

    for (column, cell) in row.iter().enumerate() {
        let mut fragments = Vec::new();
        for line in &cell.lines {
            let parts = split_fragments(line, widths[column], &overflows[column]);
            height = height.max(parts.len() + fragments.len());
            fragments.extend(parts);
        }
        fragments_per_cell.push(fragments);
    }

    let _ = alignments;
    (0..height)
        .map(|index| {
            fragments_per_cell
                .iter()
                .map(|fragments| fragments.get(index).cloned().unwrap_or_default())
                .collect()
        })
        .collect()
}

fn append_data_line(
    output: &mut String,
    fragments: &[Vec<Token>],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            output.push('|');
            for column in 0..widths.len() {
                output.push(' ');
                output.push_str(&render_aligned(
                    &fragments[column],
                    widths[column],
                    &alignments[column],
                    false,
                ));
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
        Mode::Compact => {
            for column in 0..widths.len() {
                if column > 0 {
                    output.push_str("  ");
                }
                output.push_str(&render_aligned(
                    &fragments[column],
                    widths[column],
                    &alignments[column],
                    column + 1 == widths.len(),
                ));
            }
            output.push('\n');
        }
    }
}

fn append_grid_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for width in widths {
        output.push_str(&"-".repeat(width + 2));
        output.push('+');
    }
    output.push('\n');
}

fn append_markdown_marker(output: &mut String, widths: &[usize], alignments: &[Alignment]) {
    output.push('|');
    for (width, alignment) in widths.iter().zip(alignments) {
        let marker = match alignment {
            Alignment::Left => format!(":{}", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        output.push(' ');
        output.push_str(&marker);
        output.push(' ');
        output.push('|');
    }
    output.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_width = minimum_column_width(&options.mode);
    let fixed_overhead = overhead(&options.mode, n);
    let minimum = fixed_overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let normalized = validate_and_normalize(rows, matches!(options.mode, Mode::Markdown))?;

    let mut widths = vec![minimum_width; n];
    for row in &normalized {
        for (column, cell) in row.iter().enumerate() {
            for line in &cell.lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        loop {
            let total = widths.iter().sum::<usize>() + fixed_overhead;
            if total <= options.max_width {
                break;
            }
            let candidate = (0..n)
                .rev()
                .find(|&column| widths[column] > minimum_width)
                .map(|column| {
                    let widest = (0..n)
                        .filter(|&c| widths[c] > minimum_width)
                        .map(|c| widths[c])
                        .max()
                        .unwrap();
                    (column, widest)
                });
            let Some((_, widest)) = candidate else {
                break;
            };
            let column = (0..n)
                .rev()
                .find(|&c| widths[c] > minimum_width && widths[c] == widest)
                .unwrap();
            widths[column] -= 1;
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut output, &widths);
    }

    for (row_index, row) in normalized.iter().enumerate() {
        let lines = physical_lines(row, &widths, &alignments, &overflows);
        for fragments in &lines {
            append_data_line(
                &mut output,
                fragments,
                &widths,
                &alignments,
                &options.mode,
            );
        }

        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            append_markdown_marker(&mut output, &widths, &alignments);
        }

        if matches!(options.mode, Mode::Grid) {
            if options.header && row_index == 0 && row_index + 1 < normalized.len() {
                append_grid_border(&mut output, &widths);
            }
        }
    }

    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut output, &widths);
    }

    Ok(output)
}
