use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut core = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        let digits = &bytes[start..pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut n = 0u64;
        for &d in digits {
            n = n.checked_mul(10)
                .and_then(|x| x.checked_add((d - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        core[i] = n;

        if i < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let start = pos;
        while pos < bytes.len() && bytes[pos] != b'+' {
            pos += 1;
        }
        parse_identifiers(&s[start..pos], start, true, &mut pre)?;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(&s[pos..], pos, false, &mut build)?;
        pos = bytes.len();
    }

    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn valid_ident_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    base: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let mut start = 0;
    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'.')
            .map(|x| start + x)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
        for i in start..end {
            if !valid_ident_byte(bytes[i]) {
                return Err(VersionError::InvalidChar { at: base + i });
            }
        }
        let id = &s[start..end];
        if is_pre
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        out.push(id.to_string());

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
    }
    Ok(())
}

fn cmp_ident(a: &str, b: &str) -> Ordering {
    let an = a.bytes().all(|x| x.is_ascii_digit());
    let bn = b.bytes().all(|x| x.is_ascii_digit());
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let c = cmp_ident(x, y);
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Check {
    version: Version,
    op: Op,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}
fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}
fn version(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version { major, minor, patch, pre, build: Vec::new() }
}
fn zero_pre() -> Vec<String> {
    vec!["0".to_string()]
}
fn check(v: Version, op: Op) -> Check {
    Check { version: v, op }
}
fn admit_anchor(a: &mut Arm, v: &Version) {
    if !v.pre.is_empty() && !a.anchors.contains(&core(v)) {
        a.anchors.push(core(v));
    }
}

fn split_tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) { i += 1; }
        if i == bytes.len() { break; }
        let start = i;
        while i < bytes.len() && !ws(bytes[i]) { i += 1; }
        out.push((s[start..i].to_string(), base + start));
    }
    out
}

fn parse_operand_full(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|e| RangeError::Version { at, error: e })
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}
impl Partial {
    fn k(&self) -> usize { self.nums.len() }
    fn padded(&self) -> (u64,u64,u64) {
        (self.nums.get(0).copied().unwrap_or(0),
         self.nums.get(1).copied().unwrap_or(0),
         self.nums.get(2).copied().unwrap_or(0))
    }
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|x| matches!(*x, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        return Ok(Partial { nums: Vec::new(), full: Some(parse_operand_full(s, at)?) });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() { return Err(RangeError::Syntax { at }); }
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.bytes().all(|b| b.is_ascii_digit()) ||
                (f.len() > 1 && f.as_bytes()[0] == b'0') {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial { nums, full: None })
}

fn increment_tuple(t: (u64,u64,u64), field: usize, at: usize) -> Result<(u64,u64,u64), RangeError> {
    let mut x = t;
    let value = match field { 0 => &mut x.0, 1 => &mut x.1, _ => &mut x.2 };
    *value = value.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field <= 0 { x.1 = 0; x.2 = 0; }
    else if field == 1 { x.2 = 0; }
    Ok(x)
}

fn next_partial(p: &Partial, at: usize) -> Result<Version, RangeError> {
    let t = p.padded();
    let field = if p.k() == 1 { 0 } else { 1 };
    let n = increment_tuple(t, field, at)?;
    Ok(version(n.0,n.1,n.2,zero_pre()))
}

fn lower_partial(p: &Partial) -> Version {
    if let Some(v) = &p.full { return v.clone(); }
    let (a,b,c) = p.padded();
    version(a,b,c,Vec::new())
}

fn parse_bare(token: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let p = parse_partial(token, at)?;
    if let Some(v) = p.full {
        admit_anchor(arm, &v);
        arm.checks.push(check(v, Op::Eq));
    } else if p.k() == 1 || p.k() == 2 {
        let lo = lower_partial(&p);
        arm.checks.push(check(lo, Op::Ge));
        arm.checks.push(check(next_partial(&p, at)?, Op::Lt));
    }
    Ok(())
}

fn parse_caret_tilde(token: &str, at: usize, arm: &mut Arm, caret: bool) -> Result<(), RangeError> {
    let operand = &token[1..];
    let oat = at + 1;
    let p = parse_partial(operand, oat)?;
    if p.k() == 0 && p.full.is_none() {
        return Err(RangeError::Syntax { at });
    }
    let lo = lower_partial(&p);
    if let Some(v) = &p.full { admit_anchor(arm, v); }
    arm.checks.push(check(lo.clone(), Op::Ge));

    let k = if p.full.is_some() { 3 } else { p.k() };
    let t = p.padded();
    let field = if caret {
        if t.0 > 0 || k == 1 { 0 }
        else if t.1 > 0 || k == 2 { 1 }
        else { 2 }
    } else if k == 1 { 0 } else { 1 };
    let upper = increment_tuple(t, field, oat)?;
    arm.checks.push(check(version(upper.0,upper.1,upper.2,zero_pre()), Op::Lt));
    Ok(())
}

fn parse_hyphen(tokens: &[(String,usize)], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = tokens.iter().find(|x| x.0 == "-").unwrap().1;
    if tokens.len() != 3 || tokens[1].0 != "-" {
        return Err(RangeError::Syntax { at: minus });
    }
    let lp = parse_partial(&tokens[0].0, tokens[0].1)?;
    let up = parse_partial(&tokens[2].0, tokens[2].1)?;
    if lp.k() == 0 && lp.full.is_none() {
        return Err(RangeError::Syntax { at: tokens[0].1 });
    }
    if up.k() == 0 && up.full.is_none() {
        return Err(RangeError::Syntax { at: tokens[2].1 });
    }
    let lv = lower_partial(&lp);
    if let Some(v) = &lp.full { admit_anchor(arm, v); }
    arm.checks.push(check(lv, Op::Ge));

    let upper = if let Some(v) = up.full {
        admit_anchor(arm, &v);
        v
    } else if up.k() == 3 {
        lower_partial(&up)
    } else {
        next_partial(&up, tokens[2].1)?
    };
    arm.checks.push(check(upper, if up.full.is_some() || up.k() == 3 { Op::Le } else { Op::Lt }));
    Ok(())
}

fn parse_arm(s: &str, base: usize) -> Result<Arm, RangeError> {
    let tokens = split_tokens(s, base);
    let mut arm = Arm { checks: Vec::new(), anchors: Vec::new() };
    if tokens.iter().any(|x| x.0 == "-") {
        parse_hyphen(&tokens, &mut arm)?;
        return Ok(arm);
    }
    for (tok, at) in tokens {
        if tok.is_empty() { continue; }
        if tok.starts_with('^') || tok.starts_with('~') {
            parse_caret_tilde(&tok, at, &mut arm, tok.starts_with('^'))?;
        } else if matches!(tok.as_bytes()[0], b'<' | b'>' | b'=' ) {
            let (op, n) = if tok.starts_with(">=") { (Op::Ge,2) }
                else if tok.starts_with("<=") { (Op::Le,2) }
                else if tok.starts_with('>') { (Op::Gt,1) }
                else if tok.starts_with('<') { (Op::Lt,1) }
                else { (Op::Eq,1) };
            let operand = &tok[n..];
            let oat = at + n;
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = parse_operand_full(operand, oat)?;
            admit_anchor(&mut arm, &v);
            arm.checks.push(check(v, op));
        } else if tok.as_bytes()[0].is_ascii_digit() || matches!(tok.as_str(), "*" | "x" | "X") {
            parse_bare(&tok, at, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    let mut saw_any = false;
    while i < bytes.len() {
        if ws(bytes[i]) { i += 1; continue; }
        if i + 1 < bytes.len() && bytes[i] == b'|' && bytes[i+1] == b'|' {
            let before = &s[arm_start..i];
            if split_tokens(before, arm_start).is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(before, arm_start)?);
            saw_any = true;
            i += 2;
            while i < bytes.len() && ws(bytes[i]) { i += 1; }
            arm_start = i;
            continue;
        }
        i += 1;
    }
    let tail = &s[arm_start..];
    if split_tokens(tail, arm_start).is_empty() {
        if saw_any { return Err(RangeError::Empty { at: s.len() }); }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(tail, arm_start)?);
    Ok(Range { arms })
}

fn check_passes(v: &Version, c: &Check) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}
fn arm_matches(v: &Version, a: &Arm, include: bool) -> bool {
    if !a.checks.iter().all(|c| check_passes(v,c)) { return false; }
    include || v.pre.is_empty() || a.anchors.contains(&core(v))
}
pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v,a,include_prerelease))
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v,r,false) }

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates,r,true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates,r,false)
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }
fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (i,s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index:i,error })?);
    }
    let mut best: Option<usize> = None;
    for i in 0..candidates.len() {
        if !matches(&parsed[i],r) { continue; }
        if let Some(j) = best {
            let c = compare_versions(&parsed[i],&parsed[j]);
            if (max && c == Ordering::Greater) || (!max && c == Ordering::Less) {
                best = Some(i);
            }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn tuple_version(t:(u64,u64,u64)) -> Version { version(t.0,t.1,t.2,Vec::new()) }
fn compare_core_versions(a:&Version,b:&Version)->Ordering { core(a).cmp(&core(b)) }

fn interval(a:&Arm)->(Option<(Version,bool)>,Option<(Version,bool)>) {
    let mut lo:Option<(Version,bool)>=None;
    let mut hi:Option<(Version,bool)>=None;
    for c in &a.checks {
        match c.op {
            Op::Eq => {
                lo=Some((c.version.clone(),true));
                hi=Some((c.version.clone(),true));
            }
            Op::Ge|Op::Gt => {
                let inc=matches!(c.op,Op::Ge);
                if lo.as_ref().map_or(true,|(v,old)| compare_versions(&c.version,v)==Ordering::Greater ||
                    (compare_versions(&c.version,v)==Ordering::Equal && !inc && *old)) {
                    lo=Some((c.version.clone(),inc));
                }
            }
            Op::Le|Op::Lt => {
                let inc=matches!(c.op,Op::Le);
                if hi.as_ref().map_or(true,|(v,old)| compare_versions(&c.version,v)==Ordering::Less ||
                    (compare_versions(&c.version,v)==Ordering::Equal && !inc && *old)) {
                    hi=Some((c.version.clone(),inc));
                }
            }
        }
    }
    (lo,hi)
}

fn stable_exists(lo:&Option<(Version,bool)>,hi:&Option<(Version,bool)>)->bool {
    let min=tuple_version((0,0,0));
    let max=tuple_version((u64::MAX,u64::MAX,u64::MAX));
    let low=lo.as_ref().map(|(v,inc)| (v.clone(),*inc)).unwrap_or((min,true));
    let high=hi.as_ref().map(|(v,inc)| (v.clone(),*inc)).unwrap_or((max,true));
    if compare_versions(&low.0,&high.0)==Ordering::Greater { return false; }
    if compare_versions(&low.0,&high.0)==Ordering::Equal {
        return low.1 && high.1 && low.0.pre.is_empty();
    }
    if low.0.pre.is_empty() {
        if compare_core_versions(&low.0,&high.0)==Ordering::Less { return true; }
        return high.1 && high.0.pre.is_empty();
    }
    // The next stable release after any prerelease is its own tuple.
    let release=tuple_version(core(&low.0));
    let cmp=compare_versions(&release,&low.0);
    if cmp==Ordering::Greater {
        let cl=compare_versions(&release,&high.0);
        if cl==Ordering::Less || (cl==Ordering::Equal && high.1 && high.0.pre.is_empty()) { return true; }
    }
    false
}

fn interval_possible(a:&Arm,b:&Arm,include:bool)->bool {
    let (al,ah)=interval(a);
    let (bl,bh)=interval(b);
    let lo=match (al,bl) {
        (Some(x),Some(y)) => Some(if compare_versions(&x.0,&y.0)==Ordering::Greater ||
            (compare_versions(&x.0,&y.0)==Ordering::Equal && !x.1) { x } else { y }),
        (x,None)|(None,x)=>x,
    };
    let hi=match (ah,bh) {
        (Some(x),Some(y)) => Some(if compare_versions(&x.0,&y.0)==Ordering::Less ||
            (compare_versions(&x.0,&y.0)==Ordering::Equal && !x.1) { x } else { y }),
        (x,None)|(None,x)=>x,
    };
    if stable_exists(&lo,&hi) { return true; }
    if include {
        return match (&lo,&hi) {
            (Some((l,li)),Some((h,hi))) => {
                let o=compare_versions(l,h);
                o==Ordering::Less || (o==Ordering::Equal && *li && *hi)
            }
            _=>true
        };
    }
    let mut anchors=a.anchors.clone();
    anchors.retain(|x| b.anchors.contains(x));
    anchors.into_iter().any(|t| {
        let v=version(t.0,t.1,t.2,zero_pre());
        let lower=lo.as_ref().map_or(true,|(x,inc)| {
            let o=compare_versions(&v,x); o==Ordering::Greater || (o==Ordering::Equal && *inc)
        });
        let upper=hi.as_ref().map_or(true,|(x,inc)| {
            let o=compare_versions(&v,x); o==Ordering::Less || (o==Ordering::Equal && *inc)
        });
        lower&&upper
    })
}
pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool {
    a.arms.iter().any(|x| b.arms.iter().any(|y| interval_possible(x,y,include_prerelease)))
}

fn canon_bound(v:&Version, inclusive:bool, lower:bool)->String {
    let mut s=String::new();
    s.push_str(if lower { if inclusive { ">=" } else { ">" } } else { if inclusive { "<=" } else { "<" } });
    s.push_str(&format!("{}.{}.{}",v.major,v.minor,v.patch));
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}
fn arm_canonical(a:&Arm)->Option<String> {
    let (lo,hi)=interval(a);
    if !stable_exists(&lo,&hi) && !a.anchors.iter().any(|t| {
        let v=version(t.0,t.1,t.2,zero_pre());
        lo.as_ref().map_or(true,|(x,i)| compare_versions(&v,x)==Ordering::Greater || (*i&&compare_versions(&v,x)==Ordering::Equal))
        && hi.as_ref().map_or(true,|(x,i)| compare_versions(&v,x)==Ordering::Less || (*i&&compare_versions(&v,x)==Ordering::Equal))
    }) { return None; }
    let l=lo.map(|(mut v,i)| {v.build.clear();canon_bound(&v,i,true)}).unwrap_or("*".into());
    let h=hi.map(|(mut v,i)| {v.build.clear();canon_bound(&v,i,false)}).unwrap_or("*".into());
    let mut anchors=a.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let an=if anchors.is_empty() { "-".to_string() } else {
        anchors.iter().map(|t|format!("{}.{}.{}",t.0,t.1,t.2)).collect::<Vec<_>>().join(",")
    };
    Some(format!("[{};{};pre={}]",l,h,an))
}
pub fn canonical(r:&Range)->String {
    let mut out:Vec<String>=r.arms.iter().filter_map(arm_canonical).collect();
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}
