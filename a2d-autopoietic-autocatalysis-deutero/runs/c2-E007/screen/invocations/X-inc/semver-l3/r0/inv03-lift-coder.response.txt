use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.') | Some(b'-') | Some(b'+'))
}

fn parse_digits(s: &str, start: usize) -> Result<(u64, usize), VersionError> {
    let bytes = s.as_bytes();
    let mut end = start;
    while end < bytes.len() && bytes[end].is_ascii_digit() {
        end += 1;
    }
    if end == start {
        if is_core_terminator(bytes.get(start).copied()) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = s[start..end]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok((value, end))
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;
    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            if !(bytes[i].is_ascii_alphanumeric() || bytes[i] == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..i];
        if prerelease
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_string());
        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let (major, mut pos) = parse_digits(s, 0)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        if is_core_terminator(s.as_bytes().get(pos).copied()) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let (minor, next) = parse_digits(s, pos)?;
    pos = next;
    if s.as_bytes().get(pos) != Some(&b'.') {
        if is_core_terminator(s.as_bytes().get(pos).copied()) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let (patch, mut pos) = parse_digits(s, pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if s.as_bytes().get(pos) == Some(&b'-') {
        let start = pos + 1;
        let end = s[start..]
            .find('+')
            .map(|n| start + n)
            .unwrap_or(s.len());
        pre = parse_identifiers(s, start, end, true)?;
        pos = end;
    }
    if s.as_bytes().get(pos) == Some(&b'+') {
        let start = pos + 1;
        build = parse_identifiers(s, start, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn split_tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        out.push((s[start..i].to_string(), base + start));
    }
    out
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields(Vec<Option<u64>>),
}

impl Partial {
    fn numeric_count(&self) -> usize {
        match self {
            Partial::Full(_) => 3,
            Partial::Fields(f) => f.iter().take_while(|x| x.is_some()).count(),
        }
    }

    fn lower(&self) -> Version {
        match self {
            Partial::Full(v) => v.clone(),
            Partial::Fields(f) => Version {
                major: f.first().copied().flatten().unwrap_or(0),
                minor: f.get(1).copied().flatten().unwrap_or(0),
                patch: f.get(2).copied().flatten().unwrap_or(0),
                pre: Vec::new(),
                build: Vec::new(),
            },
        }
    }
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let full_rule = s.contains('-')
        || s.contains('+')
        || (s.split('.').count() >= 3
            && !s.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if full_rule {
        return parse_version(s)
            .map(Partial::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut result = Vec::new();
    let mut wildcard = false;
    for field in fields {
        if matches!(field, "*" | "x" | "X") {
            wildcard = true;
            result.push(None);
        } else {
            if wildcard
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            result.push(Some(n));
        }
    }
    Ok(Partial::Fields(result))
}

fn make_version(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build: Vec::new(),
    }
}

fn increment_bound(p: &Partial, mode: &str, at: usize) -> Result<Version, RangeError> {
    let lower = p.lower();
    let k = p.numeric_count();
    let (mut major, mut minor, mut patch) = tuple(&lower);
    let bump_major = match mode {
        "caret" => major > 0 || k == 1,
        "tilde" => k == 1,
        "partial" => k == 1,
        _ => false,
    };
    let bump_minor = match mode {
        "caret" => !bump_major && (minor > 0 || k == 2),
        "tilde" => !bump_major,
        "partial" => k == 2,
        _ => false,
    };
    if bump_major {
        major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
        minor = 0;
        patch = 0;
    } else if bump_minor {
        minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
        patch = 0;
    } else {
        patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    }
    Ok(make_version(major, minor, patch, vec!["0".to_string()]))
}

fn push_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(tuple(&version));
    }
    arm.comparators.push(Comparator {
        op,
        version,
        anchor,
    });
}

fn add_partial(arm: &mut Arm, p: Partial, kind: &str, at: usize) -> Result<(), RangeError> {
    match kind {
        "bare" => match p {
            Partial::Full(v) => push_cmp(arm, Op::Eq, v, true),
            Partial::Fields(f) => {
                let k = f.iter().take_while(|x| x.is_some()).count();
                if k == 0 {
                    return Ok(());
                }
                let lo = Partial::Fields(f);
                let lower = lo.lower();
                push_cmp(arm, Op::Ge, lower, false);
                if k < 3 {
                    let upper = increment_bound(&lo, "partial", at)?;
                    push_cmp(arm, Op::Lt, upper, false);
                }
            }
        },
        "caret" | "tilde" => {
            if p.numeric_count() == 0 {
                return Err(RangeError::Syntax { at });
            }
            let lower = p.lower();
            let upper = increment_bound(&p, kind, at)?;
            push_cmp(arm, Op::Ge, lower, true);
            push_cmp(arm, Op::Lt, upper, false);
        }
        _ => {}
    }
    Ok(())
}

fn parse_full_comparator(token: &str, at: usize) -> Result<(Op, Version), RangeError> {
    let (op, prefix) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else if token.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &token[prefix..];
    let operand_at = at + prefix;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let v = parse_version(operand)
        .map_err(|error| RangeError::Version { at: operand_at, error })?;
    Ok((op, v))
}

fn parse_arm(tokens: &[(String, usize)], full_s: &str) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    let hyphens: Vec<usize> = tokens
        .iter()
        .enumerate()
        .filter_map(|(i, (t, _))| if t == "-" { Some(i) } else { None })
        .collect();
    if !hyphens.is_empty() {
        let h = hyphens[0];
        if hyphens.len() != 1 || h != 1 || tokens.len() != 3 {
            return Err(RangeError::Syntax { at: tokens[h].1 });
        }
        let (lo_s, lo_at) = (&tokens[0].0, tokens[0].1);
        let (hi_s, hi_at) = (&tokens[2].0, tokens[2].1);
        let lo = parse_partial(lo_s, lo_at)?;
        let hi = parse_partial(hi_s, hi_at)?;
        if lo.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: lo_at });
        }
        if hi.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: hi_at });
        }
        let lower = lo.lower();
        push_cmp(&mut arm, Op::Ge, lower, true);
        match hi {
            Partial::Full(v) => push_cmp(&mut arm, Op::Le, v, true),
            Partial::Fields(f) => {
                let k = f.iter().take_while(|x| x.is_some()).count();
                let p = Partial::Fields(f);
                if k == 3 {
                    push_cmp(&mut arm, Op::Le, p.lower(), false);
                } else {
                    let upper = increment_bound(&p, "partial", hi_at)?;
                    push_cmp(&mut arm, Op::Lt, upper, false);
                }
            }
        }
        return Ok(arm);
    }

    for (token, at) in tokens {
        if token.starts_with('^') || token.starts_with('~') {
            let kind = if token.starts_with('^') { "caret" } else { "tilde" };
            let operand = &token[1..];
            let op_at = *at + 1;
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: *at });
            }
            let p = parse_partial(operand, op_at)?;
            add_partial(&mut arm, p, kind, op_at)?;
        } else if token.starts_with('<')
            || token.starts_with('>')
            || token.starts_with('=')
        {
            let (op, v) = parse_full_comparator(token, *at)?;
            push_cmp(&mut arm, op, v, true);
        } else if token.starts_with(|c: char| c.is_ascii_digit())
            || matches!(token.as_str(), "*" | "x" | "X")
        {
            let p = parse_partial(token, *at)?;
            add_partial(&mut arm, p, "bare", *at)?;
        } else {
            return Err(RangeError::Syntax { at: *at });
        }
    }
    let _ = full_s;
    Ok(arm)
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;
    let mut saw_non_ws = false;
    while i < bytes.len() {
        if i + 1 < bytes.len() && bytes[i] == b'|' && bytes[i + 1] == b'|' {
            let standalone_left = i == 0 || is_ws(bytes[i - 1]);
            let standalone_right = i + 2 == bytes.len() || is_ws(bytes[i + 2]);
            if standalone_left && standalone_right {
                let segment = &s[start..i];
                let tokens = split_tokens(segment, start);
                if tokens.is_empty() {
                    if !saw_non_ws {
                        return Err(RangeError::Empty { at: 0 });
                    }
                    return Err(RangeError::Empty { at: i });
                }
                arms.push(parse_arm(&tokens, segment)?);
                saw_non_ws = true;
                i += 2;
                start = i;
                continue;
            }
        }
        i += 1;
    }
    let segment = &s[start..];
    let tokens = split_tokens(segment, start);
    if tokens.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&tokens, segment)?);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !include_prerelease && !v.pre.is_empty() && !arm.anchors.contains(&tuple(v)) {
        return false;
    }
    arm.comparators.iter().all(|c| cmp_matches(v, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn version_string(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn core_successor(v: &Version) -> Option<Version> {
    if v.patch < u64::MAX {
        Some(make_version(v.major, v.minor, v.patch + 1, Vec::new()))
    } else if v.minor < u64::MAX {
        Some(make_version(v.major, v.minor + 1, 0, Vec::new()))
    } else if v.major < u64::MAX {
        Some(make_version(v.major + 1, 0, 0, Vec::new()))
    } else {
        None
    }
}

fn lower_cmp(a: &Comparator, b: &Comparator) -> Comparator {
    match compare_versions(&a.version, &b.version) {
        Ordering::Greater => a.clone(),
        Ordering::Less => b.clone(),
        Ordering::Equal => match (&a.op, &b.op) {
            (Op::Gt, _) | (_, Op::Gt) => {
                let mut c = a.clone();
                c.op = Op::Gt;
                c
            }
            _ => {
                let mut c = a.clone();
                c.op = Op::Ge;
                c
            }
        },
    }
}

fn upper_cmp(a: &Comparator, b: &Comparator) -> Comparator {
    match compare_versions(&a.version, &b.version) {
        Ordering::Less => a.clone(),
        Ordering::Greater => b.clone(),
        Ordering::Equal => match (&a.op, &b.op) {
            (Op::Lt, _) | (_, Op::Lt) => {
                let mut c = a.clone();
                c.op = Op::Lt;
                c
            }
            _ => {
                let mut c = a.clone();
                c.op = Op::Le;
                c
            }
        },
    }
}

fn arm_bounds(arm: &Arm) -> (Option<Comparator>, Option<Comparator>, bool) {
    let mut lower: Option<Comparator> = None;
    let mut upper: Option<Comparator> = None;
    let mut impossible = false;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                let mut lo = c.clone();
                lo.op = Op::Ge;
                let mut hi = c.clone();
                hi.op = Op::Le;
                lower = Some(match lower {
                    Some(ref old) => lower_cmp(old, &lo),
                    None => lo,
                });
                upper = Some(match upper {
                    Some(ref old) => upper_cmp(old, &hi),
                    None => hi,
                });
            }
            Op::Ge | Op::Gt => {
                lower = Some(match lower {
                    Some(ref old) => lower_cmp(old, c),
                    None => c.clone(),
                });
            }
            Op::Le | Op::Lt => {
                upper = Some(match upper {
                    Some(ref old) => upper_cmp(old, c),
                    None => c.clone(),
                });
            }
        }
    }
    if let (Some(lo), Some(hi)) = (&lower, &upper) {
        let ord = compare_versions(&lo.version, &hi.version);
        impossible = ord == Ordering::Greater
            || (ord == Ordering::Equal && (matches!(lo.op, Op::Gt) || matches!(hi.op, Op::Lt)));
    }
    (lower, upper, impossible)
}

fn canonical_arm(arm: &Arm) -> Option<String> {
    let (lower, upper, impossible) = arm_bounds(arm);
    if impossible {
        return None;
    }
    let anchors: Vec<(u64, u64, u64)> = {
        let mut a = arm.anchors.clone();
        a.sort();
        a.dedup();
        a
    };
    let anchor_str = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors
            .iter()
            .map(|(a, b, c)| format!("{a}.{b}.{c}"))
            .collect::<Vec<_>>()
            .join(",")
    };
    let lower_str = lower
        .map(|c| {
            format!(
                "{}{}",
                if matches!(c.op, Op::Gt) { ">" } else { ">=" },
                version_string(&c.version)
            )
        })
        .unwrap_or_else(|| "*".to_string());
    let upper_str = upper
        .map(|c| {
            format!(
                "{}{}",
                if matches!(c.op, Op::Lt) { "<" } else { "<=" },
                version_string(&c.version)
            )
        })
        .unwrap_or_else(|| "*".to_string());
    Some(format!("[{lower_str};{upper_str};pre={anchor_str}]"))
}

pub fn canonical(r: &Range) -> String {
    let mut strings: Vec<String> = r.arms.iter().filter_map(canonical_arm).collect();
    strings.sort();
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".to_string()
    } else {
        strings.join(" || ")
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut combined = aa.clone();
            combined.comparators.extend(bb.comparators.clone());
            combined.anchors = aa
                .anchors
                .iter()
                .copied()
                .filter(|x| bb.anchors.contains(x))
                .collect();
            if include_prerelease {
                combined.anchors.clear();
            }
            if !arm_bounds(&combined).2 {
                if combined.comparators.is_empty() {
                    return true;
                }
                let (lo, hi, _) = arm_bounds(&combined);
                if let (Some(l), Some(h)) = (lo, hi) {
                    let ord = compare_versions(&l.version, &h.version);
                    if ord == Ordering::Less {
                        return true;
                    }
                    if ord == Ordering::Equal
                        && !matches!(l.op, Op::Gt)
                        && !matches!(h.op, Op::Lt)
                    {
                        if l.version.pre.is_empty() || include_prerelease {
                            return true;
                        }
                        if combined.anchors.contains(&tuple(&l.version)) {
                            return true;
                        }
                    }
                    if ord == Ordering::Less {
                        return true;
                    }
                } else if lo.is_some() || hi.is_some() {
                    return true;
                } else {
                    return true;
                }
            }
        }
    }
    false
}
