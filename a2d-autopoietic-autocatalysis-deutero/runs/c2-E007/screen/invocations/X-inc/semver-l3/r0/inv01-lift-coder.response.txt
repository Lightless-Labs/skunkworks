use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(b: u8) -> bool {
    matches!(b, b'.' | b'-' | b'+')
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        if start == bytes.len() || is_core_missing_byte(bytes[start]) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut n = 0u64;
    for &b in &bytes[start..*pos] {
        let d = (b - b'0') as u64;
        n = match n.checked_mul(10).and_then(|x| x.checked_add(d)) {
            Some(v) => v,
            None => return Err(VersionError::Overflow { at: start }),
        };
    }
    Ok(n)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if is_pre {
            let id = &bytes[id_start..p];
            if id.iter().all(u8::is_ascii_digit) && id.len() > 1 && id[0] == b'0' {
                return Err(VersionError::LeadingZero { at: id_start });
            }
        }
        out.push(s[id_start..p].to_string());
        if p == end {
            break;
        }
        p += 1;
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    for _ in 0..2 {
        if p >= s.len() || s.as_bytes()[p] != b'.' {
            if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: p });
            }
            return Err(VersionError::InvalidChar { at: p });
        }
        p += 1;
        if _ == 0 {}
    }
    // Reparse explicitly to preserve the required field-by-field error order.
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    if p < s.len() && !matches!(s.as_bytes()[p], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: p });
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let start = p + 1;
        let plus = s[start..].find('+').map(|x| start + x);
        let end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, start, end, true)?;
        p = end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let start = p + 1;
        build = parse_identifiers(s, start, s.len(), false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = is_numeric_id(x);
        let yn = is_numeric_id(y);
        let o = match (xn, yn) {
            (true, true) => {
                x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes()))
            }
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if o != Ordering::Equal {
            return o;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    clauses: Vec<Clause>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Clause {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let b = s.as_bytes();
    let mut p = 0;
    let mut out = Vec::new();
    while p < b.len() {
        while p < b.len() && ws(b[p]) { p += 1; }
        if p == b.len() { break; }
        let st = p;
        while p < b.len() && !ws(b[p]) { p += 1; }
        out.push((s[st..p].to_string(), base + st));
    }
    out
}

fn version_clause(op: Op, text: &str, at: usize, offset: usize) -> Result<Clause, RangeError> {
    let v = parse_version(text).map_err(|error| RangeError::Version { at: offset, error })?;
    let _ = at;
    Ok(Clause { op, version: v })
}

fn tuple_version(t: (u64, u64, u64)) -> Version {
    Version { major: t.0, minor: t.1, patch: t.2, pre: vec![], build: vec![] }
}

fn bump_major(v: &Version, at: usize) -> Result<Version, RangeError> {
    let major = v.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
    Ok(tuple_version((major, 0, 0)))
}
fn bump_minor(v: &Version, at: usize) -> Result<Version, RangeError> {
    let minor = v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
    Ok(tuple_version((v.major, minor, 0)))
}
fn bump_patch(v: &Version, at: usize) -> Result<Version, RangeError> {
    let patch = v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    Ok(tuple_version((v.major, v.minor, patch)))
}

fn partial_fields(s: &str, at: usize) -> Result<(Vec<Option<u64>>, bool), RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut vals = Vec::new();
    let mut wild = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wild = true;
            vals.push(None);
        } else {
            if wild || !f.bytes().all(|b| b.is_ascii_digit()) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            vals.push(Some(n));
        }
    }
    Ok((vals, wild))
}

fn parse_partial_operand(text: &str, at: usize) -> Result<(Version, usize, bool), RangeError> {
    let fullish = text.contains('-') || text.contains('+')
        || (text.split('.').count() >= 3 && !text.split('.').any(|x| matches!(x, "*" | "x" | "X")));
    if fullish {
        let v = parse_version(text).map_err(|error| RangeError::Version { at, error })?;
        return Ok((v, 3, true));
    }
    let (fields, _) = partial_fields(text, at)?;
    let k = fields.iter().take_while(|x| x.is_some()).count();
    let mut vals = [0u64; 3];
    for i in 0..k { vals[i] = fields[i].unwrap(); }
    Ok((tuple_version((vals[0], vals[1], vals[2])), k, false))
}

fn add_partial_bounds(arm: &mut Arm, text: &str, at: usize, mode: char) -> Result<(), RangeError> {
    let (v, k, full) = parse_partial_operand(text, at)?;
    if mode != 'b' && k == 0 { return Err(RangeError::Syntax { at }); }
    if full {
        if mode == 'b' {
            arm.clauses.push(Clause { op: Op::Eq, version: v.clone() });
        } else {
            arm.clauses.push(Clause { op: Op::Ge, version: v.clone() });
            let upper = if mode == '^' {
                if v.major > 0 || k == 1 { bump_major(&v, at)? }
                else if v.minor > 0 || k == 2 { bump_minor(&v, at)? }
                else { bump_patch(&v, at)? }
            } else {
                if k == 1 { bump_major(&v, at)? } else { bump_minor(&v, at)? }
            };
            arm.clauses.push(Clause { op: Op::Lt, version: upper });
        }
        if !v.pre.is_empty() { arm.anchors.push((v.major,v.minor,v.patch)); }
        return Ok(());
    }
    if mode == 'b' && k == 0 { return Ok(()); }
    arm.clauses.push(Clause { op: Op::Ge, version: v.clone() });
    let upper = if mode == '^' {
        if v.major > 0 || k == 1 { bump_major(&v, at)? }
        else if v.minor > 0 || k == 2 { bump_minor(&v, at)? }
        else { bump_patch(&v, at)? }
    } else if mode == '~' {
        if k == 1 { bump_major(&v, at)? } else { bump_minor(&v, at)? }
    } else if k == 1 {
        bump_major(&v, at)?
    } else {
        bump_minor(&v, at)?
    };
    arm.clauses.push(Clause { op: Op::Lt, version: upper });
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { clauses: vec![], anchors: vec![] };
    let hyphens: Vec<usize> = tokens.iter().enumerate().filter(|(_,x)| x.0 == "-").map(|(i,_)| i).collect();
    if !hyphens.is_empty() {
        if tokens.len() != 3 || hyphens[0] != 1 {
            return Err(RangeError::Syntax { at: tokens[hyphens[0]].1 });
        }
        let (lo, _, lf) = parse_partial_operand(&tokens[0].0, tokens[0].1)?;
        let (hi, hk, hf) = parse_partial_operand(&tokens[2].0, tokens[2].1)?;
        if lo.pre.is_empty() && tokens[0].0.split('.').all(|x| matches!(x, "*" | "x" | "X")) {
            return Err(RangeError::Syntax { at: tokens[0].1 });
        }
        if hi.pre.is_empty() && hk == 0 {
            return Err(RangeError::Syntax { at: tokens[2].1 });
        }
        arm.clauses.push(Clause { op: Op::Ge, version: lo.clone() });
        if hf {
            arm.clauses.push(Clause { op: Op::Le, version: hi.clone() });
        } else if hk == 1 {
            arm.clauses.push(Clause { op: Op::Lt, version: bump_major(&hi, tokens[2].1)? });
        } else if hk == 2 {
            arm.clauses.push(Clause { op: Op::Lt, version: bump_minor(&hi, tokens[2].1)? });
        } else {
            arm.clauses.push(Clause { op: Op::Le, version: hi.clone() });
        }
        for v in [lo, hi] {
            if !v.pre.is_empty() { arm.anchors.push((v.major,v.minor,v.patch)); }
        }
        let _ = lf;
        return Ok(arm);
    }
    for (tok, at) in tokens {
        if tok.starts_with(">=") || tok.starts_with("<=") || tok.starts_with('>') || tok.starts_with('<') || tok.starts_with('=') {
            let (op, n) = if tok.starts_with(">=") { (Op::Ge,2) } else if tok.starts_with("<=") { (Op::Le,2) } else if tok.starts_with('>') { (Op::Gt,1) } else if tok.starts_with('<') { (Op::Lt,1) } else { (Op::Eq,1) };
            let operand = &tok[n..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() { return Err(RangeError::Syntax { at: *at }); }
            arm.clauses.push(version_clause(op, operand, *at, *at+n)?);
            if op == Op::Ge || op == Op::Gt || op == Op::Eq || op == Op::Le || op == Op::Lt {
                let v = &arm.clauses.last().unwrap().version;
                if !v.pre.is_empty() { arm.anchors.push((v.major,v.minor,v.patch)); }
            }
            continue;
        }
        if tok.starts_with('^') || tok.starts_with('~') {
            let mode = tok.as_bytes()[0] as char;
            add_partial_bounds(&mut arm, &tok[1..], *at + 1, mode)?;
            continue;
        }
        if !tok.as_bytes()[0].is_ascii_digit() && !matches!(tok.as_str(), "*" | "x" | "X") {
            return Err(RangeError::Syntax { at: *at });
        }
        add_partial_bounds(&mut arm, tok, *at, 'b')?;
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = split_tokens(s, 0);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].0 == "||" {
            if current.is_empty() { return Err(RangeError::Empty { at: tokens[i].1 }); }
            arms.push(parse_arm(&current)?);
            current.clear();
            if i + 1 == tokens.len() {
                let end = s.len();
                return Err(RangeError::Empty { at: end });
            }
        } else {
            current.push(tokens[i].clone());
        }
        i += 1;
    }
    if !current.is_empty() { arms.push(parse_arm(&current)?); }
    Ok(Range { arms })
}

fn clause_matches(v: &Version, c: &Clause) -> bool {
    match compare_versions(v, &c.version) {
        Ordering::Equal => matches!(c.op, Op::Eq | Op::Le | Op::Ge),
        Ordering::Less => matches!(c.op, Op::Lt | Op::Le),
        Ordering::Greater => matches!(c.op, Op::Gt | Op::Ge),
    }
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.clauses.iter().all(|c| clause_matches(v, c)) { return false; }
    if include || v.pre.is_empty() { return true; }
    arm.anchors.contains(&(v.major,v.minor,v.patch))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v, r, false) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extremum(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index: i, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(j) = best {
            let o = compare_versions(&parsed[i], &parsed[j]);
            if (greatest && o == Ordering::Greater) || (!greatest && o == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extremum(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extremum(candidates, r, false)
}

pub fn intersects(_a: &Range, _b: &Range, _include_prerelease: bool) -> bool {
    false
}

pub fn canonical(_r: &Range) -> String {
    "EMPTY".to_string()
}
