#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    const MAX_VERTICES: usize = 4096;
    const COORD_LIMIT: i64 = 1_000_000_000;

    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.last() == points.first() {
        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(effective_len);

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];

        signed_area2 += (a.x as i128) * (b.y as i128) - (a.y as i128) * (b.x as i128);

        let dx = (b.x as i128) - (a.x as i128);
        let dy = (b.y as i128) - (a.y as i128);
        let scaled_squared = ((dx * dx + dy * dy) as u128) * 1_000_000;

        // Integer square root, rounded to nearest with exact half ties upward.
        let floor_root = integer_sqrt(scaled_squared);
        let lower_square = floor_root * floor_root;
        let upper_root = floor_root + 1;
        let upper_square = upper_root * upper_root;
        let rounded = if scaled_squared - lower_square >= upper_square - scaled_squared {
            upper_root
        } else {
            floor_root
        };

        edge_milli.push(rounded as u64);
    }

    let perimeter_milli = edge_milli.iter().copied().sum();

    let mut longest_edge = Some(0usize);
    for i in 1..effective_len {
        let current = longest_edge.unwrap();
        if edge_milli[i] > edge_milli[current]
            || (edge_milli[i] == edge_milli[current]
                && edge_tie_key(polygon, i) < edge_tie_key(polygon, current))
        {
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn edge_tie_key(polygon: &[Point], index: usize) -> (Point, Point) {
    let a = polygon[index];
    let b = polygon[(index + 1) % polygon.len()];
    if a <= b { (a, b) } else { (b, a) }
}

fn integer_sqrt(value: u128) -> u128 {
    if value < 2 {
        return value;
    }

    let mut x = value;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + value / x) / 2;
    }
    x
}
