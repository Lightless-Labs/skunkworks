#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128 - a.x as i128).unsigned_abs();
    let dy = (b.y as i128 - a.y as i128).unsigned_abs();

    // Coordinates are validated to lie within +/- 1e9, so this radicand fits u128.
    let radicand = (dx * dx + dy * dy) * 1_000_000;
    let mut root = (radicand as f64).sqrt() as u128;

    // Correct any floating point approximation using exact integer comparisons.
    while (root + 1) * (root + 1) <= radicand {
        root += 1;
    }
    while root * root > radicand {
        root -= 1;
    }

    let remainder = radicand - root * root;
    if remainder * 2 >= root * 2 + 1 {
        root += 1;
    }

    root as u64
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.last() == points.first() {
        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0_i128;

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }

    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];
        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(previous_index) => {
                let previous_length = edge_milli[previous_index];
                if length > previous_length {
                    true
                } else if length < previous_length {
                    false
                } else {
                    let pair = if a <= b { (a, b) } else { (b, a) };
                    let previous_a = polygon[previous_index];
                    let previous_b = polygon[(previous_index + 1) % effective_len];
                    let previous_pair = if previous_a <= previous_b {
                        (previous_a, previous_b)
                    } else {
                        (previous_b, previous_a)
                    };
                    pair < previous_pair || (pair == previous_pair && i < previous_index)
                }
            }
        };

        if replace {
            longest_edge = Some(i);
        }
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
