#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Prefix {
    Context,
    Delete,
    Add,
}

#[derive(Clone)]
struct Row {
    prefix: Prefix,
    payload: String,
    line: usize,
}

struct Hunk {
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    header_line: usize,
    rows: Vec<Row>,
}

fn physical_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }
    let mut lines: Vec<&str> = text.split_terminator('\n').collect();
    // split_terminator keeps interior empty lines and omits the final empty field.
    // For non-LF-terminated input, it retains the last line.
    if !text.ends_with('\n') {
        // split_terminator already retains the final unterminated field.
    }
    lines.shrink_to_fit();
    lines
}

fn parse_number(bytes: &[u8], pos: &mut usize) -> Option<usize> {
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return None;
    }
    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return None;
    }
    let mut value = 0usize;
    for &digit in digits {
        value = value.checked_mul(10)?.checked_add((digit - b'0') as usize)?;
        if value > 128 {
            return None;
        }
    }
    Some(value)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let bytes = line.as_bytes();
    if !bytes.starts_with(b"@@ -") || !bytes.ends_with(b" @@") {
        return None;
    }
    let mut pos = 4;
    let old_start = parse_number(bytes, &mut pos)?;
    if bytes.get(pos) != Some(&b',') {
        return None;
    }
    pos += 1;
    let old_count = parse_number(bytes, &mut pos)?;
    if bytes.get(pos..pos + 2) != Some(&b" +") {
        return None;
    }
    pos += 2;
    let new_start = parse_number(bytes, &mut pos)?;
    if bytes.get(pos) != Some(&b',') {
        return None;
    }
    pos += 1;
    let new_count = parse_number(bytes, &mut pos)?;
    if bytes.get(pos..) != Some(b" @@") {
        return None;
    }
    if pos + 3 != bytes.len() {
        return None;
    }
    if (old_count > 0 && old_start == 0) || (new_count > 0 && new_start == 0) {
        return None;
    }
    Some((old_start, old_count, new_start, new_count))
}

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 {
        start - 1
    } else {
        start
    }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for (idx, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(idx).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: idx + 1 });
        }
    }
    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end = 0usize;
    let mut cumulative_delta: i64 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((old_start, old_count, new_start, new_count)) =
            parse_hunk_header(lines[i])
        else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_full = full_start(old_start, old_count);
        let new_full = full_start(new_start, new_count);
        let required_new = old_full as i64 + cumulative_delta;

        if old_full < previous_old_end || required_new < 0 || new_full as i64 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let bytes = lines[i].as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            if lines[i].contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = lines[i][1..].to_string();
            let prefix = match bytes[0] {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                    Prefix::Context
                }
                b'-' => {
                    old_seen += 1;
                    Prefix::Delete
                }
                b'+' => {
                    new_seen += 1;
                    Prefix::Add
                }
                _ => unreachable!(),
            };
            rows.push(Row {
                prefix,
                payload,
                line: row_line,
            });
            i += 1;
        }

        if old_seen != old_count || new_seen != new_count {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows
            .iter()
            .any(|row| row.prefix == Prefix::Add || row.prefix == Prefix::Delete)
        {
            return Err(ApplyError::BadBody { line: header_line });
        }

        previous_old_end = old_full + old_count;
        cumulative_delta += new_count as i64 - old_count as i64;
        hunks.push(Hunk {
            old_start,
            old_count,
            new_start,
            new_count,
            header_line,
            rows,
        });

        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let lines = physical_lines(source);
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines.into_iter().map(str::to_owned).collect())
}

fn abs_diff_signed(a: i64, b: i64) -> u64 {
    (a - b).unsigned_abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i64 = 0;
    let mut drift: i64 = 0;

    for (hunk_idx, hunk) in hunks.iter().enumerate() {
        let nominal_old = full_start(hunk.old_start, hunk.old_count) as i64;
        let expected = nominal_old + net + drift;

        let lead = hunk
            .rows
            .iter()
            .take_while(|r| r.prefix == Prefix::Context)
            .count();
        let trail = hunk
            .rows
            .iter()
            .rev()
            .take_while(|r| r.prefix == Prefix::Context)
            .count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);

        let mut best_rank: Option<(usize, u64)> = None;
        let mut best_starts = Vec::<usize>::new();
        let mut best_l_by_start = Vec::<(usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > lead + trail {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows {
                    if row.prefix == Prefix::Add {
                        continue;
                    }
                    old_payloads.push(row.payload.as_str());
                }
                let trimmed_old: Vec<&str> = old_payloads
                    .iter()
                    .skip(l)
                    .take(old_payloads.len().saturating_sub(l + r))
                    .copied()
                    .collect();
                let rank_fuzz = l + r;

                if hunk.old_count > working.len() {
                    continue;
                }
                for s in 0..=working.len() - hunk.old_count {
                    let distance = abs_diff_signed(s as i64, expected);
                    if distance > options.max_offset as u64 {
                        continue;
                    }
                    let begin = s + l;
                    let end = s + hunk.old_count - r;
                    if end > working.len() || begin > end {
                        continue;
                    }
                    if end - begin != trimmed_old.len() {
                        continue;
                    }
                    if !trimmed_old
                        .iter()
                        .enumerate()
                        .all(|(idx, payload)| working[begin + idx] == *payload)
                    {
                        continue;
                    }

                    let rank = (rank_fuzz, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_l_by_start.clear();
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_l_by_start.clear();
                        }
                        Some(current) if rank > current => continue,
                        _ => {}
                    }

                    if !best_starts.contains(&s) {
                        best_starts.push(s);
                    }
                    match best_l_by_start.iter_mut().find(|(start, _)| *start == s) {
                        Some((_, existing_l)) if l < *existing_l => *existing_l = l,
                        None => best_l_by_start.push((s, l)),
                        _ => {}
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_idx + 1,
            });
        }
        best_starts.sort_unstable();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_idx + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let s = best_starts[0];
        let l = best_l_by_start
            .iter()
            .find(|(start, _)| *start == s)
            .map(|(_, l)| *l)
            .unwrap_or(0);

        // The best rank fixes L+R. Select the corresponding R and construct output
        // from the body with the chosen edge context removed.
        let total_fuzz = best_rank.unwrap().0;
        let r = total_fuzz - l;
        let trimmed_rows = &hunk.rows[l..hunk.rows.len() - r];

        let replacement: Vec<String> = trimmed_rows
            .iter()
            .filter(|row| row.prefix != Prefix::Delete)
            .map(|row| row.payload.clone())
            .collect();

        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i64 - hunk.old_count as i64;
        drift = s as i64 - (nominal_old + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
