use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    if start == end {
        return Err(VersionError::EmptyIdentifier { at: start });
    }
    let mut result = Vec::new();
    let mut ident_start = start;

    while ident_start <= end {
        let mut ident_end = ident_start;
        while ident_end < end && bytes[ident_end] != b'.' {
            let b = bytes[ident_end];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: ident_end });
            }
            ident_end += 1;
        }

        if ident_start == ident_end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &s[ident_start..ident_end];
        if prerelease
            && ident.bytes().all(is_digit)
            && ident.len() > 1
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        result.push(ident.to_string());

        if ident_end == end {
            break;
        }
        ident_start = ident_end + 1;
        if ident_start > end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut numbers = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        let field = &s[start..pos];
        if field.len() > 1 && field.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        numbers[i] = field
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;

        if i < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
    }

    Ok(Version {
        major: numbers[0],
        minor: numbers[1],
        patch: numbers[2],
        pre,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_identifier(x);
        let yn = numeric_identifier(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    admission: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

fn is_range_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str, start: usize, end: usize) -> Vec<Token> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && is_range_ws(bytes[i]) {
            i += 1;
        }
        if i >= end {
            break;
        }
        let token_start = i;
        while i < end && !is_range_ws(bytes[i]) {
            i += 1;
        }
        out.push(Token {
            text: s[token_start..i].to_string(),
            start: token_start,
        });
    }
    out
}

fn version_error_at(at: usize, error: VersionError) -> RangeError {
    RangeError::Version { at, error }
}

fn parse_operand_full(token: &Token, operand: &str, operand_start: usize) -> Result<Version, RangeError> {
    parse_version(operand).map_err(|e| version_error_at(operand_start, e))
}

fn parse_operator_prefix(s: &str) -> (Op, usize) {
    if s.starts_with(">=") {
        (Op::Ge, 2)
    } else if s.starts_with("<=") {
        (Op::Le, 2)
    } else if s.starts_with('>') {
        (Op::Gt, 1)
    } else if s.starts_with('<') {
        (Op::Lt, 1)
    } else if s.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    }
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

fn add_comparator(arm: &mut Arm, op: Op, version: Version, explicit_anchor: bool) {
    if explicit_anchor && !version.pre.is_empty() {
        arm.admission
            .push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator { op, version });
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_endpoint(
    major: u64,
    minor: u64,
    patch: u64,
    pre_zero: bool,
) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: if pre_zero { vec!["0".to_string()] } else { Vec::new() },
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    numeric_count: usize,
    full: Option<Version>,
}

fn has_suffix_or_three_numeric(token: &str) -> bool {
    if token.contains('-') || token.contains('+') {
        return true;
    }
    let fields: Vec<&str> = token.split('.').collect();
    fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X"))
}

fn parse_partial(token: &str) -> Result<Partial, ()> {
    if has_suffix_or_three_numeric(token) {
        let v = parse_version(token).map_err(|_| ())?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            numeric_count: 3,
            full: Some(v),
        });
    }

    let fields: Vec<&str> = token.split('.').collect();
    if fields.is_empty() || fields.len() > 3 {
        return Err(());
    }

    let mut parsed = Vec::new();
    let mut numeric_count = 0usize;
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(());
        }
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
            parsed.push(None);
        } else {
            if wildcard_seen
                || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(());
            }
            let n = f.parse::<u64>().map_err(|_| ())?;
            numeric_count += 1;
            parsed.push(Some(n));
        }
    }
    if wildcard_seen && parsed.iter().skip_while(|x| x.is_some()).any(|x| x.is_some()) {
        return Err(());
    }

    Ok(Partial {
        fields: parsed,
        numeric_count,
        full: None,
    })
}

fn partial_padded(p: &Partial) -> (u64, u64, u64) {
    let mut values = [0u64; 3];
    for (i, field) in p.fields.iter().enumerate() {
        if let Some(n) = field {
            values[i] = *n;
        }
    }
    (values[0], values[1], values[2])
}

fn bump_at(
    tuple: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let (mut major, mut minor, mut patch) = tuple;
    match field {
        0 => {
            major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            minor = 0;
            patch = 0;
        }
        1 => {
            minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            patch = 0;
        }
        _ => {
            patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
        }
    }
    Ok(upper_endpoint(major, minor, patch, true))
}

fn add_partial_bare(arm: &mut Arm, p: Partial, at: usize) -> Result<(), RangeError> {
    if let Some(v) = p.full {
        add_comparator(arm, Op::Eq, v, true);
        return Ok(());
    }

    let k = p.numeric_count;
    if k == 0 {
        return Ok(());
    }
    let tuple = partial_padded(&p);
    add_comparator(arm, Op::Ge, endpoint(tuple.0, tuple.1, tuple.2), false);
    if k < 3 {
        let field = if k == 1 { 0 } else { 1 };
        let upper = bump_at(tuple, field, at)?;
        add_comparator(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn add_caret_tilde(
    arm: &mut Arm,
    p: Partial,
    at: usize,
    caret: bool,
) -> Result<(), RangeError> {
    if p.numeric_count == 0 {
        return Err(RangeError::Syntax { at });
    }
    let tuple = partial_padded(&p);
    let lower = if let Some(v) = p.full {
        v
    } else {
        endpoint(tuple.0, tuple.1, tuple.2)
    };
    add_comparator(arm, Op::Ge, lower, true);

    let field = if caret {
        if tuple.0 > 0 || p.numeric_count == 1 {
            0
        } else if tuple.1 > 0 || p.numeric_count == 2 {
            1
        } else {
            2
        }
    } else if p.numeric_count == 1 {
        0
    } else {
        1
    };
    let upper = bump_at(tuple, field, at)?;
    add_comparator(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_full_comparator(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, op_len) = parse_operator_prefix(&token.text);
    let operand = &token.text[op_len..];
    let operand_start = token.start + op_len;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = parse_operand_full(token, operand, operand_start)?;
    add_comparator(arm, op, v, true);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = tokens.iter().find(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || tokens[1].text != "-" {
        return Err(RangeError::Syntax { at: minus.start });
    }

    let lower_token = &tokens[0];
    let upper_token = &tokens[2];
    let lower = parse_partial(&lower_token.text)
        .map_err(|_| RangeError::Syntax { at: lower_token.start })?;
    let upper = parse_partial(&upper_token.text)
        .map_err(|_| RangeError::Syntax { at: upper_token.start })?;

    if lower.numeric_count == 0 {
        return Err(RangeError::Syntax { at: lower_token.start });
    }
    if upper.numeric_count == 0 {
        return Err(RangeError::Syntax { at: upper_token.start });
    }

    let lower_tuple = partial_padded(&lower);
    let lower_version = lower
        .full
        .clone()
        .unwrap_or_else(|| endpoint(lower_tuple.0, lower_tuple.1, lower_tuple.2));
    add_comparator(arm, Op::Ge, lower_version, true);

    if let Some(v) = upper.full {
        add_comparator(arm, Op::Le, v, true);
    } else {
        let upper_tuple = partial_padded(&upper);
        if upper.numeric_count == 3 {
            add_comparator(
                arm,
                Op::Le,
                endpoint(upper_tuple.0, upper_tuple.1, upper_tuple.2),
                false,
            );
        } else {
            let field = if upper.numeric_count == 1 { 0 } else { 1 };
            let bound = bump_at(upper_tuple, field, upper_token.start)?;
            add_comparator(arm, Op::Lt, bound, false);
        }
    }
    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let tokens = tokenize(s, start, end);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: start });
    }

    let mut arm = Arm {
        comparators: Vec::new(),
        admission: Vec::new(),
    };

    if tokens.iter().any(|t| t.text == "-") {
        parse_hyphen_arm(&tokens, &mut arm)?;
        return Ok(arm);
    }

    for token in &tokens {
        let first = token.text.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_full_comparator(token, &mut arm)?;
            continue;
        }

        if first == b'^' || first == b'~' {
            let operand = &token.text[1..];
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: token.start });
            }
            let p = parse_partial(operand)
                .map_err(|_| RangeError::Syntax { at: token.start })?;
            add_caret_tilde(&mut arm, p, token.start + 1, first == b'^')?;
            continue;
        }

        if first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X') {
            let p = parse_partial(&token.text)
                .map_err(|_| RangeError::Syntax { at: token.start })?;
            add_partial_bare(&mut arm, p, token.start)?;
            continue;
        }

        return Err(RangeError::Syntax { at: token.start });
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut i = 0usize;
    let mut saw_separator = false;

    while i < bytes.len() {
        if bytes[i] == b'|' && i + 1 < bytes.len() && bytes[i + 1] == b'|' {
            let before_ws = i == 0 || is_range_ws(bytes[i - 1]);
            let after_ws = i + 2 == bytes.len() || is_range_ws(bytes[i + 2]);
            if before_ws && after_ws {
                let arm = parse_arm(s, arm_start, i)?;
                arms.push(arm);
                saw_separator = true;
                i += 2;
                arm_start = i;
                while arm_start < bytes.len() && is_range_ws(bytes[arm_start]) {
                    arm_start += 1;
                }
                i = arm_start;
                continue;
            }
        }
        i += 1;
    }

    if saw_separator && arm_start == bytes.len() {
        return Err(RangeError::Empty { at: bytes.len() });
    }

    let arm = parse_arm(s, arm_start, bytes.len())?;
    arms.push(arm);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm
                .admission
                .iter()
                .any(|&(a, b, c)| (v.major, v.minor, v.patch) == (a, b, c))
        {
            return false;
        }
        arm.comparators.iter().all(|c| cmp_matches(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let order = compare_versions(version, &parsed[j]);
                if (greatest && order == Ordering::Greater)
                    || (!greatest && order == Ordering::Less)
                {
                    best = Some(i);
                }
            }
        }
    }

    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
