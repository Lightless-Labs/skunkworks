#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
    end: usize,
}

struct Expression<'a> {
    operator: Option<char>,
    variables: Vec<Variable<'a>>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    content_start: usize,
    content_end: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut start = content_start;
    let mut operator = None;

    if start < content_end {
        let candidate = bytes[start] as char;
        if matches!(candidate, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            operator = Some(candidate);
            start += 1;
        } else if matches!(candidate, '!' | '=' | ',' | '|' | '@') {
            return Err(Error::UnsupportedOperator { pos: start });
        }
    }

    if start == content_end {
        return Err(Error::EmptyExpression { pos: content_end });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = start;

    loop {
        let mut spec_end = spec_start;
        while spec_end < content_end && bytes[spec_end] != b',' {
            spec_end += 1;
        }

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        for (offset, byte) in name_bytes.iter().copied().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.iter().any(|existing| *existing == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        variables.push(Variable {
            name,
            start: spec_start,
            end: spec_end,
        });

        if spec_end == content_end {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open = pos;
                let mut close = pos + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }

                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(offset) = bytes[open + 1..close]
                    .iter()
                    .position(|byte| *byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let expression = parse_expression(template, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == b'%' {
            if pos + 2 < bytes.len()
                && hex_value(bytes[pos + 1]).is_some()
                && hex_value(bytes[pos + 2]).is_some()
            {
                output.push('%');
                output.push(HEX[hex_value(bytes[pos + 1]).unwrap() as usize] as char);
                output.push(HEX[hex_value(bytes[pos + 2]).unwrap() as usize] as char);
                pos += 3;
                continue;
            }
            output.push_str("%25");
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }

        pos += 1;
    }

    output
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let separator = match operator {
        Some('.') => '.',
        Some('/') => '/',
        Some(';') => ';',
        Some('?') | Some('&') => '&',
        _ => ',',
    };
    let prefix = match operator {
        Some('.') | Some('/') | Some(';') | Some('?') | Some('&') => operator,
        _ => None,
    };
    let named = matches!(operator, Some(';' | '?' | '&'));
    let allow_reserved = matches!(operator, Some('+' | '#'));

    let mut fields = Vec::new();
    for variable in &expression.variables {
        let _span = (variable.start, variable.end);
        if let Some(value) = lookup(variable.name, vars) {
            let encoded = encode(value, allow_reserved);
            if named {
                if operator == Some(';') && encoded.is_empty() {
                    fields.push(variable.name.to_owned());
                } else {
                    fields.push(format!("{}={}", variable.name, encoded));
                }
            } else {
                fields.push(encoded);
            }
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut output = String::new();
    if let Some(prefix) = prefix {
        output.push(prefix);
    }
    output.push_str(&fields.join(&separator.to_string()));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for (open, after_close, expression) in expressions {
        output.push_str(&template[literal_start..open]);
        output.push_str(&expand_expression(&expression, vars));
        literal_start = after_close;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
