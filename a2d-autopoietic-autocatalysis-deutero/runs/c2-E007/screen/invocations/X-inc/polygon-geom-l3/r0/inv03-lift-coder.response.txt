#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective.last() == effective.first() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    if points.len() >= 2 {
        for i in 0..points.len() {
            let a = points[i];
            let b = points[(i + 1) % points.len()];
            sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
        }
    }
    sum
}

fn edge_mill(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let d2 = (dx * dx + dy * dy) as u128;
    let target = d2 * 1_000_000;
    let mut lo = 0u128;
    let mut hi = 3_000_000_000_000u128;
    while lo < hi {
        let mid = lo + (hi - lo) / 2;
        if mid * mid < target {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    if lo * lo == target {
        lo as u64
    } else {
        let lower = lo - 1;
        if target - lower * lower >= lo * lo - target {
            lo as u64
        } else {
            lower as u64
        }
    }
}

fn perimeter(points: &[Point]) -> (Vec<u64>, u64) {
    let n = points.len();
    if n == 0 {
        return (Vec::new(), 0);
    }
    let mut edges = Vec::with_capacity(n);
    let mut total = 0u64;
    for i in 0..n {
        let length = edge_mill(points[i], points[(i + 1) % n]);
        edges.push(length);
        total += length;
    }
    (edges, total)
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare(points)?;
    let signed_area2 = shoelace(&polygon);
    let area2 = signed_area2.unsigned_abs();
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };
    let (edge_milli, perimeter_milli) = perimeter(&polygon);

    let mut longest_edge = None;
    for i in 0..polygon.len() {
        match longest_edge {
            None => longest_edge = Some(i),
            Some(best) => {
                if edge_milli[i] > edge_milli[best] {
                    longest_edge = Some(i);
                } else if edge_milli[i] == edge_milli[best] {
                    let endpoints = {
                        let a = polygon[i];
                        let b = polygon[(i + 1) % polygon.len()];
                        if a <= b { (a, b) } else { (b, a) }
                    };
                    let best_endpoints = {
                        let a = polygon[best];
                        let b = polygon[(best + 1) % polygon.len()];
                        if a <= b { (a, b) } else { (b, a) }
                    };
                    if endpoints < best_endpoints
                        || (endpoints == best_endpoints && i < best)
                    {
                        longest_edge = Some(i);
                    }
                }
            }
        }
    }

    Ok(Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if on_segment(a, b, query) && boundary.is_none() {
                boundary = Some(i);
            }
            if (a.y > query.y) != (b.y > query.y) {
                let side = cross(a, b, query);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if let Some(edge) = boundary {
            Location::Boundary { edge }
        } else if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

fn distance_on_segment(a: Point, b: Point, p: Point) -> bool {
    on_segment(a, b, p)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(p, _)| p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|&(p, _)| p).collect();
        let source_indices: Vec<usize> = unique.iter().map(|&(_, i)| i).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|&(p, _)| cross(unique[0].0, unique[1].0, p) == 0);

    let hull_points: Vec<Point> = if all_collinear {
        match mode {
            Collinear::Include => unique.iter().map(|&(p, _)| p).collect(),
            Collinear::Exclude => vec![unique[0].0, unique[unique.len() - 1].0],
        }
    } else {
        let mut lower: Vec<(Point, usize)> = Vec::new();
        for &(point, index) in &unique {
            while lower.len() >= 2 {
                let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, point);
                if if mode == Collinear::Include { turn < 0 } else { turn <= 0 } {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push((point, index));
        }
        let mut upper: Vec<(Point, usize)> = Vec::new();
        for &(point, index) in unique.iter().rev() {
            while upper.len() >= 2 {
                let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, point);
                if if mode == Collinear::Include { turn < 0 } else { turn <= 0 } {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push((point, index));
        }
        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower.into_iter().map(|(p, _)| p).collect()
    };

    let source_indices: Vec<usize> = hull_points
        .iter()
        .map(|point| unique.iter().find(|&&(p, _)| p == *point).unwrap().1)
        .collect();
    let area2 = shoelace(&hull_points).unsigned_abs();
    let (_, perimeter_milli) = perimeter(&hull_points);

    // Keep this helper referenced in builds where boundary inclusion is optimized.
    let _ = distance_on_segment;

    Ok(Hull {
        vertices: hull_points,
        source_indices,
        area2,
        perimeter_milli,
    })
}
