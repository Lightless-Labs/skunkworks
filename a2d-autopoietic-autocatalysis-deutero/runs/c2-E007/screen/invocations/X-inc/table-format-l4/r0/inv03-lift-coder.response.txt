#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn invalid_character(ch: char) -> bool {
    (ch.is_control() && ch != '\n' && ch != '\r') || ch == '\u{2028}' || ch == '\u{2029}'
}

fn parse_cell(cell: &str, row: usize, column: usize, markdown: bool) -> Result<Vec<Vec<Token>>, TableError> {
    let mut normalized = String::with_capacity(cell.len());
    let mut iter = cell.char_indices().peekable();

    while let Some((byte, ch)) = iter.next() {
        if ch == '\r' {
            if let Some(&(next_byte, next_ch)) = iter.peek() {
                if next_ch == '\n' {
                    let _ = iter.next();
                    let _ = next_byte;
                    normalized.push('\n');
                    continue;
                }
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }
        if invalid_character(ch) {
            return Err(TableError::InvalidChar { row, column, byte });
        }
        normalized.push(ch);
    }

    let mut lines = Vec::new();
    for line in normalized.split('\n') {
        let trimmed = line.trim_end_matches(' ');
        let mut tokens = Vec::new();
        for ch in trimmed.chars() {
            if markdown && ch == '|' {
                tokens.push(Token { text: "\\|".to_owned(), width: 2 });
            } else if markdown && ch == '\\' {
                tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
            } else {
                tokens.push(Token { text: ch.to_string(), width: 1 });
            }
        }
        lines.push(tokens);
    }
    Ok(lines)
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut out = String::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align_content(content: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let extra = width.saturating_sub(scalar_len(content));
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::new();
    out.extend(std::iter::repeat_n(' ', left));
    out.push_str(content);
    if !omit_right {
        out.extend(std::iter::repeat_n(' ', right));
    }
    out
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::from("|");
    for (&width, alignment) in widths.iter().zip(alignments) {
        out.push(' ');
        match alignment {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged { row, expected: n, actual: cells.len() });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_width = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut parsed = Vec::with_capacity(rows.len());
    let markdown = options.mode == Mode::Markdown;
    for (ri, row) in rows.iter().enumerate() {
        let mut parsed_row = Vec::with_capacity(n);
        for (ci, cell) in row.iter().enumerate() {
            parsed_row.push(parse_cell(cell, ri, ci, markdown)?);
        }
        parsed.push(parsed_row);
    }

    let mut widths = vec![minimum_width; n];
    for row in &parsed {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let candidate = (0..n)
                .rev()
                .find(|&i| widths[i] > minimum_width)
                .expect("minimum budget was validated");
            widths[candidate] -= 1;
            total -= 1;
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in parsed.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let mut column_fragments = Vec::new();
            for line in &row[column] {
                column_fragments.extend(fragments(line, widths[column], &overflows[column]));
            }
            height = height.max(column_fragments.len());
            cell_fragments.push(column_fragments);
        }

        for fragment_index in 0..height {
            match options.mode {
                Mode::Grid | Mode::Markdown => {
                    output.push('|');
                    for column in 0..n {
                        let content = cell_fragments[column]
                            .get(fragment_index)
                            .map(String::as_str)
                            .unwrap_or("");
                        output.push(' ');
                        output.push_str(&align_content(content, widths[column], &alignments[column], false));
                        output.push(' ');
                        output.push('|');
                    }
                    output.push('\n');
                }
                Mode::Compact => {
                    for column in 0..n {
                        if column > 0 {
                            output.push_str("  ");
                        }
                        let content = cell_fragments[column]
                            .get(fragment_index)
                            .map(String::as_str)
                            .unwrap_or("");
                        output.push_str(&align_content(
                            content,
                            widths[column],
                            &alignments[column],
                            column + 1 == n,
                        ));
                    }
                    output.push('\n');
                }
            }
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
        if options.mode == Mode::Markdown && row_index == 0 {
            output.push_str(&markdown_marker(&widths, &alignments));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    Ok(output)
}
