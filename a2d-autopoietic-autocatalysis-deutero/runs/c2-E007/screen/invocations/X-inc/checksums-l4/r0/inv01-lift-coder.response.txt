    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

pub enum SnapshotError {

pub struct Stream {

    adler_a: u32,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

fn crc32_step(mut crc: u32, byte: u8) -> u32 {
    crc ^= byte as u32;
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb88320;
        }
    }
    crc
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    for &byte in data {
        crc = crc32_step(crc, byte);
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i != 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let (mut crc, mut a, mut b) = (0xffff_ffffu32, 1u32, 0u32);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            a = (a + byte as u32) % 65521;
            b = (b + a) % 65521;
        }
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc_raw = crc32_step(self.crc_raw, byte);
            self.adler_a = (self.adler_a + byte as u32) % 65521;
            self.adler_b = (self.adler_b + self.adler_a) % 65521;
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let mut crc = 0u32;
        let mut adler = 0u32;
        for offset in (1..9).chain(11..19) {
            let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            if offset < 9 {
                crc = (crc << 4) | digit as u32;
            } else {
                adler = (adler << 4) | digit as u32;
            }
        }
        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (0u64, 0u64);
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = u16::from_be_bytes([pair[0], pair[1]]) as u64;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }
    if let Some(&last) = chunks.remainder().first() {
        a = (a + ((last as u64) << 8)) % 65535;
        b = (b + a) % 65535;
    }
    ((b as u32) << 16) | a as u32
}

fn crc16_step(mut reg: u16, byte: u8, variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            reg ^= (byte as u16) << 8;
            for _ in 0..8 {
                let top = reg & 0x8000;
                reg = reg.wrapping_shl(1);
                if top != 0 {
                    reg ^= 0x1021;
                }
            }
        }
        Crc16::X25 => {
            reg ^= byte as u16;
            for _ in 0..8 {
                let low = reg & 1;
                reg >>= 1;
                if low != 0 {
                    reg ^= 0x8408;
                }
            }
        }
    }
    reg
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut reg = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    for &byte in data {
        reg = crc16_step(reg, byte, variant);
    }
    if variant == Crc16::X25 {
        reg ^ 0xffff
    } else {
        reg
    }
}

#[derive(Clone, Copy)]
struct Accumulator {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    f16_a: u32,
    f16_b: u32,
    f32_a: u64,
    f32_b: u64,
    f32_pending: Option<u8>,
    crc16: u16,
}

impl Accumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            f16_a: 0,
            f16_b: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
        }
    }

    fn push(&mut self, byte: u8, variant: Crc16) {
        self.crc = crc32_step(self.crc, byte);
        self.adler_a = (self.adler_a + byte as u32) % 65521;
        self.adler_b = (self.adler_b + self.adler_a) % 65521;

        self.f16_a = (self.f16_a + byte as u32) % 255;
        self.f16_b = (self.f16_b + self.f16_a) % 255;

        if let Some(high) = self.f32_pending.take() {
            let word = ((high as u64) << 8) | byte as u64;
            self.f32_a = (self.f32_a + word) % 65535;
            self.f32_b = (self.f32_b + self.f32_a) % 65535;
        } else {
            self.f32_pending = Some(byte);
        }

        self.crc16 = crc16_step(self.crc16, byte, variant);
    }

    fn finish_fletcher32(&mut self) {
        if let Some(high) = self.f32_pending.take() {
            self.f32_a = (self.f32_a + ((high as u64) << 8)) % 65535;
            self.f32_b = (self.f32_b + self.f32_a) % 65535;
        }
    }

    fn multi(self, variant: Crc16) -> Multi {
        let crc16 = if variant == Crc16::X25 {
            self.crc16 ^ 0xffff
        } else {
            self.crc16
        };
        Multi {
            base: Hashes {
                crc32: self.crc ^ 0xffff_ffff,
                adler32: (self.adler_b << 16) | self.adler_a,
            },
            fletcher16: ((self.f16_b as u16) << 8) | self.f16_a as u16,
            fletcher32: ((self.f32_b as u32) << 16) | self.f32_a as u32,
            crc16,
        }
    }
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = acc.f32_pending.take() {
                let word = ((high as u64) << 8) | byte as u64;
                acc.f32_a = (acc.f32_a + word) % 65535;
                acc.f32_b = (acc.f32_b + acc.f32_a) % 65535;
            } else {
                acc.f32_pending = Some(byte);
            }
        }
    }
    acc.finish_fletcher32();
    Ok(((acc.f32_b as u32) << 16) | acc.f32_a as u32)
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(variant);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            acc.push(byte, variant);
        }
    }
    acc.finish_fletcher32();
    Ok(acc.multi(variant))
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for n in 0..32 {
        square[n] = gf2_matrix_times(matrix, matrix[n]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb88320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut len = 0u64;
    let mut crc = 0xffff_ffffu32;
    let (mut a, mut b) = (1u32, 0u32);
    for span in spans {
        let selected = span.end - span.start;
        len += selected as u64;
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            a = (a + byte as u32) % 65521;
            b = (b + a) % 65521;
        }
    }
    Ok(format!(
        "len={len:016x};crc={:08x};adler={:08x}",
        crc ^ 0xffff_ffff,
        (b << 16) | a
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut values = [0u64; 3];
    for offset in (4..20).chain(25..33).chain(40..48) {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        let field = if offset < 20 {
            0
        } else if offset < 33 {
            1
        } else {
            2
        };
        values[field] = (values[field] << 4) | digit as u64;
    }

    let mut selected_len = 0u64;
    let mut crc = 0xffff_ffffu32;
    let (mut a, mut b) = (1u32, 0u32);
    for span in spans {
        selected_len += (span.end - span.start) as u64;
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            a = (a + byte as u32) % 65521;
            b = (b + a) % 65521;
        }
    }

    if values[0] != selected_len {
        return Err(VerifyError::Length);
    }
    let actual_crc = crc ^ 0xffff_ffff;
    let actual_adler = (b << 16) | a;
    if values[1] as u32 != actual_crc {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: values[1] as u32,
            actual: actual_crc,
        });
    }
    if values[2] as u32 != actual_adler {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: values[2] as u32,
            actual: actual_adler,
        });
    }
    Ok(())
}
