#[derive(Debug, Clone)]
struct State {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
}

impl State {
    fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    fn from_checksums(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc ^= u32::from(byte);
            for _ in 0..8 {
                self.crc = (self.crc >> 1)
                    ^ if self.crc & 1 != 0 { 0xedb8_8320 } else { 0 };
            }

            self.adler_a = (self.adler_a + u32::from(byte)) % 65521;
            self.adler_b = (self.adler_b + self.adler_a) % 65521;
        }
    }

    fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Stream {
    state: State,
}

impl Stream {
    pub fn new() -> Self {
        Self { state: State::new() }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        Ok(Self {
            state: State::from_checksums(crc32, adler32)?,
        })
    }

    pub fn checksum(&self) -> Hashes {
        self.state.checksum()
    }

    pub fn update(&mut self, data: &[u8]) {
        self.state.update(data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data, spans)?;
        for span in spans {
            self.state.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        self.state = State::new();
    }

    pub fn snapshot(&self) -> String {
        let h = self.checksum();
        format!("c{:08x}:a{:08x}", h.crc32, h.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19
            || bytes[0] != b'c'
            || bytes[9] != b':'
            || bytes[10] != b'a'
        {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_hex_u32(bytes, 1, 9)?;
        let adler = parse_hex_u32(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn validate_spans(data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
    let mut preceding_end = 0usize;

    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i > 0 && span.start < preceding_end {
            return Err(SpanError::Overlap { span: i });
        }
        preceding_end = span.end;
    }
    Ok(())
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut state = State::new();
    state.update(data);
    state.checksum().crc32
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data, spans)?;
    let mut state = State::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

#[derive(Debug, Clone, Copy)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending_high: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending_high: None,
        }
    }

    fn word(&mut self, word: u16) {
        self.a = (self.a + u32::from(word)) % 65535;
        self.b = (self.b + self.a) % 65535;
    }

    fn update(&mut self, data: &[u8]) {
        let mut index = 0;
        if let Some(high) = self.pending_high.take() {
            if let Some(&low) = data.first() {
                self.word((u16::from(high) << 8) | u16::from(low));
                index = 1;
            } else {
                self.pending_high = Some(high);
                return;
            }
        }

        while index + 1 < data.len() {
            self.word((u16::from(data[index]) << 8) | u16::from(data[index + 1]));
            index += 2;
        }

        if index < data.len() {
            self.pending_high = Some(data[index]);
        }
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending_high.take() {
            self.word(u16::from(high) << 8);
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    state.update(data);
    state.finish()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data, spans)?;
    let mut state = Fletcher32State::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut register: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                register ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000 != 0;
                    register = register.wrapping_shl(1);
                    if top {
                        register ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                register ^= u16::from(byte);
                for _ in 0..8 {
                    let low = register & 1 != 0;
                    register >>= 1;
                    if low {
                        register ^= 0x8408;
                    }
                }
            }
        }
    }

    if matches!(variant, Crc16::X25) {
        register ^ 0xffff
    } else {
        register
    }
}

#[derive(Debug, Clone, Copy)]
struct Fletcher16State {
    a: u32,
    b: u32,
}

impl Fletcher16State {
    fn new() -> Self {
        Self { a: 0, b: 0 }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.a = (self.a + u32::from(byte)) % 255;
            self.b = (self.b + self.a) % 255;
        }
    }

    fn finish(self) -> u16 {
        ((self.b << 8) | self.a) as u16
    }
}

#[derive(Debug, Clone, Copy)]
struct Crc16State {
    variant: Crc16,
    register: u16,
}

impl Crc16State {
    fn new(variant: Crc16) -> Self {
        Self {
            variant,
            register: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
        }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            match self.variant {
                Crc16::Xmodem | Crc16::False => {
                    self.register ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        let top = self.register & 0x8000 != 0;
                        self.register = self.register.wrapping_shl(1);
                        if top {
                            self.register ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    self.register ^= u16::from(byte);
                    for _ in 0..8 {
                        let low = self.register & 1 != 0;
                        self.register >>= 1;
                        if low {
                            self.register ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    fn finish(self) -> u16 {
        if matches!(self.variant, Crc16::X25) {
            self.register ^ 0xffff
        } else {
            self.register
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data, spans)?;

    let mut base = State::new();
    let mut f16 = Fletcher16State::new();
    let mut f32 = Fletcher32State::new();
    let mut c16 = Crc16State::new(variant);

    for span in spans {
        let part = &data[span.start..span.end];
        base.update(part);
        f16.update(part);
        f32.update(part);
        c16.update(part);
    }

    Ok(Multi {
        base: base.checksum(),
        fletcher16: f16.finish(),
        fletcher32: f32.finish(),
        crc16: c16.finish(),
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = gf2_matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];

    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut length = right_len;
    let mut crc = left;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }
    }

    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data, spans)?;
    let mut state = State::new();
    let mut len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        len += selected.len() as u64;
        state.update(selected);
    }
    let h = state.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, h.crc32, h.adler32
    ))
}

fn hex_value(byte: u8, offset: usize) -> Result<u32, VerifyError> {
    match byte {
        b'0'..=b'9' => Ok(u32::from(byte - b'0')),
        b'a'..=b'f' => Ok(u32::from(byte - b'a' + 10)),
        _ => Err(VerifyError::Hex { offset }),
    }
}

fn parse_hex_u64_verify(bytes: &[u8], start: usize, end: usize) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in start..end {
        value = (value << 4) | u64::from(hex_value(bytes[offset], offset)?);
    }
    Ok(value)
}

fn parse_hex_u32_verify(bytes: &[u8], start: usize, end: usize) -> Result<u32, VerifyError> {
    let mut value = 0u32;
    for offset in start..end {
        value = (value << 4) | hex_value(bytes[offset], offset)?;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data, spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let expected_len = parse_hex_u64_verify(bytes, 4, 20)?;
    let expected_crc = parse_hex_u32_verify(bytes, 25, 33)?;
    let expected_adler = parse_hex_u32_verify(bytes, 40, 48)?;

    let mut state = State::new();
    let mut selected_len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        selected_len += selected.len() as u64;
        state.update(selected);
    }

    if expected_len != selected_len {
        return Err(VerifyError::Length);
    }

    let actual = state.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

fn parse_hex_u32(
    bytes: &[u8],
    start: usize,
    end: usize,
) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => u32::from(bytes[offset] - b'0'),
            b'a'..=b'f' => u32::from(bytes[offset] - b'a' + 10),
            _ => return Err(SnapshotError::Hex { offset }),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}
