#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Clone)]
struct Component {
    start: usize,
    end: usize,
    left: Vec<Change>,
    right: Vec<Change>,
}

#[derive(Clone, Copy)]
enum Op {
    Keep,
    Delete,
    Insert,
}

fn lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0, 0, 0);
    let mut changes = Vec::new();
    let mut active_start = None;
    let mut deletes = 0usize;
    let mut replacement = Vec::new();

    while i < n || j < m {
        let op = if i == n {
            Op::Insert
        } else if j == m {
            Op::Delete
        } else if old[i] == new[j] {
            Op::Keep
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins || (del == ins && (i + j) % 2 == 0) {
                Op::Delete
            } else {
                Op::Insert
            }
        };

        match op {
            Op::Keep => {
                if let Some(start) = active_start.take() {
                    changes.push(Change {
                        start,
                        end: start + deletes,
                        replacement: std::mem::take(&mut replacement),
                    });
                    deletes = 0;
                }
                i += 1;
                j += 1;
                base_pos += 1;
            }
            Op::Delete => {
                active_start.get_or_insert(base_pos);
                i += 1;
                base_pos += 1;
                deletes += 1;
            }
            Op::Insert => {
                active_start.get_or_insert(base_pos);
                replacement.push(new[j].clone());
                j += 1;
            }
        }
    }

    if let Some(start) = active_start {
        changes.push(Change {
            start,
            end: start + deletes,
            replacement,
        });
    }
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn apply_changes(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut output = Vec::new();
    let mut cursor = start;
    for change in changes {
        if change.start < cursor || change.end > end {
            continue;
        }
        output.extend(base[cursor..change.start].iter().cloned());
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    output.extend(base[cursor..end].iter().cloned());
    output
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);
    let (base_len, left_len, right_len) =
        (base_lines.len(), left_lines.len(), right_lines.len());
    if base_len > 128 || left_len > 128 || right_len > 128 {
        return Err(MergeError::TooLarge {
            base_len,
            left_len,
            right_len,
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let total = left_changes.len() + right_changes.len();
    let mut parent: Vec<usize> = (0..total).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        x
    }

    for i in 0..total {
        for j in i + 1..total {
            let a = if i < left_changes.len() {
                &left_changes[i]
            } else {
                &right_changes[i - left_changes.len()]
            };
            let b = if j < left_changes.len() {
                &left_changes[j]
            } else {
                &right_changes[j - left_changes.len()]
            };
            if interacts(a, b) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut components: Vec<Component> = Vec::new();
    for (side_index, changes) in [&left_changes, &right_changes].iter().enumerate() {
        for (change_index, change) in changes.iter().enumerate() {
            let global = if side_index == 0 {
                change_index
            } else {
                left_changes.len() + change_index
            };
            let representative = root(&mut parent, global);
            let existing = components.iter().position(|component| {
                let any = component
                    .left
                    .iter()
                    .chain(component.right.iter())
                    .any(|c| {
                        let idx = if side_index == 0 {
                            left_changes.iter().position(|x| std::ptr::eq(x, c))
                        } else {
                            right_changes.iter().position(|x| std::ptr::eq(x, c))
                        };
                        idx.is_some()
                    });
                any && representative == root(&mut parent, global)
            });

            let _ = existing;
        }
    }

    // Build connected components from their union-find representatives.
    let mut groups: Vec<(usize, Vec<Change>, Vec<Change>)> = Vec::new();
    for (idx, change) in left_changes.iter().enumerate() {
        let rep = root(&mut parent, idx);
        let pos = groups.iter().position(|g| g.0 == rep).unwrap_or_else(|| {
            groups.push((rep, Vec::new(), Vec::new()));
            groups.len() - 1
        });
        groups[pos].1.push(change.clone());
    }
    for (idx, change) in right_changes.iter().enumerate() {
        let rep = root(&mut parent, left_changes.len() + idx);
        let pos = groups.iter().position(|g| g.0 == rep).unwrap_or_else(|| {
            groups.push((rep, Vec::new(), Vec::new()));
            groups.len() - 1
        });
        groups[pos].2.push(change.clone());
    }

    for (_, left_group, right_group) in groups {
        let all = left_group.iter().chain(right_group.iter());
        let start = all.clone().map(|c| c.start).min().unwrap_or(0);
        let end = all.map(|c| c.end).max().unwrap_or(start);
        components.push(Component {
            start,
            end,
            left: left_group,
            right: right_group,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        output.extend(base_lines[base_cursor..component.start].iter().cloned());
        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = apply_changes(
            &base_lines,
            component.start,
            component.end,
            &component.left,
        );
        let right_projection = apply_changes(
            &base_lines,
            component.start,
            component.end,
            &component.right,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {}", ordinal));
                    output.extend(left_projection.iter().cloned());
                    let base_label = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {},{}",
                        base_label,
                        component.end - component.start
                    ));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {}", ordinal));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    output.extend(base_lines[base_cursor..].iter().cloned());
    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        conflicts: conflicts.len(),
        text,
        regions: conflicts,
    })
}
