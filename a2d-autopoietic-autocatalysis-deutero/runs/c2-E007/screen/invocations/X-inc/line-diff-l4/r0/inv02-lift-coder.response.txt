#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Debug, Clone)]
struct Component {
    start: usize,
    end: usize,
    left: Vec<usize>,
    right: Vec<usize>,
}

#[derive(Debug, Clone, Copy)]
enum Edit {
    Keep,
    Delete,
    Insert,
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn validate(side: MergeSide, text: &str) -> Result<(), MergeError> {
    if let Some(byte) = text.bytes().position(|b| b == b'\r') {
        return Err(MergeError::CarriageReturn { side, byte });
    }
    if !text.is_empty() && !text.ends_with('\n') {
        return Err(MergeError::MissingFinalLf { side });
    }
    Ok(())
}

fn canonical_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut edits = Vec::new();
    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            edits.push(Edit::Keep);
            i += 1;
            j += 1;
        } else if i == n {
            edits.push(Edit::Insert);
            j += 1;
        } else if j == m {
            edits.push(Edit::Delete);
            i += 1;
        } else {
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            if delete_len > insert_len || (delete_len == insert_len && (i + j) % 2 == 0) {
                edits.push(Edit::Delete);
                i += 1;
            } else {
                edits.push(Edit::Insert);
                j += 1;
            }
        }
    }

    let mut changes = Vec::new();
    let (mut old_pos, mut new_pos) = (0, 0);
    let mut run_start = None;
    let mut run_deletes = 0usize;
    let mut replacement = Vec::new();

    for edit in edits {
        match edit {
            Edit::Keep => {
                if let Some(start) = run_start.take() {
                    changes.push(Change {
                        start,
                        end: start + run_deletes,
                        replacement: std::mem::take(&mut replacement),
                    });
                    run_deletes = 0;
                }
                old_pos += 1;
                new_pos += 1;
            }
            Edit::Delete => {
                run_start.get_or_insert(old_pos);
                run_deletes += 1;
                old_pos += 1;
            }
            Edit::Insert => {
                run_start.get_or_insert(old_pos);
                replacement.push(new[new_pos].clone());
                new_pos += 1;
            }
        }
    }
    if let Some(start) = run_start {
        changes.push(Change {
            start,
            end: start + run_deletes,
            replacement,
        });
    }

    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_len = a.end - a.start;
    let b_len = b.end - b.start;

    match (a_len > 0, b_len > 0) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn build_components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let mut nodes: Vec<(bool, usize)> = (0..left.len())
        .map(|i| (true, i))
        .chain((0..right.len()).map(|i| (false, i)))
        .collect();
    let count = nodes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        x
    }

    for i in 0..count {
        for j in i + 1..count {
            let (li, xi) = nodes[i];
            let (lj, xj) = nodes[j];
            if li == lj {
                continue;
            }
            let a = if li { &left[xi] } else { &right[xi] };
            let b = if lj { &left[xj] } else { &right[xj] };
            if interacts(a, b) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                parent[ri] = rj;
            }
        }
    }

    let mut groups: Vec<(usize, usize, Vec<usize>, Vec<usize>)> = Vec::new();
    for i in 0..count {
        let r = root(&mut parent, i);
        let (is_left, index) = nodes[i];
        let change = if is_left { &left[index] } else { &right[index] };

        if let Some(group) = groups.iter_mut().find(|g| g.0 == r) {
            group.1 = group.1.min(change.start);
            group.2 = group.2.max(change.end);
            if is_left {
                group.3.push(index);
            } else {
                group.4.push(index);
            }
        } else {
            let mut ls = Vec::new();
            let mut rs = Vec::new();
            if is_left {
                ls.push(index);
            } else {
                rs.push(index);
            }
            groups.push((r, change.start, change.end, ls));
            groups.last_mut().unwrap().4 = rs;
        }
    }

    nodes.clear();
    let mut components: Vec<Component> = groups
        .into_iter()
        .map(|(_, start, end, left_indices, right_indices)| Component {
            start,
            end,
            left: left_indices,
            right: right_indices,
        })
        .collect();

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });
    components
}

fn project(base: &[String], start: usize, end: usize, changes: &[Change], indices: &[usize]) -> Vec<String> {
    if indices.is_empty() {
        return base[start..end].to_vec();
    }
    let mut selected: Vec<&Change> = indices.iter().map(|&i| &changes[i]).collect();
    selected.sort_by_key(|change| (change.start, change.end));

    let mut output = Vec::new();
    let mut cursor = start;
    for change in selected {
        if change.start > cursor {
            output.extend_from_slice(&base[cursor..change.start]);
        }
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        output.extend_from_slice(&base[cursor..end]);
    }
    output
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate(MergeSide::Base, base)?;
    validate(MergeSide::Left, left)?;
    validate(MergeSide::Right, right)?;

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = canonical_changes(&base_lines, &left_lines);
    let right_changes = canonical_changes(&base_lines, &right_lines);
    let components = build_components(&left_changes, &right_changes);

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        if component.start > base_cursor {
            append_lines(&mut output, &base_lines[base_cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
            &component.left,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
            &component.right,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let base_label = if component.end > component.start {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_label},{}",
                        component.end - component.start
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[base_cursor..]);
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
