fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)

fn constraint_rank(c: &Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn canonicalize(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by(|a, b| {
        constraint_rank(a)
            .cmp(&constraint_rank(b))
            .then_with(|| match (a, b) {
                (Constraint::Exact(x), Constraint::Exact(y))
                | (Constraint::AtLeast(x), Constraint::AtLeast(y))
                | (Constraint::Before(x), Constraint::Before(y))
                | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(y),
                _ => std::cmp::Ordering::Equal,
            })
    });
    constraints.dedup();
    constraints
}

fn constraint_text(c: &Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(*v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(*v)),
        Constraint::Before(v) => format!("<{}", version_text(*v)),
        Constraint::Caret(v) => format!("^{}", version_text(*v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonicalize(constraints.to_vec())
        .iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

        }
    }
    out
}

type Active = std::collections::BTreeMap<String, Vec<Constraint>>;
type Chosen = std::collections::BTreeMap<String, Version>;

struct Search<'a> {
    catalog: &'a [Package],
    by_name: std::collections::BTreeMap<String, &'a Package>,
    locks: std::collections::BTreeMap<String, Version>,
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, Chosen, Active)>,
    first_error: Option<ResolveError>,
    limited: bool,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn active_for(&self, chosen: &Chosen, roots: &[Requirement]) -> Active {
        let mut active = Active::new();
        for root in roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in chosen {
            if let Some(package) = self.by_name.get(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonicalize(std::mem::take(constraints));
        }
        active
    }

    fn inspect_error(&self, active: &Active, chosen: &Chosen) -> Option<ResolveError> {
        for name in active.keys() {
            if !self.by_name.contains_key(name) {
                return Some(ResolveError::Missing {
                    package: name.clone(),
                });
            }
        }

        for (name, version) in chosen {
            if let Some(constraints) = active.get(name) {
                if !constraints.iter().all(|c| matches(*version, *c)) {
                    return Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in active {
            if chosen.contains_key(name) {
                continue;
            }
            let package = self.by_name[name];
            let found = package
                .releases
                .iter()
                .any(|r| constraints.iter().all(|c| matches(r.version, *c)));
            if !found {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }
        None
    }

    fn consider_solution(&mut self, chosen: &Chosen, active: &Active) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| chosen.get(*name) != Some(*version))
            .count();
        let vector: Vec<Option<Version>> = self
            .catalog_names
            .iter()
            .map(|name| chosen.get(name).copied())
            .collect();

        let better = match &self.best {
            None => true,
            Some((best_cost, best_vector, _, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if better {
            self.best = Some((cost, vector, chosen.clone(), active.clone()));
        }
    }

    fn visit(&mut self, roots: &[Requirement], chosen: &mut Chosen) {
        if self.limited {
            return;
        }
        let active = self.active_for(chosen, roots);
        if let Some(error) = self.inspect_error(&active, chosen) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return;
        }

        let next = active.keys().find(|name| !chosen.contains_key(*name)).cloned();
        let Some(name) = next else {
            self.consider_solution(chosen, &active);
            return;
        };

        let constraints = &active[&name];
        let package = self.by_name[&name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(locked) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == locked) {
                let version = candidates.remove(pos);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limited = true;
                return;
            }
            self.attempts += 1;
            chosen.insert(name.clone(), version);
            self.visit(roots, chosen);
            chosen.remove(&name);
            if self.limited {
                return;
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let by_name: std::collections::BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        by_name,
        locks: lock_map,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
        limited: false,
        catalog_names,
    };
    let mut chosen = Chosen::new();
    search.visit(roots, &mut chosen);

    if search.limited {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, _, solution, final_active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut package_names: std::collections::BTreeSet<String> =
        search.locks.keys().cloned().collect();
    package_names.extend(solution.keys().cloned());

    let mut changes = Vec::new();
    for package in package_names {
        let before = search.locks.get(&package).copied();
        let after = solution.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = search
                    .by_name
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == locked));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&package)
                    .is_some_and(|cs| !cs.iter().all(|c| matches(locked, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
        && bytes[bytes.len() - 1] != b'-'
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    constraint_texts(constraints.to_vec())
}

fn constraint_texts(constraints: Vec<Constraint>) -> String {
    canonical(constraints)
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn matches_all(v: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|&c| matches(v, c))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for byte in s.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

type Active = std::collections::BTreeMap<String, Vec<Constraint>>;
type Assignment = std::collections::BTreeMap<String, Version>;

struct Search<'a> {
    catalog: &'a [Package],
    package_indices: std::collections::BTreeMap<String, usize>,
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, Assignment, Active)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn active(&self, assignment: &Assignment, roots: &[Requirement]) -> Active {
        let mut active = Active::new();
        for root in roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in assignment {
            if let Some(&index) = self.package_indices.get(name) {
                if let Some(release) = self.catalog[index]
                    .releases
                    .iter()
                    .find(|r| r.version == *version)
                {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        active
    }

    fn node_error(&self, assignment: &Assignment, active: &Active) -> Option<ResolveError> {
        for name in active.keys() {
            if !self.package_indices.contains_key(name) {
                return Some(ResolveError::Missing {
                    package: name.clone(),
                });
            }
        }

        for (name, version) in assignment {
            if let Some(constraints) = active.get(name) {
                if !matches_all(*version, constraints) {
                    return Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical(constraints.clone()),
                    });
                }
            }
        }

        for (name, constraints) in active {
            if assignment.contains_key(name) {
                continue;
            }
            let index = self.package_indices[name];
            if !self.catalog[index]
                .releases
                .iter()
                .any(|r| matches_all(r.version, constraints))
            {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical(constraints.clone()),
                });
            }
        }
        None
    }

    fn lock_cost(&self, assignment: &Assignment) -> usize {
        self.locks
            .iter()
            .filter(|lock| assignment.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn version_vector(&self, assignment: &Assignment) -> Vec<Option<Version>> {
        self.package_indices
            .keys()
            .map(|name| assignment.get(name).copied())
            .collect()
    }

    fn consider(&mut self, assignment: Assignment, active: Active) {
        let cost = self.lock_cost(&assignment);
        let vector = self.version_vector(&assignment);
        let replace = match &self.best {
            None => true,
            Some((best_cost, best_vector, _, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            self.best = Some((cost, vector, assignment, active));
        }
    }

    fn explore(&mut self, roots: &[Requirement], assignment: Assignment) -> Result<(), ResolveError> {
        let active = self.active(&assignment, roots);
        if let Some(error) = self.node_error(&assignment, &active) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active.keys().find(|name| !assignment.contains_key(*name)).cloned();
        let Some(name) = next else {
            self.consider(assignment, active);
            return Ok(());
        };

        let constraints = &active[&name];
        let index = self.package_indices[&name];
        let mut candidates: Vec<Version> = self.catalog[index]
            .releases
            .iter()
            .filter(|r| matches_all(r.version, constraints))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(position) = candidates.iter().position(|v| *v == lock.version) {
                let version = candidates.remove(position);
                candidates.insert(0, version);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let mut child = assignment.clone();
            child.insert(name.clone(), candidate);
            self.explore(roots, child)?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_indices = catalog
        .iter()
        .enumerate()
        .map(|(i, package)| (package.name.clone(), i))
        .collect();

    let mut search = Search {
        catalog,
        package_indices,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.explore(roots, Assignment::new())?;

    let Some((_, _, assignment, active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = assignment
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut before = std::collections::BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut after = std::collections::BTreeMap::new();
    for (package, version) in &assignment {
        after.insert(package.clone(), *version);
    }

    let package_names: std::collections::BTreeSet<String> =
        before.keys().chain(after.keys()).cloned().collect();
    let mut changes = Vec::new();
    for package in package_names {
        let old = before.get(&package).copied();
        let new = after.get(&package).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = search
                    .package_indices
                    .get(&package)
                    .map(|&i| {
                        catalog[i]
                            .releases
                            .iter()
                            .any(|r| r.version == old_version)
                    })
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if active
                    .get(&package)
                    .is_some_and(|constraints| !matches_all(old_version, constraints))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let lock_map: BTreeMap<&str, Version> = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect();

    let mut root_constraints: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    for root in roots {
        root_constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut state = SearchState {
        catalog: &catalog_map,
        roots: &root_constraints,
        locks: &lock_map,
        all_names: catalog.iter().map(|p| p.name.clone()).collect(),
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };
    state.all_names.sort();

    let mut assigned = BTreeMap::new();
    state.search(&mut assigned)?;

    if let Some(best) = state.best {
        let selected: Vec<Selection> = best
            .iter()
            .map(|(name, version)| Selection {
                package: name.clone(),
                version: *version,
            })
            .collect();

        let mut changes = Vec::new();

        let names: BTreeSet<String> = lock_map
            .keys()
            .map(|name| (*name).to_string())
            .chain(best.keys().cloned())
            .collect();

        for name in names {
            let before = lock_map.get(name.as_str()).copied();
            let after = best.get(&name).copied();
            if before == after {
                continue;
            }
            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(locked), Some(_)) => {
                    let exists = catalog_map
                        .get(name.as_str())
                        .map(|package| package.releases.iter().any(|r| r.version == locked))
                        .unwrap_or(false);
                    if !exists {
                        ChangeReason::Unavailable
                    } else if !final_constraints
                        .get(&name)
                        .map(|cs| cs.iter().all(|c| permits(*c, locked)))
                        .unwrap_or(true)
                    {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
                (None, None) => continue,
            };
            changes.push(Change {
                package: name,
                before,
                after,
                reason,
            });
        }

        Ok(Resolution { selected, changes })
    } else {
        Err(state.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }))
    }
}

struct SearchState<'a> {
    catalog: &'a BTreeMap<&'a str, &'a Package>,
    roots: &'a BTreeMap<String, Vec<Constraint>>,
    locks: &'a BTreeMap<&'a str, Version>,
    all_names: Vec<String>,
    attempts: usize,
    max_attempts: usize,
    best: Option<BTreeMap<String, Version>>,
    first_error: Option<ResolveError>,
}

impl SearchState<'_> {
    fn search(
        &mut self,
        assigned: &mut BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, assigned, self.catalog);

        let missing: Vec<String> = active
            .keys()
            .filter(|name| !self.catalog.contains_key(name.as_str()))
            .cloned()
            .collect();
        if let Some(package) = missing.into_iter().next() {
            self.record_error(ResolveError::Missing { package });
            return Ok(());
        }

        for (package, version) in assigned.iter() {
            if let Some(requirements) = active.get(package) {
                if !requirements.iter().all(|c| permits(*c, *version)) {
                    self.record_error(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: requirements.clone(),
                    });
                    return Ok(());
                }
            }
        }

        for (package, requirements) in &active {
            if assigned.contains_key(package) {
                continue;
            }
            let releases = &self.catalog[package.as_str()].releases;
            if !releases
                .iter()
                .any(|r| requirements.iter().all(|c| permits(*c, r.version)))
            {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: requirements.clone(),
                });
                return Ok(());
            }
        }

        let next = active
            .keys()
            .find(|package| !assigned.contains_key(*package))
            .cloned();

        let Some(package) = next else {
            self.consider_solution(assigned);
            return Ok(());
        };

        let requirements = &active[&package];
        let mut candidates: Vec<Version> = self.catalog[package.as_str()]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|version| requirements.iter().all(|c| permits(*c, *version)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(package.as_str()).copied() {
            if let Some(index) = candidates.iter().position(|v| *v == locked) {
                candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(package.clone(), version);
            self.search(assigned)?;
            assigned.remove(&package);
        }
        Ok(())
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn consider_solution(&mut self, assigned: &BTreeMap<String, Version>) {
        let cost = self
            .locks
            .iter()
            .filter(|(package, version)| assigned.get(**package) != Some(*version))
            .count();

        let replace = match &self.best {
            None => true,
            Some(best) => {
                let best_cost = self
                    .locks
                    .iter()
                    .filter(|(package, version)| best.get(**package) != Some(*version))
                    .count();
                if cost != best_cost {
                    cost < best_cost
                } else {
                    self.lex_vector_greater(assigned, best)
                }
            }
        };

        if replace {
            self.best = Some(assigned.clone());
        }
    }

    fn lex_vector_greater(
        &self,
        left: &BTreeMap<String, Version>,
        right: &BTreeMap<String, Version>,
    ) -> bool {
        for name in &self.all_names {
            match (left.get(name), right.get(name)) {
                (None, None) => continue,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => continue,
            }
        }
        false
    }
}

fn active_constraints(
    roots: &BTreeMap<String, Vec<Constraint>>,
    assigned: &BTreeMap<String, Version>,
    catalog: &BTreeMap<&str, &Package>,
) -> BTreeMap<String, Vec<Constraint>> {
    let mut active = roots.clone();
    for (package, version) in assigned {
        if let Some(pkg) = catalog.get(package.as_str()) {
            if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        canonicalize(constraints);
    }
    active
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn permits(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(expected) => version == expected,
        Constraint::AtLeast(minimum) => version >= minimum,
        Constraint::Before(bound) => version < bound,
        Constraint::Caret(base) => {
            if version < base {
                false
            } else if base.major > 0 {
                version.major == base.major
            } else if base.minor > 0 {
                version.major == 0 && version.minor == base.minor
            } else {
                version == base
            }
        }
    }
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut result = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphabetic()
            || byte.is_ascii_digit()
            || matches!(*byte, b'.' | b'_' | b'-')
        {
            result.push(*byte as char);
        } else {
            result.push_str(&format!("%{:02X}", byte));
        }
    }
    result
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

type Active = BTreeMap<String, Vec<Constraint>>;
type Selected = BTreeMap<String, Version>;

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
    limit_hit: bool,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn active_for(&self, selected: &Selected, roots: &[Requirement]) -> Active {
        let mut active = Active::new();
        for root in roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in selected {
            if let Some(package) = self.catalog.get(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical_constraints(constraints);
        }
        active
    }

    fn candidates(&self, name: &str, constraints: &[Constraint]) -> Vec<Version> {
        let mut versions: Vec<Version> = self.catalog[name]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
            .collect();
        versions.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = self.locks.get(name) {
            if let Some(pos) = versions.iter().position(|v| v == locked) {
                let version = versions.remove(pos);
                versions.insert(0, version);
            }
        }
        versions
    }

    fn solution_cost(&self, selected: &Selected) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count()
    }

    fn better_solution(&self, cost: usize, selected: &[Selection]) -> bool {
        let Some((best_cost, best_selected)) = &self.best else {
            return true;
        };
        if cost != *best_cost {
            return cost < *best_cost;
        }
        let current: BTreeMap<&str, Version> = selected
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect();
        let previous: BTreeMap<&str, Version> = best_selected
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect();
        for name in &self.catalog_names {
            let a = current.get(name.as_str());
            let b = previous.get(name.as_str());
            match (a, b) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn visit(&mut self, roots: &[Requirement], selected: &mut Selected) {
        if self.limit_hit {
            return;
        }

        let active = self.active_for(selected, roots);

        if let Some(name) = active.keys().find(|name| !self.catalog.contains_key(*name)) {
            self.record_error(ResolveError::Missing {
                package: name.clone(),
            });
            return;
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints.iter().all(|c| matches(*version, *c)) {
                    self.record_error(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                    return;
                }
            }
        }

        let mut candidate_map = BTreeMap::new();
        for (name, constraints) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let candidates = self.candidates(name, constraints);
            if candidates.is_empty() {
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
                return;
            }
            candidate_map.insert(name.clone(), candidates);
        }

        let Some((name, candidates)) = candidate_map.into_iter().next() else {
            let solution: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self.solution_cost(selected);
            if self.better_solution(cost, &solution) {
                self.best = Some((cost, solution));
            }
            return;
        };

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit_hit = true;
                return;
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            self.visit(roots, selected);
            selected.remove(&name);
            if self.limit_hit {
                return;
            }
        }
    }
}

fn collect_catalog_names(catalog: &[Package]) -> Vec<String> {
    let mut names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    names.sort();
    names
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();

    let mut search = Search {
        catalog: &catalog_map,
        locks: &lock_map,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit_hit: false,
        catalog_names: collect_catalog_names(catalog),
    };

    let mut selected = Selected::new();
    search.visit(roots, &mut selected);

    if search.limit_hit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, selected_vec)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected_map: BTreeMap<String, Version> = selected_vec
        .iter()
        .map(|s| (s.package.clone(), s.version))
        .collect();
    let mut active = Active::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for selection in &selected_vec {
        if let Some(package) = catalog_map.get(&selection.package) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|r| r.version == selection.version)
            {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for cs in active.values_mut() {
        *cs = canonical_constraints(cs);
    }

    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(selected_map.keys().cloned());
    let mut changes = Vec::new();
    for name in all_names {
        let before = lock_map.get(&name).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&name)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if active
                    .get(&name)
                    .is_some_and(|cs| !cs.iter().all(|c| matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
