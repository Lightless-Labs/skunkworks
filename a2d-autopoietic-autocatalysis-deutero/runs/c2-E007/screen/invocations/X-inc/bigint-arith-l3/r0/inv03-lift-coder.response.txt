use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits; zero is represented by [0].
    digits: Vec<u8>,
}

impl Big {
    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !zero,
            digits,
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u8;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0)
                + b.get(i).copied().unwrap_or(0)
                + carry;
            out.push(sum % 10);
            carry = sum / 10;
        }
        if carry != 0 {
            out.push(carry);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i8;
        for (i, &digit) in a.iter().enumerate() {
            let mut value = digit as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
            if value < 0 {
                value += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = vec![0u8; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            let mut carry = 0u32;
            for (j, &y) in b.iter().enumerate() {
                let k = i + j;
                let value = out[k] as u32 + x as u32 * y as u32 + carry;
                out[k] = (value % 10) as u8;
                carry = value / 10;
            }
            let mut k = i + b.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let value = out[k] as u32 + carry;
                out[k] = (value % 10) as u8;
                carry = value / 10;
                k += 1;
            }
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs_digit(a: &[u8], digit: u8) -> Vec<u8> {
        if digit == 0 {
            return vec![0];
        }
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &x in a {
            let value = x as u16 * digit as u16 + carry;
            out.push((value % 10) as u8);
            carry = value / 10;
        }
        while carry != 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out
    }

    fn push_abs_digit(a: &mut Vec<u8>, digit: u8) {
        if a.len() == 1 && a[0] == 0 {
            a[0] = digit;
        } else {
            a.insert(0, digit);
        }
        while a.len() > 1 && a.last() == Some(&0) {
            a.pop();
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_count = 0usize;
        let mut first_digit_index = None;
        let mut first_invalid = None;
        for i in start..bytes.len() {
            if bytes[i].is_ascii_digit() {
                digit_count += 1;
                if first_digit_index.is_none() {
                    first_digit_index = Some(i);
                }
            } else if bytes[i] != b'_' && first_invalid.is_none() {
                first_invalid = Some(i);
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_bad_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                first_bad_separator = Some(i);
                break;
            }
        }
        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        if digit_count > 1 {
            let first = first_digit_index.expect("nonempty magnitude has digits");
            if bytes[first] == b'0' {
                return Err(Error::LeadingZero { index: first });
            }
        }

        if digit_count > max_digits {
            let excess = max_digits;
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i].is_ascii_digit() {
                    seen += 1;
                    if seen == excess + 1 {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }
        Ok(Self::from_parts(negative, digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let raw_len = self.digits.len().max(1);
        let width = raw_len.max(min_digits);
        let mut magnitude = String::with_capacity(width + width / group.max(1));
        for _ in raw_len..width {
            magnitude.push('0');
        }
        for &digit in self.digits.iter().rev() {
            magnitude.push((b'0' + digit) as char);
        }

        if group > 0 && magnitude.len() > group {
            let mut grouped = String::with_capacity(magnitude.len() + magnitude.len() / group);
            let first_group = {
                let rem = magnitude.len() % group;
                if rem == 0 { group } else { rem }
            };
            for (i, ch) in magnitude.chars().enumerate() {
                if i != 0 && (i - first_group) % (group + 1) == 0 {
                    grouped.push('_');
                }
                grouped.push(ch);
            }
            magnitude = grouped;
        }

        if self.negative {
            let mut out = String::with_capacity(magnitude.len() + 1);
            out.push('-');
            out.push_str(&magnitude);
            out
        } else {
            magnitude
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(self.negative, Self::add_abs(&self.digits, &rhs.digits))
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::from_parts(false, vec![0]),
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    Self::sub_abs(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    Self::sub_abs(&rhs.digits, &self.digits),
                ),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        Self::from_parts(!self.negative, self.digits.clone())
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::from_parts(false, vec![0]);
        }
        Self::from_parts(
            self.negative != rhs.negative,
            Self::mul_abs(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let mut quotient_be = Vec::with_capacity(self.digits.len());
        let mut remainder = vec![0];
        for &digit in self.digits.iter().rev() {
            Self::push_abs_digit(&mut remainder, digit);
            let mut qdigit = 0u8;
            for candidate in (1..=9).rev() {
                let product = Self::mul_abs_digit(&rhs.digits, candidate);
                if Self::cmp_digits(&remainder, &product) != Ordering::Less {
                    qdigit = candidate;
                    remainder = Self::sub_abs(&remainder, &product);
                    break;
                }
            }
            quotient_be.push(qdigit);
        }

        while quotient_be.len() > 1 && quotient_be[0] == 0 {
            quotient_be.remove(0);
        }
        let quotient = Self::from_parts(
            self.negative != rhs.negative,
            quotient_be.into_iter().rev().collect(),
        );
        let remainder = Self::from_parts(self.negative, remainder);
        Ok((quotient, remainder))
    }

    fn cmp_digits(a: &[u8], b: &[u8]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => {
                for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                    match x.cmp(y) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }
}
