```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrivals: Vec<usize> = (0..jobs.len()).collect();
    arrivals.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let mut now = 0u64;
    let mut next_arrival = 0usize;
    let mut completed_count = 0usize;
    let mut last_dispatched_id: Option<u32> = None;

    let admit_through = |time: u64,
                         next_arrival: &mut usize,
                         queues: &mut Vec<std::collections::VecDeque<usize>>,
                         completion: &mut Vec<Option<u64>>,
                         completed_count: &mut usize| {
        while *next_arrival < arrivals.len()
            && u64::from(jobs[arrivals[*next_arrival]].arrival) <= time
        {
            let index = arrivals[*next_arrival];
            *next_arrival += 1;
            if jobs[index].ticks == 0 {
                completion[index] = Some(u64::from(jobs[index].arrival));
                *completed_count += 1;
            } else {
                queues[jobs[index].level].push_back(index);
            }
        }
    };

    while completed_count < jobs.len() {
        admit_through(
            now,
            &mut next_arrival,
            &mut queues,
            &mut completion,
            &mut completed_count,
        );

        let selected = queues.iter().position(|queue| !queue.is_empty());
        let Some(level_with_job) = selected else {
            if next_arrival >= arrivals.len() {
                break;
            }

            let next_time = u64::from(jobs[arrivals[next_arrival]].arrival);
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let job_index = queues[level_with_job]
            .pop_front()
            .expect("selected queue is nonempty");

        if last_dispatched_id.is_some_and(|id| id != jobs[job_index].id) {
            now += u64::from(switch_cost);
        }

        admit_through(
            now,
            &mut next_arrival,
            &mut queues,
            &mut completion,
            &mut completed_count,
        );

        let selected_level = level_with_job;
        let consumed = remaining[job_index].min(quanta[selected_level]);
        let start = now;
        now += u64::from(consumed);
        remaining[job_index] -= consumed;
        let finished = remaining[job_index] == 0;

        runs.push(Run {
            id: jobs[job_index].id,
            start,
            end: now,
            finished,
            level: selected_level,
        });

        admit_through(
            now,
            &mut next_arrival,
            &mut queues,
            &mut completion,
            &mut completed_count,
        );

        if finished {
            completion[job_index] = Some(now);
            completed_count += 1;
        } else {
            let demoted_level = (selected_level + 1).min(quanta.len() - 1);
            queues[demoted_level].push_back(job_index);
        }

        last_dispatched_id = Some(jobs[job_index].id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap_or(now)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}
```
