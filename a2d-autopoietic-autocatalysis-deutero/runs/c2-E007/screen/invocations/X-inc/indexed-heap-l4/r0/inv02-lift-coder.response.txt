```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u128,
}

#[derive(Debug, Clone)]
pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    capacity: Option<usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::with_capacity_internal(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_capacity_internal(Some(capacity))
    }

    fn with_capacity_internal(capacity: Option<usize>) -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            capacity,
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn canonicalize(key: &str) -> Result<String, Error> {
        let bytes = key.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }

        if start == end {
            return Err(Error::EmptyKey);
        }

        for (i, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if i == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphanumeric() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + i });
            }
        }

        Ok(key[start..end].to_ascii_lowercase())
    }

    fn cmp_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (i, entry) in self.entries.iter().enumerate() {
            self.indices.insert(entry.item.key.clone(), i);
        }
    }

    fn normalize_ages(&mut self) {
        let mut order: Vec<usize> = (0..self.entries.len()).collect();
        order.sort_by_key(|&i| self.entries[i].age);
        for (age, index) in order.into_iter().enumerate() {
            self.entries[index].age = age as u128;
        }
        self.next_age = self.entries.len() as u128;
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            self.normalize_ages();
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    fn worst_index(&self) -> Option<usize> {
        (0..self.entries.len()).max_by(|&a, &b| {
            Self::cmp_entries(&self.entries[a], &self.entries[b])
        })
    }

    fn remove_index(&mut self, index: usize) -> Item {
        let removed = self.entries.swap_remove(index).item;
        self.rebuild_indices();
        removed
    }

    fn ordered_indices(&self) -> Vec<usize> {
        let mut indices: Vec<usize> = (0..self.entries.len()).collect();
        indices.sort_by(|&a, &b| Self::cmp_entries(&self.entries[a], &self.entries[b]));
        indices
    }

    fn trim_to_capacity(&mut self) -> Vec<Item> {
        let Some(capacity) = self.capacity else {
            return Vec::new();
        };
        let mut evicted = Vec::new();
        while self.entries.len() > capacity {
            if let Some(index) = self.worst_index() {
                evicted.push(self.remove_index(index));
            }
        }
        evicted
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let canonical = Self::canonicalize(key)?;
        if self.indices.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        let proposed = Item {
            key: canonical,
            priority,
            value: value.to_owned(),
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed));
        }

        if let Some(capacity) = self.capacity {
            if self.entries.len() >= capacity {
                let worst_index = self.worst_index().expect("full nonzero queue has an entry");
                let proposal = Entry {
                    item: proposed.clone(),
                    age: self.next_age,
                };
                if Self::cmp_entries(&proposal, &self.entries[worst_index])
                    != std::cmp::Ordering::Less
                {
                    return Ok(Admission::Discarded(proposed));
                }
                let evicted = self.remove_index(worst_index);
                let age = self.fresh_age();
                self.entries.push(Entry {
                    item: proposed,
                    age,
                });
                self.rebuild_indices();
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.fresh_age();
        self.entries.push(Entry {
            item: proposed,
            age,
        });
        self.rebuild_indices();
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.ordered_indices()
            .first()
            .map(|&i| self.entries[i].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let index = self.ordered_indices().first().copied()?;
        Some(self.remove_index(index))
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = Self::canonicalize(key)?;
        Ok(self
            .indices
            .get(&canonical)
            .map(|&index| self.entries[index].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let canonical = Self::canonicalize(key)?;
        let index = *self
            .indices
            .get(&canonical)
            .ok_or_else(|| Error::Missing {
                key: canonical.clone(),
            })?;
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let canonical = Self::canonicalize(key)?;
        let index = *self
            .indices
            .get(&canonical)
            .ok_or_else(|| Error::Missing {
                key: canonical.clone(),
            })?;
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let canonical = Self::canonicalize(key)?;
        let index = *self
            .indices
            .get(&canonical)
            .ok_or_else(|| Error::Missing {
                key: canonical.clone(),
            })?;
        Ok(std::mem::replace(
            &mut self.entries[index].item.value,
            value.to_owned(),
        ))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let canonical = Self::canonicalize(key)?;
        let index = *self
            .indices
            .get(&canonical)
            .ok_or_else(|| Error::Missing {
                key: canonical.clone(),
            })?;
        Ok(self.remove_index(index))
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = self.trim_to_capacity();
        evicted.sort_by(|a, b| {
            // Sort using the entries' former ordering is not possible after removal;
            // collect ordering before trimming below instead.
            a.key.cmp(&b.key)
        });
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.ordered_indices()
            .into_iter()
            .map(|i| self.entries[i].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order = other.ordered_indices();
        if policy == CollisionPolicy::Reject {
            for &index in &source_order {
                let key = &other.entries[index].item.key;
                if self.indices.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let source_items: Vec<Item> = source_order
            .into_iter()
            .map(|i| other.entries[i].item.clone())
            .collect();

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for item in source_items {
            if let Some(&index) = self.indices.get(&item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = item.priority;
                        self.entries[index].item.value = item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                self.entries.push(Entry { item, age });
                self.indices
                    .insert(self.entries.last().unwrap().item.key.clone(), self.entries.len() - 1);
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.indices.clear();
        report.evicted = self.trim_to_capacity();
        Ok(report)
    }
}
```
