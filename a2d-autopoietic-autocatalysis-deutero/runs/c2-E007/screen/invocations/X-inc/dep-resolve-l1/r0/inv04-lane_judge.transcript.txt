                name: root.package.clone(),

            for (k, dep) in release.deps.iter().enumerate() {

                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let package_map: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

        std::collections::BTreeMap::new();

            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);

        std::collections::BTreeMap::new();

            canonicalize(constraints);

            .keys()

                package: package.clone(),
            });
        }

            if let Some(constraints) = active.get(package) {
                if !satisfies(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let mut next: Option<String> = None;

                    .releases
                    .iter()
                    .any(|release| satisfies(release.version, constraints));

                    return Err(ResolveError::NoMatch {
                        package: package.clone(),
                        requirements: constraints.clone(),
                    });
                }
                if next.is_none() {
                    next = Some(package.clone());
                }
            }
        }

        let Some(package) = next else {
            return Ok(selected
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());

        let constraints = &active[&package];

        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }

fn matches_constraint(version: Version, constraint: Constraint) -> bool {

        Constraint::Exact(expected) => version == expected,
        Constraint::AtLeast(minimum) => version >= minimum,

        Constraint::Caret(base) => {

    }

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_indices = std::collections::BTreeMap::new();
    for (i, package) in catalog.iter().enumerate() {
        if package_indices.insert(package.name.as_str(), i).is_some() {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

    loop {
        if let Some(package) = active
            .keys()
            .find(|name| !package_indices.contains_key(name.as_str()))
        {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }

        for (package, version) in &selected {
            let requirements = canonical(active.get(package).cloned().unwrap_or_default());
            if !matches_all(*version, &requirements) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements,
                });
            }
        }

        for (package, requirements) in &active {
            if selected.contains_key(package) {
                continue;
            }
            let catalog_package = &catalog[package_indices[package.as_str()]];
            if !catalog_package
                .releases
                .iter()
                .any(|release| matches_all(release.version, requirements))
            {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical(requirements.clone()),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(package) = next else {
            break;
        };

        let requirements = &active[&package];
        let catalog_package = &catalog[package_indices[package.as_str()]];
        let release = catalog_package
            .releases
            .iter()
            .filter(|release| matches_all(release.version, requirements))
            .max_by_key(|release| release.version)
            .expect("candidate existence checked above");

        selected.insert(package, release.version);
        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

fn resolve_node(
    catalog: &[Package],
    constraints: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    for package in constraints.keys() {
        if !catalog.iter().any(|entry| entry.name == *package) {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }
    }

    for (package, version) in selected.iter() {
        if let Some(active) = constraints.get(package) {
            if !matches_all(*version, active) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: canonical_constraints(active),
                });
            }
        }
    }

    for (package, active) in constraints.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let entry = catalog.iter().find(|entry| entry.name == *package).unwrap();
        if !entry
            .releases
            .iter()
            .any(|release| matches_all(release.version, active))
        {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements: canonical_constraints(active),
            });
        }
    }

    let next = constraints
        .keys()
        .find(|package| !selected.contains_key(*package))
        .cloned();

    let Some(package_name) = next else {
        return Ok(());
    };

    let entry = catalog
        .iter()
        .find(|entry| entry.name == package_name)
        .unwrap();
    let active = constraints.get(&package_name).unwrap();
    let release = entry
        .releases
        .iter()
        .filter(|release| matches_all(release.version, active))
        .max_by_key(|release| release.version)
        .unwrap();

    let version = release.version;
    let dependencies = release.deps.clone();

    selected.insert(package_name.clone(), version);
    for dependency in &dependencies {
        constraints
            .entry(dependency.package.clone())
            .or_default()
            .push(dependency.constraint);
    }

    resolve_node(catalog, constraints, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_input(catalog, roots)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut constraints: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected = std::collections::BTreeMap::new();
    resolve_node(catalog, &mut constraints, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}
