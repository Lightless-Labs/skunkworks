fn parse_specifier<'a>(
    text: &'a str,
    start: usize,
    end: usize,
) -> Result<Specifier<'a>, Error> {
    if start == end {
        return Err(Error::EmptyVariable { pos: start });
    }

    let bytes = text.as_bytes();
    let mut modifier_at = end;
    for i in start..end {
        if bytes[i] == b'*' || bytes[i] == b':' {
            modifier_at = i;
            break;
        }
    }

    if modifier_at == start {
        return Err(Error::InvalidName { pos: start });
    }

    let name_bytes = &bytes[start..modifier_at];
    if !is_name_start(name_bytes[0]) {
        return Err(Error::InvalidName { pos: start });
    }
    for (offset, byte) in name_bytes.iter().enumerate().skip(1) {
        if !is_name_continue(*byte) {
            return Err(Error::InvalidName {
                pos: start + offset,
            });
        }
    }

    let modifier = if modifier_at == end {
        Modifier::None
    } else {
        let suffix = &bytes[modifier_at..end];
        match suffix {
            [b'*'] => Modifier::Explode,
            [b':', first, rest @ ..]
                if (b'1'..=b'9').contains(first)
                    && (1..=4).contains(&suffix.len().saturating_sub(1))
                    && rest.iter().all(u8::is_ascii_digit) =>
            {
                let mut value = 0usize;
                for digit in &suffix[1..] {
                    value = value * 10 + usize::from(*digit - b'0');
                }
                if (1..=9999).contains(&value) {
                    Modifier::Prefix(value)
                } else {
                    return Err(Error::InvalidModifier { pos: modifier_at });
                }
            }
            _ => return Err(Error::InvalidModifier { pos: modifier_at }),
        }
    };

    Ok(Specifier {
        name: &text[start..modifier_at],
        modifier,
        start,
    })
}

fn parse_expression<'a>(
    template: &'a str,
    content_start: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut pos = content_start;
    let operator = match bytes.get(pos).copied() {
        Some(b'+') => {
            pos += 1;
            Operator::Reserved
        }
        Some(b'#') => {
            pos += 1;
            Operator::Fragment
        }
        Some(b'.') => {
            pos += 1;
            Operator::Dot
        }
        Some(b'/') => {
            pos += 1;
            Operator::Slash
        }
        Some(b';') => {
            pos += 1;
            Operator::Semicolon
        }
        Some(b'?') => {
            pos += 1;
            Operator::Query
        }
        Some(b'&') => {
            pos += 1;
            Operator::Continuation
        }
        Some(b'=') | Some(b',') | Some(b'!') | Some(b'@') | Some(b'|') => {
            return Err(Error::UnsupportedOperator { pos });
        }
        _ => Operator::Simple,
    };

    if pos == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut spec_start = pos;

    for i in pos..=close {
        if i == close || bytes[i] == b',' {
            let spec = parse_specifier(template, spec_start, i)?;
            if !seen.insert(spec.name) {
                return Err(Error::DuplicateVariable { pos: spec.start });
            }
            specs.push(spec);
            spec_start = i + 1;
        }
    }

    Ok(Expression {
        operator,
        specs,
        close,
    })
}

fn encode(input: &str, reserved: bool) -> String {
    fn safe(byte: u8, reserved: bool) -> bool {
        byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || (reserved
                && matches!(
                    byte,
                    b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$'
                        | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
                ))
    }

    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' && i + 2 < bytes.len() {
            let a = bytes[i + 1];
            let b = bytes[i + 2];
            if a.is_ascii_hexdigit() && b.is_ascii_hexdigit() {
                out.push('%');
                out.push(char::from(a.to_ascii_uppercase()));
                out.push(char::from(b.to_ascii_uppercase()));
                i += 3;
                continue;
            }
        }

        let byte = bytes[i];
        if safe(byte, reserved) {
            out.push(char::from(byte));
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(char::from(HEX[(byte >> 4) as usize]));
            out.push(char::from(HEX[(byte & 0x0f) as usize]));
        }
        i += 1;
    }
    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        _ => format!("{name}={value}"),
    }
}

fn expand_expression(
    expression: &Expression<'_>,
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let operator = expression.operator;
    let reserved = matches!(operator, Operator::Reserved | Operator::Fragment);
    let exploded = |modifier: Modifier| matches!(modifier, Modifier::Explode);

    let mut fields = Vec::new();

    for spec in &expression.specs {
        let Some(value) = lookup(spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) {
            match value {
                Value::Scalar(_) => {}
                Value::List(_) | Value::Map(_) => {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
            }
        }

        match value {
            Value::Scalar(raw) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(count) => raw.chars().take(count).collect::<String>(),
                    Modifier::None | Modifier::Explode => raw.clone(),
                };
                let encoded = encode(&selected, reserved);
                if matches!(
                    operator,
                    Operator::Semicolon | Operator::Query | Operator::Continuation
                ) {
                    fields.push(named_field(spec.name, encoded, operator));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if exploded(spec.modifier) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        if matches!(
                            operator,
                            Operator::Semicolon | Operator::Query | Operator::Continuation
                        ) {
                            fields.push(named_field(spec.name, encoded, operator));
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(
                        operator,
                        Operator::Semicolon | Operator::Query | Operator::Continuation
                    ) {
                        fields.push(named_field(spec.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key.as_str()) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if entries.is_empty() {
                    continue;
                }

                if exploded(spec.modifier) {
                    for (key, map_value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(map_value, reserved);
                        let field = if operator == Operator::Semicolon
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, map_value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(map_value, reserved));
                    }
                    let joined = components.join(",");
                    if matches!(
                        operator,
                        Operator::Semicolon | Operator::Query | Operator::Continuation
                    ) {
                        fields.push(named_field(spec.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[cursor + 1..].iter().position(|b| *b == b'}')
                else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = cursor + 1 + relative_close;

                if let Some(relative_nested) =
                    bytes[cursor + 1..close].iter().position(|b| *b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + relative_nested,
                    });
                }

                let expression = parse_expression(template, cursor + 1, close)?;
                expressions.push((open, close + 1, expression));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut result = String::new();
    let mut literal_start = 0;
    for (start, end, expression) in &expressions {
        result.push_str(&template[literal_start..*start]);
        result.push_str(&expand_expression(expression, vars)?);
        literal_start = *end;
    }
    result.push_str(&template[literal_start..]);
    Ok(result)
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;
    let mut operator = None;

    if content_start < close {
        let first = bytes[content_start] as char;
        if matches!(first, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            operator = Some(first);
            content_start += 1;
        } else if matches!(first, '=' | '!' | '@' | '|') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specifiers = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut part_start = content_start;

    loop {
        let part_end = bytes[part_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| part_start + offset)
            .unwrap_or(close);

            return Err(Error::EmptyVariable { pos: part_start });
        }

        let part = &template[part_start..part_end];
        let modifier_offset = part
            .bytes()
            .position(|b| b == b'*' || b == b':')
            .unwrap_or(part.len());
        let name = &part[..modifier_offset];

        if name.is_empty() {
            return Err(Error::InvalidName { pos: part_start });
        }

            let valid = if offset == 0 {
                is_name_start(byte)
            } else {
                is_name_continue(byte)
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: part_start + offset,
                });
            }
        }

        let modifier = if modifier_offset == part.len() {
            None
        } else {
            let modifier_pos = part_start + modifier_offset;
            let suffix = &part[modifier_offset..];
            if suffix == "*" {
                Some(Modifier::Explode)
            } else if let Some(digits) = suffix.strip_prefix(':') {
                let valid_digits = (1..=4).contains(&digits.len())
                    && digits
                        .bytes()
                        .all(|b| b.is_ascii_digit())
                    && digits.as_bytes()[0] != b'0';
                if !valid_digits {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                let count = digits.parse::<usize>().unwrap_or(0);
                if !(1..=9999).contains(&count) {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Some(Modifier::Prefix(count))
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        if !seen.insert(name.to_string()) {
            return Err(Error::DuplicateVariable { pos: part_start });
        }

        specifiers.push(Specifier {
            name,
            name_start: part_start,
            start: part_start,
            modifier,
        });

        if part_end == close {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        specifiers,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };
                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }
                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn encode(input: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if hex_value(bytes[i + 1]).is_some() && hex_value(bytes[i + 2]).is_some() {
                out.push('%');
                out.push(bytes[i + 1].to_ascii_uppercase() as char);
                out.push(bytes[i + 2].to_ascii_uppercase() as char);
                i += 3;
                continue;
            }
        }
        if is_unreserved(b) || (allow_reserved && is_reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn encode_prefix(input: &str, count: usize, allow_reserved: bool) -> String {
    let prefix: String = input.chars().take(count).collect();
    encode(&prefix, allow_reserved)
}

fn lookup<'a>(vars: &'a [(&str, Value)], name: &str) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn encode_scalar(value: &str, modifier: Option<Modifier>, allow_reserved: bool) -> String {
    match modifier {
        Some(Modifier::Prefix(count)) => encode_prefix(value, count, allow_reserved),
        _ => encode(value, allow_reserved),
    }
}

fn named_field(name: &str, value: String, operator: char) -> String {
    if operator == ';' && value.is_empty() {
        name.to_string()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expr.operator.unwrap_or('\0');
    let allow_reserved = matches!(operator, '+' | '#');
    let mut fields = Vec::new();

    for spec in &expr.specifiers {
        let Some(value) = lookup(vars, spec.name) else {
            continue;
        };

        if matches!(spec.modifier, Some(Modifier::Prefix(_)))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(s) => {
                let encoded = encode_scalar(s, spec.modifier, allow_reserved);
                if matches!(operator, ';' | '?' | '&') {
                    fields.push(named_field(spec.name, encoded, operator));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                let exploded = spec.modifier == Some(Modifier::Explode);
                if items.is_empty() {
                    continue;
                }
                if exploded {
                    for item in items {
                        let encoded = encode(item, allow_reserved);
                        if matches!(operator, ';' | '?' | '&') {
                            fields.push(named_field(spec.name, encoded, operator));
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(operator, ';' | '?' | '&') {
                        fields.push(named_field(spec.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if entries.is_empty() {
                    continue;
                }
                if spec.modifier == Some(Modifier::Explode) {
                    for (key, val) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(val, allow_reserved);
                        if operator == ';' && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, allow_reserved));
                        components.push(encode(val, allow_reserved));
                    }
                    let joined = components.join(",");
                    if matches!(operator, ';' | '?' | '&') {
                        fields.push(named_field(spec.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match operator {
        '+' => ("", ","),
        '#' => ("#", ","),
        '.' => (".", "."),
        '/' => ("/", "/"),
        ';' => (";", ";"),
        '?' => ("?", "&"),
        '&' => ("&", "&"),
        _ => ("", ","),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut out = String::new();
    let mut cursor = 0;

    for expr in &expressions {
        out.push_str(&template[cursor..expr.start]);
        out.push_str(&expand_expression(expr, vars)?);
        cursor = expr.end;
    }

    out.push_str(&template[cursor..]);
    Ok(out)
}
