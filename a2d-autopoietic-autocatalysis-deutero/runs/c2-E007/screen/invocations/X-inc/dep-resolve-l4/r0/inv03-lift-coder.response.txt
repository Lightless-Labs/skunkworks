use std::collections::{BTreeMap, BTreeSet};

    pub patch: u32,
}

    Exact(Version),

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
        evidence: Vec<Evidence>,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    MissingFeature {
        package: String,
        selected: Version,
        feature: String,
        path: Vec<String>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn constraint_matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn canonical_constraints(mut values: Vec<Constraint>) -> Vec<Constraint> {
    values.sort();
    values.dedup();
    values
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sort_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut e = evidence.to_vec();
    e.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    e
}

fn append_evidence(text: &mut String, evidence: &[Evidence]) {
    for e in sort_evidence(evidence) {
        text.push_str("\n  C:");
        text.push_str(&constraint_text(e.constraint));
        text.push_str(" via ");
        text.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature { package, version, feature } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch { package, requirements, evidence } => {
            let cs = canonical_constraints(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!("no-match|package={}|constraints={}", encode(package), cs);
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict { package, selected, requirements, evidence } => {
            let cs = canonical_constraints(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                cs
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature { package, selected, feature, path } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_name(name: &str, site: String) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName { site, name: name.to_string() })
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn target_name(target: Target) -> &'static str {
    match target {
        Target::Linux => "linux",
        Target::Mac => "mac",
        Target::Windows => "windows",
    }
}

fn platform_active(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn dependency_token(source: &str, version: Version, index: usize, dep: &Dependency, target: Target) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let mut fs = dep.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs.join(","),
        gate,
        match dep.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn root_token(index: usize, req: &Requirement) -> String {
    let mut fs = req.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        req.package,
        constraint_text(req.constraint),
        fs.join(",")
    )
}

fn insert_path(paths: &mut BTreeMap<String, Vec<String>>, package: &str, path: Vec<String>) -> bool {
    let replace = match paths.get(package) {
        None => true,
        Some(old) => path.len() < old.len() || (path.len() == old.len() && path < *old),
    };
    if replace {
        paths.insert(package.to_string(), path);
    }
    replace
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut active = Active::default();
    for (i, root) in roots.iter().enumerate() {
        active.origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: vec![root_token(i, root)],
        });
        active.features.entry(root.package.clone()).or_default().extend(root.features.iter().cloned());
        insert_path(&mut active.paths, &root.package, vec![root_token(i, root)]);
    }

    loop {
        let mut changed = false;
        let packages: Vec<(String, Version)> = selected.iter().map(|(p, v)| (p.clone(), *v)).collect();

        for (package_name, version) in packages {
            let Some(release) = catalog
                .get(&package_name)
                .and_then(|p| p.releases.iter().find(|r| r.version == version))
            else {
                continue;
            };

            let requested: Vec<String> = active
                .features
                .get(&package_name)
                .map(|s| s.iter().cloned().collect())
                .unwrap_or_default();

            for feature_name in requested {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if active.features.entry(package_name.clone()).or_default().insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            let activated: BTreeSet<String> = active
                .features
                .get(&package_name)
                .cloned()
                .unwrap_or_default();

            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }

                let token = dependency_token(&package_name, version, j, dep, target);
                let source_path = active.paths.get(&package_name).cloned().unwrap_or_default();
                let mut dep_path = source_path;
                dep_path.push(token);

                let origins = active.origins.entry(dep.package.clone()).or_default();
                if !origins.iter().any(|o| o.constraint == dep.constraint && o.path == dep_path) {
                    origins.push(Origin { constraint: dep.constraint, path: dep_path.clone() });
                    changed = true;
                }

                if insert_path(&mut active.paths, &dep.package, dep_path) {
                    changed = true;
                }

                let requested_features = active.features.entry(dep.package.clone()).or_default();
                for f in &dep.features {
                    if requested_features.insert(f.clone()) {
                        changed = true;
                    }
                }
            }
        }

        if !changed {
            break;
        }
    }

    active
}

fn canonical_path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn sorted_evidence(origins: &[Origin]) -> Vec<Evidence> {
    let mut result: Vec<Evidence> = origins
        .iter()
        .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() })
        .collect();
    result.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    result
}

fn constraints_for(active: &Active, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        active
            .origins
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| o.constraint)
            .collect(),
    )
}

fn feature_error(
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    active: &Active,
) -> Option<ResolveError> {
    for (package, version) in selected {
        let Some(release) = catalog
            .get(package)
            .and_then(|p| p.releases.iter().find(|r| r.version == *version))
        else {
            continue;
        };
        let activated = active.features.get(package)?;
        for feature in activated {
            if !release.features.iter().any(|f| f.name == *feature) {
                return Some(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: active.paths.get(package).cloned().unwrap_or_default(),
                });
            }
        }
    }
    None
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
    all_names: Vec<String>,
}

impl Search<'_> {
    fn evaluate_node(&mut self, selected: &BTreeMap<String, Version>) -> Result<Option<String>, ResolveError> {
        let active = compute_active(self.catalog, self.roots, selected, self.target);

        for package in active.origins.keys() {
            if !self.catalog.contains_key(package) {
                return Err(ResolveError::Missing {
                    package: package.clone(),
                    evidence: sorted_evidence(&active.origins[package]),
                });
            }
        }

        if let Some(error) = feature_error(self.catalog, selected, &active) {
            return Err(error);
        }

        for (package, version) in selected {
            let constraints = constraints_for(&active, package);
            if constraints.iter().any(|c| !constraint_matches(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: sorted_evidence(active.origins.get(package).map(Vec::as_slice).unwrap_or(&[])),
                });
            }
        }

        for package in active.origins.keys() {
            if selected.contains_key(package) {
                continue;
            }
            let constraints = constraints_for(&active, package);
            let matches = self.catalog[package]
                .releases
                .iter()
                .any(|r| constraints.iter().all(|c| constraint_matches(*c, r.version)));
            if !matches {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints,
                    evidence: sorted_evidence(&active.origins[package]),
                });
            }
        }

        Ok(active.origins.keys().find(|p| !selected.contains_key(*p)).cloned())
    }

    fn solution_cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(p, v)| selected.get(*p) != Some(*v))
            .count()
    }

    fn better_solution(&self, cost: usize, selected: &BTreeMap<String, Version>) -> bool {
        let Some((best_cost, best)) = &self.best else {
            return true;
        };
        if cost != *best_cost {
            return cost < *best_cost;
        }
        for name in &self.all_names {
            match (selected.get(name), best.get(name)) {
                (None, None) => {}
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn search(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let next = match self.evaluate_node(selected) {
            Ok(next) => next,
            Err(error) => {
                if let ResolveError::Limit { .. } = error {
                    return Err(error);
                }
                if self.first_error.is_none() {
                    self.first_error = Some(error);
                }
                return Ok(());
            }
        };

        let Some(package) = next else {
            let cost = self.solution_cost(selected);
            if self.better_solution(cost, selected) {
                self.best = Some((cost, selected.clone()));
            }
            return Ok(());
        };

        let constraints = constraints_for(&compute_active(self.catalog, self.roots, selected, self.target), &package);
        let mut candidates: Vec<Version> = self.catalog[&package]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| constraint_matches(*c, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.get(&package) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let chosen = candidates.remove(pos);
                candidates.insert(0, chosen);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit { attempts: self.max_attempts });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let result = self.search(selected);
            selected.remove(&package);
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        validate_name(&root.package, format!("root[{}]", i))?;
        for (j, feature) in root.features.iter().enumerate() {
            validate_name(feature, format!("root[{}].feature[{}]", i, j))?;
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        validate_name(&package.name, format!("catalog[{}]", i))?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                validate_name(&dep.package, format!("catalog[{}].release[{}].dep[{}]", i, j, k))?;
                if let Some(gate) = &dep.gate {
                    validate_name(gate, format!("catalog[{}].release[{}].dep[{}].gate", i, j, k))?;
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    validate_name(
                        feature,
                        format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f),
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                validate_name(
                    &feature.name,
                    format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                )?;
                for (e, enabled) in feature.enables.iter().enumerate() {
                    validate_name(
                        enabled,
                        format!("catalog[{}].release[{}].feature[{}].enable[{}]", i, j, f, e),
                    )?;
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: package.name.clone() });
        }
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        validate_name(&lock.package, format!("lock[{}]", i))?;
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> =
        locks.iter().map(|l| (l.package.clone(), l.version)).collect();
    let mut all_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    all_names.sort();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        all_names,
    };

    let mut selected = BTreeMap::new();
    search.search(&mut selected)?;

    let Some((_, best)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_active = compute_active(&catalog_map, roots, &best, target);
    let selected_vec: Vec<Selection> = best
        .iter()
        .map(|(package, version)| Selection { package: package.clone(), version: *version })
        .collect();

    let mut packages: BTreeSet<String> = lock_map.keys().cloned().collect();
    packages.extend(best.keys().cloned());
    let mut changes = Vec::new();
    for package in packages {
        let before = lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = constraints_for(&final_active, &package);
                    if constraints.iter().any(|c| !constraint_matches(*c, old)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change { package, before, after, reason });
    }

    Ok(Resolution { selected: selected_vec, changes })
}
