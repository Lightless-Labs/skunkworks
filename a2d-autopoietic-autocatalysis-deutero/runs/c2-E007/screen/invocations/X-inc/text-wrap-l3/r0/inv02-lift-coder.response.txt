#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<bool>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    word_id: usize,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<Piece>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn piece_width(line: &Line) -> usize {
    line.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
        + line.pieces.len().saturating_sub(1)
}

fn render_line(line: &Line, width: usize, justify: bool, is_final: bool, line_index: usize) -> String {
    let content_width = piece_width(line);
    let gaps = line.pieces.len().saturating_sub(1);
    let extra = if justify && !is_final && gaps > 0 {
        width.saturating_sub(line.indent + content_width)
    } else {
        0
    };

    let base = if gaps > 0 { extra / gaps } else { 0 };
    let rem = if gaps > 0 { extra % gaps } else { 0 };
    let start = if gaps > 0 { line_index % gaps } else { 0 };

    let mut out = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            let gap_index = i - 1;
            let bonus = if gaps > 0 && (gap_index + gaps - start) % gaps < rem {
                1
            } else {
                0
            };
            out.push_str(&" ".repeat(1 + base + bonus));
        }
        out.push_str(&piece.text);
    }
    out
}

fn parse_segments(input: &str) -> Vec<Vec<Word>> {
    let normalized = input.replace("\r\n", "\n").replace('\r', "\n");
    let chars: Vec<char> = normalized.chars().collect();
    let mut segments = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            if i - start >= 2 {
                segments.push(current);
                current = Vec::new();
                for _ in 1..(i - start) {
                    segments.push(Vec::new());
                }
                continue;
            }
            current.push(Word { chars: vec!['\n'], markers: vec![false] });
            continue;
        }
        current.push(Word { chars: vec![chars[i]], markers: vec![false] });
        i += 1;
    }

    segments.push(current);

    segments
        .into_iter()
        .map(|items| {
            let mut words = Vec::new();
            let mut chars = Vec::new();
            let mut marks = Vec::new();
            for item in items {
                let c = item.chars[0];
                if is_separator(c) {
                    if !chars.is_empty() {
                        words.push(Word { chars: std::mem::take(&mut chars), markers: std::mem::take(&mut marks) });
                    }
                } else {
                    chars.push(c);
                    marks.push(false);
                }
            }
            if !chars.is_empty() {
                words.push(Word { chars, markers: marks });
            }
            words
        })
        .collect()
}

fn parse_with_markers(input: &str) -> Vec<Vec<Word>> {
    let normalized = input.replace("\r\n", "\n").replace('\r', "\n');
    let chars: Vec<char> = normalized.chars().collect();
    let mut segments: Vec<Vec<Word>> = Vec::new();
    let mut words = Vec::new();
    let mut word_chars = Vec::new();
    let mut markers = Vec::new();
    let mut i = 0;

    let flush_word = |words: &mut Vec<Word>, wc: &mut Vec<char>, wm: &mut Vec<bool>| {
        if !wc.is_empty() {
            words.push(Word { chars: std::mem::take(wc), markers: std::mem::take(wm) });
        }
    };

    while i < chars.len() {
        let c = chars[i];
        if c == '\n' {
            let mut j = i;
            while j < chars.len() && chars[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                flush_word(&mut words, &mut word_chars, &mut markers);
                segments.push(std::mem::take(&mut words));
                for _ in 1..(j - i) {
                    segments.push(Vec::new());
                }
                i = j;
                continue;
            }
        }

        if is_separator(c) {
            flush_word(&mut words, &mut word_chars, &mut markers);
        } else if c == '~' {
            word_chars.push(c);
            markers.push(true);
        } else {
            word_chars.push(c);
            markers.push(false);
        }
        i += 1;
    }
    flush_word(&mut words, &mut word_chars, &mut markers);
    segments.push(words);
    segments
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for (pos, &(byte_index, c)) in original.iter().enumerate() {
        if c != '~' {
            continue;
        }
        let valid_neighbor = |item: Option<&(usize, char)>| {
            item.map(|(_, ch)| *ch != '~' && !is_separator(*ch)).unwrap_or(false)
        };
        if !valid_neighbor(pos.checked_sub(1).and_then(|p| original.get(p)))
            || !valid_neighbor(original.get(pos + 1))
        {
            return Err(WrapError::InvalidMarker { index: byte_index });
        }
    }

    let segments = parse_with_markers(input);
    let mut output = Vec::new();

    for words in segments {
        if words.is_empty() {
            continue;
        }

        let mut lines: Vec<Line> = Vec::new();
        let mut line_no = 0usize;
        let mut line = Line { indent: options.first_indent, pieces: Vec::new() };

        for (word_id, word) in words.iter().enumerate() {
            let mut offset = 0usize;
            let n = word.chars.len();

            while offset < n {
                let capacity = options.width - line.indent;
                let current_len = piece_width(&line);
                let room = capacity.saturating_sub(current_len + usize::from(!line.pieces.is_empty()));
                let suffix_len = n - offset;
                if suffix_len <= room {
                    let text: String = word.chars[offset..]
                        .iter()
                        .enumerate()
                        .filter_map(|(k, &c)| if word.markers[offset + k] { None } else { Some(c) })
                        .collect();
                    if !text.is_empty() {
                        line.pieces.push(Piece { text, word_id });
                    }
                    offset = n;
                    continue;
                }

                let available = if line.pieces.is_empty() { capacity } else { capacity.saturating_sub(current_len) };
                let mut opportunity = None;
                for marker in offset + 1..n {
                    if word.markers[marker] {
                        let prefix_len = marker - offset;
                        if prefix_len + 1 <= available {
                            opportunity = Some(marker);
                        }
                    }
                }

                let accepted = if let Some(marker) = opportunity {
                    let used = marker - offset + 1;
                    let unused = available - used;
                    if line.pieces.is_empty() {
                        Some(marker)
                    } else {
                        let baseline = capacity.saturating_sub(current_len);
                        if unused * unused + options.hyphen_cost < baseline * baseline {
                            Some(marker)
                        } else {
                            None
                        }
                    }
                } else {
                    None
                };

                if let Some(marker) = accepted {
                    let mut text: String = word.chars[offset..marker]
                        .iter()
                        .enumerate()
                        .filter_map(|(k, &c)| if word.markers[offset + k] { None } else { Some(c) })
                        .collect();
                    text.push('-');
                    line.pieces.push(Piece { text, word_id });
                    lines.push(line);
                    line_no += 1;
                    line = Line { indent: options.rest_indent, pieces: Vec::new() };
                    offset = marker + 1;
                } else if !line.pieces.is_empty() {
                    lines.push(line);
                    line_no += 1;
                    line = Line { indent: options.rest_indent, pieces: Vec::new() };
                } else {
                    let raw_count = capacity.min(suffix_len);
                    let text: String = word.chars[offset..offset + raw_count]
                        .iter()
                        .enumerate()
                        .filter_map(|(k, &c)| if word.markers[offset + k] { None } else { Some(c) })
                        .collect();
                    line.pieces.push(Piece { text, word_id });
                    lines.push(line);
                    line_no += 1;
                    line = Line { indent: options.rest_indent, pieces: Vec::new() };
                    offset += raw_count;
                }
            }
        }

        if !line.pieces.is_empty() {
            lines.push(line);
        }

        let final_index = lines.len().saturating_sub(1);
        for (idx, line) in lines.iter().enumerate() {
            output.push(render_line(line, options.width, options.justify, idx == final_index, idx));
        }
    }

    output
}
