#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(char, usize)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].0 != '~' {
            continue;
        }
        let valid_neighbor = |j: usize| {
            j < chars.len()
                && chars[j].0 != '~'
                && !is_separator(chars[j].0)
        };
        if i == 0 || i + 1 >= chars.len() || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].1 });
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let run_start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                layout_segment(&normalized[segment_start..run_start], options, &mut output);
                for _ in 1..(i - run_start) {
                    output.push(String::new());
                }
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }
    layout_segment(&normalized[segment_start..], options, &mut output);

    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    word_id: usize,
}

struct Line {
    indent: usize,
    pieces: Vec<Piece>,
    gaps: usize,
    content_width: usize,
}

fn layout_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words = Vec::new();
    let mut i = 0;
    while i < segment.len() {
        while i < segment.len() && is_separator(segment[i]) {
            i += 1;
        }
        if i == segment.len() {
            break;
        }

        let mut chars = Vec::new();
        let mut breaks = Vec::new();
        while i < segment.len() && !is_separator(segment[i]) {
            let ch = segment[i];
            if ch == '~' {
                breaks.push(chars.len());
            } else {
                chars.push(ch);
            }
            i += 1;
        }
        words.push(Word { chars, breaks });
    }

    if words.is_empty() {
        return;
    }

    let mut lines = Vec::new();
    let mut line_number = 0;
    let mut line = new_line(line_number, options);
    let mut word_index = 0;

    while word_index < words.len() {
        let word = &words[word_index];
        let mut offset = 0;
        while offset < word.chars.len() {
            let is_empty = line.pieces.is_empty();
            let capacity = options.width - line.indent;
            let room = capacity.saturating_sub(line.content_width);
            let gap = usize::from(!is_empty);
            let suffix_len = word.chars.len() - offset;

            if suffix_len + gap <= room {
                if gap != 0 {
                    line.gaps += 1;
                    line.content_width += 1;
                }
                let text: String = word.chars[offset..].iter().collect();
                line.content_width += suffix_len;
                line.pieces.push(Piece {
                    text,
                    word_id: word_index,
                });
                offset = word.chars.len();
                continue;
            }

            let mut usable = None;
            for &marker in &word.breaks {
                if marker > offset && marker < word.chars.len() {
                    let prefix_len = marker - offset;
                    if prefix_len + 1 <= room {
                        usable = Some((marker, prefix_len));
                    }
                }
            }

            let accepted = if let Some((marker, prefix_len)) = usable {
                let unused = room - prefix_len - 1;
                if is_empty
                    || unused * unused + options.hyphen_cost
                        < room.saturating_sub(gap) * room.saturating_sub(gap)
                {
                    Some((marker, prefix_len))
                } else {
                    None
                }
            } else {
                None
            };

            if let Some((marker, prefix_len)) = accepted {
                if gap != 0 {
                    line.gaps += 1;
                    line.content_width += 1;
                }
                let mut text: String = word.chars[offset..marker].iter().collect();
                text.push('-');
                line.content_width += prefix_len + 1;
                line.pieces.push(Piece {
                    text,
                    word_id: word_index,
                });
                lines.push(line);
                line_number += 1;
                line = new_line(line_number, options);
                offset = marker;
                continue;
            }

            if !is_empty {
                lines.push(line);
                line_number += 1;
                line = new_line(line_number, options);
                continue;
            }

            if suffix_len > capacity {
                let end = offset + capacity;
                let text: String = word.chars[offset..end].iter().collect();
                line.content_width += capacity;
                line.pieces.push(Piece {
                    text,
                    word_id: word_index,
                });
                lines.push(line);
                line_number += 1;
                line = new_line(line_number, options);
                offset = end;
            } else {
                line.content_width += suffix_len;
                let text: String = word.chars[offset..].iter().collect();
                line.pieces.push(Piece {
                    text,
                    word_id: word_index,
                });
                offset = word.chars.len();
            }
        }

        word_index += 1;
    }

    if !line.pieces.is_empty() {
        lines.push(line);
    }

    let final_index = lines.len().saturating_sub(1);
    for (line_index, line) in lines.into_iter().enumerate() {
        let mut rendered = " ".repeat(line.indent);
        let justify = options.justify && line_index < final_index && line.gaps > 0;
        let extra = if justify {
            options.width.saturating_sub(line.indent + line.content_width)
        } else {
            0
        };
        let each = if justify { extra / line.gaps } else { 0 };
        let remainder = if justify { extra % line.gaps } else { 0 };
        let start_gap = if justify { line_index % line.gaps } else { 0 };

        for (piece_index, piece) in line.pieces.iter().enumerate() {
            rendered.push_str(&piece.text);
            if piece_index + 1 < line.pieces.len() {
                rendered.push(' ');
                if justify {
                    rendered.push_str(&" ".repeat(each));
                    let relative_gap = piece_index;
                    let cyclic = (relative_gap + line.gaps - start_gap) % line.gaps;
                    if cyclic < remainder {
                        rendered.push(' ');
                    }
                }
            }
        }
        output.push(rendered);
    }
}

fn new_line(line_number: usize, options: &WrapOptions) -> Line {
    let indent = if line_number == 0 {
        options.first_indent
    } else {
        options.rest_indent
    };
    Line {
        indent,
        pieces: Vec::new(),
        gaps: 0,
        content_width: 0,
    }
}
