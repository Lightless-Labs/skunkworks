#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, usize)>,
}

fn is_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let content_start = open + 1;

    let (operator, mut name_start) = if content_start < close && is_operator(bytes[content_start]) {
        let op_pos = content_start;
        let operator = match bytes[op_pos] {
            b'+' => Operator::Plus,
            b'#' => Operator::Hash,
            _ => return Err(Error::UnsupportedOperator { pos: op_pos }),
        };
        (operator, content_start + 1)
    } else {
        (Operator::Simple, content_start)
    };

    if name_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<(usize, usize)> = Vec::new();

    loop {
        let spec_start = name_start;
        let mut spec_end = spec_start;
        while spec_end < close && bytes[spec_end] != b',' {
            spec_end += 1;
        }

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        if !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let mut i = spec_start + 1;
        while i < spec_end {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
            i += 1;
        }

        if seen.iter().any(|&(start, end)| {
            template[start..end] == template[spec_start..spec_end]
        }) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }

        seen.push((spec_start, spec_end));
        names.push((spec_start, spec_end));

        if spec_end == close {
            break;
        }
        name_start = spec_end + 1;
        if name_start == close {
            return Err(Error::EmptyVariable { pos: close });
        }
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        names,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = open + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(inner) = (open + 1..close).find(|&pos| bytes[pos] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_upper(byte: u8) -> u8 {
    match byte {
        b'a'..=b'f' => byte - b'a' + b'A',
        _ => byte,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn push_encoded(output: &mut String, value: &str, allow_reserved: bool) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push(hex_upper(bytes[i + 1]) as char);
            output.push(hex_upper(bytes[i + 2]) as char);
            i += 3;
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
            i += 1;
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
            i += 1;
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let allow_reserved = expression.operator != Operator::Simple;
        let mut fields = Vec::new();

        for &(start, end) in &expression.names {
            let name = &template[start..end];
            if let Some(value) = lookup(name, vars) {
                let mut encoded = String::new();
                push_encoded(&mut encoded, value, allow_reserved);
                fields.push(encoded);
            }
        }

        if !fields.is_empty() {
            if expression.operator == Operator::Hash {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
