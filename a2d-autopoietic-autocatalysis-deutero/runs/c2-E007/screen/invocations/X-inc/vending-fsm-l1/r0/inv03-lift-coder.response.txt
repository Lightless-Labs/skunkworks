#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    ExactPayment { price: u16, credit: u16 },
    TenderMismatch {
        index: usize,
        expected: u16,
        found: u16,
    },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
}

impl Machine {
    pub fn new(products: Vec<Product>) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());

            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }

            if seen.contains(&product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if ![1, 3, 4, 10].contains(&coin) {
            return Err(Error::BadCoin { coin });
        }

        if self.credit + u32::from(coin) as u16 > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(product) = self.products.iter().find(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };

        if self.selected.as_deref() == Some(product.id.as_str()) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(product.id.clone());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };

        let product = self
            .products
            .iter()
            .find(|product| product.id == *selected_id)
            .expect("selected products always belong to the catalog");

        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        if self.credit > price {
            return Err(Error::ExactPayment {
                price,
                credit: self.credit,
            });
        }

        let mut expected = Vec::new();
        let mut remaining = price;
        for denomination in [10, 4, 3, 1] {
            while remaining >= denomination {
                expected.push(denomination);
                remaining -= denomination;
            }
        }

        let mut found = self.escrow.clone();
        found.sort_unstable_by(|a, b| b.cmp(a));

        for index in 0..expected.len().max(found.len()) {
            if expected.get(index) != found.get(index) {
                let expected_coin = expected[index];
                let found_coin = found[index];
                return Err(Error::TenderMismatch {
                    index,
                    expected: expected_coin,
                    found: found_coin,
                });
            }
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price,
            change: Vec::new(),
        };
        self.selected = None;
        self.escrow.clear();
        self.credit = 0;
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.selected = None;
        self.escrow.clear();
        self.credit = 0;
        Ok(returned)
    }
}
