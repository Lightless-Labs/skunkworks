#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn push_line(
    output: &mut Vec<String>,
    content: &mut String,
    content_len: &mut usize,
    indent: usize,
) {
    let mut line = String::with_capacity(indent + content.len());
    line.extend(std::iter::repeat(' ').take(indent));
    line.push_str(content);
    output.push(line);
    content.clear();
    *content_len = 0;
}

fn layout_segment(segment: &str, options: &WrapOptions, output: &mut Vec<String>) {
    let words: Vec<Vec<char>> = segment
        .split(is_separator)
        .filter(|word| !word.is_empty())
        .map(|word| word.chars().collect())
        .collect();

    let mut first_line = true;
    let mut content = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            let remaining = word.len() - offset;

            if remaining <= capacity {
                let separator_len = usize::from(content_len > 0);
                if content_len + separator_len + remaining <= capacity {
                    if separator_len > 0 {
                        content.push(' ');
                        content_len += 1;
                    }
                    content.extend(word[offset..].iter());
                    content_len += remaining;
                    offset = word.len();
                } else {
                    push_line(output, &mut content, &mut content_len, indent);
                    first_line = false;
                }
            } else if content_len == 0 {
                content.extend(word[offset..offset + capacity].iter());
                content_len = capacity;
                offset += capacity;
                push_line(output, &mut content, &mut content_len, indent);
                first_line = false;
            } else {
                let available = capacity - content_len - 1;
                if available >= 2 {
                    content.push(' ');
                    content.extend(word[offset..offset + available].iter());
                    content_len += 1 + available;
                    offset += available;
                    push_line(output, &mut content, &mut content_len, indent);
                    first_line = false;
                } else {
                    push_line(output, &mut content, &mut content_len, indent);
                    first_line = false;
                }
            }
        }
    }

    if content_len > 0 {
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        push_line(output, &mut content, &mut content_len, indent);
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized = input.replace("\r\n", "\n").replace('\r', "\n");
    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut index = 0usize;

    while index < chars.len() {
        if chars[index] == '\n' {
            let run_start = index;
            while index < chars.len() && chars[index] == '\n' {
                index += 1;
            }
            let run_len = index - run_start;

            if run_len >= 2 {
                let segment: String = chars[segment_start..run_start].iter().collect();
                layout_segment(&segment, options, &mut output);
                output.extend(std::iter::repeat(String::new()).take(run_len - 1));
                segment_start = index;
            }
        } else {
            index += 1;
        }
    }

    let segment: String = chars[segment_start..].iter().collect();
    layout_segment(&segment, options, &mut output);
    Ok(output)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn public_unicode_words() {
        assert_eq!(
            wrap(
                "猫犬 鳥魚",
                &WrapOptions {
                    width: 5,
                    first_indent: 0,
                    rest_indent: 0
                }
            ),
            Ok(vec!["猫犬 鳥魚".to_string()])
        );
    }

    #[test]
    fn public_hanging() {
        assert_eq!(
            wrap(
                "a bb ccc dd",
                &WrapOptions {
                    width: 7,
                    first_indent: 1,
                    rest_indent: 2
                }
            ),
            Ok(vec![
                " a bb".to_string(),
                "  ccc".to_string(),
                "  dd".to_string()
            ])
        );
    }

    #[test]
    fn public_paragraph() {
        assert_eq!(
            wrap(
                "a\n\nb",
                &WrapOptions {
                    width: 4,
                    first_indent: 0,
                    rest_indent: 1
                }
            ),
            Ok(vec!["a".to_string(), "".to_string(), "b".to_string()])
        );
    }

    #[test]
    fn public_bad_indent() {
        assert_eq!(
            wrap(
                "x",
                &WrapOptions {
                    width: 3,
                    first_indent: 3,
                    rest_indent: 0
                }
            ),
            Err(WrapError::FirstIndentTooWide { indent: 3 })
        );
    }
}
