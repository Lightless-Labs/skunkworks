#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::new();
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            output.push(payload);
            break;
        }
        output.push(payload | 0x80);
    }
    output
}

fn decode_from_valid_start(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    let mut value = 0u64;

    for group in 0..10 {
        let offset = start + group;
        let Some(&byte) = input.get(offset) else {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        };

        let payload = (byte & 0x7f) as u64;
        if group == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow { offset, start });
            }
            value |= payload << 63;
            return Ok((value, offset + 1));
        }

        value |= payload << (group * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    decode_from_valid_start(input, start)
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::new();
    let mut cursor = start;
    for _ in 0..count {
        let (value, next) = decode_from_valid_start(input, cursor)?;
        values.push(value);
        cursor = next;
    }
    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let zigzag = if value >= 0 {
        (value as u64) << 1
    } else {
        ((value.unsigned_abs()) << 1).wrapping_sub(1)
    };
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, next) = decode_u64(input, start)?;
    let decoded = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((decoded, next))
}

#[derive(Debug, Clone)]
pub struct StreamDecoder {
    position: usize,
    pending: Vec<u8>,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            pending: Vec::new(),
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut combined = self.pending.clone();
        combined.extend_from_slice(chunk);

        let base = self.position - self.pending.len();
        let mut cursor = 0usize;
        let mut values = Vec::new();

        while cursor < combined.len() {
            let integer_start = cursor;
            let mut value = 0u64;
            let mut terminated = false;

            for group in 0..10 {
                if cursor >= combined.len() {
                    break;
                }

                let byte = combined[cursor];
                let absolute_offset = base + cursor;
                let payload = (byte & 0x7f) as u64;

                if group == 9 {
                    if payload > 1 || byte & 0x80 != 0 {
                        return Err(Error::Overflow {
                            offset: absolute_offset,
                            start: base + integer_start,
                        });
                    }
                    value |= payload << 63;
                    cursor += 1;
                    terminated = true;
                    break;
                }

                value |= payload << (group * 7);
                cursor += 1;
                if byte & 0x80 == 0 {
                    terminated = true;
                    break;
                }
            }

            if terminated {
                values.push(value);
            } else {
                break;
            }
        }

        self.pending = combined[cursor..].to_vec();
        self.position += chunk.len();
        Ok(values)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending.is_empty() {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.position - self.pending.len(),
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending.len()
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending.clear();
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}
