use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    let mut end = start;
    while end < bytes.len() && is_digit(bytes[end]) {
        end += 1;
    }
    if start == end {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut n = 0u64;
    for &b in &bytes[start..end] {
        n = n.checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    *pos = end;
    Ok(n)
}

fn parse_identifiers(
    s: &str,
    pos: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut end = bytes.len();
    if pre {
        if let Some(i) = bytes[pos..].iter().position(|&b| b == b'+') {
            end = pos + i;
        }
    }
    let mut ids = Vec::new();
    let mut start = pos;
    loop {
        let mut stop = start;
        while stop < end && bytes[stop] != b'.' {
            if !bytes[stop].is_ascii_alphanumeric() && bytes[stop] != b'-' {
                return Err(VersionError::InvalidChar { at: stop });
            }
            stop += 1;
        }
        if stop == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
        if pre {
            let id = &bytes[start..stop];
            if id.iter().all(|b| is_digit(*b)) && id.len() > 1 && id[0] == b'0' {
                return Err(VersionError::LeadingZero { at: start });
            }
        }
        ids.push(s[start..stop].to_string());
        if stop == end {
            break;
        }
        start = stop + 1;
        if start == end {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
    }
    Ok(ids)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        if _ == 0 {}
    }
    unreachable!()
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() && b.is_empty() {
        return Ordering::Equal;
    }
    if a.is_empty() {
        return Ordering::Greater;
    }
    if b.is_empty() {
        return Ordering::Less;
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let o = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if o != Ordering::Equal {
            return o;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    comps: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let mut arms = Vec::new();
    let mut cur = Vec::new();
    let mut i = 0;
    let b = s.as_bytes();
    while i < b.len() {
        while i < b.len() && matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == b.len() {
            break;
        }
        let start = i;
        while i < b.len() && !matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        let token = &s[start..i];
        if token == "||" {
            if cur.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(Arm { comps: std::mem::take(&mut cur), anchors: Vec::new() });
            continue;
        }
        let (op, offset) = if token.starts_with(">=") {
            (Op::Ge, 2)
        } else if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with('>') {
            (Op::Gt, 1)
        } else if token.starts_with('<') {
            (Op::Lt, 1)
        } else if token.starts_with('=') {
            (Op::Eq, 1)
        } else {
            (Op::Eq, 0)
        };
        if offset == token.len() || !token.as_bytes()[offset].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }
        let operand = &token[offset..];
        let v = parse_version(operand).map_err(|error| RangeError::Version {
            at: start + offset,
            error,
        })?;
        cur.push(Comparator { op, version: v });
    }
    if cur.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(Arm { comps: cur, anchors: Vec::new() });
    Ok(Range { arms })
}

fn cmp_ok(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, a: &Arm, include_pre: bool) -> bool {
    if !a.comps.iter().all(|c| cmp_ok(v, c)) {
        return false;
    }
    if include_pre || v.pre.is_empty() {
        return true;
    }
    a.comps.iter().any(|c| {
        !c.version.pre.is_empty()
            && (c.version.major, c.version.minor, c.version.patch)
                == (v.major, v.minor, v.patch)
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates.iter().enumerate().map(|(index, s)| {
        parse_version(s).map_err(|error| CandidateError { index, error })
    }).collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for arm_a in &a.arms {
        for arm_b in &b.arms {
            if let Some(v) = find_witness(arm_a, arm_b, include_prerelease) {
                if arm_matches(&v, arm_a, include_prerelease)
                    && arm_matches(&v, arm_b, include_prerelease)
                {
                    return true;
                }
            }
        }
    }
    false
}

fn find_witness(a: &Arm, b: &Arm, include_pre: bool) -> Option<Version> {
    let mut candidates = vec![
        Version { major: 0, minor: 0, patch: 0, pre: vec![], build: vec![] },
        Version { major: u64::MAX, minor: u64::MAX, patch: u64::MAX, pre: vec![], build: vec![] },
    ];
    for c in a.comps.iter().chain(b.comps.iter()) {
        candidates.push(c.version.clone());
        let t = (c.version.major, c.version.minor, c.version.patch);
        candidates.push(Version { major: t.0, minor: t.1, patch: t.2, pre: vec!["0".into()], build: vec![] });
        if c.version.patch < u64::MAX {
            candidates.push(Version { major: t.0, minor: t.1, patch: t.2 + 1, pre: vec![], build: vec![] });
        }
    }
    for t in a.anchors.iter().chain(b.anchors.iter()) {
        candidates.push(Version { major: t.0, minor: t.1, patch: t.2, pre: vec!["0".into()], build: vec![] });
        candidates.push(Version { major: t.0, minor: t.1, patch: t.2, pre: vec![], build: vec![] });
    }
    candidates.into_iter().find(|v| {
        arm_matches(v, a, include_pre) && arm_matches(v, b, include_pre)
    })
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let mut anchors = arm.anchors.clone();
        for c in &arm.comps {
            if !c.version.pre.is_empty() {
                anchors.push((c.version.major, c.version.minor, c.version.patch));
            }
        }
        anchors.sort();
        anchors.dedup();
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        let mut impossible = false;
        for c in &arm.comps {
            let v = c.version.clone();
            let (is_lower, inclusive) = match c.op {
                Op::Eq => {
                    update_bound(&mut lower, v.clone(), true, true);
                    update_bound(&mut upper, v, true, false);
                    continue;
                }
                Op::Gt => (true, false),
                Op::Ge => (true, true),
                Op::Lt => (false, false),
                Op::Le => (false, true),
            };
            if is_lower {
                update_bound(&mut lower, v, inclusive, true);
            } else {
                update_bound(&mut upper, v, inclusive, false);
            }
        }
        if let (Some((l, li)), Some((u, ui))) = (&lower, &upper) {
            let o = compare_versions(l, u);
            if o == Ordering::Greater || (o == Ordering::Equal && (!li || !ui)) {
                impossible = true;
            }
        }
        if impossible {
            continue;
        }
        let lo = lower.map(|(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, version_no_build(&v))).unwrap_or_else(|| "*".into());
        let hi = upper.map(|(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, version_no_build(&v))).unwrap_or_else(|| "*".into());
        let at = if anchors.is_empty() { "-".to_string() } else {
            anchors.iter().map(|t| format!("{}.{}.{}", t.0, t.1, t.2)).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{};{};pre={}]", lo, hi, at));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

fn update_bound(bound: &mut Option<(Version, bool)>, v: Version, inclusive: bool, lower: bool) {
    match bound {
        None => *bound = Some((v, inclusive)),
        Some((old, old_inc)) => {
            let ord = compare_versions(&v, old);
            if (lower && ord == Ordering::Greater) || (!lower && ord == Ordering::Less) {
                *bound = Some((v, inclusive));
            } else if ord == Ordering::Equal && !inclusive {
                *old_inc = false;
            }
        }
    }
}

fn version_no_build(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(expr: &SetExpr, v: &Version, parsed_ranges: &[Range], idx: &mut usize, include: bool) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let r = &parsed_ranges[*idx];
            *idx += 1;
            matches_with(v, r, include)
        }
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, v, parsed_ranges, idx, include)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, v, parsed_ranges, idx, include)),
    }
}

fn gather_ranges(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    out: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            out.push(r);
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                gather_ranges(x, package, label, path, out)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn candidate_satisfies(
    v: &Version,
    constraints: &[Constraint],
    ranges: &[Range],
    include: bool,
) -> bool {
    let mut idx = 0;
    constraints.iter().all(|c| eval_expr(&c.expr, v, ranges, &mut idx, include))
}

fn conflict_labels(
    versions: &[Version],
    constraints: &[Constraint],
    ranges: &[Range],
    include: bool,
) -> Vec<String> {
    if versions.is_empty() {
        return vec![];
    }
    let n = constraints.len();
    for size in 1..=n {
        let mut chosen = Vec::new();
        if let Some(result) = find_conflict_combination(0, size, &mut chosen, versions, constraints, ranges, include) {
            let mut labels = result.into_iter().map(|i| constraints[i].label.clone()).collect::<Vec<_>>();
            labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            return labels;
        }
    }
    vec![]
}

fn find_conflict_combination(
    start: usize,
    left: usize,
    chosen: &mut Vec<usize>,
    versions: &[Version],
    constraints: &[Constraint],
    ranges: &[Range],
    include: bool,
) -> Option<Vec<usize>> {
    if left == 0 {
        let any = versions.iter().any(|v| {
            let mut offset = 0;
            chosen.iter().all(|&i| eval_expr(&constraints[i].expr, v, ranges, &mut offset, include))
        });
        return if any { None } else { Some(chosen.clone()) };
    }
    if constraints.len().saturating_sub(start) < left {
        return None;
    }
    for i in start..=constraints.len() - left {
        chosen.push(i);
        if let Some(r) = find_conflict_combination(i + 1, left - 1, chosen, versions, constraints, ranges, include) {
            return Some(r);
        }
        chosen.pop();
    }
    None
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut parsed_versions = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index: i, error
            })?);
        }
        parsed_versions.push(vs);
    }
    let mut parsed_ranges_all = Vec::new();
    for p in packages {
        let mut rs = Vec::new();
        for c in &p.constraints {
            gather_ranges(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut rs)?;
        }
        parsed_ranges_all.push(rs);
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let vs = &parsed_versions[pi];
        let ranges = &parsed_ranges_all[pi];
        let mut best: Option<usize> = None;
        for (i, v) in vs.iter().enumerate() {
            if !candidate_satisfies(v, &p.constraints, ranges, include_prerelease) {
                continue;
            }
            match best {
                None => best = Some(i),
                Some(j) => {
                    let take = match preference {
                        Preference::Highest => compare_versions(v, &vs[j]) == Ordering::Greater,
                        Preference::Lowest => compare_versions(v, &vs[j]) == Ordering::Less,
                        Preference::StableFirst => {
                            match (v.pre.is_empty(), vs[j].pre.is_empty()) {
                                (true, false) => true,
                                (false, true) => false,
                                _ => compare_versions(v, &vs[j]) == Ordering::Greater,
                            }
                        }
                    };
                    if take { best = Some(i); }
                }
            }
        }
        if let Some(i) = best {
            selections.push(Selection { package: p.name.clone(), version: p.candidates[i].clone() });
        } else {
            let labels = conflict_labels(vs, &p.constraints, ranges, include_prerelease);
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
    }
    Ok(selections)
}
