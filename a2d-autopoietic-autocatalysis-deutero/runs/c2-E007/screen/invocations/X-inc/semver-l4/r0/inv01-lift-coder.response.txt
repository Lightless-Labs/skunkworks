use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_stop(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        let b = bytes.get(start).copied();
        return Err(if is_core_stop(b) {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut id_start = start;
    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..i];
        if prerelease
            && id.len() > 1
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    for _ in 0..2 {
        match s.as_bytes().get(p).copied() {
            Some(b'.') => p += 1,
            b => {
                return Err(if is_core_stop(b) {
                    VersionError::MissingCore { at: p }
                } else {
                    VersionError::InvalidChar { at: p }
                });
            }
        }
        if _ == 0 {}
    }
    // Reparse the second and third fields explicitly; the first loop above only
    // validates separators, so retain their positions while parsing below.
    let mut p = 0;
    let _ = parse_core(s, &mut p)?;
    p += 1;
    let minor = parse_core(s, &mut p)?;
    match s.as_bytes().get(p).copied() {
        Some(b'.') => p += 1,
        b => {
            return Err(if is_core_stop(b) {
                VersionError::MissingCore { at: p }
            } else {
                VersionError::InvalidChar { at: p }
            });
        }
    }
    let patch = parse_core(s, &mut p)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let pre_start = p + 1;
        let plus = s[pre_start..].find('+').map(|x| pre_start + x);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        p = pre_end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let build_start = p + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
        p = s.len();
    }
    if p < s.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn cmp_ids(a: &str, b: &str) -> Ordering {
    let an = !a.is_empty() && a.bytes().all(|x| x.is_ascii_digit());
    let bn = !b.is_empty() && b.bytes().all(|x| x.is_ascii_digit());
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => {
            a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
        }
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let c = cmp_ids(x, y);
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

fn ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut pos = 0;
    let mut saw_any = false;
    while pos < bytes.len() {
        while pos < bytes.len() && ws(bytes[pos]) { pos += 1; }
        if pos == bytes.len() { break; }
        let arm_start = pos;
        let mut toks = Vec::<(usize, &str)>::new();
        while pos < bytes.len() {
            while pos < bytes.len() && ws(bytes[pos]) { pos += 1; }
            if pos == bytes.len() { break; }
            if pos + 1 < bytes.len() && &s[pos..pos + 2] == "||" {
                break;
            }
            let st = pos;
            while pos < bytes.len() && !ws(bytes[pos]) { pos += 1; }
            toks.push((st, &s[st..pos]));
        }
        if toks.is_empty() {
            return Err(RangeError::Empty { at: arm_start });
        }
        saw_any = true;
        if let Some((idx, _)) = toks.iter().find(|(_, t)| *t == "-") {
            if toks.len() != 3 || toks[1].1 != "-" {
                return Err(RangeError::Syntax { at: *idx });
            }
        }
        let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
        if toks.len() == 3 && toks[1].1 == "-" {
            let (lo_at, lo_t) = toks[0];
            let (hi_at, hi_t) = toks[2];
            let lo = parse_partial_or_full(lo_t, lo_at)?;
            let hi = parse_partial_or_full(hi_t, hi_at)?;
            if lo.1 == 0 { return Err(RangeError::Syntax { at: lo_at }); }
            if hi.1 == 0 { return Err(RangeError::Syntax { at: hi_at }); }
            arm.comparators.push(Comparator { op: Op::Ge, version: lo.0.clone() });
            if !lo.0.pre.is_empty() { arm.anchors.push((lo.0.major, lo.0.minor, lo.0.patch)); }
            if hi.2 {
                arm.comparators.push(Comparator { op: Op::Le, version: hi.0.clone() });
                if !hi.0.pre.is_empty() { arm.anchors.push((hi.0.major, hi.0.minor, hi.0.patch)); }
            } else {
                let upper = partial_upper(&hi.0, hi.1, hi_at, false, false)?;
                arm.comparators.push(Comparator { op: Op::Lt, version: upper });
            }
        } else {
            for (at, tok) in toks {
                parse_token(&mut arm, tok, at)?;
            }
        }
        arms.push(arm);
        while pos < bytes.len() && ws(bytes[pos]) { pos += 1; }
        if pos < bytes.len() && pos + 1 < bytes.len() && &s[pos..pos + 2] == "||" {
            let sep = pos;
            pos += 2;
            let next = {
                let mut n = pos;
                while n < bytes.len() && ws(bytes[n]) { n += 1; }
                n
            };
            if next == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
            if next + 1 < bytes.len() && &s[next..next + 2] == "||" {
                return Err(RangeError::Empty { at: next });
            }
            let _ = sep;
        } else if pos < bytes.len() {
            return Err(RangeError::Syntax { at: pos });
        }
    }
    if !saw_any { return Err(RangeError::Empty { at: 0 }); }
    Ok(Range { arms })
}

fn parse_partial_or_full(tok: &str, at: usize) -> Result<(Version, usize, bool), RangeError> {
    let fields: Vec<&str> = tok.split('.').collect();
    let full = tok.contains('-') || tok.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X")));
    if full {
        let v = full_operand(tok, at)?;
        return Ok((v, 3, true));
    }
    let (v, k) = parse_partial(tok, at)?;
    Ok((v, k, false))
}

fn parse_partial(tok: &str, at: usize) -> Result<(Version, usize), RangeError> {
    let fields: Vec<&str> = tok.split('.').collect();
    if fields.len() > 3 || fields.is_empty() { return Err(RangeError::Syntax { at }); }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in &fields {
        if matches!(*f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else if f.is_empty() || wildcard_seen || !f.bytes().all(|b| b.is_ascii_digit())
            || (f.len() > 1 && f.starts_with('0')) || f.parse::<u64>().is_err()
        {
            return Err(RangeError::Syntax { at });
        } else {
            nums.push(f.parse::<u64>().unwrap());
        }
    }
    let k = nums.len();
    let mut vals = [0u64; 3];
    for (i, n) in nums.into_iter().enumerate() { vals[i] = n; }
    Ok((Version { major: vals[0], minor: vals[1], patch: vals[2], pre: vec![], build: vec![] }, k))
}

fn partial_upper(v: &Version, k: usize, at: usize, caret: bool, tilde: bool) -> Result<Version, RangeError> {
    let mut out = v.clone();
    out.pre.clear(); out.build.clear();
    let idx = if caret {
        if v.major > 0 || k == 1 { 0 } else if v.minor > 0 || k == 2 { 1 } else { 2 }
    } else if tilde {
        if k == 1 { 0 } else { 1 }
    } else if k == 1 { 0 } else { 1 };
    let val = match idx { 0 => out.major, 1 => out.minor, _ => out.patch };
    let Some(next) = val.checked_add(1) else { return Err(RangeError::Syntax { at }); };
    match idx {
        0 => { out.major = next; out.minor = 0; out.patch = 0; }
        1 => { out.minor = next; out.patch = 0; }
        _ => out.patch = next,
    }
    out.pre = vec!["0".into()];
    Ok(out)
}

fn push_comparison(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() { arm.anchors.push((v.major, v.minor, v.patch)); }
    arm.comparators.push(Comparator { op, version: v });
}

fn parse_token(arm: &mut Arm, tok: &str, at: usize) -> Result<(), RangeError> {
    if tok.is_empty() { return Err(RangeError::Syntax { at }); }
    let first = tok.as_bytes()[0];
    if matches!(first, b'<' | b'>' | b'=') {
        let (op, n) = if tok.starts_with("<=") { (Op::Le, 2) }
            else if tok.starts_with(">=") { (Op::Ge, 2) }
            else if tok.starts_with("<") { (Op::Lt, 1) }
            else if tok.starts_with(">") { (Op::Gt, 1) }
            else { (Op::Eq, 1) };
        let operand = &tok[n..];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at });
        }
        let v = full_operand(operand, at + n)?;
        push_comparison(arm, op, v, true);
        return Ok(());
    }
    if first == b'^' || first == b'~' {
        let operand = &tok[1..];
        if operand.is_empty() { return Err(RangeError::Syntax { at }); }
        let (v, k, full) = parse_partial_or_full(operand, at + 1)?;
        if k == 0 { return Err(RangeError::Syntax { at }); }
        push_comparison(arm, Op::Ge, v.clone(), true);
        let upper = partial_upper(&v, k, at, first == b'^', first == b'~')?;
        push_comparison(arm, Op::Lt, upper, false);
        let _ = full;
        return Ok(());
    }
    if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
        return Err(RangeError::Syntax { at });
    }
    let (v, k, full) = parse_partial_or_full(tok, at)?;
    if full {
        push_comparison(arm, Op::Eq, v, true);
    } else if k == 0 {
    } else if k == 1 || k == 2 {
        push_comparison(arm, Op::Ge, v.clone(), false);
        let upper = partial_upper(&v, k, at, false, false)?;
        push_comparison(arm, Op::Lt, upper, false);
    } else {
        push_comparison(arm, Op::Eq, v, false);
    }
    Ok(())
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.comparators.iter().all(|c| cmp_matches(v, c)) { return false; }
    include || v.pre.is_empty() || arm.anchors.contains(&(v.major, v.minor, v.patch))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v, r, false) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

fn satisfying(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (i, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index: i, error })?);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) { best = Some(i); }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> { satisfying(candidates, r, true) }
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> { satisfying(candidates, r, false) }

pub fn intersects(_a: &Range, _b: &Range, _include_prerelease: bool) -> bool { false }

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let mut lo: Option<(Version, bool)> = None;
        let mut hi: Option<(Version, bool)> = None;
        for c in &arm.comparators {
            match c.op {
                Op::Eq => { lo = Some((c.version.clone(), true)); hi = Some((c.version.clone(), true)); }
                Op::Ge | Op::Gt => {
                    let inc = c.op == Op::Ge;
                    if lo.as_ref().map_or(true, |(v, i)| {
                        let o = compare_versions(&c.version, v);
                        o == Ordering::Greater || (o == Ordering::Equal && !inc && *i)
                    }) { lo = Some((c.version.clone(), inc)); }
                }
                Op::Le | Op::Lt => {
                    let inc = c.op == Op::Le;
                    if hi.as_ref().map_or(true, |(v, i)| {
                        let o = compare_versions(&c.version, v);
                        o == Ordering::Less || (o == Ordering::Equal && !inc && *i)
                    }) { hi = Some((c.version.clone(), inc)); }
                }
            }
        }
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let astr = if anchors.is_empty() { "-".to_string() } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        let fmt = |x: &Option<(Version, bool)>, lower: bool| -> String {
            match x {
                None => "*".to_string(),
                Some((v, inc)) => {
                    let op = match (lower, inc) { (true,true)=>">=", (true,false)=>">", (false,true)=>"<=", (false,false)=>"<" };
                    format!("{op}{}", display_version(v))
                }
            }
        };
        out.push(format!("[{};{};pre={}]", fmt(&lo,true), fmt(&hi,false), astr));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label: String, pub expr: SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name: String, pub candidates: Vec<String>, pub constraints: Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package: String, pub version: String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(expr: &SetExpr, ranges: &mut Vec<Range>, path: &mut Vec<usize>, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(_) => ranges.last().map_or(false, |r| matches_with(v, r, include)),
        SetExpr::Union(xs) => xs.iter().enumerate().any(|(i,x)| { path.push(i); let y=eval_expr(x,ranges,path,v,include); path.pop(); y }),
        SetExpr::Intersection(xs) => xs.iter().enumerate().all(|(i,x)| { path.push(i); let y=eval_expr(x,ranges,path,v,include); path.pop(); y }),
    }
}

fn validate_expr(expr: &SetExpr, package: &str, label: &str, path: &mut Vec<usize>, out: &mut Vec<Range>) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange { package: package.into(), label: label.into(), path: path.clone(), error })?;
            out.push(r); Ok(())
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                validate_expr(x, package, label, path, out)?;
                path.pop();
            }
            Ok(())
        }
    }
}

pub fn select_packages(packages: &[Package], preference: Preference, include_prerelease: bool) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) { return Err(SelectError::DuplicatePackage { name: p.name.clone() }); }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) { return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() }); }
        }
    }
    let mut parsed: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i,s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate { package: p.name.clone(), index: i, error })?);
        }
        parsed.push(vs);
    }
    let mut all_ranges: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut prs = Vec::new();
        for c in &p.constraints {
            let mut rs = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut rs)?;
            prs.push(rs);
        }
        all_ranges.push(prs);
    }
    let mut result = Vec::new();
    for (pi,p) in packages.iter().enumerate() {
        let valid = |ci: usize| -> bool {
            p.constraints.iter().enumerate().all(|(ki,c)| {
                let mut cursor = 0usize;
                eval_expr_indexed(&c.expr, &all_ranges[pi][ki], &mut cursor, &parsed[pi][ci], include_prerelease)
            })
        };
        let qualifying: Vec<usize> = (0..p.candidates.len()).filter(|i| valid(*i)).collect();
        if qualifying.is_empty() {
            let labels = if p.candidates.is_empty() { Vec::new() } else { minimum_conflict(p, &parsed[pi], include_prerelease) };
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let ai = &parsed[pi][i]; let ab = &parsed[pi][best];
            let chosen = match preference {
                Preference::Highest => compare_versions(ai,ab) == Ordering::Greater,
                Preference::Lowest => compare_versions(ai,ab) == Ordering::Less,
                Preference::StableFirst => {
                    if ai.pre.is_empty() != ab.pre.is_empty() { ai.pre.is_empty() }
                    else { compare_versions(ai,ab) == Ordering::Greater }
                }
            };
            if chosen { best = i; }
        }
        result.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(result)
}

fn eval_expr_indexed(expr: &SetExpr, ranges: &[Range], cursor: &mut usize, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(_) => { let i=*cursor; *cursor+=1; matches_with(v, &ranges[i], include) }
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr_indexed(x,ranges,cursor,v,include)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr_indexed(x,ranges,cursor,v,include)),
    }
}

fn minimum_conflict(p: &Package, versions: &[Version], include: bool) -> Vec<String> {
    let n = p.constraints.len();
    if n >= usize::BITS as usize { return p.constraints.iter().map(|c| c.label.clone()).collect(); }
    for size in 1..=n {
        let mut best: Option<Vec<String>> = None;
        for mask in 1usize..(1usize << n) {
            if mask.count_ones() as usize != size { continue; }
            let ok = versions.iter().any(|v| {
                p.constraints.iter().enumerate().all(|(i,c)| {
                    if mask & (1<<i) == 0 { true } else {
                        let mut rs=Vec::new();
                        let _=validate_expr(&c.expr, "", "", &mut Vec::new(), &mut rs);
                        let mut cur=0;
                        eval_expr_indexed(&c.expr,&rs,&mut cur,v,include)
                    }
                })
            });
            if !ok {
                let mut labels: Vec<String> = p.constraints.iter().enumerate().filter(|(i,_)| mask&(1<<i)!=0).map(|(_,c)| c.label.clone()).collect();
                labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                if best.as_ref().map_or(true, |b| labels.as_slice() < b.as_slice()) { best=Some(labels); }
            }
        }
        if let Some(x)=best { return x; }
    }
    Vec::new()
}
