```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
    pub maintenance: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
    MaintenanceActive,
    MaintenanceRequired,
    AlreadyMode,
    ReconcileMismatch { coin: u16, expected: u16, found: u16 },
    ValueDrift { before: u64, after: u64 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
    Maintenance { enabled: bool },
    Reconciled { before: [u16; 4], after: [u16; 4] },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Audit {
    pub inserted: u64,
    pub refunded: u64,
    pub sales: u64,
    pub change: u64,
    pub adjustment: i64,
    pub cash: u64,
}

pub struct Machine {
    products: Vec<Product>,
    product_indices: HashMap<String, usize>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const COINS: [u16; 4] = [1, 3, 4, 10];

fn coin_index(coin: u16) -> Option<usize> {
    COINS.iter().position(|&c| c == coin)
}

fn valid_id(id: &str) -> bool {
    !id.is_empty() && id.len() <= 8 && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn value_of_counts(counts: &[u16; 4]) -> u64 {
    counts
        .iter()
        .zip(COINS.iter())
        .map(|(&count, &coin)| u64::from(count) * u64::from(coin))
        .sum()
}

fn value_of_coins(coins: &[u16]) -> u64 {
    coins.iter().map(|&coin| u64::from(coin)).sum()
}

fn add_sat_u64(a: u64, b: u64) -> u64 {
    a.saturating_add(b)
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    let mut remaining = amount;
    let mut counts = [0u16; 4];
    for i in (0..4).rev() {
        let coin = COINS[i];
        let take = tubes[i].min(remaining / coin);
        counts[i] = take;
        remaining -= take * coin;
    }
    if remaining != 0 {
        return None;
    }

    let mut out = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            out.push(COINS[i]);
        }
    }
    Some(out)
}

fn better_change(candidate: &[u16], best: &[u16]) -> bool {
    if candidate.len() != best.len() {
        return candidate.len() < best.len();
    }
    for (a, b) in candidate.iter().zip(best.iter()) {
        if a != b {
            return a > b;
        }
    }
    false
}

fn search_change(
    denomination_index: i32,
    remaining: u16,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<Vec<u16>>,
) {
    if denomination_index < 0 {
        if remaining == 0 {
            let mut candidate = Vec::new();
            for i in (0..4).rev() {
                for _ in 0..counts[i] {
                    candidate.push(COINS[i]);
                }
            }
            if best
                .as_ref()
                .map_or(true, |current| better_change(&candidate, current))
            {
                *best = Some(candidate);
            }
        }
        return;
    }

    let i = denomination_index as usize;
    let coin = COINS[i];
    let max_take = tubes[i].min(remaining / coin);
    for take in 0..=max_take {
        counts[i] = take;
        search_change(
            denomination_index - 1,
            remaining - take * coin,
            tubes,
            counts,
            best,
        );
    }
    counts[i] = 0;
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }
    if let Some(change) = greedy_change(amount, tubes) {
        return Some(change);
    }
    let mut best = None;
    search_change(3, amount, tubes, &mut [0; 4], &mut best);
    best
}

fn change_counts(coins: &[u16]) -> [u16; 4] {
    let mut counts = [0u16; 4];
    for &coin in coins {
        if let Some(i) = coin_index(coin) {
            counts[i] = counts[i].saturating_add(1);
        }
    }
    counts
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashMap::new();
        for (index, product) in products.iter().enumerate() {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains_key(&product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.insert(product.id.clone(), index);
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        Ok(Self {
            products,
            product_indices: seen,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let index = coin_index(coin).ok_or(Error::BadCoin { coin })?;
        let _ = index;
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        self.inserted = add_sat_u64(self.inserted, u64::from(coin));
        self.push_event(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !self.product_indices.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        self.push_event(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let product_index = *self
            .product_indices
            .get(&id)
            .expect("selected product must exist");
        let product = &self.products[product_index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = choose_change(amount, &self.tubes).ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;
        let removed = change_counts(&change);
        for i in 0..4 {
            proposed[i] -= removed[i];
        }
        for &coin in &self.escrow {
            let i = coin_index(coin).expect("escrow coins are valid");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        self.tubes = proposed;
        self.products[product_index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = add_sat_u64(self.sales, u64::from(price));
        let change_value = value_of_coins(&change);
        self.change = add_sat_u64(self.change, change_value);
        self.push_event(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let refunded = u64::from(self.credit);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.refunded = add_sat_u64(self.refunded, refunded);
        self.push_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.cmp(&b.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = *self
            .product_indices
            .get(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.push_event(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = *self
            .product_indices
            .get(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = stock as u16;
        self.push_event(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if self.maintenance == enabled {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }

        self.maintenance = enabled;
        self.selected = None;
        self.push_event(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let cash = value_of_counts(&self.tubes).saturating_add(u64::from(self.credit));
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash,
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: COINS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        let before_value = value_of_counts(&self.tubes);
        let after_value = value_of_counts(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift {
                before: before_value,
                after: after_value,
            });
        }

        let before = self.tubes;
        self.tubes = actual;
        let delta = i128::from(after_value) - i128::from(before_value);
        let adjustment = i128::from(self.adjustment) + delta;
        self.adjustment = adjustment.clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.push_event(EventKind::Reconciled {
            before,
            after: actual,
        });
        Ok(())
    }

    fn push_event(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}
```
