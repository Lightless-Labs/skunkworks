```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous_bound = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous_bound {
                    return Err(Error::Bound { index });
                }
                previous_bound = bound;
            }
            None => {
                if index != tiers.len() - 1 {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().is_some_and(|tier| tier.up_to.is_some()) {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn band_amount(index: usize, units: u64, unit: u64, start: u64) -> Result<u64, Error> {
    let product = units.checked_mul(unit).ok_or(Error::Overflow { index })?;
    product.checked_add(start).ok_or(Error::Overflow { index })
}

fn add_band(
    bands: &mut Vec<Band>,
    total: &mut u64,
    index: usize,
    units: u64,
    unit: u64,
    start: u64,
) -> Result<(), Error> {
    if units == 0 {
        return Ok(());
    }
    let amount = band_amount(index, units, unit, start)?;
    *total = total.checked_add(amount).ok_or(Error::Overflow { index })?;
    bands.push(Band {
        index,
        units,
        amount,
    });
    Ok(())
}

fn graduated_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    let mut remaining = quantity;
    let mut previous_bound = 0u64;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous_bound;
                previous_bound = bound;
                remaining.min(width)
            }
            None => remaining,
        };

        add_band(
            &mut bands,
            &mut total,
            index,
            units,
            tier.unit,
            tier.start,
        )?;
        remaining -= units;

        if remaining == 0 {
            break;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}

fn volume_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let (index, tier) = tiers
        .iter()
        .enumerate()
        .find(|(_, tier)| tier.up_to.is_none_or(|bound| quantity <= bound))
        .expect("validated schedules always have a final unbounded tier");

    let mut bands = Vec::new();
    let mut total = 0u64;
    add_band(
        &mut bands,
        &mut total,
        index,
        quantity,
        tier.unit,
        tier.start,
    )?;

    Ok(Charge {
        amount: total,
        bands,
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }
            let mut bands = Vec::new();
            let mut total = 0u64;
            add_band(&mut bands, &mut total, 0, quantity, *unit, *start)?;
            Ok(Charge {
                amount: total,
                bands,
            })
        }
        Price::Tiers(tiers) => graduated_charge(quantity, tiers),
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Flat { .. }, _) => quote(quantity, price)?,
        (Price::Tiers(tiers), Mode::Graduated) => graduated_charge(quantity, tiers)?,
        (Price::Tiers(tiers), Mode::Volume) => volume_charge(quantity, tiers)?,
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::Overflow { index: 0 })?;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```
