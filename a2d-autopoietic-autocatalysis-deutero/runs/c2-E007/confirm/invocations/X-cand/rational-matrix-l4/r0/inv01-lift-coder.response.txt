use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn signed_from_magnitude(mag: u128, negative: bool) -> Result<i64, RationalError> {
    if negative {
        if mag == (1u128 << 63) {
            Ok(i64::MIN)
        } else if mag <= i64::MAX as u128 {
            Ok(-(mag as i64))
        } else {
            Err(RationalError::Overflow)
        }
    } else if mag <= i64::MAX as u128 {
        Ok(mag as i64)
    } else {
        Err(RationalError::Overflow)
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let negative = (num < 0) ^ (den < 0);
        let nmag = num.unsigned_abs() as u128;
        let dmag = den.unsigned_abs() as u128;
        let g = gcd_u128(nmag, dmag);
        let nm = nmag / g;
        let dm = dmag / g;
        if dm > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: signed_from_magnitude(nm, negative)?,
            den: dm as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num.unsigned_abs() as u128;
        let b = rhs.num.unsigned_abs() as u128;
        let c = self.den as u128;
        let d = rhs.den as u128;
        let g1 = gcd_u128(a, d);
        let g2 = gcd_u128(b, c);
        let nmag = (a / g1) * (b / g2);
        let den = (c / g2) * (d / g1);
        let negative = (self.num < 0) ^ (rhs.num < 0);
        if den > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: signed_from_magnitude(nmag, negative)?,
            den: den as i64,
        })
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let a = self.num.unsigned_abs() as u128;
        let b = rhs.den as u128;
        let c = self.den as u128;
        let d = rhs.num.unsigned_abs() as u128;
        let g1 = gcd_u128(a, d);
        let g2 = gcd_u128(b, c);
        let nmag = (a / g1) * (b / g2);
        let den = (c / g2) * (d / g1);
        let negative = (self.num < 0) ^ (rhs.num < 0);
        if den > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: signed_from_magnitude(nmag, negative)?,
            den: den as i64,
        })
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num: n, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let mut slash = None;
        let mut denominator_start = None;
        let mut denominator_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            denominator_start = Some(i);
            let start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            denominator_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let num_digits = &bytes[numerator_start..numerator_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (denominator_start, denominator_end) {
            let digits = &bytes[ds..de];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_error(ds + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let numerator_magnitude = parse_magnitude(num_digits);
        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if numerator_magnitude > nlimit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let numerator = signed_from_magnitude(numerator_magnitude, negative)
            .map_err(|_| parse_error(0, ParseKind::OutOfRange))?;

        let mut denominator = 1u64;
        if let (Some(ds), Some(de)) = (denominator_start, denominator_end) {
            let digits = &bytes[ds..de];
            let value = parse_magnitude(digits);
            if value > i64::MAX as u128 {
                return Err(parse_error(ds, ParseKind::OutOfRange));
            }
            denominator = value as u64;
            if denominator == 0 {
                return Err(parse_error(ds, ParseKind::ZeroDenominator));
            }
        }

        if let Some(slash_at) = slash {
            if denominator == 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if numerator == 0 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator.unsigned_abs() as u128, denominator as u128) > 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
        }

        Ok(Self {
            num: numerator,
            den: denominator as i64,
        })
    }
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(bytes: &[u8]) -> u128 {
    let mut value = 0u128;
    for &b in bytes {
        value = value * 10 + (b - b'0') as u128;
    }
    value
}

fn from_i128(n: i128, d: i128) -> Result<Rational, RationalError> {
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let negative = (n < 0) ^ (d < 0);
    let nmag = n.unsigned_abs();
    let dmag = d.unsigned_abs();
    let g = gcd_u128(nmag, dmag);
    let nm = nmag / g;
    let dm = dmag / g;
    if dm > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: signed_from_magnitude(nm, negative)?,
        den: dm as i64,
    })
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = vec![Rational { num: 0, den: 1 }; self.data.len()];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[c * self.rows + r] = self.data[r * self.cols + c];
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let lhs = self.data[r * self.cols + k];
                    let rhsv = rhs.data[k * rhs.cols + c];
                    let term = lhs.mul(&rhsv).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        let one = Rational { num: 1, den: 1 };
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = one;
        }
        let aug = Matrix { rows: n, cols: width, data };
        let reduction = eliminate(aug, n, width)?;
        let rank = reduction.pivots.len();
        if rank < n {
            return Err(MatrixError::Singular { rank });
        }
        let mut inverse_data = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                inverse_data.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix { rows: n, cols: n, data: inverse_data })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..self.rows {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            for c in 0..k {
                data[r * width + n + c] = rhs.data[r * k + c];
            }
        }
        let reduction = eliminate(Matrix { rows: self.rows, cols: width, data }, n, width)
            .map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..n).all(|c| reduction.matrix.data[r * width + c].num == 0);
            if coeff_zero {
                if let Some(c) = (0..k).find(|&c| reduction.matrix.data[r * width + n + c].num != 0) {
                    let witness = reduction.matrix.data[r * width..(r + 1) * width].to_vec();
                    return Ok(Solution::None { row: r, rhs_col: c, witness });
                }
            }
        }

        let pivots = reduction.pivots;
        let rank = pivots.len();
        let mut free = Vec::new();
        let mut is_pivot = vec![false; n];
        for &p in &pivots {
            is_pivot[p] = true;
        }
        for c in 0..n {
            if !is_pivot[c] {
                free.push(c);
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] =
                    reduction.matrix.data[j * width + n + c];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular_data };

        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots });
        }

        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        let one = Rational { num: 1, den: 1 };
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + j] = one;
            for (i, &pivot_col) in pivots.iter().enumerate() {
                let coeff = reduction.matrix.data[i * width + free_col];
                let neg = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col, col: j, step: rank, source,
                    })
                })?;
                basis_data[pivot_col * f + j] = neg;
            }
        }
        let basis = Matrix { rows: n, cols: f, data: basis_data };
        Ok(Solution::Infinite { particular, free, basis, pivots })
    }
}

fn eliminate(
    mut matrix: Matrix,
    pivot_columns: usize,
    update_columns: usize,
) -> Result<Reduction, MatrixError> {
    let mut p = 0;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();

    for col in 0..pivot_columns {
        if p == matrix.rows {
            break;
        }
        let mut chosen: Option<usize> = None;
        for row in p..matrix.rows {
            let value = matrix.data[row * matrix.cols + col];
            if value.num == 0 {
                continue;
            }
            match chosen {
                None => chosen = Some(row),
                Some(current) => {
                    let current_value = matrix.data[current * matrix.cols + col];
                    let key = (value.den, value.num.unsigned_abs(), row);
                    let current_key = (current_value.den, current_value.num.unsigned_abs(), current);
                    if key < current_key {
                        chosen = Some(row);
                    }
                }
            }
        }
        let Some(q) = chosen else { continue };

        if q != p {
            for c in 0..update_columns {
                matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * matrix.cols + col];
        let mut normalized = Vec::with_capacity(update_columns);
        for c in 0..update_columns {
            let original = matrix.data[p * matrix.cols + c];
            let value = original.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                row: p, col: c, step: p, source,
            })?;
            normalized.push(value);
        }
        for c in 0..update_columns {
            matrix.data[p * matrix.cols + c] = normalized[c];
        }

        for row in 0..matrix.rows {
            if row == p {
                continue;
            }
            let factor = matrix.data[row * matrix.cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..update_columns {
                let product = factor.mul(&matrix.data[p * matrix.cols + c]).map_err(|source| {
                    MatrixError::Arithmetic { row, col: c, step: p, source }
                })?;
                let updated = matrix.data[row * matrix.cols + c].sub(&product).map_err(|source| {
                    MatrixError::Arithmetic { row, col: c, step: p, source }
                })?;
                matrix.data[row * matrix.cols + c] = updated;
            }
        }

        pivots.push(col);
        p += 1;
    }

    Ok(Reduction { matrix, pivots, swaps })
}
