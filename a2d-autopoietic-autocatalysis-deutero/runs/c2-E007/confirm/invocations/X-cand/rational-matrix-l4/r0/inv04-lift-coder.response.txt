use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_fraction(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_i128_fraction(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_i128_fraction(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_i128_fraction(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_i128_fraction(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn err(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }

        let b = text.as_bytes();
        if b.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if b[0] == b'-' {
            i = 1;
        }
        if i == b.len() {
            return Err(err(i, ParseKind::InvalidByte));
        }
        if !b[i].is_ascii_digit() {
            return Err(err(i, ParseKind::InvalidByte));
        }
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
        let num_end = i;
        let mut den_start = None;
        if i < b.len() {
            if b[i] != b'/' {
                return Err(err(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            if i == b.len() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            if !b[i].is_ascii_digit() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            while i < b.len() && b[i].is_ascii_digit() {
                i += 1;
            }
            if i < b.len() {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let neg = b[0] == b'-';
        let num_start = if neg { 1 } else { 0 };
        if num_end - num_start >= 2 && b[num_start] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if neg && &b[..num_end] == b"-0" {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some(ds) = den_start {
            if b.len() - ds >= 2 && b[ds] == b'0' {
                return Err(err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let parse_mag = |start: usize, end: usize, limit: u128| -> Result<u128, ()> {
            let mut v = 0u128;
            for &c in &b[start..end] {
                v = v
                    .checked_mul(10)
                    .and_then(|x| x.checked_add((c - b'0') as u128))
                    .ok_or(())?;
                if v > limit {
                    return Err(());
                }
            }
            Ok(v)
        };

        let nm = parse_mag(
            num_start,
            num_end,
            if neg {
                1u128 << 63
            } else {
                i64::MAX as u128
            },
        )
        .map_err(|_| err(0, ParseKind::OutOfRange))?;
        let n = if neg {
            if nm == (1u128 << 63) {
                i64::MIN
            } else {
                -(nm as i64)
            }
        } else {
            nm as i64
        };

        let (d, ds) = if let Some(ds) = den_start {
            let dm = parse_mag(ds, b.len(), i64::MAX as u128)
                .map_err(|_| err(ds, ParseKind::OutOfRange))?;
            (dm as i64, Some(ds))
        } else {
            (1, None)
        };

        if let Some(ds) = ds {
            if d == 0 {
                return Err(err(ds, ParseKind::ZeroDenominator));
            }
            let slash = ds - 1;
            if d == 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if n == 0 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(n.unsigned_abs() as u128, d as u128) > 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
        }

        Ok(Self { num: n, den: d })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        if self.data.is_empty() {
            return Self {
                rows: self.cols,
                cols: self.rows,
                data: Vec::new(),
            };
        }

        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        if len == 0 {
            return Ok(Self {
                rows: self.rows,
                cols: rhs.cols,
                data: Vec::new(),
            });
        }

        let mut out = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                out.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data: out,
        })
    }

    fn eliminate(
        &mut self,
        candidate_cols: usize,
    ) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;
        for col in 0..candidate_cols {
            if p == self.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for r in p..self.rows {
                let v = self.data[r * self.cols + col];
                if v.num == 0 {
                    continue;
                }
                let key = (v.den, v.num.unsigned_abs(), r);
                if chosen.map_or(true, |q| {
                    let w = self.data[q * self.cols + col];
                    key < (w.den, w.num.unsigned_abs(), q)
                }) {
                    chosen = Some(r);
                }
            }
            let Some(q) = chosen else {
                continue;
            };
            if q != p {
                for c in 0..self.cols {
                    self.data.swap(p * self.cols + c, q * self.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = self.data[p * self.cols + col];
            for c in 0..self.cols {
                let idx = p * self.cols + c;
                self.data[idx] = self.data[idx].div(&pivot).map_err(|source| {
                    MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    }
                })?;
            }

            for r in 0..self.rows {
                if r == p {
                    continue;
                }
                let factor = self.data[r * self.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..self.cols {
                    let prod = factor
                        .mul(&self.data[p * self.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        })?;
                    let idx = r * self.cols + c;
                    self.data[idx] = self.data[idx].sub(&prod).map_err(|source| {
                        MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = matrix.eliminate(self.cols)?;
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(Rational {
                    num: (r == c) as i64,
                    den: 1,
                });
            }
        }

        let mut aug = Self {
            rows: n,
            cols: width,
            data,
        };
        let (pivots, _) = aug.eliminate(n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }

        let mut out = Vec::with_capacity(n.checked_mul(n).ok_or(MatrixError::ShapeOverflow)?);
        for r in 0..n {
            for c in 0..n {
                out.push(aug.data[r * width + n + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: out,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let n = self.cols;
        let k = rhs.cols;
        let width = n
            .checked_add(k)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(len);
        if width != 0 {
            for r in 0..self.rows {
                for c in 0..n {
                    data.push(self.data[r * n + c]);
                }
                for c in 0..k {
                    data.push(rhs.data[r * k + c]);
                }
            }
        }

        let mut aug = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let (pivots, _) = aug.eliminate(n).map_err(SolveError::Matrix)?;

        if width != 0 {
            for r in 0..self.rows {
                let coeff_zero = (0..n).all(|c| aug.data[r * width + c].num == 0);
                if coeff_zero {
                    if let Some(c) =
                        (0..k).find(|&c| aug.data[r * width + n + c].num != 0)
                    {
                        return Ok(Solution::None {
                            row: r,
                            rhs_col: c,
                            witness: aug.data[r * width..(r + 1) * width].to_vec(),
                        });
                    }
                }
            }
        }

        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let rank = pivots.len();
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pc) in pivots.iter().enumerate() {
            for c in 0..k {
                particular[pc * k + c] = aug.data[j * width + n + c];
            }
        }
        let particular = Self {
            rows: n,
            cols: k,
            data: particular,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let free_count = n - rank;
        let basis_len = n
            .checked_mul(free_count)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let free: Vec<usize> = (0..n).filter(|c| !pivots.contains(c)).collect();
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_len];

        for (j, &fc) in free.iter().enumerate() {
            basis[fc * free_count + j] = Rational { num: 1, den: 1 };
            for (i, &pc) in pivots.iter().enumerate() {
                let coeff = aug.data[i * width + fc];
                let value = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pc,
                        col: j,
                        step: rank,
                        source,
                    })
                })?;
                basis[pc * free_count + j] = value;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self {
                rows: n,
                cols: free_count,
                data: basis,
            },
            pivots,
        })
    }
}
