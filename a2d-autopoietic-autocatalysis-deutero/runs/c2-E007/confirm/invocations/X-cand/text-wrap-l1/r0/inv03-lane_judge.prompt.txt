#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        let allowed = (32..=126).contains(&byte) || matches!(byte, b'\t' | b'\n' | b'\r');
        if !allowed {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r'))
        .filter(|word| !word.is_empty())
    {
        let mut suffix = word;

        loop {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let separator_len = usize::from(!line.is_empty());
            if suffix.len() <= width {
                if line.len() + separator_len + suffix.len() <= width {
                    if separator_len != 0 {
                        line.push(' ');
                    }
                    line.push_str(suffix);
                    break;
                }

                if !line.is_empty() {
                    lines.push(std::mem::take(&mut line));
                    continue;
                }
            } else if line.is_empty() {
                line.push_str(&suffix[..width]);
                lines.push(std::mem::take(&mut line));
                suffix = &suffix[width..];
                continue;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&suffix[..available]);
                    lines.push(std::mem::take(&mut line));
                    suffix = &suffix[available..];
                    continue;
                }

                lines.push(std::mem::take(&mut line));
                continue;
            }

            // A suffix that fits an empty line is appended whole.
            line.push_str(suffix);
            break;
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}

    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(32..=126).contains(&byte) && !matches!(byte, b'\t' | b'\n' | b'\r') {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r')).filter(|w| !w.is_empty()) {
        let bytes = word.as_bytes();
        let mut pos = 0;

        while pos < bytes.len() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = bytes.len() - pos;
            let separator_len = usize::from(!line.is_empty());

            if remaining <= width {
                if line.len() + separator_len + remaining <= width {
                    if separator_len != 0 {
                        line.push(' ');
                    }
                    line.push_str(&word[pos..]);
                    pos = bytes.len();
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                line.push_str(&word[pos..pos + width]);
                pos += width;
                lines.push(std::mem::take(&mut line));
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&word[pos..pos + available]);
                    pos += available;
                    lines.push(std::mem::take(&mut line));
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(matches!(byte, b' '..=b'~') || matches!(byte, b'\t' | b'\n' | b'\r')) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r')).filter(|s| !s.is_empty()) {
        let mut rest = word;

        while !rest.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if rest.len() <= width {
                let needed = rest.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(rest);
                    rest = "";
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                lines.push(rest[..width].to_string());
                rest = &rest[width..];
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&rest[..available]);
                    lines.push(std::mem::take(&mut line));
                    rest = &rest[available..];
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
