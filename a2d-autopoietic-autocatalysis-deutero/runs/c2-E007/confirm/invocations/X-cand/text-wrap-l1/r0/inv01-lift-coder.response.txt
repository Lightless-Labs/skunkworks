#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        let allowed = (32..=126).contains(&byte) || matches!(byte, b'\t' | b'\n' | b'\r');
        if !allowed {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r'))
        .filter(|word| !word.is_empty())
    {
        let mut suffix = word;

        loop {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let separator_len = usize::from(!line.is_empty());
            if suffix.len() <= width {
                if line.len() + separator_len + suffix.len() <= width {
                    if separator_len != 0 {
                        line.push(' ');
                    }
                    line.push_str(suffix);
                    break;
                }

                if !line.is_empty() {
                    lines.push(std::mem::take(&mut line));
                    continue;
                }
            } else if line.is_empty() {
                line.push_str(&suffix[..width]);
                lines.push(std::mem::take(&mut line));
                suffix = &suffix[width..];
                continue;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&suffix[..available]);
                    lines.push(std::mem::take(&mut line));
                    suffix = &suffix[available..];
                    continue;
                }

                lines.push(std::mem::take(&mut line));
                continue;
            }

            // A suffix that fits an empty line is appended whole.
            line.push_str(suffix);
            break;
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
