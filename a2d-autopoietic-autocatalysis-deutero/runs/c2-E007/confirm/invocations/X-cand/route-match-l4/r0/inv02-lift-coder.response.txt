#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, Debug, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone, Debug)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(s: &str) -> bool {
    let mut bytes = s.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_' | b'-'))
}

fn valid_capture_name(s: &str) -> bool {
    let mut bytes = s.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(hi) = hex_value(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(lo) = hex_value(bytes[i + 2]) else {
                return Err(base + i);
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn split_path<'a>(path: &'a str) -> Result<(Vec<(&'a str, usize)>, bool), usize> {
    if !path.starts_with('/') {
        return Err(0);
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };

    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            parts.push((&path[start..i], start));
            start = i + 1;
        }
        i += 1;
    }

    Ok((parts, trailing))
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>, bool), usize> {
    let (parts, trailing) = split_path(pattern)?;
    let mut segments = Vec::with_capacity(parts.len());
    let mut seen = std::collections::HashSet::new();
    let mut rank = Vec::new();
    let mut has_wildcard = false;

    for (index, (raw, at)) in parts.iter().enumerate() {
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(*at);
            }
            segments.push(Segment::Param(name.to_owned()));
            rank.push(1);
        } else if let Some(name) = raw.strip_prefix('*') {
            if index + 1 != parts.len() {
                return Err(*at);
            }
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(*at);
            }
            segments.push(Segment::Wildcard(name.to_owned()));
            rank.push(0);
            has_wildcard = true;
        } else {
            let decoded = decode_static(raw, *at)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if !has_wildcard {
        rank.push(3);
    }

    Ok((segments, trailing, rank, has_wildcard))
}

fn parse_request(path: &str) -> Result<(Vec<String>, bool), usize> {
    let head_end = path
        .bytes()
        .position(|b| b == b'?' || b == b'#')
        .unwrap_or(path.len());
    let head = &path[..head_end];
    let (parts, trailing) = split_path(head)?;
    let mut segments = Vec::with_capacity(parts.len());

    for (raw, at) in parts {
        segments.push(decode_static(raw, at)?);
    }

    Ok((segments, trailing))
}

fn same_shape(a: &Route, segments: &[Segment], trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != trailing {
        return false;
    }

    a.segments.iter().zip(segments).all(|(left, right)| match (left, right) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing, rank, has_wildcard) =
            parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;

        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        if let Some(route) = self.routes.iter().find(|route| {
            route.method == method
                && same_shape(route, &segments, trailing, self.policy)
        }) {
            return Err(Error::Conflict { with: route.id });
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];

        let (request_segments, request_trailing) =
            parse_request(path).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();

        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }

            let mut captures = Vec::new();
            let mut matched = true;
            let mut request_index = 0;

            for segment in &route.segments {
                match segment {
                    Segment::Static(text) => {
                        if request_segments.get(request_index) != Some(text) {
                            matched = false;
                            break;
                        }
                        request_index += 1;
                    }
                    Segment::Param(name) => {
                        let Some(value) = request_segments.get(request_index) else {
                            matched = false;
                            break;
                        };
                        captures.push((name.clone(), value.clone()));
                        request_index += 1;
                    }
                    Segment::Wildcard(name) => {
                        let value = request_segments[request_index..].join("/");
                        captures.push((name.clone(), value));
                        request_index = request_segments.len();
                    }
                }
            }

            if matched && request_index == request_segments.len() {
                candidates.push((route, captures));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut allowed = std::collections::BTreeSet::new();
        for (route, _) in &candidates {
            allowed.insert(route.method.clone());
        }
        if allowed.contains("GET") {
            allowed.insert("HEAD".to_owned());
        }

        let mut eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing != request_trailing
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else if let Some(stripped) = head.strip_suffix('/') {
                Some(stripped.to_owned())
            } else {
                None
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(route) = self.routes.iter().find(|route| route.name == name) else {
            return Err(Error::UnknownName);
        };

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut captures = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(key) | Segment::Wildcard(key) => captures.push(key.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &captures {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !captures.contains(key))
            .collect();
        extras.sort_unstable_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(key) => {
                    let value = supplied[key.as_str()];
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || (!value.is_empty()
                            && value
                                .split('/')
                                .any(|part| part.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut out = String::from("/");
        let mut emitted = false;

        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted {
                        out.push('/');
                    }
                    out.push_str(&encode_segment(value));
                    emitted = true;
                }
                Segment::Param(key) => {
                    if emitted {
                        out.push('/');
                    }
                    out.push_str(&encode_segment(supplied[key.as_str()]));
                    emitted = true;
                }
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    if !value.is_empty() {
                        if emitted {
                            out.push('/');
                        }
                        for (index, part) in value.split('/').enumerate() {
                            if index > 0 {
                                out.push('/');
                            }
                            out.push_str(&encode_segment(part));
                        }
                        emitted = true;
                    }
                }
            }
        }

        if route.trailing && emitted {
            out.push('/');
        }

        Ok(out)
    }
}
