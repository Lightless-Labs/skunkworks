#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_lowercase()
}

fn is_capture_name(bytes: &[u8]) -> bool {
    !bytes.is_empty()
        && is_name_start(bytes[0])
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn is_route_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..].iter().all(|c| {
            c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_' || *c == b'-'
        })
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn parse_path(
    input: &str,
    pattern: bool,
) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let bytes = input.as_bytes();

    if bytes.first() != Some(&b'/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut trailing = false;
    let mut end = bytes.len();
    if bytes.last() == Some(&b'/') {
        trailing = true;
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut start = 1usize;

    while start < end {
        let mut finish = start;
        while finish < end && bytes[finish] != b'/' {
            finish += 1;
        }
        if finish == start {
            return Err(invalid(start));
        }

        let raw = &bytes[start..finish];

        if pattern && (raw[0] == b':' || raw[0] == b'*') {
            let name_bytes = &raw[1..];
            if !is_capture_name(name_bytes) {
                return Err(invalid(start));
            }
            let name = std::str::from_utf8(name_bytes).unwrap().to_string();
            if !seen.insert(name.clone()) {
                return Err(invalid(start));
            }
            if raw[0] == b'*' {
                if finish != end {
                    return Err(invalid(start));
                }
                segments.push(Segment::Wildcard(name));
            } else {
                segments.push(Segment::Param(name));
            }
        } else {
            let mut decoded = Vec::new();
            let mut i = 0;
            while i < raw.len() {
                let b = raw[i];
                if b == b'%' {
                    if i + 2 >= raw.len() {
                        return Err(invalid(start + i));
                    }
                    let Some(hi) = hex_value(raw[i + 1]) else {
                        return Err(invalid(start + i));
                    };
                    let Some(lo) = hex_value(raw[i + 2]) else {
                        return Err(invalid(start + i));
                    };
                    let value = (hi << 4) | lo;
                    if !(0x20..=0x7e).contains(&value)
                        || value == b'/'
                        || is_unreserved(value)
                    {
                        return Err(invalid(start + i));
                    }
                    decoded.push(value);
                    i += 3;
                } else {
                    if !is_unreserved(b) {
                        return Err(invalid(start + i));
                    }
                    decoded.push(b);
                    i += 1;
                }
            }
            segments.push(Segment::Static(
                String::from_utf8(decoded).expect("decoded bytes are ASCII"),
            ));
        }

        if finish < end {
            start = finish + 1;
        } else {
            start = end;
        }
    }

    if trailing && segments.is_empty() {
        return Ok((segments, false));
    }
    Ok((segments, trailing))
}

fn shape_matches(a: &Route, b: &[Segment], trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != b.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != trailing {
        return false;
    }
    a.segments.iter().zip(b.iter()).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn encode_component(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !is_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let candidate = Route {
            id: self.routes.len(),
            name: name.to_string(),
            method: method.to_string(),
            rank: segments
                .iter()
                .map(|s| match s {
                    Segment::Static(_) => 2,
                    Segment::Param(_) => 1,
                    Segment::Wildcard(_) => 0,
                })
                .collect(),
            has_wildcard: segments
                .iter()
                .any(|s| matches!(s, Segment::Wildcard(_))),
            segments: segments.clone(),
            trailing,
        };
        let mut candidate = candidate;
        if !candidate.has_wildcard {
            candidate.rank.push(3);
        }

        if let Some(existing) = self.routes.iter().find(|r| {
            r.method == method && shape_matches(r, &segments, trailing, self.policy)
        }) {
            return Err(Error::Conflict { with: existing.id });
        }

        let id = candidate.id;
        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head, false)?;

        let mut candidates = Vec::<(&Route, Vec<(String, String)>)>::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }

            let mut params = Vec::new();
            let mut ri = 0usize;
            let mut matched = true;

            for segment in &route.segments {
                match segment {
                    Segment::Static(expected) => {
                        if request_segments.get(ri).map(|s| match s {
                            Segment::Static(v) => v == expected,
                            _ => false,
                        }) != Some(true)
                        {
                            matched = false;
                            break;
                        }
                        ri += 1;
                    }
                    Segment::Param(name) => {
                        match request_segments.get(ri) {
                            Some(Segment::Static(value)) => {
                                params.push((name.clone(), value.clone()));
                                ri += 1;
                            }
                            _ => {
                                matched = false;
                                break;
                            }
                        }
                    }
                    Segment::Wildcard(name) => {
                        let tail = request_segments[ri..]
                            .iter()
                            .map(|s| match s {
                                Segment::Static(v) => v.as_str(),
                                _ => "",
                            })
                            .collect::<Vec<_>>()
                            .join("/");
                        params.push((name.clone(), tail));
                        ri = request_segments.len();
                        break;
                    }
                }
            }

            if matched && ri == request_segments.len() {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_methods: Vec<&str> = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                vec!["HEAD"]
            } else {
                vec!["GET"]
            }
        } else {
            vec![method]
        };

        let mut eligible: Vec<(&Route, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(r, _)| eligible_methods.contains(&r.method.as_str()))
            .map(|(r, p)| (*r, p.clone()))
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::<String>::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (winner, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && winner.trailing != request_trailing
        {
            if winner.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(route) = self.routes.iter().find(|r| r.name == name) else {
            return Err(Error::UnknownName);
        };

        let mut supplied = std::collections::HashMap::<&str, &str>::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut required = Vec::<&str>::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(key) | Segment::Wildcard(key) => required.push(key),
                Segment::Static(_) => {}
            }
        }

        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let required_set: std::collections::HashSet<&str> = required.iter().copied().collect();
        if let Some(extra) = supplied
            .keys()
            .filter(|key| !required_set.contains(**key))
            .min()
        {
            return Err(Error::ExtraParam {
                name: (*extra).to_string(),
            });
        }

        for segment in &route.segments {
            let (key, wildcard) = match segment {
                Segment::Param(k) => (k, false),
                Segment::Wildcard(k) => (k, true),
                Segment::Static(_) => continue,
            };
            let value = supplied[key.as_str()];
            let valid_ascii = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
            let valid = if !valid_ascii {
                false
            } else if wildcard {
                value.is_empty()
                    || value
                        .split('/')
                        .all(|part| !part.is_empty())
            } else {
                !value.is_empty() && !value.contains('/')
            };
            if !valid {
                return Err(Error::InvalidValue { name: key.clone() });
            }
        }

        let mut output = String::from("/");
        let mut emitted = false;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_component(value));
                    emitted = true;
                }
                Segment::Param(key) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_component(supplied[key.as_str()]));
                    emitted = true;
                }
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    if !value.is_empty() {
                        if emitted {
                            output.push('/');
                        }
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part));
                        }
                        emitted = true;
                    }
                }
            }
        }

        if route.trailing && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}
