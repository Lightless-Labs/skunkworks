use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn push_component(
    components: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "" | "." => Ok(()),
        ".." => {
            if components.last().is_some_and(|last| last != "..") {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                components.push("..".to_owned());
            }
            Ok(())
        }
        _ => {
            components.push(component.to_owned());
            Ok(())
        }
    }
}

fn raw_components(path: &str, offset: usize) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let mut start = 0;
    for (i, part) in path.split('/').enumerate() {
        let _ = i;
        if !part.is_empty() {
            result.push((part.to_owned(), offset + start));
        }
        start += part.len() + 1;
    }
    result
}

fn join_components(components: &[String], absolute: bool, intent: bool) -> String {
    let joined = components.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut components = Vec::new();

    for (component, at) in raw_components(path, 0) {
        push_component(&mut components, &component, absolute, at)?;
    }

    Ok(join_components(&components, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect())
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let normalized = normalize(path).map_err(|_| PathError::InvalidHome)?;
            Ok(Some(
                normalized
                    .split('/')
                    .filter(|part| !part.is_empty())
                    .map(str::to_owned)
                    .collect(),
            ))
        }
    }
}

fn tilde_remainder<'a>(path: &'a str, home: Option<&[String]>) -> Result<(bool, &'a str, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, path, 0));
    }
    let bytes = path.as_bytes();
    if bytes.len() > 1 && bytes[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if let Some(home) = home {
        let _ = home;
    } else {
        return Err(PathError::MissingHome);
    }
    if bytes.len() == 1 {
        Ok((true, "", path.len()))
    } else {
        Ok((true, &path[2..], 2))
    }
}

fn resolve_basic(path: &str, cwd: &[String], home: Option<&[String]>) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, remainder, offset) = tilde_remainder(path, home)?;

    let absolute = !tilde && path.starts_with('/');
    let mut components = if absolute {
        Vec::new()
    } else if tilde {
        home.unwrap().to_vec()
    } else {
        cwd.to_vec()
    };

    if tilde && remainder.is_empty() {
        return Ok(join_components(&components, true, intent));
    }

    for (component, at) in raw_components(remainder, offset) {
        push_component(&mut components, &component, true, at)?;
    }

    Ok(join_components(&components, true, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd = validate_context(cwd)?;
    let home = validate_home(home)?;
    resolve_basic(path, &cwd, home.as_deref())
}

fn canonical_components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn components_path(components: &[String]) -> String {
    if components.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", components.join("/"))
    }
}

fn link_table(links: &[Link]) -> Result<(Vec<Vec<String>>, HashMap<String, usize>), PathError> {
    let mut keys = Vec::with_capacity(links.len());
    let mut map = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = map.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        map.insert(link.path.clone(), index);
        keys.push(canonical_components(&link.path));
    }

    Ok((keys, map))
}

struct Traversal<'a> {
    links: &'a [Link],
    key_map: &'a HashMap<String, usize>,
    components: Vec<String>,
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl<'a> Traversal<'a> {
    fn current_path(&self) -> String {
        components_path(&self.components)
    }

    fn process_raw(&mut self, raw: &str, base_offset: usize) -> Result<(), PathError> {
        let items = raw_components(raw, base_offset);
        self.process_items(&items)
    }

    fn process_items(&mut self, items: &[(String, usize)]) -> Result<(), PathError> {
        for (component, at) in items {
            match component.as_str() {
                "." => continue,
                ".." => {
                    if self.components.pop().is_none() {
                        return Err(PathError::Escape { at: *at });
                    }
                }
                _ => {
                    self.components.push(component.clone());
                    let path = self.current_path();
                    if let Some(&index) = self.key_map.get(&path) {
                        self.expand(index)?;
                    }
                }
            }
        }
        Ok(())
    }

    fn expand(&mut self, index: usize) -> Result<(), PathError> {
        if self.active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if self.expansions >= self.max_expansions {
            return Err(PathError::Limit { index });
        }

        self.expansions += 1;
        self.active.push(index);
        let link = &self.links[index];
        let target = link.target.clone();
        let target_absolute = target.starts_with('/');

        if target_absolute {
            self.components.clear();
        } else {
            self.components.pop();
        }

        let result = self.process_raw(&target, 0);
        self.active.pop();
        result
    }
}

fn resolve_links_internal(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
    links: &[Link],
    key_map: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, remainder, offset) = tilde_remainder(path, home);
    let (tilde, remainder, offset) = if path.starts_with('~') {
        (tilde?, remainder, offset)
    } else {
        (false, path, 0)
    };

    let absolute = !tilde && path.starts_with('/');
    let mut traversal = Traversal {
        links,
        key_map,
        components: if absolute {
            Vec::new()
        } else if tilde {
            home.unwrap().to_vec()
        } else {
            cwd.to_vec()
        },
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    let mut items = Vec::new();
    if tilde {
        if remainder.is_empty() {
            return Ok(join_components(
                &traversal.components,
                true,
                intent,
            ));
        }
        items.extend(raw_components(remainder, offset));
    } else {
        items.extend(raw_components(path, 0));
    }

    traversal.process_items(&items)?;
    Ok(join_components(&traversal.components, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd = validate_context(cwd)?;
    let home = validate_home(home)?;
    let (_, key_map) = link_table(links)?;
    resolve_links_internal(
        path,
        &cwd,
        home.as_deref(),
        links,
        &key_map,
        max_expansions,
    )
}

fn route(base: &[String], target: &[String]) -> String {
    let mut common = 0;
    while common < base.len()
        && common < target.len()
        && base[common] == target[common]
    {
        common += 1;
    }

    let mut parts = Vec::new();
    for _ in common..base.len() {
        parts.push("..".to_owned());
    }
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = canonical_components(&resolved_base);
    let target_components = canonical_components(&resolved_target);

    let (_, key_map) = link_table(links)?;
    let mut candidates = Vec::new();
    candidates.push(route(&base_components, &target_components));

    for link in links {
        if let Ok(probe) = resolve_links_internal(
            &link.path,
            &[],
            None,
            links,
            &key_map,
            max_expansions,
        ) {
            let probe_components = canonical_components(&probe);
            if probe_components.len() <= target_components.len()
                && probe_components
                    .iter()
                    .zip(target_components.iter())
                    .all(|(a, b)| a == b)
            {
                let mut candidate = route(&base_components, &canonical_components(&link.path));
                let suffix = &target_components[probe_components.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let target_directory = resolved_target.ends_with('/');
    let mut valid = Vec::new();

    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }

        if let Ok(result) = resolve_links_internal(
            &candidate,
            &base_components,
            home.as_deref(),
            links,
            &key_map,
            max_expansions,
        ) {
            if canonical_components(&result) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
