pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path.bytes().position(|b| b <= 31 || b == 127) {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.split('/').filter(|part| !part.is_empty()).last()
        .is_some_and(|part| part == "." || part == "..")
}

fn finish_absolute(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return "/".to_owned();
    }
    let mut result = format!("/{}", parts.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn finish_relative(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return ".".to_owned();
    }
    let mut result = parts.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn apply_component(
    parts: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    if component.is_empty() || component == "." {
        return Ok(());
    }
    if component == ".." {
        if parts.last().is_some_and(|last| last != "..") {
            parts.pop();
        } else if absolute {
            return Err(PathError::Escape { at });
        } else {
            parts.push("..".to_owned());
        }
    } else {
        parts.push(component.to_owned());
    }
    Ok(())
}

fn reduce(path: &str, absolute: bool, initial: Vec<String>, offset: usize) -> Result<Vec<String>, PathError> {
    let mut parts = initial;
    let mut byte = offset;
    for component in path.split('/') {
        if !component.is_empty() {
            apply_component(&mut parts, component, absolute, byte)?;
        }
        byte += component.len() + 1;
    }
    Ok(parts)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let parts = reduce(path, absolute, Vec::new(), 0)?;
    if absolute {
        Ok(finish_absolute(&parts, intent))
    } else {
        Ok(finish_relative(&parts, intent))
    }
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    match normalize(path) {
        Ok(value) => Ok(value.split('/').filter(|p| !p.is_empty()).map(str::to_owned).collect()),
        Err(_) => Err(PathError::InvalidCwd),
    }
}

fn normalized_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            match normalize(path) {
                Ok(value) => Ok(Some(value.split('/').filter(|p| !p.is_empty()).map(str::to_owned).collect())),
                Err(_) => Err(PathError::InvalidHome),
            }
        }
    }
}

fn tilde_remainder<'a>(path: &'a str, home: Option<&[String]>) -> Result<(&'a str, usize, Option<Vec<String>>), PathError> {
    if !path.starts_with('~') {
        return Ok((path, 0, None));
    }
    if path.len() > 1 && !path[1..].starts_with('/') {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let start = if path.len() > 1 { 2 } else { 1 };
    let home = home.ok_or(PathError::MissingHome)?;
    Ok((&path[start..], start, Some(home.to_vec())))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd)?;
    let home_parts = normalized_home(home)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (remainder, offset, tilde_parts) = tilde_remainder(path, home_parts.as_deref())?;
    let (absolute, initial) = if let Some(home) = tilde_parts {
        (true, home)
    } else if path.starts_with('/') {
        (true, Vec::new())
    } else {
        (true, cwd_parts)
    };
    let parts = reduce(remainder, absolute, initial, offset)?;
    Ok(finish_absolute(&parts, intent))
}

#[derive(Clone)]
struct Work {
    raw: String,
    offset: usize,
    parts: Vec<String>,
    absolute: bool,
    kind: WorkKind,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum WorkKind {
    Request,
    Target,
}

fn canonical_link_path(path: &str) -> bool {
    path.starts_with('/')
        && !path.ends_with('/')
        && normalize(path).is_ok_and(|normalized| normalized == path)
        && path != "/"
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        if !canonical_link_path(&link.path) || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(first) = links[..index].iter().position(|previous| previous.path == link.path) {
            return Err(PathError::DuplicateLink { first, second: index });
        }
    }
    Ok(())
}

fn expand_tilde<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<(&'a str, usize, Option<Vec<String>>), PathError> {
    if !path.starts_with('~') {
        return Ok((path, 0, None));
    }
    if path.len() > 1 && path.as_bytes()[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let start = if path.len() > 1 { 2 } else { 1 };
    let home = home.ok_or(PathError::MissingHome)?;
    Ok((&path[start..], start, Some(home.to_vec())))
}

fn stack_path(parts: &[String]) -> String {
    if parts.is_empty() { "/".to_owned() } else { format!("/{}", parts.join("/")) }
}

fn traverse_links(
    request: &str,
    request_offset: usize,
    initial: Vec<String>,
    links: &[Link],
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    let mut work = vec![Work {
        raw: request.to_owned(),
        offset: request_offset,
        parts: Vec::new(),
        absolute: false,
        kind: WorkKind::Request,
    }];
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(item) = work.last_mut() {
        if item.parts.is_empty() && item.raw.is_empty() {
            work.pop();
            if item.kind == WorkKind::Target {
                active.pop();
            }
            continue;
        }

        let mut found = None;
        for (i, component) in item.raw.split('/').enumerate() {
            let _ = i;
            if !component.is_empty() {
                found = Some(component.to_owned());
                break;
            }
        }
        let Some(component) = found else {
            let kind = item.kind;
            work.pop();
            if kind == WorkKind::Target {
                active.pop();
            }
            continue;
        };

        let component_start = item.offset + item.raw.find(&component).unwrap_or(0);
        let consumed = item.raw.find(&component).unwrap_or(0) + component.len();
        item.raw = item.raw[consumed..].to_owned();
        item.offset = component_start + component.len();

        if component == "." {
            continue;
        }
        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at: component_start });
            }
            continue;
        }

        stack.push(component);
        let current = stack_path(&stack);
        if let Some((index, link)) = links.iter().enumerate().find(|(_, l)| l.path == current) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);

            let target = link.target.clone();
            let target_absolute = target.starts_with('/');
            if target_absolute {
                stack.clear();
            } else {
                stack.pop();
            }
            let target_parts = target.split('/').filter(|p| !p.is_empty()).map(str::to_owned).collect();
            work.push(Work {
                raw: target,
                offset: 0,
                parts: target_parts,
                absolute: target_absolute,
                kind: WorkKind::Target,
            });
        }
    }
    Ok(stack)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd)?;
    let home_parts = normalized_home(home)?;
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (remainder, offset, tilde_parts) = expand_tilde(path, home_parts.as_deref())?;
    let (initial, remainder, offset) = if let Some(home) = tilde_parts {
        (home, remainder, offset)
    } else if path.starts_with('/') {
        (Vec::new(), remainder, offset)
    } else {
        (cwd_parts, remainder, offset)
    };
    let parts = traverse_links(remainder, offset, initial, links, max_expansions)?;
    Ok(finish_absolute(&parts, intent))
}

fn result_components(path: &str) -> Vec<String> {
    path.split('/').filter(|p| !p.is_empty()).map(str::to_owned).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base.iter().zip(target.iter()).take_while(|(a, b)| a == b).count();
    let mut components = vec!["..".to_owned(); base.len() - common];
    components.extend_from_slice(&target[common..]);
    if components.is_empty() { ".".to_owned() } else { components.join("/") }
}

fn target_has_directory_intent(path: &str) -> bool {
    path.ends_with('/')
}

fn protect_tilde(candidate: &str) -> String {
    if candidate.starts_with('~') {
        format!("./{}", candidate)
    } else {
        candidate.to_owned()
    }
}

fn candidate_for(link_path: &str, base: &[String], target: &[String], matched: &[String]) -> String {
    let key = result_components(link_path);
    let key_route = route(base, &key);
    let suffix = &target[matched.len()..];
    if suffix.is_empty() {
        key_route
    } else if key_route == "." {
        suffix.join("/")
    } else {
        format!("{}/{}", key_route, suffix.join("/"))
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = result_components(&resolved_base);
    let target_parts = result_components(&resolved_target);
    let target_dir = target_has_directory_intent(&resolved_target);

    let mut candidates = vec![route(&base_parts, &target_parts)];
    for link in links {
        if let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let probe_parts = result_components(&probe);
            if probe_parts.len() <= target_parts.len()
                && probe_parts.iter().zip(target_parts.iter()).all(|(a, b)| a == b)
            {
                candidates.push(candidate_for(&link.path, &base_parts, &target_parts, &probe_parts));
            }
        }
    }

    let mut valid = Vec::new();
    for candidate in candidates {
        let mut candidate = protect_tilde(&candidate);
        if target_dir && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }
        if let Ok(resolved) = resolve_links(&candidate, &resolved_base, home, links, max_expansions) {
            if result_components(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().unwrap_or_else(|| route(&base_parts, &target_parts)).pipe(Ok)
}

fn traversal(
    initial: Vec<String>,
    request: Vec<Component<'_>>,
    links: &[Link],
    link_map: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = Vec::<String>::new();
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;
    let mut work = Vec::<Work<'_>>::new();
    work.push(Work {
        items: initial
            .iter()
            .map(|s| Component { text: s, at: 0 })
            .collect(),
        pos: 0,
        link_index: None,
    });
    work.push(Work {
        items: request,
        pos: 0,
        link_index: None,
    });

    while let Some(frame) = work.last_mut() {
        if frame.pos >= frame.items.len() {
            let finished = work.pop().unwrap();
            if let Some(index) = finished.link_index {
                if active.last() == Some(&index) {
                    active.pop();
                }
            }
            continue;
        }

        let component = frame.items[frame.pos];
        frame.pos += 1;
        match component.text {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: component.at });
                }
            }
            name => {
                stack.push(name.to_string());
                let key = join_absolute(&stack, false);
                if let Some(&index) = link_map.get(&key) {
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    expansions += 1;
                    active.push(index);

                    let link = &links[index];
                    if link.target.starts_with('/') {
                        stack.clear();
                    } else {
                        stack.pop();
                    }
                    work.push(Work {
                        items: components(&link.target, 0),
                        pos: 0,
                        link_index: Some(index),
                    });
                }
            }
        }
    }
    Ok(stack)
}

        keys.push(canonical_components(&link.path));
    }

    Ok((keys, map))
}

struct Traversal<'a> {
    links: &'a [Link],
    key_map: &'a HashMap<String, usize>,
    components: Vec<String>,
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl<'a> Traversal<'a> {
    fn current_path(&self) -> String {
        components_path(&self.components)
    }

    fn process_raw(&mut self, raw: &str, base_offset: usize) -> Result<(), PathError> {
        let items = raw_components(raw, base_offset);
        self.process_items(&items)
    }

    fn process_items(&mut self, items: &[(String, usize)]) -> Result<(), PathError> {
        for (component, at) in items {
            match component.as_str() {
                "." => continue,
                ".." => {
                    if self.components.pop().is_none() {
                        return Err(PathError::Escape { at: *at });
                    }
                }
                _ => {
                    self.components.push(component.clone());
                    let path = self.current_path();
                    if let Some(&index) = self.key_map.get(&path) {
                        self.expand(index)?;
                    }
                }
            }
        }
        Ok(())
    }

    fn expand(&mut self, index: usize) -> Result<(), PathError> {
        if self.active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if self.expansions >= self.max_expansions {
            return Err(PathError::Limit { index });
        }

        self.expansions += 1;
        self.active.push(index);
        let link = &self.links[index];
        let target = link.target.clone();
        let target_absolute = target.starts_with('/');

        if target_absolute {
            self.components.clear();
        } else {
            self.components.pop();
        }

        let result = self.process_raw(&target, 0);
        self.active.pop();
        result
    }
}

fn resolve_links_internal(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
    links: &[Link],
    key_map: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, remainder, offset) = tilde_remainder(path, home);
    let (tilde, remainder, offset) = if path.starts_with('~') {
        (tilde?, remainder, offset)
    } else {
        (false, path, 0)
    };

    let absolute = !tilde && path.starts_with('/');
    let mut traversal = Traversal {
        links,
        key_map,
        components: if absolute {
            Vec::new()
        } else if tilde {
            home.unwrap().to_vec()
        } else {
            cwd.to_vec()
        },
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    let mut items = Vec::new();
    if tilde {
        if remainder.is_empty() {
            return Ok(join_components(
                &traversal.components,
                true,
                intent,
            ));
        }
        items.extend(raw_components(remainder, offset));
    } else {
        items.extend(raw_components(path, 0));
    }

    traversal.process_items(&items)?;
    Ok(join_components(&traversal.components, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd = validate_context(cwd)?;
    let home = validate_home(home)?;
    let (_, key_map) = link_table(links)?;
    resolve_links_internal(
        path,
        &cwd,
        home.as_deref(),
        links,
        &key_map,
        max_expansions,
    )
}

fn route(base: &[String], target: &[String]) -> String {
    let mut common = 0;
    while common < base.len()
        && common < target.len()
        && base[common] == target[common]
    {
        common += 1;
    }

    let mut parts = Vec::new();
    for _ in common..base.len() {
        parts.push("..".to_owned());
    }
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = canonical_components(&resolved_base);
    let target_components = canonical_components(&resolved_target);

    let (_, key_map) = link_table(links)?;
    let mut candidates = Vec::new();
    candidates.push(route(&base_components, &target_components));

    for link in links {
        if let Ok(probe) = resolve_links_internal(
            &link.path,
            &[],
            None,
            links,
            &key_map,
            max_expansions,
        ) {
            let probe_components = canonical_components(&probe);
            if probe_components.len() <= target_components.len()
                && probe_components
                    .iter()
                    .zip(target_components.iter())
                    .all(|(a, b)| a == b)
            {
                let mut candidate = route(&base_components, &canonical_components(&link.path));
                let suffix = &target_components[probe_components.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let target_directory = resolved_target.ends_with('/');
    let mut valid = Vec::new();

    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }

        if let Ok(result) = resolve_links_internal(
            &candidate,
            &base_components,
            home.as_deref(),
            links,
            &key_map,
            max_expansions,
        ) {
            if canonical_components(&result) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
