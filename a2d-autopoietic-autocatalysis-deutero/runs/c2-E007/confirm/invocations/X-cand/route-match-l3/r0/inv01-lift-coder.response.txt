#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static(bytes: &[u8], base: usize) -> Result<String, usize> {
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn valid_name(name: &[u8]) -> bool {
    if name.is_empty() || !name[0].is_ascii_lowercase() {
        return false;
    }
    name[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_segments(
    input: &str,
    is_pattern: bool,
) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut trailing = false;
    let mut end = bytes.len();
    if bytes.last() == Some(&b'/') {
        trailing = true;
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::<String>::new();
    let mut start = 1;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(start);
        }

        let raw = &bytes[start..stop];
        if is_pattern && (raw[0] == b':' || raw[0] == b'*') {
            let wildcard = raw[0] == b'*';
            let name = &raw[1..];
            if !valid_name(name) {
                return Err(start);
            }
            let name_string = String::from_utf8(name.to_vec()).expect("name is ASCII");
            if !names.insert(name_string.clone()) {
                return Err(start);
            }
            if wildcard {
                if stop != end {
                    return Err(start);
                }
                segments.push(Segment::Wildcard(name_string));
            } else {
                segments.push(Segment::Param(name_string));
            }
        } else {
            let decoded = decode_static(raw, start)?;
            segments.push(Segment::Static(decoded));
        }

        start = stop + 1;
    }

    if end > 1 && start == end + 1 {
        // Completed the final segment.
    } else if end > 1 && start <= end {
        return Err(start);
    }

    Ok((segments, trailing))
}

fn path_head(path: &str) -> &str {
    let q = path.find('?').unwrap_or(path.len());
    let h = path.find('#').unwrap_or(path.len());
    &path[..q.min(h)]
}

fn rank_for(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    let mut wildcard = false;
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                wildcard = true;
                break;
            }
        }
    }
    if !wildcard {
        rank.push(3);
    }
    rank
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing) =
            parse_segments(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        let id = self.routes.len();
        let rank = rank_for(&segments);
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head = path_head(path);
        let (request_segments, request_trailing) =
            parse_segments(head, false).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            let mut params = Vec::new();
            let mut ri = 0;
            let mut pi = 0;
            let mut matched = true;

            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Wildcard(name) => {
                        let value = request_segments[pi..]
                            .iter()
                            .map(|s| match s {
                                Segment::Static(v) => v.as_str(),
                                _ => unreachable!(),
                            })
                            .collect::<Vec<_>>()
                            .join("/");
                        params.push((name.clone(), value));
                        pi = request_segments.len();
                        ri += 1;
                        break;
                    }
                    Segment::Static(expected) => {
                        if pi >= request_segments.len()
                            || !matches!(&request_segments[pi], Segment::Static(actual) if actual == expected)
                        {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        if pi >= request_segments.len() {
                            matched = false;
                            break;
                        }
                        if let Segment::Static(value) = &request_segments[pi] {
                            params.push((name.clone(), value.clone()));
                        } else {
                            unreachable!();
                        }
                    }
                }
                ri += 1;
                pi += 1;
            }

            if matched && pi == request_segments.len() {
                if self.policy != SlashPolicy::Strict || route.trailing == request_trailing {
                    candidates.push((route, params));
                }
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(r, _)| r.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut best = eligible[0];
        for candidate in eligible.iter().skip(1) {
            if candidate.0.rank > best.0.rank {
                best = candidate;
            }
        }

        let redirect = if self.policy == SlashPolicy::Redirect
            && best.0.trailing != request_trailing
            && head != "/"
        {
            if best.0.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: best.0.id,
            params: best.1.clone(),
            redirect,
        })
    }
}
