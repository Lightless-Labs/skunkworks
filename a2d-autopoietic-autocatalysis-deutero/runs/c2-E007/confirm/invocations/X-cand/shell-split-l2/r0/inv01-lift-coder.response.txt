#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&'a str, &'a str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.push(next);
                        }
                        i = next_i + next.len_utf8();
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if c == '$' {
                    let (consumed, value) = parse_expansion(input, i, env)?;
                    if let Some((end, value)) = value {
                        active = true;
                        word.push_str(&value);
                        i = end;
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                    i = next_i + next.len_utf8();
                } else if c == '$' {
                    let (_, expansion) = parse_expansion(input, i, env)?;
                    if let Some((end, value)) = expansion {
                        active = true;
                        word.push_str(&value);
                        i = end;
                    } else {
                        active = true;
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    word.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, Option<(usize, String)>), SplitError> {
    let after_dollar = dollar + 1;
    let Some(first) = input[after_dollar..].chars().next() else {
        return Ok((after_dollar, None));
    };

    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut closing = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += c.len_utf8();
                if cursor < input.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                closing = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let Some(close_pos) = closing else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };

        let contents = &input[content_start..close_pos];
        let (name_end, name) = read_name(contents, 0);
        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let selected = if name_end == contents.len() {
            lookup(env, name).unwrap_or("")
        } else if contents[name_end..].starts_with(":-") {
            let default_text = &contents[name_end + 2..];
            match lookup(env, name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default_text),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((close_pos + 1, Some((close_pos + 1, selected))));
    }

    if is_name_start(first) {
        let (end, name) = read_name(input, after_dollar);
        let value = lookup(env, name).unwrap_or("").to_string();
        return Ok((end, Some((end, value))));
    }

    Ok((after_dollar, None))
}

fn read_name<'a>(text: &'a str, start: usize) -> (usize, &'a str) {
    let mut end = start;
    let mut chars = text[start..].char_indices();
    if let Some((_, c)) = chars.next() {
        if !is_name_start(c) {
            return (start, &text[start..start]);
        }
        end += c.len_utf8();
        for (offset, c) in chars {
            if !is_name_continue(c) {
                break;
            }
            end = start + offset + c.len_utf8();
        }
    }
    (end, &text[start..end])
}
