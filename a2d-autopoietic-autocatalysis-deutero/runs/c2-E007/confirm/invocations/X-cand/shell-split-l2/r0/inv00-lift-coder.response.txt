#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut it = text.char_indices().peekable();
    while let Some((_, c)) = it.next() {
        if c == '\\' {
            if let Some((_, next)) = it.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(open) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i += n;
                let _ = open;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next);
                        i += next.len_utf8();
                    } else if next == '\n' {
                        i += 1;
                    } else {
                        word.push('\\');
                    }
                } else if c == '$' {
                    active = true;
                    let (end, value) = parse_expansion(input, i, env)?;
                    if let Some(value) = value {
                        word.push_str(&value);
                    }
                    i = end;
                } else {
                    word.push(c);
                    i += n;
                }
                let _ = open;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i += n;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                    i += next.len_utf8();
                } else if c == '$' {
                    let (end, value) = parse_expansion(input, i, env)?;
                    active = true;
                    if let Some(value) = value {
                        word.push_str(&value);
                    }
                    i = end;
                } else {
                    active = true;
                    word.push(c);
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, Option<String>), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((after_dollar, None));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;
        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    let next = input[cursor..].chars().next().unwrap();
                    cursor += next.len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .find_map(|(idx, c)| if is_name_char(c) { None } else { Some(idx) })
            .unwrap_or(contents.len());

        if name_end == 0 || !is_name_start(contents.chars().next().unwrap()) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let rest = &contents[name_end..];
        let value = if rest.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            let existing = lookup(name, env).unwrap_or("");
            if existing.is_empty() {
                decode_default(default)
            } else {
                existing.to_string()
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((close + 1, Some(value)));
    }

    if !is_name_start(first) {
        return Ok((after_dollar, None));
    }

    let mut end = after_dollar + first.len_utf8();
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if is_name_char(c) {
            end += c.len_utf8();
        } else {
            break;
        }
    }
    let name = &input[after_dollar..end];
    Ok((end, Some(lookup(name, env).unwrap_or("").to_string())))
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn public_basic_words() {
        assert_eq!(
            split("alpha 'two words' x\"y\" ''", &[]),
            Ok(vec![
                "alpha".to_string(),
                "two words".to_string(),
                "xy".to_string(),
                "".to_string()
            ])
        );
    }

    #[test]
    fn public_escapes() {
        assert_eq!(
            split("a\\ b \"c\\\"d\"", &[]),
            Ok(vec!["a b".to_string(), "c\"d".to_string()])
        );
    }

    #[test]
    fn public_environment_and_default() {
        assert_eq!(
            split(
                "$USER ${M:-guest} \"$HOME\"",
                &[("USER", "sam"), ("HOME", "/a b")]
            ),
            Ok(vec![
                "sam".to_string(),
                "guest".to_string(),
                "/a b".to_string()
            ])
        );
    }

    #[test]
    fn public_expansion_error() {
        assert_eq!(
            split("${!}", &[]),
            Err(SplitError::InvalidExpansion { pos: 0 })
        );
    }
}
