```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    jobs: Vec<Entry>,
    history: Vec<(String, State)>,
}

struct Entry {
    job: Job,
    sequence: u64,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: Vec::new(),
            history: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.history.iter().any(|(id, _)| id == &job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.wrapping_add(1);
        self.jobs.push(Entry {
            job: job.clone(),
            sequence,
        });
        self.history.push((job.id, State::Queued));
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let Some((_, state)) = self.history.iter_mut().find(|(known_id, _)| known_id == id) else {
            return Err(Error::Unknown);
        };

        match state {
            State::Queued => {
                *state = State::Cancelled;
                self.jobs.retain(|entry| entry.job.id != id);
                self.last_time = now;
                Ok(())
            }
            State::Cancelled => Err(Error::Cancelled),
            State::Completed => Err(Error::Completed),
        }
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let mut selected: Option<(usize, u8, u64)> = None;

        for (index, entry) in self.jobs.iter().enumerate() {
            if entry.job.ready_at > now {
                continue;
            }

            let effective_priority = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                let age = (now - entry.job.ready_at) / self.aging_ms;
                let boosted = u64::from(entry.job.priority) + age;
                boosted.min(9) as u8
            };

            match selected {
                None => selected = Some((index, effective_priority, entry.sequence)),
                Some((_, best_priority, best_sequence))
                    if effective_priority > best_priority
                        || (effective_priority == best_priority
                            && entry.sequence < best_sequence) =>
                {
                    selected = Some((index, effective_priority, entry.sequence));
                }
                _ => {}
            }
        }

        self.last_time = now;

        let Some((index, effective_priority, _)) = selected else {
            return Ok(None);
        };

        let entry = self.jobs.remove(index);
        if let Some((_, state)) = self
            .history
            .iter_mut()
            .find(|(known_id, _)| known_id == &entry.job.id)
        {
            *state = State::Completed;
        }

        Ok(Some(Dispatch {
            id: entry.job.id,
            effective_priority,
        }))
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}
```
