#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

fn append_resource_snapshot(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
        return;
    }

    for (i, amount) in used.iter().enumerate() {
        if i != 0 {
            trace.push(',');
        }
        trace.push_str(&amount.to_string());
    }
}

fn write_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    trace.push_str(&time.to_string());
    trace.push('|');
    trace.push_str(kind);
    trace.push('|');
    trace.push_str(&task.to_string());
    trace.push('|');
    trace.push_str(&worker.to_string());
    trace.push('|');
    append_resource_snapshot(trace, used);
    trace.push('\n');
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in the specified task and entry order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&cooldowns.len()) {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build edges from predecessor to dependent task.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn order, always removing the smallest available ID.
    let mut ready = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            ready.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !ready.is_empty() {
        let smallest_pos = ready
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = ready.swap_remove(smallest_pos);
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.push(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for id in 0..n {
            if indegree[id] != 0 {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_preds: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut used = vec![0u64; capacity.len()];

    // A worker is available once its cooldown expires. Running tasks are
    // tracked separately and finish at their actual finish time.
    let mut worker_available = vec![0u64; worker_count];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // (finish, task, worker)
    let mut time = 0u64;
    let mut makespan = 0u64;

    let mut trace = String::new();
    trace.push_str("workers=");
    trace.push_str(&worker_count.to_string());
    trace.push_str(";resources=");
    trace.push_str(&capacity.len().to_string());
    trace.push('\n');

    loop {
        // Complete every task finishing now, in ascending task ID order.
        let mut finished_now: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for entry in running.drain(..) {
            if entry.0 == time {
                finished_now.push(entry);
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;
        finished_now.sort_by_key(|entry| entry.1);

        for &(finish, task_id, worker) in &finished_now {
            completed[task_id] = true;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] -= u64::from(need);
            }
            makespan = makespan.max(finish);
            write_event(&mut trace, time, "end", task_id, worker, &used);

            for &child in &children[task_id] {
                remaining_preds[child] -= 1;
            }
        }

        // Repeatedly admit one task and immediately re-evaluate eligibility.
        let mut started_zero_duration = false;
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= time
                        && remaining_preds[id] == 0
                        && tasks[id].needs.iter().enumerate().all(|(r, &need)| {
                            u64::from(need) <= u64::from(capacity[r]) - used[r]
                        })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&w| worker_available[w] <= time);

            let Some(worker) = worker else {
                break;
            };

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }

            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            write_event(&mut trace, time, "start", task_id, worker, &used);

            if tasks[task_id].duration == 0 {
                // Immediate completion occurs before selecting another task.
                completed[task_id] = true;
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                makespan = makespan.max(finish);
                write_event(&mut trace, time, "end", task_id, worker, &used);
                for &child in &children[task_id] {
                    remaining_preds[child] -= 1;
                }
                started_zero_duration = true;
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if started.iter().all(|&value| value) && completed.iter().all(|&value| value) {
            break;
        }

        // If no task can start now, advance to the next relevant event.
        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |old| old.min(candidate)));
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        if let Some(next) = next_time {
            time = next;
        } else if started_zero_duration {
            // A zero-duration completion may have unblocked work at this time;
            // the admission loop already reconsidered it, so no later event
            // exists only when the schedule is complete.
            if started.iter().all(|&value| value) && completed.iter().all(|&value| value) {
                break;
            }
            return Err(Error::Cycle {
                blocked: (0..n).filter(|&id| !completed[id]).collect(),
            });
        } else {
            return Err(Error::Cycle {
                blocked: (0..n).filter(|&id| !completed[id]).collect(),
            });
        }
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();

    trace.push_str("makespan=");
    trace.push_str(&makespan.to_string());
    trace.push('\n');

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}
