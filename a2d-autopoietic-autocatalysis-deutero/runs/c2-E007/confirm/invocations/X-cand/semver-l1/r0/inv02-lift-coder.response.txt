use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut core = [0u64; 3];

    for (field, slot) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            return if pos == bytes.len()
                || matches!(bytes[pos], b'.' | b'-' | b'+')
            {
                Err(VersionError::MissingCore { at: pos })
            } else {
                Err(VersionError::InvalidChar { at: pos })
            };
        }

        let digits = &bytes[start..pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in digits {
            value = match value
                .checked_mul(10)
                .and_then(|n| n.checked_add(u64::from(digit - b'0')))
            {
                Some(value) => value,
                None => return Err(VersionError::Overflow { at: start }),
            };
        }
        *slot = value;

        if field < 2 {
            if pos >= bytes.len() || bytes[pos] != b'.' {
                return if pos == bytes.len()
                    || matches!(bytes[pos], b'-' | b'+')
                {
                    Err(VersionError::MissingCore { at: pos })
                } else {
                    Err(VersionError::InvalidChar { at: pos })
                };
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pos + offset)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(bytes, pos, end, true)?;
        pos = end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(bytes, pos, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut identifiers = Vec::new();
    let mut ident_start = start;

    loop {
        let ident_end = bytes[ident_start..end]
            .iter()
            .position(|&b| b == b'.')
            .map(|offset| ident_start + offset)
            .unwrap_or(end);

        if ident_start == ident_end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &bytes[ident_start..ident_end];
        for (offset, &byte) in ident.iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || byte == b'-') {
                return Err(VersionError::InvalidChar {
                    at: ident_start + offset,
                });
            }
        }

        let numeric = ident.iter().all(u8::is_ascii_digit);
        if prerelease && numeric && ident.len() > 1 && ident[0] == b'0' {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        identifiers.push(String::from_utf8(ident.to_vec()).unwrap());

        if ident_end == end {
            break;
        }
        ident_start = ident_end + 1;
    }

    Ok(identifiers)
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_prereleases(&a.pre, &b.pre))
}

fn compare_prereleases(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.iter().zip(b) {
        let left_numeric = left.bytes().all(|byte| byte.is_ascii_digit());
        let right_numeric = right.bytes().all(|byte| byte.is_ascii_digit());

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => left
                .len()
                .cmp(&right.len())
                .then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CompareOp,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum CompareOp {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();
    let mut found_token = false;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        found_token = true;
        let token_start = pos;
        let mut op = CompareOp::Equal;

        match bytes[pos] {
            b'=' => {
                op = CompareOp::Equal;
                pos += 1;
            }
            b'<' => {
                if bytes.get(pos + 1) == Some(&b'=') {
                    op = CompareOp::LessEqual;
                    pos += 2;
                } else {
                    op = CompareOp::Less;
                    pos += 1;
                }
            }
            b'>' => {
                if bytes.get(pos + 1) == Some(&b'=') {
                    op = CompareOp::GreaterEqual;
                    pos += 2;
                } else {
                    op = CompareOp::Greater;
                    pos += 1;
                }
            }
            _ => {}
        }

        let operand_start = pos;
        if operand_start >= bytes.len() || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand_end = bytes[operand_start..]
            .iter()
            .position(|&byte| is_range_whitespace(byte))
            .map(|offset| operand_start + offset)
            .unwrap_or(bytes.len());

        let operand = &s[operand_start..operand_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        comparators.push(Comparator { op, version });
        pos = operand_end;
    }

    if !found_token {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let prerelease_admitted = !v.pre.is_empty()
            && arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && (v.major, v.minor, v.patch)
                        == (
                            comparator.version.major,
                            comparator.version.minor,
                            comparator.version.patch,
                        )
            });

        if !v.pre.is_empty() && !prerelease_admitted {
            return false;
        }

        arm.iter().all(|comparator| {
            let ordering = compare_versions(v, &comparator.version);
            match comparator.op {
                CompareOp::Equal => ordering == Ordering::Equal,
                CompareOp::Less => ordering == Ordering::Less,
                CompareOp::LessEqual => ordering != Ordering::Greater,
                CompareOp::Greater => ordering == Ordering::Greater,
                CompareOp::GreaterEqual => ordering != Ordering::Less,
            }
        })
    })
}

fn is_range_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\r' | b'\n')
}
