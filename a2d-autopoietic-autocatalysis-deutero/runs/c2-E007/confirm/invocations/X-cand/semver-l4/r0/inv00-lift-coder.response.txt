use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_sep_missing(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(if is_sep_missing(bytes.get(start).copied()) {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    stop_at_plus: bool,
    prerelease: bool,
) -> Result<(Vec<String>, usize), VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < bytes.len()
            && bytes[p] != b'.'
            && !(stop_at_plus && bytes[p] == b'+')
        {
            if !bytes[p].is_ascii_alphanumeric() && bytes[p] != b'-' {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if prerelease
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if p >= bytes.len() || (stop_at_plus && bytes[p] == b'+') {
            break;
        }
        p += 1;
        if p == bytes.len() || (stop_at_plus && bytes[p] == b'+') {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok((out, p))
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(if is_sep_missing(s.as_bytes().get(p).copied()) {
            VersionError::MissingCore { at: p }
        } else {
            VersionError::InvalidChar { at: p }
        });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(if is_sep_missing(s.as_bytes().get(p).copied()) {
            VersionError::MissingCore { at: p }
        } else {
            VersionError::InvalidChar { at: p }
        });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() {
        match s.as_bytes()[p] {
            b'-' => {
                let (ids, q) = parse_identifiers(s, p + 1, true, true)?;
                pre = ids;
                p = q;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: p }),
        }
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let (ids, q) = parse_identifiers(s, p + 1, false, false)?;
        build = ids;
        p = q;
    }
    if p != s.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_id(a: &str, b: &str) -> Ordering {
    let an = numeric_id(a);
    let bn = numeric_id(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let c = cmp_id(x, y);
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    v: Version,
    anchor: bool,
}
#[derive(Clone, Copy, Debug)]
enum Op { Lt, Le, Eq, Ge, Gt }

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn tuple(v: &Version) -> (u64, u64, u64) { (v.major, v.minor, v.patch) }

fn valid_cmp(c: &Cmp, v: &Version) -> bool {
    match c.op {
        Op::Lt => compare_versions(v, &c.v) == Ordering::Less,
        Op::Le => compare_versions(v, &c.v) != Ordering::Greater,
        Op::Eq => compare_versions(v, &c.v) == Ordering::Equal,
        Op::Ge => compare_versions(v, &c.v) != Ordering::Less,
        Op::Gt => compare_versions(v, &c.v) == Ordering::Greater,
    }
}

fn admitted(arm: &Arm, v: &Version) -> bool {
    v.pre.is_empty() || arm.anchors.contains(&tuple(v))
}

fn spaces(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn parse_full_operand(s: &str, start: usize, text: &str) -> Result<Version, RangeError> {
    parse_version(text).map_err(|error| RangeError::Version { at: start, error })
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: vec!["0".into()], build: Vec::new() }
}
fn release(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: Vec::new(), build: Vec::new() }
}

#[derive(Clone, Debug)]
struct Partial {
    v: Version,
    k: usize,
    full: bool,
}

fn partial(s: &str, start: usize, text: &str) -> Result<Partial, RangeError> {
    if text.contains('-') || text.contains('+') ||
        (text.split('.').count() >= 3 && !text.split('.').any(|x| matches!(x, "*" | "x" | "X"))) {
        return Ok(Partial { v: parse_full_operand(s, start, text)?, k: 3, full: true });
    }
    let fields: Vec<&str> = text.split('.').collect();
    if fields.is_empty() || fields.len() > 3 { return Err(RangeError::Syntax { at: start }); }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for f in &fields {
        if f.is_empty() { return Err(RangeError::Syntax { at: start }); }
        if matches!(*f, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || !f.bytes().all(|b| b.is_ascii_digit()) ||
                (f.len() > 1 && f.as_bytes()[0] == b'0') {
                return Err(RangeError::Syntax { at: start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            nums.push(n);
        }
    }
    let k = nums.len();
    let mut v = release(*nums.get(0).unwrap_or(&0), *nums.get(1).unwrap_or(&0), *nums.get(2).unwrap_or(&0));
    Ok(Partial { v, k, full: false })
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (mut a, mut b, mut c) = tuple(v);
    match field {
        0 => { a = a.checked_add(1).ok_or(RangeError::Syntax { at })?; b = 0; c = 0; }
        1 => { b = b.checked_add(1).ok_or(RangeError::Syntax { at })?; c = 0; }
        _ => { c = c.checked_add(1).ok_or(RangeError::Syntax { at })?; }
    }
    Ok(endpoint(a,b,c))
}

fn add(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() { arm.anchors.push(tuple(&v)); }
    arm.cmps.push(Cmp { op, v, anchor });
}

fn tokens(s: &str, base: usize) -> Vec<(usize, &str)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && spaces(b[i]) { i += 1; }
        if i == b.len() { break; }
        let st = i;
        while i < b.len() && !spaces(b[i]) { i += 1; }
        out.push((base + st, &s[st..i]));
    }
    out
}

fn parse_arm(s: &str, base: usize) -> Result<Arm, RangeError> {
    let ts = tokens(s, base);
    if ts.is_empty() { return Err(RangeError::Empty { at: base }); }
    let mut arm = Arm { cmps: Vec::new(), anchors: Vec::new() };
    if let Some((minus_i, _)) = ts.iter().find(|(_, t)| *t == "-") {
        if ts.len() != 3 || ts[1].1 != "-" {
            return Err(RangeError::Syntax { at: *minus_i });
        }
        let (li, lt) = ts[0];
        let (ui, ut) = ts[2];
        let l = partial(s, li - base, lt)?;
        let u = partial(s, ui - base, ut)?;
        if l.k == 0 { return Err(RangeError::Syntax { at: li }); }
        if u.k == 0 { return Err(RangeError::Syntax { at: ui }); }
        add(&mut arm, Op::Ge, l.v.clone(), !l.v.pre.is_empty());
        if u.full {
            add(&mut arm, Op::Le, u.v.clone(), !u.v.pre.is_empty());
        } else {
            let up = if u.k == 1 { bump(&u.v, 0, ui)? } else { bump(&u.v, 1, ui)? };
            add(&mut arm, Op::Lt, up, false);
        }
        return Ok(arm);
    }

    for (at, t) in ts {
        if t == "||" || t.contains(',') || t.contains('(') || t.contains(')') || t == "&" {
            return Err(RangeError::Syntax { at });
        }
        let (op, op_len) = if t.starts_with(">=") { (Some(Op::Ge), 2) }
            else if t.starts_with("<=") { (Some(Op::Le), 2) }
            else if t.starts_with(">") { (Some(Op::Gt), 1) }
            else if t.starts_with("<") { (Some(Op::Lt), 1) }
            else if t.starts_with("=") { (Some(Op::Eq), 1) }
            else { (None, 0) };
        if let Some(op) = op {
            let operand = &t[op_len..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = parse_full_operand(operand, 0, operand).map_err(|e| match e {
                RangeError::Version { at: _, error } => RangeError::Version { at: at + op_len, error },
                _ => e
            })?;
            add(&mut arm, op, v.clone(), !v.pre.is_empty());
            continue;
        }
        let (caret, tilde, offset) = if t.starts_with('^') { (true,false,1) }
            else if t.starts_with('~') { (false,true,1) } else { (false,false,0) };
        let operand = &t[offset..];
        if !operand.as_bytes().first().map_or(false, |b| b.is_ascii_digit() || matches!(b, b'*'|b'x'|b'X')) {
            return Err(RangeError::Syntax { at });
        }
        let p = partial(operand, 0, operand).map_err(|e| match e {
            RangeError::Version { error, .. } => RangeError::Version { at: at + offset, error },
            _ => RangeError::Syntax { at }
        })?;
        if caret || tilde {
            if p.k == 0 { return Err(RangeError::Syntax { at }); }
            add(&mut arm, Op::Ge, p.v.clone(), !p.v.pre.is_empty());
            let f = if caret {
                if p.v.major > 0 || p.k == 1 { 0 }
                else if p.v.minor > 0 || p.k == 2 { 1 }
                else { 2 }
            } else if p.k == 1 { 0 } else { 1 };
            add(&mut arm, Op::Lt, bump(&p.v, f, at)?, false);
        } else if p.full {
            add(&mut arm, Op::Eq, p.v.clone(), !p.v.pre.is_empty());
        } else if p.k == 1 || p.k == 2 {
            add(&mut arm, Op::Ge, p.v.clone(), false);
            add(&mut arm, Op::Lt, bump(&p.v, if p.k == 1 { 0 } else { 1 }, at)?, false);
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut sep = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'|' && b.get(i+1) == Some(&b'|') {
            let before = i == 0 || spaces(b[i-1]);
            let after = i + 2 == b.len() || spaces(b[i+2]);
            if before && after { sep.push(i); }
            i += 2;
        } else { i += 1; }
    }
    if s.trim_matches(|c| matches!(c, ' ' | '\t' | '\r' | '\n')).is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut start = 0;
    for pos in sep {
        let segment = &s[start..pos];
        if tokens(segment, start).is_empty() {
            return Err(RangeError::Empty { at: pos });
        }
        arms.push(parse_arm(segment, start)?);
        start = pos + 2;
    }
    let tail = &s[start..];
    if tokens(tail, start).is_empty() {
        if !arms.is_empty() { return Err(RangeError::Empty { at: s.len() }); }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(tail, start)?);
    Ok(Range { arms })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| {
        (include_prerelease || admitted(a, v)) && a.cmps.iter().all(|c| valid_cmp(c, v))
    })
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v, r, false) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

fn extreme(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index: i, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if matches(v, r) && best.map_or(true, |j| {
            let o = compare_versions(v, &parsed[j]);
            if greatest { o == Ordering::Greater } else { o == Ordering::Less }
        }) { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extreme(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extreme(candidates, r, false)
}

fn fmt_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}
pub fn canonical(r: &Range) -> String {
    let mut all = Vec::new();
    for a in &r.arms {
        let mut lower: Option<(Version,bool)> = None;
        let mut upper: Option<(Version,bool)> = None;
        for c in &a.cmps {
            match c.op {
                Op::Eq => {
                    lower = Some((c.v.clone(), true));
                    upper = Some((c.v.clone(), true));
                }
                Op::Ge | Op::Gt => {
                    let strict = matches!(c.op, Op::Gt);
                    if lower.as_ref().map_or(true, |(v, inc)| {
                        compare_versions(&c.v, v) == Ordering::Greater ||
                        (compare_versions(&c.v, v) == Ordering::Equal && !strict && *inc)
                    }) { lower = Some((c.v.clone(), !strict)); }
                }
                Op::Le | Op::Lt => {
                    let strict = matches!(c.op, Op::Lt);
                    if upper.as_ref().map_or(true, |(v, inc)| {
                        compare_versions(&c.v, v) == Ordering::Less ||
                        (compare_versions(&c.v, v) == Ordering::Equal && !strict && *inc)
                    }) { upper = Some((c.v.clone(), !strict)); }
                }
            }
        }
        let lo = lower.map_or("*".to_string(), |(v,i)| format!("{}{}", if i {">="} else {">"}, fmt_version(&v)));
        let hi = upper.map_or("*".to_string(), |(v,i)| format!("{}{}", if i {"<="} else {"<"}, fmt_version(&v)));
        let mut anchors = a.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anc = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|(x,y,z)| format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")
        };
        all.push(format!("[{lo};{hi};pre={anc}]"));
    }
    all.sort();
    all.dedup();
    if all.is_empty() { "EMPTY".into() } else { all.join(" || ") }
}
pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    // Probe all finite boundaries and nearby releases; intervals with unconstrained
    // bounds are handled by the same candidate set plus a stable fallback.
    let mut candidates = vec![release(0,0,0)];
    for arm in a.arms.iter().chain(b.arms.iter()) {
        for c in &arm.cmps {
            candidates.push(c.v.clone());
            if c.v.pre.is_empty() {
                if let Some(n) = c.v.patch.checked_add(1) { candidates.push(release(c.v.major,c.v.minor,n)); }
            } else {
                candidates.push(endpoint(c.v.major,c.v.minor,c.v.patch));
            }
        }
    }
    candidates.into_iter().any(|v| {
        matches_with(&v, a, include_prerelease) && matches_with(&v, b, include_prerelease)
    })
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label: String, pub expr: SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name: String, pub candidates: Vec<String>, pub constraints: Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package: String, pub version: String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn walk_ranges<F: FnMut(&[usize], &str)>(e: &SetExpr, path: &mut Vec<usize>, f: &mut F) {
    match e {
        SetExpr::Range(s) => f(path, s),
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i,x) in xs.iter().enumerate() {
                path.push(i); walk_ranges(x,path,f); path.pop();
            }
        }
    }
}
fn eval(e: &SetExpr, v: &Version, parsed: &[Range], ix: &mut usize, incl: bool) -> bool {
    match e {
        SetExpr::Range(_) => { let r=&parsed[*ix]; *ix+=1; matches_with(v,r,incl) }
        SetExpr::Union(xs) => xs.iter().any(|x| eval(x,v,parsed,ix,incl)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval(x,v,parsed,ix,incl)),
    }
}
fn subset_conflict(
    constraints: &[Constraint], parsed: &[Vec<Range>], candidates: &[Version], include: bool
) -> Vec<String> {
    let n = constraints.len();
    if candidates.is_empty() { return Vec::new(); }
    for size in 1..=n {
        let mut found: Vec<Vec<String>> = Vec::new();
        fn rec(
            start: usize, size: usize, chosen: &mut Vec<usize>, n: usize,
            constraints: &[Constraint], parsed: &[Vec<Range>], candidates: &[Version], include: bool,
            found: &mut Vec<Vec<String>>
        ) {
            if chosen.len() == size {
                let sat = candidates.iter().any(|v| chosen.iter().all(|&j| {
                    let mut ix=0; eval(&constraints[j].expr,v,&parsed[j],&mut ix,include)
                }));
                if !sat {
                    let mut labels=chosen.iter().map(|&j| constraints[j].label.clone()).collect::<Vec<_>>();
                    labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                    found.push(labels);
                }
                return;
            }
            for i in start..n {
                chosen.push(i); rec(i+1,size,chosen,n,constraints,parsed,candidates,include,found); chosen.pop();
            }
        }
        rec(0,size,&mut Vec::new(),n,constraints,parsed,candidates,include,&mut found);
        if !found.is_empty() {
            found.sort_by(|a,b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
            return found.remove(0);
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    for i in 0..packages.len() {
        if packages[..i].iter().any(|p| p.name == packages[i].name) {
            return Err(SelectError::DuplicatePackage { name: packages[i].name.clone() });
        }
    }
    for p in packages {
        for i in 0..p.constraints.len() {
            if p.constraints[..i].iter().any(|c| c.label == p.constraints[i].label) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: p.constraints[i].label.clone() });
            }
        }
    }
    let mut versions = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i,s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index: i, error
            })?);
        }
        versions.push(vs);
    }
    let mut ranges_by_pkg: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut prs = Vec::new();
        for c in &p.constraints {
            let mut rs = Vec::new();
            let mut path = Vec::new();
            let mut err = None;
            walk_ranges(&c.expr, &mut path, &mut |pth, s| {
                if err.is_none() {
                    match parse_range(s) {
                        Ok(r) => rs.push(r),
                        Err(error) => err = Some((pth.to_vec(), error)),
                    }
                }
            });
            if let Some((path,error)) = err {
                return Err(SelectError::InvalidRange { package:p.name.clone(), label:c.label.clone(), path, error });
            }
            prs.push(rs);
        }
        ranges_by_pkg.push(prs);
    }
    let mut result = Vec::new();
    for (pi,p) in packages.iter().enumerate() {
        let mut qualifying = Vec::new();
        for (i,v) in versions[pi].iter().enumerate() {
            let ok = p.constraints.iter().enumerate().all(|(ci,c)| {
                let mut ix=0; eval(&c.expr,v,&ranges_by_pkg[pi][ci],&mut ix,include_prerelease)
            });
            if ok { qualifying.push(i); }
        }
        if qualifying.is_empty() {
            let labels = subset_conflict(&p.constraints, &ranges_by_pkg[pi], &versions[pi], include_prerelease);
            return Err(SelectError::NoSolution { package:p.name.clone(), labels });
        }
        let stable_exists = qualifying.iter().any(|&i| versions[pi][i].pre.is_empty());
        let pool = qualifying.into_iter().filter(|&i| {
            preference != Preference::StableFirst || !stable_exists || versions[pi][i].pre.is_empty()
        });
        let mut best: Option<usize> = None;
        for i in pool {
            if best.map_or(true, |j| match preference {
                Preference::Lowest => compare_versions(&versions[pi][i], &versions[pi][j]) == Ordering::Less,
                _ => compare_versions(&versions[pi][i], &versions[pi][j]) == Ordering::Greater,
            }) { best = Some(i); }
        }
        result.push(Selection { package:p.name.clone(), version:p.candidates[best.unwrap()].clone() });
    }
    Ok(result)
}
