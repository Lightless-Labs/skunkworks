use std::cmp::Ordering;
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_identifier_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            if !valid_identifier_byte(bytes[p]) {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if is_pre
            && id.len() > 1
            && id.bytes().all(is_digit)
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_owned());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    for _ in 0..2 {
        if p >= s.len() || s.as_bytes()[p] != b'.' {
            let at = p;
            return Err(if at == s.len() || matches!(s.as_bytes()[at], b'-' | b'+') {
                VersionError::MissingCore { at }
            } else {
                VersionError::InvalidChar { at }
            });
        }
        p += 1;
        if _ == 0 {
            // The loop variable is only used to retain the left-to-right parse order.
        }
    }
    // Parse the remaining core fields explicitly so validation precedes separator checks.
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        let at = p;
        return Err(if at == s.len() || matches!(s.as_bytes()[at], b'-' | b'+') {
            VersionError::MissingCore { at }
        } else {
            VersionError::InvalidChar { at }
        });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        let at = p;
        return Err(if at == s.len() || matches!(s.as_bytes()[at], b'-' | b'+') {
            VersionError::MissingCore { at }
        } else {
            VersionError::InvalidChar { at }
        });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    if p < s.len() && !matches!(s.as_bytes()[p], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: p });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let pre_start = p + 1;
        let plus = s[pre_start..].find('+').map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        p = pre_end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        build = parse_identifiers(s, p + 1, s.len(), false)?;
        p = s.len();
    }
    if p != s.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_identifier(a: &str, b: &str) -> Ordering {
    match (numeric_id(a), numeric_id(b)) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let c = cmp_identifier(x, y);
                if c != Ordering::Equal {
                    return c;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
    explicit_pre: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn is_wild(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parse_version(s)
            .map(|v| Partial { nums: vec![v.major, v.minor, v.patch], full: Some(v) })
            .map_err(|error| RangeError::Version { at, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && fields.iter().all(|x| !is_wild(x)) {
        return parse_version(s)
            .map(|v| Partial { nums: vec![v.major, v.minor, v.patch], full: Some(v) })
            .map_err(|error| RangeError::Version { at, error });
    }
    if fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for field in fields {
        if is_wild(field) {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !field.bytes().all(is_digit)
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial { nums, full: None })
}

fn version_at(nums: &[u64], pre: Vec<String>) -> Version {
    Version {
        major: nums.first().copied().unwrap_or(0),
        minor: nums.get(1).copied().unwrap_or(0),
        patch: nums.get(2).copied().unwrap_or(0),
        pre,
        build: Vec::new(),
    }
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, explicit_pre: bool) {
    if explicit_pre {
        arm.anchors.push(core(&version));
    }
    arm.cmps.push(Cmp { op, version, explicit_pre });
}

fn bump(nums: &mut [u64; 3], field: usize, at: usize) -> Result<(), RangeError> {
    nums[field] = nums[field].checked_add(1).ok_or(RangeError::Syntax { at })?;
    for x in nums.iter_mut().skip(field + 1) {
        *x = 0;
    }
    Ok(())
}

fn partial_upper(p: &Partial, at: usize, mode: u8) -> Result<Option<Version>, RangeError> {
    let k = p.nums.len();
    if k == 0 {
        return Ok(None);
    }
    let mut n = [0, 0, 0];
    for (i, x) in p.nums.iter().enumerate() {
        n[i] = *x;
    }
    let field = match mode {
        0 => if k == 1 { 0 } else { 1 },
        1 => {
            if n[0] > 0 || k == 1 { 0 }
            else if n[1] > 0 || k == 2 { 1 }
            else { 2 }
        }
        2 => if k == 1 { 0 } else { 1 },
        _ => 0,
    };
    bump(&mut n, field, at)?;
    Ok(Some(Version { major: n[0], minor: n[1], patch: n[2], pre: vec!["0".into()], build: Vec::new() }))
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    if s.is_empty() || !s.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn apply_bare(arm: &mut Arm, token: &str, at: usize) -> Result<(), RangeError> {
    let p = parse_partial(token, at)?;
    if let Some(v) = p.full {
        let anchor = !v.pre.is_empty();
        add_cmp(arm, Op::Eq, v, anchor);
    } else if !p.nums.is_empty() {
        let lower = version_at(&p.nums, Vec::new());
        add_cmp(arm, Op::Ge, lower, false);
        if let Some(upper) = partial_upper(&p, at, 0)? {
            add_cmp(arm, Op::Lt, upper, false);
        }
    }
    Ok(())
}

fn apply_caret_tilde(arm: &mut Arm, token: &str, at: usize, caret: bool) -> Result<(), RangeError> {
    let operand = &token[1..];
    let operand_at = at + 1;
    let p = parse_partial(operand, operand_at)?;
    if p.nums.is_empty() {
        return Err(RangeError::Syntax { at });
    }
    let lower = p.full.clone().unwrap_or_else(|| version_at(&p.nums, Vec::new()));
    let anchor = !lower.pre.is_empty();
    add_cmp(arm, Op::Ge, lower, anchor);
    if let Some(upper) = partial_upper(&p, operand_at, if caret { 1 } else { 2 })? {
        add_cmp(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_hyphen_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let minus = tokens.iter().position(|(_, t)| *t == "-").unwrap();
    if tokens.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax { at: tokens[minus].0 });
    }
    let (lo_at, lo_s) = tokens[0];
    let (hi_at, hi_s) = tokens[2];
    let lo = parse_partial(lo_s, lo_at)?;
    let hi = parse_partial(hi_s, hi_at)?;
    if lo.nums.is_empty() {
        return Err(RangeError::Syntax { at: lo_at });
    }
    if hi.nums.is_empty() {
        return Err(RangeError::Syntax { at: hi_at });
    }
    let mut arm = Arm { cmps: Vec::new(), anchors: Vec::new() };
    let lower = lo.full.clone().unwrap_or_else(|| version_at(&lo.nums, Vec::new()));
    add_cmp(&mut arm, Op::Ge, lower.clone(), !lower.pre.is_empty());
    if let Some(v) = hi.full {
        add_cmp(&mut arm, Op::Le, v.clone(), !v.pre.is_empty());
    } else if let Some(upper) = partial_upper(&hi, hi_at, 0)? {
        add_cmp(&mut arm, Op::Lt, upper, false);
    } else {
        let upper = version_at(&hi.nums, Vec::new());
        add_cmp(&mut arm, Op::Le, upper, false);
    }
    Ok(arm)
}

fn parse_arm(s: &str, base: usize, tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    if let Some((_, _)) = tokens.iter().find(|(_, t)| *t == "-") {
        return parse_hyphen_arm(tokens);
    }
    let mut arm = Arm { cmps: Vec::new(), anchors: Vec::new() };
    for &(at, token) in tokens {
        if token.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        match token.as_bytes()[0] {
            b'^' => apply_caret_tilde(&mut arm, token, at, true)?,
            b'~' => apply_caret_tilde(&mut arm, token, at, false)?,
            b'<' | b'>' | b'=' => {
                let (op, n) = if token.starts_with("<=") { (Op::Le, 2) }
                    else if token.starts_with(">=") { (Op::Ge, 2) }
                    else if token.starts_with('<') { (Op::Lt, 1) }
                    else if token.starts_with('>') { (Op::Gt, 1) }
                    else { (Op::Eq, 1) };
                let operand_at = at + n;
                let operand = &token[n..];
                let v = parse_full_operand(operand, operand_at)?;
                add_cmp(&mut arm, op, v.clone(), !v.pre.is_empty());
            }
            b'0'..=b'9' | b'*' | b'x' | b'X' => apply_bare(&mut arm, token, at)?,
            _ => return Err(RangeError::Syntax { at }),
        }
    }
    let _ = (s, base);
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut p = 0;
    let mut arm_start = 0;
    let mut arm_tokens: Vec<(usize, &str)> = Vec::new();
    let mut saw_any = false;
    while p < bytes.len() {
        while p < bytes.len() && whitespace(bytes[p]) { p += 1; }
        if p == bytes.len() { break; }
        if p + 1 < bytes.len() && &bytes[p..p + 2] == b"||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at: p });
            }
            arms.push(parse_arm(s, arm_start, &arm_tokens)?);
            arm_tokens.clear();
            saw_any = true;
            p += 2;
            arm_start = p;
            continue;
        }
        let start = p;
        while p < bytes.len() && !whitespace(bytes[p]) { p += 1; }
        let token = &s[start..p];
        if token == "||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(parse_arm(s, arm_start, &arm_tokens)?);
            arm_tokens.clear();
            saw_any = true;
            arm_start = p;
        } else {
            arm_tokens.push((start, token));
        }
    }
    if !arm_tokens.is_empty() {
        arms.push(parse_arm(s, arm_start, &arm_tokens)?);
    } else if saw_any {
        return Err(RangeError::Empty { at: s.len() });
    } else {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

fn cmp_satisfies(c: &Cmp, v: &Version) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.cmps.iter().all(|c| cmp_satisfies(c, v)) {
        return false;
    }
    include || v.pre.is_empty() || arm.anchors.iter().any(|a| *a == core(v))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut best: Option<(usize, Version)> = None;
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    for (index, v) in parsed.into_iter().enumerate() {
        if !matches(&v, r) { continue; }
        let replace = match &best {
            None => true,
            Some((_, b)) => {
                let ord = compare_versions(&v, b);
                if greatest { ord == Ordering::Greater } else { ord == Ordering::Less }
            }
        };
        if replace { best = Some((index, v)); }
    }
    Ok(best.map(|(i, _)| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

fn cmp_bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.cmps {
        match c.op {
            Op::Eq => {
                lower = Some((c.version.clone(), true));
                upper = Some((c.version.clone(), true));
            }
            Op::Gt | Op::Ge => {
                let inclusive = matches!(c.op, Op::Ge);
                let replace = lower.as_ref().map_or(true, |(v, inc)| {
                    let o = compare_versions(&c.version, v);
                    o == Ordering::Greater || (o == Ordering::Equal && !inclusive && *inc)
                });
                if replace { lower = Some((c.version.clone(), inclusive)); }
            }
            Op::Lt | Op::Le => {
                let inclusive = matches!(c.op, Op::Le);
                let replace = upper.as_ref().map_or(true, |(v, inc)| {
                    let o = compare_versions(&c.version, v);
                    o == Ordering::Less || (o == Ordering::Equal && !inclusive && *inc)
                });
                if replace { upper = Some((c.version.clone(), inclusive)); }
            }
        }
    }
    (lower, upper)
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn arm_string(arm: &Arm) -> String {
    let (lo, hi) = cmp_bounds(arm);
    let lo = lo.map_or("*".to_owned(), |(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, display_version(&v)));
    let hi = hi.map_or("*".to_owned(), |(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, display_version(&v)));
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let a = if anchors.is_empty() { "-".to_owned() } else {
        anchors.iter().map(|(x,y,z)| format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")
    };
    format!("[{lo};{hi};pre={a}]")
}

pub fn canonical(r: &Range) -> String {
    let mut strings: Vec<String> = r.arms.iter().map(arm_string).collect();
    strings.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() { "EMPTY".to_owned() } else { strings.join(" || ") }
}

fn bump_core_tuple(c: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    let (mut a, mut b, mut d) = c;
    if let Some(x) = d.checked_add(1) { d = x; }
    else if let Some(x) = b.checked_add(1) { b = x; d = 0; }
    else if let Some(x) = a.checked_add(1) { a = x; b = 0; d = 0; }
    else { return None; }
    Some((a,b,d))
}

fn candidate_witnesses(arm_a: &Arm, arm_b: &Arm) -> Vec<Version> {
    let mut out = Vec::new();
    for arm in [arm_a, arm_b] {
        for c in &arm.cmps {
            out.push(c.version.clone());
            let v = &c.version;
            out.push(Version { major: v.major, minor: v.minor, patch: v.patch, pre: vec!["0".into()], build: Vec::new() });
            if let Some((a,b,d)) = bump_core_tuple(core(v)) {
                out.push(Version { major:a, minor:b, patch:d, pre:Vec::new(), build:Vec::new() });
                out.push(Version { major:a, minor:b, patch:d, pre:vec!["0".into()], build:Vec::new() });
            }
        }
    }
    out.push(Version { major:0, minor:0, patch:0, pre:Vec::new(), build:Vec::new() });
    for c in &arm_a.anchors {
        if arm_b.anchors.contains(c) {
            out.push(Version { major:c.0, minor:c.1, patch:c.2, pre:vec!["0".into()], build:Vec::new() });
        }
    }
    out
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            for v in candidate_witnesses(aa, bb) {
                if arm_matches(&v, aa, include_prerelease) && arm_matches(&v, bb, include_prerelease) {
                    return true;
                }
            }
        }
    }
    false
}

}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    parsed: &mut Vec<(Vec<usize>, Range)>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_owned(),
                label: label.to_owned(),
                path: path.clone(),
                error,
            })?;
            parsed.push((path.clone(), r));
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, parsed)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn eval_expr(
    expr: &SetExpr,
    v: &Version,
    include: bool,
    ranges: &[(Vec<usize>, Range)],
    path: &mut Vec<usize>,
) -> bool {
    match expr {
        SetExpr::Range(_) => ranges.iter().find(|(p, _)| p == path).map_or(false, |(_, r)| matches_with(v, r, include)),
        SetExpr::Union(children) => children.iter().enumerate().any(|(i, child)| {
            path.push(i);
            let ok = eval_expr(child, v, include, ranges, path);
            path.pop();
            ok
        }),
        SetExpr::Intersection(children) => children.iter().enumerate().all(|(i, child)| {
            path.push(i);
            let ok = eval_expr(child, v, include, ranges, path);
            path.pop();
            ok
        }),
    }
}

fn conflict_labels(
    package: &Package,
    candidates: &[Version],
    ranges: &[Vec<(Vec<usize>, Range)>],
    include: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if candidates.is_empty() || n == 0 { return Vec::new(); }
    for size in 1..=n {
        let mut best: Option<Vec<String>> = None;
        let mut idx = Vec::new();
        fn combinations(
            start: usize,
            left: usize,
            n: usize,
            idx: &mut Vec<usize>,
            best: &mut Option<Vec<String>>,
            package: &Package,
            candidates: &[Version],
            ranges: &[Vec<(Vec<usize>, Range)>],
            include: bool,
        ) {
            if left == 0 {
                let satisfiable = candidates.iter().any(|v| {
                    idx.iter().all(|&i| {
                        eval_expr(&package.constraints[i].expr, v, include, &ranges[i], &mut Vec::new())
                    })
                });
                if !satisfiable {
                    let mut labels: Vec<String> = idx.iter().map(|&i| package.constraints[i].label.clone()).collect();
                    labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                    if best.as_ref().map_or(true, |b| labels.as_slice() < b.as_slice()) {
                        *best = Some(labels);
                    }
                }
                return;
            }
            if n - start < left { return; }
            for i in start..=n-left {
                idx.push(i);
                combinations(i+1, left-1, n, idx, best, package, candidates, ranges, include);
                idx.pop();
            }
        }
        combinations(0, size, n, &mut idx, &mut best, package, candidates, ranges, include);
        if let Some(labels) = best { return labels; }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }

    for p in packages {
        let mut labels = HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }

    let mut all_versions = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            match parse_version(s) {
                Ok(v) => vs.push(v),
                Err(error) => return Err(SelectError::InvalidCandidate { package: p.name.clone(), index: i, error }),
            }
        }
        all_versions.push(vs);
    }

    let mut all_ranges = Vec::new();
    for p in packages {
        let mut package_ranges = Vec::new();
        for c in &p.constraints {
            let mut leaves = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut leaves)?;
            package_ranges.push(leaves);
        }
        all_ranges.push(package_ranges);
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &all_versions[pi];
        let ranges = &all_ranges[pi];
        let mut qualifying: Vec<usize> = Vec::new();
        for (i, v) in versions.iter().enumerate() {
            if p.constraints.iter().enumerate().all(|(ci, c)| {
                eval_expr(&c.expr, v, include_prerelease, &ranges[ci], &mut Vec::new())
            }) {
                qualifying.push(i);
            }
        }
        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflict_labels(p, versions, ranges, include_prerelease),
            });
        }
        let stable_exists = matches!(preference, Preference::StableFirst)
            && qualifying.iter().any(|&i| versions[i].pre.is_empty());
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let vi = &versions[i];
            let vb = &versions[best];
            if stable_exists && (vi.pre.is_empty() != vb.pre.is_empty()) {
                if vi.pre.is_empty() { best = i; }
                continue;
            }
            let order = compare_versions(vi, vb);
            let replace = match preference {
                Preference::Lowest => order == Ordering::Less,
                Preference::Highest | Preference::StableFirst => order == Ordering::Greater,
            };
            if replace { best = i; }
        }
        selections.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(selections)
}
