#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    burst_index: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

fn validate(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<(), Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }
    Ok(())
}

fn admit_through(
    events: &mut Vec<Event>,
    time: u64,
    queues: &mut [std::collections::VecDeque<usize>],
    jobs: &mut [Process],
) {
    events.sort_by_key(|event| (event.time, event.kind, event.index));
    let count = events.partition_point(|event| event.time <= time);
    let admitted: Vec<Event> = events.drain(..count).collect();
    for event in admitted {
        let level = if event.kind == 0 {
            jobs[event.index].level
        } else {
            event.level
        };
        queues[level].push_back(event.index);
    }
}

fn mean_x100(total: u64, count: u64) -> u64 {
    if count == 0 {
        return 0;
    }
    (total * 100 + count / 2) / count
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    validate(jobs, quanta, switch_cost)?;

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            burst_index: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut completed = 0usize;

    while completed < processes.len() {
        admit_through(&mut events, time, &mut queues, &mut processes);

        let selected = queues.iter_mut().find_map(|queue| queue.pop_front());
        let Some(index) = selected else {
            if events.is_empty() {
                break;
            }
            events.sort_by_key(|event| (event.time, event.kind, event.index));
            let next_time = events[0].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch = None;
            }
            continue;
        };

        let level = processes[index].level;
        let id = processes[index].id;
        if last_dispatch.is_some_and(|last| last != id) {
            time += u64::from(switch_cost);
        }
        admit_through(&mut events, time, &mut queues, &mut processes);

        let start = time;
        let consumed = processes[index]
            .remaining
            .min(quanta[level]);
        time += u64::from(consumed);
        processes[index].remaining -= consumed;
        let burst_done = processes[index].remaining == 0;

        let final_burst = burst_done && processes[index].burst_index + 1 == processes[index].bursts.len();
        runs.push(Run {
            id,
            start,
            end: time,
            finished: final_burst,
            level,
        });

        if final_burst {
            processes[index].completion = Some(time);
            completed += 1;
        } else if burst_done {
            processes[index].burst_index += 2;
            processes[index].remaining = processes[index].bursts[processes[index].burst_index];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: time + u64::from(processes[index].bursts[processes[index].burst_index - 1]),
                kind: 1,
                index,
                level: wake_level,
            });
        }

        admit_through(&mut events, time, &mut queues, &mut processes);

        if !final_burst {
            if burst_done {
                // The process is blocked until its I/O event.
            } else {
                processes[index].level = (level + 1).min(quanta.len() - 1);
                queues[processes[index].level].push_back(index);
            }
        }
        last_dispatch = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut waiting_total = 0u64;
    let mut turnaround_total = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = processes[index].completion.unwrap_or(time);
        let turnaround = completion - u64::from(job.arrival);
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let waiting = turnaround - cpu - io;
        waiting_total += waiting;
        turnaround_total += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: mean_x100(waiting_total, jobs.len() as u64),
        mean_turnaround_x100: mean_x100(turnaround_total, jobs.len() as u64),
        end: time,
    })
}
