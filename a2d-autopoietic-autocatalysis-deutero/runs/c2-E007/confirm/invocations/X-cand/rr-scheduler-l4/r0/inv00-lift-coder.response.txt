```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    #[derive(Clone, Copy)]
    struct Event {
        time: u64,
        kind: u8,
        index: usize,
        level: usize,
    }

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            level: job.level,
        })
        .collect();
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut event_cursor = 0usize;
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut burst_index = vec![0usize; jobs.len()];
    let mut remaining_cpu: Vec<u32> = jobs.iter().map(|job| job.bursts[0]).collect();
    let mut current_level: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completion = vec![0u64; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completed = 0usize;

    fn admit(
        through: u64,
        events: &[Event],
        cursor: &mut usize,
        queues: &mut [VecDeque<usize>],
    ) {
        while *cursor < events.len() && events[*cursor].time <= through {
            let event = events[*cursor];
            queues[event.level].push_back(event.index);
            *cursor += 1;
        }
    }

    while completed < jobs.len() {
        admit(time, &events, &mut event_cursor, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().expect("selected queue is nonempty");
        let id = jobs[index].id;
        if last_dispatch_id.is_some_and(|last_id| last_id != id) {
            time += switch_cost as u64;
        }
        admit(time, &events, &mut event_cursor, &mut queues);

        let start = time;
        let consumed = remaining_cpu[index].min(quanta[level]);
        time += consumed as u64;
        remaining_cpu[index] -= consumed;

        let burst_finished = remaining_cpu[index] == 0;
        let job_finished = burst_finished && burst_index[index] + 1 == jobs[index].bursts.len() - 1;
        runs.push(Run {
            id,
            start,
            end: time,
            finished: job_finished,
            level,
        });

        admit(time, &events, &mut event_cursor, &mut queues);

        if !burst_finished {
            current_level[index] = (level + 1).min(quanta.len() - 1);
            queues[current_level[index]].push_back(index);
        } else if job_finished {
            completion[index] = time;
            completed += 1;
        } else {
            let io_position = burst_index[index] + 1;
            let io_delay = jobs[index].bursts[io_position] as u64;
            burst_index[index] += 2;
            remaining_cpu[index] = jobs[index].bursts[burst_index[index]];
            current_level[index] = level.saturating_sub(1);
            events.push(Event {
                time: time + io_delay,
                kind: 1,
                index,
                level: current_level[index],
            });
            events[event_cursor..].sort_by_key(|event| (event.time, event.kind, event.index));
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let turnaround = completion[index] - job.arrival as u64;
        let total_work: u64 = job.bursts.iter().map(|&burst| burst as u64).sum();
        let waiting = turnaround - total_work;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion: completion[index],
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = if count == 0 {
        0
    } else {
        (total_waiting * 100 + count / 2) / count
    };
    let mean_turnaround_x100 = if count == 0 {
        0
    } else {
        (total_turnaround * 100 + count / 2) / count
    };

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}

use std::collections::VecDeque;
```
