#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/');

    let mut components: Vec<(String, usize)> = Vec::new();
    let mut last_nonempty: Option<&str> = None;
    let mut offset = 0;

    for component in path.split('/') {
        let start = offset;
        offset += component.len() + 1;

        if component.is_empty() {
            continue;
        }

        last_nonempty = Some(component);

        match component {
            "." => {}
            ".." => {
                if components
                    .last()
                    .is_some_and(|(name, _)| name != "..")
                {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push(("..".to_owned(), start));
                }
            }
            _ => components.push((component.to_owned(), start)),
        }
    }

    let directory_intent =
        directory_intent || matches!(last_nonempty, Some("." | ".."));

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    for (index, (component, _)) in components.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let bytes = path.as_bytes();
    let absolute = bytes[0] == b'/';
    let directory_intent = bytes.last() == Some(&b'/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..");

    let mut components: Vec<(usize, &str)> = Vec::new();
    let mut start = 0;

    while start < bytes.len() {
        while start < bytes.len() && bytes[start] == b'/' {
            start += 1;
        }
        if start == bytes.len() {
            break;
        }

        let component_start = start;
        while start < bytes.len() && bytes[start] != b'/' {
            start += 1;
        }
        let component = &path[component_start..start];

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|(_, previous)| *previous != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component_start });
                } else {
                    components.push((component_start, component));
                }
            }
            _ => components.push((component_start, component)),
        }
    }

    let mut result = String::new();
    if absolute {
        result.push('/');
    }
    result.push_str(
        &components
            .iter()
            .map(|(_, component)| *component)
            .collect::<Vec<_>>()
            .join("/"),
    );

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.bytes().position(|byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .bytes()
        .position(|byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let trailing_slash = path.ends_with('/');

    let mut components: Vec<(String, usize)> = Vec::new();
    let mut last_nonempty: Option<(&str, usize)> = None;
    let mut offset = 0;

    for component in path.split('/') {
        if !component.is_empty() {
            last_nonempty = Some((component, offset));

            match component {
                "." => {}
                ".." => {
                    if components
                        .last()
                        .is_some_and(|(name, _)| name != "..")
                    {
                        components.pop();
                    } else if absolute {
                        return Err(PathError::Escape { at: offset });
                    } else {
                        components.push(("..".to_string(), offset));
                    }
                }
                _ => components.push((component.to_string(), offset)),
            }
        }
        offset += component.len() + 1;
    }

    let directory_intent = trailing_slash
        || last_nonempty.is_some_and(|(component, _)| component == "." || component == "..");

    let joined = components
        .iter()
        .map(|(component, _)| component.as_str())
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
