#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (index, &byte) in local_bytes.iter().enumerate() {
        if byte == b'.' {
            if index == 0
                || index + 1 == local_bytes.len()
                || local_bytes[index - 1] == b'.'
            {
                return Err(EmailError::LocalDot { at: index });
            }
        } else if !byte.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
        {
            return Err(EmailError::LocalChar { at: index });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_bytes = if has_root_dot {
        &raw_domain[..raw_domain.len() - 1]
    } else {
        raw_domain
    };
    let domain_len = domain_bytes.len();

    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0;

    for index in 0..=domain_len {
        if index == domain_len || domain_bytes[index] == b'.' {
            if label_start == index {
                let at = if index == domain_len && has_root_dot {
                    input.len() - 1
                } else {
                    domain_start + label_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label = &domain_bytes[label_start..index];
            let absolute_start = domain_start + label_start;

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }
            if let Some(offset) = label
                .iter()
                .position(|&byte| !byte.is_ascii_alphanumeric() && byte != b'-')
            {
                return Err(EmailError::DomainChar {
                    at: absolute_start + offset,
                });
            }

            labels.push((absolute_start, label));
            label_start = index + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let local = String::from_utf8(local_bytes.to_vec()).expect("ASCII input");
    let domain = String::from_utf8(domain_bytes.to_vec())
        .expect("ASCII input")
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn public_atom_address() {
        assert_eq!(
            parse_email("Ada.Lovelace@Mail.Example"),
            Ok(Address {
                local: "Ada.Lovelace".into(),
                domain: "mail.example".into(),
                quoted: false
            })
        );
    }

    #[test]
    fn public_domain_label() {
        assert_eq!(
            parse_email("info@post-box.example"),
            Ok(Address {
                local: "info".into(),
                domain: "post-box.example".into(),
                quoted: false
            })
        );
    }

    #[test]
    fn public_separator_error() {
        assert_eq!(
            parse_email("missing.example"),
            Err(EmailError::MissingAt)
        );
    }
}
