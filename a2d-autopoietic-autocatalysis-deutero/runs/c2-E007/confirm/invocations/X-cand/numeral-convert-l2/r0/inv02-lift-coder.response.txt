#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let payload_start = if let Some(hash_index) = bytes.iter().position(|&b| b == b'#') {
        let expected = base.to_string();
        if &bytes[..hash_index] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash_index + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        if !matches!(digit_value(byte), Some(value) if value < base) {
            return Err(BaseError::InvalidDigit {
                operand,
                index: payload_start + offset,
            });
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let digits: Vec<u8> = payload
        .iter()
        .filter_map(|&byte| if byte == b'_' { None } else { digit_value(byte) })
        .collect();

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for digit in value.iter_mut() {
        let n = *digit as u16 * multiplier as u16 + carry;
        *digit = (n % 10) as u8;
        carry = n / 10;
    }
    while carry > 0 {
        value.push((carry % 10) as u8);
        carry /= 10;
    }
}

fn add_values(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    let length = left.len().max(right.len());

    for index in 0..length {
        let sum = left.get(index).copied().unwrap_or(0) as u16
            + right.get(index).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum % 10) as u8);
        carry = sum / 10;
    }
    if carry > 0 {
        result.push(carry as u8);
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn convert_digits_to_decimal(digits: &[u8], base: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_add(&mut value, base, digit);
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    value
}

fn decimal_to_base(mut value: Vec<u8>, base: u8) -> Vec<u8> {
    if value.iter().all(|&digit| digit == 0) {
        return vec![0];
    }

    let mut output = Vec::new();
    while value.iter().any(|&digit| digit != 0) {
        let mut remainder = 0u16;
        for digit in value.iter_mut().rev() {
            let current = remainder * 10 + *digit as u16;
            *digit = (current / base as u16) as u8;
            remainder = current % base as u16;
        }
        output.push(remainder as u8);
        while value.len() > 1 && value.last() == Some(&0) {
            value.pop();
        }
    }
    output
}

fn format_result(mut digits: Vec<u8>, style: BaseStyle) -> String {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    while digits.len() < style.min_width {
        digits.push(0);
    }

    let mut chars = Vec::with_capacity(digits.len());
    for &digit in digits.iter().rev() {
        let byte = match digit {
            0..=9 => b'0' + digit,
            10..=35 if style.upper => b'A' + digit - 10,
            10..=35 => b'a' + digit - 10,
            _ => unreachable!(),
        };
        chars.push(byte as char);
    }

    if style.group == 0 {
        return chars.into_iter().collect();
    }

    let mut output = String::with_capacity(chars.len() + chars.len() / style.group);
    for (index, ch) in chars.into_iter().enumerate() {
        if index > 0 && (chars_len_placeholder() - index) % style.group == 0 {
            output.push('_');
        }
        output.push(ch);
    }
    output
}

fn chars_len_placeholder() -> usize {
    0
}

fn format_result(mut digits: Vec<u8>, style: BaseStyle) -> String {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    while digits.len() < style.min_width {
        digits.push(0);
    }

    let chars: Vec<char> = digits
        .iter()
        .rev()
        .map(|&digit| match digit {
            0..=9 => (b'0' + digit) as char,
            10..=35 if style.upper => (b'A' + digit - 10) as char,
            10..=35 => (b'a' + digit - 10) as char,
            _ => unreachable!(),
        })
        .collect();

    let mut output = String::with_capacity(chars.len() + chars.len() / style.group.max(1));
    for (index, ch) in chars.iter().enumerate() {
        if style.group != 0 && index > 0 && (chars.len() - index) % style.group == 0 {
            output.push('_');
        }
        output.push(*ch);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let decimal = convert_digits_to_decimal(&digits, from);
    let converted = decimal_to_base(decimal, to);
    Ok(format_result(converted, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_decimal = convert_digits_to_decimal(&left_digits, base);
    let right_decimal = convert_digits_to_decimal(&right_digits, base);
    let sum = add_values(&left_decimal, &right_decimal);
    let result = decimal_to_base(sum, base);
    Ok(format_result(result, style))
}
