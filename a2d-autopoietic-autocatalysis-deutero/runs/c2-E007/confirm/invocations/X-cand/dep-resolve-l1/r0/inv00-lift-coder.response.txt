#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|&b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints.to_vec())
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut out = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(required) => version == required,
        Constraint::AtLeast(required) => version >= required,
        Constraint::Before(bound) => version < bound,
        Constraint::Caret(required) => {
            if version < required {
                false
            } else if required.major > 0 {
                version.major == required.major
            } else if required.minor > 0 {
                version.major == 0 && version.minor == required.minor
            } else {
                version == required
            }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{index}]"),
                name: root.package.clone(),
            });
        }
    }

    for (package_index, package) in catalog.iter().enumerate() {
        let package_site = format!("catalog[{package_index}]");
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: package_site.clone(),
                name: package.name.clone(),
            });
        }
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!(
                            "{package_site}.release[{release_index}].dep[{dependency_index}]"
                        ),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn check_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())
}

fn resolve_search(
    catalog: &std::collections::BTreeMap<&str, &Package>,
    active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    for package in active.keys() {
        if !catalog.contains_key(package.as_str()) {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }
    }

    for (package, version) in selected.iter() {
        if let Some(constraints) = active.get(package) {
            if !matches_all(*version, constraints) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: canonical_constraints(constraints.clone()),
                });
            }
        }
    }

    let mut candidates = Vec::new();
    for (package, constraints) in active.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let catalog_package = catalog[package.as_str()];
        let mut matching: Vec<Version> = catalog_package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| matches_all(*version, constraints))
            .collect();
        matching.sort();
        if matching.is_empty() {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements: canonical_constraints(constraints.clone()),
            });
        }
        candidates.push((package.clone(), matching.pop().unwrap()));
    }

    let Some((package, version)) = candidates.into_iter().next() else {
        return Ok(());
    };

    selected.insert(package.clone(), version);
    let release = catalog[package.as_str()]
        .releases
        .iter()
        .find(|release| release.version == version)
        .unwrap();

    let mut added = Vec::new();
    for dependency in &release.deps {
        let constraints = active.entry(dependency.package.clone()).or_default();
        constraints.push(dependency.constraint);
        added.push(dependency.package.clone());
    }

    let result = resolve_search(catalog, active, selected);

    for dependency in added {
        if let Some(constraints) = active.get_mut(&dependency) {
            // Restore the exact prior active set by removing one matching contribution.
            // The recursive call has already returned, so its additions were restored
            // by their own stack frames.
            let constraint = release
                .deps
                .iter()
                .find(|requirement| requirement.package == dependency)
                .map(|requirement| requirement.constraint);
            if let Some(constraint) = constraint {
                if let Some(position) = constraints.iter().rposition(|item| *item == constraint) {
                    constraints.remove(position);
                }
            }
        }
    }
    active.retain(|_, constraints| !constraints.is_empty());

    selected.remove(&package);
    result
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    check_duplicates(catalog)?;

    let catalog_map: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected = std::collections::BTreeMap::new();
    resolve_search(&catalog_map, &mut active, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}
