#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_char(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_str(word: &mut Word, s: &str, active_glob: bool) {
    for c in s.chars() {
        append_char(word, c, active_glob);
    }
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < chars.len() {
        let (pos, c) = chars[i];

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut word, c, false);
                }
                i += 1;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i += 1;
                } else if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_char(&mut word, next, false);
                        }
                        i += 2;
                    } else {
                        append_char(&mut word, '\\', false);
                        i += 1;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len() && is_name_start(chars[i + 1].1) {
                        active = true;
                        let start = i + 1;
                        let mut end = start + 1;
                        while end < chars.len() && is_name_continue(chars[end].1) {
                            end += 1;
                        }
                        let name: String = chars[start..end].iter().map(|(_, ch)| *ch).collect();
                        if let Some(value) = env_value(env, &name) {
                            append_str(&mut word, value, false);
                        }
                        i = end;
                    } else if i + 1 < chars.len() && chars[i + 1].1 == '{' {
                        active = true;
                        let mut end = i + 2;
                        let mut close = None;
                        while end < chars.len() {
                            if chars[end].1 == '\\' {
                                end += 2;
                            } else if chars[end].1 == '}' {
                                close = Some(end);
                                break;
                            } else {
                                end += 1;
                            }
                        }
                        let Some(close) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos });
                        };
                        let content: String =
                            chars[i + 2..close].iter().map(|(_, ch)| *ch).collect();
                        let (name, default) = if let Some(split_at) = content.find(":-") {
                            (&content[..split_at], Some(&content[split_at + 2..]))
                        } else {
                            (content.as_str(), None)
                        };
                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos });
                        }
                        let selected = if let Some(default) = default {
                            match env_value(env, name) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => {
                                    let mut decoded = String::new();
                                    let mut chars = default.chars();
                                    while let Some(ch) = chars.next() {
                                        if ch == '\\' {
                                            if let Some(next) = chars.next() {
                                                decoded.push(next);
                                            }
                                        } else {
                                            decoded.push(ch);
                                        }
                                    }
                                    decoded
                                }
                            }
                        } else {
                            env_value(env, name).unwrap_or("").to_string()
                        };
                        append_str(&mut word, &selected, false);
                        i = close + 1;
                    } else {
                        append_char(&mut word, '$', false);
                        i += 1;
                    }
                } else {
                    append_char(&mut word, c, false);
                    i += 1;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += 1;
                } else if c == '\'' {
                    quote = Quote::Single(pos);
                    active = true;
                    i += 1;
                } else if c == '"' {
                    quote = Quote::Double(pos);
                    active = true;
                    i += 1;
                } else if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if next == '\n' {
                        i += 2;
                    } else {
                        active = true;
                        append_char(&mut word, next, false);
                        i += 2;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len() && is_name_start(chars[i + 1].1) {
                        active = true;
                        let start = i + 1;
                        let mut end = start + 1;
                        while end < chars.len() && is_name_continue(chars[end].1) {
                            end += 1;
                        }
                        let name: String = chars[start..end].iter().map(|(_, ch)| *ch).collect();
                        if let Some(value) = env_value(env, &name) {
                            append_str(&mut word, value, true);
                        }
                        i = end;
                    } else if i + 1 < chars.len() && chars[i + 1].1 == '{' {
                        active = true;
                        let mut end = i + 2;
                        let mut close = None;
                        while end < chars.len() {
                            if chars[end].1 == '\\' {
                                end += 2;
                            } else if chars[end].1 == '}' {
                                close = Some(end);
                                break;
                            } else {
                                end += 1;
                            }
                        }
                        let Some(close) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos });
                        };
                        let content: String =
                            chars[i + 2..close].iter().map(|(_, ch)| *ch).collect();
                        let (name, default) = if let Some(split_at) = content.find(":-") {
                            (&content[..split_at], Some(&content[split_at + 2..]))
                        } else {
                            (content.as_str(), None)
                        };
                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos });
                        }
                        let selected = if let Some(default) = default {
                            match env_value(env, name) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => {
                                    let mut decoded = String::new();
                                    let mut chars = default.chars();
                                    while let Some(ch) = chars.next() {
                                        if ch == '\\' {
                                            if let Some(next) = chars.next() {
                                                decoded.push(next);
                                            }
                                        } else {
                                            decoded.push(ch);
                                        }
                                    }
                                    decoded
                                }
                            }
                        } else {
                            env_value(env, name).unwrap_or("").to_string()
                        };
                        append_str(&mut word, &selected, true);
                        i = close + 1;
                    } else {
                        active = true;
                        append_char(&mut word, '$', false);
                        i += 1;
                    }
                } else {
                    active = true;
                    append_char(&mut word, c, matches!(c, '*' | '?' | '['));
                    i += 1;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }
    if active {
        words.push(word);
    }
    Ok(words)
}
