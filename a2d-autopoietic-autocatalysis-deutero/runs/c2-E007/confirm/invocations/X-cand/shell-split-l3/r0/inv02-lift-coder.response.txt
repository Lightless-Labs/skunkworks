#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn push_char(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn push_string(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        push_char(word, c, active_glob);
    }
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut it = text.char_indices().peekable();

    while let Some((_, c)) = it.next() {
        if c == '\\' {
            if let Some((_, next)) = it.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }

    out
}

fn find_braced_close(input: &str, start: usize) -> Option<usize> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next()?;
        let n = c.len_utf8();
        if c == '\\' {
            i += n;
            if i < input.len() {
                let next = input[i..].chars().next()?;
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += n;
        }
    }
    None
}

fn parse_braced<'a>(
    input: &'a str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(&'a str, usize), SplitError> {
    let content_start = dollar + 2;
    let close = find_braced_close(input, content_start)
        .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[content_start..close];

    let name_end = content
        .char_indices()
        .find_map(|(i, c)| {
            if !is_name_continue(c) {
                Some(i)
            } else {
                None
            }
        })
        .unwrap_or(content.len());

    let name = &content[..name_end];
    let mut chars = name.chars();
    let valid_name = chars.next().map(is_name_start).unwrap_or(false)
        && chars.all(is_name_continue);

    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let tail = &content[name_end..];
    let selected = if tail.is_empty() {
        env_value(env, name).unwrap_or("")
    } else if let Some(default_text) = tail.strip_prefix(":-") {
        match env_value(env, name) {
            Some(value) if !value.is_empty() => value,
            _ => {
                // The borrowed return type is tied to input, so defaults are decoded
                // by the caller through this marker path.
                return Ok((default_text, close + 1));
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((selected, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    push_char(&mut word, c, false);
                }
                i += n;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += n;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let next_len = next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            push_char(&mut word, next, false);
                        }
                        i += next_len;
                    } else {
                        push_char(&mut word, '\\', false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let (value, next_i) = parse_braced(input, i, env)?;
                        if value.as_ptr() == input[i + 2..].as_ptr() {
                            let close = next_i - 1;
                            let content = &input[i + 2..close];
                            let name_end = content
                                .char_indices()
                                .find_map(|(j, ch)| {
                                    if !is_name_continue(ch) { Some(j) } else { None }
                                })
                                .unwrap_or(content.len());
                            let default_text = &content[name_end + 2..];
                            let decoded = decode_default(default_text);
                            push_string(&mut word, &decoded, false);
                        } else {
                            push_string(&mut word, value, false);
                        }
                        active = true;
                        i = next_i;
                    } else if let Some((_, first)) = input[i + n..].char_indices().next() {
                        if is_name_start(first) {
                            let mut end = i + n;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[i + n..end];
                            push_string(&mut word, env_value(env, name).unwrap_or(""), false);
                            active = true;
                            i = end;
                        } else {
                            push_char(&mut word, '$', false);
                            active = true;
                            i += n;
                        }
                    } else {
                        push_char(&mut word, '$', false);
                        active = true;
                        i += n;
                    }
                } else {
                    push_char(&mut word, c, false);
                    i += n;
                }

                if matches!(quote, Quote::Double(_)) && i == input.len() {
                    let _ = open_pos;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i += n;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += n;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '\n' {
                        i += next.len_utf8();
                    } else {
                        push_char(&mut word, next, false);
                        active = true;
                        i += next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let (value, next_i) = parse_braced(input, i, env)?;
                        if value.as_ptr() == input[i + 2..].as_ptr() {
                            let close = next_i - 1;
                            let content = &input[i + 2..close];
                            let name_end = content
                                .char_indices()
                                .find_map(|(j, ch)| {
                                    if !is_name_continue(ch) { Some(j) } else { None }
                                })
                                .unwrap_or(content.len());
                            let default_text = &content[name_end + 2..];
                            let decoded = decode_default(default_text);
                            push_string(&mut word, &decoded, true);
                        } else {
                            push_string(&mut word, value, true);
                        }
                        active = true;
                        i = next_i;
                    } else if let Some((_, first)) = input[i + n..].char_indices().next() {
                        if is_name_start(first) {
                            let mut end = i + n;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[i + n..end];
                            push_string(&mut word, env_value(env, name).unwrap_or(""), true);
                            active = true;
                            i = end;
                        } else {
                            push_char(&mut word, '$', false);
                            active = true;
                            i += n;
                        }
                    } else {
                        push_char(&mut word, '$', false);
                        active = true;
                        i += n;
                    }
                } else {
                    push_char(&mut word, c, matches!(c, '*' | '?' | '['));
                    active = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
