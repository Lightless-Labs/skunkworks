#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn push_text(word: &mut Word, text: &str, glob_active: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if glob_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> &'a str {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
        .unwrap_or("")
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;
    let bytes = input.as_bytes();

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let width = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                    i += width;
                } else {
                    word.text.push(c);
                    i += width;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i += width;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.text.push(next);
                        }
                        i += next.len_utf8();
                    } else {
                        word.text.push('\\');
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let dollar = i;
                    let next_i = i + width;
                    if next_i < input.len() && bytes[next_i] == b'(' {
                        return Err(SplitError::CommandSubstitution { pos: dollar });
                    }
                    if next_i < input.len() && bytes[next_i] == b'{' {
                        let (end, name, default) = parse_braced(input, dollar)?;
                        let value = match default {
                            Some(text) => {
                                let found = lookup(name, env);
                                if found.is_empty() {
                                    decode_default(text)
                                } else {
                                    found.to_string()
                                }
                            }
                            None => lookup(name, env).to_string(),
                        };
                        push_text(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(&input[next_i..end], env);
                            push_text(&mut word, value, false);
                            active = true;
                            i = end;
                        } else {
                            word.text.push('$');
                            i += width;
                        }
                    } else {
                        word.text.push('$');
                        i += width;
                    }
                } else {
                    word.text.push(c);
                    i += width;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += width;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_pos = i;
                    active = true;
                    i += width;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_pos = i;
                    active = true;
                    i += width;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '\n' {
                        i += next.len_utf8();
                    } else {
                        word.text.push(next);
                        active = true;
                        i += next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let dollar = i;
                    let next_i = i + width;
                    if next_i < input.len() && bytes[next_i] == b'(' {
                        return Err(SplitError::CommandSubstitution { pos: dollar });
                    }
                    if next_i < input.len() && bytes[next_i] == b'{' {
                        let (end, name, default) = parse_braced(input, dollar)?;
                        let value = match default {
                            Some(text) => {
                                let found = lookup(name, env);
                                if found.is_empty() {
                                    decode_default(text)
                                } else {
                                    found.to_string()
                                }
                            }
                            None => lookup(name, env).to_string(),
                        };
                        push_text(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(&input[next_i..end], env);
                            push_text(&mut word, value, true);
                            active = true;
                            i = end;
                        } else {
                            word.text.push('$');
                            active = true;
                            i += width;
                        }
                    } else {
                        word.text.push('$');
                        active = true;
                        i += width;
                    }
                } else {
                    let offset = word.text.len();
                    word.text.push(c);
                    if matches!(c, '*' | '?' | '[') {
                        word.glob.push(offset);
                    }
                    active = true;
                    i += width;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_braced(input: &str, dollar: usize) -> Result<(usize, &str, Option<&str>), SplitError> {
    let content_start = dollar + 2;
    let mut i = content_start;
    let mut close = None;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let width = c.len_utf8();
        if c == '\\' {
            i += width;
            if i < input.len() {
                let next = input[i..].chars().next().unwrap();
                i += next.len_utf8();
            }
        } else if c == '}' {
            close = Some(i);
            break;
        } else {
            i += width;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[content_start..close];

    let mut name_end = 0usize;
    let mut chars = content.char_indices();
    let Some((_, first)) = chars.next() else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    if !is_name_start(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    name_end += first.len_utf8();
    for (offset, c) in chars {
        if !is_name_continue(c) {
            break;
        }
        name_end = offset + c.len_utf8();
    }

    let name = &content[..name_end];
    if name_end == content.len() {
        Ok((close + 1, name, None))
    } else if content[name_end..].starts_with(":-") {
        Ok((close + 1, name, Some(&content[name_end + 2..])))
    } else {
        Err(SplitError::InvalidExpansion { pos: dollar })
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0usize;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}
