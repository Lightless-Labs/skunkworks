```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_name(name: &str, site: String) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_owned(),
        })
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut result = constraints.to_vec();
    result.sort();
    result.dedup();
    result
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut result = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            result.push(*byte as char);
        } else {
            result.push('%');
            result.push_str(&format!("{:02X}", byte));
        }
    }
    result
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|c| matches_constraint(version, *c))
}

fn validate_input(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        validate_name(&root.package, format!("root[{i}]"))?;
    }

    for (i, package) in catalog.iter().enumerate() {
        validate_name(&package.name, format!("catalog[{i}]"))?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                validate_name(
                    &dependency.package,
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                )?;
            }
        }
    }
    Ok(())
}

fn search(
    catalog: &[Package],
    selected: &mut Vec<(String, Version)>,
    attempts: &mut usize,
    max_attempts: usize,
    first_failure: &mut Option<ResolveError>,
) -> Result<Vec<Selection>, ResolveError> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

    // Root constraints are seeded by the caller through a private sentinel-free list.
    // The selected releases' dependencies are added here; roots are supplied separately
    // in the wrapper via the active-root argument below.
    let _ = (&mut active, catalog, selected, attempts, max_attempts, first_failure);
    unreachable!()
}

fn resolve_search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut Vec<(String, Version)>,
    attempts: &mut usize,
    max_attempts: usize,
    first_failure: &mut Option<ResolveError>,
) -> Result<Vec<Selection>, ResolveError> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in selected.iter() {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    if let Some(name) = active
        .keys()
        .filter(|name| !catalog.iter().any(|p| p.name == ***name))
        .next()
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    let mut selected_sorted = selected.clone();
    selected_sorted.sort_by(|a, b| a.0.cmp(&b.0));
    for (name, version) in &selected_sorted {
        let constraints = canonical_constraints(active.get(name).map_or(&[], |v| v));
        if !matches_all(*version, &constraints) {
            return Err(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: constraints,
            });
        }
    }

    for (name, constraints) in &active {
        if selected.iter().any(|(selected_name, _)| selected_name == name) {
            continue;
        }
        let package = catalog.iter().find(|p| p.name == *name).unwrap();
        if !package
            .releases
            .iter()
            .any(|r| matches_all(r.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: canonical_constraints(constraints),
            });
        }
    }

    let next = active
        .keys()
        .find(|name| !selected.iter().any(|(selected_name, _)| selected_name == *name))
        .cloned();

    let Some(name) = next else {
        let mut result = selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect::<Vec<_>>();
        result.sort_by(|a, b| a.package.cmp(&b.package));
        return Ok(result);
    };

    let constraints = active.get(&name).map(Vec::as_slice).unwrap_or(&[]);
    let package = catalog.iter().find(|p| p.name == name).unwrap();
    let mut candidates = package
        .releases
        .iter()
        .filter(|r| matches_all(r.version, constraints))
        .map(|r| r.version)
        .collect::<Vec<_>>();
    candidates.sort_by(|a, b| b.cmp(a));

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.push((name.clone(), version));
        match resolve_search(catalog, roots, selected, attempts, max_attempts, first_failure) {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => {
                selected.pop();
                return Err(error);
            }
            Err(error) => {
                if first_failure.is_none() {
                    *first_failure = Some(error);
                }
                selected.pop();
            }
        }
    }

    Err(first_failure
        .clone()
        .unwrap_or(ResolveError::NoMatch {
            package: name.clone(),
            requirements: canonical_constraints(constraints),
        }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_input(catalog, roots)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut attempts = 0;
    let mut first_failure = None;
    resolve_search(
        catalog,
        roots,
        &mut Vec::new(),
        &mut attempts,
        max_attempts,
        &mut first_failure,
    )
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}
```
