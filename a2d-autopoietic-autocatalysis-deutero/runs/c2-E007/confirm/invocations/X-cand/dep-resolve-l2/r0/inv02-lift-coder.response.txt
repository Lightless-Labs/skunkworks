```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut out = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn check_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    for i in 0..catalog.len() {
        if catalog[..i]
            .iter()
            .any(|previous| previous.name == catalog[i].name)
        {
            return Err(ResolveError::DuplicatePackage {
                package: catalog[i].name.clone(),
            });
        }
    }

    for package in catalog {
        for i in 0..package.releases.len() {
            if package.releases[..i]
                .iter()
                .any(|previous| previous.version == package.releases[i].version)
            {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: package.releases[i].version,
                });
            }
        }
    }
    Ok(())
}

fn search(
    catalog: &[Package],
    active: &mut Vec<(String, Vec<Constraint>)>,
    selected: &mut Vec<(String, Version, usize, usize)>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    active.sort_by(|a, b| a.0.cmp(&b.0));
    for (_, constraints) in active.iter_mut() {
        canonicalize(constraints);
    }

    let selected_names: Vec<&str> = selected.iter().map(|s| s.0.as_str()).collect();

    let mut missing: Vec<String> = active
        .iter()
        .filter(|(name, _)| !catalog.iter().any(|p| p.name == *name))
        .map(|(name, _)| name.clone())
        .collect();
    missing.sort();
    missing.dedup();
    if let Some(package) = missing.into_iter().next() {
        return Err(ResolveError::Missing { package });
    }

    let mut selected_sorted: Vec<_> = selected.iter().collect();
    selected_sorted.sort_by(|a, b| a.0.cmp(&b.0));
    for (name, version, _, _) in selected_sorted {
        let constraints = active
            .iter()
            .find(|(active_name, _)| active_name == name)
            .map(|(_, constraints)| constraints)
            .unwrap_or(&Vec::new());
        if !matches_all(*version, constraints) {
            return Err(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: constraints.clone(),
            });
        }
    }

    for (name, constraints) in &active {
        if selected_names.contains(&name.as_str()) {
            continue;
        }
        let package = catalog.iter().find(|p| p.name == *name).unwrap();
        if !package
            .releases
            .iter()
            .any(|release| matches_all(release.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let next = active
        .iter()
        .find(|(name, _)| !selected_names.contains(&name.as_str()))
        .map(|(name, constraints)| (name.clone(), constraints.clone()));

    let Some((name, constraints)) = next else {
        let mut result: Vec<Selection> = selected
            .iter()
            .map(|(package, version, _, _)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();
        result.sort_by(|a, b| a.package.cmp(&b.package));
        return Ok(result);
    };

    let package_index = catalog.iter().position(|p| p.name == name).unwrap();
    let mut candidates: Vec<usize> = catalog[package_index]
        .releases
        .iter()
        .enumerate()
        .filter(|(_, release)| matches_all(release.version, &constraints))
        .map(|(i, _)| i)
        .collect();
    candidates.sort_by(|a, b| {
        catalog[package_index].releases[*b]
            .version
            .cmp(&catalog[package_index].releases[*a].version)
    });

    let mut first_error: Option<ResolveError> = None;
    for release_index in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;

        let release = &catalog[package_index].releases[release_index];
        selected.push((
            name.clone(),
            release.version,
            package_index,
            release_index,
        ));
        let old_len = active.len();
        for dependency in &release.deps {
            if let Some((_, deps)) = active
                .iter_mut()
                .find(|(active_name, _)| active_name == &dependency.package)
            {
                deps.push(dependency.constraint);
            } else {
                active.push((
                    dependency.package.clone(),
                    vec![dependency.constraint],
                ));
            }
        }

        let result = search(catalog, active, selected, attempts, max_attempts);
        active.truncate(old_len);
        if let Err(ResolveError::Limit { attempts }) = result {
            return Err(ResolveError::Limit { attempts });
        }
        selected.pop();

        match result {
            Ok(solution) => return Ok(solution),
            Err(error) => {
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
        package: name,
        requirements: constraints,
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    check_duplicates(catalog)?;

    let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();
    for root in roots {
        if let Some((_, constraints)) = active
            .iter_mut()
            .find(|(name, _)| name == &root.package)
        {
            constraints.push(root.constraint);
        } else {
            active.push((root.package.clone(), vec![root.constraint]));
        }
    }

    search(
        catalog,
        &mut active,
        &mut Vec::new(),
        &mut 0,
        max_attempts,
    )
}
```
