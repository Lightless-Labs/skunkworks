use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let src = input.as_bytes();
    let mut out = Vec::with_capacity(src.len());
    let mut offsets = Vec::with_capacity(src.len());
    let mut i = 0;

    while i < src.len() {
        if src[i] == b'%' {
            if i + 2 >= src.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(src[i + 1]);
            let lo = hex_value(src[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    out.push((h << 4) | l);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            out.push(src[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&out) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let decoded_at = e.valid_up_to();
            let original_at = offsets.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn component_safe(component: Component, b: u8) -> bool {
    is_unreserved(b)
        || match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if component_safe(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn global_byte_scan(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn parse_port(port: &str, start: usize) -> Result<Option<u16>, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for b in port.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(Some(value as u16))
}

fn map_codec_parse(err: CodecError, offset: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_byte_scan(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| auth_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + at_positions[1],
        });
    }
    let (userinfo, hostport_start) = if let Some(&p) = at_positions.first() {
        if p == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (Some(authority[..p].to_owned()), auth_start + p + 1)
    } else {
        (None, auth_start)
    };

    let hostport = &input[hostport_start..auth_end];
    let (host, port, ipv6) = if hostport.starts_with('[') {
        let open = hostport_start;
        let close_rel = hostport[1..].find(']').map(|n| n + 1);
        let close = match close_rel {
            Some(n) => n,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close];
        let parsed = match Ipv6Addr::from_str(interior) {
            Ok(v) => v,
            Err(_) => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let _ = parsed;
        let suffix = &hostport[close + 1..];
        let p = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            parse_port(&suffix[1..], hostport_start + close + 2)?
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), p, true)
    } else {
        let rel_colon = hostport.find(':');
        let (host_part, p) = if let Some(c) = rel_colon {
            (&hostport[..c], parse_port(&hostport[c + 1..], hostport_start + c + 1)?)
        } else {
            (hostport, None)
        };
        if host_part.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host_part.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hostport_start + i });
            }
        }
        (host_part.to_ascii_lowercase(), p, false)
    };

    let component_tail = &input[auth_end..];
    let tail_base = auth_end;
    let fragment_rel = component_tail.find('#');
    let before_fragment = fragment_rel.map(|p| &component_tail[..p]).unwrap_or(component_tail);
    let fragment = fragment_rel.map(|p| component_tail[p + 1..].to_owned());
    let fragment_offset = fragment_rel.map(|p| tail_base + p + 1);

    let query_rel = before_fragment.find('?');
    let path_raw = query_rel
        .map(|p| &before_fragment[..p])
        .unwrap_or(before_fragment);
    let query = query_rel.map(|p| before_fragment[p + 1..].to_owned());
    let query_offset = query_rel.map(|p| tail_base + p + 1);
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(ref u) = userinfo {
        decode_component(Component::UserInfo, u)
            .map_err(|e| map_codec_parse(e, auth_start))?;
    }
    if !path_raw.is_empty() {
        decode_component(Component::Path, path_raw)
            .map_err(|e| map_codec_parse(e, tail_base))?;
    }
    if let Some(ref q) = query {
        decode_component(Component::Query, q)
            .map_err(|e| map_codec_parse(e, query_offset.unwrap_or(tail_base)))?;
    }
    if let Some(ref f) = fragment {
        decode_component(Component::Fragment, f)
            .map_err(|e| map_codec_parse(e, fragment_offset.unwrap_or(tail_base)))?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn map_codec_path(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        if decoded == "." {
            terminal_navigation = index + 1 == raw_segments.len();
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            terminal_navigation = index + 1 == raw_segments.len();
        } else {
            stack.push((*segment).to_owned());
            terminal_navigation = false;
        }
    }

    if terminal_navigation {
        stack.push(String::new());
    }
    Ok(format!("/{}", stack.join("/")))
}

fn map_path_resolve(err: PathError) -> ResolveError {
    ResolveError::Path(err)
}

fn validate_reference_bytes(reference: &str) -> Result<(), ResolveError> {
    for (at, b) in reference.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
        }
    }
    Ok(())
}

fn split_reference_tail(s: &str) -> (&str, Option<&str>, Option<&str>) {
    let frag_pos = s.find('#');
    let before_frag = frag_pos.map(|i| &s[..i]).unwrap_or(s);
    let frag = frag_pos.map(|i| &s[i + 1..]);
    let query_pos = before_frag.find('?');
    let path = query_pos.map(|i| &before_frag[..i]).unwrap_or(before_frag);
    let query = query_pos.map(|i| &before_frag[i + 1..]);
    (path, query, frag)
}

fn normalize_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(map_path_resolve)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalize_url_path(base_url)?;
    validate_reference_bytes(reference)?;

    let (ref_path_before_delims, _, _) = split_reference_tail(reference);
    let first_delim = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delim].contains(':') {
        let u = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_url_path(u);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let u = parse_url(&absolute).map_err(ResolveError::Reference)?;
        return normalize_url_path(u);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference_tail(reference);
    let selected_path = if ref_path.is_empty() {
        base_url.path.clone()
    } else if ref_path.starts_with('/') {
        ref_path.to_owned()
    } else {
        let cut = base_url.path.rfind('/').map(|i| i + 1).unwrap_or(0);
        format!("{}{}", &base_url.path[..cut], ref_path)
    };
    let selected_query = if ref_path.is_empty() && ref_query.is_none() {
        base_url.query.clone()
    } else {
        ref_query.map(str::to_owned)
    };

    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(ref info) = base_url.userinfo {
        constructed.push_str(info);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&selected_path);
    if let Some(q) = selected_query {
        constructed.push('?');
        constructed.push_str(&q);
    }
    if let Some(f) = ref_fragment {
        constructed.push('#');
        constructed.push_str(f);
    }

    let _ = ref_path_before_delims;
    let u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalize_url_path(u)
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();

    for (field_start, field) in {
        let mut v = Vec::new();
        let mut start = 0;
        for (i, b) in input.bytes().enumerate() {
            if b == b'&' {
                v.push((start, &input[start..i]));
                start = i + 1;
            }
        }
        v.push((start, &input[start..]));
        v
    } {
        if field.is_empty() {
            continue;
        }
        let eq = field.find('=');
        let (raw_key, raw_value, value_offset) = match eq {
            Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
            None => (field, "", field_start + field.len()),
        };
        let key_plus = raw_key.replace('+', " ");
        let key = decode_component(Component::Query, &key_plus).map_err(|e| match e {
            CodecError::BadEscape { at } => QueryError::BadEscape {
                at: field_start + at,
            },
            CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 {
                at: field_start + at,
            },
        })?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }

        let value_plus = raw_value.replace('+', " ");
        let value = decode_component(Component::Query, &value_plus).map_err(|e| match e {
            CodecError::BadEscape { at } => QueryError::BadEscape {
                at: value_offset + at,
            },
            CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 {
                at: value_offset + at,
            },
        })?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else if b == b' ' {
            out.push('+');
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

fn map_query_canonical(err: QueryError) -> CanonicalError {
    CanonicalError::Query(err)
}

fn canonical_segment(segment: &str) -> Result<String, CodecError> {
    let decoded = decode_component(Component::Path, segment)?;
    let mut out = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in decoded.as_bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    Ok(out)
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(q) => Some(parse_query(q).map_err(map_query_canonical)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(info) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, info).map_err(|e| {
            CanonicalError::Parse(map_codec_parse(e, 0))
        })?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host)
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let omit = (url.scheme == "http" && port == 80)
            || (url.scheme == "https" && port == 443);
        if !omit {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_path_segments = Vec::new();
    for segment in path.split('/') {
        canonical_path_segments.push(
            canonical_segment(segment)
                .map_err(|e| CanonicalError::Path(map_codec_path(e)))?,
        );
    }
    out.push_str(&canonical_path_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment).map_err(|e| {
            CanonicalError::Parse(map_codec_parse(e, 0))
        })?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}
