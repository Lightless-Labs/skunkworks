#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn append_under_hundred(n: u128, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10) as usize - 2]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

fn append_group(n: u128, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_under_hundred(remainder, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    while remaining != 0 {
        groups.push((remaining % 1000) as u16);
        remaining /= 1000;
    }

    let highest = groups.len() - 1;
    let mut emitted = false;
    for index in (0..=highest).rev() {
        let group = groups[index] as u128;
        if group == 0 {
            continue;
        }
        if emitted {
            out.push(' ');
        }
        append_group(group, &mut out);
        if index > 0 {
            out.push(' ');
            out.push_str(SCALES[index - 1]);
        }
        emitted = true;
    }

    // Replace the final group separator with " and " when the units group is 1..99.
    if groups.len() > 1 && (1..100).contains(&(groups[0] as u128)) {
        let units = groups[0] as u128;
        let unit_words = {
            let mut s = String::new();
            append_under_hundred(units, &mut s);
            s
        };
        if let Some(pos) = out.rfind(&unit_words) {
            if pos > 0 && out.as_bytes()[pos - 1] == b' ' {
                out.replace_range(pos - 1..pos, " and ");
            }
        }
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<(u128, WordKind)> {
    if word == "zero" {
        return Some((0, WordKind::Number));
    }
    for (i, item) in SMALL.iter().enumerate().skip(1) {
        if word == *item {
            return Some((i as u128, WordKind::Number));
        }
    }
    for (i, item) in TENS.iter().enumerate() {
        if word == *item {
            return Some(((i as u128 + 2) * 10, WordKind::Number));
        }
    }
    match word {
        "hundred" => Some((100, WordKind::Hundred)),
        "and" => Some((0, WordKind::And)),
        "thousand" => Some((1_000, WordKind::Scale)),
        "million" => Some((1_000_000, WordKind::Scale)),
        "billion" => Some((1_000_000_000, WordKind::Scale)),
        "trillion" => Some((1_000_000_000_000, WordKind::Scale)),
        "quadrillion" => Some((1_000_000_000_000_000, WordKind::Scale)),
        "quintillion" => Some((1_000_000_000_000_000_000, WordKind::Scale)),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((start, &input[start..index]));
            start = index + 1;
        }
    }
    tokens.push((start, &input[start..]));

    for (i, &(position, token)) in tokens.iter().enumerate() {
        if token == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: position });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: position });
        }
    }

    let negative = tokens.first().is_some_and(|(_, token)| *token == "minus");
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for &(_, token) in arithmetic {
        if token == "minus" {
            continue;
        }
        let (value, kind) = word_value(token).expect("dictionary scan validated tokens");
        match kind {
            WordKind::Number => group += value,
            WordKind::Hundred => group *= 100,
            WordKind::Scale => {
                total += group * value;
                group = 0;
            }
            WordKind::And => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != input.as_bytes() {
        let mismatch = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(value)
}
