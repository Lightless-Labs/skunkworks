#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten",
    "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand",
    "million",
    "billion",
    "trillion",
    "quadrillion",
    "quintillion",
];

fn encode_under_thousand(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            encode_under_thousand(remainder, out);
        }
    } else if n >= 20 {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    } else if n != 0 {
        out.push_str(SMALL[n as usize]);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    while remaining != 0 {
        groups.push((remaining % 1000) as u16);
        remaining /= 1000;
    }

    let mut first = true;
    for index in (0..groups.len()).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }
        if !first {
            let units_group = index == 0 && group < 100;
            if units_group {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        encode_under_thousand(group, &mut out);
        if index > 0 {
            out.push(' ');
            out.push_str(SCALES[index - 1]);
        }
        first = false;
    }

    Ok(out)
}

fn dictionary_value(word: &str) -> Option<DictWord> {
    if word == "zero" {
        return Some(DictWord::Zero);
    }
    if word == "hundred" {
        return Some(DictWord::Hundred);
    }
    if word == "and" {
        return Some(DictWord::And);
    }
    for (i, candidate) in SMALL.iter().enumerate().skip(1) {
        if word == *candidate {
            return Some(DictWord::Number(i as u128));
        }
    }
    for (i, candidate) in TENS.iter().enumerate() {
        if word == *candidate {
            return Some(DictWord::Number((i as u128 + 2) * 10));
        }
    }
    for (i, candidate) in SCALES.iter().enumerate() {
        if word == *candidate {
            return Some(DictWord::Scale(1000u128.pow((i + 1) as u32)));
        }
    }
    if word == "minus" {
        return Some(DictWord::Minus);
    }
    None
}

#[derive(Clone, Copy)]
enum DictWord {
    Zero,
    Number(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(usize, &str, DictWord)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            let word = &input[start..index];
            let dict = dictionary_value(word)
                .ok_or(WordsError::InvalidWord { index: start })?;
            tokens.push((start, word, dict));
            start = index + 1;
        }
    }
    let word = &input[start..];
    let dict = dictionary_value(word).ok_or(WordsError::InvalidWord { index: start })?;
    tokens.push((start, word, dict));

    let negative = matches!(tokens.first().map(|t| t.2), Some(DictWord::Minus));
    for (i, (index, _, token)) in tokens.iter().enumerate() {
        if matches!(token, DictWord::Minus) && i != 0 {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let arithmetic_start = if negative { 1 } else { 0 };
    let mut total = 0u128;
    let mut group = 0u128;

    for (_, _, token) in tokens.iter().skip(arithmetic_start) {
        match token {
            DictWord::Zero | DictWord::And => {}
            DictWord::Number(n) => group += *n,
            DictWord::Hundred => group *= 100,
            DictWord::Scale(scale) => {
                total += group * *scale;
                group = 0;
            }
            DictWord::Minus => unreachable!(),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if magnitude == 0 {
        0
    } else if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != input.as_bytes() {
        let mismatch = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }
    Ok(value)
}
