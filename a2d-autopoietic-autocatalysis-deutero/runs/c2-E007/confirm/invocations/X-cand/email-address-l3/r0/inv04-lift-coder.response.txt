#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut clean = Vec::<u8>::with_capacity(bytes.len());
    let mut map = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<Run>::new();
    let mut i = 0usize;
    let mut quote_prefix = false;
    let mut quote_closed = false;

    while i < bytes.len() {
        let b = bytes[i];

        if quote_prefix && !quote_closed {
            clean.push(b);
            map.push(i);
            if b == b'\\' {
                if i + 1 < bytes.len() {
                    clean.push(bytes[i + 1]);
                    map.push(i + 1);
                    i += 2;
                    continue;
                }
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        let is_ws = is_hws(b);
        let is_fold = b == b'\r'
            && i + 2 < bytes.len()
            && bytes[i + 1] == b'\n'
            && is_hws(bytes[i + 2]);

        if is_ws || is_fold || b == b'(' {
            let start = i;
            let mut depth = 0usize;
            let mut outer = 0usize;
            let mut in_construct = true;

            while in_construct && i < bytes.len() {
                let c = bytes[i];

                if depth > 0 {
                    if c == b'\\' {
                        if i + 1 >= bytes.len()
                            || !(0x20..=0x7e).contains(&bytes[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: i });
                        }
                        i += 2;
                        continue;
                    }

                    if c == b'\r' || c == b'\n' {
                        if c == b'\r'
                            && i + 2 < bytes.len()
                            && bytes[i + 1] == b'\n'
                            && is_hws(bytes[i + 2])
                        {
                            i += 2;
                            while i < bytes.len() && is_hws(bytes[i]) {
                                i += 1;
                            }
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }

                    if !(c == b'\t' || (0x20..=0x7e).contains(&c)) {
                        return Err(EmailError::CommentChar { at: i });
                    }

                    if c == b'(' {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        depth += 1;
                    } else if c == b')' {
                        depth -= 1;
                    }
                    i += 1;
                    continue;
                }

                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && i + 2 < bytes.len()
                        && bytes[i + 1] == b'\n'
                        && is_hws(bytes[i + 2])
                    {
                        i += 2;
                        while i < bytes.len() && is_hws(bytes[i]) {
                            i += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    if depth == 0 {
                        outer = i;
                    }
                    depth = 1;
                    i += 1;
                    continue;
                }
                in_construct = false;
            }

            if depth > 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if b == b'\r'
                && i + 2 < bytes.len()
                && bytes[i + 1] == b'\n'
                && is_hws(bytes[i + 2])
            {
                let start = i;
                i += 2;
                while i < bytes.len() && is_hws(bytes[i]) {
                    i += 1;
                }
                runs.push(Run { start, end: i });
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        map.push(i);
        if clean.len() == 1 && b == b'"' {
            quote_prefix = true;
        }
        i += 1;
    }

    let end_pos = input.len();
    for run in runs {
        let retained_before = map.iter().rposition(|&p| p < run.start);
        let retained_after = map.iter().position(|&p| p >= run.end);
        let at_start = retained_before.is_none();
        let at_end = retained_after.is_none();
        let adjacent_at = retained_before
            .map(|idx| clean[idx] == b'@')
            .unwrap_or(false)
            || retained_after
                .map(|idx| clean[idx] == b'@')
                .unwrap_or(false);
        if !(at_start || at_end || adjacent_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    if clean[0] == b'"' {
        let mut j = 1usize;
        let mut found = None;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                found = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        match found {
            Some(j) => protected_end = j + 1,
            None => return Err(EmailError::UnclosedQuote { at: map[0] }),
        }
    }

    let ats: Vec<usize> = (protected_end..clean.len())
        .filter(|&j| clean[j] == b'@')
        .collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: map[ats[1]] });
    }
    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: map[at_idx] });
    }

    let mut local = String::new();
    let quoted = clean[0] == b'"';
    if !quoted {
        if at_idx > 64 {
            return Err(EmailError::LocalTooLong { at: map[64] });
        }
        for j in 0..at_idx {
            let b = clean[j];
            if b == b'.' {
                if j == 0 || j + 1 == at_idx || clean[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: map[j] });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar { at: map[j] });
            }
            local.push(b as char);
        }
    } else {
        let closing = protected_end - 1;
        if closing + 1 != at_idx {
            return Err(EmailError::TrailingLocal {
                at: map[closing + 1],
            });
        }

        let mut decoded = Vec::<(u8, usize)>::new();
        let mut j = 1usize;
        while j < closing {
            let b = clean[j];
            if b == b'\\' {
                if j + 1 >= closing {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                let next = clean[j + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                decoded.push((next, map[j]));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: map[j] });
                }
                decoded.push((b, map[j]));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: map[closing] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded[64].1,
            });
        }
        for (b, _) in decoded {
            local.push(b as char);
        }
    }

    let domain_start_idx = at_idx + 1;
    let mut domain_end = clean.len();
    let removed_root = if domain_end > domain_start_idx && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(map[domain_end])
    } else {
        None
    };

    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain {
            at: map.get(domain_start_idx).copied().unwrap_or(end_pos),
        });
    }
    if domain_end - domain_start_idx > 253 {
        return Err(EmailError::DomainTooLong {
            at: map[domain_start_idx + 253],
        });
    }

    let mut labels = Vec::<(usize, usize)>::new();
    let mut label_start = domain_start_idx;
    let mut j = domain_start_idx;

    while j <= domain_end {
        if j == domain_end || clean[j] == b'.' {
            if j == label_start {
                let at = if j == domain_end {
                    removed_root.unwrap_or_else(|| {
                        map.get(domain_end).copied().unwrap_or(end_pos)
                    })
                } else {
                    map[j]
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label_end = j;
            if label_end - label_start > 63 {
                return Err(EmailError::LabelTooLong {
                    at: map[label_start + 63],
                });
            }
            if clean[label_start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: map[label_start],
                });
            }
            if clean[label_end - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: map[label_end - 1],
                });
            }
            for k in label_start..label_end {
                let b = clean[k];
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar { at: map[k] });
                }
            }

            labels.push((label_start, label_end));
            label_start = j + 1;
        }
        j += 1;
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: map[domain_start_idx],
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[last_start] });
    }

    let mut domain = String::new();
    for &b in &clean[domain_start_idx..domain_end] {
        domain.push((b as char).to_ascii_lowercase());
    }

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
