#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut out = Vec::with_capacity(input.len());
    let mut map = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut first_retained = true;

    while i < input.len() {
        if first_retained && input[i] == b'"' {
            let mut j = i;
            while j < input.len() {
                let b = input[j];
                out.push(b);
                map.push(j);
                if b == b'\\' {
                    j += 1;
                    if j < input.len() {
                        out.push(input[j]);
                        map.push(j);
                    }
                } else if b == b'"' && j != i {
                    j += 1;
                    break;
                }
                j += 1;
            }
            first_retained = false;
            i = j;
            continue;
        }

        let construct_start = i;
        let mut consumed = false;

        if input[i] == b' ' || input[i] == b'\t' {
            i += 1;
            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }
            consumed = true;
        } else if input[i] == b'\r' || input[i] == b'\n' {
            if let Some(end) = fold_end(input, i) {
                i = end;
                consumed = true;
            } else {
                return Err(EmailError::BadFold { at: i });
            }
        } else if input[i] == b'(' {
            let outer = i;
            let mut stack = vec![i];
            i += 1;
            while i < input.len() && !stack.is_empty() {
                let b = input[i];
                if b == b'\\' {
                    if i + 1 >= input.len()
                        || !(0x20..=0x7e).contains(&input[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }
                if b == b'\r' || b == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if b == b'(' {
                    if stack.len() == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    stack.push(i);
                    i += 1;
                    continue;
                }
                if b == b')' {
                    stack.pop();
                    i += 1;
                    continue;
                }
                if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }
            if !stack.is_empty() {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            consumed = true;
        } else if input[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if consumed {
            let run_start = construct_start;
            let mut run_end = i;
            loop {
                if i >= input.len() {
                    break;
                }
                if input[i] == b' ' || input[i] == b'\t' {
                    while i < input.len() && is_hws(input[i]) {
                        i += 1;
                    }
                    run_end = i;
                } else if input[i] == b'\r' || input[i] == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        run_end = i;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if input[i] == b'(' {
                    let outer = i;
                    let mut stack = vec![i];
                    i += 1;
                    while i < input.len() && !stack.is_empty() {
                        let b = input[i];
                        if b == b'\\' {
                            if i + 1 >= input.len()
                                || !(0x20..=0x7e).contains(&input[i + 1])
                            {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if b == b'\r' || b == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if b == b'(' {
                            if stack.len() == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            stack.push(i);
                            i += 1;
                        } else if b == b')' {
                            stack.pop();
                            i += 1;
                        } else if b == b'\t' || (0x20..=0x7e).contains(&b) {
                            i += 1;
                        } else {
                            return Err(EmailError::CommentChar { at: i });
                        }
                    }
                    if !stack.is_empty() {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    run_end = i;
                } else {
                    break;
                }
            }
            runs.push(Run {
                start: run_start,
                end: run_end,
            });
            continue;
        }

        out.push(input[i]);
        map.push(i);
        first_retained = false;
        i += 1;
    }

    Ok((out, map, runs))
}

fn orig(map: &[usize], i: usize, input_len: usize) -> usize {
    map.get(i).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, map, runs) = cleanup(raw)?;
    for run in runs {
        let left_at = run.start > 0 && clean.get(run.start - 1) == Some(&b'@');
        let right_at = run.end < clean.len() && clean.get(run.end) == Some(&b'@');
        let boundary = run.start == 0 || run.end == raw.len() || left_at || right_at;
        if !boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: orig(&map, 0, raw.len()) });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < clean.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if clean[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: orig(&map, ats[1], raw.len()),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: orig(&map, sep, raw.len()),
        });
    }

    let quoted = clean[0] == b'"';
    let local;
    if !quoted {
        let len = sep;
        if len > 64 {
            return Err(EmailError::LocalTooLong {
                at: orig(&map, 64, raw.len()),
            });
        }
        let mut prev_dot = false;
        for j in 0..len {
            let b = clean[j];
            if b == b'.' {
                if j == 0 || j + 1 == len || prev_dot {
                    return Err(EmailError::LocalDot {
                        at: orig(&map, j, raw.len()),
                    });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !(b.is_ascii_alphanumeric()
                    || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
                {
                    return Err(EmailError::LocalChar {
                        at: orig(&map, j, raw.len()),
                    });
                }
            }
        }
        local = String::from_utf8(clean[..sep].to_vec()).unwrap();
    } else {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: orig(&map, end + 1, raw.len()),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            if clean[j] == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: orig(&map, j, raw.len()),
                    });
                }
                let b = clean[j + 1];
                if b != b'"' && b != b'\\' && b != b'@' {
                    return Err(EmailError::BadEscape {
                        at: orig(&map, j, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(orig(&map, j, raw.len()));
                j += 2;
            } else {
                let b = clean[j];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: orig(&map, j, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(orig(&map, j, raw.len()));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: orig(&map, end, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).unwrap();
    }

    let domain_start = sep + 1;
    let domain_end = clean.len();
    let mut effective_end = domain_end;
    let mut root_dot = None;
    if effective_end > domain_start && clean[effective_end - 1] == b'.' {
        effective_end -= 1;
        root_dot = Some(effective_end);
    }
    if effective_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: orig(&map, domain_start, raw.len()),
        });
    }
    if effective_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: orig(&map, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    let mut j = domain_start;
    while j <= effective_end {
        if j == effective_end || clean[j] == b'.' {
            labels.push((start, j));
            start = j + 1;
        }
        j += 1;
    }

    for (idx, &(a, b)) in labels.iter().enumerate() {
        if a == b {
            let pos = if idx + 1 == labels.len() {
                root_dot.map(|x| orig(&map, x, raw.len()))
                    .unwrap_or_else(|| orig(&map, a, raw.len()))
            } else {
                orig(&map, a, raw.len())
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        if b - a > 63 {
            return Err(EmailError::LabelTooLong {
                at: orig(&map, a + 63, raw.len()),
            });
        }
        if clean[a] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: orig(&map, a, raw.len()),
            });
        }
        if clean[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: orig(&map, b - 1, raw.len()),
            });
        }
        for k in a..b {
            let c = clean[k];
            if !(c.is_ascii_alphanumeric() || c == b'-') {
                return Err(EmailError::DomainChar {
                    at: orig(&map, k, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: orig(&map, domain_start, raw.len()),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: orig(&map, last_start, raw.len()),
        });
    }

    let domain = clean[domain_start..effective_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();
    Ok(Address {
        local,
        domain,
        quoted,
    })
}
