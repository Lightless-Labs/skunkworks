#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut names = std::collections::BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    catalog_by_name: std::collections::BTreeMap<&'a str, &'a Package>,
    locks: std::collections::BTreeMap<&'a str, Version>,
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, std::collections::BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
        root_requirements: &std::collections::BTreeMap<String, Vec<Constraint>>,
    ) -> Result<(), ResolveError> {
        let mut active = root_requirements.clone();

        for (name, version) in selected.iter() {
            if let Some(package) = self.catalog_by_name.get(name.as_str()) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }

        for constraints in active.values_mut() {
            *constraints = canonical_constraints(std::mem::take(constraints));
        }

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog_by_name.contains_key(name.as_str()))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if !satisfies(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.catalog_by_name[name.as_str()];
            if !package
                .releases
                .iter()
                .any(|release| satisfies(release.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            self.consider(selected);
            return Ok(());
        };

        let constraints = &active[&name];
        let package = self.catalog_by_name[name.as_str()];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| satisfies(*v, constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(name.as_str()).copied() {
            if let Some(index) = candidates.iter().position(|v| *v == locked) {
                candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), candidate);
            match self.inspect(selected, root_requirements) {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => {
                    selected.remove(&name);
                    return Err(error);
                }
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
            selected.remove(&name);
        }
        Ok(())
    }

    fn consider(&mut self, selected: &std::collections::BTreeMap<String, Version>) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| selected.get(**name) != Some(*version))
            .count();
        let replace = match &self.best {
            None => true,
            Some((best_cost, best)) => {
                if cost != *best_cost {
                    cost < *best_cost
                } else {
                    self.lex_vector_greater(selected, best)
                }
            }
        };
        if replace {
            self.best = Some((cost, selected.clone()));
        }
    }

    fn lex_vector_greater(
        &self,
        left: &std::collections::BTreeMap<String, Version>,
        right: &std::collections::BTreeMap<String, Version>,
    ) -> bool {
        let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        names.sort_unstable();
        for name in names {
            match (left.get(name), right.get(name)) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let catalog_by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let lock_map: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|l| (l.package.as_str(), l.version))
        .collect();
    let mut root_requirements: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        root_requirements
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut search = Search {
        catalog,
        catalog_by_name,
        locks: lock_map,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    search.inspect(&mut selected, &root_requirements)?;

    let Some((_, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected_vec: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names = std::collections::BTreeSet::new();
    for lock in locks {
        all_names.insert(lock.package.as_str());
    }
    for selection in &selected_vec {
        all_names.insert(selection.package.as_str());
    }

    let mut changes = Vec::new();
    for name in all_names {
        let before = lock_map.get(name).copied();
        let after = chosen.get(name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let package = catalog_by_name[name];
                let exists = package.releases.iter().any(|r| r.version == old);
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let final_constraints = final_constraints(&chosen, &root_requirements, &catalog_by_name);
                    if final_constraints
                        .get(name)
                        .map(|cs| !satisfies(old, cs))
                        .unwrap_or(false)
                    {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name.to_string(),
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}

fn final_constraints(
    selected: &std::collections::BTreeMap<String, Version>,
    roots: &std::collections::BTreeMap<String, Vec<Constraint>>,
    catalog: &std::collections::BTreeMap<&str, &Package>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = roots.clone();
    for (name, version) in selected {
        if let Some(package) = catalog.get(name.as_str()) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical_constraints(std::mem::take(constraints));
    }
    active
}
