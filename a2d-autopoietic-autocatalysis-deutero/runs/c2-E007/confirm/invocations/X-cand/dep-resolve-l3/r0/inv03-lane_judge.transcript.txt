fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    (1..=24).contains(&b.len())
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn matches(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| match *c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    })
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

    for b in s.bytes() {

type Active = std::collections::BTreeMap<String, Vec<Constraint>>;
type Assigned = std::collections::BTreeMap<String, Version>;

struct Search<'a> {
    packages: &'a std::collections::BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a std::collections::BTreeMap<String, Version>,
    catalog_names: &'a [String],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, Assigned, Active)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn active(&self, assigned: &Assigned) -> Active {
        let mut active = Active::new();
        for req in self.roots {
            active
                .entry(req.package.clone())
                .or_default()
                .push(req.constraint);
        }
        for (name, version) in assigned {
            if let Some(package) = self.packages.get(name) {
                if let Some(release) = package
                    .releases
                    .iter()
                    .find(|release| release.version == *version)
                {
                    for req in &release.deps {
                        active
                            .entry(req.package.clone())
                            .or_default()
                            .push(req.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }
        active
    }

    fn node_error(&self, assigned: &Assigned, active: &Active) -> Option<ResolveError> {
        if let Some(name) = active.keys().find(|name| !self.packages.contains_key(*name)) {
            return Some(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assigned {
            if let Some(cs) = active.get(name) {
                if !matches(*version, cs) {
                    return Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for (name, cs) in active {
            if assigned.contains_key(name) {
                continue;
            }
            let package = self.packages[name];
            if !package.releases.iter().any(|r| matches(r.version, cs)) {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.clone(),
                });
            }
        }
        None
    }

    fn record_solution(&mut self, assigned: &Assigned, active: &Active) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| assigned.get(*name) != Some(*version))
            .count();
        let vector = self
            .catalog_names
            .iter()
            .map(|name| assigned.get(name).copied())
            .collect::<Vec<_>>();

        let better = match &self.best {
            None => true,
            Some((best_cost, best_vector, _, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if better {
            self.best = Some((cost, vector, assigned.clone(), active.clone()));
        }
    }

    fn visit(&mut self, assigned: &mut Assigned) -> Result<(), ResolveError> {
        let active = self.active(assigned);
        if let Some(error) = self.node_error(assigned, &active) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active.keys().find(|name| !assigned.contains_key(*name)).cloned();
        let Some(name) = next else {
            self.record_solution(assigned, &active);
            return Ok(());
        };

        let cs = &active[&name];
        let package = self.packages[&name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches(*v, cs))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(&name) {
            if let Some(i) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), version);
            let result = self.visit(assigned);
            assigned.remove(&name);
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, req) in roots.iter().enumerate() {
        if !valid_name(&req.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: req.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, req) in release.deps.iter().enumerate() {
                if !valid_name(&req.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: req.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_map = std::collections::BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_map: std::collections::BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut search = Search {
        packages: &package_map,
        roots,
        locks: &lock_map,
        catalog_names: &catalog_names,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut Assigned::new())?;

    let Some((_, _, selected_map, final_active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut all_names = std::collections::BTreeSet::new();
    all_names.extend(lock_map.keys().cloned());
    all_names.extend(selected_map.keys().cloned());

    let mut changes = Vec::new();
    for name in all_names {
        let before = lock_map.get(&name).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = package_map
                    .get(&name)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&name)
                    .is_some_and(|cs| !matches(old, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
