#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn validate_and_split(text: &str, side: Side) -> Result<Vec<&str>, DiffError> {
    if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn { side, byte });
    }

    if !text.is_empty() && !text.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side });
    }

    if text.is_empty() {
        return Ok(Vec::new());
    }

    Ok(text[..text.len() - 1].split('\n').collect())
}

fn build_script(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let m = old.len();
    let n = new.len();
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::with_capacity(m + n);
    let (mut i, mut j) = (0, 0);

    while i < m && j < n {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_lcs = lcs[i + 1][j];
            let insert_lcs = lcs[i][j + 1];
            if delete_lcs > insert_lcs || (delete_lcs == insert_lcs && (i + j) % 2 == 0) {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].to_owned(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].to_owned(),
                });
                j += 1;
            }
        }
    }

    while i < m {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].to_owned(),
        });
        i += 1;
    }

    while j < n {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].to_owned(),
        });
        j += 1;
    }

    edits
}

fn is_change(edit: &Edit) -> bool {
    matches!(edit, Edit::Delete { .. } | Edit::Insert { .. })
}

fn keep_count(edits: &[Edit], start: usize, end: usize) -> usize {
    edits[start..end]
        .iter()
        .filter(|edit| matches!(edit, Edit::Keep { .. }))
        .count()
}

fn emit_hunk(output: &mut String, edits: &[Edit], start: usize, end: usize) {
    let old_count = edits[start..end]
        .iter()
        .filter(|edit| !matches!(edit, Edit::Insert { .. }))
        .count();
    let new_count = edits[start..end]
        .iter()
        .filter(|edit| !matches!(edit, Edit::Delete { .. }))
        .count();

    let old_before = edits[..start]
        .iter()
        .filter(|edit| !matches!(edit, Edit::Insert { .. }))
        .count();
    let new_before = edits[..start]
        .iter()
        .filter(|edit| !matches!(edit, Edit::Delete { .. }))
        .count();

    let old_start = if old_count > 0 {
        old_before + 1
    } else {
        old_before
    };
    let new_start = if new_count > 0 {
        new_before + 1
    } else {
        new_before
    };

    output.push_str(&format!(
        "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
    ));

    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { line, .. } => {
                output.push(' ');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Delete { line, .. } => {
                output.push('-');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Insert { line, .. } => {
                output.push('+');
                output.push_str(line);
                output.push('\n');
            }
        }
    }
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }

    let old_lines = validate_and_split(old, Side::Old)?;
    let new_lines = validate_and_split(new, Side::New)?;

    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let edits = build_script(&old_lines, &new_lines);
    if !edits.iter().any(is_change) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut pos = 0;
    while pos < edits.len() {
        if is_change(&edits[pos]) {
            let start = pos;
            while pos < edits.len() && is_change(&edits[pos]) {
                pos += 1;
            }
            groups.push((start, pos));
        } else {
            pos += 1;
        }
    }

    let merge_limit = 2 * context;
    let mut merged: Vec<(usize, usize)> = Vec::new();
    for group in groups {
        if let Some(last) = merged.last_mut() {
            if keep_count(&edits, last.1, group.0) <= merge_limit {
                last.1 = group.1;
                continue;
            }
        }
        merged.push(group);
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (group_start, group_end) in merged {
        let mut start = group_start;
        let mut preceding = 0;
        while start > 0 && preceding < context {
            start -= 1;
            if matches!(edits[start], Edit::Keep { .. }) {
                preceding += 1;
            }
        }

        let mut end = group_end;
        let mut following = 0;
        while end < edits.len() && following < context {
            if matches!(edits[end], Edit::Keep { .. }) {
                following += 1;
            }
            end += 1;
        }

        emit_hunk(&mut output, &edits, start, end);
    }

    Ok(output)
}
