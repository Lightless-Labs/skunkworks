    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {

        let mut rank = Vec::with_capacity(segments.len() + 1);

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let segments = parse_path(head)?;

        let mut best: Option<(usize, Vec<u8>, Vec<(String, String)>)> = None;

                    .as_ref()

                    .unwrap_or(true);
                if replace {
                    best = Some((route_id, route.rank.clone(), params));
                }
            }
        }

        match best {
            Some((route, _, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let body = &input[1..];
    let parts: Vec<(usize, &str)> = body
        .split('/')
        .scan(1usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .collect();

    let mut end = parts.len();
    let trailing = parts.last().map(|(_, s)| s.is_empty()).unwrap_or(false);
    if trailing {
        end -= 1;
    }

    let mut result = Vec::with_capacity(end);
    let mut names = std::collections::HashSet::new();

    for &(start, part) in &parts[..end] {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: start });
        }

        if part.starts_with(':') || part.starts_with('*') {
            let wildcard = part.starts_with('*');
            let name = &part[1..];

            if wildcard && end > 0 && result.len() + 1 != end {
                return Err(Error::InvalidPattern { at: start });
            }
            if !valid_name(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }

            result.push(if wildcard {
                Segment::Wildcard(name.to_string())
            } else {
                Segment::Param(name.to_string())
            });
        } else {
            let decoded = decode_segment(part, start, ErrorKind::Pattern)?;
            result.push(Segment::Static(decoded));
        }
    }

    if trailing {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }

    Ok(result)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let body = &input[1..];
    let parts: Vec<(usize, &str)> = body
        .split('/')
        .scan(1usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .collect();

    let mut result = Vec::with_capacity(parts.len());
    for &(start, part) in &parts {
        if part.is_empty() {
            return Err(Error::InvalidPath { at: start });
        }
        result.push(decode_segment(part, start, ErrorKind::Path)?);
    }

    Ok(result)
}

#[derive(Clone, Copy)]
enum ErrorKind {
    Pattern,
    Path,
}

fn invalid(kind: ErrorKind, at: usize) -> Error {
    match kind {
        ErrorKind::Pattern => Error::InvalidPattern { at },
        ErrorKind::Path => Error::InvalidPath { at },
    }
}

fn decode_segment(segment: &str, base: usize, kind: ErrorKind) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(kind, base + i));
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            let value = match (hi, lo) {
                (Some(hi), Some(lo)) => (hi << 4) | lo,
                _ => return Err(invalid(kind, base + i)),
            };
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(kind, base + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid(kind, base + i));
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segments are ASCII"))
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = path.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, has_wildcard) = parse_pattern(pattern)?;
        let id = self.routes.len();
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let parsed = parse_path_head(path, head_end)?;
        let parts = match parsed {
            ParsedPath::Root => Vec::new(),
            ParsedPath::Segments(parts) => parts,
        };

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(route, &parts) {
                let replace = match &best {
                    None => true,
                    Some((current, _)) => route.rank > current.rank,
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {

    }

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(end);

        if segment_end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..segment_end];
        let first = bytes[start];
        if first == b':' || first == b'*' {
            let name_bytes = &bytes[start + 1..segment_end];
            let valid = !name_bytes.is_empty()
                && name_bytes[0].is_ascii_lowercase()
                && name_bytes[1..]
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid {
                return Err(Error::InvalidPattern { at: start });
            }
            let name = raw[1..].to_owned();
            if !names.insert(name.clone()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' {
                if segment_end != end {
                    return Err(Error::InvalidPattern { at: start });
                }
                segments.push(Segment::Wildcard(name));
            } else {
                segments.push(Segment::Param(name));
            }
        } else {
            let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }

        if segment_end == end {
            break;
        }
        start = segment_end + 1;
        if start == end {
            break;
        }
    }

    if trailing {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }

    Ok((
        segments.clone(),
        segments.iter().any(|s| matches!(s, Segment::Wildcard(_))),
    ))
}

fn parse_path_head(path: &str, end: usize) -> Result<ParsedPath, Error> {
    let bytes = path.as_bytes();
    if end == 0 || bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if end == 1 {
        return Ok(ParsedPath::Root);
    }

    let trailing = bytes[end - 1] == b'/';
    let segment_end = if trailing { end - 1 } else { end };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < segment_end {
        let stop = bytes[start..segment_end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(segment_end);

        if stop == start {
            return Err(Error::InvalidPath { at: start });
        }

        let raw = &path[start..stop];
        let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);

        if stop == segment_end {
            break;
        }
        start = stop + 1;
    }

    if trailing {
        return Err(Error::InvalidPath { at: end - 1 });
    }

    Ok(ParsedPath::Segments(segments))
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;

            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)

                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}
