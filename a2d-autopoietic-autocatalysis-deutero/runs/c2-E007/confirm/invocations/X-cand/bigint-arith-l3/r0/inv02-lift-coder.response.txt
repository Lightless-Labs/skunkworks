use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] with negative=false.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let magnitude = &bytes[start..];

        // Invalid bytes take precedence over all separator errors.
        if let Some(i) = magnitude
            .iter()
            .enumerate()
            .find_map(|(i, &b)| {
                if b != b'_' && !b.is_ascii_digit() {
                    Some(i)
                } else {
                    None
                }
            })
        {
            return Err(Error::InvalidDigit { index: start + i });
        }

        // Validate separators after the full invalid-byte scan.
        for (i, &b) in magnitude.iter().enumerate() {
            if b == b'_'
                && (i == 0
                    || i + 1 == magnitude.len()
                    || !magnitude[i - 1].is_ascii_digit()
                    || !magnitude[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: start + i });
            }
        }

        let digit_count = magnitude.iter().filter(|&&b| b != b'_').count();
        let first_digit_index = magnitude.iter().position(|&b| b != b'_').unwrap();

        if digit_count > 1 && magnitude[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: start + first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess = max_digits;
            let index = magnitude
                .iter()
                .enumerate()
                .filter(|(_, b)| **b != b'_')
                .nth(excess)
                .map(|(i, _)| start + i)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits: Vec<u8> = magnitude
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| b - b'0')
            .rev()
            .collect();

        if digits.is_empty() {
            digits.push(0);
        }
        let mut value = Self { negative, digits };
        value.normalize();
        Ok(value)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(min_digits).max(1);
        let mut magnitude = String::with_capacity(width + width / group.max(1));
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                magnitude.push('_');
            }
            let digit = if i < width - self.digits.len() {
                0
            } else {
                self.digits[self.digits.len() - 1 - (i - (width - self.digits.len()))]
            };
            magnitude.push((b'0' + digit) as char);
        }

        if self.negative {
            let mut result = String::with_capacity(magnitude.len() + 1);
            result.push('-');
            result.push_str(&magnitude);
            result
        } else {
            magnitude
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (a, b) if a != b => a.cmp(&b),
            (0, 0) => Ordering::Equal,
            (1, 1) => cmp_magnitude(&self.digits, &rhs.digits),
            (-1, -1) => cmp_magnitude(&rhs.digits, &self.digits),
            _ => Ordering::Equal,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = if self.negative == rhs.negative {
            Self {
                negative: self.negative,
                digits: add_magnitude(&self.digits, &rhs.digits),
            }
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => Self {
                    negative: self.negative,
                    digits: sub_magnitude(&self.digits, &rhs.digits),
                },
                Ordering::Less => Self {
                    negative: rhs.negative,
                    digits: sub_magnitude(&rhs.digits, &self.digits),
                },
                Ordering::Equal => Self::zero(),
            }
        };
        result.normalize();
        result
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            digits: mul_magnitude(&self.digits, &rhs.digits),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient, remainder) = div_rem_magnitude(&self.digits, &rhs.digits);
        let mut q = Self {
            negative: self.negative ^ rhs.negative,
            digits: quotient,
        };
        let mut r = Self {
            negative: self.negative,
            digits: remainder,
        };
        q.normalize();
        r.normalize();
        Ok((q, r))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for (&x, &y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(&y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry > 0 {
        result.push(carry);
    }
    result
}

// Requires a >= b.
fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(digit as u8);
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = vec![0u8; a.len() + b.len()];
    for (i, &x) in a.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &y) in b.iter().enumerate() {
            let index = i + j;
            let product = result[index] as u16 + x as u16 * y as u16 + carry;
            result[index] = (product % 10) as u8;
            carry = product / 10;
        }
        let mut index = i + b.len();
        while carry > 0 {
            let total = result[index] as u16 + carry;
            result[index] = (total % 10) as u8;
            carry = total / 10;
            index += 1;
            if index == result.len() && carry > 0 {
                result.push(0);
            }
        }
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

// Returns nonnegative quotient and remainder in little-endian decimal digits.
fn div_rem_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_magnitude(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0u8];

    for &digit in dividend.iter().rev() {
        remainder.insert(0, digit);
        trim_magnitude(&mut remainder);

        let mut qdigit = 0u8;
        while cmp_magnitude(&remainder, divisor) != Ordering::Less {
            remainder = sub_magnitude(&remainder, divisor);
            qdigit += 1;
        }
        quotient_be.push(qdigit);
    }

    while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
        quotient_be.remove(0);
    }
    quotient_be.reverse();
    trim_magnitude(&mut remainder);
    (quotient_be, remainder)
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
}
