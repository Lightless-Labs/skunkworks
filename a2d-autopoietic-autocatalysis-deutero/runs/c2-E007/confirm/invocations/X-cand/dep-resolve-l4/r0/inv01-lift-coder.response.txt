use std::collections::{BTreeMap, BTreeSet};

    pub patch: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateFeature { package: String, version: Version, feature: String },
    DuplicateLock { package: String },
    Missing { package: String, evidence: Vec<Evidence> },
    NoMatch { package: String, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    Conflict { package: String, selected: Version, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    MissingFeature { package: String, selected: Version, feature: String, path: Vec<String> },
    Limit { attempts: usize },
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(v: &[Constraint]) -> Vec<Constraint> {
    let mut out = v.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(e: &[Evidence]) -> Vec<Evidence> {
    let mut out = e.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature { package, version, feature } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::NoMatch { package, requirements, evidence } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!("no-match|package={}|constraints={}", encode(package), cs);
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::Conflict { package, selected, requirements, evidence } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                cs
            );
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::MissingFeature { package, selected, feature, path } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
}

fn token_features(features: &[String]) -> String {
    let mut fs = features.to_vec();
    fs.sort();
    fs.dedup();
    fs.join(",")
}

fn root_token(r: &Requirement, i: usize) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        token_features(&r.features)
    )
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn target_matches(p: Platform, t: Target) -> bool {
    p == Platform::Any
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn dep_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        token_features(&dep.features),
        dep.gate.as_deref().unwrap_or("*"),
        platform_text(dep.platform)
    )
}

fn choose_path(paths: &mut BTreeMap<String, Vec<String>>, package: &str, path: Vec<String>) -> bool {
    match paths.get(package) {
        None => {
            paths.insert(package.to_string(), path);
            true
        }
        Some(old) if (path.len(), &path) < (old.len(), old) => {
            paths.insert(package.to_string(), path);
            true
        }
        _ => false,
    }
}

fn compute_active(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let cmap: BTreeMap<&str, &Package> = catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut active = Active::default();
    let mut queue = Vec::<(String, Vec<String>)>::new();

    for (i, root) in roots.iter().enumerate() {
        let tok = root_token(root, i);
        active.origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: vec![tok.clone()],
        });
        active.requests.entry(root.package.clone()).or_default().extend(root.features.iter().cloned());
        choose_path(&mut active.paths, &root.package, vec![tok.clone()]);
        queue.push((root.package.clone(), vec![tok]));
    }

    let mut processed_edges = BTreeSet::<(String, usize, Vec<String>)>::new();
    loop {
        let mut changed = false;

        let selected_names: Vec<String> = selected.keys().cloned().collect();
        for name in selected_names {
            let Some(package) = cmap.get(name.as_str()) else { continue };
            let version = selected[&name];
            let Some(release) = package.releases.iter().find(|r| r.version == version) else { continue };
            let reqs = active.requests.get(&name).cloned().unwrap_or_default();
            let activated = active.activated.entry(name.clone()).or_default();
            let mut feat_queue: Vec<String> = reqs.into_iter().collect();
            while let Some(fname) = feat_queue.pop() {
                if !activated.insert(fname.clone()) {
                    continue;
                }
                changed = true;
                if let Some(f) = release.features.iter().find(|f| f.name == fname) {
                    for enabled in &f.enables {
                        active.requests.entry(name.clone()).or_default().insert(enabled.clone());
                        feat_queue.push(enabled.clone());
                    }
                }
            }
        }

        for (name, version) in selected {
            let Some(package) = cmap.get(name.as_str()) else { continue };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else { continue };
            let source_path = active.paths.get(name).cloned().unwrap_or_default();
            let enabled = active.activated.get(name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !enabled.contains(g))
                {
                    continue;
                }
                let token = dep_token(name, *version, j, dep);
                let key = (name.clone(), j, source_path.clone());
                if !processed_edges.insert(key) {
                    continue;
                }
                let mut path = source_path.clone();
                path.push(token);
                active.origins.entry(dep.package.clone()).or_default().push(Origin {
                    constraint: dep.constraint,
                    path: path.clone(),
                });
                active.requests.entry(dep.package.clone()).or_default().extend(dep.features.iter().cloned());
                if choose_path(&mut active.paths, &dep.package, path.clone()) {
                    queue.push((dep.package.clone(), path));
                }
                changed = true;
            }
        }

        while let Some((name, path)) = queue.pop() {
            if choose_path(&mut active.paths, &name, path) {
                changed = true;
            }
        }
        if !changed {
            break;
        }
    }
    active
}

fn evidence_for(active: &Active, package: &str) -> Vec<Evidence> {
    active.origins.get(package).into_iter().flatten()
        .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() })
        .collect()
}

fn constraints_for(active: &Active, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        &active.origins.get(package).into_iter().flatten().map(|o| o.constraint).collect::<Vec<_>>()
    )
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]", i), name: r.package.clone() });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName { site: format!("root[{}].feature[{}]", i, f), name: name.clone() });
            }
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]", i), name: p.name.clone() });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName { site: base.clone(), name: d.package.clone() });
                }
                if let Some(g) = &d.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName { site: format!("{}.gate", base), name: g.clone() });
                    }
                }
                for (f, name) in d.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("{}.feature[{}]", base, f), name: name.clone() });
                    }
                }
            }
            for (f, feature) in r.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName { site: base.clone(), name: feature.name.clone() });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("{}.enable[{}]", base, e), name: name.clone() });
                    }
                }
            }
        }
    }

    let mut seen = BTreeSet::new();
    for p in catalog {
        if !seen.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature { package: p.name.clone(), version: r.version, feature: f.name.clone() });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{}]", i), name: lock.package.clone() });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    budget: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>, Active)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = compute_active(self.catalog, self.roots, selected, self.target);
        let cmap: BTreeMap<&str, &Package> = self.catalog.iter().map(|p| (p.name.as_str(), p)).collect();

        let missing: Vec<String> = active.origins.keys().filter(|p| !cmap.contains_key(p.as_str())).cloned().collect();
        if let Some(package) = missing.first() {
            let err = ResolveError::Missing { package: package.clone(), evidence: evidence_for(&active, package) };
            if self.first_error.is_none() { self.first_error = Some(err); }
            return Ok(());
        }

        for (package, version) in selected.iter() {
            let constraints = constraints_for(&active, package);
            if constraints.iter().any(|c| !matches(*version, *c)) {
                let err = ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: evidence_for(&active, package),
                };
                if self.first_error.is_none() { self.first_error = Some(err); }
                return Ok(());
            }
        }

        for package in active.origins.keys() {
            if selected.contains_key(package) { continue; }
            let p = cmap[package.as_str()];
            let constraints = constraints_for(&active, package);
            if !p.releases.iter().any(|r| constraints.iter().all(|c| matches(r.version, *c))) {
                let err = ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints,
                    evidence: evidence_for(&active, package),
                };
                if self.first_error.is_none() { self.first_error = Some(err); }
                return Ok(());
            }
        }

        for package in selected.keys() {
            let version = selected[package];
            let activated = active.activated.get(package).cloned().unwrap_or_default();
            let p = cmap[package.as_str()];
            let release = p.releases.iter().find(|r| r.version == version).unwrap();
            for feature in activated {
                if !release.features.iter().any(|f| f.name == feature) {
                    let err = ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: version,
                        feature,
                        path: active.paths.get(package).cloned().unwrap_or_default(),
                    };
                    if self.first_error.is_none() { self.first_error = Some(err); }
                    return Ok(());
                }
            }
        }

        let next = active.origins.keys().find(|p| !selected.contains_key(*p)).cloned();
        if let Some(package) = next {
            let p = cmap[package.as_str()];
            let constraints = constraints_for(&active, &package);
            let mut candidates: Vec<Version> = p.releases.iter()
                .map(|r| r.version)
                .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
                .collect();
            candidates.sort_by(|a, b| b.cmp(a));
            if let Some(lock) = self.locks.get(&package) {
                if let Some(pos) = candidates.iter().position(|v| v == lock) {
                    let v = candidates.remove(pos);
                    candidates.insert(0, v);
                }
            }
            let mut had_error = false;
            for candidate in candidates {
                if self.attempts == self.budget {
                    return Err(ResolveError::Limit { attempts: self.budget });
                }
                self.attempts += 1;
                selected.insert(package.clone(), candidate);
                self.visit(selected)?;
                selected.remove(&package);
                had_error = true;
            }
            let _ = had_error;
            return Ok(());
        }

        let selections: Vec<Selection> = selected.iter().map(|(p, v)| Selection { package: p.clone(), version: *v }).collect();
        let cost = self.locks.iter().filter(|(p, v)| selected.get(*p) != Some(*v)).count();
        let replace = match &self.best {
            None => true,
            Some((best_cost, best_sel, _)) => {
                if cost != *best_cost {
                    cost < *best_cost
                } else {
                    let names: BTreeSet<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
                    let mut better = false;
                    for name in names {
                        let a = selections.iter().find(|s| s.package == name).map(|s| s.version);
                        let b = best_sel.iter().find(|s| s.package == name).map(|s| s.version);
                        if a != b {
                            better = match (a, b) {
                                (None, Some(_)) => false,
                                (Some(_), None) => true,
                                (Some(x), Some(y)) => x > y,
                                _ => false,
                            };
                            break;
                        }
                    }
                    better
                }
            }
        };
        if replace {
            self.best = Some((cost, selections, active));
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let mut search = Search {
        catalog,
        roots,
        locks: locks.iter().map(|l| (l.package.clone(), l.version)).collect(),
        target,
        budget: max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;
    let Some((_, selected, active)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: vec![],
            evidence: vec![],
        }));
    };

    let selected_map: BTreeMap<String, Version> = selected.iter().map(|s| (s.package.clone(), s.version)).collect();
    let mut packages: BTreeSet<String> = search.locks.keys().cloned().collect();
    packages.extend(selected_map.keys().cloned());
    let cmap: BTreeMap<&str, &Package> = catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut changes = Vec::new();
    for package in packages {
        let before = search.locks.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after { continue; }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = cmap.get(package.as_str()).is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if constraints_for(&active, &package).iter().any(|c| !matches(old, *c)) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            _ => continue,
        };
        changes.push(Change { package, before, after, reason });
    }
    Ok(Resolution { selected, changes })
}
