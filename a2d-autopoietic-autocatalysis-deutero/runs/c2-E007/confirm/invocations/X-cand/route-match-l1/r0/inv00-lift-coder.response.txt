#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let route_id = self.routes.len();
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
            })
            .collect();

        self.routes.push(Route { segments, rank });
        Ok(route_id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .bytes()
            .position(|byte| byte == b'?' || byte == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let segments = parse_path(head)?;

        let mut best: Option<(usize, Vec<u8>, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_segment, path_segment) in route.segments.iter().zip(&segments) {
                match pattern_segment {
                    Segment::Static(value) if value == path_segment => {}
                    Segment::Static(_) => {
                        matches = false;
                        break;
                    }
                    Segment::Param(name) => params.push((name.clone(), path_segment.clone())),
                }
            }

            if !matches {
                continue;
            }

            let replace = match &best {
                None => true,
                Some((_, best_rank, _)) => route.rank > *best_rank,
            };

            if replace {
                best = Some((route_id, route.rank.clone(), params));
            }
        }

        match best {
            Some((route, _, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    for (index, part) in pattern[1..].split('/').enumerate() {
        let segment_start = start;
        let is_last = index == pattern[1..].split('/').count() - 1;

        if part.is_empty() {
            if is_last && segment_start == pattern.len() {
                return Err(Error::InvalidPattern {
                    at: pattern.len() - 1,
                });
            }
            return Err(Error::InvalidPattern { at: segment_start });
        }

        if part.starts_with(':') {
            let name = &part[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: segment_start });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if part.starts_with('*') {
            return Err(Error::InvalidPattern { at: segment_start });
        } else {
            segments.push(Segment::Static(
                decode_segment(part, segment_start).map_err(|at| Error::InvalidPattern { at })?,
            ));
        }

        start += part.len() + 1;
    }

    if pattern.ends_with('/') {
        return Err(Error::InvalidPattern {
            at: pattern.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let mut segments = Vec::new();
    let mut start = 1;

    for (index, part) in path[1..].split('/').enumerate() {
        let segment_start = start;
        let is_last = index == path[1..].split('/').count() - 1;

        if part.is_empty() {
            if is_last && segment_start == path.len() {
                return Err(Error::InvalidPath { at: path.len() - 1 });
            }
            return Err(Error::InvalidPath { at: segment_start });
        }

        segments.push(
            decode_segment(part, segment_start).map_err(|at| Error::InvalidPath { at })?,
        );
        start += part.len() + 1;
    }

    if path.ends_with('/') {
        return Err(Error::InvalidPath {
            at: path.len() - 1,
        });
    }

    Ok(segments)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' {
            if index + 2 >= bytes.len() {
                return Err(base + index);
            }
            let high = uppercase_hex(bytes[index + 1]).ok_or(base + index)?;
            let low = uppercase_hex(bytes[index + 2]).ok_or(base + index)?;
            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + index);
            }
            decoded.push(value);
            index += 3;
        } else {
            if !byte.is_ascii() || !is_unreserved(byte) {
                return Err(base + index);
            }
            decoded.push(byte);
            index += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segments are ASCII"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}
