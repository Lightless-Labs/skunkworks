pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {

    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|item| item == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|item| item == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut quoted = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;
    let mut field_started = false;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if !quoted && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        if quoted {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote as char);
                    pos += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    pos += 1;
                }
            } else {
                let ch = input[pos..].chars().next().expect("valid UTF-8");
                field.push(ch);
                pos += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                pos += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_index += 1;
                after_quote = false;
                field_started = false;
                pos += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: pos,
            });
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            field_started = false;
            pos += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            field_started = false;
            pos += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
            quoted = true;
            quote_start = pos;
            field_started = true;
            pos += 1;
        } else {
            let ch = input[pos..].chars().next().expect("valid UTF-8");
            field.push(ch);
            field_started = true;
            pos += ch.len_utf8();
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if after_quote || field_started || !row.is_empty() || !field.is_empty() {
        row.push(field);
        records.push(row);
    } else if !records.is_empty() {
        // A terminating newline already emitted its record.
    } else if bytes.len() > 0 {
        // Nonempty input can end with an empty field (for example ",").
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(existing, _)| *existing == name) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }
    if records.is_empty() {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    }

    let header = records[0].clone();
    if config.duplicates == DuplicatePolicy::Reject {
        if let Some(error) = duplicate_header(&header) {
            return Err(error);
        }
    }

    for (i, row) in records.iter().enumerate().skip(1) {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: i,
                expected: header.len(),
                actual: row.len(),
            });
        }
    }

    Ok(Table {
        header: Some(header),
        rows: records.into_iter().skip(1).collect(),
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start])
        && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    if b.len() < start + 4 {
        return false;
    }
    let dot = b.len() - 3;
    if b[dot] != b'.'
        || !b[dot + 1..].iter().all(u8::is_ascii_digit)
        || b[dot + 1..].len() != 2
    {
        return false;
    }
    let integral = &b[start..dot];
    if integral.is_empty() {
        return false;
    }
    if integral == b"0" {
        return !s.starts_with("-0.");
    }
    (b'1'..=b'9').contains(&integral[0])
        && integral[1..].iter().all(u8::is_ascii_digit)
}

fn parse_cell(s: &str, kind: ColumnType) -> Result<Value, CellErrorKind> {
    match kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) {
                return Err(CellErrorKind::InvalidInteger);
            }
            s.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = s.starts_with('-');
            let magnitude = if negative { &s[1..] } else { s };
            let (whole, fraction) = magnitude.split_once('.').expect("validated decimal");
            let whole_value = whole
                .parse::<i128>()
                .map_err(|_| CellErrorKind::DecimalOverflow)?;
            let fraction_value = fraction.parse::<i128>().expect("two digits");
            let total = whole_value
                .checked_mul(100)
                .and_then(|v| v.checked_add(fraction_value))
                .ok_or(CellErrorKind::DecimalOverflow)?;
            let signed = if negative { -total } else { total };
            i64::try_from(signed)
                .map(Value::Decimal)
                .map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };
    let mut stopped = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stopped {
            break;
        }
        let mut output = vec![None; columns.len()];
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                break;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, cell) = match resolved {
                None => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    });
                    row_failed = true;
                    if policy == ErrorPolicy::Stop {
                        stopped = true;
                    }
                    continue;
                }
                Some(index) => match row.get(index) {
                    None => {
                        result.errors.push(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(index),
                            kind: CellErrorKind::MissingCell,
                        });
                        row_failed = true;
                        if policy == ErrorPolicy::Stop {
                            stopped = true;
                        }
                        continue;
                    }
                    Some(cell) => (index, cell),
                },
            };

            let _ = column;
            if cell.is_empty() {
                if spec.nullable {
                    output[spec_index] = Some(Value::Null);
                    continue;
                }
                if spec.kind != ColumnType::Text {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column),
                        kind: CellErrorKind::Empty,
                    });
                    row_failed = true;
                    if policy == ErrorPolicy::Stop {
                        stopped = true;
                    }
                    continue;
                }
            }

            match parse_cell(cell, spec.kind) {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column),
                        kind,
                    });
                    row_failed = true;
                    if policy == ErrorPolicy::Stop {
                        stopped = true;
                    }
                }
            }
        }

        result.rows.push(output);
        if row_failed && policy == ErrorPolicy::FirstPerRow {
            continue;
        }
    }

    result
}

fn first_zero(s: &str) -> Option<usize> {
    s.as_bytes().iter().position(|b| *b == 0)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

fn encode_field(
    out: &mut String,
    value: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) {
    let bytes = value.as_bytes();
    let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let has_special = bytes
        .iter()
        .any(|b| *b == delimiter || *b == quote || *b == b'\r' || *b == b'\n');
    let bom = first_field_first_record && value.starts_with('\u{feff}');
    let needs_quote = edge_space || has_special || bom || sole_empty;

    if !needs_quote {
        out.push_str(value);
        return;
    }
    out.push(quote as char);
    for ch in value.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    original_records.extend(table.rows.iter().map(Vec::as_slice));

    for (record, row) in original_records.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            if let Some(byte) = first_zero(cell) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(CsvError::DuplicateHeader {
            name,
            first,
            second,
        }) = table.header.as_ref().and_then(|h| duplicate_header(h))
        {
            return Err(WriteError::DuplicateHeader {
                name,
                first,
                second,
            });
        }
    }

    let mut output_header: Option<Vec<String>> = None;
    let mut output_rows: Vec<Vec<String>> = Vec::new();

    if config.mode == WriteMode::Preserve {
        output_header = table.header.clone();
        output_rows = table.rows.clone();
    } else if let Some(header) = &table.header {
        let mut chosen: Vec<usize> = Vec::new();
        for (i, name) in header.iter().enumerate() {
            let prior = chosen.iter().any(|&j| header[j] == *name);
            match table.duplicates {
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                    if !prior {
                        chosen.push(i);
                    }
                }
                DuplicatePolicy::KeepLast => {
                    if !header.iter().skip(i + 1).any(|next| next == name) {
                        chosen.push(i);
                    }
                }
            }
        }
        chosen.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
        output_header = Some(chosen.iter().map(|&i| header[i].clone()).collect());
        output_rows = table
            .rows
            .iter()
            .map(|row| chosen.iter().map(|&i| row[i].clone()).collect())
            .collect();
        output_rows.sort_by(|a, b| compare_rows(a, b));
    } else {
        output_rows = table.rows.clone();
        output_rows.sort_by(|a, b| compare_rows(a, b));
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    if let Some(header) = output_header {
        records.push(header);
    }
    records.extend(output_rows);

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let terminator = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            out.push_str(terminator);
        }
        for (column_index, value) in row.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            encode_field(
                &mut out,
                value,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                row.len() == 1 && value.is_empty(),
            );
        }
    }

    if canonical || config.final_terminator {
        out.push_str(terminator);
    }
    Ok(out)
}
