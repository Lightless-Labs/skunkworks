#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<u8>,
    names: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    // Validate the entire template before looking up any bindings.
    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[cursor + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => cursor + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[cursor + 1..close]
                    .iter()
                    .position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                let operator = match bytes.get(content_start).copied() {
                    Some(op @ (b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')) => {
                        content_start += 1;
                        if op != b'+' && op != b'#' {
                            return Err(Error::UnsupportedOperator { pos: open + 1 });
                        }
                        Some(op)
                    }
                    _ => None,
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &bytes[spec_start..spec_end];
                    if !is_name_start(spec[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    if let Some(offset) = spec[1..].iter().position(|&b| !is_name_continue(b)) {
                        return Err(Error::InvalidName {
                            pos: spec_start + 1 + offset,
                        });
                    }

                    let name = template[spec_start..spec_end].to_owned();
                    if seen.iter().any(|existing| existing == &name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name.clone());
                    names.push(name);

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression { operator, names });
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    cursor = 0;

    while cursor < bytes.len() {
        if bytes[cursor] == b'{' {
            output.push_str(&template[literal_start..cursor]);

            let close = bytes[cursor + 1..]
                .iter()
                .position(|&b| b == b'}')
                .map(|offset| cursor + 1 + offset)
                .expect("template was validated");

            // Each opening brace corresponds to the next validated expression.
            let expression_index = expressions
                .iter()
                .position(|_| true)
                .expect("validated expression exists");
            let _ = expression_index;

            let content_start = cursor + 1;
            let operator = match bytes[content_start] {
                b'+' | b'#' => Some(bytes[content_start]),
                _ => None,
            };
            let names_start = content_start + usize::from(operator.is_some());
            let mut names = Vec::new();
            let mut spec_start = names_start;
            loop {
                let spec_end = bytes[spec_start..close]
                    .iter()
                    .position(|&b| b == b',')
                    .map(|offset| spec_start + offset)
                    .unwrap_or(close);
                names.push(&template[spec_start..spec_end]);
                if spec_end == close {
                    break;
                }
                spec_start = spec_end + 1;
            }

            let mut fields = Vec::new();
            for name in names {
                if let Some(value) = vars
                    .iter()
                    .rev()
                    .find(|(binding_name, _)| *binding_name == name)
                    .map(|(_, value)| *value)
                {
                    fields.push(encode(value, operator.is_some()));
                }
            }

            if !fields.is_empty() {
                if operator == Some(b'#') {
                    output.push('#');
                }
                for (index, field) in fields.iter().enumerate() {
                    if index > 0 {
                        output.push(',');
                    }
                    output.push_str(field);
                }
            }

            cursor = close + 1;
            literal_start = cursor;
        } else {
            cursor += 1;
        }
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn encode(value: &str, reserved_allowed: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved_allowed && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
