pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<(String, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn validate_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let content_start = open + 1;

    if let Some(relative) = bytes[content_start..close]
        .iter()
        .position(|&byte| byte == b'{')
    {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let mut spec_start = content_start;
    let operator = if spec_start < close
        && matches!(bytes[spec_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let byte = bytes[spec_start];
        let pos = spec_start;
        spec_start += 1;

        match byte {
            b'+' => Some('+'),
            b'#' => Some('#'),
            _ => return Err(Error::UnsupportedOperator { pos }),
        }
    } else {
        None
    };

    if spec_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut part_start = spec_start;

    loop {
        let part_end = bytes[part_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |relative| part_start + relative);

        if part_start == part_end {
            return Err(Error::EmptyVariable { pos: part_start });
        }

        let first = bytes[part_start];
        if !is_name_start(first) {
            return Err(Error::InvalidName { pos: part_start });
        }

        for pos in part_start + 1..part_end {
            if !is_name_byte(bytes[pos]) {
                return Err(Error::InvalidName { pos });
            }
        }

        let name = template[part_start..part_end].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: part_start });
        }
        names.push((name, part_start));

        if part_end == close {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        names,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[open + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(relative) => open + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                expressions.push(validate_expression(template, open, close)?);
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        let byte = bytes[cursor];

        if byte == b'%' && cursor + 2 < bytes.len() {
            let high = bytes[cursor + 1];
            let low = bytes[cursor + 2];
            if high.is_ascii_hexdigit() && low.is_ascii_hexdigit() {
                output.push('%');
                output.push((high as char).to_ascii_uppercase());
                output.push((low as char).to_ascii_uppercase());
                cursor += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved = allow_reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        cursor += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let allow_reserved = matches!(expression.operator, Some('+') | Some('#'));
        let mut fields = Vec::new();

        for (name, _) in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(encode_value(value, allow_reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

        return Err(Error::NestedOpen {
            pos: open + 1 + offset,
        });

            let op = bytes[content_start] as char;
            content_start += 1;
            if op != '+' && op != '#' {
                return Err(Error::UnsupportedOperator {
                    pos: content_start - 1,
                });
            }
            Some(op)
        }
        _ => None,
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map_or(close, |offset| spec_start + offset);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &bytes[spec_start..spec_end];
        if !is_name_start(spec[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(offset) = spec[1..].iter().position(|&b| !is_name_continue(b)) {
            return Err(Error::InvalidName {
                pos: spec_start + 1 + offset,
            });
        }

        let name = std::str::from_utf8(spec).expect("validated ASCII variable name");
        if !seen.insert(name.to_owned()) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        variables.push((spec_start, name.to_owned()));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        variables,
    })
}

    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

                expressions.push(validate_expression(template, i, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

    )

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + value - 10
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.extend_from_slice(&[b'%', hex_digit(high), hex_digit(low)]);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte);
        } else {
            output.extend_from_slice(&[b'%', hex_digit(byte >> 4), hex_digit(byte & 0x0f)]);
        }
        i += 1;
    }

    String::from_utf8(output).expect("percent encoding produces valid UTF-8")
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {

    let mut expressions = Vec::new();
    let mut cursor = 0;

    // Validate the entire template before looking up any bindings.
    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[cursor + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => cursor + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[cursor + 1..close]
                    .iter()
                    .position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                let operator = match bytes.get(content_start).copied() {
                    Some(op @ (b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')) => {
                        content_start += 1;
                        if op != b'+' && op != b'#' {
                            return Err(Error::UnsupportedOperator { pos: open + 1 });
                        }
                        Some(op)
                    }
                    _ => None,
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &bytes[spec_start..spec_end];
                    if !is_name_start(spec[0]) {
                        return Err(Error::InvalidName { pos: spec_start });

                    if let Some(offset) = spec[1..].iter().position(|&b| !is_name_continue(b)) {
                        return Err(Error::InvalidName {
                            pos: spec_start + 1 + offset,
                        });
                    }

                    let name = template[spec_start..spec_end].to_owned();
                    if seen.iter().any(|existing| existing == &name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name.clone());
                    names.push(name);

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression { operator, names });
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    cursor = 0;

            output.push_str(&template[literal_start..cursor]);

                .position(|&b| b == b'}')
                .map(|offset| cursor + 1 + offset)
                .expect("template was validated");

                .iter()
                .position(|_| true)
                .expect("validated expression exists");
            let _ = expression_index;

            let content_start = cursor + 1;
            let operator = match bytes[content_start] {
                b'+' | b'#' => Some(bytes[content_start]),
                _ => None,
            };
            let names_start = content_start + usize::from(operator.is_some());
            let mut names = Vec::new();
            let mut spec_start = names_start;
            loop {
                let spec_end = bytes[spec_start..close]
                    .iter()
                    .position(|&b| b == b',')
                    .map(|offset| spec_start + offset)
                    .unwrap_or(close);
                names.push(&template[spec_start..spec_end]);
                if spec_end == close {
                    break;
                }
                spec_start = spec_end + 1;
            }

            let mut fields = Vec::new();
            for name in names {
                if let Some(value) = vars
                    .iter()
                    .rev()
                    .find(|(binding_name, _)| *binding_name == name)

                    fields.push(encode(value, operator.is_some()));

                }

                    if index > 0 {
                        output.push(',');
                    }
                    output.push_str(field);
                }
            }

            cursor = close + 1;
            literal_start = cursor;
        } else {
            cursor += 1;
        }
    }

    output.push_str(&template[literal_start..]);
    Ok(output)

    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
