```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn components(path: &str) -> Vec<&str> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn normalize_from(path: &str, base: &[String]) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            base.to_vec()
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }
        Self::canonical(&parts)
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        Ok(Self::normalize_from(path, &Self::components(&self.cwd)))
    }

    fn parent(path: &str) -> String {
        let mut parts: Vec<String> = Self::components(path)
            .iter()
            .map(|s| (*s).to_string())
            .collect();
        parts.pop();
        Self::canonical(&parts)
    }

    fn basename(path: &str) -> String {
        Self::components(path)
            .last()
            .copied()
            .unwrap_or("")
            .to_string()
    }

    fn join_child(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            path.starts_with('/')
        } else {
            path == dir
                || path
                    .strip_prefix(dir)
                    .is_some_and(|suffix| suffix.starts_with('/'))
        }
    }

    fn children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| **path != dir && Self::parent(path) == dir)
            .cloned()
            .collect()
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        allow_missing_final: bool,
    ) -> Result<String, Error> {
        let mut pending: Vec<String> = Self::components(path)
            .iter()
            .map(|s| (*s).to_string())
            .collect();
        let mut physical: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            let candidate_parts = {
                let mut p = physical.clone();
                p.push(component);
                p
            };
            let candidate = Self::canonical(&candidate_parts);

            let current = Self::canonical(&physical);
            match self.entries.get(&current) {
                Some(Entry::Dir) => {}
                Some(_) => return Err(Error::NotDir(current)),
                None => return Err(Error::Missing(current)),
            }

            match self.entries.get(&candidate) {
                None => {
                    if allow_missing_final && pending.is_empty() {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
                Some(Entry::Link(target)) if follow_final || !pending.is_empty() => {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;

                    let base = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        physical.clone()
                    };
                    let expanded = Self::normalize_from(target, &base);
                    let mut expanded_parts: Vec<String> = Self::components(&expanded)
                        .iter()
                        .map(|s| (*s).to_string())
                        .collect();
                    expanded_parts.extend(pending.drain(..));
                    let expanded_path = Self::canonical(&expanded_parts);
                    let normalized = Self::normalize_from(&expanded_path, &[]);
                    pending = Self::components(&normalized)
                        .iter()
                        .map(|s| (*s).to_string())
                        .collect();
                    physical.clear();
                }
                Some(_) => physical.push(component),
            }
        }

        let resolved = Self::canonical(&physical);
        if !self.entries.contains_key(&resolved) {
            if allow_missing_final {
                return Ok(resolved);
            }
            return Err(Error::Missing(resolved));
        }
        Ok(resolved)
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, Self::basename(path))),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn insert_child(&mut self, parent: &str, name: &str, entry: Entry) {
        self.entries.insert(Self::join_child(parent, name), entry);
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&normalized)?;
        let destination = Self::join_child(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.insert_child(&parent, &name, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, true)?;
        if let Some(entry) = self.entries.get(&resolved) {
            if matches!(entry, Entry::Dir) {
                return Err(Error::IsDir(resolved));
            }
        }
        let parent = Self::parent(&resolved);
        let name = Self::basename(&resolved);
        let (physical_parent, _) = self.resolve_parent(&resolved)?;
        if parent != physical_parent {
            return Err(Error::NotDir(parent));
        }
        self.entries
            .insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut out = self
            .children(&resolved)
            .into_iter()
            .filter_map(|child| {
                self.entries
                    .get(&child)
                    .map(|entry| (Self::basename(&child), entry.kind()))
            })
            .collect::<Vec<_>>();
        out.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(out)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::join_child(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = Self::join_child(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let existing = self.entries.get(&destination_path).cloned();
        if existing.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(existing_entry) = existing.as_ref() {
            match (&source_entry, existing_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        if existing.is_some() {
            self.remove_subtree(&destination_path);
        }
        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &source_path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();
        for (old, _) in &moving {
            self.entries.remove(old);
        }
        for (old, entry) in moving {
            let suffix = &old[source_path.len()..];
            let new_path = format!("{}{}", destination_path, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    fn remove_subtree(&mut self, path: &str) {
        let doomed: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, path))
            .cloned()
            .collect();
        for candidate in doomed {
            self.entries.remove(&candidate);
        }
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let resolved = Self::join_child(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;
        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&resolved).is_empty() {
            return Err(Error::NotEmpty(resolved));
        }
        self.remove_subtree(&resolved);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized_link = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&normalized_link)?;
        let destination = Self::join_child(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.insert_child(&parent, &name, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let mut paths = if matches!(entry, Entry::Dir) {
            self.entries
                .keys()
                .filter(|candidate| Self::is_within(candidate, &resolved))
                .cloned()
                .collect::<Vec<_>>()
        } else {
            vec![resolved.clone()]
        };
        paths.sort();

        let mut sizes = BTreeMap::<String, u64>::new();
        for current in paths.iter().rev() {
            let size = match self.entries.get(current).expect("listed entry exists") {
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => {
                    let children = self.children(current);
                    children
                        .iter()
                        .map(|child| *sizes.get(child).unwrap_or(&0))
                        .sum()
                }
            };
            sizes.insert(current.clone(), size);
        }
        let mut result = paths
            .into_iter()
            .map(|current| {
                let size = *sizes.get(&current).unwrap_or(&0);
                (current, size)
            })
            .collect::<Vec<_>>();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
