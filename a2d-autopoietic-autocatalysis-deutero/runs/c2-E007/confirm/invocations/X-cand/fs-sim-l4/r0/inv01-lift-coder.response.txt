```rust
use std::collections::{BTreeMap, BTreeSet, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|part| !part.is_empty() && *part != ".")
            .fold(Vec::new(), |mut acc, part| {
                if part == ".." {
                    acc.pop();
                } else {
                    acc.push(part.to_string());
                }
                acc
            })
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize_from(path: &str, base: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(base)
        };
        for part in path.split('/') {
            if part.is_empty() || part == "." {
                continue;
            }
            if part == ".." {
                parts.pop();
            } else {
                parts.push(part.to_string());
            }
        }
        Self::canonical(&parts)
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        Ok(Self::normalize_from(path, &self.cwd))
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            _ => "/".to_string(),
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            String::new()
        } else {
            path.rsplit('/').next().unwrap_or("").to_string()
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            path.starts_with('/')
        } else {
            path == dir || path.strip_prefix(dir).is_some_and(|tail| tail.starts_with('/'))
        }
    }

    fn join_child(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let mut current = Self::components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;
        let mut index = 0usize;

        loop {
            if index >= current.len() {
                return Ok(Self::canonical(&resolved));
            }

            let name = current[index].clone();
            let candidate_parts: Vec<String> = resolved
                .iter()
                .cloned()
                .chain(std::iter::once(name.clone()))
                .collect();
            let candidate = Self::canonical(&candidate_parts);
            let final_component = index + 1 == current.len();

            let parent_path = Self::canonical(&resolved);
            match self.entries.get(&parent_path) {
                Some(Entry::Dir) => {}
                Some(_) => return Err(Error::NotDir(parent_path)),
                None => return Err(Error::Missing(parent_path)),
            }

            let entry = self.entries.get(&candidate);
            if entry.is_none() {
                if final_component && create_final {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            }

            if let Some(Entry::Link(target)) = entry {
                if !final_component || follow_final {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let target_path = if target.starts_with('/') {
                        Self::normalize_from(target, "/")
                    } else {
                        Self::normalize_from(target, &parent_path)
                    };
                    let mut expanded = Self::components(&target_path);
                    expanded.extend(current[index + 1..].iter().cloned());
                    let expanded_path = Self::canonical(&expanded);
                    current = Self::components(&expanded_path);
                    resolved.clear();
                    index = 0;
                    continue;
                }
            }

            if final_component {
                return Ok(candidate);
            }

            match entry {
                Some(Entry::Dir) => {
                    resolved.push(name);
                    index += 1;
                }
                Some(_) => return Err(Error::NotDir(candidate)),
                None => unreachable!(),
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let name = Self::basename(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| *path != dir && Self::parent(path) == dir)
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&normalized)?;
        let actual = Self::join_child(&parent, &name);
        if self.entries.contains_key(&actual) {
            return Err(Error::Exists(actual));
        }
        self.entries.insert(actual, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let actual = self.resolve(&normalized, true, true)?;
        match self.entries.get(&actual) {
            Some(Entry::Dir) => return Err(Error::IsDir(actual)),
            Some(Entry::File(_)) | None => {}
            Some(Entry::Link(_)) => return Err(Error::Loop(actual)),
        }
        self.entries
            .insert(actual, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = self.normalize(path)?;
        let actual = self.resolve(&normalized, true, false)?;
        match self.entries.get(&actual) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(actual)),
            Some(Entry::Link(_)) => Err(Error::Loop(actual)),
            None => Err(Error::Missing(actual)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = self.normalize(path)?;
        let actual = self.resolve(&normalized, true, false)?;
        match self.entries.get(&actual) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(actual)),
            None => return Err(Error::Missing(actual)),
        }
        let mut out: Vec<(String, Kind)> = self
            .children(&actual)
            .into_iter()
            .filter_map(|child| {
                self.entries
                    .get(&child)
                    .map(|entry| (Self::basename(&child), entry.kind()))
            })
            .collect();
        out.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(out)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let actual = self.resolve(&normalized, true, false)?;
        match self.entries.get(&actual) {
            Some(Entry::Dir) => {
                self.cwd = actual;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(actual)),
            None => Err(Error::Missing(actual)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&source)?;
        let src_actual = Self::join_child(&src_parent, &src_name);
        let src_entry = self
            .entries
            .get(&src_actual)
            .cloned()
            .ok_or_else(|| Error::Missing(src_actual.clone()))?;

        if matches!(src_entry, Entry::Dir) && Self::is_within(&self.cwd, &src_actual) {
            return Err(Error::Busy(src_actual));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&destination)?;
        let dst_actual = Self::join_child(&dst_parent, &dst_name);

        if matches!(self.entries.get(&dst_actual), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst_actual)
        {
            return Err(Error::Busy(dst_actual));
        }

        if src_actual == dst_actual {
            return Ok(());
        }

        if matches!(src_entry, Entry::Dir) && Self::is_within(&dst_actual, &src_actual) {
            return Err(Error::Cycle(dst_actual));
        }

        let dst_entry = self.entries.get(&dst_actual).cloned();
        if dst_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst_actual));
        }

        if let Some(ref dst) = dst_entry {
            match (&src_entry, dst) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&dst_actual).is_empty() {
                        return Err(Error::NotEmpty(dst_actual));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst_actual)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst_actual)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &src_actual))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(dst) = dst_entry {
            if matches!(dst, Entry::Dir) {
                self.entries.remove(&dst_actual);
            } else {
                self.entries.remove(&dst_actual);
            }
        }
        for (path, _) in &moving {
            self.entries.remove(path);
        }
        for (old_path, entry) in moving {
            let suffix = &old_path[src_actual.len()..];
            let new_path = format!("{}{}", dst_actual, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let actual = Self::join_child(&parent, &name);
        let entry = self
            .entries
            .get(&actual)
            .cloned()
            .ok_or_else(|| Error::Missing(actual.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &actual) {
            return Err(Error::Busy(actual));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&actual).is_empty() {
            return Err(Error::NotEmpty(actual));
        }

        let removing: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &actual))
            .cloned()
            .collect();
        for p in removing {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized_link = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&normalized_link)?;
        let actual = Self::join_child(&parent, &name);
        if self.entries.contains_key(&actual) {
            return Err(Error::Exists(actual));
        }
        self.entries.insert(actual, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&normalized)?;
        let actual = Self::join_child(&parent, &name);
        match self.entries.get(&actual) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(actual)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = self.normalize(path)?;
        let actual = self.resolve(&normalized, true, false)?;
        let entry = self
            .entries
            .get(&actual)
            .ok_or_else(|| Error::Missing(actual.clone()))?;

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &actual))
            .cloned()
            .collect();

        let mut sizes = BTreeMap::new();
        for p in &paths {
            let size = match self.entries.get(p).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p.clone(), size);
        }

        for p in paths.iter().rev() {
            if matches!(self.entries.get(p), Some(Entry::Dir)) {
                let total = paths
                    .iter()
                    .filter(|child| child != &p && Self::is_within(child, p))
                    .filter(|child| {
                        let parent = Self::parent(child);
                        parent == *p || Self::is_within(&parent, p)
                    })
                    .map(|child| {
                        let mut ancestor = Self::parent(child);
                        while ancestor != *p && Self::is_within(&ancestor, p) {
                            ancestor = Self::parent(&ancestor);
                        }
                        if ancestor == *p {
                            *sizes.get(child).unwrap_or(&0)
                        } else {
                            0
                        }
                    })
                    .sum::<u64>();
                sizes.insert(p.clone(), total);
            }
        }

        let _ = entry;
        Ok(sizes.into_iter().collect())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
