#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy: Option<(u64, u32)>, // (finish time, customer id)
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_arrival = 0usize;
    let mut finished_customers = 0usize;
    let mut now = 0u64;

    loop {
        // Finish phase: ascending server ID.
        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| {
                s.busy
                    .filter(|(finish, _)| *finish == now)
                    .map(|_| i)
            })
            .collect();
        finishing.sort_by_key(|&i| states[i].server.id);
        for i in finishing {
            let (_, customer) = states[i].busy.take().unwrap();
            events.push(Event::Finish {
                time: now,
                customer,
                server: states[i].server.id,
            });
            states[i].idle_since = now;
            finished_customers += 1;
        }

        // Open phase: descending server ID.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.server.opens == now).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[i].server.id,
            });
        }

        // Arrival phase: arrivals are already ordered by time and descending ID.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = &arrivals[next_arrival];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(next_arrival);
            next_arrival += 1;
        }

        // Dispatch until no customer/server pair remains.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.busy.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &arrivals[customer_index];
            let server = &states[server_index].server;

            let finish = now
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
                finished_customers += 1;
            } else {
                states[server_index].busy = Some((finish, customer.id));
            }
        }

        if finished_customers == customers.len()
            && states.iter().all(|s| s.opened)
        {
            break;
        }

        let next = arrivals
            .get(next_arrival)
            .map(|c| c.arrival)
            .into_iter()
            .chain(
                states
                    .iter()
                    .filter(|s| !s.opened)
                    .map(|s| s.server.opens),
            )
            .chain(states.iter().filter_map(|s| s.busy.map(|(t, _)| t)))
            .filter(|&t| t > now)
            .min();

        if let Some(time) = next {
            now = time;
        } else {
            // All pending work is complete; this also handles empty input.
            break;
        }
    }

    Ok(events)
}
