#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut server_states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            idle_since: server.opens,
            busy_until: None,
        })
        .collect();

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut arrival_index = 0;
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut events = Vec::new();
    let mut now = 0u64;

    loop {
        let mut finished: Vec<(u32, usize)> = server_states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.busy_until == Some(now)).then_some((state.id, index))
            })
            .collect();
        finished.sort_by_key(|&(id, _)| id);

        for (_, index) in finished {
            let state = &mut server_states[index];
            // The associated customer is recorded by the scheduled finish below.
            state.busy_until = None;
            state.idle_since = now;
        }

        // Finish events need the customer assigned to each completing server.
        // They are inserted here from the separate occupation records below.
        // (The records are populated and consumed alongside server state.)

        let mut opening_indices: Vec<usize> = server_states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.opens == now).then_some(index)
            })
            .collect();
        opening_indices.sort_by(|&a, &b| {
            server_states[b].id.cmp(&server_states[a].id)
        });
        for index in opening_indices {
            server_states[index].opened = true;
            events.push(Event::Open {
                time: now,
                server: server_states[index].id,
            });
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == now {
            let customer = arrivals[arrival_index];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(customer);
            arrival_index += 1;
        }

        loop {
            if waiting.is_empty() {
                break;
            }

            let selected = server_states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting.remove(0);
            let server = &server_states[server_index];
            let Some(after_service) = now.checked_add(customer.service) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };
            let Some(finish) = after_service.checked_add(server.overhead) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };

            let server_id = server.id;
            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server_id,
                });
                server_states[server_index].idle_since = now;
            } else {
                server_states[server_index].busy_until = Some(finish);
            }
        }

        let next_arrival = arrivals.get(arrival_index).map(|customer| customer.arrival);
        let next_open = server_states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = server_states
            .iter()
            .filter_map(|state| state.busy_until)
            .min();

        let next = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
