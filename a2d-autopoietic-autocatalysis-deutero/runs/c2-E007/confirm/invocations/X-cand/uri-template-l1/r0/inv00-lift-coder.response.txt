#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, String)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(relative) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: start + 1 + relative,
        });
    }

    let mut content_start = start + 1;
    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    match bytes[content_start] {
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => {}
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();

    loop {
        let spec_start = content_start;
        let spec_end = bytes[spec_start..end]
            .iter()
            .position(|&b| b == b',')
            .map_or(end, |relative| spec_start + relative);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &bytes[spec_start..spec_end];
        if !is_name_start(spec[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (offset, &byte) in spec.iter().enumerate().skip(1) {
            if !is_name_byte(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|prior| prior == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push((spec_start, name));

        if spec_end == end {
            break;
        }
        content_start = spec_end + 1;
    }

    Ok(Expression { start, end, names })
}

fn validate_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let Some(relative_end) = bytes[pos + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = pos + 1 + relative_end;
                expressions.push(parse_expression(template, start, end)?);
                pos = end + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn encode_value(value: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];
        if byte == b'%' && pos + 2 < bytes.len() {
            let high = bytes[pos + 1];
            let low = bytes[pos + 2];
            if high.is_ascii_hexdigit() && low.is_ascii_hexdigit() {
                output.push('%');
                output.push((high as char).to_ascii_uppercase());
                output.push((low as char).to_ascii_uppercase());
                pos += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        pos += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut first = true;
        for (_, name) in expression.names {
            let mut found = None;
            for &(binding_name, value) in vars {
                if binding_name == name {
                    found = Some(value);
                }
            }

            if let Some(value) = found {
                if !first {
                    output.push(',');
                }
                output.push_str(&encode_value(value));
                first = false;
            }
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
