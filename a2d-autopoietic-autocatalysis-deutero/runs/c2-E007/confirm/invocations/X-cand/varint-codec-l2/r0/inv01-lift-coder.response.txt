#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::new();
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            output.push(payload);
            break;
        }
        output.push(payload | 0x80);
    }
    output
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    let mut cursor = start;

    for group in 0..10 {
        if cursor >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[cursor];
        let payload = (byte & 0x7f) as u64;

        if group == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow {
                    offset: cursor,
                    start,
                });
            }
            value |= payload << 63;
            return Ok((value, cursor + 1));
        }

        value |= payload << (group * 7);
        cursor += 1;

        if byte & 0x80 == 0 {
            return Ok((value, cursor));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count);
    let mut cursor = start;

    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }

    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let zigzag = if value >= 0 {
        (value as u64) << 1
    } else {
        ((value.unsigned_abs()) << 1).wrapping_sub(1)
    };
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, cursor) = decode_u64(input, start)?;
    let decoded = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((decoded, cursor))
}

pub struct StreamDecoder {
    position: usize,
    pending: Vec<u8>,
    pending_start: usize,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            pending: Vec::new(),
            pending_start: 0,
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut combined = Vec::with_capacity(self.pending.len().saturating_add(chunk.len()));
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(chunk);

        let base = self.pending_start;
        let mut cursor = 0usize;
        let mut values = Vec::new();

        while cursor < combined.len() {
            let integer_start = cursor;
            let mut value = 0u64;
            let mut terminated = false;

            for group in 0..10 {
                if cursor >= combined.len() {
                    break;
                }

                let byte = combined[cursor];
                let payload = (byte & 0x7f) as u64;
                let absolute = base + cursor;

                if group == 9 {
                    if payload > 1 || byte & 0x80 != 0 {
                        return Err(Error::Overflow {
                            offset: absolute,
                            start: base + integer_start,
                        });
                    }
                    value |= payload << 63;
                    cursor += 1;
                    values.push(value);
                    terminated = true;
                    break;
                }

                value |= payload << (group * 7);
                cursor += 1;

                if byte & 0x80 == 0 {
                    values.push(value);
                    terminated = true;
                    break;
                }
            }

            if !terminated {
                break;
            }
        }

        let consumed_total = self.position + chunk.len();
        let consumed_from_combined = cursor;
        let remaining = &combined[consumed_from_combined..];

        self.position = consumed_total;
        self.pending.clear();
        self.pending.extend_from_slice(remaining);
        if self.pending.is_empty() {
            self.pending_start = self.position;
        } else {
            self.pending_start = base + consumed_from_combined;
        }

        Ok(values)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending.is_empty() {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.position - self.pending.len(),
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending.len()
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending.clear();
        self.pending_start = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}
