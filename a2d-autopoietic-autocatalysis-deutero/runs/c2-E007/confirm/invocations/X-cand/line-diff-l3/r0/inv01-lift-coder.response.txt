#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Row {
    prefix: u8,
    payload: String,
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(bytes: &[u8]) -> Option<usize> {
    if bytes.is_empty() || (bytes.len() > 1 && bytes[0] == b'0') {
        return None;
    }
    let mut value = 0usize;
    for &b in bytes {
        if !b.is_ascii_digit() {
            return None;
        }
        value = value.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if value > 128 {
            return None;
        }
    }
    Some(value)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let b = line.as_bytes();
    if b.len() < 14 || !b.starts_with(b"@@ -") || !b.ends_with(b" @@") {
        return None;
    }
    let inner = &b[4..b.len() - 3];
    let space = inner.iter().position(|&c| c == b' ')?;
    let old = &inner[..space];
    let new = &inner[space + 1..];
    if new.first() != Some(&b'+') {
        return None;
    }
    let old = old.strip_prefix(b"")?;
    if old.first() != Some(&b'-') {
        return None;
    }
    let old_fields = &old[1..];
    let new_fields = &new[1..];
    let comma_old = old_fields.iter().position(|&c| c == b',')?;
    let comma_new = new_fields.iter().position(|&c| c == b',')?;
    if old_fields[comma_old + 1..].contains(&b',')
        || new_fields[comma_new + 1..].contains(&b',')
    {
        return None;
    }
    Some((
        parse_number(&old_fields[..comma_old])?,
        parse_number(&old_fields[comma_old + 1..])?,
        parse_number(&new_fields[..comma_new])?,
        parse_number(&new_fields[comma_new + 1..])?,
    ))
}

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 { start - 1 } else { start }
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let mut source_lines = Vec::<String>::new();
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if !source.is_empty() {
        source_lines.extend(source[..source.len() - 1].split('\n').map(str::to_owned));
    }
    if source_lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    if patch.is_empty() {
        return Ok(source.to_owned());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    if lines.first() != Some(&"--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if lines.get(1) != Some(&"+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::<Hunk>::new();
    let mut i = 2usize;
    let mut previous_old_end = 0usize;
    let mut cumulative_delta: i128 = 0;

    while i < lines.len() {
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: i + 1 });
        }
        let header_line = i + 1;
        let (a, b, c, d) = parse_hunk_header(lines[i])
            .ok_or(ApplyError::BadHunkHeader { line: header_line })?;
        if (b > 0 && a == 0) || (d > 0 && c == 0) {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let line = lines[i];
            let bytes = line.as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') || line.contains('\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            let payload = &line[1..];
            let prefix = bytes[0];
            if prefix == b' ' || prefix == b'-' {
                old_seen += 1;
            }
            if prefix == b' ' || prefix == b'+' {
                new_seen += 1;
            }
            rows.push(Row { prefix, payload: payload.to_owned() });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_end = old_start.saturating_add(b);
        if old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i128 + cumulative_delta;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_end;
        cumulative_delta += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    let mut working = source_lines;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.prefix == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.prefix == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut best_by_start = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.rows.len() {
                    continue;
                }
                let trimmed_end = hunk.rows.len() - r;
                let kept = &hunk.rows[l..trimmed_end];
                let match_payloads: Vec<&str> = kept
                    .iter()
                    .filter(|row| row.prefix == b' ' || row.prefix == b'-')
                    .map(|row| row.payload.as_str())
                    .collect();

                if match_payloads.len() != hunk.old_count - l - r {
                    continue;
                }

                for s in 0..=working.len() {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (s as i128 - expected).unsigned_abs();
                    if distance > options.max_offset as u128 {
                        continue;
                    }
                    let start = s + l;
                    let end = s + hunk.old_count - r;
                    if match_payloads.len() != end - start
                        || !match_payloads
                            .iter()
                            .zip(&working[start..end])
                            .all(|(expected, actual)| *expected == actual)
                    {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, l, r));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, l, r));
                        }
                        Some(current) if rank == current => {
                            best_by_start.push((s, l, r));
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_by_start.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_index + 1 });
        }
        best_by_start.sort_by_key(|&(s, l, _)| (s, l));
        let distinct_starts: Vec<usize> = {
            let mut v = Vec::new();
            for &(s, _, _) in &best_by_start {
                if v.last() != Some(&s) {
                    v.push(s);
                }
            }
            v
        };
        if distinct_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: distinct_starts[0],
                second: distinct_starts[1],
            });
        }

        let (s, l, r) = best_by_start[0];
        let trimmed_end = hunk.rows.len() - r;
        let replacement: Vec<String> = hunk.rows[l..trimmed_end]
            .iter()
            .filter(|row| row.prefix == b' ' || row.prefix == b'+')
            .map(|row| row.payload.clone())
            .collect();
        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        drift = s as i128 - (hunk.old_start as i128 + net);
        net += hunk.new_count as i128 - hunk.old_count as i128;
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
