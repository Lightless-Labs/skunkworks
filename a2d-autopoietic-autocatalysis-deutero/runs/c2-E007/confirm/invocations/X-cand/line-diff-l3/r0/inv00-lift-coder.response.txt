#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Row {
    kind: u8,
    payload: String,
    line: usize,
}

struct Hunk {
    line: usize,
    old_start: i64,
    old_count: usize,
    new_start: i64,
    new_count: usize,
    rows: Vec<Row>,
}

fn patch_lines(patch: &str) -> Vec<&str> {
    patch.split_terminator('\n').collect()
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty()
        || !s.bytes().all(|b| b.is_ascii_digit())
        || (s.len() > 1 && s.starts_with('0'))
    {
        return None;
    }
    let mut n = 0usize;
    for b in s.bytes() {
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let inner = line.strip_prefix("@@ -").ok_or_else(bad)?;
    let (old, rest) = inner.split_once(" +").ok_or_else(bad)?;
    let (new, suffix) = rest.split_once(" @@").ok_or_else(bad)?;
    if !suffix.is_empty() {
        return Err(bad());
    }
    let (a, b) = old.split_once(',').ok_or_else(bad)?;
    let (c, d) = new.split_once(',').ok_or_else(bad)?;
    let a = parse_number(a).ok_or_else(bad)?;
    let b = parse_number(b).ok_or_else(bad)?;
    let c = parse_number(c).ok_or_else(bad)?;
    let d = parse_number(d).ok_or_else(bad)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    Ok((a, b, c, d))
}

fn full_start(anchor: usize, count: usize) -> i64 {
    if count > 0 {
        anchor as i64 - 1
    } else {
        anchor as i64
    }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }
    let lines = patch_lines(patch);
    if lines.get(0) != Some(&"--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if lines.get(1) != Some(&"+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_old_end = 0i64;
    let mut net = 0i64;

    while i < lines.len() {
        let header_line = i + 1;
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let (a, b, c, d) = parse_hunk_header(lines[i], header_line)?;
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);
        i += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let text = lines[i];
            let Some((&kind, payload)) = text.as_bytes().split_first() else {
                return Err(ApplyError::BadBody { line: row_line });
            };
            if !matches!(kind, b' ' | b'-' | b'+') || text.as_bytes().contains(&b'\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = std::str::from_utf8(payload)
                .map_err(|_| ApplyError::BadBody { line: row_line })?
                .to_string();
            if kind != b'+' {
                old_seen += 1;
            }
            if kind != b'-' {
                new_seen += 1;
            }
            rows.push(Row { kind, payload, line: row_line });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end || new_start != old_start + net {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_start + b as i64;
        net += d as i64 - b as i64;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source
        .strip_suffix('\n')
        .unwrap()
        .split('\n')
        .map(str::to_string)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net = 0i64;
    let mut drift = 0i64;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
        let expected = hunk.old_start + net + drift;
        let b = hunk.old_count;
        let mut best_rank: Option<(usize, i64)> = None;
        let mut best_by_start: Vec<(usize, usize, usize)> = Vec::new();

        for left in 0..=options.fuzz.min(lead) {
            for right in 0..=options.fuzz.min(trail) {
                if left + right > b {
                    continue;
                }
                let mut match_payloads = Vec::new();
                let mut output_payloads = Vec::new();
                for (idx, row) in hunk.rows.iter().enumerate() {
                    if idx < left || idx >= hunk.rows.len().saturating_sub(right) {
                        continue;
                    }
                    if row.kind != b'+' {
                        match_payloads.push(row.payload.as_str());
                    }
                    if row.kind != b'-' {
                        output_payloads.push(row.payload.clone());
                    }
                }
                let fuzz_cost = left + right;
                let max_s = working.len().saturating_sub(b);
                for s in 0..=max_s {
                    let distance = (s as i64 - expected).abs();
                    if distance > options.max_offset as i64 {
                        continue;
                    }
                    if s + left + (b - left - right) > working.len() {
                        continue;
                    }
                    let slice = &working[s + left..s + b - right];
                    if slice.len() != match_payloads.len()
                        || !slice.iter().zip(&match_payloads).all(|(x, y)| x == y)
                    {
                        continue;
                    }
                    let rank = (fuzz_cost, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, left, right));
                        }
                        Some(old) if rank < old => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, left, right));
                        }
                        Some(old) if rank == old => {
                            if let Some(existing) = best_by_start.iter_mut().find(|v| v.0 == s) {
                                if left < existing.1 {
                                    *existing = (s, left, right);
                                }
                            } else {
                                best_by_start.push((s, left, right));
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_by_start.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_index + 1 });
        }
        best_by_start.sort_by_key(|v| v.0);
        if best_by_start.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_by_start[0].0,
                second: best_by_start[1].0,
            });
        }

        let (s, left, right) = best_by_start[0];
        let mut replacement = Vec::new();
        for (idx, row) in hunk.rows.iter().enumerate() {
            if idx < left || idx >= hunk.rows.len().saturating_sub(right) {
                continue;
            }
            if row.kind != b'-' {
                replacement.push(row.payload.clone());
            }
        }
        let start = s + left;
        let end = s + b - right;
        working.splice(start..end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
        let previous_net = net;
        net += hunk.new_count as i64 - hunk.old_count as i64;
        drift = s as i64 - (hunk.old_start + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
