#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn foreign_alphabet(byte: u8, alphabet: Alphabet) -> bool {
    match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    }
}

fn encode_digit(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 if alphabet == Alphabet::Standard => b'+',
        63 if alphabet == Alphabet::Standard => b'/',
        62 => b'-',
        _ => b'_',
    }
}

fn encoded_group(bytes: &[u8], options: Options, final_group: bool) -> Vec<u8> {
    let mut out = Vec::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(encode_digit(a >> 2, options.alphabet));
    out.push(encode_digit(((a & 0x03) << 4) | (b >> 4), options.alphabet));
    if bytes.len() > 1 {
        out.push(encode_digit(((b & 0x0f) << 2) | (c >> 6), options.alphabet));
    } else if final_group && options.padding != Padding::Forbidden {
        out.push(b'=');
    }
    if bytes.len() > 2 {
        out.push(encode_digit(c & 0x3f, options.alphabet));
    } else if final_group && options.padding != Padding::Forbidden {
        out.push(b'=');
    }
    out
}

fn encode_bytes(input: &[u8], options: Options) -> Vec<u8> {
    let mut symbols = Vec::new();
    let mut chunks = input.chunks_exact(3);
    for group in &mut chunks {
        symbols.extend(encoded_group(group, options, false));
    }
    let rem = chunks.remainder();
    if !rem.is_empty() {
        symbols.extend(encoded_group(rem, options, true));
    }

    if let Whitespace::Wrapped { width, ending } = options.whitespace {
        let separator: &[u8] = match ending {
            LineEnding::Lf => b"\n",
            LineEnding::CrLf => b"\r\n",
        };
        let mut out = Vec::with_capacity(symbols.len() + symbols.len() / width * separator.len());
        for (i, symbol) in symbols.into_iter().enumerate() {
            if i != 0 && i % width == 0 {
                out.extend_from_slice(separator);
            }
            out.push(symbol);
        }
        out
    } else {
        symbols
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    Ok(String::from_utf8(encode_bytes(input, options)).expect("base64 output is ASCII"))
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    column: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffered: Vec::new(),
            column: 0,
            closed: false,
        })
    }

    fn emit_symbols(&mut self, symbols: &[u8], out: &mut Vec<u8>) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            for &symbol in symbols {
                if self.column == width {
                    match ending {
                        LineEnding::Lf => out.push(b'\n'),
                        LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
                    }
                    self.column = 0;
                }
                out.push(symbol);
                self.column += 1;
            }
        } else {
            out.extend_from_slice(symbols);
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut data = Vec::with_capacity(self.buffered.len() + input.len());
        data.extend_from_slice(&self.buffered);
        data.extend_from_slice(input);
        self.buffered.clear();

        let complete_len = data.len() / 3 * 3;
        let mut out = Vec::new();
        for group in data[..complete_len].chunks_exact(3) {
            let symbols = encoded_group(group, self.options, false);
            self.emit_symbols(&symbols, &mut out);
        }
        self.buffered.extend_from_slice(&data[complete_len..]);
        Ok(String::from_utf8(out).expect("base64 output is ASCII"))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let mut out = Vec::new();
        if !self.buffered.is_empty() {
            let symbols = encoded_group(&self.buffered, self.options, true);
            self.emit_symbols(&symbols, &mut out);
            self.buffered.clear();
        }
        Ok(String::from_utf8(out).expect("base64 output is ASCII"))
    }
}

#[derive(Clone)]
struct DecoderCore {
    options: Options,
    offset: usize,
    symbols: usize,
    group: Vec<(u8, usize)>,
    pending_second_pad: Option<usize>,
    padded_finished: bool,
    wrap_column: usize,
    wrap_state: u8,
    wrap_separator_offset: usize,
    after_wrap_separator: bool,
}

impl DecoderCore {
    fn new(options: Options) -> Self {
        Self {
            options,
            offset: 0,
            symbols: 0,
            group: Vec::with_capacity(4),
            pending_second_pad: None,
            padded_finished: false,
            wrap_column: 0,
            wrap_state: 0,
            wrap_separator_offset: 0,
            after_wrap_separator: false,
        }
    }

    fn is_ignored(&self, byte: u8) -> bool {
        self.options.whitespace == Whitespace::IgnoreAscii
            && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn decode_group(&mut self, out: &mut Vec<u8>, padded: bool) -> Result<(), Error> {
        let g = &self.group;
        let a = digit(g[0].0, self.options.alphabet).unwrap();
        let b = digit(g[1].0, self.options.alphabet).unwrap();
        if g[2].0 == b'=' {
            if self.options.strict && b & 0x0f != 0 {
                return Err(Error::NonCanonical { offset: g[1].1 });
            }
            out.push((a << 2) | (b >> 4));
        } else {
            let c = digit(g[2].0, self.options.alphabet).unwrap();
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            if !padded {
                let d = digit(g[3].0, self.options.alphabet).unwrap();
                out.push((c << 6) | d);
            } else if self.options.strict && c & 0x03 != 0 {
                return Err(Error::NonCanonical { offset: g[2].1 });
            }
        }
        Ok(())
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbols += 1;

        if self.padded_finished {
            return Err(Error::BadPadding { offset });
        }
        if self.pending_second_pad.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        let pos = self.group.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if pos < 2 || pos > 3 {
                return Err(Error::BadPadding { offset });
            }
            self.group.push((byte, offset));
            if pos == 2 {
                self.pending_second_pad = Some(offset);
            } else {
                self.pending_second_pad = None;
                self.decode_group(out, true)?;
                self.group.clear();
                self.padded_finished = true;
            }
            return Ok(());
        }

        if let Some(value) = digit(byte, self.options.alphabet) {
            self.group.push((byte, offset));
            if self.group.len() == 4 {
                self.decode_group(out, false)?;
                self.group.clear();
            }
            return Ok(());
        }

        if self.options.strict && foreign_alphabet(byte, self.options.alphabet) {
            return Err(Error::MixedAlphabet { offset });
        }
        Err(Error::InvalidByte { offset, byte })
    }

    fn process_wrapped_byte(
        &mut self,
        byte: u8,
        offset: usize,
        out: &mut Vec<u8>,
    ) -> Result<(), Error> {
        let (width, ending) = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => (width, ending),
            _ => unreachable!(),
        };

        if self.wrap_state != 0 {
            match (self.wrap_state, ending, byte) {
                (1, LineEnding::Lf, b'\n') => {
                    self.wrap_state = 0;
                    self.after_wrap_separator = true;
                    self.wrap_column = 0;
                    return Ok(());
                }
                (1, LineEnding::CrLf, b'\n') => {
                    self.wrap_state = 0;
                    self.after_wrap_separator = true;
                    self.wrap_column = 0;
                    return Ok(());
                }
                (1, LineEnding::CrLf, _) => return Err(Error::BadWrap { offset }),
                (1, LineEnding::Lf, _) => return Err(Error::BadWrap { offset }),
                _ => unreachable!(),
            }
        }

        if self.wrap_column == width {
            let expected = match ending {
                LineEnding::Lf => b'\n',
                LineEnding::CrLf => b'\r',
            };
            if byte != expected {
                return Err(Error::BadWrap { offset });
            }
            self.wrap_separator_offset = offset;
            self.wrap_state = 1;
            return Ok(());
        }

        if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
            return Err(Error::BadWrap { offset });
        }

        self.after_wrap_separator = false;
        self.wrap_column += 1;
        self.process_symbol(byte, offset, out)
    }

    fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        let mut out = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            match self.options.whitespace {
                Whitespace::None => self.process_symbol(byte, offset, &mut out)?,
                Whitespace::IgnoreAscii => {
                    if !self.is_ignored(byte) {
                        self.process_symbol(byte, offset, &mut out)?;
                    }
                }
                Whitespace::Wrapped { .. } => {
                    self.process_wrapped_byte(byte, offset, &mut out)?
                }
            }
        }
        Ok(out)
    }

    fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if self.wrap_state != 0 {
            return Err(Error::BadWrap {
                offset: self.wrap_separator_offset,
            });
        }
        if self.after_wrap_separator {
            return Err(Error::BadWrap {
                offset: self.wrap_separator_offset,
            });
        }
        if let Some(offset) = self.pending_second_pad {
            return Err(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            match self.options.padding {
                Padding::Required => {
                    return Err(Error::Truncated {
                        offset: self.group[0].1,
                    });
                }
                Padding::Forbidden | Padding::Optional => {
                    if self.group.len() == 1 {
                        return Err(Error::Truncated {
                            offset: self.group[0].1,
                        });
                    }
                    if self.group.iter().any(|(b, _)| *b == b'=') {
                        return Err(Error::BadPadding {
                            offset: self.group.iter().find(|(b, _)| *b == b'=').unwrap().1,
                        });
                    }
                    let mut out = Vec::new();
                    let a = digit(self.group[0].0, self.options.alphabet).unwrap();
                    let b = digit(self.group[1].0, self.options.alphabet).unwrap();
                    out.push((a << 2) | (b >> 4));
                    if self.group.len() == 3 {
                        let c = digit(self.group[2].0, self.options.alphabet).unwrap();
                        out.push((b << 4) | (c >> 2));
                        if self.options.strict && c & 0x03 != 0 {
                            return Err(Error::NonCanonical {
                                offset: self.group[2].1,
                            });
                        }
                    } else if self.options.strict && b & 0x0f != 0 {
                        return Err(Error::NonCanonical {
                            offset: self.group[1].1,
                        });
                    }
                    self.group.clear();
                    return Ok(out);
                }
            }
        }
        Ok(Vec::new())
    }
}

pub struct Decoder {
    core: DecoderCore,
    sticky: Option<Error>,
    closed: bool,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            core: DecoderCore::new(options),
            sticky: None,
            closed: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        match self.core.push(input) {
            Ok(out) => Ok(out),
            Err(error) => {
                self.sticky = Some(error.clone());
                Err(error)
            }
        }
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        match self.core.finish() {
            Ok(out) => {
                self.closed = true;
                Ok(out)
            }
            Err(error) => {
                self.sticky = Some(error.clone());
                Err(error)
            }
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
