```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Base-10 digits, most significant first; zero is [0] and never negative.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan all magnitude bytes for invalid characters before checking separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut first_bad_separator = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_' {
                let valid_before = index > start && bytes[index - 1].is_ascii_digit();
                let valid_after = index + 1 < bytes.len() && bytes[index + 1].is_ascii_digit();
                if !(valid_before && valid_after) {
                    first_bad_separator = Some(index);
                    break;
                }
            }
        }
        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len())
            .find(|&index| bytes[index].is_ascii_digit())
            .expect("nonempty validated magnitude");

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for index in start..bytes.len() {
                if bytes[index].is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let digits: Vec<u8> = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| b - b'0')
            .collect();

        let is_zero = digits.iter().all(|&digit| digit == 0);
        Ok(Self {
            negative: negative && !is_zero,
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let padding = width - self.digits.len();
        let mut out = String::with_capacity(width + width / group.max(1) + usize::from(self.negative));

        if self.negative {
            out.push('-');
        }

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }
            if position < padding {
                out.push('0');
            } else {
                out.push(char::from(b'0' + self.digits[position - padding]));
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        match (self.negative, rhs.negative) {
            (false, false) => Self::from_parts(false, add_magnitude(&self.digits, &rhs.digits)),
            (true, true) => Self::from_parts(true, add_magnitude(&self.digits, &rhs.digits)),
            (false, true) => match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => Self::from_parts(
                    false,
                    sub_magnitude(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    true,
                    sub_magnitude(&rhs.digits, &self.digits),
                ),
                Ordering::Equal => Self::zero(),
            },
            (true, false) => match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => Self::from_parts(
                    true,
                    sub_magnitude(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    false,
                    sub_magnitude(&rhs.digits, &self.digits),
                ),
                Ordering::Equal => Self::zero(),
            },
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().rev().enumerate() {
            for (j, &b) in rhs.digits.iter().rev().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }

        for index in 0..accum.len() - 1 {
            let carry = accum[index] / 10;
            accum[index] %= 10;
            accum[index + 1] += carry;
        }

        let mut digits: Vec<u8> = accum.into_iter().rev().map(|n| n as u8).collect();
        trim_leading_zeros(&mut digits);
        Self::from_parts(self.negative ^ rhs.negative, digits)
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        trim_leading_zeros(&mut digits);
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }
}

fn trim_leading_zeros(digits: &mut Vec<u8>) {
    let first_nonzero = digits
        .iter()
        .position(|&digit| digit != 0)
        .unwrap_or(digits.len().saturating_sub(1));
    if first_nonzero > 0 {
        digits.drain(0..first_nonzero);
    }
    if digits.is_empty() {
        digits.push(0);
    }
}

fn cmp_magnitude(lhs: &[u8], rhs: &[u8]) -> Ordering {
    lhs.len()
        .cmp(&rhs.len())
        .then_with(|| lhs.cmp(rhs))
}

fn add_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(lhs.len().max(rhs.len()) + 1);
    let mut i = lhs.len();
    let mut j = rhs.len();
    let mut carry = 0u8;

    while i > 0 || j > 0 || carry > 0 {
        let a = if i > 0 {
            i -= 1;
            lhs[i]
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            rhs[j]
        } else {
            0
        };
        let sum = a + b + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }

    result.reverse();
    result
}

fn sub_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_magnitude(lhs, rhs) != Ordering::Less);

    let mut result = Vec::with_capacity(lhs.len());
    let mut i = lhs.len();
    let mut j = rhs.len();
    let mut borrow = 0i8;

    while i > 0 {
        i -= 1;
        let a = i16::from(lhs[i]) - i16::from(borrow);
        let b = if j > 0 {
            j -= 1;
            i16::from(rhs[j])
        } else {
            0
        };

        if a < b {
            result.push((a + 10 - b) as u8);
            borrow = 1;
        } else {
            result.push((a - b) as u8);
            borrow = 0;
        }
    }

    result.reverse();
    trim_leading_zeros(&mut result);
    result
}
```
