fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();

            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()

            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());

                expressions.push((literal_start, open, expr));

                literal_start = cursor;

            _ => cursor += 1,
        }

        .rfind('}')
        .map(|p| p + 1)
        .unwrap_or(0);

                    && digits.as_bytes().iter().all(u8::is_ascii_digit)
                    && digits.as_bytes()[0] != b'0';
                let value = if valid {
                    digits.parse::<usize>().ok()
                } else {
                    None
                };
                match value {
                    Some(n) if (1..=9999).contains(&n) => Modifier::Prefix(n),
                    _ => {
                        return Err(Error::InvalidModifier {
                            pos: spec_start + i,
                        });
                    }
                }
            } else {
                return Err(Error::InvalidModifier {
                    pos: spec_start + i,
                });
            }
        } else {
            Modifier::None
        };

        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            start: spec_start,
            modifier,
        });

        if end == remaining.len() {
            break;
        }
        segment_start = end + 1;
    }

    Ok(Expression { operator, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || b"-._~".contains(&b);
        let is_reserved = reserved && b":/?#[]@!$&'()*+,;=".contains(&b);
        if unreserved || is_reserved {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: char) -> String {
    if operator == ';' && value.is_empty() {
        name.to_string()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expr.operator;
    let reserved = operator == '+' || operator == '#';
    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if let Modifier::Prefix(_) = spec.modifier {
            if !matches!(value, Value::Scalar(_)) {
                return Err(Error::TypeMismatch { pos: spec.start });
            }
        }

        if let Value::Map(entries) = value {
            let mut keys = Vec::<&str>::new();
            for (key, _) in entries {
                if keys.iter().any(|previous| *previous == key) {
                    return Err(Error::DuplicateKey { pos: spec.start });
                }
                keys.push(key);
            }
        }

        match value {
            Value::Scalar(s) => {
                let value = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                fields.push(if matches!(operator, ';' | '?' | '&') {
                    named_field(&spec.name, encode(&value, reserved), operator)
                } else {
                    encode(&value, reserved)
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(operator, ';' | '?' | '&') {
                            named_field(&spec.name, encoded, operator)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(operator, ';' | '?' | '&') {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let key = encode(key, reserved);
                        let value = encode(value, reserved);
                        let field = if operator == ';' && value.is_empty() {
                            key
                        } else {
                            format!("{key}={value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(operator, ';' | '?' | '&') {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match operator {
        '#' => ("#", ','),
        '.' => (".", '.'),
        '/' => ("/", '/'),
        ';' => (";", ';'),
        '?' => ("?", '&'),
        '&' => ("&", '&'),
        _ => ("", ','),
    };
    Ok(format!("{prefix}{}", fields.join(&separator.to_string())))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut cursor = 0;
    let mut literal_start = 0;
    let mut expressions = Vec::<(usize, usize, Expression)>::new();

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = template[open + 1..].find('}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;
                if let Some(inner) = template[open + 1..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner,
                    });
                }
                let content = &template[open + 1..close];
                let expr = parse_expression(content, open + 1, close)?;
                expressions.push((literal_start, open, expr));
                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    for (start, open, expr) in expressions {
        output.push_str(&template[start..open]);
        output.push_str(&expand_expression(&expr, vars)?);
    }
    output.push_str(&template[literal_start..]);

    Ok(output)
}
