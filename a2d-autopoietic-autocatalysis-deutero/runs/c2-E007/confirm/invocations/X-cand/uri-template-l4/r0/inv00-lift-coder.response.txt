#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
struct Spec {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Query,
    Ampersand,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Plus | Self::Hash => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Query | Self::Ampersand => '&',
        }
    }

    fn prefix(self) -> &'static str {
        match self {
            Self::Simple | Self::Plus => "",
            Self::Hash => "#",
            Self::Dot => ".",
            Self::Slash => "/",
            Self::Semicolon => ";",
            Self::Query => "?",
            Self::Ampersand => "&",
        }
    }

    fn named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Ampersand)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Plus | Self::Hash)
    }
}

struct Expression {
    operator: Operator,
    specs: Vec<Spec>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_modifier(suffix: &str, pos: usize) -> Result<Modifier, Error> {
    if suffix == "*" {
        return Ok(Modifier::Explode);
    }
    if let Some(digits) = suffix.strip_prefix(':') {
        if !digits.is_empty()
            && digits.len() <= 4
            && digits.as_bytes()[0].is_ascii_digit()
            && digits.as_bytes()[0] != b'0'
            && digits.bytes().all(|b| b.is_ascii_digit())
        {
            let n = digits.parse::<usize>().unwrap_or(0);
            if (1..=9999).contains(&n) {
                return Ok(Modifier::Prefix(n));
            }
        }
    }
    Err(Error::InvalidModifier { pos })
}

fn parse_expression(content: &str, start: usize, close: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = match bytes.first().copied() {
        Some(b'+') => (Operator::Plus, 1),
        Some(b'#') => (Operator::Hash, 1),
        Some(b'.') => (Operator::Dot, 1),
        Some(b'/') => (Operator::Slash, 1),
        Some(b';') => (Operator::Semicolon, 1),
        Some(b'?') => (Operator::Query, 1),
        Some(b'&') => (Operator::Ampersand, 1),
        _ => (Operator::Simple, 0),
    };

    if offset != 0 && !matches!(operator, Operator::Plus | Operator::Hash | Operator::Dot | Operator::Slash | Operator::Semicolon | Operator::Query | Operator::Ampersand) {
        return Err(Error::UnsupportedOperator { pos: start });
    }

    let rest = &content[offset..];
    let base_pos = start + offset;
    if rest.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut piece_start = 0usize;

    for piece in rest.split(',') {
        let spec_start = base_pos + piece_start;
        if piece.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let raw = piece.as_bytes();
        let modifier_index = raw.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_index.unwrap_or(raw.len());
        if name_end == 0 {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let name_bytes = &raw[..name_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(i) = name_bytes.iter().position(|&b| !is_name_byte(b)) {
            return Err(Error::InvalidName { pos: spec_start + i });
        }

        let modifier = if let Some(i) = modifier_index {
            parse_modifier(&piece[i..], spec_start + i)?
        } else {
            Modifier::None
        };

        let name = piece[..name_end].to_string();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        specs.push(Spec {
            name,
            start: spec_start,
            modifier,
        });
        piece_start += piece.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0usize;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                if let Some(inner) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + inner });
                }
                let content = &template[i + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }
    Ok(expressions)
}

fn should_preserve(byte: u8, reserved: bool) -> bool {
    byte.is_ascii_alphanumeric()
        || matches!(byte, b'-' | b'.' | b'_' | b'~')
        || (reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            ))
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    b"0123456789ABCDEF"[value as usize] as char
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0usize;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h1), Some(h2)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(hex_digit(h1));
                out.push(hex_digit(h2));
                i += 3;
                continue;
            }
        }
        if should_preserve(b, reserved) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
        i += 1;
    }
    out
}

fn named_field(name: &str, value: String, op: Operator) -> String {
    if op == Operator::Semicolon && value.is_empty() {
        name.to_string()
    } else {
        format!("{name}={value}")
    }
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expression.operator;
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(s) => {
                let value = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                let encoded = encode(&value, op.reserved());
                fields.push(if op.named() {
                    named_field(&spec.name, encoded, op)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for item in items {
                        let encoded = encode(item, op.reserved());
                        fields.push(if op.named() {
                            named_field(&spec.name, encoded, op)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, op.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if op.named() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if entries.is_empty() {
                    continue;
                }
                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, op.reserved());
                        let encoded_value = encode(value, op.reserved());
                        let field = if op == Operator::Semicolon && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, op.reserved()));
                        components.push(encode(value, op.reserved()));
                    }
                    let joined = components.join(",");
                    fields.push(if op.named() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let joined = fields.join(&op.separator().to_string());
    Ok(format!("{}{joined}", op.prefix()))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut out = String::new();
    let mut cursor = 0usize;

    for (start, end, expression) in expressions {
        out.push_str(&template[cursor..start]);
        out.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
