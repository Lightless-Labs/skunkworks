fn parse_expression(content: &str, base: usize, close: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let mut operator = '\0';
    let mut offset = 0;

    if let Some(&first) = bytes.first() {
        if b"+#./;?&".contains(&first) {
            operator = first as char;
            offset = 1;
        } else if b"@$!|".contains(&first) {
            return Err(Error::UnsupportedOperator { pos: base });
        }
    }

    let remaining = &content[offset..];
    if remaining.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut segment_start = 0;

    loop {
        let end = remaining[segment_start..]
            .find(',')
            .map(|n| segment_start + n)
            .unwrap_or(remaining.len());

        if end == segment_start {
            return Err(Error::EmptyVariable {
                pos: base + offset + segment_start,
            });
        }

        let spec = &remaining[segment_start..end];
        let spec_start = base + offset + segment_start;
        let bytes = spec.as_bytes();
        let modifier_index = bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_index.unwrap_or(bytes.len());

        if name_end == 0 {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let name_bytes = &bytes[..name_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(i) = name_bytes.iter().position(|&b| !is_name_continue(b)) {
            return Err(Error::InvalidName {
                pos: spec_start + i,
            });
        }

        let name = spec[..name_end].to_string();
        let modifier = if let Some(i) = modifier_index {
            let suffix = &spec[i..];
            if suffix == "*" {
                Modifier::Explode
            } else if suffix.starts_with(':') {
                let digits = &suffix[1..];
                let valid = (1..=4).contains(&digits.len())
                    && digits.as_bytes().iter().all(u8::is_ascii_digit)
                    && digits.as_bytes()[0] != b'0';
                let value = if valid {
                    digits.parse::<usize>().ok()
                } else {
                    None
                };
                match value {
                    Some(n) if (1..=9999).contains(&n) => Modifier::Prefix(n),
                    _ => {
                        return Err(Error::InvalidModifier {
                            pos: spec_start + i,
                        });
                    }
                }
            } else {
                return Err(Error::InvalidModifier {
                    pos: spec_start + i,
                });
            }
        } else {
            Modifier::None
        };

        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            start: spec_start,
            modifier,
        });

        if end == remaining.len() {
            break;
        }
        segment_start = end + 1;
    }

    Ok(Expression { operator, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || b"-._~".contains(&b);
        let is_reserved = reserved && b":/?#[]@!$&'()*+,;=".contains(&b);
        if unreserved || is_reserved {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: char) -> String {
    if operator == ';' && value.is_empty() {
        name.to_string()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expr.operator;
    let reserved = operator == '+' || operator == '#';
    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if let Modifier::Prefix(_) = spec.modifier {
            if !matches!(value, Value::Scalar(_)) {
                return Err(Error::TypeMismatch { pos: spec.start });
            }
        }

        if let Value::Map(entries) = value {
            let mut keys = Vec::<&str>::new();
            for (key, _) in entries {
                if keys.iter().any(|previous| *previous == key) {
                    return Err(Error::DuplicateKey { pos: spec.start });
                }
                keys.push(key);
            }
        }

        match value {
            Value::Scalar(s) => {
                let value = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                fields.push(if matches!(operator, ';' | '?' | '&') {
                    named_field(&spec.name, encode(&value, reserved), operator)
                } else {
                    encode(&value, reserved)
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(operator, ';' | '?' | '&') {
                            named_field(&spec.name, encoded, operator)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(operator, ';' | '?' | '&') {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let key = encode(key, reserved);
                        let value = encode(value, reserved);
                        let field = if operator == ';' && value.is_empty() {
                            key
                        } else {
                            format!("{key}={value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(operator, ';' | '?' | '&') {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match operator {
        '#' => ("#", ','),
        '.' => (".", '.'),
        '/' => ("/", '/'),
        ';' => (";", ';'),
        '?' => ("?", '&'),
        '&' => ("&", '&'),
        _ => ("", ','),
    };
    Ok(format!("{prefix}{}", fields.join(&separator.to_string())))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut cursor = 0;
    let mut literal_start = 0;
    let mut expressions = Vec::<(usize, usize, Expression)>::new();

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = template[open + 1..].find('}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;
                if let Some(inner) = template[open + 1..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner,
                    });
                }
                let content = &template[open + 1..close];
                let expr = parse_expression(content, open + 1, close)?;
                expressions.push((literal_start, open, expr));
                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut emitted_until = 0;
    for (literal_start, open, expr) in expressions {
        output.push_str(&template[literal_start..open]);
        output.push_str(&expand_expression(&expr, vars)?);
        emitted_until = open;
        // The following literal starts after this expression's closing brace.
        let _ = emitted_until;
    }

    // Append trailing literal, locating its start from the last closing brace.
    let trailing_start = template[..cursor]
        .rfind('}')
        .map(|p| p + 1)
        .unwrap_or(0);
    output.push_str(&template[trailing_start..]);

    Ok(output)
}

                Modifier::Prefix(value)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
        };

        if used.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        used.push(name.clone());
        specifiers.push(Specifier {
            name,
            start: item_start,
            modifier,
        });

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        specifiers,
    })
}

fn encode(input: &str, allow_reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved = allow_reserved
            && matches!(byte, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'=');

        if unreserved || reserved {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn encode_named(name: &str, value: &str, operator: char, allow_reserved: bool) -> String {
    let encoded = encode(value, allow_reserved);
    if operator == ';' && encoded.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={encoded}")
    }
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator.unwrap_or('\0');
    let allow_reserved = matches!(operator, '+' | '#');
    let mut fields = Vec::<String>::new();

    for specifier in &expression.specifiers {
        let Some(value) = selected_value(&specifier.name, vars) else {
            continue;
        };

        if matches!(specifier.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: specifier.start });
        }

        match value {
            Value::Scalar(raw) => {
                let truncated;
                let selected = match specifier.modifier {
                    Modifier::Prefix(limit) => {
                        truncated = raw.chars().take(limit).collect::<String>();
                        truncated.as_str()
                    }
                    _ => raw.as_str(),
                };
                let encoded = encode(selected, allow_reserved);
                if matches!(operator, ';' | '?' | '&') {
                    fields.push(encode_named(&specifier.name, selected, operator, allow_reserved));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let explode = matches!(specifier.modifier, Modifier::Explode);
                if explode {
                    for item in items {
                        if matches!(operator, ';' | '?' | '&') {
                            fields.push(encode_named(&specifier.name, item, operator, allow_reserved));
                        } else {
                            fields.push(encode(item, allow_reserved));
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(operator, ';' | '?' | '&') {
                        fields.push(if operator == ';' && joined.is_empty() {
                            specifier.name.clone()
                        } else {
                            format!("{}={joined}", specifier.name)
                        });
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut seen = Vec::<&str>::new();
                for (key, _) in entries {
                    if seen.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: specifier.start });
                    }
                    seen.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                let explode = matches!(specifier.modifier, Modifier::Explode);
                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(value, allow_reserved);
                        fields.push(if operator == ';' && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else {
                    let components = entries
                        .iter()
                        .flat_map(|(key, value)| {
                            [encode(key, allow_reserved), encode(value, allow_reserved)]
                        })
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(operator, ';' | '?' | '&') {
                        fields.push(if operator == ';' && components.is_empty() {
                            specifier.name.clone()
                        } else {
                            format!("{}={components}", specifier.name)
                        });
                    } else {
                        fields.push(components);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match operator {
        '+' => (",", ""),
        '#' => (",", "#"),
        '.' => (".", "."),
        '/' => ("/", "/"),
        ';' => (";", ";"),
        '?' => ("&", "?"),
        '&' => ("&", "&"),
        _ => (",", ""),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<Expression>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(offset) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + offset;
                if let Some(inner_offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner_offset,
                    });
                }
                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for expression in expressions {
        out.push_str(&template[cursor..expression.start]);
        out.push_str(&expand_expression(&expression, vars)?);
        cursor = expression.end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
