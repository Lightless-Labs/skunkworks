#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    specifiers: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;
    let mut operator = None;

    if content_start < close {
        let first = bytes[content_start] as char;
        if matches!(first, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            operator = Some(first);
            content_start += 1;
        } else if matches!(first, '=' | '!' | '@' | '|' | '$' | '%' | '^' | '(' | ')' | '[' | ']' | '<' | '>' | '\\' | '\'' | '"' | '`') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specifiers = Vec::new();
    let mut used = Vec::<String>::new();
    let mut item_start = content_start;

    loop {
        let item_end = bytes[item_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| item_start + offset)
            .unwrap_or(close);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let modifier_at = bytes[item_start..item_end]
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .map(|offset| item_start + offset)
            .unwrap_or(item_end);

        if modifier_at == item_start {
            return Err(Error::InvalidName { pos: item_start });
        }

        let name_bytes = &bytes[item_start..modifier_at];
        let mut name_error = None;
        for (i, &byte) in name_bytes.iter().enumerate() {
            let valid = if i == 0 {
                is_name_start(byte)
            } else {
                is_name_continue(byte)
            };
            if !valid {
                name_error = Some(item_start + i);
                break;
            }
        }
        if let Some(pos) = name_error {
            return Err(Error::InvalidName { pos });
        }

        let name = template[item_start..modifier_at].to_owned();
        let modifier = if modifier_at == item_end {
            Modifier::None
        } else {
            let suffix = &bytes[modifier_at..item_end];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.first() == Some(&b':')
                && suffix.len() >= 2
                && suffix.len() <= 5
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut value = 0usize;
                for &digit in &suffix[1..] {
                    value = value * 10 + (digit - b'0') as usize;
                }
                if value > 9999 {
                    return Err(Error::InvalidModifier { pos: modifier_at });
                }
                Modifier::Prefix(value)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
        };

        if used.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        used.push(name.clone());
        specifiers.push(Specifier {
            name,
            start: item_start,
            modifier,
        });

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        specifiers,
    })
}

fn encode(input: &str, allow_reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved = allow_reserved
            && matches!(byte, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'=');

        if unreserved || reserved {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn encode_named(name: &str, value: &str, operator: char, allow_reserved: bool) -> String {
    let encoded = encode(value, allow_reserved);
    if operator == ';' && encoded.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={encoded}")
    }
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator.unwrap_or('\0');
    let allow_reserved = matches!(operator, '+' | '#');
    let mut fields = Vec::<String>::new();

    for specifier in &expression.specifiers {
        let Some(value) = selected_value(&specifier.name, vars) else {
            continue;
        };

        if matches!(specifier.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: specifier.start });
        }

        match value {
            Value::Scalar(raw) => {
                let truncated;
                let selected = match specifier.modifier {
                    Modifier::Prefix(limit) => {
                        truncated = raw.chars().take(limit).collect::<String>();
                        truncated.as_str()
                    }
                    _ => raw.as_str(),
                };
                let encoded = encode(selected, allow_reserved);
                if matches!(operator, ';' | '?' | '&') {
                    fields.push(encode_named(&specifier.name, selected, operator, allow_reserved));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let explode = matches!(specifier.modifier, Modifier::Explode);
                if explode {
                    for item in items {
                        if matches!(operator, ';' | '?' | '&') {
                            fields.push(encode_named(&specifier.name, item, operator, allow_reserved));
                        } else {
                            fields.push(encode(item, allow_reserved));
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(operator, ';' | '?' | '&') {
                        fields.push(if operator == ';' && joined.is_empty() {
                            specifier.name.clone()
                        } else {
                            format!("{}={joined}", specifier.name)
                        });
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut seen = Vec::<&str>::new();
                for (key, _) in entries {
                    if seen.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: specifier.start });
                    }
                    seen.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                let explode = matches!(specifier.modifier, Modifier::Explode);
                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(value, allow_reserved);
                        fields.push(if operator == ';' && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else {
                    let components = entries
                        .iter()
                        .flat_map(|(key, value)| {
                            [encode(key, allow_reserved), encode(value, allow_reserved)]
                        })
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(operator, ';' | '?' | '&') {
                        fields.push(if operator == ';' && components.is_empty() {
                            specifier.name.clone()
                        } else {
                            format!("{}={components}", specifier.name)
                        });
                    } else {
                        fields.push(components);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match operator {
        '+' => (",", ""),
        '#' => (",", "#"),
        '.' => (".", "."),
        '/' => ("/", "/"),
        ';' => (";", ";"),
        '?' => ("&", "?"),
        '&' => ("&", "&"),
        _ => (",", ""),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<Expression>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(offset) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + offset;
                if let Some(inner_offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner_offset,
                    });
                }
                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for expression in expressions {
        out.push_str(&template[cursor..expression.start]);
        out.push_str(&expand_expression(&expression, vars)?);
        cursor = expression.end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
