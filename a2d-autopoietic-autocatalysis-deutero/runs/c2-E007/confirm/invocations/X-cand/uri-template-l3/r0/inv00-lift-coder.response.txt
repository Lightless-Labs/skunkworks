#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

struct Expression {
    operator: Operator,
    variables: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (open, close, expression) in expressions {
        output.push_str(&template[cursor..open]);
        expand_expression(&mut output, expression, vars);
        cursor = close + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;

                if let Some(relative_inner) =
                    bytes[open + 1..close].iter().position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_inner,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close, expression));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut start = open + 1;
    let mut operator = Operator::Simple;

    if start < close {
        operator = match bytes[start] {
            b'+' => Operator::Reserved,
            b'#' => Operator::Fragment,
            b'.' => Operator::Dot,
            b'/' => Operator::Slash,
            b';' => Operator::Semicolon,
            b'?' => Operator::Query,
            b'&' => Operator::Continuation,
            _ => Operator::Simple,
        };
        if operator != Operator::Simple {
            start += 1;
        }
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        for pos in spec_start..spec_end {
            let byte = bytes[pos];
            let valid = if pos == spec_start {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName { pos });
            }
        }

        let name = template[spec_start..spec_end].to_string();
        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        variables.push((name, spec_start));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn expand_expression(output: &mut String, expression: Expression, vars: &[(&str, &str)]) {
    let mut fields = Vec::<String>::new();

    for (name, _) in expression.variables {
        if let Some(value) = vars.iter().rev().find_map(|(key, value)| {
            if *key == name {
                Some(*value)
            } else {
                None
            }
        }) {
            let encoded = encode(value, matches!(expression.operator, Operator::Reserved | Operator::Fragment));
            let field = match expression.operator {
                Operator::Semicolon => {
                    if encoded.is_empty() {
                        name
                    } else {
                        format!("{name}={encoded}")
                    }
                }
                Operator::Query | Operator::Continuation => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return;
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };

    output.push_str(prefix);
    output.push_str(&fields.join(separator));
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            let high = bytes[index + 1];
            let low = bytes[index + 2];
            if high.is_ascii_hexdigit() && low.is_ascii_hexdigit() {
                output.push('%');
                output.push((high as char).to_ascii_uppercase());
                output.push((low as char).to_ascii_uppercase());
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + (value - 10)) as char,
    }
}
