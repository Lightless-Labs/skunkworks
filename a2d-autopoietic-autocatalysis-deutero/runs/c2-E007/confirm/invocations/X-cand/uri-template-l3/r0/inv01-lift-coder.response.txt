#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon | Self::Query | Self::Continuation => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continuation => Some('&'),
        }
    }

    fn is_named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continuation)
    }

    fn omit_equals_for_empty(self) -> bool {
        matches!(self, Self::Semicolon)
    }

    fn allows_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

#[derive(Clone)]
struct Variable {
    name: String,
    start: usize,
}

#[derive(Clone)]
struct Expression {
    operator: Operator,
    variables: Vec<Variable>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;

    let operator = if content_start < end {
        match bytes[content_start] {
            b'+' => {
                content_start += 1;
                Operator::Reserved
            }
            b'#' => {
                content_start += 1;
                Operator::Fragment
            }
            b'.' => {
                content_start += 1;
                Operator::Dot
            }
            b'/' => {
                content_start += 1;
                Operator::Slash
            }
            b';' => {
                content_start += 1;
                Operator::Semicolon
            }
            b'?' => {
                content_start += 1;
                Operator::Query
            }
            b'&' => {
                content_start += 1;
                Operator::Continuation
            }
            b'!' | b'=' | b'<' | b'>' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: content_start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut variables = Vec::new();
    let mut seen = Vec::<&str>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(end);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &template[spec_start..spec_end];
        let spec_bytes = spec.as_bytes();
        if !is_name_start(spec_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (offset, &byte) in spec_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        if seen.iter().any(|&name| name == spec) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(spec);
        variables.push(Variable {
            name: spec.to_owned(),
            start: spec_start,
        });

        if spec_end == end {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_upper(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            out.push('%');
            out.push(hex_upper(bytes[i + 1]) as char);
            out.push(hex_upper(bytes[i + 2]) as char);
            i += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0F) as usize] as char);
        }
        i += 1;
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();

    for variable in &expression.variables {
        if let Some(value) = lookup(&variable.name, vars) {
            let encoded = encode(value, expression.operator.allows_reserved());
            if expression.operator.is_named() {
                if expression.operator.omit_equals_for_empty() && encoded.is_empty() {
                    fields.push(variable.name.clone());
                } else {
                    fields.push(format!("{}={}", variable.name, encoded));
                }
            } else {
                fields.push(encoded);
            }
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut out = String::new();
    if let Some(prefix) = expression.operator.prefix() {
        out.push(prefix);
    }
    let separator = expression.operator.separator();
    for (index, field) in fields.iter().enumerate() {
        if index > 0 {
            out.push(separator);
        }
        out.push_str(field);
    }
    out
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let Some(relative_end) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = i + 1 + relative_end;

                if let Some(relative_open) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + relative_open,
                    });
                }

                expressions.push((start, end + 1, parse_expression(template, start, end)?));
                i = end + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for (start, end, expression) in expressions {
        out.push_str(&template[cursor..start]);
        out.push_str(&expand_expression(&expression, vars));
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
