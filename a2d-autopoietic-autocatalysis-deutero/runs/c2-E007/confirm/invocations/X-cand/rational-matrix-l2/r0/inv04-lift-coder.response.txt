```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn signed_from_magnitude(magnitude: u128, negative: bool) -> Option<i64> {
    if negative {
        if magnitude == (i64::MAX as u128) + 1 {
            Some(i64::MIN)
        } else {
            i64::try_from(magnitude).ok().map(|n| -n)
        }
    } else {
        i64::try_from(magnitude).ok()
    }
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let negative = (num < 0) ^ (den < 0);
        let nmag = num.unsigned_abs();
        let dmag = den.unsigned_abs();
        let divisor = gcd_u128(nmag, dmag);
        let nmag = nmag / divisor;
        let dmag = dmag / divisor;

        let normalized_num =
            signed_from_magnitude(nmag, negative).ok_or(RationalError::Overflow)?;
        let normalized_den = i64::try_from(dmag).map_err(|_| RationalError::Overflow)?;

        Ok(Self {
            num: normalized_num,
            den: normalized_den,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let negative = (num < 0) ^ (den < 0);
        let nmag = (num as i128).unsigned_abs();
        let dmag = (den as i128).unsigned_abs();
        let divisor = gcd_u128(nmag, dmag);
        let normalized_num = signed_from_magnitude(nmag / divisor, negative)
            .ok_or(RationalError::Overflow)?;
        let normalized_den =
            i64::try_from(dmag / divisor).map_err(|_| RationalError::Overflow)?;

        Ok(Self {
            num: normalized_num,
            den: normalized_den,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let den = (self.den as i128) * (rhs.den as i128);
        Self::from_i128(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let den = (self.den as i128) * (rhs.den as i128);
        Self::from_i128(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = (self.num as i128) * (rhs.num as i128);
        let den = (self.den as i128) * (rhs.den as i128);
        Self::from_i128(num, den)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let num = (self.num as i128) * (rhs.den as i128);
        let den = (self.den as i128) * (rhs.num as i128);
        Self::from_i128(num, den)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let num = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let mut denominator_start = None;
        let mut denominator_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            denominator_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start.unwrap() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            denominator_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let numerator_digits = numerator_end - numerator_start;
        if numerator_digits >= 2 && bytes[numerator_start] == b'0' {
            return Err(parse_err(
                numerator_start + 1,
                ParseKind::NonCanonical,
            ));
        }
        if negative && numerator_digits == 1 && bytes[numerator_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_err(start + 1, ParseKind::NonCanonical));
            }
        }

        let numerator_magnitude =
            parse_magnitude(&bytes[numerator_start..numerator_end])
                .ok_or_else(|| parse_err(0, ParseKind::OutOfRange))?;
        let numerator_limit = if negative {
            (i64::MAX as u128) + 1
        } else {
            i64::MAX as u128
        };
        if numerator_magnitude > numerator_limit {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }

        let numerator = signed_from_magnitude(numerator_magnitude, negative)
            .ok_or_else(|| parse_err(0, ParseKind::OutOfRange))?;

        let Some(start) = denominator_start else {
            return Ok(Self {
                num: numerator,
                den: 1,
            });
        };
        let end = denominator_end.unwrap();
        let denominator_magnitude = parse_magnitude(&bytes[start..end])
            .ok_or_else(|| parse_err(start, ParseKind::OutOfRange))?;
        if denominator_magnitude > i64::MAX as u128 {
            return Err(parse_err(start, ParseKind::OutOfRange));
        }
        if denominator_magnitude == 0 {
            return Err(parse_err(start, ParseKind::ZeroDenominator));
        }

        let slash = start - 1;
        if denominator_magnitude == 1 {
            return Err(parse_err(slash, ParseKind::NonCanonical));
        }
        if numerator_magnitude == 0 {
            return Err(parse_err(slash, ParseKind::NonCanonical));
        }
        if gcd_u128(numerator_magnitude, denominator_magnitude) > 1 {
            return Err(parse_err(slash, ParseKind::NonCanonical));
        }

        Ok(Self {
            num: numerator,
            den: denominator_magnitude as i64,
        })
    }
}

fn parse_magnitude(digits: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &byte in digits {
        value = value
            .checked_mul(10)?
            .checked_add((byte - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(
        rows: usize,
        cols: usize,
        data: Vec<Rational>,
    ) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        if self.rows == 0 || self.cols == 0 {
            return Self {
                rows: self.cols,
                cols: self.rows,
                data: Vec::new(),
            };
        }

        let mut data = Vec::with_capacity(self.data.len());
        for col in 0..self.cols {
            for row in 0..self.rows {
                data.push(self.data[row * self.cols + col]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);

        if self.rows == 0 || rhs.cols == 0 {
            return Ok(Self {
                rows: self.rows,
                cols: rhs.cols,
                data,
            });
        }

        let zero = Rational { num: 0, den: 1 };
        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut accumulator = zero;
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step,
                            source,
                        }
                    })?;
                    accumulator = accumulator.add(&term).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step,
                            source,
                        }
                    })?;
                }
                data.push(accumulator);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
```
