use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    const ZERO: Self = Self { num: 0, den: 1 };

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let gcd = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / gcd;
        let right_scale = self.den as i128 / gcd;
        let num = (self.num as i128)
            .checked_mul(left_scale)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(right_scale)
                    .and_then(|b| a.checked_add(b))
            })
            .ok_or(RationalError::Overflow)?;
        let den = (self.den as i128)
            .checked_mul(left_scale)
            .ok_or(RationalError::Overflow)?;
        Self::from_i128(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let gcd = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / gcd;
        let right_scale = self.den as i128 / gcd;
        let num = (self.num as i128)
            .checked_mul(left_scale)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(right_scale)
                    .and_then(|b| a.checked_sub(b))
            })
            .ok_or(RationalError::Overflow)?;
        let den = (self.den as i128)
            .checked_mul(left_scale)
            .ok_or(RationalError::Overflow)?;
        Self::from_i128(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(abs_i64(self.num), rhs.den as u128) as i128;
        let g2 = gcd_u128(abs_i64(rhs.num), self.den as u128) as i128;
        let a = self.num as i128 / g1;
        let d = rhs.den as i128 / g1;
        let c = rhs.num as i128 / g2;
        let b = self.den as i128 / g2;
        let num = a.checked_mul(c).ok_or(RationalError::Overflow)?;
        let den = b.checked_mul(d).ok_or(RationalError::Overflow)?;
        Self::from_i128(num, den)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(abs_i64(self.num), abs_i64(rhs.num)) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.num as i128 / g1;
        let c = rhs.num as i128 / g1;
        let b = self.den as i128 / g2;
        let d = rhs.den as i128 / g2;
        let num = a.checked_mul(d).ok_or(RationalError::Overflow)?;
        let den = b.checked_mul(c).ok_or(RationalError::Overflow)?;
        Self::from_i128(num, den)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let num = (self.num as i128)
            .checked_neg()
            .ok_or(RationalError::Overflow)?;
        Self::from_i128(num, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let mut denominator_start = None;
        let mut denominator_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            denominator_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            denominator_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let numerator_digits = numerator_end - numerator_start;
        if numerator_digits >= 2 && bytes[numerator_start] == b'0' {
            return Err(parse_error(
                numerator_start + 1,
                ParseKind::NonCanonical,
            ));
        }
        if negative && numerator_digits == 1 && bytes[numerator_start] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let numerator_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let numerator_magnitude =
            parse_magnitude(&bytes[numerator_start..numerator_end], numerator_limit)
                .ok_or_else(|| parse_error(0, ParseKind::OutOfRange))?;

        let numerator = if negative {
            if numerator_magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        let (denominator, slash_at) =
            if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
                let magnitude =
                    parse_magnitude(&bytes[start..end], i64::MAX as u128)
                        .ok_or_else(|| parse_error(start, ParseKind::OutOfRange))?;
                if magnitude == 0 {
                    return Err(parse_error(start, ParseKind::ZeroDenominator));
                }
                (Some(magnitude as i64), start - 1)
            } else {
                (None, 0)
            };

        if let Some(den) = denominator {
            if den == 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if numerator == 0 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(abs_i64(numerator), den as u128) > 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            Ok(Self { num: numerator, den })
        } else {
            Ok(Self {
                num: numerator,
                den: 1,
            })
        }
    }

    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = num.checked_neg().ok_or(RationalError::Overflow)?;
            den = den.checked_neg().ok_or(RationalError::Overflow)?;
        }
        if num == 0 {
            return Ok(Self::ZERO);
        }

        let gcd = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= gcd;
        den /= gcd;

        if num < i64::MIN as i128
            || num > i64::MAX as i128
            || den < 1
            || den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn abs_i64(value: i64) -> u128 {
    value.unsigned_abs() as u128
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn parse_magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &digit in digits {
        value = value
            .checked_mul(10)?
            .checked_add((digit - b'0') as u128)?;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        if output_len == 0 {
            return Ok(Self {
                rows: self.rows,
                cols: rhs.cols,
                data: Vec::new(),
            });
        }

        let mut data = Vec::with_capacity(output_len);
        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = Rational::ZERO;
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for row in 0..n {
            for col in 0..n {
                data.push(self.data[row * n + col]);
            }
            for col in 0..n {
                data.push(if row == col {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational::ZERO
                });
            }
        }

        let augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let reduction = eliminate(&augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for row in 0..n {
            for col in 0..n {
                inverse_data.push(reduction.matrix.data[row * width + n + col]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }
}

fn eliminate(matrix: &Matrix, candidate_cols: usize) -> Result<Reduction, MatrixError> {
    let mut working = matrix.clone();
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut pivot_row = 0;

    for pivot_col in 0..candidate_cols {
        if pivot_row == working.rows {
            break;
        }

        let mut chosen: Option<usize> = None;
        for row in pivot_row..working.rows {
            let value = working.data[row * working.cols + pivot_col];
            if value.num == 0 {
                continue;
            }
            let replace = match chosen {
                None => true,
                Some(current|) => {
                    let old = working.data[current * working.cols + pivot_col];
                    (value.den, abs_i64(value.num), row)
                        < (old.den, abs_i64(old.num), current)
                }
            };
            if replace {
                chosen = Some(row);
            }
        }

        let Some(chosen_row) = chosen else {
            continue;
        };

        if chosen_row != pivot_row {
            for col in 0..working.cols {
                working
                    .data
                    .swap(pivot_row * working.cols + col, chosen_row * working.cols + col);
            }
            swaps.push((pivot_row, chosen_row));
        }

        let pivot = working.data[pivot_row * working.cols + pivot_col];
        for col in 0..working.cols {
            let index = pivot_row * working.cols + col;
            working.data[index] =
                working.data[index]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: pivot_row,
                        col,
                        step: pivot_row,
                        source,
                    })?;
        }

        for row in 0..working.rows {
            if row == pivot_row {
                continue;
            }
            let factor = working.data[row * working.cols + pivot_col];
            if factor.num == 0 {
                continue;
            }
            for col in 0..working.cols {
                let pivot_entry = working.data[pivot_row * working.cols + col];
                let product = factor.mul(&pivot_entry).map_err(|source| {
                    MatrixError::Arithmetic {
                        row,
                        col,
                        step: pivot_row,
                        source,
                    }
                })?;
                let index = row * working.cols + col;
                working.data[index] =
                    working.data[index]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col,
                            step: pivot_row,
                            source,
                        })?;
            }
        }

        pivots.push(pivot_col);
        pivot_row += 1;
    }

    Ok(Reduction {
        matrix: working,
        pivots,
        swaps,
    })
}
