#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn ascii_pos(map: &[usize], index: usize, input_len: usize) -> usize {
    map.get(index).copied().unwrap_or(input_len)
}

fn valid_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    map: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::with_capacity(input.len());
    let mut map = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

    let mut i = 0usize;
    let mut comment_stack: Vec<usize> = Vec::new();
    let mut quote_mode = false;
    let mut first_retained = true;

    while i < input.len() {
        if quote_mode {
            let b = input[i];
            out.push(b);
            map.push(i);
            first_retained = false;
            if b == b'\\' {
                if i + 1 < input.len() {
                    out.push(input[i + 1]);
                    map.push(i + 1);
                    i += 2;
                    continue;
                }
                i += 1;
                continue;
            }
            if b == b'"' {
                quote_mode = false;
            }
            i += 1;
            continue;
        }

        if !comment_stack.is_empty() {
            let b = input[i];
            if b == b'\\' {
                if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = valid_fold(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_stack.len() == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_stack.push(i);
                i += 1;
                continue;
            }
            if b == b')' {
                comment_stack.pop();
                i += 1;
                continue;
            }
            if !(is_printable(b) || b == b'\t') {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let b = input[i];

        if first_retained && b == b'"' {
            quote_mode = true;
            out.push(b);
            map.push(i);
            first_retained = false;
            i += 1;
            continue;
        }

        if b == b'(' {
            let run_start = i;
            comment_stack.push(i);
            first_retained = false;
            i += 1;
            while i < input.len() && !comment_stack.is_empty() {
                let c = input[i];
                if c == b'\\' {
                    if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                } else if c == b'\r' || c == b'\n' {
                    if let Some(end) = valid_fold(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'(' {
                    if comment_stack.len() == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    comment_stack.push(i);
                    i += 1;
                } else if c == b')' {
                    comment_stack.pop();
                    i += 1;
                } else if is_printable(c) || c == b'\t' {
                    i += 1;
                } else {
                    return Err(EmailError::CommentChar { at: i });
                }
            }
            if !comment_stack.is_empty() {
                return Err(EmailError::UnclosedComment {
                    at: comment_stack[0],
                });
            }
            runs.push((run_start, i));
            continue;
        }

        if is_hws(b) || b == b'\r' || b == b'\n' {
            let start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                if is_hws(input[i]) {
                    i += 1;
                } else if input[i] == b'\r' || input[i] == b'\n' {
                    if let Some(end) = valid_fold(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if input[i] == b'(' {
                    i += 1;
                    comment_stack.push(i - 1);
                    while i < input.len() && !comment_stack.is_empty() {
                        let c = input[i];
                        if c == b'\\' {
                            if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if c == b'\r' || c == b'\n' {
                            if let Some(end) = valid_fold(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if c == b'(' {
                            if comment_stack.len() == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            comment_stack.push(i);
                            i += 1;
                        } else if c == b')' {
                            comment_stack.pop();
                            i += 1;
                        } else if is_printable(c) || c == b'\t' {
                            i += 1;
                        } else {
                            return Err(EmailError::CommentChar { at: i });
                        }
                    }
                    if !comment_stack.is_empty() {
                        return Err(EmailError::UnclosedComment {
                            at: comment_stack[0],
                        });
                    }
                } else {
                    break;
                }
            }
            runs.push((start, i));
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        out.push(b);
        map.push(i);
        first_retained = false;
        i += 1;
    }

    if !comment_stack.is_empty() {
        return Err(EmailError::UnclosedComment {
            at: comment_stack[0],
        });
    }

    Ok(Cleaned {
        bytes: out,
        map,
        runs,
    })
}

fn position(cleaned: &Cleaned, i: usize, input_len: usize) -> usize {
    ascii_pos(&cleaned.map, i, input_len)
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn validate_domain(
    bytes: &[u8],
    map: &[usize],
    input_len: usize,
    allow_single_label: bool,
) -> Result<String, EmailError> {
    let start = 0usize;
    if bytes.is_empty() {
        return Err(EmailError::EmptyDomain {
            at: ascii_pos(map, start, input_len),
        });
    }

    let mut end = bytes.len();
    let root_pos = if bytes[end - 1] == b'.' {
        end -= 1;
        Some(ascii_pos(map, end, input_len))
    } else {
        None
    };

    if end == 0 {
        return Err(EmailError::EmptyDomain {
            at: root_pos.unwrap_or(input_len),
        });
    }
    if end > 253 {
        return Err(EmailError::DomainTooLong {
            at: ascii_pos(map, 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0usize;
    for j in 0..=end {
        if j == end || bytes[j] == b'.' {
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for (idx, &(a, b)) in labels.iter().enumerate() {
        if a == b {
            let at = if idx + 1 == labels.len() {
                root_pos.unwrap_or_else(|| ascii_pos(map, a, input_len))
            } else {
                ascii_pos(map, a, input_len)
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if b - a > 63 {
            return Err(EmailError::LabelTooLong {
                at: ascii_pos(map, a + 63, input_len),
            });
        }
        if bytes[a] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: ascii_pos(map, a, input_len),
            });
        }
        if bytes[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: ascii_pos(map, b - 1, input_len),
            });
        }
        for j in a..b {
            if !(bytes[j].is_ascii_alphanumeric() || bytes[j] == b'-') {
                return Err(EmailError::DomainChar {
                    at: ascii_pos(map, j, input_len),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: ascii_pos(map, start, input_len),
        });
    }

    let (ta, tb) = *labels.last().unwrap();
    if bytes[ta..tb].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: ascii_pos(map, ta, input_len),
        });
    }

    Ok(bytes[..end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect())
}

fn parse_impl(input: &str, policy: Option<&Policy>) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    for &(a, b) in &cleaned.runs {
        let left = a == 0;
        let right = b == raw.len();
        let preceding_at = a > 0 && cleaned.bytes.get(a - 1) == Some(&b'@');
        let following_at = cleaned.bytes.get(a).or_else(|| cleaned.bytes.get(b.saturating_sub(1)));
        let adjacent_at = preceding_at || following_at == Some(&b'@');
        if !left && !right && !adjacent_at {
            return Err(EmailError::CfwsBoundary { at: a });
        }
    }

    if cleaned.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let bytes = &cleaned.bytes;
    let mut quote_end = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i = (i + 2).min(bytes.len());
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: position(&cleaned, 0, raw.len()),
            });
        }
    }

    let protected_end = quote_end.map(|v| v + 1).unwrap_or(0);
    let ats: Vec<usize> = (protected_end..bytes.len())
        .filter(|&i| bytes[i] == b'@')
        .collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: position(&cleaned, ats[1], raw.len()),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: position(&cleaned, sep, raw.len()),
        });
    }

    let local_bytes = &bytes[..sep];
    let mut local = Vec::<u8>::new();
    let mut origins = Vec::<usize>::new();
    let quoted;

    if local_bytes[0] != b'"' {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: position(&cleaned, 64, raw.len()),
            });
        }
        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes.get(i.wrapping_sub(1)) == Some(&b'.')
                {
                    return Err(EmailError::LocalDot {
                        at: position(&cleaned, i, raw.len()),
                    });
                }
            } else if !is_atom_char(b) {
                return Err(EmailError::LocalChar {
                    at: position(&cleaned, i, raw.len()),
                });
            }
            local.push(b);
            origins.push(position(&cleaned, i, raw.len()));
        }
        quoted = false;
    } else {
        let close = quote_end.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: position(&cleaned, close + 1, raw.len()),
            });
        }
        let mut i = 1usize;
        while i < close {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape {
                        at: position(&cleaned, i, raw.len()),
                    });
                }
                let escaped = local_bytes[i + 1];
                if ![b'"', b'\\', b'@'].contains(&escaped) {
                    return Err(EmailError::BadEscape {
                        at: position(&cleaned, i, raw.len()),
                    });
                }
                local.push(escaped);
                origins.push(position(&cleaned, i, raw.len()));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: position(&cleaned, i, raw.len()),
                    });
                }
                local.push(b);
                origins.push(position(&cleaned, i, raw.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: position(&cleaned, close, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        quoted = true;
    }

    let domain_start = sep + 1;
    let domain = validate_domain(
        &bytes[domain_start..],
        &cleaned.map[domain_start..],
        raw.len(),
        policy.map(|p| p.allow_single_label).unwrap_or(false),
    )?;

    let mut final_local = local;
    let mut final_origins = origins;
    if let Some(p) = policy {
        if p.local_case == LocalCase::Lower {
            for b in &mut final_local {
                *b = b.to_ascii_lowercase();
            }
        }
        if p.strip_subaddress {
            if let Some(plus) = final_local.iter().position(|&b| b == b'+') {
                if plus == 0 {
                    return Err(EmailError::EmptyNormalizedLocal {
                        at: final_origins[plus],
                    });
                }
                final_local.truncate(plus);
                final_origins.truncate(plus);
            }
        }
    }

    let safe_atom = !final_local.is_empty()
        && final_local.iter().enumerate().all(|(i, &b)| {
            if b == b'.' {
                i != 0
                    && i + 1 != final_local.len()
                    && final_local.get(i - 1) != Some(&b'.')
            } else {
                is_atom_char(b)
            }
        });

    let final_quoted = if !safe_atom {
        true
    } else if policy.map(|p| p.unquote_safe).unwrap_or(false) {
        false
    } else {
        quoted
    };

    Ok(Address {
        local: final_local.into_iter().map(char::from).collect(),
        domain,
        quoted: final_quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_impl(input, None)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_impl(input, Some(policy))
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: std::collections::HashMap<(String, String), usize> =
        std::collections::HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = keys.get(&key) {
                    result.duplicate_of[index] = Some(first);
                } else {
                    keys.insert(key, index);
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }

    result
}
