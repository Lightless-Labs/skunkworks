#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut quote_protected = false;

    while i < input.len() {
        let b = input[i];

        if quote_protected {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    bytes.push(input[i + 1]);
                    positions.push(i + 1);
                    i += 2;
                    continue;
                }
            } else if b == b'"' {
                quote_protected = false;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if b == b'\r'
                    && i + 2 < input.len()
                    && input[i + 1] == b'\n'
                    && is_hws(input[i + 2])
                {
                    i += 2;
                    while i < input.len() && is_hws(input[i]) {
                        i += 1;
                    }
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !is_printable(b) && !is_hws(b) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let is_cfws_start = is_hws(b)
            || b == b'('
            || b == b'\r'
            || b == b'\n';

        if is_cfws_start {
            let start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if is_hws(c) {
                    i += 1;
                } else if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && i + 2 < input.len()
                        && input[i + 1] == b'\n'
                        && is_hws(input[i + 2])
                    {
                        i += 2;
                        while i < input.len() && is_hws(input[i]) {
                            i += 1;
                        }
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'(' {
                    depth = 1;
                    outer_open = i;
                    i += 1;
                    while i < input.len() && depth > 0 {
                        let x = input[i];
                        if x == b'\\' {
                            if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if x == b'\r' || x == b'\n' {
                            if x == b'\r'
                                && i + 2 < input.len()
                                && input[i + 1] == b'\n'
                                && is_hws(input[i + 2])
                            {
                                i += 2;
                                while i < input.len() && is_hws(input[i]) {
                                    i += 1;
                                }
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if x == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            i += 1;
                        } else if x == b')' {
                            depth -= 1;
                            i += 1;
                        } else if !is_printable(x) && !is_hws(x) {
                            return Err(EmailError::CommentChar { at: i });
                        } else {
                            i += 1;
                        }
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer_open });
                    }
                } else {
                    break;
                }
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if bytes.is_empty() && b == b'"' {
            quote_protected = true;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn mapped(pos: &[usize], idx: usize, input_len: usize) -> usize {
    pos.get(idx).copied().unwrap_or(input_len)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if b.is_ascii_alphanumeric()
            || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_internal(
    input: &str,
    policy: Option<&Policy>,
) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let b = &cleaned.bytes;
    let pos = &cleaned.positions;

    for run in &cleaned.runs {
        let left_at = run.start > 0 && b[run.start - 1] == b'@';
        let right_at = run.end < b.len() && b[run.end] == b'@';
        if run.start != 0 && run.end != b.len() && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    let quoted_prefix = b[0] == b'"';
    if quoted_prefix {
        let mut i = 1;
        let mut close = None;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match close {
            Some(c) => protected_end = c + 1,
            None => return Err(EmailError::UnclosedQuote { at: mapped(pos, 0, raw.len()) }),
        }
    }

    let separators: Vec<usize> = (0..b.len())
        .filter(|&i| b[i] == b'@' && (!quoted_prefix || i >= protected_end))
        .collect();
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: mapped(pos, separators[1], raw.len()),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: mapped(pos, sep, raw.len()),
        });
    }

    let local_bytes = &b[..sep];
    let mut local = String::new();
    let mut quoted = false;
    let mut local_origins = Vec::new();

    if local_bytes[0] == b'"' {
        quoted = true;
        let close = protected_end - 1;
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: mapped(pos, close + 1, raw.len()),
            });
        }

        let mut i = 1;
        while i < close {
            let c = local_bytes[i];
            if c == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape {
                        at: mapped(pos, i, raw.len()),
                    });
                }
                let next = local_bytes[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: mapped(pos, i, raw.len()),
                    });
                }
                local.push(next as char);
                local_origins.push(mapped(pos, i, raw.len()));
                i += 2;
            } else {
                if !is_printable(c) {
                    return Err(EmailError::QuotedChar {
                        at: mapped(pos, i, raw.len()),
                    });
                }
                local.push(c as char);
                local_origins.push(mapped(pos, i, raw.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: mapped(pos, close, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: mapped(pos, 64, raw.len()),
            });
        }
        let mut prev_dot = false;
        for (i, &c) in local_bytes.iter().enumerate() {
            if c == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || prev_dot {
                    return Err(EmailError::LocalDot {
                        at: mapped(pos, i, raw.len()),
                    });
                }
                prev_dot = true;
            } else {
                if !(c.is_ascii_alphanumeric()
                    || b"!#$%&'*+-/=?^_`{|}~".contains(&c))
                {
                    return Err(EmailError::LocalChar {
                        at: mapped(pos, i, raw.len()),
                    });
                }
                prev_dot = false;
            }
            local.push(c as char);
            local_origins.push(mapped(pos, i, raw.len()));
        }
    }

    let domain_start = sep + 1;
    let mut domain_end = b.len();
    let mut root_position = None;
    if domain_end > domain_start && b[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_position = Some(domain_end);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: mapped(pos, domain_start, raw.len()),
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: mapped(pos, domain_start + 253, raw.len()),
        });
    }

    let domain = &b[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }

    for (start, end) in &labels {
        if start == end {
            let idx = if *start == 0 {
                domain_start
            } else if *start == domain.len() {
                root_position.unwrap_or(domain_start + *start)
            } else {
                domain_start + *start - 1
            };
            return Err(EmailError::EmptyLabel {
                at: mapped(pos, idx, raw.len()),
            });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: mapped(pos, domain_start + start + 63, raw.len()),
            });
        }
        let label = &domain[*start..*end];
        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(pos, domain_start + *start, raw.len()),
            });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(pos, domain_start + end - 1, raw.len()),
            });
        }
        for (j, &c) in label.iter().enumerate() {
            if !(c.is_ascii_alphanumeric() || c == b'-') {
                return Err(EmailError::DomainChar {
                    at: mapped(pos, domain_start + start + j, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2
        && !policy.map_or(false, |p| p.allow_single_label)
    {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(pos, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    let last_label = &domain[last_start..last_end];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: mapped(pos, domain_start + last_start, raw.len()),
        });
    }

    let mut domain_string = String::with_capacity(domain_len);
    for &c in domain {
        domain_string.push((c as char).to_ascii_lowercase());
    }

    if let Some(p) = policy {
        if p.local_case == LocalCase::Lower {
            local.make_ascii_lowercase();
        }

        let mut origins = local_origins;
        if p.strip_subaddress {
            if let Some(plus) = local.as_bytes().iter().position(|&c| c == b'+') {
                if plus == 0 {
                    return Err(EmailError::EmptyNormalizedLocal {
                        at: origins.first().copied().unwrap_or(mapped(pos, 0, raw.len())),
                    });
                }
                local.truncate(plus);
                origins.truncate(plus);
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyNormalizedLocal {
                at: origins.first().copied().unwrap_or(mapped(pos, 0, raw.len())),
            });
        }

        if !valid_dot_atom(local.as_bytes()) {
            quoted = true;
        } else if p.unquote_safe {
            quoted = false;
        }
    }

    Ok(Address {
        local,
        domain: domain_string,
        quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, None)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_internal(input, Some(policy))
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut first_by_key: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => errors.push((i, e)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = first_by_key.get(&key) {
                    duplicate_of[i] = Some(first);
                } else {
                    first_by_key.insert(key, i);
                    source_indices.push(i);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
