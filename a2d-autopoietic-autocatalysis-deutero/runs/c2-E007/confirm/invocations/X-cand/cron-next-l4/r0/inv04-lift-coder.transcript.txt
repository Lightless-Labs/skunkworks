fn parse_number(
    field: Field,
    s: &str,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut n = 0u32;
    for b in s.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    let (lo, hi) = bounds(field);
    if step {
        let width = (hi - lo + 1) as u32;
        if n == 0 || n > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else if n < lo as u32 || n > hi as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(n)
}

fn parse_field(field: Field, text: &str) -> Result<BTreeSet<u8>, ParseError> {
    for (offset, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (lo, hi) = bounds(field);
    let mut selected = BTreeSet::new();
    let mut item_start = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_parts: Vec<&str> = item.split('/').collect();
        if slash_parts.len() > 2 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let base = slash_parts[0];
        let step_text = if slash_parts.len() == 2 {
            Some(slash_parts[1])
        } else {
            None
        };

        let is_star = base == "*";
        let range_parts: Vec<&str> = if is_star {
            vec!["*"]
        } else {
            base.split('-').collect()
        };

        if (is_star && base != "*")
            || (!is_star && !(range_parts.len() == 1 || range_parts.len() == 2))
            || range_parts.iter().any(|part| part.is_empty() || *part == "*")
            || step_text.is_some_and(|s| !is_digits(s))
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        // Check all base atoms before reading any numeric values, so malformed
        // endpoints take precedence over range and step numeric errors.
        if !is_star && range_parts.iter().any(|part| !valid_atom(field, part)) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let mut start = lo;
        let mut end = hi;
        if !is_star {
            start = parse_atom(field, range_parts[0], item_start)?;
            end = start;
            if range_parts.len() == 2 {
                let end_offset = item_start + range_parts[0].len() + 1;
                end = parse_atom(field, range_parts[1], end_offset)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
            }
        }

        let step = if let Some(s) = step_text {
            let offset = item_start + base.len() + 1;
            Some(parse_number(field, s, offset, true)? as u8)
        } else {
            None
        };

        if !is_star && range_parts.len() == 1 && step.is_some() {
            end = hi;
        }

        let mut value = start;
        while value <= end {
            selected.insert(value);
            match value.checked_add(stride) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        item_start += item.len() + 1;
    }

    Ok(selected)
}
