#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_day_of_month: bool,
    unrestricted_day_of_week: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn parse_atom(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    let b = s.as_bytes();
    let is_digits = !b.is_empty() && b.iter().all(u8::is_ascii_digit);
    if is_digits {
        if b.len() > 1 && b[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut n = 0u32;
        for &d in b {
            n = n
                .checked_mul(10)
                .and_then(|v| v.checked_add((d - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let (lo, hi) = bounds(field);
        if n < lo || n > hi {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok(n);
    }

    if let Some(list) = names(field) {
        let upper = s.to_ascii_uppercase();
        if let Some(pos) = list.iter().position(|name| *name == upper) {
            return Ok(pos as u32 + if field == Field::Month { 1 } else { 0 });
        }
    }

    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_step(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    let b = s.as_bytes();
    if b.is_empty() || !b.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if b.len() > 1 && b[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for &d in b {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((d - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (lo, hi) = bounds(field);
    let width = hi - lo + 1;
    if n == 0 || n > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(n)
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (lo, hi) = bounds(field);
    let mut selected = vec![false; (hi + 1) as usize];

    let mut start = 0usize;
    loop {
        let end = text[start..]
            .find(',')
            .map(|n| start + n)
            .unwrap_or(text.len());
        let item = &text[start..end];
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: start,
            });
        }

        let slash_positions: Vec<usize> = item.match_indices('/').map(|(i, _)| i).collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }

        let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), start + slash + 1)
        } else {
            (item, None, 0)
        };

        let hyphen_positions: Vec<usize> = base.match_indices('-').map(|(i, _)| i).collect();
        let is_star = base == "*";
        let structure_ok = if is_star {
            hyphen_positions.is_empty()
        } else {
            match hyphen_positions.as_slice() {
                [] => !base.is_empty(),
                [h] => *h > 0 && *h + 1 < base.len(),
                _ => false,
            }
        };
        if !structure_ok || step_text == Some("") {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }

        let (left, right) = if let Some(&h) = hyphen_positions.first() {
            (&base[..h], Some((&base[h + 1..], start + h + 1)))
        } else {
            (base, None)
        };

        let mut left_num = None;
        let mut right_num = None;
        if !is_star {
            left_num = Some(parse_atom(field, left, start)?);
            if let Some((r, roff)) = right {
                right_num = Some(parse_atom(field, r, roff)?);
                if left_num.unwrap() > right_num.unwrap() {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: roff,
                    });
                }
            }
        }

        let step = if let Some(s) = step_text {
            Some(parse_step(field, s, step_offset)?)
        } else {
            None
        };

        let (range_start, range_end) = if is_star {
            (lo, hi)
        } else if let Some(r) = right_num {
            (left_num.unwrap(), r)
        } else if step.is_some() {
            (left_num.unwrap(), hi)
        } else {
            (left_num.unwrap(), left_num.unwrap())
        };
        let stride = step.unwrap_or(1);
        let mut value = range_start;
        while value <= range_end {
            selected[value as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        if end == text.len() {
            break;
        }
        start = end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let input = if input.as_bytes().first() == Some(&b'@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets = Vec::with_capacity(5);
    for (i, part) in parts.iter().enumerate() {
        sets.push(parse_field(FIELDS[i], part)?);
    }

    Ok(Cron {
        selected: sets.try_into().ok().unwrap(),
        unrestricted_day_of_month: parts[2] == "*",
        unrestricted_day_of_week: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| if yes { Some(value as u8) } else { None })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = after;
        loop {
            let hours = self.allowed(Field::Hour);
            let minutes = self.allowed(Field::Minute);
            for hour in hours {
                for minute in &minutes {
                    let candidate = CivilDateTime { hour, minute: *minute, ..date };
                    if candidate > after && self.matches_valid(candidate) {
                        return Ok(Some(candidate));
                    }
                }
            }

            match next_date(date) {
                Some(next) => date = next,
                None => return Ok(None),
            }
            date.hour = 0;
            date.minute = 0;
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let hours = self.allowed(Field::Hour);
        let minutes = self.allowed(Field::Minute);
        let mut date = before;
        loop {
            for &hour in hours.iter().rev() {
                for &minute in minutes.iter().rev() {
                    let candidate = CivilDateTime { hour, minute, ..date };
                    if candidate < before && self.matches_valid(candidate) {
                        return Ok(Some(candidate));
                    }
                }
            }
            match previous_date(date) {
                Some(prev) => date = prev,
                None => return Ok(None),
            }
            date.hour = 23;
            date.minute = 59;
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        if start == end {
            return Ok(FireWindow {
                fires: Vec::new(),
                truncated: false,
            });
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(next) = self.next_fire(cursor).map_err(|_| WindowError::InvalidStart {
            component: DatePart::Year,
        })? {
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(next);
            cursor = next;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut parts = Vec::new();
        for field in FIELDS {
            let i = field_index(field);
            let set = &self.selected[i];
            let (lo, hi) = bounds(field);
            let unrestricted = match field {
                Field::DayOfMonth => self.unrestricted_day_of_month,
                Field::DayOfWeek => self.unrestricted_day_of_week,
                _ => false,
            };
            let full = (lo..=hi).all(|n| set[n as usize]);
            let rendered = if unrestricted {
                "*".to_string()
            } else if full {
                match field {
                    Field::Minute | Field::Hour | Field::Month => "*".to_string(),
                    Field::DayOfMonth | Field::DayOfWeek => "*/1".to_string(),
                }
            } else {
                (lo..=hi)
                    .filter(|&n| set[n as usize])
                    .map(|n| n.to_string())
                    .collect::<Vec<_>>()
                    .join(",")
            };
            parts.push(rendered);
        }
        parts.join(" ")
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let weekday = weekday(at.year, at.month, at.day);
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom_unrestricted = self.unrestricted_day_of_month;
        let dow_unrestricted = self.unrestricted_day_of_week;
        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday as usize];

        match (dom_unrestricted, dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn next_date(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
    } else if at.month < 12 {
        at.month += 1;
        at.day = 1;
    } else if at.year < 2399 {
        at.year += 1;
        at.month = 1;
        at.day = 1;
    } else {
        return None;
    }
    Some(at)
}

fn previous_date(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.day > 1 {
        at.day -= 1;
    } else if at.month > 1 {
        at.month -= 1;
        at.day = days_in_month(at.year, at.month);
    } else if at.year > 2000 {
        at.year -= 1;
        at.month = 12;
        at.day = 31;
    } else {
        return None;
    }
    Some(at)
}
