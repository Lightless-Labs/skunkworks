#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn is_invalid(c: char) -> bool {
    c == '\r' || c == '\u{2028}' || c == '\u{2029}' || (c.is_control() && c != '\n')
}

fn normalize_lines(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for c in trimmed.chars() {
                if markdown && c == '|' {
                    tokens.push(Token { text: "\\|".to_owned(), width: 2 });
                } else if markdown && c == '\\' {
                    tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
                } else {
                    tokens.push(Token { text: c.to_string(), width: 1 });
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }
    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let limit = width.saturating_sub(1);
            let mut used = 0;
            let mut out = String::new();
            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut index = 0;
            while index < tokens.len() {
                let mut used = 0;
                let start = index;
                while index < tokens.len() && used + tokens[index].width <= width {
                    used += tokens[index].width;
                    index += 1;
                }
                result.push(tokens[start..index].iter().map(|t| t.text.as_str()).collect());
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn pad(content: &str, width: usize, alignment: &Alignment) -> (String, String) {
    let extra = width.saturating_sub(scalar_len(content));
    match alignment {
        Alignment::Left => (content.to_owned(), " ".repeat(extra)),
        Alignment::Right => (" ".repeat(extra), content.to_owned()),
        Alignment::Center => {
            let left = extra / 2;
            (" ".repeat(left), format!("{}{}", content, " ".repeat(extra - left)))
        }
    }
}

fn border(widths: &[usize]) -> String {
    let mut s = String::from("+");
    for &w in widths {
        s.push_str(&"-".repeat(w + 2));
        s.push('+');
    }
    s.push('\n');
    s
}

fn data_line(contents: &[String], widths: &[usize], aligns: &[Alignment], mode: &Mode) -> String {
    if matches!(mode, Mode::Compact) {
        let mut out = String::new();
        for i in 0..contents.len() {
            if i > 0 {
                out.push_str("  ");
            }
            let (left, right) = pad(&contents[i], widths[i], &aligns[i]);
            out.push_str(&left);
            out.push_str(&right);
            if i + 1 == contents.len() {
                // Last-column right alignment padding is omitted.
                if matches!(aligns[i], Alignment::Left | Alignment::Center) {
                    let extra = widths[i].saturating_sub(scalar_len(&contents[i]));
                    let right_len = match aligns[i] {
                        Alignment::Left => extra,
                        Alignment::Center => extra - extra / 2,
                        Alignment::Right => 0,
                    };
                    let keep = out.chars().count().saturating_sub(right_len);
                    out = out.chars().take(keep).collect();
                }
            }
        }
        out.push('\n');
        return out;
    }

    let mut out = String::from("|");
    for i in 0..contents.len() {
        let (left, right) = pad(&contents[i], widths[i], &aligns[i]);
        out.push(' ');
        out.push_str(&left);
        out.push_str(&right);
        out.push_str(" |");
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged { row, expected: n, actual: cells.len() });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount { expected: n, actual: options.alignments.len() });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount { expected: n, actual: options.overflow.len() });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let overhead = match options.mode {
        Mode::Compact => 2 * n.saturating_sub(1),
        _ => 3 * n + 1,
    };
    let min_col = if matches!(options.mode, Mode::Markdown) { 3 } else { 1 };
    let minimum = overhead + n * min_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    for (r, cells) in rows.iter().enumerate() {
        for (c, cell) in cells.iter().enumerate() {
            let mut byte = 0;
            let chars: Vec<char> = cell.chars().collect();
            let mut i = 0;
            while i < chars.len() {
                let ch = chars[i];
                if ch == '\r' {
                    if i + 1 >= chars.len() || chars[i + 1] != '\n' {
                        return Err(TableError::InvalidChar { row: r, column: c, byte });
                    }
                    byte += 1;
                    i += 1;
                } else if is_invalid(ch) {
                    return Err(TableError::InvalidChar { row: r, column: c, byte });
                }
                byte += ch.len_utf8();
                i += 1;
            }
        }
    }

    let aligns: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };
    let markdown = matches!(options.mode, Mode::Markdown);

    let normalized: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| normalize_lines(cell, markdown)).collect())
        .collect();

    let mut widths = vec![min_col; n];
    for row in &normalized {
        for (c, cell_lines) in row.iter().enumerate() {
            for line in cell_lines {
                widths[c] = widths[c].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&c| widths[c] > min_col)
                .max_by_key(|&c| (widths[c], c));
            match candidate {
                Some(c) => widths[c] -= 1,
                None => break,
            }
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }

    for (r, row) in normalized.iter().enumerate() {
        let cell_fragments: Vec<Vec<String>> = row
            .iter()
            .enumerate()
            .map(|(c, lines)| {
                lines
                    .iter()
                    .flat_map(|line| fragments(line, widths[c], &overflows[c]))
                    .collect()
            })
            .collect();
        let height = cell_fragments.iter().map(Vec::len).max().unwrap_or(1);

        for i in 0..height {
            let contents: Vec<String> = cell_fragments
                .iter()
                .map(|frags| frags.get(i).cloned().unwrap_or_default())
                .collect();
            output.push_str(&data_line(&contents, &widths, &aligns, &options.mode));
        }

        if markdown && options.header && r == 0 {
            output.push('|');
            for c in 0..n {
                output.push(' ');
                let marker = match aligns[c] {
                    Alignment::Left => format!(":{}", "-".repeat(widths[c] - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(widths[c] - 1)),
                    Alignment::Center => format!(":{}:", "-".repeat(widths[c] - 2)),
                };
                output.push_str(&marker);
                output.push_str(" |");
            }
            output.push('\n');
        }

        if matches!(options.mode, Mode::Grid) && options.header && r == 0 && r + 1 < rows.len() {
            output.push_str(&border(&widths));
        }
    }

    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
