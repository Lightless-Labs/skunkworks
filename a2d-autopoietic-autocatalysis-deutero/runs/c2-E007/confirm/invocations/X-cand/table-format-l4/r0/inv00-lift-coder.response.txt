#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn invalid_char(cell: &str) -> Option<usize> {
    let chars: Vec<(usize, char)> = cell.char_indices().collect();
    let mut i = 0;
    while i < chars.len() {
        let (byte, ch) = chars[i];
        if ch == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 2;
                continue;
            }
            return Some(byte);
        }
        if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
            return Some(byte);
        }
        i += 1;
    }
    None
}

fn logical_lines(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            trimmed
                .chars()
                .map(|ch| {
                    if markdown && ch == '|' {
                        Token {
                            text: "\\|".to_owned(),
                            width: 2,
                        }
                    } else if markdown && ch == '\\' {
                        Token {
                            text: "\\\\".to_owned(),
                            width: 2,
                        }
                    } else {
                        Token {
                            text: ch.to_string(),
                            width: 1,
                        }
                    }
                })
                .collect()
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragment_text(tokens: &[Token]) -> String {
    tokens.iter().map(|t| t.text.as_str()).collect()
}

fn split_line(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![fragment_text(tokens)];
            }
            let mut used = 0;
            let mut end = 0;
            for (i, token) in tokens.iter().enumerate() {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                end = i + 1;
            }
            let mut result = fragment_text(&tokens[..end]);
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut used = 0;
                let mut end = start;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // All tokens have width at most two and Markdown's minimum is three.
                if end == start {
                    end += 1;
                }
                result.push(fragment_text(&tokens[start..end]));
                start = end;
            }
            result
        }
    }
}

fn pad(text: &str, content_width: usize, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::new();
    out.extend(std::iter::repeat_n(' ', left));
    out.push_str(text);
    out.extend(std::iter::repeat_n(' ', right));
    out
}

fn render_data_line(
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut out = String::new();
    if !matches!(mode, Mode::Compact) {
        out.push('|');
    }

    for col in 0..widths.len() {
        let text = fragments.get(col).map(String::as_str).unwrap_or("");
        let content_width = text.chars().count();
        let aligned = pad(text, content_width, widths[col], &alignments[col]);

        if matches!(mode, Mode::Compact) {
            if col > 0 {
                out.push_str("  ");
            }
            if col + 1 == widths.len() {
                let keep = widths[col].saturating_sub(
                    widths[col].saturating_sub(content_width),
                );
                // Omit only alignment padding on the right in the final column.
                let target_len = match alignments[col] {
                    Alignment::Left => content_width,
                    Alignment::Right | Alignment::Center => widths[col],
                };
                let _ = keep;
                out.push_str(&aligned.chars().take(target_len).collect::<String>());
            } else {
                out.push_str(&aligned);
            }
        } else {
            out.push(' ');
            out.push_str(&aligned);
            out.push_str(" |");
        }
    }
    out.push('\n');
    out
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::from("|");
    for (width, alignment) in widths.iter().zip(alignments) {
        out.push(' ');
        match alignment {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push_str(" |");
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_col = if matches!(options.mode, Mode::Markdown) {
        3
    } else {
        1
    };
    let overhead = if matches!(options.mode, Mode::Compact) {
        2 * (n - 1)
    } else {
        3 * n + 1
    };
    let minimum = overhead + n * min_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            if let Some(byte) = invalid_char(cell) {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let transformed: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| logical_lines(cell, markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![min_col; n];
    for row in &transformed {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_col)
                .max_by_key(|&i| (widths[i], i));
            if let Some(i) = candidate {
                widths[i] -= 1;
            } else {
                break;
            }
        }
    }

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        out.push_str(&border(&widths));
    }

    for (row_index, row) in transformed.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for (column, lines) in row.iter().enumerate() {
            let mut fragments = Vec::new();
            for line in lines {
                fragments.extend(split_line(line, widths[column], &overflows[column]));
            }
            height = height.max(fragments.len());
            cell_fragments.push(fragments);
        }

        for fragment_index in 0..height {
            let fragments: Vec<String> = cell_fragments
                .iter()
                .map(|cell| cell.get(fragment_index).cloned().unwrap_or_default())
                .collect();
            out.push_str(&render_data_line(
                &fragments,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if matches!(options.mode, Mode::Grid)
            && options.header
            && row_index == 0
            && rows.len() > 1
        {
            out.push_str(&border(&widths));
        }
        if markdown && options.header && row_index == 0 {
            out.push_str(&markdown_marker(&widths, &alignments));
        }
    }

    if matches!(options.mode, Mode::Grid) {
        out.push_str(&border(&widths));
    }
    Ok(out)
}
