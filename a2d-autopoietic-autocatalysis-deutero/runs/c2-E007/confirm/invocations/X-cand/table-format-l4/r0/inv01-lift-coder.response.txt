#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type CellLines = Vec<Vec<Token>>;

fn invalid_char(c: char, next: Option<char>) -> bool {
    if c == '\r' {
        return next != Some('\n');
    }
    c.is_control() || c == '\u{2028}' || c == '\u{2029}'
}

fn validate(rows: &[Vec<String>]) -> Result<(), TableError> {
    for (row_idx, row) in rows.iter().enumerate() {
        for (col_idx, cell) in row.iter().enumerate() {
            let chars: Vec<(usize, char)> = cell.char_indices().collect();
            for i in 0..chars.len() {
                let (byte, ch) = chars[i];
                let next = chars.get(i + 1).map(|(_, c)| *c);
                if invalid_char(ch, next) {
                    return Err(TableError::InvalidChar {
                        row: row_idx,
                        column: col_idx,
                        byte,
                    });
                }
                if ch == '\r' {
                    // The following LF is part of this accepted CRLF pair.
                    continue;
                }
            }
        }
    }
    Ok(())
}

fn tokenize_line(line: &str, markdown: bool) -> Vec<Token> {
    let trimmed = line.trim_end_matches(' ');
    let mut tokens = Vec::new();
    for ch in trimmed.chars() {
        if markdown && ch == '|' {
            tokens.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && ch == '\\' {
            tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            tokens.push(Token { text: ch.to_string(), width: 1 });
        }
    }
    tokens
}

fn split_cell(cell: &str, markdown: bool) -> CellLines {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| tokenize_line(line, markdown))
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }
    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut out = String::new();
            let mut used = 0usize;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0usize;
            while start < tokens.len() {
                let mut used = 0usize;
                let mut end = start;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // All tokens have width at most two and Markdown's minimum is three.
                if end == start {
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align_text(text: &str, width: usize, alignment: &Alignment) -> String {
    let len = scalar_len(text);
    let extra = width.saturating_sub(len);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut result = String::with_capacity(text.len() + extra);
    result.extend(std::iter::repeat_n(' ', left));
    result.push_str(text);
    result.extend(std::iter::repeat_n(' ', right));
    result
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn data_line(parts: &[String]) -> String {
    let mut out = String::from("|");
    for part in parts {
        out.push(' ');
        out.push_str(part);
        out.push_str(" |");
    }
    out.push('\n');
    out
}

fn markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::from("|");
    for (&width, alignment) in widths.iter().zip(alignments) {
        out.push(' ');
        match alignment {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push_str(" |");
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_idx, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_idx });
        }
    }

    let n = rows[0].len();
    for (row_idx, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_idx,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_width = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    validate(rows)?;

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = options.mode == Mode::Markdown;
    let cells: Vec<Vec<CellLines>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| split_cell(cell, markdown)).collect())
        .collect();

    let mut widths = vec![minimum_width; n];
    for row in &cells {
        for (col, lines) in row.iter().enumerate() {
            for line in lines {
                widths[col] = widths[col].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let candidate = (0..n)
                .filter(|&col| widths[col] > minimum_width)
                .max_by_key(|&col| (widths[col], col));
            let Some(col) = candidate else { break };
            widths[col] -= 1;
            total -= 1;
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_idx, row) in cells.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1usize;
        for (col, lines) in row.iter().enumerate() {
            let mut all = Vec::new();
            for line in lines {
                all.extend(fragments(line, widths[col], &overflows[col]));
            }
            height = height.max(all.len());
            cell_fragments.push(all);
        }

        for fragment_idx in 0..height {
            let mut parts = Vec::with_capacity(n);
            for col in 0..n {
                let content = cell_fragments[col]
                    .get(fragment_idx)
                    .map(String::as_str)
                    .unwrap_or("");
                let mut aligned = align_text(content, widths[col], &alignments[col]);
                if options.mode == Mode::Compact && col + 1 == n {
                    let content_len = scalar_len(content);
                    let desired_left = match alignments[col] {
                        Alignment::Left => 0,
                        Alignment::Right => widths[col].saturating_sub(content_len),
                        Alignment::Center => widths[col].saturating_sub(content_len) / 2,
                    };
                    aligned = format!("{}{}", " ".repeat(desired_left), content);
                }
                parts.push(aligned);
            }

            match options.mode {
                Mode::Compact => {
                    output.push_str(&parts.join("  "));
                    output.push('\n');
                }
                Mode::Grid | Mode::Markdown => output.push_str(&data_line(&parts)),
            }
        }

        if options.mode == Mode::Markdown && row_idx == 0 {
            output.push_str(&markdown_marker(&widths, &alignments));
        }
        if options.mode == Mode::Grid && options.header && row_idx == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
