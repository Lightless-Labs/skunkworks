#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') {
        '\u{feff}'.len_utf8()
    } else {
        0
    };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            if !record_started {
                // A terminator by itself still represents an empty field.
                record.push(String::new());
            } else {
                record.push(std::mem::take(&mut field));
            }
            records.push(std::mem::take(&mut record));
            record_index += 1;
            record_started = false;
            i += 2;
            continue;
        }

        if b == b'\n' {
            if !record_started {
                record.push(String::new());
            } else {
                record.push(std::mem::take(&mut field));
            }
            records.push(std::mem::take(&mut record));
            record_index += 1;
            record_started = false;
            i += 1;
            continue;
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            record_started = true;
            i += 1;
            continue;
        }

        if b == quote {
            if !record_started && field.is_empty() {
                let opening = i;
                i += 1;
                let mut closed = false;
                while i < bytes.len() {
                    let inner = bytes[i];
                    if inner == 0 {
                        return Err(CsvError::Nul {
                            record: record_index,
                            byte: i,
                        });
                    }
                    if inner == quote {
                        if i + 1 < bytes.len() && bytes[i + 1] == quote {
                            field.push(quote as char);
                            i += 2;
                            continue;
                        }
                        i += 1;
                        closed = true;
                        break;
                    }

                    let ch = input[i..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    i += ch.len_utf8();
                }
                if !closed {
                    return Err(CsvError::UnclosedQuote {
                        record: record_index,
                        byte: opening,
                    });
                }

                if i < bytes.len() {
                    let after = bytes[i];
                    if after == 0 {
                        return Err(CsvError::Nul {
                            record: record_index,
                            byte: i,
                        });
                    }
                    if after == b'\r' {
                        if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                            return Err(CsvError::BareCr {
                                record: record_index,
                                byte: i,
                            });
                        }
                        record.push(std::mem::take(&mut field));
                        records.push(std::mem::take(&mut record));
                        record_index += 1;
                        record_started = false;
                        i += 2;
                        continue;
                    }
                    if after == b'\n' {
                        record.push(std::mem::take(&mut field));
                        records.push(std::mem::take(&mut record));
                        record_index += 1;
                        record_started = false;
                        i += 1;
                        continue;
                    }
                    if after == delimiter {
                        record.push(std::mem::take(&mut field));
                        record_started = true;
                        i += 1;
                        continue;
                    }
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }

                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                return Ok(records);
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
        }

        let ch = input[i..].chars().next().expect("valid UTF-8");
        field.push(ch);
        record_started = true;
        i += ch.len_utf8();
    }

    if record_started || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn build_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut records = records.into_iter();
    let mut header = if config.headers {
        records.next()
    } else {
        None
    };

    if let (Some(values), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        for second in 0..values.len() {
            if let Some(first) = values[..second].iter().position(|name| name == &values[second]) {
                return Err(CsvError::DuplicateHeader {
                    name: values[second].clone(),
                    first,
                    second,
                });
            }
        }
    }

    let mut rows: Vec<Vec<String>> = records.collect();
    if let Some(values) = &header {
        for (index, row) in rows.iter().enumerate() {
            if row.len() != values.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: values.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header: header.take(),
        rows: std::mem::take(&mut rows),
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
