#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
}

#[derive(Debug, Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<usize>,
}

fn lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0, 0, 0);
    let mut changes = Vec::new();
    let mut in_run = false;
    let mut run_start = 0;
    let mut deletes = 0usize;
    let mut inserted = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 in_run: &mut bool,
                 run_start: &mut usize,
                 deletes: &mut usize,
                 inserted: &mut Vec<String>| {
        if *in_run {
            changes.push(Change {
                start: *run_start,
                end: *run_start + *deletes,
                replacement: std::mem::take(inserted),
                side,
            });
            *in_run = false;
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(
                &mut changes,
                &mut in_run,
                &mut run_start,
                &mut deletes,
                &mut inserted,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            if !in_run {
                in_run = true;
                run_start = base_pos;
            }
            inserted.push(new[j].clone());
            j += 1;
        } else if j == m {
            if !in_run {
                in_run = true;
                run_start = base_pos;
            }
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            let do_delete = if del > ins {
                true
            } else if ins > del {
                false
            } else {
                (i + j) % 2 == 0
            };

            if !in_run {
                in_run = true;
                run_start = base_pos;
            }
            if do_delete {
                deletes += 1;
                i += 1;
                base_pos += 1;
            } else {
                inserted.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(
        &mut changes,
        &mut in_run,
        &mut run_start,
        &mut deletes,
        &mut inserted,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let al = a.end - a.start;
    let bl = b.end - b.start;
    match (al > 0, bl > 0) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn build_components(changes: &[Change]) -> Vec<Component> {
    let n = changes.len();
    let mut parent: Vec<usize> = (0..n).collect();

    fn root(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = root(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if interacts(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<Component> = Vec::new();
    for i in 0..n {
        let r = root(&mut parent, i);
        if let Some(component) = groups.iter_mut().find(|c| c.changes[0] == r) {
            component.start = component.start.min(changes[i].start);
            component.end = component.end.max(changes[i].end);
            component.changes.push(i);
        } else {
            groups.push(Component {
                start: changes[i].start,
                end: changes[i].end,
                changes: vec![r, i],
            });
            groups.last_mut().unwrap().changes.remove(0);
        }
    }

    groups.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.end > a.start).cmp(&(b.end > b.start)))
            .then(a.end.cmp(&b.end))
    });
    groups
}

fn projection(base: &[String], component: &Component, changes: &[Change], side: usize) -> Vec<String> {
    let mut selected: Vec<&Change> = component
        .changes
        .iter()
        .map(|&idx| &changes[idx])
        .filter(|change| change.side == side)
        .collect();
    selected.sort_by_key(|change| (change.start, change.end));

    let mut output = Vec::new();
    let mut pos = component.start;
    for change in selected {
        while pos < change.start {
            output.push(base[pos].clone());
            pos += 1;
        }
        output.extend(change.replacement.iter().cloned());
        pos = change.end;
    }
    while pos < component.end {
        output.push(base[pos].clone());
        pos += 1;
    }
    output
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = edit_changes(&base_lines, &left_lines, 0);
    changes.extend(edit_changes(&base_lines, &right_lines, 1));
    let components = build_components(&changes);

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        while base_cursor < component.start {
            output.push(base_lines[base_cursor].clone());
            base_cursor += 1;
        }

        let base_projection: Vec<String> = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(&base_lines, &component, &changes, 0);
        let right_projection = projection(&base_lines, &component, &changes, 1);

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let base_label_start = if component.end > component.start {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_label_start},{}",
                        component.end - component.start
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len = output.len() - output_start;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    while base_cursor < base_lines.len() {
        output.push(base_lines[base_cursor].clone());
        base_cursor += 1;
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
