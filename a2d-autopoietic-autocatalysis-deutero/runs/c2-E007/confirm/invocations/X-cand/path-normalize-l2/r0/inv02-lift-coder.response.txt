#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .filter(|(_, part)| !part.is_empty())
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, part)| *part == "." || *part == "..")
}

fn reduce(
    stack: &mut Vec<String>,
    parts: impl IntoIterator<Item = (usize, String)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, part) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, dir_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if dir_intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    reduce(
        components(path).into_iter().map(|(at, part)| (at, part.to_string())),
        &mut stack,
        absolute,
    )?;

    Ok(render(&stack, absolute, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_normalized = normalize(cwd).map_err(|_| PathError::InvalidCwd)?;
    if !cwd.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }

    let cwd_stack: Vec<String> = components(&cwd_normalized)
        .into_iter()
        .map(|(_, part)| part.to_string())
        .collect();

    let home_stack = if let Some(home_path) = home {
        if !home_path.starts_with('/') {
            return Err(PathError::InvalidHome);
        }
        let normalized = normalize(home_path).map_err(|_| PathError::InvalidHome)?;
        Some(
            components(&normalized)
                .into_iter()
                .map(|(_, part)| part.to_string())
                .collect::<Vec<String>>(),
        )
    } else {
        None
    };

    validate_raw(path)?;

    let tilde = path.starts_with('~');
    let (absolute, mut stack, remainder, remainder_offset) = if tilde {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_stack = home_stack.ok_or(PathError::MissingHome)?;
        let skip = if bytes.len() > 1 { 2 } else { 1 };
        (true, home_stack, &path[skip..], skip)
    } else if path.starts_with('/') {
        (true, Vec::new(), path, 0)
    } else {
        (true, cwd_stack, path, 0)
    };

    let intent = directory_intent(path);
    let parts = components(remainder)
        .into_iter()
        .map(|(at, part)| (at + remainder_offset, part.to_string()));

    reduce(&mut stack, parts, absolute)?;
    Ok(render(&stack, true, intent))
}
