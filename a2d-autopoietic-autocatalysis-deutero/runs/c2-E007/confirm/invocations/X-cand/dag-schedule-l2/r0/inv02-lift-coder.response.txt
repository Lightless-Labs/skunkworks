#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Analysis {
    pub order: Vec<usize>,
    pub earliest: Vec<Timing>,
    pub latest: Vec<Timing>,
    pub slack: Vec<u64>,
    pub critical: Vec<usize>,
    pub path: Vec<usize>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
    TargetTooSmall {
        minimum: u64,
    },
}

fn validate_dependencies(tasks: &[Task]) -> Result<(), Error> {
    let count = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = Vec::with_capacity(task.deps.len());

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    Ok(())
}

fn kahn_order(tasks: &[Task]) -> Result<Vec<usize>, Error> {
    let count = tasks.len();
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children = vec![Vec::new(); count];

    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    let mut available = Vec::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(task_id);
        }
    }

    let mut order = Vec::with_capacity(count);
    while !available.is_empty() {
        let smallest_index = available
            .iter()
            .enumerate()
            .min_by_key(|(_, task_id)| **task_id)
            .map(|(index, _)| index)
            .unwrap();
        let task_id = available.swap_remove(smallest_index);
        order.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != count {
        let removed = {
            let mut removed = vec![false; count];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        let blocked = (0..count).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    Ok(order)
}

pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    validate_dependencies(tasks)?;

    let order = kahn_order(tasks)?;
    let count = tasks.len();
    let mut earliest = vec![Timing { start: 0, finish: 0 }; count];

    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&predecessor| earliest[predecessor].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;
        earliest[task_id] = Timing { start, finish };
    }

    let makespan = earliest.iter().map(|timing| timing.finish).max().unwrap_or(0);
    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut children = vec![Vec::new(); count];
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    let mut latest = vec![Timing { start: 0, finish: 0 }; count];
    for &task_id in order.iter().rev() {
        let finish = if children[task_id].is_empty() {
            target
        } else {
            children[task_id]
                .iter()
                .map(|&child| latest[child].start)
                .min()
                .unwrap()
        };
        let start = finish - tasks[task_id].duration;
        latest[task_id] = Timing { start, finish };
    }

    let slack: Vec<u64> = (0..count)
        .map(|id| latest[id].start - earliest[id].start)
        .collect();
    let baseline = target - makespan;
    let critical = (0..count)
        .filter(|&id| slack[id] == baseline)
        .collect();

    let path = if count == 0 {
        Vec::new()
    } else {
        let mut sinks: Vec<usize> = (0..count)
            .filter(|&id| children[id].is_empty() && earliest[id].finish == makespan)
            .collect();
        sinks.sort_unstable();
        let mut current = *sinks.last().unwrap();
        let mut reversed_path = vec![current];

        while !tasks[current].deps.is_empty() {
            current = *tasks[current]
                .deps
                .iter()
                .max_by_key(|&&predecessor| (earliest[predecessor].finish, predecessor))
                .unwrap();
            reversed_path.push(current);
        }

        reversed_path.reverse();
        reversed_path
    };

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}
