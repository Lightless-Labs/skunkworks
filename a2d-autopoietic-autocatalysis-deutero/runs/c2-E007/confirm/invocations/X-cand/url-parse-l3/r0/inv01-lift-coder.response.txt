#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn scan_bytes(s: &str) -> Result<(), ParseError> {
    for (i, b) in s.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let (Some(hi), Some(lo)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((hi << 4) | lo);
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for b in text.bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(char::from(b));
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(char::from(HEX[(b >> 4) as usize]));
            out.push(char::from(HEX[(b & 0x0f) as usize]));
        }
    }
    out
}

fn validate_component(input: &str, base: usize, component: Component) -> Result<(), ParseError> {
    decode_component(component, input).map(|_| ()).map_err(|e| match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    })
}

fn parse_port(port: &str, port_start: usize) -> Result<Option<u16>, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    if let Some(i) = port.bytes().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: port_start + i });
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let mut value: u32 = 0;
    for b in port.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: port_start })?;
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(Some(value as u16))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let rest = &input[auth_start..];
    let auth_len = rest
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(rest.len());
    let authority = &rest[..auth_len];
    let auth_end = auth_start + auth_len;

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + at_positions[1],
        });
    }
    let (userinfo, hostpart, hostpart_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            auth_start + at + 1,
        )
    } else {
        (None, authority, auth_start)
    };

    let mut ipv6 = false;
    let (host_raw, port) = if hostpart.starts_with('[') {
        let close_rel = match hostpart[1..].find(']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: hostpart_start }),
        };
        let interior = &hostpart[1..close_rel];
        if interior.contains('%') || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostpart_start });
        }
        let suffix = &hostpart[close_rel + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if let Some(port_text) = suffix.strip_prefix(':') {
            parse_port(port_text, hostpart_start + close_rel + 2)?
        } else {
            return Err(ParseError::InvalidHost {
                at: hostpart_start + close_rel + 1,
            });
        };
        ipv6 = true;
        (interior, parsed_port)
    } else {
        let split = hostpart.find(':');
        let (host, parsed_port) = if let Some(i) = split {
            (&hostpart[..i], parse_port(&hostpart[i + 1..], hostpart_start + i + 1)?)
        } else {
            (hostpart, None)
        };
        if host.is_empty() {
            return Err(ParseError::InvalidHost { at: hostpart_start });
        }
        for (i, b) in host.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hostpart_start + i });
            }
        }
        (host, parsed_port)
    };

    let host = host_raw.to_ascii_lowercase();
    let component_start = auth_end;
    let suffix = &input[component_start..];

    let frag_rel = suffix.find('#');
    let before_fragment = frag_rel.map_or(suffix, |i| &suffix[..i]);
    let fragment = frag_rel.map(|i| suffix[i + 1..].to_owned());
    let query_rel = before_fragment.find('?');
    let path_raw = query_rel.map_or(before_fragment, |i| &before_fragment[..i]);
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());

    if let Some(info) = userinfo.as_deref() {
        validate_component(info, auth_start, Component::UserInfo)?;
    }
    if !path_raw.is_empty() {
        validate_component(path_raw, component_start, Component::Path)?;
    }
    if let Some(q) = query.as_deref() {
        validate_component(q, component_start + query_rel.unwrap() + 1, Component::Query)?;
    }
    if let Some(f) = fragment.as_deref() {
        validate_component(
            f,
            component_start + frag_rel.unwrap() + 1,
            Component::Fragment,
        )?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path: if path_raw.is_empty() {
            "/".to_owned()
        } else {
            path_raw.to_owned()
        },
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (i, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 },
        })?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if is_dotdot {
            stack.pop();
        } else if !is_dot {
            stack.push(segment);
        }
        if i + 1 == raw_segments.len() {
            terminal_navigation = is_dot || is_dotdot;
        }
    }

    if terminal_navigation {
        stack.push("");
    }
    let joined = stack.join("/");
    if joined.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{joined}"))
    }
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let frag = reference.find('#');
    let before_frag = frag.map_or(reference, |i| &reference[..i]);
    let fragment = frag.map(|i| &reference[i + 1..]);
    let query_at = before_frag.find('?');
    let path = query_at.map_or(before_frag, |i| &before_frag[..i]);
    let query = query_at.map(|i| &before_frag[i + 1..]);
    (path, query, fragment)
}

fn with_path(mut url: Url, path: String) -> Url {
    url.path = path;
    url
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    scan_bytes(reference).map_err(ResolveError::Reference)?;

    let first_delimiter = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.clone(),
            if ref_query.is_some() {
                ref_query.map(str::to_owned)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query.map(str::to_owned))
    };

    let authority = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let userinfo = base_url
        .userinfo
        .as_ref()
        .map(|s| format!("{s}@"))
        .unwrap_or_default();
    let port = base_url
        .port
        .map(|p| format!(":{p}"))
        .unwrap_or_default();
    let path_part = if selected_path.is_empty() {
        String::new()
    } else {
        selected_path.clone()
    };
    let query_part = selected_query
        .as_ref()
        .map(|q| format!("?{q}"))
        .unwrap_or_default();
    let fragment_part = ref_fragment
        .map(|f| format!("#{f}"))
        .unwrap_or_default();
    let constructed = format!(
        "{}://{}{}{}{}{}",
        base_url.scheme, userinfo, authority, port, path_part, query_part
    ) + &fragment_part;

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(with_path(url, url.path.clone()))
}
