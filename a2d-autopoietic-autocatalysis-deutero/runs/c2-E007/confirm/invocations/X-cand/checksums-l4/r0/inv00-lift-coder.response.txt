#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

const ADLER_MOD: u32 = 65_521;

fn crc32_step(mut crc: u32, byte: u8) -> u32 {
    crc ^= byte as u32;
    for _ in 0..8 {
        crc = (crc >> 1) ^ if crc & 1 != 0 { 0xedb8_8320 } else { 0 };
    }
    crc
}

fn adler_step(a: &mut u32, b: &mut u32, byte: u8) {
    *a = (*a + byte as u32) % ADLER_MOD;
    *b = (*b + *a) % ADLER_MOD;
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    for &byte in data {
        crc = crc32_step(crc, byte);
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    for &byte in data {
        adler_step(&mut a, &mut b, byte);
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index != 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            adler_step(&mut a, &mut b, byte);
        }
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc = crc32_step(self.crc, byte);
            adler_step(&mut self.a, &mut self.b, byte);
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let mut crc = 0u32;
        let mut adler = 0u32;
        for (offset, byte) in bytes.iter().enumerate() {
            if (1..9).contains(&offset) || (11..19).contains(&offset) {
                let digit = hex_value(*byte).ok_or(SnapshotError::Hex { offset })?;
                if offset < 9 {
                    crc = (crc << 4) | digit as u32;
                } else {
                    adler = (adler << 4) | digit as u32;
                }
            }
        }
        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_word(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for chunk in &mut chunks {
        fletcher32_word(&mut a, &mut b, u16::from_be_bytes([chunk[0], chunk[1]]));
    }
    if let Some(&byte) = chunks.remainder().first() {
        fletcher32_word(&mut a, &mut b, (byte as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut high_byte = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = high_byte.take() {
                fletcher32_word(&mut a, &mut b, ((high as u16) << 8) | byte as u16);
            } else {
                high_byte = Some(byte);
            }
        }
    }
    if let Some(high) = high_byte {
        fletcher32_word(&mut a, &mut b, (high as u16) << 8);
    }
    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    crc = (crc << 1)
                        ^ if crc & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            Crc16::X25 => {
                crc ^= byte as u16;
                for _ in 0..8 {
                    crc = (crc >> 1) ^ if crc & 1 != 0 { 0x8408 } else { 0 };
                }
            }
        }
    }
    if matches!(variant, Crc16::X25) {
        crc ^ 0xffff
    } else {
        crc
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffff;
    let mut adler_a = 1u32;
    let mut adler_b = 0u32;
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_high = None;
    let mut crc16_reg: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            adler_step(&mut adler_a, &mut adler_b, byte);

            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_high.take() {
                fletcher32_word(&mut f32_a, &mut f32_b, ((high as u16) << 8) | byte as u16);
            } else {
                f32_high = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_reg ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        crc16_reg = (crc16_reg << 1)
                            ^ if crc16_reg & 0x8000 != 0 { 0x1021 } else { 0 };
                    }
                }
                Crc16::X25 => {
                    crc16_reg ^= byte as u16;
                    for _ in 0..8 {
                        crc16_reg = (crc16_reg >> 1)
                            ^ if crc16_reg & 1 != 0 { 0x8408 } else { 0 };
                    }
                }
            }
        }
    }

    if let Some(high) = f32_high {
        fletcher32_word(&mut f32_a, &mut f32_b, (high as u16) << 8);
    }
    if matches!(variant, Crc16::X25) {
        crc16_reg ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_reg,
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn crc32_zero_byte_operator(mut value: u32) -> u32 {
    for _ in 0..8 {
        value = (value >> 1) ^ if value & 1 != 0 { 0xedb8_8320 } else { 0 };
    }
    value
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut operator = [0u32; 32];
    for (index, slot) in operator.iter_mut().enumerate() {
        *slot = crc32_zero_byte_operator(1u32 << index);
    }

    let mut remaining = right_len;
    let mut result = left;
    while remaining != 0 {
        if remaining & 1 != 0 {
            result = gf2_matrix_times(&operator, result);
        }
        remaining >>= 1;
        if remaining != 0 {
            let mut squared = [0u32; 32];
            gf2_matrix_square(&mut squared, &operator);
            operator = squared;
        }
    }
    result ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut length = 0u64;
    for span in spans {
        length += (span.end - span.start) as u64;
    }
    let sums = checksums_spans(data, spans)?;
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length, sums.crc32, sums.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut length = 0u64;
    let mut crc = 0u32;
    let mut adler = 0u32;
    for offset in 4..48 {
        let is_digit = (4..20).contains(&offset)
            || (25..33).contains(&offset)
            || (40..48).contains(&offset);
        if is_digit {
            let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
            if offset < 20 {
                length = (length << 4) | digit as u64;
            } else if offset < 33 {
                crc = (crc << 4) | digit as u32;
            } else {
                adler = (adler << 4) | digit as u32;
            }
        }
    }

    let mut selected_len = 0u64;
    for span in spans {
        selected_len += (span.end - span.start) as u64;
    }
    if length != selected_len {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
