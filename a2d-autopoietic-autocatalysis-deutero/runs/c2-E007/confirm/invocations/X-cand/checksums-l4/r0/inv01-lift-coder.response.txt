#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

const ADLER_MOD: u32 = 65_521;

fn crc32_step(mut register: u32, byte: u8) -> u32 {
    register ^= byte as u32;
    for _ in 0..8 {
        let low = register & 1;
        register >>= 1;
        if low != 0 {
            register ^= 0xedb8_8320;
        }
    }
    register
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut register = 0xffff_ffff;
    for &byte in data {
        register = crc32_step(register, byte);
    }
    register ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % ADLER_MOD;
        b = (b + a) % ADLER_MOD;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = ((pair[0] as u64) << 8) | pair[1] as u64;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    if let Some(&byte) = chunks.remainder().first() {
        a = (a + ((byte as u64) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    ((b as u32) << 16) | a as u32
}

#[derive(Debug, Clone, Copy)]
struct Fletcher32State {
    a: u64,
    b: u64,
    high: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            high: None,
        }
    }

    fn update(&mut self, bytes: &[u8]) {
        for &byte in bytes {
            if let Some(high) = self.high.take() {
                let word = ((high as u64) << 8) | byte as u64;
                self.a = (self.a + word) % 65_535;
                self.b = (self.b + self.a) % 65_535;
            } else {
                self.high = Some(byte);
            }
        }
    }

    fn finish(self) -> u32 {
        let mut a = self.a;
        let mut b = self.b;
        if let Some(high) = self.high {
            a = (a + ((high as u64) << 8)) % 65_535;
            b = (b + a) % 65_535;
        }
        ((b as u32) << 16) | a as u32
    }
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Fletcher32State::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut register: u16 = if variant == Crc16::False { 0xffff } else { 0 };
            for &byte in data {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if top != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            register
        }
        Crc16::X25 => {
            let mut register: u16 = 0xffff;
            for &byte in data {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
            register ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32 = Fletcher32State::new();
    let mut c16 = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);
        for &byte in selected {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;
            match variant {
                Crc16::Xmodem | Crc16::False => {
                    c16 ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = c16 & 0x8000;
                        c16 = c16.wrapping_shl(1);
                        if top != 0 {
                            c16 ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    c16 ^= byte as u16;
                    for _ in 0..8 {
                        let low = c16 & 1;
                        c16 >>= 1;
                        if low != 0 {
                            c16 ^= 0x8408;
                        }
                    }
                }
            }
        }
        f32.update(selected);
    }

    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }
    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16_b << 8) | f16_a) as u16,
        fletcher32: f32.finish(),
        crc16: c16,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc = crc32_step(self.crc, byte);
            self.a = (self.a + byte as u32) % ADLER_MOD;
            self.b = (self.b + self.a) % ADLER_MOD;
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let crc = parse_hex_u32(bytes, 1, 9).map_err(|offset| SnapshotError::Hex { offset })?;
        let adler = parse_hex_u32(bytes, 11, 19).map_err(|offset| SnapshotError::Hex { offset })?;
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> Result<u32, usize> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => bytes[offset] - b'0',
            b'a'..=b'f' => bytes[offset] - b'a' + 10,
            _ => return Err(offset),
        };
        value = (value << 4) | digit as u32;
    }
    Ok(value)
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for n in 0..32 {
        square[n] = gf2_matrix_times(matrix, matrix[n]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut crc = left;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

fn selected_count(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |sum, span| {
        sum + (span.end - span.start) as u64
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    let hashes = stream.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        selected_count(spans),
        hashes.crc32,
        hashes.adler32
    ))
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> Result<u64, usize> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => bytes[offset] - b'0',
            b'a'..=b'f' => bytes[offset] - b'a' + 10,
            _ => return Err(offset),
        };
        value = (value << 4) | digit as u64;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let length = parse_hex_u64(bytes, 4, 20).map_err(|offset| VerifyError::Hex { offset })?;
    let crc = parse_hex_u32(bytes, 25, 33).map_err(|offset| VerifyError::Hex { offset })?;
    let adler = parse_hex_u32(bytes, 40, 48).map_err(|offset| VerifyError::Hex { offset })?;

    if length != selected_count(spans) {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
