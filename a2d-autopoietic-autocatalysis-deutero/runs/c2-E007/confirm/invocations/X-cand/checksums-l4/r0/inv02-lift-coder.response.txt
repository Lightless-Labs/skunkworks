#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffffu32;
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 65_521;
        b = (b + a) % 65_521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

#[derive(Clone, Copy)]
struct Accumulator {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    fletcher16_a: u32,
    fletcher16_b: u32,
    fletcher32_a: u64,
    fletcher32_b: u64,
    fletcher32_high: Option<u8>,
    crc16: u16,
    crc16_variant: Crc16,
}

impl Accumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            fletcher16_a: 0,
            fletcher16_b: 0,
            fletcher32_a: 0,
            fletcher32_b: 0,
            fletcher32_high: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            crc16_variant: variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc ^= byte as u32;
            for _ in 0..8 {
                let low = self.crc & 1;
                self.crc >>= 1;
                if low != 0 {
                    self.crc ^= 0xedb8_8320;
                }
            }

            self.adler_a = (self.adler_a + byte as u32) % 65_521;
            self.adler_b = (self.adler_b + self.adler_a) % 65_521;

            self.fletcher16_a = (self.fletcher16_a + byte as u32) % 255;
            self.fletcher16_b = (self.fletcher16_b + self.fletcher16_a) % 255;

            if let Some(high) = self.fletcher32_high.take() {
                self.fletcher32_word(((high as u16) << 8) | byte as u16);
            } else {
                self.fletcher32_high = Some(byte);
            }

            match self.crc16_variant {
                Crc16::Xmodem | Crc16::False => {
                    self.crc16 ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = self.crc16 & 0x8000;
                        self.crc16 = self.crc16.wrapping_shl(1);
                        if top != 0 {
                            self.crc16 ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    self.crc16 ^= byte as u16;
                    for _ in 0..8 {
                        let low = self.crc16 & 1;
                        self.crc16 >>= 1;
                        if low != 0 {
                            self.crc16 ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    fn fletcher32_word(&mut self, word: u16) {
        self.fletcher32_a = (self.fletcher32_a + word as u64) % 65_535;
        self.fletcher32_b = (self.fletcher32_b + self.fletcher32_a) % 65_535;
    }

    fn finish_fletcher32(&mut self) {
        if let Some(high) = self.fletcher32_high.take() {
            self.fletcher32_word((high as u16) << 8);
        }
    }

    fn hashes(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        acc.update(&data[span.start..span.end]);
    }
    Ok(acc.hashes())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = u16::from_be_bytes([pair[0], pair[1]]) as u64;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    if let Some(&last) = chunks.remainder().first() {
        a = (a + ((last as u64) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    ((b as u32) << 16) | a as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        acc.update(&data[span.start..span.end]);
    }
    acc.finish_fletcher32();
    Ok(((acc.fletcher32_b as u32) << 16) | acc.fletcher32_a as u32)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut acc = Accumulator::new(variant);
    acc.update(data);
    if variant == Crc16::X25 {
        acc.crc16 ^ 0xffff
    } else {
        acc.crc16
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(variant);
    for span in spans {
        acc.update(&data[span.start..span.end]);
    }
    acc.finish_fletcher32();
    let crc16 = if variant == Crc16::X25 {
        acc.crc16 ^ 0xffff
    } else {
        acc.crc16
    };
    Ok(Multi {
        base: acc.hashes(),
        fletcher16: ((acc.fletcher16_b as u16) << 8) | acc.fletcher16_a as u16,
        fletcher32: ((acc.fletcher32_b as u32) << 16) | acc.fletcher32_a as u32,
        crc16,
    })
}

fn crc32_zero_byte(mut crc: u32) -> u32 {
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb8_8320;
        }
    }
    crc
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = gf2_matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];

    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }

    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }
    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

fn selected_len(spans: &[Span]) -> usize {
    spans.iter().map(|span| span.end - span.start).sum()
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let base = checksums_spans(data, spans)?;
    let len = selected_len(spans) as u64;
    Ok(format!(
        "len={len:016x};crc={:08x};adler={:08x}",
        base.crc32, base.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex(bytes: &[u8]) -> u64 {
    bytes.iter().fold(0u64, |value, &byte| {
        (value << 4) | hex_value(byte).unwrap() as u64
    })
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let expected_len = parse_hex(&bytes[4..20]);
    let expected_crc = parse_hex(&bytes[25..33]) as u32;
    let expected_adler = parse_hex(&bytes[40..48]) as u32;
    if expected_len != selected_len(spans) as u64 {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc ^= byte as u32;
            for _ in 0..8 {
                let low = self.crc & 1;
                self.crc >>= 1;
                if low != 0 {
                    self.crc ^= 0xedb8_8320;
                }
            }
            self.adler_a = (self.adler_a + byte as u32) % 65_521;
            self.adler_b = (self.adler_b + self.adler_a) % 65_521;
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksums = self.checksum();
        format!("c{:08x}:a{:08x}", checksums.crc32, checksums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        for offset in (1..9).chain(11..19) {
            if hex_value(bytes[offset]).is_none() {
                return Err(SnapshotError::Hex { offset });
            }
        }
        let crc = parse_hex(&bytes[1..9]) as u32;
        let adler = parse_hex(&bytes[11..19]) as u32;
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}
