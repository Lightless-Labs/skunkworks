#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_symbol(value: u8, alphabet: Alphabet) -> u8 {
    match (value, alphabet) {
        (62, Alphabet::Standard) => b'+',
        (63, Alphabet::Standard) => b'/',
        (62, Alphabet::UrlSafe) => b'-',
        (63, Alphabet::UrlSafe) => b'_',
        (v @ 0..=25, _) => b'A' + v,
        (v @ 26..=51, _) => b'a' + (v - 26),
        (v @ 52..=61, _) => b'0' + (v - 52),
        _ => unreachable!(),
    }
}

fn decode_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8], alphabet: Alphabet, padding: bool) -> String {
    let mut out = String::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    let digits = [
        a >> 2,
        ((a & 0x03) << 4) | (b >> 4),
        ((b & 0x0f) << 2) | (c >> 6),
        c & 0x3f,
    ];
    let count = if bytes.len() == 1 { 2 } else if bytes.len() == 2 { 3 } else { 4 };
    for digit in digits.iter().take(count) {
        out.push(alphabet_symbol(*digit, alphabet) as char);
    }
    if padding {
        for _ in count..4 {
            out.push('=');
        }
    }
    out
}

fn wrap_encoded(raw: &str, whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return raw.to_owned();
    };
    let separator = match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };
    let mut out = String::with_capacity(raw.len() + raw.len() / width * separator.len());
    for (i, chunk) in raw.as_bytes().chunks(width).enumerate() {
        if i != 0 {
            out.push_str(separator);
        }
        out.push_str(std::str::from_utf8(chunk).unwrap());
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut raw = String::new();
    for chunk in input.chunks(3) {
        let pad = options.padding != Padding::Forbidden;
        raw.push_str(&encode_group(chunk, options.alphabet, pad));
    }
    Ok(wrap_encoded(&raw, options.whitespace))
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::with_capacity(2),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut bytes = Vec::with_capacity(self.tail.len() + input.len());
        bytes.extend_from_slice(&self.tail);
        bytes.extend_from_slice(input);
        self.tail.clear();

        let complete_len = bytes.len() / 3 * 3;
        let mut raw = String::new();
        for chunk in bytes[..complete_len].chunks(3) {
            raw.push_str(&encode_group(chunk, self.options.alphabet, false));
        }
        self.tail.extend_from_slice(&bytes[complete_len..]);
        Ok(wrap_encoded(&raw, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let raw = if self.tail.is_empty() {
            String::new()
        } else {
            encode_group(
                &self.tail,
                self.options.alphabet,
                self.options.padding != Padding::Forbidden,
            )
        };
        self.tail.clear();
        Ok(wrap_encoded(&raw, self.options.whitespace))
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum FrameState {
    Row,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    symbols: Vec<Symbol>,
    total_offset: usize,
    finished: bool,
    sticky: Option<Error>,
    padded: bool,
    frame_symbols: usize,
    frame_state: FrameState,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            symbols: Vec::with_capacity(4),
            total_offset: 0,
            finished: false,
            sticky: None,
            padded: false,
            frame_symbols: 0,
            frame_state: FrameState::Row,
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky = Some(error.clone());
        error
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.padded {
            return Err(Error::BadPadding { offset });
        }

        let pos = self.symbols.len();
        if pos <= 1 && byte == b'=' {
            return Err(Error::BadPadding { offset });
        }
        if pos == 2 && byte == b'=' {
            self.symbols.push(Symbol { byte, offset });
            return Ok(());
        }
        if pos == 3 && self.symbols[2].byte == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if pos == 2 {
                return Err(Error::BadPadding { offset });
            }
            if pos != 3 {
                return Err(Error::BadPadding { offset });
            }
        } else if decode_digit(byte, self.options.alphabet).is_none() {
            return Err(Error::InvalidByte { offset, byte });
        }

        self.symbols.push(Symbol { byte, offset });
        if self.symbols.len() == 4 {
            let s = self.symbols;
            let first = decode_digit(s[0].byte, self.options.alphabet).unwrap();
            let second = decode_digit(s[1].byte, self.options.alphabet).unwrap();

            if s[2].byte == b'=' {
                if s[3].byte != b'=' {
                    return Err(Error::BadPadding { offset: s[3].offset });
                }
                out.push((first << 2) | (second >> 4));
                self.padded = true;
            } else {
                let third = decode_digit(s[2].byte, self.options.alphabet).unwrap();
                out.push((first << 2) | (second >> 4));
                if s[3].byte == b'=' {
                    out.push((second << 4) | (third >> 2));
                    self.padded = true;
                } else {
                    let fourth = decode_digit(s[3].byte, self.options.alphabet).unwrap();
                    out.push((second << 4) | (third >> 2));
                    out.push((third << 6) | fourth);
                }
            }
            self.symbols.clear();
        }
        Ok(())
    }

    fn process_byte(
        &mut self,
        byte: u8,
        offset: usize,
        out: &mut Vec<u8>,
    ) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.process_symbol(byte, offset, out),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.process_symbol(byte, offset, out)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.frame_state {
                    FrameState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.frame_state = FrameState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        Ok(())
                    }
                    FrameState::NeedSymbol { .. } => {
                        self.frame_state = FrameState::Row;
                        self.frame_symbols = 0;
                        if byte == b'\r' || byte == b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.process_symbol(byte, offset, out)?;
                        self.frame_symbols = 1;
                        Ok(())
                    }
                    FrameState::Row => {
                        if self.frame_symbols == width {
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.frame_state = FrameState::NeedSymbol {
                                        separator_offset: offset,
                                    };
                                    return Ok(());
                                }
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.frame_state = FrameState::NeedLf { cr_offset: offset };
                                    return Ok(());
                                }
                                _ => return Err(Error::BadWrap { offset }),
                            }
                        }
                        if byte == b'\r' || byte == b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.process_symbol(byte, offset, out)?;
                        self.frame_symbols += 1;
                        Ok(())
                    }
                }
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.total_offset;
            self.total_offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut out) {
                return Err(self.fail(error));
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedLf { cr_offset } => {
                    return Err(self.fail(Error::BadWrap { offset: cr_offset }));
                }
                FrameState::NeedSymbol { separator_offset } => {
                    return Err(self.fail(Error::BadWrap {
                        offset: separator_offset,
                    }));
                }
                FrameState::Row => {}
            }
        }

        if self.symbols.len() == 3 && self.symbols[2].byte == b'=' {
            return Err(self.fail(Error::BadPadding {
                offset: self.symbols[2].offset,
            }));
        }

        let mut out = Vec::new();
        match self.symbols.len() {
            0 => {}
            1 => {
                return Err(self.fail(Error::Truncated {
                    offset: self.symbols[0].offset,
                }));
            }
            2 | 3 if self.options.padding != Padding::Required => {
                let a = decode_digit(self.symbols[0].byte, self.options.alphabet).unwrap();
                let b = decode_digit(self.symbols[1].byte, self.options.alphabet).unwrap();
                out.push((a << 2) | (b >> 4));
                if self.symbols.len() == 3 {
                    let c = decode_digit(self.symbols[2].byte, self.options.alphabet).unwrap();
                    out.push((b << 4) | (c >> 2));
                }
            }
            _ => {
                return Err(self.fail(Error::Truncated {
                    offset: self.symbols[0].offset,
                }));
            }
        }

        self.finished = true;
        Ok(out)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
