#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn emit_digit(value: u8, alphabet: Alphabet, out: &mut String) {
    let byte = match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + value - 26,
        52..=61 => b'0' + value - 52,
        62 if alphabet == Alphabet::Standard => b'+',
        63 if alphabet == Alphabet::Standard => b'/',
        62 => b'-',
        _ => b'_',
    };
    out.push(byte as char);
}

fn encode_group(input: &[u8], alphabet: Alphabet, padding: Padding, out: &mut String) {
    let a = input[0];
    let b = if input.len() > 1 { input[1] } else { 0 };
    let c = if input.len() > 2 { input[2] } else { 0 };

    emit_digit(a >> 2, alphabet, out);
    emit_digit(((a & 0x03) << 4) | (b >> 4), alphabet, out);

    if input.len() > 1 {
        emit_digit(((b & 0x0f) << 2) | (c >> 6), alphabet, out);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }

    if input.len() > 2 {
        emit_digit(c & 0x3f, alphabet, out);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }
}

fn wrap_encoded(raw: String, whitespace: Whitespace) -> String {
    let (width, ending) = match whitespace {
        Whitespace::Wrapped { width, ending } => (width, ending),
        _ => return raw,
    };

    let separator = match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };
    let mut out = String::with_capacity(raw.len() + raw.len() / width * separator.len());
    for (i, chunk) in raw.as_bytes().chunks(width).enumerate() {
        if i != 0 {
            out.push_str(separator);
        }
        out.push_str(std::str::from_utf8(chunk).unwrap());
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut raw = String::with_capacity((input.len() + 2) / 3 * 4);
    for chunk in input.chunks(3) {
        encode_group(chunk, options.alphabet, options.padding, &mut raw);
    }
    Ok(wrap_encoded(raw, options.whitespace))
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            pending: Vec::with_capacity(2),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.pending.len() + input.len());
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(input);

        let complete = combined.len() / 3 * 3;
        let mut raw = String::with_capacity(complete / 3 * 4);
        for chunk in combined[..complete].chunks(3) {
            encode_group(chunk, self.options.alphabet, self.options.padding, &mut raw);
        }
        self.pending.clear();
        self.pending.extend_from_slice(&combined[complete..]);

        Ok(wrap_encoded(raw, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut raw = String::new();
        if !self.pending.is_empty() {
            encode_group(
                &self.pending,
                self.options.alphabet,
                self.options.padding,
                &mut raw,
            );
            self.pending.clear();
        }
        Ok(wrap_encoded(raw, self.options.whitespace))
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
    value: Option<u8>,
}

#[derive(Clone, Copy)]
enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    offset: usize,
    row_symbols: usize,
    frame: FrameState,
    group: Vec<Symbol>,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            row_symbols: 0,
            frame: FrameState::Normal,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky: None,
        })
    }

    fn push_inner(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        let mut out = Vec::new();
        for &byte in input {
            let here = self.offset;
            self.offset += 1;

            match self.options.whitespace {
                Whitespace::IgnoreAscii
                    if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) =>
                {
                    continue;
                }
                Whitespace::Wrapped { width, ending } => {
                    match self.frame {
                        FrameState::NeedLf { cr_offset } => {
                            if byte != b'\n' {
                                return Err(Error::BadWrap { offset: here });
                            }
                            self.frame = FrameState::NeedSymbol {
                                separator_offset: cr_offset,
                            };
                            continue;
                        }
                        FrameState::NeedSymbol { separator_offset } => {
                            if matches!(byte, b'\r' | b'\n') {
                                return Err(Error::BadWrap { offset: here });
                            }
                            self.frame = FrameState::Normal;
                            self.row_symbols = 0;
                            let _ = separator_offset;
                        }
                        FrameState::Normal => {}
                    }

                    if self.row_symbols == width {
                        match ending {
                            LineEnding::Lf if byte == b'\n' => {
                                self.frame = FrameState::NeedSymbol {
                                    separator_offset: here,
                                };
                                continue;
                            }
                            LineEnding::CrLf if byte == b'\r' => {
                                self.frame = FrameState::NeedLf { cr_offset: here };
                                continue;
                            }
                            _ => return Err(Error::BadWrap { offset: here }),
                        }
                    }

                    if matches!(byte, b'\r' | b'\n' | 9 | 11 | 12 | 32) {
                        return Err(Error::BadWrap { offset: here });
                    }
                    self.row_symbols += 1;
                }
                _ => {}
            }

            if self.terminal_padding {
                return Err(Error::BadPadding { offset: here });
            }

            if let Some(pad_offset) = self.pending_second_padding {
                if byte != b'=' {
                    return Err(Error::BadPadding { offset: here });
                }
                self.pending_second_padding = None;
                self.group.push(Symbol {
                    byte,
                    offset: here,
                    value: None,
                });
                self.terminal_padding = true;
                self.decode_complete_group(&mut out)?;
                continue;
            }

            let pos = self.group.len();
            if byte == b'=' {
                if pos <= 1 || self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset: here });
                }
                if pos == 2 {
                    self.pending_second_padding = Some(here);
                    self.group.push(Symbol {
                        byte,
                        offset: here,
                        value: None,
                    });
                    continue;
                }
                self.group.push(Symbol {
                    byte,
                    offset: here,
                    value: None,
                });
                self.terminal_padding = true;
                self.decode_complete_group(&mut out)?;
                continue;
            }

            let value = alphabet_digit(byte, self.options.alphabet);
            if value.is_none() {
                return Err(Error::InvalidByte { offset: here, byte });
            }
            self.group.push(Symbol {
                byte,
                offset: here,
                value,
            });
            if self.group.len() == 4 {
                self.decode_complete_group(&mut out)?;
            }
        }
        Ok(out)
    }

    fn decode_complete_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.group.len() != 4 {
            return Ok(());
        }
        let g = &self.group;
        let vals = [
            g[0].value.unwrap_or(0),
            g[1].value.unwrap_or(0),
            g[2].value.unwrap_or(0),
            g[3].value.unwrap_or(0),
        ];

        if g[0].byte == b'=' || g[1].byte == b'=' {
            return Err(Error::BadPadding {
                offset: if g[0].byte == b'=' { g[0].offset } else { g[1].offset },
            });
        }

        if g[2].byte == b'=' {
            if g[3].byte != b'=' {
                return Err(Error::BadPadding { offset: g[3].offset });
            }
            out.push((vals[0] << 2) | (vals[1] >> 4));
        } else if g[3].byte == b'=' {
            out.push((vals[0] << 2) | (vals[1] >> 4));
            out.push((vals[1] << 4) | (vals[2] >> 2));
        } else {
            out.push((vals[0] << 2) | (vals[1] >> 4));
            out.push((vals[1] << 4) | (vals[2] >> 2));
            out.push((vals[2] << 6) | vals[3]);
        }
        self.group.clear();
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        match self.push_inner(input) {
            Ok(out) => Ok(out),
            Err(error) => {
                self.sticky = Some(error.clone());
                Err(error)
            }
        }
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let eof_error = match self.frame {
            FrameState::NeedLf { cr_offset } => Some(Error::BadWrap { offset: cr_offset }),
            FrameState::NeedSymbol { separator_offset } => {
                Some(Error::BadWrap { offset: separator_offset })
            }
            FrameState::Normal => None,
        };
        if let Some(error) = eof_error {
            self.sticky = Some(error.clone());
            return Err(error);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky = Some(error.clone());
            return Err(error);
        }

        let mut out = Vec::new();
        match self.group.len() {
            0 => {}
            1 => {
                let error = Error::Truncated {
                    offset: self.group[0].offset,
                };
                self.sticky = Some(error.clone());
                return Err(error);
            }
            2 | 3 if self.options.padding != Padding::Required => {
                let a = self.group[0].value.unwrap();
                let b = self.group[1].value.unwrap();
                out.push((a << 2) | (b >> 4));
                if self.group.len() == 3 {
                    let c = self.group[2].value.unwrap();
                    out.push((b << 4) | (c >> 2));
                }
                self.group.clear();
            }
            _ => {
                let error = Error::Truncated {
                    offset: self.group[0].offset,
                };
                self.sticky = Some(error.clone());
                return Err(error);
            }
        }

        self.finished = true;
        Ok(out)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
