use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

const BASE: u64 = 1_000_000_000;
const POW_LIMIT: u32 = 1024;

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64
                + carry;
            out.push((sum % BASE) as u32);
            carry = sum / BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut value = a[i] as i64
                - b.get(i).copied().unwrap_or(0) as i64
                - borrow;
            if value < 0 {
                value += BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u32);
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_small_abs(a: &[u32], m: u32) -> Vec<u32> {
        if m == 0 || a.is_empty() {
            return Vec::new();
        }
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u64;
        for &limb in a {
            let value = limb as u64 * m as u64 + carry;
            out.push((value % BASE) as u32);
            carry = value / BASE;
        }
        while carry != 0 {
            out.push((carry % BASE) as u32);
            carry /= BASE;
        }
        out
    }

    fn add_small_abs(a: &mut Vec<u32>, value: u32) {
        if value == 0 {
            return;
        }
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == a.len() {
                a.push(0);
            }
            let sum = a[i] as u64 + carry;
            a[i] = (sum % BASE) as u32;
            carry = sum / BASE;
            i += 1;
        }
    }

    fn div_small_abs(a: &[u32], divisor: u32) -> (Vec<u32>, u32) {
        let mut quotient = vec![0; a.len()];
        let mut remainder = 0u64;
        for i in (0..a.len()).rev() {
            let value = remainder * BASE + a[i] as u64;
            quotient[i] = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        while quotient.last() == Some(&0) {
            quotient.pop();
        }
        (quotient, remainder as u32)
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
            b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        let mut digit_values = Vec::new();
        let mut digit_offsets = Vec::new();
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            match Self::digit_value(byte) {
                Some(value) if value < radix => {
                    digit_values.push(value);
                    digit_offsets.push(index);
                }
                _ => {
                    if invalid.is_none() {
                        invalid = Some(index);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                return Err(Error::Separator { index });
            }
        }

        if digit_values.len() > 1 && digit_values[0] == 0 {
            return Err(Error::LeadingZero {
                index: digit_offsets[0],
            });
        }
        if digit_values.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_offsets[max_digits],
            });
        }

        let mut result = Self::zero();
        for digit in digit_values {
            result.limbs = Self::mul_small_abs(&result.limbs, radix);
            Self::add_small_abs(&mut result.limbs, digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.abs();
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(0u32);
        } else {
            while !value.is_zero() {
                let (quotient, remainder) = Self::div_small_abs(&value.limbs, radix);
                value.limbs = quotient;
                digits.push(remainder);
            }
            digits.reverse();
        }

        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut magnitude = String::new();
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                magnitude.push('_');
            }
            magnitude.push(if digit < 10 {
                (b'0' + digit as u8) as char
            } else {
                (b'a' + (digit - 10) as u8) as char
            });
        }

        if self.negative && !self.is_zero() {
            Ok(format!("-{magnitude}"))
        } else {
            Ok(magnitude)
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = if self.negative == rhs.negative {
            Self {
                negative: self.negative,
                limbs: Self::add_abs(&self.limbs, &rhs.limbs),
            }
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Greater => Self {
                    negative: self.negative,
                    limbs: Self::sub_abs(&self.limbs, &rhs.limbs),
                },
                Ordering::Less => Self {
                    negative: rhs.negative,
                    limbs: Self::sub_abs(&rhs.limbs, &self.limbs),
                },
                Ordering::Equal => Self::zero(),
            }
        };
        result.normalize();
        result
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = limbs[index] as u64 + a as u64 * b as u64 + carry;
                limbs[index] = (value % BASE) as u32;
                carry = value / BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == limbs.len() {
                    limbs.push(0);
                }
                let value = limbs[index] as u64 + carry;
                limbs[index] = (value % BASE) as u32;
                carry = value / BASE;
                index += 1;
            }
        }
        let mut result = Self {
            negative: self.negative != rhs.negative,
            limbs,
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        if self.is_zero() {
            return Ok((Self::zero(), Self::zero()));
        }

        let dividend = self.abs();
        let divisor = rhs.abs();
        if dividend.abs_cmp(&divisor) == Ordering::Less {
            return Ok((Self::zero(), self.clone()));
        }

        let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
        let mut remainder = Vec::<u32>::new();
        for &limb in dividend.limbs.iter().rev() {
            remainder.insert(0, limb);
            while remainder.last() == Some(&0) {
                remainder.pop();
            }
            let mut low = 0u32;
            let mut high = (BASE - 1) as u32;
            while low < high {
                let mid = low + ((high - low) / 2) + 1;
                let product = Self::mul_small_abs(&divisor.limbs, mid);
                let cmp = Self {
                    negative: false,
                    limbs: remainder.clone(),
                }
                .abs_cmp(&Self {
                    negative: false,
                    limbs: product,
                });
                if cmp != Ordering::Less {
                    low = mid;
                } else {
                    high = mid - 1;
                }
            }
            quotient_be.push(low);
            if low != 0 {
                let product = Self::mul_small_abs(&divisor.limbs, low);
                remainder = Self::sub_abs(&remainder, &product);
            }
        }
        quotient_be.reverse();
        while quotient_be.last() == Some(&0) {
            quotient_be.pop();
        }

        let mut quotient = Self {
            negative: self.negative != rhs.negative,
            limbs: quotient_be,
        };
        quotient.normalize();
        let mut rem = Self {
            negative: self.negative,
            limbs: remainder,
        };
        rem.normalize();
        Ok((quotient, rem))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: POW_LIMIT,
            });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();

            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

        let mut b = rhs.abs();

            let (_, remainder) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = remainder.abs();
        }
        a.negative = false;
        a
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn abs_cmp(&self, other: &Self) -> Ordering {
        match self.limbs.len().cmp(&other.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(other.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        non_equal => return non_equal,
                    }
                }
                Ordering::Equal
            }
            non_equal => non_equal,
        }
    }

    fn mul_small_assign(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }

        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry > 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_assign(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut index = 0;
        while carry > 0 {
            if index == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[index] as u64 + carry;
            self.limbs[index] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            index += 1;
        }
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let len = a.limbs.len().max(b.limbs.len());
        let mut result = Vec::with_capacity(len + 1);
        let mut carry = 0u64;
        for i in 0..len {
            let av = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = av + bv + carry;
            result.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry > 0 {
            result.push(carry as u32);
        }
        Self {
            negative: false,
            limbs: result,
        }
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut result = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let av = a.limbs[i] as i64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut value = av - bv - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(value as u32);
        }
        let mut answer = Self {
            negative: false,
            limbs: result,
        };
        answer.normalize();
        answer
    }

    fn mul_abs(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut accum = vec![0u128; a.limbs.len() + b.limbs.len() + 1];
        for (i, &av) in a.limbs.iter().enumerate() {
            for (j, &bv) in b.limbs.iter().enumerate() {
                accum[i + j] += av as u128 * bv as u128;
            }
        }

        let base = Self::BASE as u128;
        for i in 0..accum.len() - 1 {
            let carry = accum[i] / base;
            accum[i] %= base;
            accum[i + 1] += carry;
        }

        let mut result = Self {
            negative: false,
            limbs: accum.into_iter().map(|v| v as u32).collect(),
        };
        result.normalize();
        result
    }

    fn div_rem_abs(dividend: &Self, divisor: &Self) -> (Self, Self) {
        if dividend.abs_cmp(divisor) == Ordering::Less {
            let mut rem = dividend.clone();
            rem.negative = false;
            return (Self::zero(), rem);
        }

        let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
        let mut remainder = Self::zero();

        for &limb in dividend.limbs.iter().rev() {
            if !remainder.is_zero() {
                remainder.limbs.insert(0, limb);
            } else if limb != 0 {
                remainder.limbs.push(limb);
            }
            remainder.normalize();

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = divisor.mul_small_abs(mid as u32);
                match product.abs_cmp(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }

            if best != 0 {
                let product = divisor.mul_small_abs(best as u32);
                remainder = Self::sub_abs(&remainder, &product);
            }
            quotient_be.push(best as u32);
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn mul_small_abs(&self, factor: u32) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result.mul_small_assign(factor);
        result
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
            b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let mut start = 0;
        let mut negative = false;
        match bytes[0] {
            b'+' => start = 1,
            b'-' => {
                negative = true;
                start = 1;
            }
            _ => {}
        }
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            if Self::digit_value(bytes[i]).is_none_or(|d| d >= radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                malformed_separator = Some(i);
                break;
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = start;
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == max_digits.saturating_add(1) {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte != b'_' {
                value.mul_small_assign(radix);
                value.add_small_assign(Self::digit_value(byte).unwrap());
            }
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.clone();
        value.negative = false;
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(b'0');
        } else {
            while !value.is_zero() {
                let (quotient, remainder) = value.div_small(radix);
                digits.push(Self::digit_char(remainder));
                value = quotient;
            }
            digits.reverse();
        }

        while digits.len() < min_digits {
            digits.insert(0, b'0');
        }

        let mut output = String::new();
        if self.negative && !self.is_zero() {
            output.push('-');
        }
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                output.push('_');
            }
            output.push(digit as char);
        }
        Ok(output)
    }

    fn div_small(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = vec![0u32; self.limbs.len()];
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let value = remainder * Self::BASE + self.limbs[i] as u64;
            quotient[i] = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        let mut result = Self {
            negative: false,
            limbs: quotient,
        };
        result.normalize();
        (result, remainder as u32)
    }

    fn digit_char(digit: u32) -> u8 {
        if digit < 10 {
            b'0' + digit as u8
        } else {
            b'a' + (digit - 10) as u8
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::sub_abs(self, rhs);
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = Self::sub_abs(rhs, self);
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut result = Self::mul_abs(self, rhs);
        result.negative = !result.is_zero() && (self.negative != rhs.negative);
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::div_rem_abs(self, rhs);
        quotient.negative =
            !quotient.is_zero() && (self.negative != rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }

        let mut result = Self::one();
        let mut base = self.clone();
        let mut n = exponent;
        while n > 0 {
            if n & 1 == 1 {
                result = result.mul(&base);
            }
            n >>= 1;
            if n > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, remainder) = Self::div_rem_abs(&a, &b);
            a = b;
            b = remainder;
        }
        a.negative = false;
        a
    }
}

    fn mul_small_abs(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_abs(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }

    fn div_small_abs(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let value = rem * Self::BASE + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            rem = value % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    fn mul_abs(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u32; a.limbs.len() + b.limbs.len()];
        for (i, &x) in a.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.limbs.iter().enumerate() {
                let k = i + j;
                let value = out[k] as u64 + x as u64 * y as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut k = i + b.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let value = out[k] as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self {
            negative: false,
            limbs: out,
        };
        result.normalize();
        result
    }

    fn shift_base_add(&mut self, digit: u32) {
        if self.is_zero() {
            if digit != 0 {
                self.limbs.push(digit);
            }
            return;
        }
        self.limbs.insert(0, digit);
    }

    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        let mut quotient = vec![0u32; a.limbs.len()];
        let mut remainder = Self::zero();

        for i in (0..a.limbs.len()).rev() {
            remainder.shift_base_add(a.limbs[i]);
            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let mut product = b.clone();
                product.mul_small_abs(mid as u32);
                match product.abs_cmp(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }
            quotient[i] = best as u32;
            if best != 0 {
                let mut product = b.clone();
                product.mul_small_abs(best as u32);
                remainder = Self::sub_abs(&remainder, &product);
            }
        }

        let mut q = Self {
            negative: false,
            limbs: quotient,
        };
        q.normalize();
        (q, remainder)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let digit_value = |byte: u8| -> Option<u32> {
            match byte {
                b'0'..=b'9' => Some((byte - b'0') as u32),
                b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
                _ => None,
            }
        };

        let mut first_invalid = None;
        for i in start..bytes.len() {
            if bytes[i] != b'_' {
                match digit_value(bytes[i]) {
                    Some(v) if v < radix => {}
                    _ => {
                        first_invalid = Some(i);
                        break;
                    }
                }
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let before = i > start && digit_value(bytes[i - 1]).is_some_and(|v| v < radix);
                let after = i + 1 < bytes.len()
                    && digit_value(bytes[i + 1]).is_some_and(|v| v < radix);
                if !(before && after) {
                    first_separator = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = first_separator {
            return Err(Error::Separator { index });
        }

        let digits: Vec<(usize, u32)> = (start..bytes.len())
            .filter(|&i| bytes[i] != b'_')
            .map(|i| (i, digit_value(bytes[i]).unwrap()))
            .collect();

        if digits.len() > 1 && digits[0].1 == 0 {
            return Err(Error::LeadingZero { index: digits[0].0 });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digits[max_digits].0,
            });
        }

        let mut result = Self::zero();
        for &(_, digit) in &digits {
            result.mul_small_abs(radix);
            result.add_small_abs(digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let alphabet = b"0123456789abcdefghijklmnopqrstuvwxyz";
        let mut value = self.clone();
        value.negative = false;
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(b'0');
        } else {
            while !value.is_zero() {
                let rem = value.div_small_abs(radix);
                digits.push(alphabet[rem as usize]);
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, b'0');
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(digit as char);
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::sub_abs(self, rhs);
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = Self::sub_abs(rhs, self);
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut result = Self::mul_abs(self, rhs);
        result.negative = !result.is_zero() && (self.negative != rhs.negative);
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::div_rem_abs(self, rhs);
        quotient.negative =
            !quotient.is_zero() && (self.negative != rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: LIMIT,
            });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power != 0 {
            if power & 1 != 0 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, remainder) = Self::div_rem_abs(&a, &b);
            a = b;
            b = remainder;
        }
        a.negative = false;
        a
    }
}
