```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;
        if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left = a * (d / g);
        let right = c * (b / g);
        Self::from_i128(left + right, (b / g) * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left = a * (d / g);
        let right = c * (b / g);
        Self::from_i128(left - right, (b / g) * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(c.unsigned_abs(), b as u128) as i128;
        Self::from_i128((a / g1) * (c / g2), (b / g2) * (d / g1))
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g1 = gcd_u128(a.unsigned_abs(), c.unsigned_abs()) as i128;
        let g2 = gcd_u128(b as u128, d as u128) as i128;
        Self::from_i128((a / g1) * (d / g2), (b / g2) * (c / g1))
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }
        let num_end = i;

        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let num_digits = num_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(parse_err(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[num_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_mag = parse_magnitude(&bytes[num_start..num_end], num_limit)
            .ok_or_else(|| parse_err(0, ParseKind::OutOfRange))?;

        let den_mag = if let (Some(ds), Some(de)) = (den_start, den_end) {
            Some(
                parse_magnitude(&bytes[ds..de], i64::MAX as u128)
                    .ok_or_else(|| parse_err(ds, ParseKind::OutOfRange))?,
            )
        } else {
            None
        };

        if let (Some(ds), Some(mag)) = (den_start, den_mag) {
            if mag == 0 {
                return Err(parse_err(ds, ParseKind::ZeroDenominator));
            }
        }

        if let (Some(ds), Some(de), Some(mag)) = (den_start, den_end, den_mag) {
            let slash = ds - 1;
            if mag == 1 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, mag) > 1 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
            let _ = de;
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        Ok(Self {
            num: signed_num,
            den: den_mag.unwrap_or(1) as i64,
        })
    }
}

fn parse_err(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &byte in digits {
        let digit = (byte - b'0') as u128;
        if value > (limit.saturating_sub(digit)) / 10 {
            return None;
        }
        value = value * 10 + digit;
    }
    Some(value)
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
