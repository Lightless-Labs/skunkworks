```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    paused_at: Option<u64>,
    remaining: u64,
    started: bool,
}

#[derive(Clone)]
struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_before(a: &Waiting, b: &Waiting) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }

    match (a.paused_at, b.paused_at) {
        (Some(at_a), Some(at_b)) => {
            at_a < at_b || (at_a == at_b && a.customer.id < b.customer.id)
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            a.customer.arrival < b.customer.arrival
                || (a.customer.arrival == b.customer.arrival
                    && a.customer.id > b.customer.id)
        }
    }
}

fn best_waiter_index(waiting: &[Waiting]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| {
            if waiting_before(a, b) {
                std::cmp::Ordering::Less
            } else if waiting_before(b, a) {
                std::cmp::Ordering::Greater
            } else {
                std::cmp::Ordering::Equal
            }
        })
        .map(|(index, _)| index)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut seen = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut now = 0u64;

    loop {
        let mut had_phase = false;

        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let finished = running.take().unwrap();
                events.push(Event {
                    time: now,
                    customer: finished.customer.id,
                    kind: Kind::Finish,
                });
                had_phase = true;
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                paused_at: None,
                started: false,
            });
            next_arrival += 1;
            had_phase = true;
        }

        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                if let Some(index) = best_waiter_index(&waiting) {
                    if waiting[index].customer.priority < active.customer.priority {
                        let by = waiting[index].customer.id;
                        let active = running.take().unwrap();
                        let remaining = active.finish - now;
                        events.push(Event {
                            time: now,
                            customer: active.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            customer: active.customer,
                            paused_at: Some(now),
                            remaining,
                            started: true,
                        });
                        had_phase = true;
                    }
                }
            }
        }

        if running.is_none() {
            while let Some(index) = best_waiter_index(&waiting) {
                let item = waiting.remove(index);
                let cost = if last_dispatched.is_some_and(|id| id != item.customer.id) {
                    switch_cost
                } else {
                    0
                };
                let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: item.customer.id,
                    time: now,
                })?;
                let finish = work_at.checked_add(item.remaining).ok_or(Error::TimeOverflow {
                    customer: item.customer.id,
                    time: now,
                })?;

                let kind = if item.started {
                    Kind::Resume {
                        remaining: item.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: item.remaining,
                        work_at,
                        finish,
                    }
                };
                events.push(Event {
                    time: now,
                    customer: item.customer.id,
                    kind,
                });
                last_dispatched = Some(item.customer.id);
                running = Some(Running {
                    customer: item.customer,
                    remaining: item.remaining,
                    work_at,
                    finish,
                });
                had_phase = true;

                if finish == now {
                    let finished = running.take().unwrap();
                    events.push(Event {
                        time: now,
                        customer: finished.customer.id,
                        kind: Kind::Finish,
                    });
                    continue;
                }
                break;
            }
        }

        if next_arrival == arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for time in [active.finish, active.work_at] {
                if time > now && next_time.is_none_or(|current| time < current) {
                    next_time = Some(time);
                }
            }
        }

        if let Some(next) = next_time {
            now = next;
        } else if had_phase {
            // All current-time work was handled; any remaining activity must have
            // a future arrival or active boundary.
            break;
        } else {
            break;
        }
    }

    Ok(events)
}
```
