use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    ready_at: u64,
    remaining: Option<u64>,
    segment_order: u64,
}

pub struct Pool {
    entries: Vec<Entry>,
    by_id: HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
    next_segment_order: u64,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_token(tag, 1, 8) {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut resolved = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (j, name) in job.dependencies.iter().enumerate() {
                if !seen.insert(name.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(name) {
                    Some(&dep) => deps.push(dep),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            resolved.push(deps);
        }

        if has_cycle(&resolved) {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                ready_at: job.release_at,
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                dependencies: resolved[i].clone(),
                remaining: None,
                segment_order: 0,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
            next_segment_order: 0,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &index = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.entries[index].state.clone())
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.has_run {
            self.horizon
        } else {
            0
        };

        loop {
            self.process_timestamp(time, &mut lines);
            let Some(next) = self.next_event_after(time, until) else {
                break;
            };
            time = next;
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn process_timestamp(&mut self, time: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.state {
                State::Running { finish_at, .. } if finish_at == time => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in completions {
            let attempt = match self.entries[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            if attempt > self.entries[i].job.failures {
                lines.push(format!("{} finish {} {} ok", time, self.entries[i].job.id, attempt));
                self.entries[i].state = State::Succeeded;
                self.entries[i].remaining = None;
            } else {
                lines.push(format!("{} finish {} {} fail", time, self.entries[i].job.id, attempt));
                self.entries[i].remaining = None;
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let delay = self.retry_delay(attempt);
                    let ready_at = time.saturating_add(delay);
                    lines.push(format!(
                        "{} retry {} {} {}",
                        time, self.entries[i].job.id, attempt, ready_at
                    ));
                    self.entries[i].ready_at = ready_at;
                    self.entries[i].state = State::Waiting {
                        ready_at,
                        attempt,
                    };
                }
            }
        }

        self.propagate_blocks(time, lines);
        self.schedule(time, lines);
    }

    fn retry_delay(&self, attempt: u8) -> u64 {
        let mut delay = self.retry_base;
        for _ in 1..attempt {
            delay = delay.saturating_mul(2).min(self.retry_cap);
            if delay == self.retry_cap {
                break;
            }
        }
        delay.min(self.retry_cap)
    }

    fn propagate_blocks(&mut self, time: u64, lines: &mut Vec<String>) {
        let mut newly = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.entries.len() {
                if !matches!(self.entries[i].state, State::Waiting { .. }) {
                    continue;
                }
                let bad_dep = self.entries[i]
                    .dependencies
                    .iter()
                    .find(|&&d| {
                        matches!(
                            self.entries[d].state,
                            State::Failed | State::Blocked
                        )
                    })
                    .copied();
                if let Some(dep) = bad_dep {
                    self.entries[i].state = State::Blocked;
                    newly.push((i, dep));
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly.sort_by(|(a, _), (b, _)| self.entries[*a].job.id.cmp(&self.entries[*b].job.id));
        for (i, dep) in newly {
            lines.push(format!(
                "{} skip {} {}",
                time, self.entries[i].job.id, self.entries[dep].job.id
            ));
        }
    }

    fn schedule(&mut self, time: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| {
                    if !matches!(e.state, State::Waiting { .. }) || e.ready_at > time {
                        return None;
                    }
                    if e.dependencies
                        .iter()
                        .any(|&d| !matches!(self.entries[d].state, State::Succeeded))
                    {
                        return None;
                    }
                    Some(i)
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut action = None;
            for candidate in candidates {
                let tag = self.entries[candidate].job.tag.as_deref();
                let running: Vec<usize> = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| matches!(e.state, State::Running { .. }).then_some(i))
                    .collect();

                let tag_busy = tag.is_some_and(|t| {
                    running.iter().any(|&i| self.entries[i].job.tag.as_deref() == Some(t))
                });
                if running.len() < self.capacity && !tag_busy {
                    action = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&v| {
                        let victim_tag = self.entries[v].job.tag.as_deref();
                        let slot_after = running.len() - 1 < self.capacity;
                        let tag_after = tag.map_or(true, |t| {
                            !running.iter().any(|&r| {
                                r != v && self.entries[r].job.tag.as_deref() == Some(t)
                            })
                        });
                        slot_after
                            && tag_after
                            && self.entries[candidate].job.priority
                                >= self.entries[v].job.priority.saturating_add(2)
                            && (victim_tag != tag || tag_busy)
                    })
                    .collect();
                victims.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            self.entries[b]
                                .segment_order
                                .cmp(&self.entries[a].segment_order)
                        })
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });
                if let Some(victim) = victims.first().copied() {
                    action = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = action else {
                break;
            };

            if let Some(v) = victim {
                let finish_at = match self.entries[v].state {
                    State::Running { finish_at, .. } => finish_at,
                    _ => continue,
                };
                let attempt = match self.entries[v].state {
                    State::Running { attempt, .. } => attempt,
                    _ => continue,
                };
                let remaining = finish_at.saturating_sub(time);
                self.entries[v].ready_at = time;
                self.entries[v].remaining = Some(remaining);
                self.entries[v].state = State::Waiting {
                    ready_at: time,
                    attempt,
                };
                lines.push(format!(
                    "{} preempt {} by {}",
                    time, self.entries[v].job.id, self.entries[candidate].job.id
                ));
            }

            let attempt;
            let remaining;
            match self.entries[candidate].remaining.take() {
                Some(saved) => {
                    remaining = saved;
                    attempt = match self.entries[candidate].state {
                        State::Waiting { attempt, .. } => attempt,
                        _ => 0,
                    };
                }
                None => {
                    attempt = match self.entries[candidate].state {
                        State::Waiting { attempt, .. } => attempt.saturating_add(1),
                        _ => 1,
                    };
                    remaining = self.entries[candidate].job.duration;
                }
            }
            let finish_at = time.saturating_add(remaining);
            self.next_segment_order = self.next_segment_order.saturating_add(1);
            self.entries[candidate].segment_order = self.next_segment_order;
            self.entries[candidate].state = State::Running { attempt, finish_at };
            lines.push(format!(
                "{} start {} {} {}",
                time, self.entries[candidate].job.id, attempt, remaining
            ));
        }
    }

    fn effective_priority(&self, index: usize, time: u64) -> u8 {
        let base = self.entries[index].job.priority;
        if self.aging_ms == 0 {
            return base;
        }
        let waited = time.saturating_sub(self.entries[index].ready_at);
        let gain = (waited / self.aging_ms).min(9) as u8;
        base.saturating_add(gain).min(9)
    }

    fn next_event_after(&self, time: u64, until: u64) -> Option<u64> {
        self.entries
            .iter()
            .filter_map(|e| match e.state {
                State::Running { finish_at, .. } if finish_at > time && finish_at <= until => {
                    Some(finish_at)
                }
                State::Waiting { .. } if e.ready_at > time && e.ready_at <= until => {
                    Some(e.ready_at)
                }
                _ => None,
            })
            .min()
    }
}

fn valid_token(s: &str, min: usize, max: usize) -> bool {
    let bytes = s.as_bytes();
    (min..=max).contains(&bytes.len())

            .iter()

            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
            sequence: 0,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.entries
            .iter()
            .find(|e| e.job.id == id)
            .map(|e| e.state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut now = if self.has_run { self.horizon } else { 0 };
        let mut process_now = !self.has_run;

        loop {
            if process_now {
                self.process_timestamp(now, &mut lines);
                process_now = false;
            }
            if now == until {
                break;
            }

            let mut next = None;
            for e in &self.entries {
                match e.state {
                    State::Running { finish_at, .. } if finish_at > now => {
                        next = Some(next.map_or(finish_at, |n: u64| n.min(finish_at)));
                    }
                    State::Waiting { ready_at, .. } if ready_at > now => {
                        next = Some(next.map_or(ready_at, |n: u64| n.min(ready_at)));
                    }
                    _ => {}
                }
            }
            let Some(t) = next.filter(|&t| t <= until) else {
                break;
            };
            now = t;
            process_now = true;
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }
}

fn valid_name(s: &str, min: usize, max: usize) -> bool {
    let b = s.as_bytes();
    (min..=max).contains(&b.len())
        && b.iter()
            .all(|&c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == b'-')
}

impl Pool {
    fn process_timestamp(&mut self, t: u64, lines: &mut Vec<String>) {
        let mut finished: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.state {
                State::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        finished.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in finished {
            let attempt = match self.entries[i].state {
                State::Running { attempt, .. } => attempt,
                _ => unreachable!(),
            };
            let id = self.entries[i].job.id.clone();
            if attempt > self.entries[i].job.failures {
                self.entries[i].state = State::Succeeded;
                self.entries[i].remaining = None;
                lines.push(format!("{t} finish {id} {attempt} ok"));
            } else {
                lines.push(format!("{t} finish {id} {attempt} fail"));
                self.entries[i].remaining = None;
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self
                        .retry_base
                        .checked_mul(1u64.checked_shl(shift).unwrap_or(u64::MAX))
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready = t.saturating_add(delay);
                    self.entries[i].waiting_since = ready;
                    self.entries[i].state = State::Waiting {
                        ready_at: ready,
                        attempt,
                    };
                    lines.push(format!("{t} retry {id} {attempt} {ready}"));
                }
            }
        }

        let mut blocked = Vec::new();
        loop {
            let newly: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| {
                    if !matches!(e.state, State::Waiting { .. }) {
                        return None;
                    }
                    e.dependencies.iter().find_map(|&d| {
                        matches!(self.entries[d].state, State::Failed | State::Blocked)
                            .then_some((i, d))
                    })
                })
                .collect();
            if newly.is_empty() {
                break;
            }
            for (i, dep) in newly {
                let dependency_id = self.entries[dep].job.id.clone();
                self.entries[i].state = State::Blocked;
                blocked.push((i, dependency_id));
            }
        }
        blocked.sort_by(|(a, _), (b, _)| self.entries[*a].job.id.cmp(&self.entries[*b].job.id));
        for (i, dep) in blocked {
            lines.push(format!("{t} skip {} {dep}", self.entries[i].job.id));
        }

        loop {
            let mut candidates: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| {
                    let ready = match e.state {
                        State::Waiting { ready_at, .. } => ready_at <= t,
                        _ => false,
                    };
                    let deps_ok = e
                        .dependencies
                        .iter()
                        .all(|&d| matches!(self.entries[d].state, State::Succeeded));
                    (ready && deps_ok).then_some(i)
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa).then(a.cmp(&b))
            });

            let mut started = false;
            for c in candidates {
                let tag = self.entries[c].job.tag.clone();
                let running: Vec<usize> = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| matches!(e.state, State::Running { .. }).then_some(i))
                    .collect();
                let slots_free = running.len() < self.capacity;
                let tag_free = tag.as_ref().is_none_or(|tag| {
                    !running
                        .iter()
                        .any(|&i| self.entries[i].job.tag.as_ref() == Some(tag))
                });

                if slots_free && tag_free {
                    self.start_job(c, t, lines);
                    started = true;
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&v| {
                        self.entries[c].job.priority
                            >= self.entries[v].job.priority.saturating_add(2)
                    })
                    .filter(|&v| {
                        let slots_after = running.len() - 1 < self.capacity;
                        let tag_after = tag.as_ref().is_none_or(|candidate_tag| {
                            !running.iter().any(|&other| {
                                other != v
                                    && self.entries[other].job.tag.as_ref()
                                        == Some(candidate_tag)
                            })
                        });
                        slots_after && tag_after
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            self.entries[b]
                                .started_order
                                .cmp(&self.entries[a].started_order)
                        })
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });

                if let Some(v) = victims.first().copied() {
                    let (attempt, finish_at) = match self.entries[v].state {
                        State::Running { attempt, finish_at } => (attempt, finish_at),
                        _ => unreachable!(),
                    };
                    let remaining = finish_at - t;
                    let victim_id = self.entries[v].job.id.clone();
                    let candidate_id = self.entries[c].job.id.clone();
                    lines.push(format!("{t} preempt {victim_id} by {candidate_id}"));
                    self.entries[v].remaining = Some(remaining);
                    self.entries[v].waiting_since = t;
                    self.entries[v].state = State::Waiting {
                        ready_at: t,
                        attempt,
                    };
                    self.start_job(c, t, lines);
                    started = true;
                    break;
                }
            }

            if !started {
                break;
            }
        }
    }

    fn effective_priority(&self, i: usize, t: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.entries[i].job.priority;
        }
        let waited = t.saturating_sub(self.entries[i].waiting_since);
        let age = waited / self.aging_ms;
        (u64::from(self.entries[i].job.priority) + age).min(9) as u8
    }

    fn start_job(&mut self, i: usize, t: u64, lines: &mut Vec<String>) {
        let (attempt, remaining, resumed) = match self.entries[i].state {
            State::Waiting { attempt, .. } => {
                if let Some(rem) = self.entries[i].remaining.take() {
                    (attempt, rem, true)
                } else {
                    (attempt.saturating_add(1), self.entries[i].job.duration, false)
                }
            }
            _ => unreachable!(),
        };
        let finish = t.saturating_add(remaining);
        if !resumed {
            self.entries[i].waiting_since = t;
        }
        self.sequence = self.sequence.saturating_add(1);
        self.entries[i].started_order = self.sequence;
        self.entries[i].state = State::Running {
            attempt,
            finish_at: finish,
        };
        lines.push(format!(
            "{t} start {} {attempt} {remaining}",
            self.entries[i].job.id
        ));
    }
}
