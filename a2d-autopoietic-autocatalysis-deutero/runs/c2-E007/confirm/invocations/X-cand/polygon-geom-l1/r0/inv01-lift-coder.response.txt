#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

fn prepare(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if !(-1_000_000_000..=1_000_000_000).contains(&point.x)
            || !(-1_000_000_000..=1_000_000_000).contains(&point.y)
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.last() == points.first() {
        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    Ok(&points[..effective_len])
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128).unsigned_abs();
    let dy = (a.y as i128 - b.y as i128).unsigned_abs();
    let squared_milli = (dx * dx + dy * dy) * 1_000_000;

    // Binary search for floor(sqrt(squared_milli)).
    let mut low = 0_u128;
    let mut high = 1_u128 << 42;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid * mid <= squared_milli {
            low = mid;
        } else {
            high = mid;
        }
    }

    // Compare against the exact half-integer threshold: (low + 1/2)^2.
    if 4 * squared_milli >= 4 * low * low + 4 * low + 1 {
        (low + 1) as u64
    } else {
        low as u64
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare(points)?;
    let n = polygon.len();

    let mut signed_area2 = 0_i128;
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge = None;
    let mut longest_pair = None;

    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];

        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let pair = if a <= b { (a, b) } else { (b, a) };
        match longest_edge {
            None => {
                longest_edge = Some(i);
                longest_pair = Some(pair);
            }
            Some(best_index) => {
                let best_length = edge_milli[best_index];
                if length > best_length
                    || (length == best_length
                        && (Some(pair) < longest_pair
                            || (Some(pair) == longest_pair && i < best_index)))
                {
                    longest_edge = Some(i);
                    longest_pair = Some(pair);
                }
            }
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
