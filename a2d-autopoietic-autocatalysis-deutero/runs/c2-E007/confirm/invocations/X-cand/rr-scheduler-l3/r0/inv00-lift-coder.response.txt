```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            completion: Vec::new(),
            end: 0,
        });
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completed: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut next_event = 0usize;
    let mut last_dispatched: Option<u32> = None;

    let admit_through = |now: u64,
                         next_event: &mut usize,
                         queues: &mut Vec<VecDeque<usize>>,
                         remaining: &mut Vec<u32>,
                         completed: &mut Vec<Option<u64>>| {
        while *next_event < events.len() {
            let index = events[*next_event];
            let job = &jobs[index];
            let arrival = u64::from(job.arrival);
            if arrival > now {
                break;
            }
            *next_event += 1;

            if remaining[index] == 0 {
                completed[index] = Some(arrival);
            } else {
                queues[job.level].push_back(index);
            }
        }
    };

    loop {
        admit_through(
            now,
            &mut next_event,
            &mut queues,
            &mut remaining,
            &mut completed,
        );

        let selected = queues.iter_mut().find_map(VecDeque::pop_front);

        let Some(index) = selected else {
            if next_event == events.len() {
                break;
            }

            let next_time = u64::from(jobs[events[next_event]].arrival);
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatched = None;
            }
            continue;
        };

        let job = &jobs[index];
        if last_dispatched.is_some_and(|last_id| last_id != job.id) {
            now += u64::from(switch_cost);
        }

        admit_through(
            now,
            &mut next_event,
            &mut queues,
            &mut remaining,
            &mut completed,
        );

        let level = job.level;
        let consumed = remaining[index].min(quanta[level]);
        let start = now;
        now += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id: job.id,
            start,
            end: now,
            finished,
            level,
        });

        admit_through(
            now,
            &mut next_event,
            &mut queues,
            &mut remaining,
            &mut completed,
        );

        if finished {
            completed[index] = Some(now);
        } else {
            let next_level = (level + 1).min(quanta.len() - 1);
            queues[next_level].push_back(index);
        }

        last_dispatched = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completed[index].unwrap_or(0)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}

use std::collections::VecDeque;
```
