#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone, Copy)]
struct Component<'a> {
    text: &'a str,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<Component<'_>> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }

        result.push(Component {
            text: &path[start..i],
            at: start,
        });
    }

    result
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|c| c.text == "." || c.text == "..")
}

fn finish_path(parts: &[String], trailing: bool) -> String {
    if parts.is_empty() {
        return "/".to_string();
    }

    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if trailing {
        result.push('/');
    }
    result
}

fn reduce_component(
    parts: &mut Vec<String>,
    component: &str,
    at: usize,
    absolute: bool,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if parts.last().is_some_and(|last| last != "..") {
                parts.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                parts.push("..".to_string());
            }
        }
        _ => parts.push(component.to_string()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let trailing = directory_intent(path);
    let mut parts = Vec::new();

    for component in components(path) {
        reduce_component(&mut parts, component.text, component.at, absolute)?;
    }

    if absolute {
        let mut result = finish_path(&parts, trailing && !parts.is_empty());
        if result == "/" {
            return Ok(result);
        }
        if trailing && !result.ends_with('/') {
            result.push('/');
        }
        Ok(result)
    } else {
        let mut result = if parts.is_empty() {
            ".".to_string()
        } else {
            parts.join("/")
        };

        if trailing && result != "." {
            result.push('/');
        }
        Ok(result)
    }
}

fn validated_context(value: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !value.starts_with('/') {
        return Err(error);
    }

    match normalize(value) {
        Ok(normalized) => {
            if normalized == "/" {
                Ok(Vec::new())
            } else {
                Ok(normalized
                    .trim_start_matches('/')
                    .trim_end_matches('/')
                    .split('/')
                    .map(str::to_string)
                    .collect())
            }
        }
        Err(_) => Err(error),
    }
}

fn request_base<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, &'a str, usize), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((false, &path[consumed..], consumed));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    if path.starts_with('/') {
        Ok((true, path, 0))
    } else {
        Ok((false, path, 0))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let (absolute, remainder, consumed) = request_base(path, home)?;
    let trailing = directory_intent(path);
    let mut parts = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        home_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    for component in components(remainder) {
        let at = if consumed == 0 {
            component.at
        } else {
            component.at + consumed
        };
        reduce_component(&mut parts, component.text, at, true)?;
    }

    Ok(finish_path(&parts, trailing && !parts.is_empty()))
}

fn link_table<'a>(links: &'a [Link]) -> Result<Vec<(usize, &'a str, &'a str)>, PathError> {
    let mut valid = Vec::with_capacity(links.len());

    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);

        if !path_valid || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((first, _, _)) = valid.iter().find(|(_, path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }

        valid.push((index, link.path.as_str(), link.target.as_str()));
    }

    Ok(valid)
}

struct LinkTraversal<'a> {
    links: &'a [(usize, &'a str, &'a str)],
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
    parts: Vec<String>,
}

impl LinkTraversal<'_> {
    fn current_path(&self) -> String {
        finish_path(&self.parts, false)
    }

    fn walk<'s>(
        &mut self,
        raw: &'s str,
        offset: usize,
        absolute: bool,
        allow_lookup: bool,
    ) -> Result<(), PathError> {
        validate_raw(raw)?;

        if absolute {
            self.parts.clear();
        }

        for component in components(raw) {
            match component.text {
                "." => {}
                ".." => {
                    if self.parts.is_empty() {
                        return Err(PathError::Escape {
                            at: offset + component.at,
                        });
                    }
                    self.parts.pop();
                }
                name => {
                    self.parts.push(name.to_string());

                    if allow_lookup {
                        let current = self.current_path();
                        if let Some(&(index, _, target)) =
                            self.links.iter().find(|(_, path, _)| *path == current)
                        {
                            if self.active.contains(&index) {
                                return Err(PathError::Loop { index });
                            }
                            if self.expansions >= self.max_expansions {
                                return Err(PathError::Limit { index });
                            }

                            self.expansions += 1;
                            self.active.push(index);

                            let target_absolute = target.starts_with('/');
                            if target_absolute {
                                self.parts.clear();
                            } else {
                                self.parts.pop();
                            }

                            let result = self.walk(target, 0, target_absolute, true);
                            self.active.pop();
                            result?;
                        }
                    }
                }
            }
        }

        Ok(())
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value, PathError::InvalidHome)?),
        None => None,
    };

    let links = link_table(links)?;
    validate_raw(path)?;

    let (absolute, remainder, consumed) = request_base(path, home)?;
    let trailing = directory_intent(path);
    let mut traversal = LinkTraversal {
        links: &links,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
        parts: Vec::new(),
    };

    if absolute {
        traversal.parts.clear();
    } else if path.starts_with('~') {
        traversal.parts = home_parts.unwrap_or_default();
    } else {
        traversal.parts.clear();
        for component in cwd_parts {
            traversal.parts.push(component);
            let current = traversal.current_path();
            if let Some(&(index, _, target)) =
                traversal.links.iter().find(|(_, link_path, _)| *link_path == current)
            {
                if traversal.active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if traversal.expansions >= traversal.max_expansions {
                    return Err(PathError::Limit { index });
                }

                traversal.expansions += 1;
                traversal.active.push(index);
                let target_absolute = target.starts_with('/');
                if target_absolute {
                    traversal.parts.clear();
                } else {
                    traversal.parts.pop();
                }
                let result = traversal.walk(target, 0, target_absolute, true);
                traversal.active.pop();
                result?;
            }
        }
    }

    traversal.walk(remainder, consumed, false, true)?;

    let mut result = finish_path(&traversal.parts, false);
    if trailing && result != "/" {
        result.push('/');
    }
    Ok(result)
}
