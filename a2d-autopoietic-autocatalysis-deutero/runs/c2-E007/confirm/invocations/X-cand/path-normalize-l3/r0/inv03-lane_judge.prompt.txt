#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str, base: usize) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut start = 0;
    while start < bytes.len() {
        if bytes[start] == b'/' {
            start += 1;
            continue;
        }
        let mut end = start;
        while end < bytes.len() && bytes[end] != b'/' {
            end += 1;
        }
        result.push((path[start..end].to_owned(), base + start));
        start = end;
    }
    result
}

fn apply_component(
    stack: &mut Vec<String>,
    component: &str,
    at: usize,
    absolute: bool,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_owned());
            }
        }
        _ => stack.push(component.to_owned()),
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = stack.join("/");
    if absolute {
        result.insert(0, '/');
    } else if result.is_empty() {
        result.push('.');
    }
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    for (component, at) in components(path, 0) {
        apply_component(&mut stack, &component, at, absolute)?;
    }
    Ok(render(&stack, absolute, intent))
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized, 0)
        .into_iter()
        .map(|(component, _)| component)
        .collect())
}

fn validated_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_components = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(normalized_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_components, home_components))
}

enum RequestStart {
    Absolute,
    Relative,
    Home { consumed: usize },
}

fn request_start(path: &str, home: Option<&[String]>) -> Result<RequestStart, PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            if home.is_none() {
                return Err(PathError::MissingHome);
            }
            return Ok(RequestStart::Home {
                consumed: if bytes.len() > 1 { 2 } else { 1 },
            });
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if path.starts_with('/') {
        Ok(RequestStart::Absolute)
    } else {
        Ok(RequestStart::Relative)
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (cwd_components, home_components) = validated_contexts(cwd, home)?;
    validate_raw(path)?;
    let start = request_start(path, home_components.as_deref())?;
    let intent = directory_intent(path);

    let (mut stack, request, base) = match start {
        RequestStart::Absolute => (Vec::new(), path, 0),
        RequestStart::Relative => (cwd_components, path, 0),
        RequestStart::Home { consumed } => (
            home_components.unwrap_or_default(),
            &path[consumed..],
            consumed,
        ),
    };

    for (component, at) in components(request, base) {
        apply_component(&mut stack, &component, at, true)?;
    }
    Ok(render(&stack, true, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path)
            && link.path != "/";
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

struct Traversal<'a> {
    stack: Vec<String>,
    links: &'a [Link],
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl Traversal<'_> {
    fn process(
        &mut self,
        component: &str,
        at: usize,
        links: &[Link],
        active: &mut Vec<usize>,
        expansions: &mut usize,
        max_expansions: usize,
    ) -> Result<(), PathError> {
        match component {
            "." => return Ok(()),
            ".." => {
                if self.stack.is_empty() {
                    return Err(PathError::Escape { at });
                }
                self.stack.pop();
                return Ok(());
            }
            _ => self.stack.push(component.to_owned()),
        }

        let matched = self
            .links
            .iter()
            .position(|link| link.path == render(&self.stack, true, false));
        let Some(index) = matched else {
            return Ok(());
        };

        if active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if *expansions >= max_expansions {
            return Err(PathError::Limit { index });
        }
        *expansions += 1;
        active.push(index);

        let link = &links[index];
        let target = link.target.as_str();
        if target.starts_with('/') {
            self.stack.clear();
        } else {
            self.stack.pop();
        }

        for (target_component, target_at) in components(target, 0) {
            self.process(
                &target_component,
                target_at,
                links,
                active,
                expansions,
                max_expansions,
            )?;
        }

        active.pop();
        Ok(())
    }

    fn process_sequence(
        &mut self,
        sequence: &[(String, usize)],
        links: &[Link],
        active: &mut Vec<usize>,
        expansions: &mut usize,
        max_expansions: usize,
    ) -> Result<(), PathError> {
        for (component, at) in sequence {
            self.process(
                component,
                *at,
                links,
                active,
                expansions,
                max_expansions,
            )?;
        }
        Ok(())
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_components, home_components) = validated_contexts(cwd, home)?;
    validate_links(links)?;
    validate_raw(path)?;

    let start = request_start(path, home_components.as_deref())?;
    let intent = directory_intent(path);
    let (initial, request, base) = match start {
        RequestStart::Absolute => (Vec::new(), path, 0),
        RequestStart::Relative => (cwd_components, path, 0),
        RequestStart::Home { consumed } => (
            home_components.unwrap_or_default(),
            &path[consumed..],
            consumed,
        ),
    };

    let mut traversal = Traversal {
        stack: Vec::new(),
        links,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };
    let mut active = Vec::new();
    let mut expansions = 0;

    if !path.starts_with('/') {
        for component in initial {
            traversal.process(
                &component,
                0,
                links,
                &mut active,
                &mut expansions,
                max_expansions,
            )?;
        }
    }

    let request_components = components(request, base);
    traversal.process_sequence(
        &request_components,
        links,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;
    Ok(render(&traversal.stack, true, intent))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone, Copy)]
struct Component<'a> {
    text: &'a str,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<Component<'_>> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }

        result.push(Component {
            text: &path[start..i],
            at: start,
        });
    }

    result
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|c| c.text == "." || c.text == "..")
}

fn finish_path(parts: &[String], trailing: bool) -> String {
    if parts.is_empty() {
        return "/".to_string();
    }

    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if trailing {
        result.push('/');
    }
    result
}

fn reduce_component(
    parts: &mut Vec<String>,
    component: &str,
    at: usize,
    absolute: bool,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if parts.last().is_some_and(|last| last != "..") {
                parts.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                parts.push("..".to_string());
            }
        }
        _ => parts.push(component.to_string()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let trailing = directory_intent(path);
    let mut parts = Vec::new();

    for component in components(path) {
        reduce_component(&mut parts, component.text, component.at, absolute)?;
    }

    if absolute {
        let mut result = finish_path(&parts, trailing && !parts.is_empty());
        if result == "/" {
            return Ok(result);
        }
        if trailing && !result.ends_with('/') {
            result.push('/');
        }
        Ok(result)
    } else {
        let mut result = if parts.is_empty() {
            ".".to_string()
        } else {
            parts.join("/")
        };

        if trailing && result != "." {
            result.push('/');
        }
        Ok(result)
    }
}

fn validated_context(value: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !value.starts_with('/') {
        return Err(error);
    }

    match normalize(value) {
        Ok(normalized) => {
            if normalized == "/" {
                Ok(Vec::new())
            } else {
                Ok(normalized
                    .trim_start_matches('/')
                    .trim_end_matches('/')
                    .split('/')
                    .map(str::to_string)
                    .collect())
            }
        }
        Err(_) => Err(error),
    }
}

fn request_base<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, &'a str, usize), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((false, &path[consumed..], consumed));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    if path.starts_with('/') {
        Ok((true, path, 0))
    } else {
        Ok((false, path, 0))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let (absolute, remainder, consumed) = request_base(path, home)?;
    let trailing = directory_intent(path);
    let mut parts = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        home_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    for component in components(remainder) {
        let at = if consumed == 0 {
            component.at
        } else {
            component.at + consumed
        };
        reduce_component(&mut parts, component.text, at, true)?;
    }

    Ok(finish_path(&parts, trailing && !parts.is_empty()))
}

fn link_table<'a>(links: &'a [Link]) -> Result<Vec<(usize, &'a str, &'a str)>, PathError> {
    let mut valid = Vec::with_capacity(links.len());

    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);

        if !path_valid || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((first, _, _)) = valid.iter().find(|(_, path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }

        valid.push((index, link.path.as_str(), link.target.as_str()));
    }

    Ok(valid)
}

struct LinkTraversal<'a> {
    links: &'a [(usize, &'a str, &'a str)],
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
    parts: Vec<String>,
}

impl LinkTraversal<'_> {
    fn current_path(&self) -> String {
        finish_path(&self.parts, false)
    }

    fn walk<'s>(
        &mut self,
        raw: &'s str,
        offset: usize,
        absolute: bool,
        allow_lookup: bool,
    ) -> Result<(), PathError> {
        validate_raw(raw)?;

        if absolute {
            self.parts.clear();
        }

        for component in components(raw) {
            match component.text {
                "." => {}
                ".." => {
                    if self.parts.is_empty() {
                        return Err(PathError::Escape {
                            at: offset + component.at,
                        });
                    }
                    self.parts.pop();
                }
                name => {
                    self.parts.push(name.to_string());

                    if allow_lookup {
                        let current = self.current_path();
                        if let Some(&(index, _, target)) =
                            self.links.iter().find(|(_, path, _)| *path == current)
                        {
                            if self.active.contains(&index) {
                                return Err(PathError::Loop { index });
                            }
                            if self.expansions >= self.max_expansions {
                                return Err(PathError::Limit { index });
                            }

                            self.expansions += 1;
                            self.active.push(index);

                            let target_absolute = target.starts_with('/');
                            if target_absolute {
                                self.parts.clear();
                            } else {
                                self.parts.pop();
                            }

                            let result = self.walk(target, 0, target_absolute, true);
                            self.active.pop();
                            result?;
                        }
                    }
                }
            }
        }

        Ok(())
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value, PathError::InvalidHome)?),
        None => None,
    };

    let links = link_table(links)?;
    validate_raw(path)?;

    let (absolute, remainder, consumed) = request_base(path, home)?;
    let trailing = directory_intent(path);
    let mut traversal = LinkTraversal {
        links: &links,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
        parts: Vec::new(),
    };

    if absolute {
        traversal.parts.clear();
    } else if path.starts_with('~') {
        traversal.parts = home_parts.unwrap_or_default();
    } else {
        traversal.parts.clear();
        for component in cwd_parts {
            traversal.parts.push(component);
            let current = traversal.current_path();
            if let Some(&(index, _, target)) =
                traversal.links.iter().find(|(_, link_path, _)| *link_path == current)
            {
                if traversal.active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if traversal.expansions >= traversal.max_expansions {
                    return Err(PathError::Limit { index });
                }

                traversal.expansions += 1;
                traversal.active.push(index);
                let target_absolute = target.starts_with('/');
                if target_absolute {
                    traversal.parts.clear();
                } else {
                    traversal.parts.pop();
                }
                let result = traversal.walk(target, 0, target_absolute, true);
                traversal.active.pop();
                result?;
            }
        }
    }

    traversal.walk(remainder, consumed, false, true)?;

    let mut result = finish_path(&traversal.parts, false);
    if trailing && result != "/" {
        result.push('/');
    }
    Ok(result)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_string(), start));
    }

    result
}

fn reduce(
    stack: &mut Vec<String>,
    parts: impl IntoIterator<Item = (String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|component| component != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let joined = stack.join("/");
    if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            let mut result = format!("/{joined}");
            if intent {
                result.push('/');
            }
            result
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        let mut result = joined;
        if intent {
            result.push('/');
        }
        result
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, components(path), absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized)
        .into_iter()
        .map(|(part, _)| part)
        .collect())
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let normalized = normalize(path).map_err(|_| PathError::InvalidHome)?;
            Ok(Some(
                components(&normalized)
                    .into_iter()
                    .map(|(part, _)| part)
                    .collect(),
            ))
        }
    }
}

enum RequestStart {
    Absolute,
    Cwd,
    Home,
}

fn request_start(
    path: &str,
    home: Option<&[String]>,
) -> Result<(RequestStart, usize), PathError> {
    if path.as_bytes()[0] == b'/' {
        return Ok((RequestStart::Absolute, 0));
    }

    if path.as_bytes()[0] == b'~' {
        if path.len() == 1 {
            return if home.is_some() {
                Ok((RequestStart::Home, 1))
            } else {
                Err(PathError::MissingHome)
            };
        }
        if path.as_bytes()[1] == b'/' {
            return if home.is_some() {
                Ok((RequestStart::Home, 2))
            } else {
                Err(PathError::MissingHome)
            };
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    Ok((RequestStart::Cwd, 0))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (start, skip) = request_start(path, home_parts.as_deref())?;
    let mut stack = match start {
        RequestStart::Absolute => Vec::new(),
        RequestStart::Cwd => cwd_parts,
        RequestStart::Home => home_parts.unwrap_or_default(),
    };

    reduce(
        &mut stack,
        components(&path[skip..]).into_iter().map(|(part, at)| (part, at + skip)),
        true,
    )?;

    Ok(render(&stack, true, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(first) = links[..index]
            .iter()
            .position(|previous| previous.path == link.path)
        {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
    }
    Ok(())
}

struct Traversal<'a> {
    stack: Vec<String>,
    links: &'a [Link],
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl<'a> Traversal<'a> {
    fn absolute_path(&self) -> String {
        if self.stack.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", self.stack.join("/"))
        }
    }

    fn traverse_parts(
        &mut self,
        parts: Vec<(String, usize)>,
        absolute: bool,
    ) -> Result<(), PathError> {
        for (part, at) in parts {
            match part.as_str() {
                "." => {}
                ".." => {
                    if self.stack.pop().is_none() {
                        if absolute {
                            return Err(PathError::Escape { at });
                        }
                    }
                }
                _ => {
                    self.stack.push(part);
                    if let Some(index) = self
                        .links
                        .iter()
                        .position(|link| link.path == self.absolute_path())
                    {
                        if self.active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if self.expansions >= self.max_expansions {
                            return Err(PathError::Limit { index });
                        }

                        self.expansions += 1;
                        self.active.push(index);
                        let link = &self.links[index];
                        let target_absolute = link.target.starts_with('/');
                        if target_absolute {
                            self.stack.clear();
                        } else {
                            self.stack.pop();
                        }

                        let target_parts = components(&link.target);
                        let result = self.traverse_parts(target_parts, true);
                        self.active.pop();
                        result?;
                    }
                }
            }
        }
        Ok(())
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (start, skip) = request_start(path, home_parts.as_deref())?;
    let mut traversal = Traversal {
        stack: Vec::new(),
        links,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    match start {
        RequestStart::Absolute => {}
        RequestStart::Cwd => {
            traversal.traverse_parts(cwd_parts.into_iter().map(|part| (part, 0)).collect(), true)?;
        }
        RequestStart::Home => {
            traversal.traverse_parts(
                home_parts.unwrap_or_default().into_iter().map(|part| (part, 0)).collect(),
                true,
            )?;
        }
    }

    let request_parts = components(&path[skip..])
        .into_iter()
        .map(|(part, at)| (part, at + skip))
        .collect();
    traversal.traverse_parts(request_parts, true)?;

    Ok(render(&traversal.stack, true, intent))
}
