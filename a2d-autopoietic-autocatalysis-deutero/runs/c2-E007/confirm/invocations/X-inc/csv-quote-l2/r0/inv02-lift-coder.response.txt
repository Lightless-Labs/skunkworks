#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = 0usize;
    if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        i = 3;
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Quoted { opening } => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(b);
                    i += 1;
                }
                let _ = opening;
            }
            State::Start => {
                if b == quote {
                    state = State::Quoted { opening: i };
                    record_started = true;
                    i += 1;
                } else if b == delimiter {
                    record.push(String::new());
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    record_started = false;
                    i += 1;
                } else if b == b'\r' {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
                        record.push(String::new());
                        records.push(record);
                        record = Vec::new();
                        record_index += 1;
                        record_started = false;
                        i += 2;
                    } else {
                        return Err(CsvError::BareCr {
                            record: record_index,
                            byte: i,
                        });
                    }
                } else {
                    field.push(b);
                    state = State::Unquoted;
                    record_started = true;
                    i += 1;
                }
            }
            State::Unquoted => {
                if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else if b == delimiter {
                    record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    state = State::Start;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    record_started = false;
                    state = State::Start;
                    i += 1;
                } else if b == b'\r' {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
                        record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                        field = Vec::new();
                        records.push(record);
                        record = Vec::new();
                        record_index += 1;
                        record_started = false;
                        state = State::Start;
                        i += 2;
                    } else {
                        return Err(CsvError::BareCr {
                            record: record_index,
                            byte: i,
                        });
                    }
                } else {
                    field.push(b);
                    i += 1;
                }
            }
            State::AfterQuote => {
                if b == delimiter {
                    record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    state = State::Start;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    record_started = false;
                    state = State::Start;
                    i += 1;
                } else if b == b'\r' {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
                        record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                        field = Vec::new();
                        records.push(record);
                        record = Vec::new();
                        record_index += 1;
                        record_started = false;
                        state = State::Start;
                        i += 2;
                    } else {
                        return Err(CsvError::BareCr {
                            record: record_index,
                            byte: i,
                        });
                    }
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started || !record.is_empty() || !field.is_empty() || matches!(state, State::AfterQuote) {
        record.push(String::from_utf8(field).expect("input is valid UTF-8"));
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            let header = iter.next();
            (header, iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(header_values) = header.as_ref() {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..header_values.len() {
                if let Some(first) = header_values[..second]
                    .iter()
                    .position(|value| value == &header_values[second])
                {
                    return Err(CsvError::DuplicateHeader {
                        name: header_values[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }

        for (row_index, row) in rows.iter().enumerate() {
            if row.len() != header_values.len() {
                return Err(CsvError::HeaderWidth {
                    record: row_index + 1,
                    expected: header_values.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }

        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
