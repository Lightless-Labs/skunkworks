#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: MergeSide,
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn make_changes(old: &[String], new: &[String], side: MergeSide) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut changes = Vec::new();
    let (mut i, mut j, mut base_pos) = (0usize, 0usize, 0usize);
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
                side: side.clone(),
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        let operation = if i == n {
            1 // Insert
        } else if j == m {
            0 // Delete
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            if delete_score > insert_score {
                0
            } else if insert_score > delete_score {
                1
            } else if (i + j) % 2 == 0 {
                0
            } else {
                1
            }
        };

        run_start.get_or_insert(base_pos);
        if operation == 0 {
            i += 1;
            base_pos += 1;
            deletes += 1;
        } else {
            inserts.push(new[j].clone());
            j += 1;
        }
    }

    flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;

    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

#[derive(Debug)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<Change>,
}

fn components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let all: Vec<Change> = left.iter().chain(right.iter()).cloned().collect();
    let mut seen = vec![false; all.len()];
    let mut result = Vec::new();

    for seed in 0..all.len() {
        if seen[seed] {
            continue;
        }
        seen[seed] = true;
        let mut stack = vec![seed];
        let mut indexes = Vec::new();

        while let Some(index) = stack.pop() {
            indexes.push(index);
            for other in 0..all.len() {
                if !seen[other] && interacts(&all[index], &all[other]) {
                    seen[other] = true;
                    stack.push(other);
                }
            }
        }

        let mut group: Vec<Change> = indexes.into_iter().map(|i| all[i].clone()).collect();
        let start = group.iter().map(|c| c.start).min().unwrap();
        let end = group.iter().map(|c| c.end).max().unwrap();
        group.sort_by_key(|c| (c.start, c.end));
        result.push(Component {
            start,
            end,
            changes: group,
        });
    }

    result.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));
    result
}

fn projection(base: &[String], component: &Component, side: MergeSide) -> Vec<String> {
    let mut changes: Vec<&Change> = component
        .changes
        .iter()
        .filter(|change| change.side == side)
        .collect();
    changes.sort_by_key(|change| (change.start, change.end));

    let mut output = Vec::new();
    let mut cursor = component.start;

    for change in changes {
        if change.start > cursor {
            output.extend(base[cursor..change.start].iter().cloned());
        }
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < component.end {
        output.extend(base[cursor..component.end].iter().cloned());
    }
    output
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = make_changes(&base_lines, &left_lines, MergeSide::Left);
    let right_changes = make_changes(&base_lines, &right_lines, MergeSide::Right);
    let components = components(&left_changes, &right_changes);

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0usize;

    for component in components {
        if cursor < component.start {
            append_lines(&mut output, &base_lines[cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(&base_lines, &component, MergeSide::Left);
        let right_projection = projection(&base_lines, &component, MergeSide::Right);

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let base_start = if component.end > component.start {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_start},{}",
                        component.end - component.start
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        cursor = component.end;
    }

    if cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[cursor..]);
    }

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut serialized = output.join("\n");
        serialized.push('\n');
        serialized
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
