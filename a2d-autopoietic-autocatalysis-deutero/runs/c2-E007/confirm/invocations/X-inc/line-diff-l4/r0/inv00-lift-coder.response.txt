#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

fn lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0, 0, 0);
    let mut changes = Vec::new();
    let mut active_start = None;
    let mut deleted = 0usize;
    let mut replacement = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active_start: &mut Option<usize>,
                 deleted: &mut usize,
                 replacement: &mut Vec<String>| {
        if let Some(start) = active_start.take() {
            changes.push(Change {
                start,
                end: start + *deleted,
                replacement: std::mem::take(replacement),
            });
            *deleted = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(
                &mut changes,
                &mut active_start,
                &mut deleted,
                &mut replacement,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            active_start.get_or_insert(base_pos);
            replacement.push(new[j].clone());
            j += 1;
        } else if j == m {
            active_start.get_or_insert(base_pos);
            deleted += 1;
            i += 1;
            base_pos += 1;
        } else {
            let delete_lcs = lcs[i + 1][j];
            let insert_lcs = lcs[i][j + 1];
            let delete = if delete_lcs > insert_lcs {
                true
            } else if insert_lcs > delete_lcs {
                false
            } else {
                (i + j) % 2 == 0
            };

            active_start.get_or_insert(base_pos);
            if delete {
                deleted += 1;
                i += 1;
                base_pos += 1;
            } else {
                replacement.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(
        &mut changes,
        &mut active_start,
        &mut deleted,
        &mut replacement,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;

    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

#[derive(Debug)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<(bool, Change)>,
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;

    for change in changes {
        if change.start > cursor {
            result.extend(base[cursor..change.start].iter().cloned());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

fn append_line(out: &mut String, line: &str) {
    out.push_str(line);
    out.push('\n');
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_changes(&base_lines, &left_lines);
    let right_changes = edit_changes(&base_lines, &right_lines);
    let mut all_changes: Vec<(bool, Change)> = left_changes
        .into_iter()
        .map(|c| (true, c))
        .chain(right_changes.into_iter().map(|c| (false, c)))
        .collect();

    let count = all_changes.len();
    let mut visited = vec![false; count];
    let mut components = Vec::new();

    for seed in 0..count {
        if visited[seed] {
            continue;
        }
        visited[seed] = true;
        let mut stack = vec![seed];
        let mut members = Vec::new();

        while let Some(index) = stack.pop() {
            members.push(all_changes[index].clone());
            for other in 0..count {
                if !visited[other]
                    && interacts(&all_changes[index].1, &all_changes[other].1)
                {
                    visited[other] = true;
                    stack.push(other);
                }
            }
        }

        let start = members.iter().map(|(_, c)| c.start).min().unwrap();
        let end = members.iter().map(|(_, c)| c.end).max().unwrap();
        components.push(Component {
            start,
            end,
            changes: members,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.end != a.start).cmp(&(b.end != b.start)))
            .then(a.end.cmp(&b.end))
    });

    let mut output_lines = Vec::<String>::new();
    let mut cursor = 0usize;
    let mut conflicts = Vec::new();

    for component in components {
        if cursor < component.start {
            output_lines.extend(base_lines[cursor..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let mut lc = Vec::new();
        let mut rc = Vec::new();
        for (is_left, change) in &component.changes {
            if *is_left {
                lc.push(change.clone());
            } else {
                rc.push(change.clone());
            }
        }
        lc.sort_by_key(|c| (c.start, c.end));
        rc.sort_by_key(|c| (c.start, c.end));

        let left_projection = projection(&base_lines, component.start, component.end, &lc);
        let right_projection = projection(&base_lines, component.start, component.end, &rc);

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines.len();
                    output_lines.push(format!("<<<<<<< LEFT {ordinal}"));
                    output_lines.extend(left_projection.iter().cloned());
                    let region_start = if component.end > component.start {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output_lines.push(format!(
                        "||||||| BASE {region_start},{}",
                        component.end - component.start
                    ));
                    output_lines.extend(base_projection.iter().cloned());
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection.iter().cloned());
                    output_lines.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: output_lines.len() - output_start,
                    });
                }
            }
        }
        cursor = component.end;
    }

    if cursor < base_lines.len() {
        output_lines.extend(base_lines[cursor..].iter().cloned());
    }

    let mut text = String::new();
    for line in &output_lines {
        append_line(&mut text, line);
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
