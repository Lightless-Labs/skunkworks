#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
}

fn split_lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn canonical_changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut base_pos = 0;
    let mut changes = Vec::new();
    let mut active: Option<Change> = None;

    let flush = |active: &mut Option<Change>, changes: &mut Vec<Change>| {
        if let Some(change) = active.take() {
            changes.push(change);
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut active, &mut changes);
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        let delete = if i == n {
            false
        } else if j == m {
            true
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins {
                true
            } else if ins > del {
                false
            } else {
                (i + j) % 2 == 0
            }
        };

        let change = active.get_or_insert_with(|| Change {
            start: base_pos,
            end: base_pos,
            replacement: Vec::new(),
            side,
        });

        if delete {
            i += 1;
            base_pos += 1;
            change.end = base_pos;
        } else {
            change.replacement.push(new[j].clone());
            j += 1;
        }
    }
    flush(&mut active, &mut changes);
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[Change],
    side: usize,
) -> Vec<String> {
    let mut selected: Vec<&Change> = changes
        .iter()
        .filter(|c| c.side == side)
        .collect();
    selected.sort_by_key(|c| (c.start, c.end));

    let mut out = Vec::new();
    let mut cursor = start;
    for change in selected {
        if change.start < start || change.end > end {
            continue;
        }
        if change.start > cursor {
            out.extend(base[cursor..change.start].iter().cloned());
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        out.extend(base[cursor..end].iter().cloned());
    }
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = canonical_changes(&base_lines, &left_lines, 0);
    changes.extend(canonical_changes(&base_lines, &right_lines, 1));

    let count = changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = root(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..count {
        for j in (i + 1)..count {
            if interacts(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<Vec<usize>> = Vec::new();
    for i in 0..count {
        let r = root(&mut parent, i);
        if let Some(group) = groups
            .iter_mut()
            .find(|group| root(&mut parent, group[0]) == r)
        {
            group.push(i);
        } else {
            groups.push(vec![i]);
        }
    }

    groups.sort_by_key(|group| {
        let start = group.iter().map(|&i| changes[i].start).min().unwrap();
        let end = group.iter().map(|&i| changes[i].end).max().unwrap();
        (start, usize::from(end > start))
    });

    let mut output_lines = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0;

    for group in groups {
        let start = group.iter().map(|&i| changes[i].start).min().unwrap();
        let end = group.iter().map(|&i| changes[i].end).max().unwrap();

        if start > base_cursor {
            output_lines.extend(base_lines[base_cursor..start].iter().cloned());
        }

        let component: Vec<Change> = group.iter().map(|&i| changes[i].clone()).collect();
        let base_projection = base_lines[start..end].to_vec();
        let left_projection = projection(&base_lines, start, end, &component, 0);
        let right_projection = projection(&base_lines, start, end, &component, 1);

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines.len();
                    let base_label_start = if end > start { start + 1 } else { start };
                    output_lines.push(format!("<<<<<<< LEFT {ordinal}"));
                    output_lines.extend(left_projection.iter().cloned());
                    output_lines.push(format!(
                        "||||||| BASE {base_label_start},{}",
                        end - start
                    ));
                    output_lines.extend(base_projection.iter().cloned());
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection.iter().cloned());
                    output_lines.push(format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len = output_lines.len() - output_start;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: start,
                        base_len: end - start,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        if end > base_cursor {
            base_cursor = end;
        }
    }

    if base_cursor < base_lines.len() {
        output_lines.extend(base_lines[base_cursor..].iter().cloned());
    }

    let mut text = String::new();
    for line in output_lines {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
