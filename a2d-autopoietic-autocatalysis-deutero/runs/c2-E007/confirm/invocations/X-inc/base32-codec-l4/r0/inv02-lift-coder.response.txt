#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut bytes = Vec::with_capacity(self.pending.len() + input.len());
        bytes.extend_from_slice(&self.pending);
        bytes.extend_from_slice(input);

        let complete_len = (bytes.len() / 5) * 5;
        let mut output = String::with_capacity((complete_len / 5) * 8);
        for chunk in bytes[..complete_len].chunks_exact(5) {
            encode_bytes(chunk, self.config.alphabet, &mut output);
        }

        self.pending.clear();
        self.pending.extend_from_slice(&bytes[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let symbols = encode_partial(&self.pending, self.config.alphabet, &mut output);
            if self.config.padding == Padding::Required {
                for _ in symbols..8 {
                    output.push('=');
                }
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(bytes: &[u8], alphabet: Alphabet, output: &mut String) {
    let chars = alphabet_bytes(alphabet);
    let mut acc = 0u64;
    for &byte in bytes {
        acc = (acc << 8) | u64::from(byte);
    }
    for shift in (0..40).rev().step_by(5) {
        output.push(chars[((acc >> shift) & 31) as usize] as char);
    }
}

fn encode_partial(bytes: &[u8], alphabet: Alphabet, output: &mut String) -> usize {
    let chars = alphabet_bytes(alphabet);
    let bit_len = bytes.len() * 8;
    let symbol_count = (bit_len + 4) / 5;
    let mut acc = 0u64;
    for &byte in bytes {
        acc = (acc << 8) | u64::from(byte);
    }
    let padded = acc << (symbol_count * 5 - bit_len);
    for i in 0..symbol_count {
        let shift = (symbol_count - 1 - i) * 5;
        output.push(chars[((padded >> shift) & 31) as usize] as char);
    }
    symbol_count
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    tail_value: u8,
    last_data_offset: Option<usize>,
    first_pad_offset: Option<usize>,
    pad_count: usize,
    expected_pads: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            tail_value: 0,
            last_data_offset: None,
            first_pad_offset: None,
            pad_count: 0,
            expected_pads: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let is_pad = byte == b'=';
            let recognized = if is_pad {
                None
            } else {
                recognize_symbol(byte, next.config.alphabet)
            };

            if !is_pad && recognized.is_none() {
                return Err(StreamError::InvalidByte { offset, byte });
            }

            let value = if let Some(value) = recognized {
                let upper = byte.to_ascii_uppercase();
                if next.config.canonical && byte != upper {
                    return Err(StreamError::NonCanonical {
                        offset,
                        expected: upper,
                    });
                }
                Some(value)
            } else {
                None
            };

            if is_pad {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }
                if next.first_pad_offset.is_none() {
                    let residue = next.data_count % 8;
                    let Some(expected) = expected_padding(residue) else {
                        return Err(StreamError::Padding { offset });
                    };
                    if expected == 0 {
                        return Err(StreamError::Padding { offset });
                    }
                    next.first_pad_offset = Some(offset);
                    next.expected_pads = expected;
                }
                if next.pad_count >= next.expected_pads {
                    return Err(StreamError::Padding { offset });
                }
                next.pad_count += 1;
            } else {
                if next.first_pad_offset.is_some() {
                    return Err(StreamError::Padding { offset });
                }
                let value = value.unwrap();
                next.data_count = next.data_count.checked_add(1).unwrap_or(usize::MAX);
                next.tail_value = value;
                next.last_data_offset = Some(offset);

                next.bit_buffer = (next.bit_buffer << 5) | value;
                next.bit_count += 5;
                while next.bit_count >= 8 {
                    next.bit_count -= 8;
                    output.push((next.bit_buffer >> next.bit_count) & 0xff);
                    if next.bit_count == 0 {
                        next.bit_buffer = 0;
                    } else {
                        next.bit_buffer &= (1u16
                            .checked_shl(u32::from(next.bit_count))
                            .unwrap_or(0) as u8)
                            - 1;
                    }
                }
            }

            next.position = next.position.checked_add(1).unwrap_or(usize::MAX);
        }

        *self = next.into_decoder();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let Some(expected) = expected_padding(residue) else {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        };

        if self.config.padding == Padding::Required && self.pad_count != expected {
            return Err(StreamError::Padding {
                offset: self.first_pad_offset.unwrap_or(self.position),
            });
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => unreachable!(),
        };
        if unused_bits != 0
            && (self.tail_value & ((1u8 << unused_bits) - 1)) != 0
        {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

impl Decoder {
    fn clone_state(&self) -> DecoderState {
        DecoderState {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            tail_value: self.tail_value,
            last_data_offset: self.last_data_offset,
            first_pad_offset: self.first_pad_offset,
            pad_count: self.pad_count,
            expected_pads: self.expected_pads,
            closed: self.closed,
        }
    }
}

struct DecoderState {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    tail_value: u8,
    last_data_offset: Option<usize>,
    first_pad_offset: Option<usize>,
    pad_count: usize,
    expected_pads: usize,
    closed: bool,
}

impl DecoderState {
    fn into_decoder(self) -> Decoder {
        Decoder {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            tail_value: self.tail_value,
            last_data_offset: self.last_data_offset,
            first_pad_offset: self.first_pad_offset,
            pad_count: self.pad_count,
            expected_pads: self.expected_pads,
            closed: self.closed,
        }
    }
}

fn expected_padding(residue: usize) -> Option<usize> {
    match residue {
        0 => Some(0),
        2 => Some(6),
        4 => Some(4),
        5 => Some(3),
        7 => Some(1),
        _ => None,
    }
}

fn recognize_symbol(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let upper = byte.to_ascii_uppercase();
    alphabet_bytes(alphabet)
        .iter()
        .position(|&symbol| symbol == upper)
        .map(|value| value as u8)
}
