```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

impl Machine {
    const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::<String>::new();
        for product in &products {
            if !Self::valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    fn valid_id(id: &str) -> bool {
        (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|product| product.id == id)
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = Self::DENOMINATIONS.iter().position(|&d| d == coin);
        let Some(_) = index else {
            return Err(Error::BadCoin { coin });
        };

        let resulting_credit = u32::from(self.credit) + u32::from(coin);
        if resulting_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit = resulting_credit as u16;
        self.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.product_index(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        let id = id.to_owned();
        self.selected = Some(id.clone());
        self.log(EventKind::Selected { id });
        Ok(())
    }

    fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
        let mut remaining = amount;
        let mut counts = [0u16; 4];

        for index in (0..4).rev() {
            let denomination = Self::DENOMINATIONS[index];
            let taken = tubes[index].min(remaining / denomination);
            counts[index] = taken;
            remaining -= taken * denomination;
        }

        (remaining == 0).then_some(counts)
    }

    fn better_change(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
        let candidate_total: u32 = candidate.iter().map(|&count| u32::from(count)).sum();
        let current_total: u32 = current.iter().map(|&count| u32::from(count)).sum();

        if candidate_total != current_total {
            return candidate_total < current_total;
        }

        for index in (0..4).rev() {
            if candidate[index] != current[index] {
                return candidate[index] > current[index];
            }
        }
        false
    }

    fn find_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
        if let Some(greedy) = Self::greedy_change(amount, tubes) {
            return Some(greedy);
        }

        let target = usize::from(amount);
        let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
        best[0] = Some([0; 4]);

        for value in 1..=target {
            for index in 0..4 {
                let denomination = usize::from(Self::DENOMINATIONS[index]);
                if value < denomination || tubes[index] == 0 {
                    continue;
                }
                let Some(previous) = best[value - denomination] else {
                    continue;
                };
                if previous[index] >= tubes[index] {
                    continue;
                }

                let mut candidate = previous;
                candidate[index] += 1;
                if best[value]
                    .as_ref()
                    .is_none_or(|current| Self::better_change(&candidate, current))
                {
                    best[value] = Some(candidate);
                }
            }
        }

        best[target]
    }

    fn change_vector(counts: &[u16; 4]) -> Vec<u16> {
        let mut coins = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..counts[index] {
                coins.push(Self::DENOMINATIONS[index]);
            }
        }
        coins
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };

        let index = self
            .product_index(&selected_id)
            .expect("selected products always belong to the catalog");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut {
                id: selected_id,
            });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change_counts = Self::find_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for tube_index in 0..4 {
            proposed_tubes[tube_index] -= change_counts[tube_index];
        }

        for &coin in &self.escrow {
            let tube_index = Self::DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            let proposed = u32::from(proposed_tubes[tube_index]) + 1;
            if proposed > 100 {
                return Err(Error::TubeLimit { coin });
            }
            proposed_tubes[tube_index] = proposed as u16;
        }

        let change = Self::change_vector(&change_counts);
        self.tubes = proposed_tubes;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: selected_id.clone(),
            price,
            change: change.clone(),
        };
        self.log(EventKind::Vended {
            id: selected_id,
            price,
            change,
        });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|left, right| left.id.cmp(&right.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let resulting_stock = u32::from(self.products[index].stock) + u32::from(count);
        if resulting_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = resulting_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: resulting_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}
```
