pub enum EventKind {

    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },

    pub kind: EventKind,

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::<String>::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.record(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(_) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        self.record(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let index = self.product_index(&id).expect("selected catalog product");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change_counts = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for i in 0..4 {
            proposed_tubes[i] -= change_counts[i];
        }
        for &coin in &self.escrow {
            let i = denomination_index(coin).expect("escrow contains accepted coins");
            let proposed = proposed_tubes[i] + 1;
            if proposed > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
            proposed_tubes[i] = proposed;
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change_counts[i] {
                change.push(DENOMINATIONS[i]);
            }
        }

        self.products[index].stock -= 1;
        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        self.record(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.record(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.cmp(&b.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.record(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });

            return Err(Error::ZeroRestock);

            return Err(Error::StockLimit { id: id.to_owned() });

        self.record(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {

    }

    fn product_index(&self, id: &str) -> Option<usize> {

            kind,
        });
    }

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&denomination| denomination == coin)
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount as u32;

    for i in (0..4).rev() {
        let denomination = DENOMINATIONS[i] as u32;
        let count = (remaining / denomination).min(tubes[i] as u32) as u16;
        greedy[i] = count;
        remaining -= count as u32 * denomination;
    }

    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut current = [0u16; 4];
    search_change(3, amount as i32, tubes, &mut current, &mut best);
    best
}

fn search_change(
    index: isize,
    remaining: i32,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if remaining == 0 {
        let replace = match best {
            None => true,
            Some(existing) => {
                let current_total: u32 = current.iter().map(|&x| x as u32).sum();
                let existing_total: u32 = existing.iter().map(|&x| x as u32).sum();
                current_total < existing_total
                    || (current_total == existing_total
                        && lex_greater_descending(current, existing))
            }
        };
        if replace {
            *best = Some(*current);
        }
        return;
    }
    if index < 0 || remaining < 0 {
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i] as i32;
    let max_count = (remaining / denomination).min(tubes[i] as i32);

    for count in (0..=max_count).rev() {
        current[i] = count as u16;
        search_change(
            index - 1,
            remaining - count * denomination,
            tubes,
            current,
            best,
        );
    }
    current[i] = 0;
}

fn lex_greater_descending(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if left[i] != right[i] {
            return left[i] > right[i];
        }
    }
    false
}

        self.credit = resulting_credit as u16;

        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.product_index(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        let id = id.to_owned();
        self.selected = Some(id.clone());
        self.log(EventKind::Selected { id });
        Ok(())
    }

    fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
        let mut remaining = amount;
        let mut counts = [0u16; 4];

        for index in (0..4).rev() {
            let denomination = Self::DENOMINATIONS[index];
            let taken = tubes[index].min(remaining / denomination);
            counts[index] = taken;
            remaining -= taken * denomination;
        }

        (remaining == 0).then_some(counts)
    }

    fn better_change(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
        let candidate_total: u32 = candidate.iter().map(|&count| u32::from(count)).sum();
        let current_total: u32 = current.iter().map(|&count| u32::from(count)).sum();

        if candidate_total != current_total {
            return candidate_total < current_total;
        }

        for index in (0..4).rev() {
            if candidate[index] != current[index] {
                return candidate[index] > current[index];
            }
        }
        false
    }

    fn find_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
        if let Some(greedy) = Self::greedy_change(amount, tubes) {
            return Some(greedy);
        }

        let target = usize::from(amount);
        let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
        best[0] = Some([0; 4]);

        for value in 1..=target {
            for index in 0..4 {
                let denomination = usize::from(Self::DENOMINATIONS[index]);
                if value < denomination || tubes[index] == 0 {
                    continue;
                }
                let Some(previous) = best[value - denomination] else {
                    continue;
                };
                if previous[index] >= tubes[index] {
                    continue;
                }

                let mut candidate = previous;
                candidate[index] += 1;
                if best[value]
                    .as_ref()
                    .is_none_or(|current| Self::better_change(&candidate, current))
                {
                    best[value] = Some(candidate);
                }
            }
        }

        best[target]
    }

    fn change_vector(counts: &[u16; 4]) -> Vec<u16> {
        let mut coins = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..counts[index] {
                coins.push(Self::DENOMINATIONS[index]);
            }
        }
        coins
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };

        let index = self
            .product_index(&selected_id)
            .expect("selected products always belong to the catalog");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut {
                id: selected_id,
            });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change_counts = Self::find_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for tube_index in 0..4 {
            proposed_tubes[tube_index] -= change_counts[tube_index];
        }

        for &coin in &self.escrow {
            let tube_index = Self::DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            let proposed = u32::from(proposed_tubes[tube_index]) + 1;
            if proposed > 100 {
                return Err(Error::TubeLimit { coin });
            }
            proposed_tubes[tube_index] = proposed as u16;
        }

        let change = Self::change_vector(&change_counts);
        self.tubes = proposed_tubes;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: selected_id.clone(),
            price,
            change: change.clone(),
        };
        self.log(EventKind::Vended {
            id: selected_id,
            price,
            change,
        });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|left, right| left.id.cmp(&right.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let resulting_stock = u32::from(self.products[index].stock) + u32::from(count);
        if resulting_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = resulting_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: resulting_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}

fn make_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0u16; 4];
    let mut remaining = amount;
    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index];
        let count = tubes[index].min(remaining / denomination);
        greedy_counts[index] = count;
        remaining -= count * denomination;
    }
    if remaining == 0 {
        return Some(counts_to_descending_coins(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    search_change(
        3,
        amount,
        tubes,
        [0; 4],
        &mut best,
    );

    best.map(counts_to_descending_coins)
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: [u16; 4],
    mut counts: [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if remaining == 0 {
        let replace = match best {
            None => true,
            Some(current) => {
                let total: u32 = counts.iter().map(|&n| u32::from(n)).sum();
                let best_total: u32 = current.iter().map(|&n| u32::from(n)).sum();
                total < best_total || (total == best_total && (3..=0).rev().any(|_| false))
                    || (total == best_total && lex_greater_desc(&counts, current))
            }
        };
        if replace {
            *best = Some(counts);
        }
        return;
    }
    if index < 0 {
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i];
    let maximum = tubes[i].min(remaining / denomination);
    for count in (0..=maximum).rev() {
        counts[i] = count;
        search_change(
            index - 1,
            remaining - count * denomination,
            tubes,
            counts,
            best,
        );
    }
}

fn lex_greater_desc(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for index in (0..4).rev() {
        if left[index] != right[index] {
            return left[index] > right[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        coins.extend(std::iter::repeat_n(DENOMINATIONS[index], counts[index] as usize));
    }
    coins
}

            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.push_event(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = match self.products.iter().position(|product| product.id == id) {
            Some(index) => index,
            None => return Err(Error::UnknownProduct { id: id.to_owned() }),
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let new_stock = self.products[index].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        let stock = new_stock as u16;
        self.products[index].stock = stock;
        self.push_event(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn push_event(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&denomination| denomination == coin)
}

fn expand_change(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remainder = amount as u32;

    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index] as u32;
        let take = (tubes[index] as u32).min(remainder / denomination);
        greedy[index] = take as u16;
        remainder -= take * denomination;
    }

    if remainder == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;

    for c10 in 0..=tubes[3].min(amount / 10) {
        let value10 = c10 as u32 * 10;
        if value10 > amount as u32 {
            break;
        }
        for c4 in 0..=tubes[2].min(((amount as u32 - value10) / 4) as u16) {
            let value4 = value10 + c4 as u32 * 4;
            if value4 > amount as u32 {
                break;
            }
            for c3 in 0..=tubes[1].min(((amount as u32 - value4) / 3) as u16) {
                let value3 = value4 + c3 as u32 * 3;
                if value3 > amount as u32 {
                    break;
                }
                let remaining = amount as u32 - value3;
                if remaining % 1 != 0 || remaining > tubes[0] as u32 {
                    continue;
                }
                let c1 = remaining as u16;
                let candidate = [c1, c3, c4, c10];
                let count = c1 as u32 + c3 as u32 + c4 as u32 + c10 as u32;
                if count < best_count
                    || (count == best_count
                        && best
                            .as_ref()
                            .map_or(true, |current| descending_lex_greater(candidate, *current)))
                {
                    best_count = count;
                    best = Some(candidate);
                }
            }
        }
    }

    best
}

fn descending_lex_greater(left: [u16; 4], right: [u16; 4]) -> bool {
    for index in (0..4).rev() {
        if left[index] != right[index] {
            return left[index] > right[index];
        }
    }
    false
}
