#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize, usize)>, // cleaned insertion index, original first byte
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::new();
    let mut positions = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_protected = false;
    let mut quote_closed = false;

    while i < input.len() {
        if quote_protected && !quote_closed {
            let b = input[i];
            out.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    out.push(input[i + 1]);
                    positions.push(i + 1);
                    i += 2;
                    continue;
                }
                i += 1;
                continue;
            }
            if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if !quote_protected && out.is_empty() && input[i] == b'"' {
            quote_protected = true;
            quote_closed = false;
            out.push(input[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        if input[i] == b'\r' || input[i] == b'\n' {
            let end = fold_end(input, i).ok_or(EmailError::BadFold { at: i })?;
            let start = i;
            i = end;
            if runs.last().is_some_and(|r| r.0 + 0 == out.len() && r.1 + r.2 == start) {
                let last = runs.last_mut().unwrap();
                last.2 = end - last.1;
            } else {
                runs.push((out.len(), start, end - start));
            }
            continue;
        }

        if is_hws(input[i]) {
            let start = i;
            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }
            if runs.last().is_some_and(|r| r.0 == out.len() && r.1 + r.2 == start) {
                let last = runs.last_mut().unwrap();
                last.2 = i - last.1;
            } else {
                runs.push((out.len(), start, i - start));
            }
            continue;
        }

        if input[i] == b'(' {
            let outer = i;
            let mut depth = 1usize;
            i += 1;
            while i < input.len() {
                let b = input[i];
                if b == b'\\' {
                    let esc = i;
                    if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                        return Err(EmailError::CommentEscape { at: esc });
                    }
                    i += 2;
                    continue;
                }
                if b == b'\r' || b == b'\n' {
                    i = fold_end(input, i).ok_or(EmailError::BadFold { at: i })?;
                    continue;
                }
                if b == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                    continue;
                }
                if b == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 {
                        break;
                    }
                    continue;
                }
                if !(is_printable(b) || b == b'\t') {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }
            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            if runs.last().is_some_and(|r| r.0 == out.len() && r.1 + r.2 == outer) {
                let last = runs.last_mut().unwrap();
                last.2 = i - last.1;
            } else {
                runs.push((out.len(), outer, i - outer));
            }
            continue;
        }

        if input[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        out.push(input[i]);
        positions.push(i);
        i += 1;
    }

    Ok(Cleaned { bytes: out, positions, runs })
}

fn pos(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

fn parse_inner(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let clean = cleanup(raw)?;
    let n = clean.bytes.len();

    for &(boundary, original, _) in &clean.runs {
        let legal = boundary == 0
            || boundary == n
            || clean.bytes.get(boundary).copied() == Some(b'@')
            || (boundary > 0 && clean.bytes[boundary - 1] == b'@');
        if !legal {
            return Err(EmailError::CfwsBoundary { at: original });
        }
    }

    if n == 0 {
        return Err(EmailError::EmptyInput);
    }

    let bytes = &clean.bytes;
    let mut protected_end = 0usize;
    if bytes[0] == b'"' {
        let mut i = 1usize;
        let mut close = None;
        while i < n {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        let Some(end) = close else {
            return Err(EmailError::UnclosedQuote { at: pos(&clean, 0, raw.len()) });
        };
        protected_end = end + 1;
    }

    let mut separators = Vec::new();
    for i in 0..n {
        if bytes[i] == b'@' && !(i < protected_end) {
            separators.push(i);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: pos(&clean, separators[1], raw.len()) });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: pos(&clean, sep, raw.len()) });
    }

    let quoted = bytes[0] == b'"';
    let mut local = Vec::new();

    if !quoted {
        let len = sep;
        if len > 64 {
            return Err(EmailError::LocalTooLong { at: pos(&clean, 64, raw.len()) });
        }
        for i in 0..len {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == len || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: pos(&clean, i, raw.len()) });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: pos(&clean, i, raw.len()) });
            }
            local.push(b);
        }
    } else {
        let quote_end = protected_end - 1;
        if quote_end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&clean, quote_end + 1, raw.len()),
            });
        }
        let mut origins = Vec::new();
        let mut i = 1usize;
        while i < quote_end {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= quote_end
                    || !matches!(bytes[i + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape { at: pos(&clean, i, raw.len()) });
                }
                local.push(bytes[i + 1]);
                origins.push(pos(&clean, i, raw.len()));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: pos(&clean, i, raw.len()) });
                }
                local.push(b);
                origins.push(pos(&clean, i, raw.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(&clean, quote_end, raw.len()) });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    }

    let domain_start = sep + 1;
    if domain_start == n {
        return Err(EmailError::EmptyDomain { at: raw.len() });
    }
    let mut domain_end = n;
    let mut root_dot = None;
    if bytes[domain_end - 1] == b'.' {
        root_dot = Some(domain_end - 1);
        domain_end -= 1;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: pos(&clean, domain_start, raw.len()) });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong { at: pos(&clean, domain_start + 253, raw.len()) });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || bytes[i] == b'.' {
            if start == i {
                let at = if i == domain_end {
                    root_dot.map(|x| pos(&clean, x, raw.len())).unwrap_or_else(|| pos(&clean, i, raw.len()))
                } else {
                    pos(&clean, i, raw.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = i - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: pos(&clean, start + 63, raw.len()) });
            }
            if bytes[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&clean, start, raw.len()) });
            }
            if bytes[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&clean, i - 1, raw.len()) });
            }
            for j in start..i {
                if !bytes[j].is_ascii_alphanumeric() && bytes[j] != b'-' {
                    return Err(EmailError::DomainChar { at: pos(&clean, j, raw.len()) });
                }
            }
            labels.push((start, i));
            start = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: pos(&clean, domain_start, raw.len()) });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&clean, last_start, raw.len()) });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: String::from_utf8(local).unwrap(),
        domain,
        quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_inner(input)
}
