#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut clean = Vec::<u8>::new();
    let mut positions = Vec::<usize>::new();
    let mut runs = Vec::<Run>::new();
    let mut i = 0usize;
    let mut first_retained = true;
    let mut protected = false;

    while i < raw.len() {
        if protected {
            let b = raw[i];
            clean.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' && i < raw.len() {
                clean.push(raw[i]);
                positions.push(i);
                i += 1;
            } else if b == b'"' {
                protected = false;
            }
            continue;
        }

        let b = raw[i];
        let starts_cfws = is_hws(b) || b == b'(' || b == b'\r' || b == b'\n';
        if starts_cfws {
            let run_start = i;
            loop {
                if i >= raw.len() {
                    break;
                }
                let c = raw[i];
                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && i + 2 < raw.len()
                        && raw[i + 1] == b'\n'
                        && is_hws(raw[i + 2])
                    {
                        i += 2;
                        while i < raw.len() && is_hws(raw[i]) {
                            i += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    let outer = i;
                    let mut depth = 0usize;
                    while i < raw.len() {
                        let x = raw[i];
                        if x == b'\\' {
                            if i + 1 >= raw.len() || !printable(raw[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                            continue;
                        }
                        if x == b'\r' || x == b'\n' {
                            if x == b'\r'
                                && i + 2 < raw.len()
                                && raw[i + 1] == b'\n'
                                && is_hws(raw[i + 2])
                            {
                                i += 2;
                                while i < raw.len() && is_hws(raw[i]) {
                                    i += 1;
                                }
                                continue;
                            }
                            return Err(EmailError::BadFold { at: i });
                        }
                        if x == b'(' {
                            depth += 1;
                            if depth > 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            i += 1;
                            continue;
                        }
                        if x == b')' {
                            depth -= 1;
                            i += 1;
                            if depth == 0 {
                                break;
                            }
                            continue;
                        }
                        if !printable(x) && x != b'\t' {
                            return Err(EmailError::CommentChar { at: i });
                        }
                        i += 1;
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    continue;
                }
                break;
            }
            runs.push(Run { start: run_start, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        positions.push(i);
        if first_retained {
            first_retained = false;
            if b == b'"' {
                protected = true;
            }
        }
        i += 1;
    }

    for run in &runs {
        let left_at = run.start > 0 && raw[run.start - 1] == b'@';
        let right_at = run.end < raw.len() && raw[run.end] == b'@';
        let at_boundary = run.start == 0 || run.end == raw.len();
        if !at_boundary && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1usize;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: positions[0] });
        }
    }

    let mut ats = Vec::new();
    let mut j = 0usize;
    while j < clean.len() {
        if let Some(end) = quote_end {
            if j <= end {
                j = end + 1;
                continue;
            }
        }
        if clean[j] == b'@' {
            ats.push(j);
        }
        j += 1;
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[ats[1]],
        });
    }

    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[at_idx],
        });
    }

    let mut local = String::new();
    let quoted = quote_end.is_some();
    if let Some(end) = quote_end {
        if end + 1 != at_idx {
            let pos = if end + 1 < positions.len() {
                positions[end + 1]
            } else {
                input.len()
            };
            return Err(EmailError::TrailingLocal { at: pos });
        }
        let mut decoded = Vec::<(u8, usize)>::new();
        let mut k = 1usize;
        while k < end {
            let b = clean[k];
            if b == b'\\' {
                if k + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: positions[k],
                    });
                }
                let escaped = clean[k + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: positions[k],
                    });
                }
                decoded.push((escaped, positions[k]));
                k += 2;
            } else {
                if !printable(b) {
                    return Err(EmailError::QuotedChar { at: positions[k] });
                }
                decoded.push((b, positions[k]));
                k += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: positions[end] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded[64].1,
            });
        }
        for (b, _) in decoded {
            local.push(b as char);
        }
    } else {
        if at_idx > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for k in 0..at_idx {
            let b = clean[k];
            if b == b'.' {
                if k == 0 || k + 1 == at_idx || clean[k - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: positions[k] });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: positions[k] });
            }
            local.push(b as char);
        }
    }

    let domain_start_idx = at_idx + 1;
    let domain_start_pos = if domain_start_idx < positions.len() {
        positions[domain_start_idx]
    } else {
        input.len()
    };
    if domain_start_idx >= clean.len() {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    let mut domain_end = clean.len();
    let mut removed_dot = None;
    if clean[domain_end - 1] == b'.' {
        removed_dot = Some(positions[domain_end - 1]);
        domain_end -= 1;
    }
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }
    if domain_end - domain_start_idx > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start_idx + 253],
        });
    }

    let mut labels = Vec::<(usize, usize)>::new();
    let mut start = domain_start_idx;
    for k in domain_start_idx..domain_end {
        if clean[k] == b'.' {
            labels.push((start, k));
            start = k + 1;
        }
    }
    labels.push((start, domain_end));

    for (li, &(ls, le)) in labels.iter().enumerate() {
        if ls == le {
            let p = if li + 1 == labels.len() {
                removed_dot.unwrap_or(input.len())
            } else if ls < positions.len() {
                positions[ls]
            } else {
                input.len()
            };
            return Err(EmailError::EmptyLabel { at: p });
        }
        if le - ls > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[ls + 63],
            });
        }
        if clean[ls] == b'-' {
            return Err(EmailError::LabelHyphen { at: positions[ls] });
        }
        if clean[le - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[le - 1],
            });
        }
        for k in ls..le {
            let b = clean[k];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: positions[k] });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start_idx..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
