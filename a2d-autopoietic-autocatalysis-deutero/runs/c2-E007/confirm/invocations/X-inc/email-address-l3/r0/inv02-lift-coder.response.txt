#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::with_capacity(bytes.len());
    let mut positions = Vec::with_capacity(bytes.len());
    let mut runs = Vec::<Run>::new();
    let mut i = 0;
    let mut protected = false;
    let mut quote_closed = false;

    while i < bytes.len() {
        if protected && !quote_closed {
            let b = bytes[i];
            clean.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    quote_closed = true;
                }
                i += 1;
            }
            continue;
        }

        if protected && quote_closed {
            protected = false;
        }

        let b = bytes[i];
        if b == b'"' && clean.is_empty() {
            protected = true;
            quote_closed = false;
            clean.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' {
            let run_start = i;
            loop {
                if i >= bytes.len() {
                    break;
                }
                match bytes[i] {
                    b' ' | b'\t' => {
                        i += 1;
                    }
                    b'\r' | b'\n' => {
                        if bytes[i] != b'\r'
                            || i + 2 >= bytes.len()
                            || bytes[i + 1] != b'\n'
                            || !(bytes[i + 2] == b' ' || bytes[i + 2] == b'\t')
                        {
                            return Err(EmailError::BadFold { at: i });
                        }
                        i += 2;
                        while i < bytes.len() && (bytes[i] == b' ' || bytes[i] == b'\t') {
                            i += 1;
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut stack = Vec::with_capacity(3);
                        stack.push(i);
                        i += 1;
                        while !stack.is_empty() {
                            if i >= bytes.len() {
                                return Err(EmailError::UnclosedComment { at: outer });
                            }
                            match bytes[i] {
                                b'\\' => {
                                    let esc = i;
                                    if i + 1 >= bytes.len() || !printable(bytes[i + 1]) {
                                        return Err(EmailError::CommentEscape { at: esc });
                                    }
                                    i += 2;
                                }
                                b'(' => {
                                    if stack.len() == 3 {
                                        return Err(EmailError::CommentDepth { at: i });
                                    }
                                    stack.push(i);
                                    i += 1;
                                }
                                b')' => {
                                    stack.pop();
                                    i += 1;
                                }
                                b'\r' | b'\n' => {
                                    if bytes[i] != b'\r'
                                        || i + 2 >= bytes.len()
                                        || bytes[i + 1] != b'\n'
                                        || !(bytes[i + 2] == b' ' || bytes[i + 2] == b'\t')
                                    {
                                        return Err(EmailError::BadFold { at: i });
                                    }
                                    i += 2;
                                    while i < bytes.len()
                                        && (bytes[i] == b' ' || bytes[i] == b'\t')
                                    {
                                        i += 1;
                                    }
                                }
                                c if printable(c) || c == b'\t' => i += 1,
                                _ => return Err(EmailError::CommentChar { at: i }),
                            }
                        }
                    }
                    _ => break,
                }
            }
            runs.push(Run { start: run_start, end: i });
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if b != b'\r'
                || i + 2 >= bytes.len()
                || bytes[i + 1] != b'\n'
                || !(bytes[i + 2] == b' ' || bytes[i + 2] == b'\t')
            {
                return Err(EmailError::BadFold { at: i });
            }
            let start = i;
            i += 2;
            while i < bytes.len() && (bytes[i] == b' ' || bytes[i] == b'\t') {
                i += 1;
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        positions.push(i);
        i += 1;
    }

    let at_positions: Vec<usize> = clean
        .iter()
        .enumerate()
        .filter_map(|(idx, &b)| if b == b'@' { Some(idx) } else { None })
        .collect();

    for run in &runs {
        let before = positions.iter().position(|&p| p >= run.start);
        let after = positions.iter().position(|&p| p >= run.end);
        let legal = before.is_none()
            || after.is_none()
            || before.is_some_and(|idx| clean[idx] == b'@')
            || after.is_some_and(|idx| idx > 0 && clean[idx - 1] == b'@');
        if !legal {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean.first() == Some(&b'"') {
        let mut j = 1;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: positions[0],
            });
        }
    }

    let mut separators = Vec::new();
    for (idx, &b) in clean.iter().enumerate() {
        if b == b'@' && !quote_end.is_some_and(|end| idx <= end) {
            separators.push(idx);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[sep],
        });
    }

    let mut local = String::new();
    let quoted = clean[0] == b'"';
    if quoted {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: positions[end + 1],
            });
        }
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            if clean[j] == b'\\' {
                if j + 1 >= end
                    || !matches!(clean[j + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: positions[j],
                    });
                }
                local.push(clean[j + 1] as char);
                origins.push(positions[j]);
                j += 2;
            } else {
                if !printable(clean[j]) {
                    return Err(EmailError::QuotedChar {
                        at: positions[j],
                    });
                }
                local.push(clean[j] as char);
                origins.push(positions[j]);
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[end],
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for idx in 0..sep {
            let b = clean[idx];
            if b == b'.' {
                if idx == 0 || idx + 1 == sep || clean[idx - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: positions[idx],
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: positions[idx],
                });
            }
        }
        local = String::from_utf8(clean[..sep].to_vec()).unwrap();
    }

    let domain_start_idx = sep + 1;
    let domain_start = positions
        .get(domain_start_idx)
        .copied()
        .unwrap_or(input.len());
    if domain_start_idx == clean.len() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let mut domain_end = clean.len();
    let mut root_pos = None;
    if clean.last() == Some(&b'.') {
        domain_end -= 1;
        root_pos = Some(positions[domain_end]);
    }
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_end - domain_start_idx > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start_idx + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start_idx;
    for idx in domain_start_idx..domain_end {
        if clean[idx] == b'.' {
            if idx == start {
                return Err(EmailError::EmptyLabel {
                    at: positions[idx],
                });
            }
            labels.push((start, idx));
            start = idx + 1;
        }
    }
    if start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: root_pos.unwrap_or(input.len()),
        });
    }
    labels.push((start, domain_end));

    for &(begin, end) in &labels {
        if end - begin > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[begin + 63],
            });
        }
        if clean[begin] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[begin],
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for idx in begin..end {
            if !clean[idx].is_ascii_alphanumeric() && clean[idx] != b'-' {
                return Err(EmailError::DomainChar {
                    at: positions[idx],
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_begin, last_end) = *labels.last().unwrap();
    if clean[last_begin..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_begin],
        });
    }

    let domain = clean[domain_start_idx..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
