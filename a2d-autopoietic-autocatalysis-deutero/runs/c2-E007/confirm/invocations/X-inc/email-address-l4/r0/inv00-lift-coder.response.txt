use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_options(input, false).map(|parsed| parsed.address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let mut parsed = parse_with_options(input, policy.allow_single_label)?;
    let mut bytes = parsed.address.local.into_bytes();
    let mut origins = parsed.local_origins;

    if policy.local_case == LocalCase::Lower {
        for byte in &mut bytes {
            if byte.is_ascii_uppercase() {
                *byte = byte.to_ascii_lowercase();
            }
        }
    }

    if policy.strip_subaddress {
        if let Some(plus) = bytes.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            bytes.truncate(plus);
            origins.truncate(plus);
        }
    }

    let valid_atom = valid_dot_atom(&bytes);
    parsed.address.local = String::from_utf8(bytes).expect("ASCII local");
    if !valid_atom {
        parsed.address.quoted = true;
    } else if policy.unquote_safe {
        parsed.address.quoted = false;
    }

    Ok(parsed.address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut first_by_key: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = first_by_key.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    first_by_key.insert(key, index);
                    addresses.push(address);
                    source_indices.push(index);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

struct Parsed {
    address: Address,
    local_origins: Vec<usize>,
}

fn parse_with_options(input: &str, allow_single_label: bool) -> Result<Parsed, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some((at, _)) = input.bytes().enumerate().find(|(_, b)| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(input)?;
    check_cfws_boundaries(&cleaned)?;
    if cleaned.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let bytes = &cleaned.bytes;
    let mut protected_end = 0;
    if bytes[0] == b'"' {
        let mut i = 1;
        let mut closing = None;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                closing = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match closing {
            Some(end) => protected_end = end + 1,
            None => return Err(EmailError::UnclosedQuote { at: cleaned.origins[0] }),
        }
    }

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if i >= protected_end && bytes[i] == b'@' {
            separators.push(i);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: cleaned.origins[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: cleaned.origins[sep],
        });
    }

    let (local, quoted, local_origins) = if bytes[0] == b'"' {
        if protected_end != sep {
            return Err(EmailError::TrailingLocal {
                at: cleaned.origins[protected_end],
            });
        }
        let closing = protected_end - 1;
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < closing {
            let byte = bytes[i];
            if byte == b'\\' {
                if i + 1 >= closing {
                    return Err(EmailError::BadEscape {
                        at: cleaned.origins[i],
                    });
                }
                let escaped = bytes[i + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: cleaned.origins[i],
                    });
                }
                decoded.push(escaped);
                origins.push(cleaned.origins[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&byte) {
                    return Err(EmailError::QuotedChar {
                        at: cleaned.origins[i],
                    });
                }
                decoded.push(byte);
                origins.push(cleaned.origins[i]);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: cleaned.origins[closing],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (decoded, true, origins)
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: cleaned.origins[64],
            });
        }
        validate_atom(&bytes[..sep], &cleaned.origins[..sep])?;
        (
            bytes[..sep].to_vec(),
            false,
            cleaned.origins[..sep].to_vec(),
        )
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let mut root_dot = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(cleaned.origins[domain_end]);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: cleaned.origins.get(domain_start).copied().unwrap_or(input.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: cleaned.origins[domain_start + 253],
        });
    }

    let domain = &bytes[domain_start..domain_end];
    let domain_origins = &cleaned.origins[domain_start..domain_end];
    let mut labels = Vec::new();
    let mut start = 0;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            let label = &domain[start..i];
            let origins = &domain_origins[start..i];
            if label.is_empty() {
                let at = if start == domain.len() {
                    root_dot.unwrap_or_else(|| {
                        domain_origins
                            .last()
                            .copied()
                            .map(|p| p + 1)
                            .unwrap_or(input.len())
                    })
                } else if i < domain.len() {
                    domain_origins[i]
                } else {
                    root_dot.unwrap_or(input.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong { at: origins[63] });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: origins[0] });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: origins[label.len() - 1],
                });
            }
            for (j, &byte) in label.iter().enumerate() {
                if !(byte.is_ascii_alphanumeric() || byte == b'-') {
                    return Err(EmailError::DomainChar { at: origins[j] });
                }
            }
            labels.push((label, origins[0]));
            start = i + 1;
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: cleaned.origins[domain_start],
        });
    }
    let (last_label, last_start) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let local = String::from_utf8(local).expect("ASCII local");
    let domain = domain.iter().map(u8::to_ascii_lowercase).collect::<Vec<_>>();
    Ok(Parsed {
        address: Address {
            local,
            domain: String::from_utf8(domain).expect("ASCII domain"),
            quoted,
        },
        local_origins,
    })
}

struct Cleaned {
    bytes: Vec<u8>,
    origins: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn cleanup(input: &str) -> Result<Cleaned, EmailError> {
    let src = input.as_bytes();
    let mut bytes = Vec::with_capacity(src.len());
    let mut origins = Vec::with_capacity(src.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_open = 0usize;
    let mut quote_open = false;
    let mut started_quote = false;
    let mut current_run_start: Option<usize> = None;

    while i < src.len() {
        if comment_depth > 0 {
            let b = src[i];
            if b == b'\\' {
                if i + 1 >= src.len() || !(0x20..=0x7e).contains(&src[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                let next = fold_end(src, i).ok_or(EmailError::BadFold { at: i })?;
                i = next;
                continue;
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                if comment_depth == 0 {
                    if let Some(start) = current_run_start.take() {
                        runs.push((start, i));
                    }
                }
                continue;
            }
            if b == b'\t' || (0x20..=0x7e).contains(&b) {
                i += 1;
                continue;
            }
            return Err(EmailError::CommentChar { at: i });
        }

        if quote_open {
            let b = src[i];
            bytes.push(b);
            origins.push(i);
            if b == b'\\' {
                i += 1;
                if i < src.len() {
                    bytes.push(src[i]);
                    origins.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    quote_open = false;
                }
                i += 1;
            }
            continue;
        }

        let b = src[i];
        if b == b' ' || b == b'\t' || b == b'(' || b == b'\r' || b == b'\n' {
            current_run_start.get_or_insert(i);
            if b == b' ' || b == b'\t' {
                i += 1;
            } else if b == b'(' {
                if comment_depth == 0 {
                    outer_open = i;
                }
                comment_depth = 1;
                i += 1;
            } else {
                i = fold_end(src, i).ok_or(EmailError::BadFold { at: i })?;
            }
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if !started_quote && bytes.is_empty() && b == b'"' {
            started_quote = true;
            quote_open = true;
            bytes.push(b);
            origins.push(i);
            i += 1;
            continue;
        }

        if let Some(start) = current_run_start.take() {
            runs.push((start, i));
        }
        bytes.push(b);
        origins.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = current_run_start.take() {
        runs.push((start, src.len()));
    }

    Ok(Cleaned {
        bytes,
        origins,
        runs,
    })
}

fn fold_end(src: &[u8], at: usize) -> Option<usize> {
    if src.get(at) != Some(&b'\r') || src.get(at + 1) != Some(&b'\n') {
        return None;
    }
    let mut end = at + 2;
    if !matches!(src.get(end), Some(b' ' | b'\t')) {
        return None;
    }
    while matches!(src.get(end), Some(b' ' | b'\t')) {
        end += 1;
    }
    Some(end)
}

fn check_cfws_boundaries(cleaned: &Cleaned) -> Result<(), EmailError> {
    for &(start, _) in &cleaned.runs {
        let before = cleaned.origins.iter().position(|&p| p >= start);
        let first_after = cleaned.origins.iter().position(|&p| p > start);
        let _ = first_after;

        let left = cleaned
            .origins
            .iter()
            .rposition(|&p| p < start)
            .and_then(|i| cleaned.bytes.get(i).copied());
        let right = cleaned
            .origins
            .iter()
            .position(|&p| p > start)
            .and_then(|i| cleaned.bytes.get(i).copied());

        let at_beginning = before == Some(0);
        let at_end = !cleaned.origins.iter().any(|&p| p > start);
        if !at_beginning && !at_end && left != Some(b'@') && right != Some(b'@') {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }
    Ok(())
}

fn validate_atom(bytes: &[u8], origins: &[usize]) -> Result<(), EmailError> {
    for i in 0..bytes.len() {
        let b = bytes[i];
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: origins[i] });
            }
        } else if !(b.is_ascii_alphanumeric()
            || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
        {
            return Err(EmailError::LocalChar { at: origins[i] });
        }
    }
    Ok(())
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    for i in 0..bytes.len() {
        let b = bytes[i];
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return false;
            }
        } else if !(b.is_ascii_alphanumeric()
            || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
        {
            return false;
        }
    }
    true
}
