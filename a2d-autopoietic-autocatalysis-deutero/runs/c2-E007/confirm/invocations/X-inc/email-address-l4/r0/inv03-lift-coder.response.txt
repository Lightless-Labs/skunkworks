#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= bytes.len() || bytes[i] != b'\r' || bytes[i + 1] != b'\n' {
        return None;
    }
    let mut j = i + 2;
    if !is_hws(bytes[j]) {
        return None;
    }
    while j < bytes.len() && is_hws(bytes[j]) {
        j += 1;
    }
    Some(j)
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_prefix = false;
    let mut quote_closed = false;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        if quote_prefix && !quote_closed {
            let b = input[i];
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' && i != 0 {
                    quote_closed = true;
                }
                i += 1;
            }
            continue;
        }

        if depth > 0 {
            let b = input[i];
            if b == b'\\' {
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
            } else if b == b')' {
                depth -= 1;
            } else if !(0x20..=0x7e).contains(&b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let b = input[i];
        if quote_prefix && quote_closed {
            // Only the initial quote pair is protected; later syntax is ordinary.
            quote_prefix = false;
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            if b == b' ' || b == b'\t' {
                i += 1;
            } else if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else {
                outer_open = i;
                depth = 1;
                i += 1;
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if let Some(start) = run_start.take() {
            runs.push(Run { start, end: i });
        }
        if bytes.is_empty() && b == b'"' {
            quote_prefix = true;
            quote_closed = false;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        runs.push(Run {
            start,
            end: input.len(),
        });
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn at_pos(c: &Cleaned, idx: usize, input_len: usize) -> usize {
    c.positions.get(idx).copied().unwrap_or(input_len)
}

fn validate_local_atom(local: &[u8], positions: &[usize]) -> Result<(), EmailError> {
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong {
            at: positions[64],
        });
    }
    let allowed = b"!#$%&'*+-/=?^_`{|}~";
    for i in 0..local.len() {
        let b = local[i];
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: positions[i] });
            }
        } else if !b.is_ascii_alphanumeric() && !allowed.contains(&b) {
            return Err(EmailError::LocalChar { at: positions[i] });
        }
    }
    Ok(())
}

fn parse_cleaned(
    c: &Cleaned,
    input_len: usize,
    allow_single_label: bool,
) -> Result<(Address, usize), EmailError> {
    let b = &c.bytes;
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    for run in &c.runs {
        let left = c.positions.iter().position(|&p| p >= run.end);
        let right = c.positions.iter().rposition(|&p| p < run.start);
        let left_is_at = right.map_or(false, |idx| b[idx] == b'@');
        let right_is_at = left.map_or(false, |idx| b[idx] == b'@');
        let at_edge = left.is_none() || right.is_none();
        if !at_edge && !left_is_at && !right_is_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    let mut quote_end = None;
    if b[0] == b'"' {
        let mut i = 1;
        while i < b.len() {
            if b[i] == b'\\' {
                if i + 1 < b.len() {
                    i += 2;
                } else {
                    break;
                }
            } else if b[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: c.positions[0],
            });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i += 1;
                continue;
            }
        }
        if b[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: c.positions[ats[1]],
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: c.positions[sep],
        });
    }

    let local_bytes = &b[..sep];
    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let mut quoted = false;
    if local_bytes[0] == b'"' {
        let end = quote_end.expect("quoted prefix has a closing quote");
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: c.positions[end + 1],
            });
        }
        quoted = true;
        let mut j = 1;
        while j < end {
            let byte = local_bytes[j];
            if byte == b'\\' {
                if j + 1 >= end
                    || !matches!(local_bytes[j + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: c.positions[j],
                    });
                }
                local.push(local_bytes[j + 1]);
                local_origins.push(c.positions[j]);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&byte) {
                    return Err(EmailError::QuotedChar {
                        at: c.positions[j],
                    });
                }
                local.push(byte);
                local_origins.push(c.positions[j]);
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: c.positions[end],
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        validate_local_atom(local_bytes, &c.positions[..sep])?;
        local.extend_from_slice(local_bytes);
        local_origins.extend_from_slice(&c.positions[..sep]);
    }

    let domain_start = sep + 1;
    let raw_domain = &b[domain_start..];
    let raw_positions = &c.positions[domain_start..];
    let remove_root = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(remove_root);
    if domain_len == 0 {
        return Err(EmailError::EmptyDomain {
            at: at_pos(c, domain_start, input_len),
        });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: raw_positions[253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0;
    while start <= domain_len {
        let end = raw_domain[start..domain_len]
            .iter()
            .position(|&x| x == b'.')
            .map(|p| start + p)
            .unwrap_or(domain_len);
        if end == start {
            let at = if start == domain_len && remove_root {
                raw_positions[domain_len]
            } else {
                raw_positions[start]
            };
            return Err(EmailError::EmptyLabel { at });
        }
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: raw_positions[start + 63],
            });
        }
        if raw_domain[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: raw_positions[start],
            });
        }
        if raw_domain[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: raw_positions[end - 1],
            });
        }
        for k in start..end {
            let ch = raw_domain[k];
            if !ch.is_ascii_alphanumeric() && ch != b'-' {
                return Err(EmailError::DomainChar {
                    at: raw_positions[k],
                });
            }
        }
        labels.push((start, end));
        if end == domain_len {
            break;
        }
        start = end + 1;
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: at_pos(c, domain_start, input_len),
        });
    }
    let (last_start, last_end) = *labels.last().expect("nonempty domain labels");
    if raw_domain[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: raw_positions[last_start],
        });
    }

    let local_string = String::from_utf8(local).expect("validated ASCII local");
    let domain_string = String::from_utf8(raw_domain[..domain_len].to_vec())
        .expect("validated ASCII domain")
        .to_ascii_lowercase();

    Ok((
        Address {
            local: local_string,
            domain: domain_string,
            quoted,
        },
        local_origins[0],
    ))
}

fn is_valid_atom(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() || bytes.len() > 64 {
        return false;
    }
    let allowed = b"!#$%&'*+-/=?^_`{|}~";
    for i in 0..bytes.len() {
        let b = bytes[i];
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return false;
            }
        } else if !b.is_ascii_alphanumeric() && !allowed.contains(&b) {
            return false;
        }
    }
    true
}

fn parse_with_policy(
    input: &str,
    policy: &Policy,
    allow_single_label: bool,
) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let cleaned = cleanup(raw)?;
    let (mut address, local_start) =
        parse_cleaned(&cleaned, raw.len(), allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(pos) = address.local.find('+') {
            if pos == 0 {
                // The plus's original origin is recovered from the decoded local.
                let origin = if address.quoted {
                    let mut decoded_index = 0usize;
                    let local_slice = &cleaned.bytes[..];
                    let _ = (&mut decoded_index, local_slice);
                    local_start
                } else {
                    local_start
                };
                return Err(EmailError::EmptyNormalizedLocal { at: origin });
            }
            address.local.truncate(pos);
        }
    }

    if !is_valid_atom(&address.local) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), false)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, policy.allow_single_label)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| {
                        local == &address.local && domain == &address.domain
                    })
                {
                    result.duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    result.addresses.push(address);
                    result.source_indices.push(index);
                }
            }
        }
    }

    result
}
