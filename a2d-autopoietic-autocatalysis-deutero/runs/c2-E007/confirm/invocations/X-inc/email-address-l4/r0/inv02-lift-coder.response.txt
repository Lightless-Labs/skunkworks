#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut kept = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut quote_prefix = false;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        if quote_prefix {
            let b = input[i];
            kept.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    kept.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                quote_prefix = false;
            }
            i += 1;
            continue;
        }

        if depth == 0 && kept.is_empty() && input[i] == b'"' {
            quote_prefix = true;
            kept.push(input[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        let b = input[i];
        if depth == 0 {
            if b == b' ' || b == b'\t' || b == b'(' || (b == b'\r' && fold_end(input, i).is_some()) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                if b == b'(' {
                    outer_open = i;
                    depth = 1;
                    i += 1;
                    continue;
                }
                if b == b'\r' {
                    i = fold_end(input, i).unwrap();
                    continue;
                }
                i += 1;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                return Err(EmailError::BadFold { at: i });
            }
            if b == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            if let Some(start) = run_start.take() {
                runs.push(Run { start, end: i });
            }
            kept.push(b);
            positions.push(i);
            i += 1;
        } else {
            if b == b'\\' {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                let next = input.get(i + 1).copied();
                match next {
                    Some(c) if (0x20..=0x7e).contains(&c) => {
                        i += 2;
                    }
                    _ => return Err(EmailError::CommentEscape { at: i }),
                }
                continue;
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                if depth == 0 {
                    if run_start.is_none() {
                        run_start = Some(outer_open);
                    }
                }
                continue;
            }
            if b == b'\r' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'\n' {
                return Err(EmailError::BadFold { at: i });
            }
            if b == b' ' || b == b'\t' {
                i += 1;
                continue;
            }
            if !(0x20..=0x7e).contains(&b) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
        }
    }

    if depth != 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        runs.push(Run {
            start,
            end: input.len(),
        });
    }

    Ok((kept, positions, runs))
}

fn at_pos(pos: &[usize], idx: usize, input_len: usize) -> usize {
    pos.get(idx).copied().unwrap_or(input_len)
}

fn valid_atom(local: &[u8]) -> bool {
    if local.is_empty() || local.len() > 64 {
        return false;
    }
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return false;
            }
        } else if !is_atom_char(b) {
            return false;
        }
    }
    true
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn parse_core(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let (clean, positions, runs) = cleanup(raw)?;
    for run in &runs {
        let before = positions.iter().position(|&p| p > run.start);
        let after = positions.iter().rposition(|&p| p < run.end);
        let boundary = before.is_none()
            || after.is_none()
            || before.map(|i| clean[i] == b'@').unwrap_or(false)
            || after.map(|i| clean[i] == b'@').unwrap_or(false);
        if !boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }
    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: at_pos(&positions, 0, raw.len()),
            });
        }
    }

    let mut ats = Vec::new();
    for i in 0..clean.len() {
        if clean[i] == b'@' && !quote_end.map(|e| i <= e).unwrap_or(false) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: at_pos(&positions, ats[1], raw.len()),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: at_pos(&positions, sep, raw.len()),
        });
    }

    let quoted = clean[0] == b'"';
    let mut local = Vec::new();
    let mut origins = Vec::new();
    if !quoted {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: at_pos(&positions, 64, raw.len()),
            });
        }
        for i in 0..sep {
            let b = clean[i];
            if b == b'.' {
                if i == 0 || i + 1 == sep || clean[i - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: at_pos(&positions, i, raw.len()),
                    });
                }
            } else if !is_atom_char(b) {
                return Err(EmailError::LocalChar {
                    at: at_pos(&positions, i, raw.len()),
                });
            }
            local.push(b);
            origins.push(positions[i]);
        }
    } else {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: at_pos(&positions, end + 1, raw.len()),
            });
        }
        let mut i = 1;
        while i < end {
            let b = clean[i];
            if b == b'\\' {
                if i + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: at_pos(&positions, i, raw.len()),
                    });
                }
                let c = clean[i + 1];
                if c != b'"' && c != b'\\' && c != b'@' {
                    return Err(EmailError::BadEscape {
                        at: at_pos(&positions, i, raw.len()),
                    });
                }
                local.push(c);
                origins.push(positions[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: at_pos(&positions, i, raw.len()),
                    });
                }
                local.push(b);
                origins.push(positions[i]);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: at_pos(&positions, end, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    }

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut removed_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_dot = Some(domain_end);
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: at_pos(&positions, domain_start, raw.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: at_pos(&positions, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if clean[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    labels.push((start, domain_end));

    for (li, &(s, e)) in labels.iter().enumerate() {
        if s == e {
            let p = if li + 1 == labels.len() {
                removed_dot.map(|x| at_pos(&positions, x, raw.len()))
                    .unwrap_or_else(|| at_pos(&positions, s, raw.len()))
            } else {
                at_pos(&positions, s, raw.len())
            };
            return Err(EmailError::EmptyLabel { at: p });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: at_pos(&positions, s + 63, raw.len()),
            });
        }
        if clean[s] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: at_pos(&positions, s, raw.len()),
            });
        }
        if clean[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: at_pos(&positions, e - 1, raw.len()),
            });
        }
        for i in s..e {
            if !clean[i].is_ascii_alphanumeric() && clean[i] != b'-' {
                return Err(EmailError::DomainChar {
                    at: at_pos(&positions, i, raw.len()),
                });
            }
        }
    }
    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: at_pos(&positions, domain_start, raw.len()),
        });
    }
    let (ts, te) = *labels.last().unwrap();
    if clean[ts..te].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: at_pos(&positions, ts, raw.len()),
        });
    }

    let local_string = String::from_utf8(local).unwrap();
    let domain_string = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();
    Ok((
        Address {
            local: local_string,
            domain: domain_string,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(index) = address.local.find('+') {
            if index == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(index);
        }
    }

    let bytes = address.local.as_bytes();
    if !valid_atom(bytes) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                result.push('\\');
            }
            result.push(c);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((i, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_input) = seen.get(&key) {
                    duplicate_of[i] = Some(first_input);
                } else {
                    seen.insert(key, i);
                    source_indices.push(i);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
