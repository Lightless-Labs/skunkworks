pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }

    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge_index, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 1,
                vertex: target,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse_adjacency = vec![Vec::new(); n];
    let mut has_self_loop = vec![false; n];

    for &(source, target) in edges {
        adjacency[source].push(target);
        reverse_adjacency[target].push(source);
        if source == target {
            has_self_loop[source] = true;
        }
    }

    // Iterative depth-first search to compute finishing order.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);

    for start in 0..n {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];

        while let Some((vertex, next_index)) = stack.last_mut() {
            if *next_index < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_index];
                *next_index += 1;

                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                finish_order.push(finished);
            }
        }
    }

    // Traverse the transpose graph in reverse finishing order.
    visited.fill(false);
    let mut components = Vec::new();
    let mut component_cyclic = Vec::new();

    for &start in finish_order.iter().rev() {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![start];
        let mut component = Vec::new();
        let mut has_loop = false;

        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            has_loop |= has_self_loop[vertex];

            for &neighbor in &reverse_adjacency[vertex] {
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }

        component.sort_unstable();
        component_cyclic.push(component.len() > 1 || has_loop);
        components.push(component);
    }

    let mut order: Vec<usize> = (0..components.len()).collect();
    order.sort_unstable_by(|&a, &b| {
        components[b]
            .len()
            .cmp(&components[a].len())
            .then_with(|| components[a][0].cmp(&components[b][0]))
    });

    let mut sorted_components = Vec::with_capacity(components.len());
    let mut sorted_cyclic = Vec::with_capacity(components.len());
    let mut component_of = vec![0; n];

    for (new_id, old_id) in order.into_iter().enumerate() {
        for &vertex in &components[old_id] {
            component_of[vertex] = new_id;
        }
        sorted_components.push(std::mem::take(&mut components[old_id]));
        sorted_cyclic.push(component_cyclic[old_id]);
    }

    Ok(Graph {
        components: sorted_components,
        component_of,
        cyclic: sorted_cyclic,
    })
}
