#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let columns = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != columns {
            return Err(TableError::Ragged {
                row: row_index,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let fixed_overhead = 3 * columns + 1;
    let minimum = fixed_overhead + columns;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let mut trimmed = Vec::with_capacity(rows.len());
    let mut widths = vec![1usize; columns];

    for (row_index, row) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(columns);
        for (column_index, cell) in row.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }

            let end = cell.trim_end_matches(' ');
            let content = cell[..end.len()].to_owned();
            let width = content.chars().count().max(1);
            widths[column_index] = widths[column_index].max(width);
            trimmed_row.push(content);
        }
        trimmed.push(trimmed_row);
    }

    if options.max_width != 0 {
        let mut total: usize = fixed_overhead + widths.iter().sum::<usize>();
        while total > options.max_width {
            let candidate = widths
                .iter()
                .enumerate()
                .filter(|(_, width)| **width > 1)
                .max_by_key(|(index, width)| (**width, *index))
                .map(|(index, _)| index);

            if let Some(index) = candidate {
                widths[index] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    append_border(&mut output, &widths);

    for (row_index, row) in trimmed.iter().enumerate() {
        output.push('|');
        for (column_index, cell) in row.iter().enumerate() {
            output.push(' ');

            let width = widths[column_index];
            let content_width = cell.chars().count();
            let rendered = if content_width > width {
                let mut clipped: String = cell.chars().take(width - 1).collect();
                clipped.push('…');
                clipped
            } else {
                cell.clone()
            };

            let extra = width - rendered.chars().count();
            let alignment = options
                .alignments
                .get(column_index)
                .unwrap_or(&Alignment::Left);
            let (left, right) = match alignment {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            output.extend(std::iter::repeat_n(' ', left));
            output.push_str(&rendered);
            output.extend(std::iter::repeat_n(' ', right));
            output.push_str(" |");
        }
        output.push('\n');

        if options.header && row_index == 0 && trimmed.len() > 1 {
            append_border(&mut output, &widths);
        }
    }

    append_border(&mut output, &widths);
    Ok(output)
}

fn append_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for &width in widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');
}
