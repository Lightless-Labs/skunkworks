#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .filter(|(_, part)| !part.is_empty())
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, part)| *part == "." || *part == "..")
}

fn render(stack: &[String], trailing: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if trailing && !stack.is_empty() {
        result.push('/');
    }
    result
}

fn reduce_normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let trailing = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    for (at, part) in components(path) {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|s| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(part.to_owned()),
        }
    }

    if absolute {
        Ok(render(&stack, trailing))
    } else if stack.is_empty() {
        Ok(".".to_owned())
    } else {
        let mut result = stack.join("/");
        if trailing {
            result.push('/');
        }
        Ok(result)
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    reduce_normalize(path)
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = reduce_normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect())
}

fn validate_home(home: &str) -> Result<Vec<String>, PathError> {
    if !home.starts_with('/') {
        return Err(PathError::InvalidHome);
    }
    let normalized = reduce_normalize(home).map_err(|_| PathError::InvalidHome)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect())
}

fn request_parts<'a>(path: &'a str) -> Result<(&'a str, usize, bool, bool), PathError> {
    validate_raw(path)?;
    let tilde = path.as_bytes().first() == Some(&b'~');
    if tilde && path.len() > 1 && path.as_bytes()[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let expanded = tilde;
    let absolute = !tilde && path.starts_with('/');
    let skip = if tilde {
        if path.len() == 1 { 1 } else { 2 }
    } else {
        0
    };
    let remainder = &path[skip..];
    Ok((remainder, skip, absolute, expanded))
}

fn resolve_impl(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: Option<usize>,
) -> Result<String, PathError> {
    let cwd_stack = validate_context(cwd)?;
    let home_stack = match home {
        Some(value) => Some(validate_home(value)?),
        None => None,
    };

    let mut link_map: Vec<(&str, &str, usize)> = Vec::new();
    if max_expansions.is_some() {
        for (index, link) in links.iter().enumerate() {
            let valid_path = link.path.starts_with('/')
                && !link.path.ends_with('/')
                && link.path != "/"
                && reduce_normalize(&link.path)
                    .is_ok_and(|normalized| normalized == link.path);
            let valid_target = validate_raw(&link.target).is_ok();
            if !valid_path || !valid_target {
                return Err(PathError::InvalidLink { index });
            }
            if let Some((_, _, first)) = link_map.iter().find(|(path, _, _)| *path == link.path) {
                return Err(PathError::DuplicateLink {
                    first: *first,
                    second: index,
                });
            }
            link_map.push((&link.path, &link.target, index));
        }
    }

    let trailing = directory_intent(path);
    let (remainder, request_offset, absolute, expanded) = request_parts(path)?;

    let mut stack = if absolute {
        Vec::new()
    } else if expanded {
        home_stack.ok_or(PathError::MissingHome)?
    } else {
        cwd_stack
    };

    let mut queue: Vec<(String, usize, usize, bool)> = Vec::new();
    for (at, part) in components(remainder) {
        queue.push((part.to_owned(), at + request_offset, 0, false));
    }

    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    while let Some((part, at, target_id, from_target)) = queue.first().cloned() {
        queue.remove(0);

        match part.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    let _ = target_id;
                    let _ = from_target;
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(part);
                if max_expansions.is_none() {
                    continue;
                }

                let matched = {
                    let absolute_path = render(&stack, false);
                    link_map
                        .iter()
                        .find(|(link_path, _, _)| *link_path == absolute_path)
                        .map(|(_, target, index)| (*target, *index))
                };

                if let Some((target, index)) = matched {
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    let budget = max_expansions.unwrap();
                    if expansions >= budget {
                        return Err(PathError::Limit { index });
                    }
                    expansions += 1;
                    active.push(index);

                    let target_parts = components(target);
                    let target_absolute = target.starts_with('/');
                    if target_absolute {
                        stack.clear();
                    } else {
                        stack.pop();
                    }

                    let target_id = index + 1;
                    let mut inserted: Vec<(String, usize, usize, bool)> = target_parts
                        .into_iter()
                        .map(|(offset, component)| {
                            (
                                component.to_owned(),
                                offset,
                                target_id,
                                true,
                            )
                        })
                        .collect();

                    // Keep the expansion active until its target and recursively
                    // expanded targets finish. A sentinel marks its completion.
                    inserted.push((String::new(), 0, target_id, true));
                    queue.splice(0..0, inserted);
                }
            }
        }

        // Target completion sentinel: release the corresponding active link.
        if part.is_empty() && target_id != 0 {
            if let Some(pos) = active.iter().rposition(|&v| v + 1 == target_id) {
                active.remove(pos);
            }
        }
    }

    Ok(render(&stack, trailing))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_impl(path, cwd, home, &[], None)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_impl(path, cwd, home, links, Some(max_expansions))
}
