#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone, Copy, Debug)]
struct Component<'a> {
    text: &'a str,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<Component<'_>> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let at = *offset;
            *offset += part.len() + 1;
            Some((part, at))
        })
        .filter(|(part, _)| !part.is_empty())
        .map(|(text, at)| Component { text, at })
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .map(|c| c.text == "." || c.text == "..")
            .unwrap_or(false)
}

fn format_absolute(stack: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent && !stack.is_empty() {
        result.push('/');
    }
    result
}

fn format_normalized(stack: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return format_absolute(stack, intent);
    }

    if stack.is_empty() {
        return ".".to_string();
    }

    let mut result = stack.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce_component(
    stack: &mut Vec<String>,
    component: &str,
    at: usize,
    absolute: bool,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        }
        _ => stack.push(component.to_string()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    for component in components(path) {
        reduce_component(&mut stack, component.text, component.at, absolute)?;
    }

    Ok(format_normalized(&stack, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized)
        .into_iter()
        .map(|component| component.text.to_string())
        .collect())
}

fn validate_home(home: &str) -> Result<Vec<String>, PathError> {
    if !home.starts_with('/') {
        return Err(PathError::InvalidHome);
    }
    let normalized = normalize(home).map_err(|_| PathError::InvalidHome)?;
    Ok(components(&normalized)
        .into_iter()
        .map(|component| component.text.to_string())
        .collect())
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, Vec<Component<'a>>, Option<Vec<String>>), PathError> {
    validate_raw(path)?;

    let mut components = components(path);
    let mut expanded_home = None;

    if path.starts_with('~') {
        let first = path.as_bytes().get(1).copied();
        if first.is_some() && first != Some(b'/') {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let Some(home_components) = home else {
            return Err(PathError::MissingHome);
        };
        expanded_home = Some(home_components.to_vec());
        components.retain(|component| component.at != 0);
        if path.starts_with("~/") {
            components.retain(|component| component.at > 1);
        }
    }

    Ok((path.starts_with('/'), components, expanded_home))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd)?;
    let home_components = match home {
        Some(value) => Some(validate_home(value)?),
        None => None,
    };

    let (absolute, request_components, expanded_home) =
        prepare_request(path, home_components.as_deref())?;

    let intent = directory_intent(path);
    let mut stack = if absolute {
        Vec::new()
    } else if let Some(expanded) = expanded_home {
        expanded
    } else {
        cwd_components
    };

    for component in request_components {
        reduce_component(&mut stack, component.text, component.at, true)?;
    }

    Ok(format_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();

    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .map(|normalized| normalized == link.path)
                .unwrap_or(false);
        if !path_valid || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }

    Ok(())
}

#[derive(Clone, Copy)]
enum Source<'a> {
    Request,
    Target(usize),
}

#[derive(Clone, Copy)]
struct WorkComponent<'a> {
    text: &'a str,
    at: usize,
    source: Source<'a>,
}

fn absolute_key(stack: &[String]) -> String {
    format_absolute(stack, false)
}

fn traverse_links(
    stack: &mut Vec<String>,
    initial_work: Vec<WorkComponent<'_>>,
    links: &[Link],
    max_expansions: usize,
) -> Result<(), PathError> {
    let mut work = initial_work;
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(component) = work.pop() {
        let at = component.at;
        match component.text {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
                continue;
            }
            _ => stack.push(component.text.to_string()),
        }

        let key = absolute_key(stack);
        if let Some((index, link)) = links.iter().enumerate().find(|(_, link)| link.path == key) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);

            // The link name is replaced by its target. A relative target starts
            // at the matched link's parent.
            stack.pop();

            let target_components = components(&link.target);
            for target_component in target_components.into_iter().rev() {
                work.push(WorkComponent {
                    text: target_component.text,
                    at: target_component.at,
                    source: Source::Target(index),
                });
            }
            work.push(WorkComponent {
                text: "",
                at: usize::MAX,
                source: Source::Target(index),
            });
        }

        while work
            .last()
            .is_some_and(|item| item.text.is_empty() && item.at == usize::MAX)
        {
            work.pop();
            if let Some(index) = active.pop() {
                let _ = index;
            }
        }
    }

    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd)?;
    let home_components = match home {
        Some(value) => Some(validate_home(value)?),
        None => None,
    };

    validate_links(links)?;

    validate_raw(path)?;

    let mut request = components(path);
    let mut tilde_home = None;
    let absolute = path.starts_with('/');

    if path.starts_with('~') {
        let next = path.as_bytes().get(1).copied();
        if next.is_some() && next != Some(b'/') {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let Some(home_components) = home_components.as_ref() else {
            return Err(PathError::MissingHome);
        };
        tilde_home = Some(home_components.clone());
        request.retain(|component| component.at != 0);
        if path.starts_with("~/") {
            request.retain(|component| component.at > 1);
        }
    }

    let intent = directory_intent(path);
    let mut stack = if absolute {
        Vec::new()
    } else if let Some(home) = tilde_home {
        home
    } else {
        cwd_components
    };

    let mut initial_work = Vec::new();

    // Context components are traversed first, from root, so links in the
    // starting directory are expanded before the request is processed.
    if !absolute && tilde_home.is_none() {
        for component in components(cwd) {
            initial_work.push(WorkComponent {
                text: component.text,
                at: 0,
                source: Source::Request,
            });
        }
    } else if !absolute {
        if let Some(home) = home {
            for component in components(home) {
                initial_work.push(WorkComponent {
                    text: component.text,
                    at: 0,
                    source: Source::Request,
                });
            }
        }
    }

    for component in request {
        initial_work.push(WorkComponent {
            text: component.text,
            at: component.at,
            source: Source::Request,
        });
    }

    // Traversal begins at root so context components can trigger links.
    stack.clear();
    traverse_links(&mut stack, initial_work, links, max_expansions)?;

    Ok(format_absolute(&stack, intent))
}
