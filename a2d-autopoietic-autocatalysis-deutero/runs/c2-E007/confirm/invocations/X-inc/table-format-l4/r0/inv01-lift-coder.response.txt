#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn invalid_char(s: &str) -> Option<usize> {
    let mut iter = s.char_indices().peekable();
    while let Some((byte, ch)) = iter.next() {
        if ch == '\r' {
            if matches!(iter.peek(), Some((_, '\n'))) {
                iter.next();
            } else {
                return Some(byte);
            }
        } else if ch != '\n' && (ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}') {
            return Some(byte);
        }
    }
    None
}

fn tokens_for_line(line: &str, markdown: bool) -> Vec<Token> {
    let trimmed = line.trim_end_matches(' ');
    let mut out = Vec::new();
    for ch in trimmed.chars() {
        if markdown && ch == '|' {
            out.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && ch == '\\' {
            out.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            out.push(Token { text: ch.to_string(), width: 1 });
        }
    }
    out
}

fn split_lines(s: &str) -> Vec<String> {
    let normalized = s.replace("\r\n", "\n");
    normalized.split('\n').map(str::to_owned).collect()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }
    match overflow {
        Overflow::Clip => {
            let total: usize = tokens.iter().map(|t| t.width).sum();
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut used = 0;
            let mut result = String::new();
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                result.push_str(&token.text);
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                let text = tokens[start..end].iter().map(|t| t.text.as_str()).collect();
                result.push(text);
                start = end;
            }
            result
        }
    }
}

fn align(text: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let used = text.chars().count();
    let extra = width.saturating_sub(used);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let right = if omit_right { 0 } else { right };
    format!("{}{}{}", " ".repeat(left), text, " ".repeat(right))
}

fn emit_data_line(
    output: &mut String,
    cells: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            output.push('|');
            for (i, cell) in cells.iter().enumerate() {
                output.push(' ');
                output.push_str(&align(cell, widths[i], &alignments[i], false));
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
        Mode::Compact => {
            for i in 0..cells.len() {
                if i > 0 {
                    output.push_str("  ");
                }
                output.push_str(&align(
                    &cells[i],
                    widths[i],
                    &alignments[i],
                    i + 1 == cells.len(),
                ));
            }
            output.push('\n');
        }
    }
}

fn emit_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for width in widths {
        output.push_str(&"-".repeat(width + 2));
        output.push('+');
    }
    output.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_col = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (r, cells) in rows.iter().enumerate() {
        for (c, cell) in cells.iter().enumerate() {
            if let Some(byte) = invalid_char(cell) {
                return Err(TableError::InvalidChar {
                    row: r,
                    column: c,
                    byte,
                });
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };
    let markdown = options.mode == Mode::Markdown;

    let mut logical: Vec<Vec<Vec<Vec<Token>>>> = Vec::with_capacity(rows.len());
    let mut widths = vec![minimum_col; n];
    for row in rows {
        let mut row_cells = Vec::with_capacity(n);
        for (c, cell) in row.iter().enumerate() {
            let lines = split_lines(cell);
            let token_lines: Vec<Vec<Token>> = lines
                .iter()
                .map(|line| tokens_for_line(line, markdown))
                .collect();
            for line in &token_lines {
                let line_width: usize = line.iter().map(|t| t.width).sum();
                widths[c] = widths[c].max(line_width);
            }
            row_cells.push(token_lines);
        }
        logical.push(row_cells);
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > minimum_col)
                .max_by_key(|&i| (widths[i], i));
            if let Some(i) = candidate {
                widths[i] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        emit_border(&mut output, &widths);
    }

    for (row_idx, row) in logical.iter().enumerate() {
        let mut cell_frags: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for c in 0..n {
            let mut joined = Vec::new();
            for line in &row[c] {
                joined.extend(fragments(line, widths[c], &overflows[c]));
            }
            height = height.max(joined.len());
            cell_frags.push(joined);
        }

        for fragment_idx in 0..height {
            let cells: Vec<String> = (0..n)
                .map(|c| cell_frags[c].get(fragment_idx).cloned().unwrap_or_default())
                .collect();
            emit_data_line(&mut output, &cells, &widths, &alignments, &options.mode);
        }

        if options.mode == Mode::Markdown && row_idx == 0 {
            output.push('|');
            for c in 0..n {
                output.push(' ');
                let marker = match alignments[c] {
                    Alignment::Left => format!(":{}", "-".repeat(widths[c] - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(widths[c] - 1)),
                    Alignment::Center => {
                        format!(":{}:", "-".repeat(widths[c] - 2))
                    }
                };
                output.push_str(&marker);
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }

        if options.mode == Mode::Grid && options.header && row_idx == 0 && rows.len() > 1 {
            emit_border(&mut output, &widths);
        }
    }

    if options.mode == Mode::Grid {
        emit_border(&mut output, &widths);
    }
    Ok(output)
}
