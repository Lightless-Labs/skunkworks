struct Token {
    text: String,
    width: usize,

fn valid_scalar(c: char) -> bool {
    if c == '\n' || c == '\r' || c == '\u{2028}' || c == '\u{2029}' {
        return false;
    }
    !c.is_control()
}

fn validate(rows: &[Vec<String>]) -> Result<(), TableError> {
    for (ri, row) in rows.iter().enumerate() {
        for (ci, cell) in row.iter().enumerate() {
            let mut iter = cell.char_indices().peekable();
            while let Some((byte, c)) = iter.next() {
                if c == '\r' {
                    if matches!(iter.peek(), Some((_, '\n'))) {
                        iter.next();
                    } else {
                        return Err(TableError::InvalidChar {
                            row: ri,
                            column: ci,
                            byte,
                        });
                    }
                } else if c != '\n' && !valid_scalar(c) {
                    return Err(TableError::InvalidChar {
                        row: ri,
                        column: ci,
                        byte,
                    });
                }
            }
        }
    }
    Ok(())
}

fn logical_lines(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            let chars: Vec<char> = trimmed.chars().collect();
            let mut tokens = Vec::new();
            let mut i = 0;
            while i < chars.len() {
                let c = chars[i];
                if markdown && (c == '|' || c == '\\') {
                    let mut text = String::new();
                    text.push('\\');
                    text.push(c);
                    tokens.push(Token { text, width: 2 });
                } else {
                    tokens.push(Token {
                        text: c.to_string(),
                        width: 1,
                    });
                }
                i += 1;
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragment_text(tokens: &[Token]) -> String {
    let mut s = String::new();
    for token in tokens {
        s.push_str(&token.text);
    }
    s
}

fn split_line(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }
    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![fragment_text(tokens)];
            }
            let limit = width.saturating_sub(1);
            let mut used = 0;
            let mut end = 0;
            while end < tokens.len() && used + tokens[end].width <= limit {
                used += tokens[end].width;
                end += 1;
            }
            let mut out = fragment_text(&tokens[..end]);
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut used = 0;
                let mut end = start;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(fragment_text(&tokens[start..end]));
                start = end;
            }
            result
        }
    }
}

fn align_fragment(s: &str, width: usize, alignment: &Alignment) -> String {
    let actual = s.chars().count();
    let extra = width.saturating_sub(actual);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), s, " ".repeat(right))
}

fn output_data_line(
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut line = String::new();
    if !matches!(mode, Mode::Compact) {
        line.push('|');
    }
    for i in 0..widths.len() {
        let aligned = align_fragment(&fragments[i], widths[i], &alignments[i]);
        if matches!(mode, Mode::Compact) {
            if i > 0 {
                line.push_str("  ");
            }
            if i + 1 == widths.len() {
                let left_pad = match alignments[i] {
                    Alignment::Left => 0,
                    Alignment::Right => widths[i].saturating_sub(fragments[i].chars().count()),
                    Alignment::Center => {
                        widths[i].saturating_sub(fragments[i].chars().count()) / 2
                    }
                };
                line.push_str(&" ".repeat(left_pad));
                line.push_str(&fragments[i]);
            } else {
                line.push_str(&aligned);
            }
        } else {
            line.push(' ');
            line.push_str(&aligned);
            line.push_str(" |");
        }
    }
    line.push('\n');
    line
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for &width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_width = if matches!(options.mode, Mode::Markdown) {
        3
    } else {
        1
    };
    let overhead = if matches!(options.mode, Mode::Compact) {
        2 * (n - 1)
    } else {
        3 * n + 1
    };
    let minimum = overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    validate(rows)?;

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let transformed: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| logical_lines(cell, matches!(options.mode, Mode::Markdown)))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_width; n];
    for row in &transformed {
        for (ci, cell_lines) in row.iter().enumerate() {
            for line in cell_lines {
                widths[ci] = widths[ci].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let mut candidate: Option<usize> = None;
            for i in 0..n {
                if widths[i] > minimum_width
                    && candidate.map_or(true, |j| widths[i] >= widths[j])
                {
                    candidate = Some(i);
                }
            }
            if let Some(i) = candidate {
                widths[i] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        out.push_str(&border(&widths));
    }

    for (ri, row) in transformed.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for (ci, lines) in row.iter().enumerate() {
            let mut fragments = Vec::new();
            for line in lines {
                fragments.extend(split_line(line, widths[ci], &overflows[ci]));
            }
            height = height.max(fragments.len());
            cell_fragments.push(fragments);
        }

        for fi in 0..height {
            let current: Vec<String> = (0..n)
                .map(|ci| {
                    cell_fragments[ci]
                        .get(fi)
                        .cloned()
                        .unwrap_or_else(String::new)
                })
                .collect();
            out.push_str(&output_data_line(
                &current,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if matches!(options.mode, Mode::Grid) {
            if options.header && ri == 0 && rows.len() > 1 {
                out.push_str(&border(&widths));
            } else if ri + 1 == rows.len() {
                out.push_str(&border(&widths));
            }
        } else if matches!(options.mode, Mode::Markdown) && ri == 0 {
            out.push('|');
            for ci in 0..n {
                out.push(' ');
                let marker = match alignments[ci] {
                    Alignment::Left => format!(":{}", "-".repeat(widths[ci] - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(widths[ci] - 1)),
                    Alignment::Center => {
                        format!(":{}:", "-".repeat(widths[ci] - 2))
                    }
                };
                out.push_str(&marker);
                out.push_str(" |");
            }
            out.push('\n');
        }
    }

    Ok(out)
}

fn invalid_char(s: &str) -> Option<usize> {
    let mut iter = s.char_indices().peekable();
    while let Some((byte, ch)) = iter.next() {
        if ch == '\r' {
            if matches!(iter.peek(), Some((_, '\n'))) {

            } else {
                return Some(byte);
            }

        }
    }
    None

fn tokens_for_line(line: &str, markdown: bool) -> Vec<Token> {
    let mut tokens = Vec::new();
    for ch in line.chars() {
        if markdown && ch == '|' {
            tokens.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && ch == '\\' {
            tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            tokens.push(Token { text: ch.to_string(), width: 1 });
        }
    }
    tokens
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {

                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                out.push(token.clone());
            }
            out.push(Token { text: "…".to_owned(), width: 1 });
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }
            result
        }
    }
}

fn render_cell(tokens: &[Token], width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let text: String = tokens.iter().map(|t| t.text.as_str()).collect();
    let used = token_width(tokens);
    let extra = width.saturating_sub(used);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!(
        "{}{}{}",
        " ".repeat(left),
        text,
        if omit_right { String::new() } else { " ".repeat(right) }
    )
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged { row, expected: n, actual: cells.len() });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount { expected: n, actual: options.alignments.len() });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount { expected: n, actual: options.overflow.len() });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_col = if matches!(options.mode, Mode::Markdown) { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * n.saturating_sub(1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * min_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            if let Some(byte) = invalid_char(cell) {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let prepared: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| logical_lines(cell, matches!(options.mode, Mode::Markdown))).collect())
        .collect();

    let mut widths = vec![min_col; n];
    for row in &prepared {
        for (column, cell_lines) in row.iter().enumerate() {
            for line in cell_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_col)
                .max_by_key(|&i| (widths[i], i));
            match candidate {
                Some(i) => {
                    widths[i] -= 1;
                    total -= 1;
                }
                None => break,
            }
        }
    }

    let mut output = String::new();

    let emit_border = |out: &mut String, widths: &[usize]| {
        out.push('+');
        for &w in widths {
            out.push_str(&"-".repeat(w + 2));
            out.push('+');
        }
        out.push('\n');
    };

    let emit_data_line = |out: &mut String, row_frags: &[Vec<Vec<Token>>], idx: usize, compact: bool| {
        if compact {
            for column in 0..n {
                if column > 0 {
                    out.push_str("  ");
                }
                let empty = Vec::new();
                let content = row_frags[column].get(idx).unwrap_or(&empty);
                out.push_str(&render_cell(content, widths[column], &alignments[column], column + 1 == n));
            }
        } else {
            out.push('|');
            for column in 0..n {
                let empty = Vec::new();
                let content = row_frags[column].get(idx).unwrap_or(&empty);
                out.push(' ');
                out.push_str(&render_cell(content, widths[column], &alignments[column], false));
                out.push_str(" |");
            }
        }
        out.push('\n');
    };

    if matches!(options.mode, Mode::Grid) {
        emit_border(&mut output, &widths);
    }

    for (row_index, row) in prepared.iter().enumerate() {
        let row_frags: Vec<Vec<Vec<Token>>> = row
            .iter()
            .enumerate()
            .map(|(column, lines)| {
                let mut all = Vec::new();
                for line in lines {
                    all.extend(fragments(line, widths[column], &overflows[column]));
                }
                all
            })
            .collect();
        let height = row_frags.iter().map(Vec::len).max().unwrap_or(1);
        for idx in 0..height {
            emit_data_line(
                &mut output,
                &row_frags,
                idx,
                matches!(options.mode, Mode::Compact),
            );
        }

        match options.mode {
            Mode::Grid if options.header && row_index == 0 && rows.len() > 1 => {
                emit_border(&mut output, &widths);
            }
            Mode::Markdown if row_index == 0 => {
                output.push('|');
                for (column, &width) in widths.iter().enumerate() {
                    let marker = match alignments[column] {
                        Alignment::Left => format!(":{}", "-".repeat(width - 1)),
                        Alignment::Right => format!("{}:", "-".repeat(width - 1)),
                        Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
                    };
                    output.push(' ');
                    output.push_str(&marker);
                    output.push_str(" |");
                }
                output.push('\n');
            }
            _ => {}
        }
    }

    if matches!(options.mode, Mode::Grid) {
        emit_border(&mut output, &widths);
    }

    Ok(output)
}

fn invalid_char(s: &str) -> Option<usize> {
    let mut iter = s.char_indices().peekable();
    while let Some((byte, ch)) = iter.next() {
        if ch == '\r' {
            if matches!(iter.peek(), Some((_, '\n'))) {
                iter.next();
            } else {
                return Some(byte);
            }
        } else if ch != '\n' && (ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}') {
            return Some(byte);
        }
    }
    None
}

fn tokens_for_line(line: &str, markdown: bool) -> Vec<Token> {
    let trimmed = line.trim_end_matches(' ');
    let mut out = Vec::new();
    for ch in trimmed.chars() {
        if markdown && ch == '|' {
            out.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && ch == '\\' {
            out.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            out.push(Token { text: ch.to_string(), width: 1 });
        }
    }
    out
}

fn split_lines(s: &str) -> Vec<String> {
    let normalized = s.replace("\r\n", "\n");
    normalized.split('\n').map(str::to_owned).collect()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }
    match overflow {
        Overflow::Clip => {
            let total: usize = tokens.iter().map(|t| t.width).sum();
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut used = 0;
            let mut result = String::new();
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                result.push_str(&token.text);
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                let text = tokens[start..end].iter().map(|t| t.text.as_str()).collect();
                result.push(text);
                start = end;
            }
            result
        }
    }
}

fn align(text: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let used = text.chars().count();
    let extra = width.saturating_sub(used);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let right = if omit_right { 0 } else { right };
    format!("{}{}{}", " ".repeat(left), text, " ".repeat(right))
}

fn emit_data_line(
    output: &mut String,
    cells: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            output.push('|');
            for (i, cell) in cells.iter().enumerate() {
                output.push(' ');
                output.push_str(&align(cell, widths[i], &alignments[i], false));
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
        Mode::Compact => {
            for i in 0..cells.len() {
                if i > 0 {
                    output.push_str("  ");
                }
                output.push_str(&align(
                    &cells[i],
                    widths[i],
                    &alignments[i],
                    i + 1 == cells.len(),
                ));
            }
            output.push('\n');
        }
    }
}

}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_col = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (r, cells) in rows.iter().enumerate() {
        for (c, cell) in cells.iter().enumerate() {
            if let Some(byte) = invalid_char(cell) {
                return Err(TableError::InvalidChar {
                    row: r,
                    column: c,
                    byte,
                });
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };
    let markdown = options.mode == Mode::Markdown;

    let mut logical: Vec<Vec<Vec<Vec<Token>>>> = Vec::with_capacity(rows.len());
    let mut widths = vec![minimum_col; n];
    for row in rows {
        let mut row_cells = Vec::with_capacity(n);
        for (c, cell) in row.iter().enumerate() {
            let lines = split_lines(cell);
            let token_lines: Vec<Vec<Token>> = lines
                .iter()
                .map(|line| tokens_for_line(line, markdown))
                .collect();
            for line in &token_lines {
                let line_width: usize = line.iter().map(|t| t.width).sum();
                widths[c] = widths[c].max(line_width);
            }
            row_cells.push(token_lines);
        }
        logical.push(row_cells);
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > minimum_col)
                .max_by_key(|&i| (widths[i], i));
            if let Some(i) = candidate {
                widths[i] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        emit_border(&mut output, &widths);
    }

    for (row_idx, row) in logical.iter().enumerate() {
        let mut cell_frags: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for c in 0..n {
            let mut joined = Vec::new();
            for line in &row[c] {
                joined.extend(fragments(line, widths[c], &overflows[c]));
            }
            height = height.max(joined.len());
            cell_frags.push(joined);
        }

        for fragment_idx in 0..height {
            let cells: Vec<String> = (0..n)
                .map(|c| cell_frags[c].get(fragment_idx).cloned().unwrap_or_default())
                .collect();
            emit_data_line(&mut output, &cells, &widths, &alignments, &options.mode);
        }

        if options.mode == Mode::Markdown && row_idx == 0 {
            output.push('|');
            for c in 0..n {
                output.push(' ');
                let marker = match alignments[c] {
                    Alignment::Left => format!(":{}", "-".repeat(widths[c] - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(widths[c] - 1)),

                        format!(":{}:", "-".repeat(widths[c] - 2))
                    }
                };
                output.push_str(&marker);
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }

        if options.mode == Mode::Grid && options.header && row_idx == 0 && rows.len() > 1 {
            emit_border(&mut output, &widths);
        }
    }

    if options.mode == Mode::Grid {
        emit_border(&mut output, &widths);
    }
    Ok(output)
}
