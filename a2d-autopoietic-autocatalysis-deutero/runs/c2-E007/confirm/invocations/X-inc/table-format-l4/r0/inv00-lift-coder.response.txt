#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type CellLines = Vec<Vec<Token>>;

fn invalid_char(c: char) -> bool {
    c.is_control() && c != '\n' && c != '\r' || c == '\u{2028}' || c == '\u{2029}'
}

fn tokenize_line(line: &str, markdown: bool) -> Vec<Token> {
    let mut chars: Vec<char> = line.chars().collect();
    while chars.last() == Some(&' ') {
        chars.pop();
    }

    let mut tokens = Vec::new();
    for ch in chars {
        if markdown && ch == '|' {
            tokens.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && ch == '\\' {
            tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            tokens.push(Token { text: ch.to_string(), width: 1 });
        }
    }
    tokens
}

fn cell_lines(cell: &str, markdown: bool) -> CellLines {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| tokenize_line(line, markdown))
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.to_vec()];
            }
            let mut result = Vec::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                result.push(token.clone());
            }
            result.push(Token { text: "…".to_owned(), width: 1 });
            vec![result]
        }
        Overflow::Wrap => {
            let mut output = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // Width floors guarantee every token is individually usable.
                output.push(tokens[start..end].to_vec());
                start = end;
            }
            output
        }
    }
}

fn render_aligned(tokens: &[Token], width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let content: String = tokens.iter().map(|t| t.text.as_str()).collect();
    let used = token_width(tokens);
    let extra = width.saturating_sub(used);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), content, if omit_right { String::new() } else { " ".repeat(right) })
}

fn render_data_line(
    parts: &[Vec<Token>],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    match mode {
        Mode::Compact => {
            let mut line = String::new();
            for i in 0..parts.len() {
                if i > 0 {
                    line.push_str("  ");
                }
                line.push_str(&render_aligned(
                    &parts[i],
                    widths[i],
                    &alignments[i],
                    i + 1 == parts.len(),
                ));
            }
            line.push('\n');
            line
        }
        Mode::Grid | Mode::Markdown => {
            let mut line = String::from("|");
            for i in 0..parts.len() {
                line.push(' ');
                line.push_str(&render_aligned(&parts[i], widths[i], &alignments[i], false));
                line.push_str(" |");
            }
            line.push('\n');
            line
        }
    }
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

fn markdown_separator(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut line = String::from("|");
    for (width, alignment) in widths.iter().zip(alignments) {
        let marker = match alignment {
            Alignment::Left => format!(":{}", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        line.push(' ');
        line.push_str(&marker);
        line.push_str(" |");
    }
    line.push('\n');
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged { row, expected: n, actual: cells.len() });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount { expected: n, actual: options.alignments.len() });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount { expected: n, actual: options.overflow.len() });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_width = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * min_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            let mut byte = 0;
            let chars: Vec<(usize, char)> = cell.char_indices().collect();
            let mut i = 0;
            while i < chars.len() {
                let (offset, ch) = chars[i];
                if ch == '\r' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                        i += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar { row, column, byte: offset });
                }
                if invalid_char(ch) {
                    return Err(TableError::InvalidChar { row, column, byte: offset });
                }
                byte = offset;
                let _ = byte;
                i += 1;
            }
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = options.mode == Mode::Markdown;
    let logical: Vec<Vec<CellLines>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| cell_lines(cell, markdown)).collect())
        .collect();

    let mut widths = vec![min_width; n];
    for row in &logical {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_width)
                .max_by_key(|&i| (widths[i], i));
            if let Some(i) = candidate {
                widths[i] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in logical.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<Vec<Token>>> = Vec::with_capacity(n);
        let mut height = 1;
        for (column, lines) in row.iter().enumerate() {
            let mut all = Vec::new();
            for line in lines {
                let mut fs = fragments(line, widths[column], &overflows[column]);
                height = height.max(fs.len());
                all.append(&mut fs);
            }
            cell_fragments.push(all);
        }

        for fragment_index in 0..height {
            let parts: Vec<Vec<Token>> = (0..n)
                .map(|column| cell_fragments[column].get(fragment_index).cloned().unwrap_or_default())
                .collect();
            output.push_str(&render_data_line(&parts, &widths, &alignments, &options.mode));
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && row_index + 1 < logical.len() {
            output.push_str(&border(&widths));
        }
        if options.mode == Mode::Markdown && row_index == 0 {
            output.push_str(&markdown_separator(&widths, &alignments));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
