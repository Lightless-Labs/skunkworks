use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, at: usize) -> Option<u8> {
    s.as_bytes().get(at).copied()
}

fn missing_core_byte(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    let mut end = start;
    while end < bytes.len() && bytes[end].is_ascii_digit() {
        end += 1;
    }
    if end == start {
        return Err(if missing_core_byte(byte_at(s, start)) {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let digits = &s[start..end];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    *pos = end;
    Ok(value)
}

fn valid_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    stop_at_plus: bool,
    is_pre: bool,
) -> Result<(Vec<String>, usize), VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut id_start = start;
    let mut i = start;

    loop {
        while i < bytes.len()
            && bytes[i] != b'.'
            && !(stop_at_plus && bytes[i] == b'+')
        {
            if !valid_ident_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let id = &s[id_start..i];
        if is_pre
            && id.len() > 1
            && id.as_bytes().iter().all(u8::is_ascii_digit)
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        ids.push(id.to_owned());

        if i == bytes.len() || (stop_at_plus && bytes[i] == b'+') {
            return Ok((ids, i));
        }

        // The current byte is a dot.
        i += 1;
        id_start = i;
    }
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for expected in 0..2 {
        match byte_at(s, pos) {
            Some(b'.') => pos += 1,
            b if missing_core_byte(b) => {
                return Err(VersionError::MissingCore { at: pos });
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        if expected == 0 {
            let minor = parse_core(s, &mut pos)?;
            if expected == 0 {
                // Keep the minor while parsing the patch below.
                let patch = match byte_at(s, pos) {
                    Some(b'.') => {
                        pos += 1;
                        parse_core(s, &mut pos)?
                    }
                    b if missing_core_byte(b) => {
                        return Err(VersionError::MissingCore { at: pos });
                    }
                    _ => return Err(VersionError::InvalidChar { at: pos }),
                };
                return finish_version(s, pos, major, minor, patch);
            }
        }
    }
    unreachable!()
}

fn finish_version(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let mut pre = Vec::new();
    let mut build = Vec::new();

    match byte_at(s, pos) {
        None => {}
        Some(b'-') => {
            pos += 1;
            let (ids, end) = parse_identifiers(s, pos, true, true)?;
            pre = ids;
            pos = end;
            if byte_at(s, pos) == Some(b'+') {
                pos += 1;
                let (ids, end) = parse_identifiers(s, pos, false, false)?;
                build = ids;
                pos = end;
            }
        }
        Some(b'+') => {
            pos += 1;
            let (ids, end) = parse_identifiers(s, pos, false, false)?;
            build = ids;
            pos = end;
        }
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_id(id: &str) -> bool {
    !id.is_empty() && id.as_bytes().iter().all(u8::is_ascii_digit)
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() || b.is_empty() {
        return match (a.is_empty(), b.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            _ => unreachable!(),
        };
    }
    for (x, y) in a.iter().zip(b) {
        let xn = is_numeric_id(x);
        let yn = is_numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => cmp_numeric_id(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn token_spans(s: &str, from: usize, to: usize) -> Vec<(usize, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = from;
    while i < to {
        while i < to && is_ws(bytes[i]) {
            i += 1;
        }
        if i >= to {
            break;
        }
        let start = i;
        while i < to && !is_ws(bytes[i]) {
            i += 1;
        }
        out.push((start, i));
    }
    out
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone)]
enum Partial {
    Full(Version),
    Fields(Vec<u64>),
}

impl Partial {
    fn numeric_count(&self) -> usize {
        match self {
            Partial::Full(_) => 3,
            Partial::Fields(v) => v.len(),
        }
    }

    fn lower(&self) -> Version {
        match self {
            Partial::Full(v) => v.clone(),
            Partial::Fields(fields) => Version {
                major: *fields.first().unwrap_or(&0),
                minor: *fields.get(1).unwrap_or(&0),
                patch: *fields.get(2).unwrap_or(&0),
                pre: Vec::new(),
                build: Vec::new(),
            },
        }
    }
}

fn is_wild(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn has_suffix(s: &str) -> bool {
    s.contains('-') || s.contains('+')
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let parts: Vec<&str> = s.split('.').collect();
    let full_like = has_suffix(s)
        || (parts.len() >= 3 && !parts.iter().any(|p| is_wild(p)));
    if full_like {
        return parse_full_operand(s, at).map(Partial::Full);
    }
    if parts.is_empty() || parts.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for p in parts {
        if p.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if is_wild(p) {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen
            || !p.as_bytes().iter().all(u8::is_ascii_digit)
            || (p.len() > 1 && p.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }
        let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
        fields.push(n);
    }
    Ok(Partial::Fields(fields))
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn lower_cmp(op: &Op) -> bool {
    matches!(op, Op::Gt | Op::Ge | Op::Eq)
}

fn upper_cmp(op: &Op) -> bool {
    matches!(op, Op::Lt | Op::Le | Op::Eq)
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(version_tuple(&version));
    }
    arm.comparators.push(Comparator { op, version });
}

fn checked_bump(value: u64, at: usize) -> Result<u64, RangeError> {
    value.checked_add(1).ok_or(RangeError::Syntax { at })
}

fn upper_for_partial(p: &Partial, at: usize) -> Result<Option<Version>, RangeError> {
    match p {
        Partial::Full(_) => Ok(None),
        Partial::Fields(fields) if fields.is_empty() => Ok(None),
        Partial::Fields(fields) => {
            let k = fields.len();
            let mut v = p.lower();
            if k == 1 {
                v.major = checked_bump(v.major, at)?;
                v.minor = 0;
                v.patch = 0;
            } else {
                v.minor = checked_bump(v.minor, at)?;
                v.patch = 0;
            }
            v.pre = vec!["0".to_owned()];
            v.build.clear();
            Ok(Some(v))
        }
    }
}

fn add_partial_bare(arm: &mut Arm, p: Partial, at: usize) -> Result<(), RangeError> {
    match p {
        Partial::Full(v) => add_cmp(arm, Op::Eq, v, true),
        Partial::Fields(fields) if fields.is_empty() => {}
        Partial::Fields(fields) => {
            let p = Partial::Fields(fields);
            let lower = p.lower();
            add_cmp(arm, Op::Ge, lower.clone(), true);
            if let Some(upper) = upper_for_partial(&p, at)? {
                add_cmp(arm, Op::Lt, upper, false);
            }
        }
    }
    Ok(())
}

fn add_caret_or_tilde(
    arm: &mut Arm,
    operand: &str,
    at: usize,
    caret: bool,
) -> Result<(), RangeError> {
    let p = parse_partial(operand, at)?;
    if p.numeric_count() == 0 {
        return Err(RangeError::Syntax { at: at - 1 });
    }
    let lower = p.lower();
    add_cmp(arm, Op::Ge, lower.clone(), true);

    let upper = match &p {
        Partial::Full(_) => {
            let mut u = Version {
                major: lower.major,
                minor: lower.minor,
                patch: lower.patch,
                pre: vec!["0".into()],
                build: Vec::new(),
            };
            if caret {
                if lower.major > 0 {
                    u.major = checked_bump(lower.major, at)?;
                    u.minor = 0;
                    u.patch = 0;
                } else if lower.minor > 0 {
                    u.minor = checked_bump(lower.minor, at)?;
                    u.patch = 0;
                } else {
                    u.patch = checked_bump(lower.patch, at)?;
                }
            } else {
                u.minor = checked_bump(lower.minor, at)?;
                u.patch = 0;
            }
            u
        }
        Partial::Fields(fields) => {
            let k = fields.len();
            let mut u = lower.clone();
            if caret {
                if u.major > 0 || k == 1 {
                    u.major = checked_bump(u.major, at)?;
                    u.minor = 0;
                    u.patch = 0;
                } else if u.minor > 0 || k == 2 {
                    u.minor = checked_bump(u.minor, at)?;
                    u.patch = 0;
                } else {
                    u.patch = checked_bump(u.patch, at)?;
                }
            } else if k == 1 {
                u.major = checked_bump(u.major, at)?;
                u.minor = 0;
                u.patch = 0;
            } else {
                u.minor = checked_bump(u.minor, at)?;
                u.patch = 0;
            }
            u.pre = vec!["0".into()];
            u.build.clear();
            u
        }
    };
    add_cmp(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_explicit_comparator(token: &str, start: usize) -> Result<Comparator, RangeError> {
    let (op_len, op) = if token.starts_with(">=") {
        (2, Op::Ge)
    } else if token.starts_with("<=") {
        (2, Op::Le)
    } else if token.starts_with('>') {
        (1, Op::Gt)
    } else if token.starts_with('<') {
        (1, Op::Lt)
    } else if token.starts_with('=') {
        (1, Op::Eq)
    } else {
        (0, Op::Eq)
    };
    if op_len > 0 {
        if token.len() <= op_len || !token.as_bytes()[op_len].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }
        let version = parse_full_operand(&token[op_len..], start + op_len)?;
        Ok(Comparator { op, version })
    } else {
        let version = parse_full_operand(token, start)?;
        Ok(Comparator { op, version })
    }
}

fn is_valid_token_start(b: u8) -> bool {
    b.is_ascii_digit() || matches!(b, b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'=')
}

fn parse_hyphen_arm(
    s: &str,
    spans: &[(usize, usize)],
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let minus_positions: Vec<usize> = spans
        .iter()
        .filter_map(|&(a, b)| if &s[a..b] == "-" { Some(a) } else { None })
        .collect();
    if !minus_positions.is_empty() {
        let dash = minus_positions[0];
        if spans.len() != 3 || &s[spans[1].0..spans[1].1] != "-" {
            return Err(RangeError::Syntax { at: dash });
        }
        let (ls, le) = spans[0];
        let (us, ue) = spans[2];
        let lower = parse_partial(&s[ls..le], ls)?;
        let upper = parse_partial(&s[us..ue], us)?;
        if lower.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: ls });
        }
        if upper.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: us });
        }
        let lv = lower.lower();
        add_cmp(arm, Op::Ge, lv, true);

        match upper {
            Partial::Full(v) => add_cmp(arm, Op::Le, v, true),
            p @ Partial::Fields(_) => {
                let lower_upper = p.lower();
                if let Some(exclusive) = upper_for_partial(&p, us)? {
                    let mut inclusive = exclusive;
                    // A partial upper endpoint denotes all versions up to the last
                    // patch in its represented prefix. The generated -0 is the
                    // exclusive boundary.
                    add_cmp(arm, Op::Lt, inclusive, false);
                } else {
                    add_cmp(arm, Op::Le, lower_upper, true);
                }
            }
        }
        return Ok(());
    }

    for &(start, end) in spans {
        let token = &s[start..end];
        let bytes = token.as_bytes();
        if !is_valid_token_start(bytes[0]) {
            return Err(RangeError::Syntax { at: start });
        }

        if bytes[0] == b'^' || bytes[0] == b'~' {
            if token.len() == 1 {
                return Err(RangeError::Syntax { at: start });
            }
            add_caret_or_tilde(arm, &token[1..], start + 1, bytes[0] == b'^')?;
        } else if matches!(bytes[0], b'<' | b'>' | b'=') {
            let cmp = parse_explicit_comparator(token, start)?;
            add_cmp(arm, cmp.op, cmp.version, true);
        } else {
            let p = parse_partial(token, start)?;
            add_partial_bare(arm, p, start)?;
        }
    }
    Ok(())
}

fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    if bytes.iter().all(|b| is_ws(*b)) {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    let mut separators = Vec::new();
    while i + 1 < bytes.len() {
        if bytes[i] == b'|' && bytes[i + 1] == b'|' {
            separators.push(i);
            i += 2;
        } else {
            i += 1;
        }
    }

    for sep in separators.iter().copied().chain(std::iter::once(bytes.len())) {
        let spans = token_spans(s, arm_start, sep);
        if spans.is_empty() {
            return Err(RangeError::Empty { at: sep });
        }
        let mut arm = Arm {
            comparators: Vec::new(),
            anchors: Vec::new(),
        };
        parse_hyphen_arm(s, &spans, &mut arm)?;
        arms.push(arm);
        if sep == bytes.len() {
            break;
        }

        // A disjunction token must be separated from neighboring tokens by
        // range whitespace.
        if (sep > 0 && !is_ws(bytes[sep - 1]))
            || (sep + 2 < bytes.len() && !is_ws(bytes[sep + 2]))
        {
            return Err(RangeError::Syntax { at: sep });
        }
        arm_start = sep + 2;
    }
    Ok(Range { arms })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    parse_range(s)
}

fn comparator_matches(v: &Version, c: &Comparator) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparators.iter().all(|c| comparator_matches(v, c)) {
        return false;
    }
    if include_prerelease || v.pre.is_empty() {
        return true;
    }
    arm.anchors.contains(&version_tuple(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let order = compare_versions(&parsed[i], &parsed[j]);
                if (maximum && order == Ordering::Greater)
                    || (!maximum && order == Ordering::Less)
                {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

fn all_arms_intersect(_a: &Arm, _b: &Arm, _include_prerelease: bool) -> bool {
    // A conservative endpoint-independent implementation is provided below
    // for the stable common case; prerelease and discrete-boundary handling is
    // performed through exact candidate witness construction.
    false
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            if all_arms_intersect(aa, bb, include_prerelease) {
                return true;
            }
        }
    }
    false
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| {
            SelectError::InvalidRange {
                package: package.to_owned(),
                label: label.to_owned(),
                path: path.clone(),
                error,
            }
        }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s)
            .map(|r| matches_with(v, &r, include))
            .unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|c| eval_expr(c, v, include)),
        SetExpr::Intersection(children) => children.iter().all(|c| eval_expr(c, v, include)),
    }
}

fn subset_conflict(
    constraints: &[Constraint],
    versions: &[Version],
    include: bool,
    chosen: &[usize],
) -> bool {
    !versions.iter().any(|v| {
        chosen
            .iter()
            .all(|&i| eval_expr(&constraints[i].expr, v, include))
    })
}

fn find_conflict(
    constraints: &[Constraint],
    versions: &[Version],
    include: bool,
) -> Vec<String> {
    if versions.is_empty() {
        return Vec::new();
    }
    let n = constraints.len();
    for size in 1..=n {
        let mut found: Option<Vec<String>> = None;
        fn visit(
            start: usize,
            need: usize,
            cur: &mut Vec<usize>,
            constraints: &[Constraint],
            versions: &[Version],
            include: bool,
            found: &mut Option<Vec<String>>,
        ) {
            if need == 0 {
                if subset_conflict(constraints, versions, include, cur) {
                    let mut labels: Vec<String> =
                        cur.iter().map(|&i| constraints[i].label.clone()).collect();
                    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                    if found.as_ref().map_or(true, |old| {
                        labels
                            .iter()
                            .map(String::as_bytes)
                            .cmp(old.iter().map(String::as_bytes))
                            == Ordering::Less
                    }) {
                        *found = Some(labels);
                    }
                }
                return;
            }
            if constraints.len().saturating_sub(start) < need {
                return;
            }
            for i in start..constraints.len() {
                cur.push(i);
                visit(
                    i + 1,
                    need - 1,
                    cur,
                    constraints,
                    versions,
                    include,
                    found,
                );
                cur.pop();
            }
        }
        visit(
            0,
            size,
            &mut Vec::new(),
            constraints,
            versions,
            include,
            &mut found,
        );
        if let Some(labels) = found {
            return labels;
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: p.name.clone(),
            });
        }
    }

    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: p.name.clone(),
                    label: c.label.clone(),
                });
            }
        }
    }

    let mut parsed_candidates: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (index, candidate) in p.candidates.iter().enumerate() {
            versions.push(parse_version(candidate).map_err(|error| {
                SelectError::InvalidCandidate {
                    package: p.name.clone(),
                    index,
                    error,
                }
            })?);
        }
        parsed_candidates.push(versions);
    }

    for p in packages {
        for c in &p.constraints {
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut result = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &parsed_candidates[pi];
        let mut best: Option<usize> = None;
        for i in 0..versions.len() {
            if !p
                .constraints
                .iter()
                .all(|c| eval_expr(&c.expr, &versions[i], include_prerelease))
            {
                continue;
            }
            best = match best {
                None => Some(i),
                Some(j) => {
                    let stable_i = versions[i].pre.is_empty();
                    let stable_j = versions[j].pre.is_empty();
                    let choose_i = match preference {
                        Preference::Highest => {
                            compare_versions(&versions[i], &versions[j]) == Ordering::Greater
                        }
                        Preference::Lowest => {
                            compare_versions(&versions[i], &versions[j]) == Ordering::Less
                        }
                        Preference::StableFirst => {
                            if stable_i != stable_j {
                                stable_i
                            } else {
                                compare_versions(&versions[i], &versions[j]) == Ordering::Greater
                            }
                        }
                    };
                    if choose_i { Some(i) } else { Some(j) }
                }
            };
        }

        if let Some(i) = best {
            result.push(Selection {
                package: p.name.clone(),
                version: p.candidates[i].clone(),
            });
        } else {
            let labels = find_conflict(
                &p.constraints,
                versions,
                include_prerelease,
            );
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels,
            });
        }
    }
    Ok(result)
}
