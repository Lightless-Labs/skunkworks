```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    state: Option<State>,
    attempts: u8,
}

pub struct Runner {
    entries: Vec<Entry>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dependency) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(resolved);
        }

        let mut color = vec![0u8; jobs.len()];
        fn visit(node: usize, dependencies: &[Vec<usize>], color: &mut [u8]) -> bool {
            if color[node] == 1 {
                return true;
            }
            if color[node] == 2 {
                return false;
            }
            color[node] = 1;
            for &dependency in &dependencies[node] {
                if visit(dependency, dependencies, color) {
                    return true;
                }
            }
            color[node] = 2;
            false
        }
        for i in 0..jobs.len() {
            if visit(i, &dependencies, &mut color) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .zip(dependencies)
            .map(|(job, dependencies)| Entry {
                job,
                dependencies,
                state: None,
                attempts: 0,
            })
            .collect();

        Ok(Self {
            entries,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.entries.iter().any(|entry| {
            matches!(entry.state, Some(State::Waiting { .. } | State::Running { .. }))
        }) {
            return Err(Error::Busy);
        }

        for entry in &mut self.entries {
            entry.state = Some(State::Waiting { ready_at: now });
            entry.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;
        for (i, entry) in self.entries.iter().enumerate() {
            let ready_at = match entry.state {
                Some(State::Waiting { ready_at }) if ready_at <= now => ready_at,
                _ => continue,
            };
            if !entry.dependencies.iter().all(|&dependency| {
                matches!(self.entries[dependency].state, Some(State::Succeeded))
            }) {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (u64::from(entry.job.priority) + age).min(9) as u8
            };

            if selected
                .map(|(best_i, best_priority)| {
                    effective > best_priority || (effective == best_priority && i < best_i)
                })
                .unwrap_or(true)
            {
                selected = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = selected else {
            return Ok(None);
        };

        let entry = &mut self.entries[index];
        entry.attempts += 1;
        let attempt = entry.attempts;
        entry.state = Some(State::Running { attempt });
        Ok(Some(Ticket {
            id: entry.job.id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = self
            .entries
            .iter()
            .position(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;
        if !matches!(self.entries[index].state, Some(State::Running { .. })) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.entries[index].state = Some(State::Succeeded);
            return Ok(());
        }

        let attempts = self.entries[index].attempts;
        let max_attempts = self.entries[index].job.max_attempts;
        if attempts == max_attempts {
            self.entries[index].state = Some(State::Failed);
            self.propagate_blocked();
        } else {
            let shift = u32::from(attempts.saturating_sub(1));
            let delay = self
                .retry_base
                .checked_shl(shift)
                .unwrap_or(u64::MAX)
                .min(self.retry_cap);
            let ready_at = now.saturating_add(delay);
            self.entries[index].state = Some(State::Waiting { ready_at });
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let entry = self
            .entries
            .iter()
            .find(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;
        entry.state.clone().ok_or(Error::NoCycle)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();
            for (i, entry) in self.entries.iter().enumerate() {
                if !matches!(entry.state, Some(State::Waiting { .. })) {
                    continue;
                }
                if entry.dependencies.iter().any(|&dependency| {
                    matches!(
                        self.entries[dependency].state,
                        Some(State::Failed | State::Blocked)
                    )
                }) {
                    newly_blocked.push(i);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for index in newly_blocked {
                self.entries[index].state = Some(State::Blocked);
            }
        }
    }
}
```
