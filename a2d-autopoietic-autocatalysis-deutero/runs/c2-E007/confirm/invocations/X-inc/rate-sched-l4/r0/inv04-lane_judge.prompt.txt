#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

struct Entry {
    job: Job,
    state: State,
    dependencies: Vec<usize>,
    remaining: Option<u64>,
    start_order: u64,
}

pub struct Pool {
    entries: Vec<Entry>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    first_run: bool,
    sequence: u64,
}

fn valid_token(s: &str, min: usize, max: usize) -> bool {
    let b = s.as_bytes();
    (min..=max).contains(&b.len())
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|t| !valid_token(t, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut deps = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, name) in job.dependencies.iter().enumerate() {
                if !seen.insert(name) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(name) {
                    Some(&idx) => resolved.push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            deps.push(resolved);
        }

        fn visit(
            i: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[i] == 1 {
                return false;
            }
            if marks[i] == 2 {
                return true;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[i] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &deps, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                dependencies: deps[i].clone(),
                remaining: None,
                start_order: 0,
            })
            .collect();

        Ok(Self {
            entries,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
            sequence: 0,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.entries
            .iter()
            .find(|e| e.job.id == id)
            .map(|e| e.state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut t = if self.first_run { 0 } else { self.horizon };
        let initial = self.first_run;
        self.first_run = false;

        loop {
            if initial && t == 0 || !initial && t == self.horizon {
                self.process_timestamp(t, &mut lines);
            }

            let next = self.next_event_after(t);
            match next {
                Some(nt) if nt <= until => {
                    t = nt;
                    self.process_timestamp(t, &mut lines);
                }
                _ => break,
            }
        }

        self.horizon = until;
        Ok(lines)
    }

    fn next_event_after(&self, t: u64) -> Option<u64> {
        let mut next = None;
        for e in &self.entries {
            let candidate = match e.state {
                State::Running { finish_at, .. } if finish_at > t => Some(finish_at),
                State::Waiting { ready_at, .. } if ready_at > t => Some(ready_at),
                _ => None,
            };
            if let Some(v) = candidate {
                next = Some(next.map_or(v, |n: u64| n.min(v)));
            }
        }
        next
    }

    fn process_timestamp(&mut self, t: u64, lines: &mut Vec<String>) {
        let mut done: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.state {
                State::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        done.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in done {
            let attempt = match self.entries[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let ok = attempt > self.entries[i].job.failures;
            if ok {
                self.entries[i].state = State::Succeeded;
                lines.push(format!("{} finish {} {} ok", t, self.entries[i].job.id, attempt));
            } else {
                lines.push(format!("{} finish {} {} fail", t, self.entries[i].job.id, attempt));
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self.retry_base.checked_shl(shift).unwrap_or(u64::MAX).min(self.retry_cap);
                    let ready = t.saturating_add(delay);
                    self.entries[i].state = State::Waiting {
                        ready_at: ready,
                        attempt,
                    };
                    self.entries[i].remaining = None;
                    lines.push(format!("{} retry {} {} {}", t, self.entries[i].job.id, attempt, ready));
                }
            }
        }

        loop {
            let newly: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| {
                    if !matches!(e.state, State::Waiting { .. }) {
                        return None;
                    }
                    e.dependencies
                        .iter()
                        .find(|&&d| matches!(self.entries[d].state, State::Failed | State::Blocked))
                        .map(|&d| (i, d))
                })
                .map(|(i, _)| i)
                .collect();
            if newly.is_empty() {
                break;
            }
            for i in newly {
                let dep = self.entries[i]
                    .dependencies
                    .iter()
                    .find(|&&d| matches!(self.entries[d].state, State::Failed | State::Blocked))
                    .copied()
                    .unwrap();
                self.entries[i].state = State::Blocked;
                lines.push(format!(
                    "{} skip {} {}",
                    t, self.entries[i].job.id, self.entries[dep].job.id
                ));
            }
        }

        // Skip lines are required in ASCII ID order across the closure.
        if blocked_lines_start > 0 {
            let mut skips = lines.split_off(blocked_lines_start);
            skips.sort_by(|a, b| {
                let id_a = a.split_whitespace().nth(2).unwrap_or("");
                let id_b = b.split_whitespace().nth(2).unwrap_or("");
                id_a.cmp(id_b)
            });
            lines.extend(skips);
        }

            let mut candidates: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| {
                    let State::Waiting { ready_at, .. } = e.state else {
                        return None;
                    };
                    if ready_at > t
                        || !e.dependencies.iter().all(|&d| matches!(self.entries[d].state, State::Succeeded))
                    {
                        return None;
                    }
                    Some(i)
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa).then(a.cmp(&b))
            });

            let mut started = false;
            for candidate in candidates {
                let tag = self.entries[candidate].job.tag.as_deref();
                let running: Vec<usize> = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| matches!(e.state, State::Running { .. }).then_some(i))
                    .collect();
                let tag_busy = tag.is_some_and(|tag| {
                    running.iter().any(|&i| self.entries[i].job.tag.as_deref() == Some(tag))
                });

                if running.len() < self.capacity && !tag_busy {
                    self.start_job(candidate, t, lines, None);
                    started = true;
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&v| {
                        self.entries[v].job.priority.saturating_add(2)
                            <= self.entries[candidate].job.priority
                    })
                    .filter(|&v| {
                        let remaining_running = running.iter().filter(|&&r| r != v).count();

                        let tag_free = tag.is_none()
                            || !running.iter().filter(|&&r| r != v).any(|&r| {
                                self.entries[r].job.tag.as_deref() == tag
                            });
                        slot_free && tag_free
                    })
                    .collect();

                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| self.entries[b].start_order.cmp(&self.entries[a].start_order))
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });

                        State::Running { finish_at, .. } => finish_at - t,
                        _ => 0,
                    };

                        State::Running { attempt, .. } => attempt,
                        _ => 0,

                        ready_at: t,
                        attempt,

                    lines.push(format!(

                        t, victim_id, self.entries[candidate].job.id

                    started = true;
                    break;

            if !started {

            }
        }
        blocked_lines_start = 0;
        let _ = blocked_lines_start;

    fn effective_priority(&self, i: usize, t: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.entries[i].job.priority;
        }
        let ready_at = match self.entries[i].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => t,
        };
        let aged = (t - ready_at) / self.aging_ms;
        (u64::from(self.entries[i].job.priority) + aged).min(9) as u8
    }

    fn start_job(&mut self, i: usize, t: u64, lines: &mut Vec<String>, _unused: Option<()>) {
        let (attempt, remaining) = match self.entries[i].state {
            State::Waiting { attempt, .. } => {
                if let Some(rem) = self.entries[i].remaining.take() {
                    (attempt, rem)
                } else {
                    let next = attempt.saturating_add(1);
                    (next, self.entries[i].job.duration)
                }
            }
            _ => return,
        };
        self.sequence = self.sequence.saturating_add(1);
        self.entries[i].start_order = self.sequence;
        let finish_at = t.saturating_add(remaining);
        self.entries[i].state = State::Running { attempt, finish_at };
        lines.push(format!(
            "{} start {} {} {}",
            t, self.entries[i].job.id, attempt, remaining

    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),

pub struct Pool {
    jobs: Vec<Job>,
    dependencies: Vec<Vec<usize>>,
    runtime: Vec<Runtime>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,

}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_name(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|tag| !valid_name(tag, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(spelling) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(resolved);
        }

        let mut marks = vec![0u8; jobs.len()];
        fn visit(node: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dependency in &deps[node] {
                if !visit(dependency, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let runtime = jobs
            .iter()
            .map(|job| Runtime {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                remaining: None,
                segment_start: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            dependencies,
            runtime,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.jobs
            .iter()
            .position(|job| job.id == id)
            .map(|i| self.runtime[i].state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.first_run {
            0
        } else {
            self.horizon
        };
        let mut process_current = self.first_run;

        loop {
            if process_current {
                self.process_timestamp(time, &mut lines);
            }
            process_current = false;

            let next = self.next_event_after(time);
            match next {
                Some(next_time) if next_time <= until => {
                    time = next_time;
                    process_current = true;
                }
                _ => break,
            }
        }

        self.horizon = until;
        self.first_run = false;
        Ok(lines)
    }

    fn process_timestamp(&mut self, time: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .runtime
            .iter()
            .enumerate()
            .filter_map(|(i, runtime)| match runtime.state {
                State::Running { finish_at, .. } if finish_at == time => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));

        for i in completions {
            let attempt = match self.runtime[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            if attempt > self.jobs[i].failures {
                lines.push(format!("{time} finish {} {attempt} ok", self.jobs[i].id));
                self.runtime[i].state = State::Succeeded;
                self.runtime[i].remaining = None;
            } else {
                lines.push(format!("{time} finish {} {attempt} fail", self.jobs[i].id));
                self.runtime[i].remaining = None;
                if attempt >= self.jobs[i].max_attempts {
                    self.runtime[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt - 1);
                    let delay = self
                        .retry_base
                        .checked_shl(shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready_at = time.saturating_add(delay);
                    lines.push(format!("{time} retry {} {attempt} {ready_at}", self.jobs[i].id));
                    self.runtime[i].state = State::Waiting { ready_at, attempt };
                }
            }
        }

        let mut blocked = Vec::new();
        loop {
            let mut newly = Vec::new();
            for i in 0..self.jobs.len() {
                if !matches!(self.runtime[i].state, State::Waiting { .. }) {
                    continue;
                }
                if let Some(&failed_dep) = self.dependencies[i].iter().find(|&&dep| {
                    matches!(
                        self.runtime[dep].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    newly.push((i, failed_dep));
                }
            }
            if newly.is_empty() {
                break;
            }
            for (i, dep) in newly {
                self.runtime[i].state = State::Blocked;
                blocked.push((i, dep));
            }
        }
        blocked.sort_by(|(a, _), (b, _)| self.jobs[*a].id.cmp(&self.jobs[*b].id));
        for (i, dep) in blocked {
            lines.push(format!(
                "{time} skip {} {}",
                self.jobs[i].id, self.jobs[dep].id
            ));
        }

        self.schedule(time, lines);
    }

    fn schedule(&mut self, time: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = (0..self.jobs.len())
                .filter(|&i| {
                    matches!(
                        self.runtime[i].state,
                        State::Waiting { ready_at, .. }
                            if ready_at <= time
                                && self.dependencies[i].iter().all(|&d| {
                                    matches!(self.runtime[d].state, State::Succeeded)
                                })
                    )
                })
                .collect();
            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut started = false;
            for candidate in candidates {
                let occupied = self.running_indices();
                let slot_free = occupied.len() < self.capacity;
                let tag_free = self.tag_is_free(candidate, &occupied);
                if slot_free && tag_free {
                    self.start(candidate, time, lines, None);
                    started = true;
                    break;
                }

                let mut victims: Vec<usize> = occupied
                    .into_iter()
                    .filter(|&victim| {
                        self.jobs[candidate].priority >= self.jobs[victim].priority.saturating_add(2)
                            && self.can_free_candidate_resources(candidate, victim)
                    })
                    .collect();
                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .priority
                        .cmp(&self.jobs[b].priority)
                        .then_with(|| {
                            self.runtime[b]
                                .segment_start
                                .cmp(&self.runtime[a].segment_start)
                        })
                        .then_with(|| self.jobs[b].id.cmp(&self.jobs[a].id))
                });
                if let Some(victim) = victims.first().copied() {
                    let remaining = match self.runtime[victim].state {
                        State::Running { finish_at, .. } => finish_at - time,
                        _ => 0,
                    };
                    let attempt = match self.runtime[victim].state {
                        State::Running { attempt, .. } => attempt,
                        _ => 0,
                    };
                    lines.push(format!(
                        "{time} preempt {} by {}",
                        self.jobs[victim].id, self.jobs[candidate].id
                    ));
                    self.runtime[victim].state = State::Waiting {
                        ready_at: time,
                        attempt,
                    };
                    self.runtime[victim].remaining = Some(remaining);
                    self.start(candidate, time, lines, None);
                    started = true;
                    break;
                }
            }
            if !started {
                break;
            }
        }
    }

    fn start(&mut self, index: usize, time: u64, lines: &mut Vec<String>, _unused: Option<()>) {
        let (attempt, duration) = match self.runtime[index].state {
            State::Waiting { attempt, .. } => {
                if let Some(remaining) = self.runtime[index].remaining.take() {
                    (attempt, remaining)
                } else {
                    let next_attempt = attempt.saturating_add(1);
                    (next_attempt, self.jobs[index].duration)
                }
            }
            _ => return,
        };
        let finish_at = time.saturating_add(duration);
        lines.push(format!(
            "{time} start {} {attempt} {duration}",
            self.jobs[index].id
        ));
        self.runtime[index].state = State::Running { attempt, finish_at };
        self.runtime[index].segment_start = time;
    }

    fn running_indices(&self) -> Vec<usize> {
        self.runtime
            .iter()
            .enumerate()
            .filter_map(|(i, runtime)| matches!(runtime.state, State::Running { .. }).then_some(i))
            .collect()
    }

    fn tag_is_free(&self, candidate: usize, running: &[usize]) -> bool {
        match self.jobs[candidate].tag.as_deref() {
            None => true,
            Some(tag) => running
                .iter()
                .all(|&i| self.jobs[i].tag.as_deref() != Some(tag)),
        }
    }

    fn can_free_candidate_resources(&self, candidate: usize, victim: usize) -> bool {
        let running = self.running_indices();
        let after = running.len().saturating_sub(1);
        let slot_free = after < self.capacity;
        let tag_free = match self.jobs[candidate].tag.as_deref() {
            None => true,
            Some(tag) => running.iter().all(|&i| {
                i == victim || self.jobs[i].tag.as_deref() != Some(tag)
            }),
        };
        slot_free && tag_free
    }

    fn effective_priority(&self, index: usize, time: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[index].priority;
        }
        let ready_at = match self.runtime[index].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => time,
        };
        let aged = (time - ready_at) / self.aging_ms;
        self.jobs[index]
            .priority
            .saturating_add(aged.min(9) as u8)
            .min(9)
    }

    fn next_event_after(&self, time: u64) -> Option<u64> {
        self.runtime
            .iter()
            .filter_map(|runtime| match runtime.state {
                State::Waiting { ready_at, .. } if ready_at > time => Some(ready_at),
                State::Running { finish_at, .. } if finish_at > time => Some(finish_at),
                _ => None,
            })
            .min()
    }
}

fn valid_name(value: &str, min: usize, max: usize) -> bool {
    let bytes = value.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes.iter().all(|&byte| {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-'
        })
}

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut states = self.states.clone();
        let mut lines = Vec::new();
        let mut current = if self.has_run { self.horizon } else { 0 };
        let mut first_phase = !self.has_run;

        loop {
            if first_phase || current > self.horizon {
                self.process_timestamp(current, &mut states, &mut lines);
                first_phase = false;
            } else if self.has_run && current == self.horizon {
                // The prior horizon was already processed. Continue only if
                // a strictly later event exists.
            }

            let next = next_event(&states, current);
            match next {
                Some(t) if t <= until => {
                    current = t;
                    self.process_timestamp(current, &mut states, &mut lines);
                }
                _ => break,
            }
        }

        self.states = states;
        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let idx = self
            .jobs
            .iter()
            .position(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;
        Ok(match self.states[idx] {
            InternalState::Waiting {
                ready_at, attempt, ..
            } => State::Waiting { ready_at, attempt },
            InternalState::Running {
                attempt, finish_at, ..
            } => State::Running { attempt, finish_at },
            InternalState::Succeeded => State::Succeeded,
            InternalState::Failed => State::Failed,
            InternalState::Blocked => State::Blocked,
        })
    }

    fn process_timestamp(
        &self,
        time: u64,
        states: &mut [InternalState],
        lines: &mut Vec<String>,
    ) {
        let mut completed: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| match state {
                InternalState::Running { finish_at, .. } if *finish_at == time => Some(i),
                _ => None,
            })
            .collect();
        completed.sort_by(|&a, &b| self.jobs[a].job.id.cmp(&self.jobs[b].job.id));

        for &i in &completed {
            let attempt = match states[i] {
                InternalState::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let ok = attempt > self.jobs[i].job.failures;
            lines.push(format!(
                "{} finish {} {} {}",
                time,
                self.jobs[i].job.id,
                attempt,
                if ok { "ok" } else { "fail" }
            ));
            if ok {
                states[i] = InternalState::Succeeded;
            } else if attempt >= self.jobs[i].job.max_attempts {
                states[i] = InternalState::Failed;
            } else {
                let shift = u32::from(attempt.saturating_sub(1));
                let delay = self
                    .retry_base
                    .checked_shl(shift)
                    .unwrap_or(u64::MAX)
                    .min(self.retry_cap);
                let ready_at = time.saturating_add(delay);
                states[i] = InternalState::Waiting {
                    ready_at,
                    attempt,
                    remaining: None,
                };
                lines.push(format!(
                    "{} retry {} {} {}",
                    time, self.jobs[i].job.id, attempt, ready_at
                ));
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..states.len() {
                if matches!(states[i], InternalState::Waiting { .. })
                    && self.jobs[i].dependencies.iter().any(|&d| {
                        matches!(
                            states[d],
                            InternalState::Failed | InternalState::Blocked
                        )
                    })
                {
                    states[i] = InternalState::Blocked;
                    newly_blocked.push(i);
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|&a, &b| self.jobs[a].job.id.cmp(&self.jobs[b].job.id));
        for &i in &newly_blocked {
            let dep = self.jobs[i]
                .dependencies
                .iter()
                .find(|&&d| {
                    matches!(
                        states[d],
                        InternalState::Failed | InternalState::Blocked
                    )
                })
                .copied()
                .unwrap_or(i);
            lines.push(format!(
                "{} skip {} {}",
                time, self.jobs[i].job.id, self.jobs[dep].job.id
            ));
        }

        self.schedule(time, states, lines);
    }

    fn schedule(&self, time: u64, states: &mut [InternalState], lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = (0..states.len())
                .filter(|&i| {
                    matches!(
                        states[i],
                        InternalState::Waiting { ready_at, .. } if ready_at <= time
                    ) && self.jobs[i]
                        .dependencies
                        .iter()
                        .all(|&d| matches!(states[d], InternalState::Succeeded))
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time, states);
                let pb = self.effective_priority(b, time, states);
                pb.cmp(&pa).then(a.cmp(&b))
            });

            let running_count = states
                .iter()
                .filter(|state| matches!(state, InternalState::Running { .. }))
                .count();
            let mut selected = None;

            for candidate in candidates {
                let tag = self.jobs[candidate].job.tag.as_deref();
                let tag_busy = tag.is_some_and(|t| {
                    states.iter().enumerate().any(|(i, state)| {
                        i != candidate
                            && matches!(state, InternalState::Running { .. })
                            && self.jobs[i].job.tag.as_deref() == Some(t)
                    })
                });
                if running_count < self.capacity && !tag_busy {
                    selected = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = states
                    .iter()
                    .enumerate()
                    .filter_map(|(i, state)| {
                        if !matches!(state, InternalState::Running { .. })
                            || self.jobs[candidate].job.priority
                                < self.jobs[i].job.priority.saturating_add(2)
                        {
                            return None;
                        }
                        let after_count = running_count - 1;
                        let slot_free = after_count < self.capacity;
                        let tag_free = tag.is_none()
                            || !states.iter().enumerate().any(|(j, other)| {
                                j != i
                                    && matches!(other, InternalState::Running { .. })
                                    && self.jobs[j].job.tag.as_deref() == tag
                            });
                        (slot_free && tag_free).then_some(i)
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .job
                        .priority
                        .cmp(&self.jobs[b].job.priority)
                        .then_with(|| {
                            let sa = match states[a] {
                                InternalState::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            let sb = match states[b] {
                                InternalState::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            sb.cmp(&sa)
                        })
                        .then_with(|| self.jobs[b].job.id.cmp(&self.jobs[a].job.id))
                });

                if let Some(&victim) = victims.first() {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(victim) = victim {
                let (attempt, finish_at) = match states[victim] {
                    InternalState::Running {
                        attempt, finish_at, ..
                    } => (attempt, finish_at),
                    _ => continue,
                };
                let remaining = finish_at.saturating_sub(time);
                lines.push(format!(
                    "{} preempt {} by {}",
                    time, self.jobs[victim].job.id, self.jobs[candidate].job.id
                ));
                states[victim] = InternalState::Waiting {
                    ready_at: time,
                    attempt,
                    remaining: Some(remaining),
                };
            }

            let (old_attempt, remaining) = match states[candidate] {
                InternalState::Waiting {
                    attempt, remaining, ..
                } => (attempt, remaining),
                _ => continue,
            };
            let (attempt, duration) = match remaining {
                Some(rem) => (old_attempt, rem),
                None => (old_attempt.saturating_add(1), self.jobs[candidate].job.duration),
            };
            let finish_at = time.saturating_add(duration);
            lines.push(format!(
                "{} start {} {} {}",
                time, self.jobs[candidate].job.id, attempt, duration
            ));
            states[candidate] = InternalState::Running {
                attempt,
                finish_at,
                started_at: time,
            };
        }
    }

    fn effective_priority(&self, index: usize, time: u64, states: &[InternalState]) -> u8 {
        let base = self.jobs[index].job.priority;
        if self.aging_ms == 0 {
            return base;
        }
        let ready_at = match states[index] {
            InternalState::Waiting { ready_at, .. } => ready_at,
            _ => time,
        };
        let increase = ((time.saturating_sub(ready_at)) / self.aging_ms).min(9) as u8;
        base.saturating_add(increase).min(9)
    }
}

fn valid_name(value: &str, min: usize, max: usize) -> bool {
    let bytes = value.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes.iter().all(|b| {
            b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-'
        })
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
        if marks[node] == 1 {
            return true;
        }
        if marks[node] == 2 {
            return false;
        }
        marks[node] = 1;
        for &dep in &deps[node] {
            if visit(dep, deps, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    (0..dependencies.len()).any(|i| visit(i, dependencies, &mut marks))
}

fn next_event(states: &[InternalState], current: u64) -> Option<u64> {
    states
        .iter()
        .filter_map(|state| match state {
            InternalState::Waiting { ready_at, .. } if *ready_at > current => Some(*ready_at),
            InternalState::Running { finish_at, .. } if *finish_at > current => Some(*finish_at),
            _ => None,
        })
        .min()
}
