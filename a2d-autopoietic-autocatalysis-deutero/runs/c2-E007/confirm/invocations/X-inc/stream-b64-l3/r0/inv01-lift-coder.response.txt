#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_byte(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn skipped(byte: u8, whitespace: Whitespace) -> bool {
    matches!(whitespace, Whitespace::IgnoreAscii)
        && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            closed: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut bytes = Vec::with_capacity(self.tail.len() + input.len());
        bytes.extend_from_slice(&self.tail);
        bytes.extend_from_slice(input);
        let complete_len = bytes.len() / 3 * 3;
        self.tail.clear();
        self.tail.extend_from_slice(&bytes[complete_len..]);

        let mut symbols = Vec::with_capacity(complete_len / 3 * 4);
        for chunk in bytes[..complete_len].chunks_exact(3) {
            let a = chunk[0];
            let b = chunk[1];
            let c = chunk[2];
            symbols.push(alphabet_byte(a >> 2, self.options.alphabet));
            symbols.push(alphabet_byte(((a & 3) << 4) | (b >> 4), self.options.alphabet));
            symbols.push(alphabet_byte(((b & 15) << 2) | (c >> 6), self.options.alphabet));
            symbols.push(alphabet_byte(c & 63, self.options.alphabet));
        }
        Ok(format_symbols(&symbols, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;

        let mut symbols = Vec::new();
        match self.tail.len() {
            0 => {}
            1 => {
                let a = self.tail[0];
                symbols.push(alphabet_byte(a >> 2, self.options.alphabet));
                symbols.push(alphabet_byte((a & 3) << 4, self.options.alphabet));
                if self.options.padding != Padding::Forbidden {
                    symbols.extend_from_slice(b"==");
                }
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                symbols.push(alphabet_byte(a >> 2, self.options.alphabet));
                symbols.push(alphabet_byte(((a & 3) << 4) | (b >> 4), self.options.alphabet));
                symbols.push(alphabet_byte((b & 15) << 2, self.options.alphabet));
                if self.options.padding != Padding::Forbidden {
                    symbols.push(b'=');
                }
            }
            _ => unreachable!(),
        }
        self.tail.clear();
        Ok(format_symbols(&symbols, self.options.whitespace))
    }
}

fn format_symbols(symbols: &[u8], whitespace: Whitespace) -> String {
    let mut output = Vec::with_capacity(symbols.len() + 8);
    if let Whitespace::Wrapped { width, ending } = whitespace {
        for (i, &byte) in symbols.iter().enumerate() {
            if i > 0 && i % width == 0 {
                match ending {
                    LineEnding::Lf => output.push(b'\n'),
                    LineEnding::CrLf => output.extend_from_slice(b"\r\n"),
                }
            }
            output.push(byte);
        }
    } else {
        output.extend_from_slice(symbols);
    }
    String::from_utf8(output).expect("base64 output is ASCII")
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum WrapState {
    Row,
    NeedSeparator { first_offset: usize },
    NeedLf { cr_offset: usize },
    NeedSymbol { first_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: Vec<Symbol>,
    pending_second_pad: Option<usize>,
    terminal_padding: bool,
    offset: usize,
    symbols_in_row: usize,
    wrap_state: WrapState,
    closed: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            pending_second_pad: None,
            terminal_padding: false,
            offset: 0,
            symbols_in_row: 0,
            wrap_state: WrapState::Row,
            closed: false,
            sticky_error: None,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut output) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let result = self.finish_inner();
        match result {
            Ok(output) => {
                self.closed = true;
                Ok(output)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }

    fn process_byte(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            match self.wrap_state {
                WrapState::NeedSeparator { first_offset } => match ending {
                    LineEnding::Lf => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = WrapState::NeedSymbol { first_offset };
                        return Ok(());
                    }
                    LineEnding::CrLf => {
                        if byte != b'\r' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = WrapState::NeedLf { cr_offset: offset };
                        return Ok(());
                    }
                },
                WrapState::NeedLf { cr_offset } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.wrap_state = WrapState::NeedSymbol {
                        first_offset: cr_offset,
                    };
                    return Ok(());
                }
                WrapState::NeedSymbol { first_offset } => {
                    if byte == b'\r' || byte == b'\n' || skipped(byte, self.options.whitespace) {
                        return Err(Error::BadWrap { offset });
                    }
                    self.wrap_state = WrapState::Row;
                    self.symbols_in_row = 0;
                    let _ = first_offset;
                }
                WrapState::Row => {
                    if self.symbols_in_row == width {
                        self.wrap_state = WrapState::NeedSeparator {
                            first_offset: offset,
                        };
                        return self.process_byte(byte, offset, output);
                    }
                    if byte == b'\r' || byte == b'\n' || skipped(byte, self.options.whitespace) {
                        return Err(Error::BadWrap { offset });
                    }
                }
            }
            self.symbols_in_row += 1;
        } else if skipped(byte, self.options.whitespace) {
            return Ok(());
        }

        self.process_symbol(byte, offset, output)
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if let Some(pending_offset) = self.pending_second_pad {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_second_pad = None;
            self.terminal_padding = true;
            self.group.clear();
            let _ = pending_offset;
            return Ok(());
        }

        match self.group.len() {
            0 | 1 => {
                if byte == b'=' {
                    return Err(Error::BadPadding { offset });
                }
                if digit(byte, self.options.alphabet).is_none() {
                    return Err(Error::InvalidByte { offset, byte });
                }
                self.group.push(Symbol { byte, offset });
            }
            2 => {
                if byte == b'=' {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.pending_second_pad = Some(offset);
                } else if digit(byte, self.options.alphabet).is_none() {
                    return Err(Error::InvalidByte { offset, byte });
                } else {
                    self.group.push(Symbol { byte, offset });
                }
            }
            3 => {
                if byte == b'=' {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.emit_group(2, output);
                    self.group.clear();
                    self.terminal_padding = true;
                } else if digit(byte, self.options.alphabet).is_none() {
                    return Err(Error::InvalidByte { offset, byte });
                } else {
                    self.group.push(Symbol { byte, offset });
                    self.emit_group(3, output);
                    self.group.clear();
                }
            }
            _ => unreachable!(),
        }
        Ok(())
    }

    fn emit_group(&self, count: usize, output: &mut Vec<u8>) {
        let a = digit(self.group[0].byte, self.options.alphabet).unwrap();
        let b = digit(self.group[1].byte, self.options.alphabet).unwrap();
        let c = if count >= 3 {
            digit(self.group[2].byte, self.options.alphabet).unwrap()
        } else {
            0
        };
        let d = if count >= 4 {
            digit(self.group[3].byte, self.options.alphabet).unwrap()
        } else {
            0
        };

        output.push((a << 2) | (b >> 4));
        if count >= 3 {
            output.push((b << 4) | (c >> 2));
        }
        if count >= 4 {
            output.push((c << 6) | d);
        }
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        if let WrapState::NeedLf { cr_offset } = self.wrap_state {
            return Err(Error::BadWrap { offset: cr_offset });
        }
        if let WrapState::NeedSymbol { first_offset } = self.wrap_state {
            return Err(Error::BadWrap {
                offset: first_offset,
            });
        }
        if matches!(self.wrap_state, WrapState::NeedSeparator { .. }) {
            // EOF after a full row is valid.
            self.wrap_state = WrapState::Row;
        }

        if let Some(offset) = self.pending_second_pad {
            return Err(Error::BadPadding { offset });
        }

        let mut output = Vec::new();
        if self.terminal_padding {
            return Ok(output);
        }

        match self.group.len() {
            0 => Ok(output),
            1 => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
            2 | 3 if self.options.padding != Padding::Required => {
                let count = self.group.len();
                self.emit_group(count, &mut output);
                self.group.clear();
                Ok(output)
            }
            _ => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
        }
    }
}
