#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/",
        Alphabet::UrlSafe => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_",
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate_options(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    column: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            buffered: Vec::with_capacity(2),
            column: 0,
            finished: false,
        })
    }

    fn emit_symbol(&mut self, output: &mut String, byte: u8) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            if self.column == width {
                match ending {
                    LineEnding::Lf => output.push('\n'),
                    LineEnding::CrLf => output.push_str("\r\n"),
                }
                self.column = 0;
            }
            self.column += 1;
        }
        output.push(byte as char);
    }

    fn emit_group(&mut self, bytes: &[u8], output: &mut String, padded: bool) {
        let alphabet = alphabet_bytes(self.options.alphabet);
        let a = bytes[0];
        let b = if bytes.len() > 1 { bytes[1] } else { 0 };
        let c = if bytes.len() > 2 { bytes[2] } else { 0 };

        let indices = [
            (a >> 2) as usize,
            (((a & 0x03) << 4) | (b >> 4)) as usize,
            (((b & 0x0f) << 2) | (c >> 6)) as usize,
            (c & 0x3f) as usize,
        ];

        self.emit_symbol(output, alphabet[indices[0]]);
        self.emit_symbol(output, alphabet[indices[1]]);
        if bytes.len() > 1 {
            self.emit_symbol(output, alphabet[indices[2]]);
        } else if padded {
            self.emit_symbol(output, b'=');
        }
        if bytes.len() > 2 {
            self.emit_symbol(output, alphabet[indices[3]]);
        } else if padded {
            self.emit_symbol(output, b'=');
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut offset = 0;

        if !self.buffered.is_empty() {
            while self.buffered.len() < 3 && offset < input.len() {
                self.buffered.push(input[offset]);
                offset += 1;
            }
            if self.buffered.len() == 3 {
                let group = self.buffered.clone();
                self.emit_group(&group, &mut output, false);
                self.buffered.clear();
            }
        }

        while offset + 3 <= input.len() {
            self.emit_group(&input[offset..offset + 3], &mut output, false);
            offset += 3;
        }
        self.buffered.extend_from_slice(&input[offset..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        if !self.buffered.is_empty() {
            let padded = self.options.padding != Padding::Forbidden;
            let tail = self.buffered.clone();
            self.emit_group(&tail, &mut output, padded);
        }
        self.buffered.clear();
        self.finished = true;
        Ok(output)
    }
}

#[derive(Clone, Copy)]
enum FramingState {
    Ready,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    symbols: Vec<(u8, usize)>,
    physical_offset: usize,
    symbols_in_row: usize,
    framing: FramingState,
    padded_finished: bool,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            symbols: Vec::with_capacity(4),
            physical_offset: 0,
            symbols_in_row: 0,
            framing: FramingState::Ready,
            padded_finished: false,
            finished: false,
            sticky_error: None,
        })
    }

    fn is_ignored_ascii(byte: u8) -> bool {
        matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn is_wrapped(&self) -> Option<(usize, LineEnding)> {
        match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => Some((width, ending)),
            _ => None,
        }
    }

    fn decode_group(&mut self) -> Result<Vec<u8>, Error> {
        let group = &self.symbols;
        let offset = group[0].1;
        let mut values = [0u8; 4];
        let alphabet = alphabet_bytes(self.options.alphabet);

        for (index, (byte, byte_offset)) in group.iter().enumerate() {
            if *byte == b'=' {
                continue;
            }
            let Some(value) = alphabet.iter().position(|candidate| candidate == byte) else {
                return Err(Error::InvalidByte {
                    offset: *byte_offset,
                    byte: *byte,
                });
            };
            values[index] = value as u8;
        }

        let mut output = Vec::with_capacity(3);
        output.push((values[0] << 2) | (values[1] >> 4));
        if group[2].0 != b'=' {
            output.push((values[1] << 4) | (values[2] >> 2));
        }
        if group[3].0 != b'=' {
            output.push((values[2] << 6) | values[3]);
        }

        if group[2].0 == b'=' || group[3].0 == b'=' {
            self.padded_finished = true;
        }
        self.symbols.clear();
        let _ = offset;
        Ok(output)
    }

    fn scan_symbol(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.padded_finished {
            return Err(Error::BadPadding { offset });
        }

        let position = self.symbols.len();
        if position <= 1 && byte == b'=' {
            return Err(Error::BadPadding { offset });
        }
        if position == 2 && self.symbols[2].0 == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        if byte != b'=' && !alphabet_bytes(self.options.alphabet).contains(&byte) {
            return Err(Error::InvalidByte { offset, byte });
        }

        if byte == b'=' && self.options.padding == Padding::Forbidden {
            return Err(Error::BadPadding { offset });
        }

        self.symbols.push((byte, offset));
        if self.symbols.len() == 4 {
            let decoded = self.decode_group()?;
            output.extend(decoded);
        }
        Ok(())
    }

    fn scan_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.options.whitespace == Whitespace::IgnoreAscii && Self::is_ignored_ascii(byte) {
            return Ok(());
        }

        if let Some((width, ending)) = self.is_wrapped() {
            match self.framing {
                FramingState::NeedLf { cr_offset } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.framing = FramingState::NeedSymbol {
                        separator_offset: cr_offset,
                    };
                    return Ok(());
                }
                FramingState::NeedSymbol { .. } => {
                    self.framing = FramingState::Ready;
                    self.symbols_in_row = 0;
                }
                FramingState::Ready => {
                    if self.symbols_in_row == width {
                        match ending {
                            LineEnding::Lf => {
                                if byte != b'\n' {
                                    return Err(Error::BadWrap { offset });
                                }
                                self.framing = FramingState::NeedSymbol {
                                    separator_offset: offset,
                                };
                                return Ok(());
                            }
                            LineEnding::CrLf => {
                                if byte != b'\r' {
                                    return Err(Error::BadWrap { offset });
                                }
                                self.framing = FramingState::NeedLf { cr_offset: offset };
                                return Ok(());
                            }
                        }
                    }

                    if Self::is_ignored_ascii(byte) {
                        return Err(Error::BadWrap { offset });
                    }
                }
            }

            if matches!(self.framing, FramingState::Ready | FramingState::NeedSymbol { .. }) {
                self.symbols_in_row += 1;
            }
        }

        self.scan_symbol(byte, offset, output)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.physical_offset;
            self.physical_offset += 1;
            if let Err(error) = self.scan_byte(byte, offset, &mut output) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some((_, ending)) = self.is_wrapped() {
            match self.framing {
                FramingState::NeedLf { cr_offset } => {
                    let error = Error::BadWrap { offset: cr_offset };
                    self.sticky_error = Some(error.clone());
                    return Err(error);
                }
                FramingState::NeedSymbol { separator_offset } => {
                    let error = Error::BadWrap {
                        offset: separator_offset,
                    };
                    self.sticky_error = Some(error.clone());
                    return Err(error);
                }
                FramingState::Ready => {}
            }
            let _ = ending;
        }

        if self.symbols.len() == 3 && self.symbols[2].0 == b'=' {
            let error = Error::BadPadding {
                offset: self.symbols[2].1,
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let residual_len = self.symbols.len();
        let mut output = Vec::new();
        match residual_len {
            0 => {}
            1 => {
                let error = Error::Truncated {
                    offset: self.symbols[0].1,
                };
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
            2 | 3 => {
                if self.options.padding == Padding::Required {
                    let error = Error::Truncated {
                        offset: self.symbols[0].1,
                    };
                    self.sticky_error = Some(error.clone());
                    return Err(error);
                }

                let alphabet = alphabet_bytes(self.options.alphabet);
                let mut values = [0u8; 3];
                for index in 0..residual_len {
                    let (byte, offset) = self.symbols[index];
                    let Some(value) = alphabet.iter().position(|candidate| *candidate == byte) else {
                        let error = Error::InvalidByte { offset, byte };
                        self.sticky_error = Some(error.clone());
                        return Err(error);
                    };
                    values[index] = value as u8;
                }
                output.push((values[0] << 2) | (values[1] >> 4));
                if residual_len == 3 {
                    output.push((values[1] << 4) | (values[2] >> 2));
                }
            }
            _ => unreachable!(),
        }

        self.symbols.clear();
        self.finished = true;
        Ok(output)
    }
}
