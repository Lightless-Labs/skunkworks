#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, &byte) in input.as_bytes().iter().enumerate() {
        if !(matches!(byte, b' '..=b'~' | b'\t' | b'\n' | b'\r')) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c| matches!(c, ' ' | '\t' | '\n' | '\r')).filter(|word| !word.is_empty()) {
        let mut remaining = word;

        while !remaining.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if remaining.len() <= width {
                let separator_len = usize::from(!line.is_empty());
                if line.len() + separator_len + remaining.len() <= width {
                    if separator_len == 1 {
                        line.push(' ');
                    }
                    line.push_str(remaining);
                    break;
                }

                if !line.is_empty() {
                    lines.push(std::mem::take(&mut line));
                    continue;
                }
            }

            if line.is_empty() {
                let take = width.min(remaining.len());
                line.push_str(&remaining[..take]);
                remaining = &remaining[take..];
                if !remaining.is_empty() {
                    lines.push(std::mem::take(&mut line));
                }
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&remaining[..available]);
                    remaining = &remaining[available..];
                    lines.push(std::mem::take(&mut line));
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
