#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(matches!(byte, b' '..=b'~' | b'\t' | b'\n' | b'\r')) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let is_separator = |byte: u8| matches!(byte, b' ' | b'\t' | b'\n' | b'\r');
    let mut lines = Vec::new();
    let mut line = String::new();
    let mut position = 0;

    while position < bytes.len() {
        while position < bytes.len() && is_separator(bytes[position]) {
            position += 1;
        }
        if position == bytes.len() {
            break;
        }

        let start = position;
        while position < bytes.len() && !is_separator(bytes[position]) {
            position += 1;
        }
        let word = &input[start..position];
        let mut consumed = 0;

        while consumed < word.len() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = &word[consumed..];
            let remaining_len = remaining.len();

            if remaining_len <= width {
                let needed = remaining_len + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(remaining);
                    consumed = word.len();
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                line.push_str(&remaining[..width]);
                lines.push(std::mem::take(&mut line));
                consumed += width;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&remaining[..available]);
                    lines.push(std::mem::take(&mut line));
                    consumed += available;
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
