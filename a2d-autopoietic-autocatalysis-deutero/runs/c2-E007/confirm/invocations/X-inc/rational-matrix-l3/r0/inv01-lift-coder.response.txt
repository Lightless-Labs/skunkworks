#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let n = num as i128;
        let mut d = den as i128;
        let mut n = n;
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        make_i128(self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
                  self.den as i128 * rhs.den as i128)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        make_i128(self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
                  self.den as i128 * rhs.den as i128)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        make_i128(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        make_i128(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::new(self.num.checked_neg().ok_or(RationalError::Overflow)?, self.den)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut i = 0;
        if b[0] == b'-' { i = 1; }
        let num_start = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == num_start {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < b.len() {
            if b[i] != b'/' {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            i += 1;
            den_start = Some(i);
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == den_start.unwrap() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            den_end = Some(i);
            if i < b.len() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
        }

        let err = |at, kind| RationalError::Parse { at, kind };
        let neg = b[0] == b'-';
        if num_end - num_start >= 2 && b[num_start] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if neg && num_end - num_start == 1 && b[num_start] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let (Some(s), Some(e)) = (den_start, den_end) {
            if e - s >= 2 && b[s] == b'0' {
                return Err(err(s + 1, ParseKind::NonCanonical));
            }
        }

        let nmag = parse_magnitude(&b[num_start..num_end], if neg { 1u128 << 63 } else { i64::MAX as u128 })
            .ok_or_else(|| err(0, ParseKind::OutOfRange))?;
        let n = if neg {
            if nmag == (1u128 << 63) { i64::MIN } else { -(nmag as i64) }
        } else { nmag as i64 };

        let mut d = 1u64;
        if let (Some(s), Some(e)) = (den_start, den_end) {
            d = parse_magnitude(&b[s..e], i64::MAX as u128)
                .ok_or_else(|| err(s, ParseKind::OutOfRange))? as u64;
            if d == 0 { return Err(err(s, ParseKind::ZeroDenominator)); }
            if d == 1 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            if n == 0 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            if gcd_u128((n as i128).unsigned_abs(), d as u128) > 1 {
                return Err(err(s - 1, ParseKind::NonCanonical));
            }
        }
        Ok(Self { num: n, den: d as i64 })
    }
}

impl std::fmt::Display for Rational {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) }
        else { write!(f, "{}/{}", self.num, self.den) }
    }
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 { let r = a % b; a = b; b = r; }
    a
}

fn make_i128(mut n: i128, mut d: i128) -> Result<Rational, RationalError> {
    if d == 0 { return Err(RationalError::ZeroDenominator); }
    if d < 0 { n = -n; d = -d; }
    if n == 0 { return Ok(Rational { num: 0, den: 1 }); }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

fn parse_magnitude(bytes: &[u8], limit: u128) -> Option<u128> {
    let mut v = 0u128;
    for &c in bytes {
        let digit = (c - b'0') as u128;
        v = v.checked_mul(10)?.checked_add(digit)?;
        if v > limit { return None; }
    }
    Some(v)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = Rational { num: 1, den: 1 };
        }
        let aug = Matrix { rows: n, cols: width, data };
        let red = eliminate(aug, n, width)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                out.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix { rows: n, cols: n, data: out })
    }
}

fn eliminate(mut matrix: Matrix, pivot_cols: usize, width: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let mut p = 0usize;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();

    for col in 0..pivot_cols {
        if p == rows { break; }
        let mut best: Option<usize> = None;
        for r in p..rows {
            let v = matrix.data[r * width + col];
            if v.num == 0 { continue; }
            let key = (v.den, (v.num as i128).unsigned_abs(), r);
            if best.map_or(true, |br| {
                let b = matrix.data[br * width + col];
                key < (b.den, (b.num as i128).unsigned_abs(), br)
            }) {
                best = Some(r);
            }
        }
        let Some(q) = best else { continue };
        if q != p {
            for c in 0..width {
                matrix.data.swap(p * width + c, q * width + c);
            }
            swaps.push((p, q));
        }
        let pivot = matrix.data[p * width + col];
        for c in 0..width {
            let idx = p * width + c;
            matrix.data[idx] = matrix.data[idx].div(&pivot)
                .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
        }
        for r in 0..rows {
            if r == p { continue; }
            let factor = matrix.data[r * width + col];
            if factor.num == 0 { continue; }
            for c in 0..width {
                let product = factor.mul(&matrix.data[p * width + c])
                    .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                let idx = r * width + c;
                matrix.data[idx] = matrix.data[idx].sub(&product)
                    .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
            }
        }
        pivots.push(col);
        p += 1;
    }
    Ok(Reduction { matrix, pivots, swaps })
}
