#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32, usize) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_atom(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<(u32, bool), ParseError> {
    if start >= end {
        return Err(syntax_error(field, start));
    }

    let part = &bytes[start..end];
    if part.iter().all(u8::is_ascii_digit) {
        if part.len() > 1 && part[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset: start });
        }
        let mut value = 0u32;
        for &byte in part {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add(u32::from(byte - b'0')))
                .ok_or(ParseError::NumberOverflow { field, offset: start })?;
        }
        let (min, max, _) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset: start });
        }
        return Ok((value, true));
    }

    if part.iter().all(u8::is_ascii_alphabetic) {
        if let Some(allowed) = names(field) {
            let upper: Vec<u8> = part.iter().map(u8::to_ascii_uppercase).collect();
            if let Some(index) = allowed
                .iter()
                .position(|name| name.as_bytes() == upper.as_slice())
            {
                return Ok((index as u32 + bounds(field).0, false));
            }
        }
    }

    Err(syntax_error(field, start))
}

fn parse_step(bytes: &[u8], start: usize, end: usize, field: Field) -> Result<u32, ParseError> {
    if start >= end || !bytes[start..end].iter().all(u8::is_ascii_digit) {
        return Err(syntax_error(field, start));
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(u32::from(byte - b'0')))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    let (_, _, width) = bounds(field);
    if value == 0 || value > width as u32 {
        return Err(ParseError::BadStep { field, offset: start });
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(syntax_error(field, item_start));
    }

    let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
        (item_start + slash, Some(item_start + slash + 1))
    } else {
        (item_end, None)
    };

    let base_start = item_start;
    let base = &bytes[base_start..base_end];
    if base.is_empty() {
        return Err(syntax_error(field, item_start));
    }

    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if hyphens.len() > 1 || (is_star && !hyphens.is_empty()) {
        return Err(syntax_error(field, item_start));
    }

    let (start_value, end_value, range_end_offset) = if is_star {
        let (min, max, _) = bounds(field);
        (min, max, None)
    } else if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(syntax_error(field, item_start));
        }
        let left_end = base_start + hyphen;
        let right_start = left_end + 1;
        let (start_value, _) = parse_atom(bytes, base_start, left_end, field)?;
        let (end_value, _) = parse_atom(bytes, right_start, base_end, field)?;
        if start_value > end_value {
            return Err(ParseError::ReversedRange {
                field,
                offset: right_start,
            });
        }
        (start_value, end_value, Some(right_start))
    } else {
        let (value, _) = parse_atom(bytes, base_start, base_end, field)?;
        let end = if step_start.is_some() {
            bounds(field).1
        } else {
            value
        };
        (value, end, None)
    };

    let step = match step_start {
        Some(start) => parse_step(bytes, start, item_end, field)?,
        None => 1,
    };

    let mut value = start_value;
    loop {
        selected[value as usize] = true;
        if value > end_value.saturating_sub(step) {
            break;
        }
        value += step;
    }

    let _ = range_end_offset;
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, _, width) = bounds(field);
    let mut selected = vec![false; width];
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(bytes, start, index, field, &mut selected)?;
            start = index + 1;
        }
    }
    parse_item(bytes, start, bytes.len(), field, &mut selected)?;
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut parsed = Vec::with_capacity(5);
    for (index, piece) in pieces.iter().enumerate() {
        parsed.push(parse_field(piece, FIELDS[index])?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().unwrap(),
        day_of_month_unrestricted: pieces[2] == "*",
        day_of_week_unrestricted: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _, _) = bounds(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &is_selected)| {
                is_selected.then_some((i as u32 + min) as u8)
            })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _, _) = bounds(field);
        let position = u32::from(value).saturating_sub(min) as usize;
        self.selected[index].get(position).copied().unwrap_or(false)
    }

    fn date_matches(&self, at: CivilDateTime, weekday: u8) -> bool {
        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday);

        match (self.day_of_month_unrestricted, self.day_of_week_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if at.day == days_in_month(at.year, at.month) => dom && dow,
            (false, false) => dom || dow,
        }
    }

    fn validate(at: CivilDateTime) -> Result<(), DateError> {
        if !(2000..=2399).contains(&at.year) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Year,
            });
        }
        if !(1..=12).contains(&at.month) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Month,
            });
        }
        if at.day == 0 || at.day > days_in_month(at.year, at.month) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Day,
            });
        }
        if at.hour > 23 {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Hour,
            });
        }
        if at.minute > 59 {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Minute,
            });
        }
        Ok(())
    }

    fn weekday(at: CivilDateTime) -> u8 {
        let mut days = 0u32;
        for year in 2000..at.year {
            days += if is_leap(year) { 366 } else { 365 };
        }
        for month in 1..at.month {
            days += u32::from(days_in_month(at.year, month));
        }
        days += u32::from(at.day - 1);
        ((6 + days) % 7) as u8
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        Self::validate(at)?;
        let weekday = Self::weekday(at);
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.date_matches(at, weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        Self::validate(after)?;
        let mut current = after;

        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    if current.day < days_in_month(current.year, current.month) {
                        current.day += 1;
                    } else {
                        current.day = 1;
                        if current.month < 12 {
                            current.month += 1;
                        } else {
                            current.month = 1;
                            current.year += 1;
                        }
                    }
                }
            }

            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}
