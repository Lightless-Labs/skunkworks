#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<u8>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Clone, Copy)]
struct Domain {
    min: u8,
    max: u8,
}

impl Domain {
    fn width(self) -> u8 {
        self.max - self.min + 1
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u8> {
    let names: &[(&[u8], u8)] = match field {
        Field::Month => &[
            (b"JAN", 1), (b"FEB", 2), (b"MAR", 3), (b"APR", 4),
            (b"MAY", 5), (b"JUN", 6), (b"JUL", 7), (b"AUG", 8),
            (b"SEP", 9), (b"OCT", 10), (b"NOV", 11), (b"DEC", 12),
        ],
        Field::DayOfWeek => &[
            (b"SUN", 0), (b"MON", 1), (b"TUE", 2), (b"WED", 3),
            (b"THU", 4), (b"FRI", 5), (b"SAT", 6),
        ],
        _ => return None,
    };
    names.iter().find_map(|(name, value)| {
        (bytes.len() == name.len()
            && bytes.iter().zip(name.iter()).all(|(a, b)| a.to_ascii_uppercase() == *b))
            .then_some(*value)
    })
}

fn parse_atom(
    field: Field,
    bytes: &[u8],
    offset: usize,
    is_step: bool,
) -> Result<u8, ParseError> {
    if bytes.is_empty() {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(parse_error(field, offset, 1));
        }
        let mut value: u32 = 0;
        for &b in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or_else(|| parse_error(field, offset, 2))?;
        }
        if is_step {
            if value < 1 || value > domain(field).width() as u32 {
                return Err(parse_error(field, offset, 5));
            }
        } else {
            let d = domain(field);
            if value < d.min as u32 || value > d.max as u32 {
                return Err(parse_error(field, offset, 3));
            }
        }
        return Ok(value as u8);
    }
    if !is_step {
        if let Some(value) = name_value(field, bytes) {
            return Ok(value);
        }
    }
    Err(parse_error(field, offset, 0))
}

fn parse_field(field: Field, text: &str) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        let legal = b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b',');
        if !legal {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; 256];
    let mut item_start = 0usize;

    for item_end in (0..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if item_start == item_end {
            return Err(ParseError::EmptyItem { field, offset: item_start });
        }
        let item = &bytes[item_start..item_end];

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }

        let (base, step_part) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some((&item[slash + 1..], item_start + slash + 1)))
        } else {
            (item, None)
        };

        if base.is_empty() || step_part.is_some_and(|(s, _)| s.is_empty()) {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1 || (is_star && !hyphens.is_empty()) {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }

        let (start_part, end_part) = if is_star {
            (None, None)
        } else if let Some(&hyphen) = hyphens.first() {
            if hyphen == 0 || hyphen + 1 == base.len() {
                return Err(ParseError::InvalidSyntax { field, offset: item_start });
            }
            (Some((&base[..hyphen], item_start)), Some((&base[hyphen + 1..], item_start + hyphen + 1)))
        } else {
            (Some((base, item_start)), None)
        };

        let start = if let Some((part, offset)) = start_part {
            Some(parse_atom(field, part, offset, false)?)
        } else {
            None
        };
        let end = if let Some((part, offset)) = end_part {
            Some(parse_atom(field, part, offset, false)?)
        } else {
            None
        };

        if let (Some(s), Some(e)) = (start, end) {
            if s > e {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_part.unwrap().1,
                });
            }
        }

        let step = if let Some((part, offset)) = step_part {
            Some(parse_atom(field, part, offset, true)?)
        } else {
            None
        };

        let d = domain(field);
        let (range_start, range_end) = if is_star {
            (d.min, d.max)
        } else if let Some(s) = start {
            (s, if let Some(e) = end { e } else if step.is_some() { d.max } else { s })
        } else {
            (d.min, d.max)
        };

        let increment = step.unwrap_or(1);
        let mut value = range_start;
        loop {
            selected[value as usize] = true;
            if value > range_end.saturating_sub(increment) {
                break;
            }
            value += increment;
        }

        item_start = item_end + 1;
    }

    let d = domain(field);
    Ok((d.min..=d.max).filter(|&v| selected[v as usize]).collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut allowed = std::array::from_fn(|_| Vec::new());

    for (i, field) in field_kinds.iter().copied().enumerate() {
        allowed[i] = parse_field(field, fields[i])?;
    }

    Ok(Cron {
        allowed,
        day_of_month_unrestricted: fields[2] == "*",
        day_of_week_unrestricted: fields[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;

        loop {
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23 && current.minute == 59
            {
                return Ok(None);
            }

            current = advance_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let has = |field: Field, value: u8| self.allowed[field_index(field)].binary_search(&value).is_ok();

        if !has(Field::Minute, at.minute)
            || !has(Field::Hour, at.hour)
            || !has(Field::Month, at.month)
        {
            return false;
        }

        let dom = has(Field::DayOfMonth, at.day);
        let dow = has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        let dom_unrestricted = self.day_of_month_unrestricted;
        let dow_unrestricted = self.day_of_week_unrestricted;

        match (dom_unrestricted, dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
