#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::EmptyItem { field, offset },
        1 => ParseError::InvalidSyntax { field, offset },
        2 => ParseError::LeadingZero { field, offset },
        3 => ParseError::NumberOverflow { field, offset },
        4 => ParseError::OutOfRange { field, offset },
        5 => ParseError::ReversedRange { field, offset },
        6 => ParseError::BadStep { field, offset },
        _ => unreachable!(),
    }
}

fn name_value(field: Field, s: &str) -> Option<u32> {
    let upper = s.to_ascii_uppercase();
    match field {
        Field::Month => [
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]
        .iter()
        .position(|&n| n == upper)
        .map(|i| (i + 1) as u32),
        Field::DayOfWeek => ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]
            .iter()
            .position(|&n| n == upper)
            .map(|i| i as u32),
        _ => None,
    }
}

enum Atom {
    Number(u32, usize),
    Name(u32),
}

fn parse_atom(field: Field, s: &str, offset: usize) -> Result<Atom, ParseError> {
    if s.is_empty() {
        return Err(err(field, offset, 1));
    }
    if s.bytes().all(|b| b.is_ascii_digit()) {
        if s.len() > 1 && s.as_bytes()[0] == b'0' {
            return Err(err(field, offset, 2));
        }
        let mut value = 0u32;
        for b in s.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or_else(|| err(field, offset, 3))?;
        }
        let (min, max) = domain(field);
        if value < min || value > max {
            return Err(err(field, offset, 4));
        }
        Ok(Atom::Number(value, offset))
    } else if let Some(v) = name_value(field, s) {
        Ok(Atom::Name(v))
    } else {
        Err(err(field, offset, 1))
    }
}

fn atom_val(atom: Atom) -> (u32, Option<usize>) {
    match atom {
        Atom::Number(v, off) => (v, Some(off)),
        Atom::Name(v) => (v, None),
    }
}

fn parse_item(
    field: Field,
    item: &str,
    offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(err(field, offset, 0));
    }

    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    let base_end = item.find('/').unwrap_or(item.len());
    let base = &item[..base_end];
    let step_text = if slash_count == 1 {
        Some(&item[base_end + 1..])
    } else {
        None
    };

    let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
    if slash_count > 1
        || (slash_count == 1 && step_text.unwrap_or("").is_empty())
        || hyphen_count > 1
        || base.is_empty()
        || base == "*"
            && hyphen_count != 0
        || (base != "*" && hyphen_count == 0 && base.contains('-'))
    {
        return Err(err(field, offset, 1));
    }

    let (start, end, numeric_end_offset) = if base == "*" {
        let (min, max) = domain(field);
        (min, max, None)
    } else if hyphen_count == 1 {
        let hyphen = base.find('-').unwrap();
        let left = &base[..hyphen];
        let right = &base[hyphen + 1..];
        if left == "*" || right == "*" || left.is_empty() || right.is_empty() {
            return Err(err(field, offset, 1));
        }
        let a = parse_atom(field, left, offset)?;
        let b = parse_atom(field, right, offset + hyphen + 1)?;
        let (start, _) = atom_val(a);
        let (end, end_offset) = atom_val(b);
        if start > end {
            return Err(err(field, end_offset.unwrap_or(offset + hyphen + 1), 5));
        }
        (start, end, end_offset)
    } else {
        let a = parse_atom(field, base, offset)?;
        let (start, _) = atom_val(a);
        let (_, max) = domain(field);
        let end = if step_text.is_some() { max } else { start };
        (start, end, None)
    };

    let step = if let Some(text) = step_text {
        if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) {
            return Err(err(field, offset + base_end + 1, 1));
        }
        if text.len() > 1 && text.as_bytes()[0] == b'0' {
            return Err(err(field, offset + base_end + 1, 2));
        }
        let mut value = 0u32;
        for b in text.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or_else(|| err(field, offset + base_end + 1, 3))?;
        }
        let (_, max) = domain(field);
        if value == 0 || value > max + 1 {
            return Err(err(field, offset + base_end + 1, 6));
        }
        value
    } else {
        1
    };

    let _ = numeric_end_offset;
    let mut value = start;
    while value <= end {
        let index = (value - domain(field).0) as usize;
        if index < selected.len() {
            selected[index] = true;
        }
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (min, max) = domain(FIELDS[i]);
        vec![false; (max - min + 1) as usize]
    });

    let mut unrestricted = [false; 5];
    for i in 0..5 {
        let field = FIELDS[i];
        let text = parts[i];

        for (offset, b) in text.bytes().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        unrestricted[i] = (field == Field::DayOfMonth || field == Field::DayOfWeek) && text == "*";

        let mut start = 0;
        for item in text.split(',') {
            parse_item(field, item, start, &mut selected[i])?;
            start += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        day_of_month_unrestricted: unrestricted[2],
        day_of_week_unrestricted: unrestricted[4],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let idx = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = domain(field);
        self.selected[idx]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let idx = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, max) = domain(field);
        value as u32 >= min
            && value as u32 <= max
            && self.selected[idx][(value as u32 - min) as usize]
    }

    fn validate(at: CivilDateTime) -> Result<(), DateError> {
        if !(2000..=2399).contains(&at.year) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Year,
            });
        }
        if !(1..=12).contains(&at.month) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Month,
            });
        }
        if at.day == 0 || at.day > month_length(at.year, at.month) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Day,
            });
        }
        if at.hour > 23 {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Hour,
            });
        }
        if at.minute > 59 {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Minute,
            });
        }
        Ok(())
    }

    fn day_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match (self.day_of_month_unrestricted, self.day_of_week_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                let last = at.day == month_length(at.year, at.month);
                if last {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        Self::validate(at)?;
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.day_matches(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        Self::validate(after)?;
        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            current = next_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut result = 6u8;
    for y in 2000..year {
        result = (result + if leap(y) { 366 } else { 365 }) % 7;
    }
    for m in 1..month {
        result = (result + month_length(year, m)) % 7;
    }
    (result + day - 1) % 7
}

fn next_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
        } else {
            at.hour = 0;
            if at.day < month_length(at.year, at.month) {
                at.day += 1;
            } else {
                at.day = 1;
                if at.month < 12 {
                    at.month += 1;
                } else {
                    at.month = 1;
                    at.year += 1;
                }
            }
        }
    }
    at
}
