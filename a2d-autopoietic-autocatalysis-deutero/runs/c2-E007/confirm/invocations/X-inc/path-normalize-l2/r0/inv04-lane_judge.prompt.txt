#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }

    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a>(
    components: impl Iterator<Item = (usize, &'a str)>,
    parts: &mut Vec<String>,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, component) in components {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if parts.last().is_some_and(|part| part != "..") {
                parts.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                parts.push("..".to_owned());
            }
        } else {
            parts.push(component.to_owned());
        }
    }

    Ok(())
}

fn assemble(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = String::new();

    if absolute {
        result.push('/');
    }

    result.push_str(&parts.join("/"));

    if result.is_empty() {
        result.push('.');
    }

    if intent && result != "/" && result != "." {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut parts = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        let at = offset;
        offset += component.len() + 1;
        reduce_components(std::iter::once((at, component)), &mut parts, absolute)?;
    }

    Ok(assemble(&parts, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(PathError::InvalidCwd);
    }

    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;

    let home_parts = match home {
        Some(home_path) => Some(validate_context(home_path).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde_expanded = bytes.first() == Some(&b'~');
    let mut start = 0;
    let (mut parts, absolute) = if tilde_expanded {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }

        let parts = home_parts.ok_or(PathError::MissingHome)?;
        start = if bytes.len() > 1 { 2 } else { 1 };
        (parts, true)
    } else if bytes.first() == Some(&b'/') {
        (Vec::new(), true)
    } else {
        (cwd_parts, true)
    };

    let intent = directory_intent(path);
    let remainder = &path[start..];
    let mut offset = start;

    for component in remainder.split('/') {
        let at = offset;
        offset += component.len() + 1;
        reduce_components(std::iter::once((at, component)), &mut parts, absolute)?;
    }

    Ok(assemble(&parts, absolute, intent))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .filter(|(_, part)| !part.is_empty())
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, part)| *part == "." || *part == "..")
}

fn reduce_into(
    stack: &mut Vec<String>,
    path: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, part) in components(path) {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: offset_base + at,
                    });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(part.to_string()),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, dir_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if dir_intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let mut stack = Vec::new();
    reduce_into(&mut stack, path, 0, absolute)?;
    Ok(render(&stack, absolute, directory_intent(path)))
}

fn normalize_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    validate_raw(path).map_err(|_| PathError::InvalidCwd)?;
    let mut stack = Vec::new();
    reduce_into(&mut stack, path, 0, true).map_err(|_| PathError::InvalidCwd)?;
    Ok(stack)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_stack = normalize_context(cwd)?;

    let home_stack = match home {
        Some(value) => Some(normalize_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let tilde = path.as_bytes().first() == Some(&b'~');
    let (mut stack, remainder, offset_base, absolute) = if tilde {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_stack = home_stack.ok_or(PathError::MissingHome)?;
        let consumed = if bytes.len() > 1 { 2 } else { 1 };
        (home_stack, &path[consumed..], consumed, true)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_stack, path, 0, true)
    };

    reduce_into(&mut stack, remainder, offset_base, absolute)?;
    Ok(render(&stack, true, directory_intent(path)))
}

    InvalidHome,

        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn normalize_validated(path: &str) -> Result<String, PathError> {
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack: Vec<&str> = Vec::new();

    for component in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| *last != "..") {
                    stack.pop();
                } else if absolute {
                    let at = path.find(component).unwrap_or(0);
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(component);
                }
            }
            _ => stack.push(component),
        }
    }

    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    normalize_validated(path)
}

fn normalize_context(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    validate_raw(path).map_err(|_| PathError::InvalidCwd)?;
    normalize_validated(path).map_err(|_| PathError::InvalidCwd)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_normalized = normalize_context(cwd)?;
    let home_normalized = match home {
        Some(value) => {
            if !value.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(
                normalize_context(value)
                    .map_err(|_| PathError::InvalidHome)?,
            )
        }
        None => None,
    };

    validate_raw(path)?;

    let (base, remainder, offset) = if path.starts_with('/') {
        ("/".to_string(), path, 0)
    } else if path.starts_with('~') {
        if path.len() == 1 {
            (
                home_normalized.ok_or(PathError::MissingHome)?,
                "",
                1,
            )
        } else if path.as_bytes()[1] == b'/' {
            (
                home_normalized.ok_or(PathError::MissingHome)?,
                &path[2..],
                2,
            )
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (cwd_normalized, path, 0)
    };

    let intent = directory_intent(path);
    let mut stack: Vec<String> = components(&base).into_iter().map(str::to_owned).collect();

    for component in components(remainder) {
        match component {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    let at = path[offset..]
                        .match_indices(component)
                        .next()
                        .map(|(index, _)| offset + index)
                        .unwrap_or(offset);
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(component.to_owned()),
        }
    }

    let mut result = format!("/{}", stack.join("/"));
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}
