#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .next_back()
            .is_some_and(|component| component == "." || component == "..")
}

fn apply_component(
    components: &mut Vec<&str>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "" | "." => Ok(()),
        ".." => {
            if components.last().is_some_and(|last| *last != "..") {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                components.push("..");
            }
            Ok(())
        }
        _ => {
            components.push(component);
            Ok(())
        }
    }
}

fn normalized_parts(path: &str, absolute: bool) -> Result<Vec<&str>, PathError> {
    let mut components = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        if !component.is_empty() {
            apply_component(&mut components, component, absolute, offset)?;
        }
        offset += component.len() + 1;
    }

    Ok(components)
}

fn join_parts(parts: &[&str], absolute: bool, intent: bool) -> String {
    let joined = parts.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let parts = normalized_parts(path, absolute)?;

    Ok(join_parts(&parts, absolute, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = if cwd.as_bytes().first() == Some(&b'/') {
        normalize(cwd)
            .map_err(|_| PathError::InvalidCwd)?
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_owned)
            .collect::<Vec<_>>()
    } else {
        return Err(PathError::InvalidCwd);
    };

    let home_parts = if let Some(home) = home {
        if home.as_bytes().first() != Some(&b'/') {
            return Err(PathError::InvalidHome);
        }
        Some(
            normalize(home)
                .map_err(|_| PathError::InvalidHome)?
                .split('/')
                .filter(|part| !part.is_empty())
                .map(str::to_owned)
                .collect::<Vec<_>>(),
        )
    } else {
        None
    };

    validate_raw(path)?;

    let (mut components, remainder, base_offset) =
        if path.starts_with('~') {
            if path.len() == 1 || path.as_bytes().get(1) == Some(&b'/') {
                let parts = home_parts.ok_or(PathError::MissingHome)?;
                let skip = if path.len() > 1 { 2 } else { 1 };
                (parts, &path[skip..], skip)
            } else {
                return Err(PathError::UnsupportedTilde { at: 1 });
            }
        } else if path.starts_with('/') {
            (Vec::new(), path, 0)
        } else {
            (cwd_parts, path, 0)
        };

    let intent = directory_intent(path);
    let mut offset = base_offset;

    for component in remainder.split('/') {
        if !component.is_empty() {
            match component {
                "." => {}
                ".." => {
                    if components.pop().is_none() {
                        return Err(PathError::Escape { at: offset });
                    }
                }
                _ => components.push(component.to_owned()),
            }
        }
        offset += component.len() + 1;
    }

    let refs = components.iter().map(String::as_str).collect::<Vec<_>>();
    Ok(join_parts(&refs, true, intent))
}
