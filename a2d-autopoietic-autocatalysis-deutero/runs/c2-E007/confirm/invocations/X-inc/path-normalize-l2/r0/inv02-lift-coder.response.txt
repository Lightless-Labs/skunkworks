#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn normalize_validated(path: &str) -> Result<String, PathError> {
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack: Vec<&str> = Vec::new();

    for component in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| *last != "..") {
                    stack.pop();
                } else if absolute {
                    let at = path.find(component).unwrap_or(0);
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(component);
                }
            }
            _ => stack.push(component),
        }
    }

    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    normalize_validated(path)
}

fn normalize_context(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    validate_raw(path).map_err(|_| PathError::InvalidCwd)?;
    normalize_validated(path).map_err(|_| PathError::InvalidCwd)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_normalized = normalize_context(cwd)?;
    let home_normalized = match home {
        Some(value) => {
            if !value.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(
                normalize_context(value)
                    .map_err(|_| PathError::InvalidHome)?,
            )
        }
        None => None,
    };

    validate_raw(path)?;

    let (base, remainder, offset) = if path.starts_with('/') {
        ("/".to_string(), path, 0)
    } else if path.starts_with('~') {
        if path.len() == 1 {
            (
                home_normalized.ok_or(PathError::MissingHome)?,
                "",
                1,
            )
        } else if path.as_bytes()[1] == b'/' {
            (
                home_normalized.ok_or(PathError::MissingHome)?,
                &path[2..],
                2,
            )
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (cwd_normalized, path, 0)
    };

    let intent = directory_intent(path);
    let mut stack: Vec<String> = components(&base).into_iter().map(str::to_owned).collect();

    for component in components(remainder) {
        match component {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    let at = path[offset..]
                        .match_indices(component)
                        .next()
                        .map(|(index, _)| offset + index)
                        .unwrap_or(offset);
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(component.to_owned()),
        }
    }

    let mut result = format!("/{}", stack.join("/"));
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}
