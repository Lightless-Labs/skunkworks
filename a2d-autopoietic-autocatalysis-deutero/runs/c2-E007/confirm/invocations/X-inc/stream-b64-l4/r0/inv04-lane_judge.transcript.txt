fn alphabet_digit(alphabet: Alphabet, byte: u8) -> Option<u8> {

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    column: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            column: 0,
            closed: false,
        })
    }

    fn emit_symbol(&mut self, output: &mut String, symbol: u8) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            if self.column == width {
                match ending {
                    LineEnding::Lf => output.push('\n'),
                    LineEnding::CrLf => output.push_str("\r\n"),
                }
                self.column = 0;
            }
            output.push(symbol as char);
            self.column += 1;
        } else {
            output.push(symbol as char);
        }
    }

    fn emit_triplet(&mut self, bytes: [u8; 3], output: &mut String) {
        let digits = [
            bytes[0] >> 2,
            ((bytes[0] & 0x03) << 4) | (bytes[1] >> 4),
            ((bytes[1] & 0x0f) << 2) | (bytes[2] >> 6),
            bytes[2] & 0x3f,
        ];
        for digit in digits {
            self.emit_symbol(output, encode_digit(self.options.alphabet, digit));
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut output = String::new();
        for &byte in input {
            self.tail.push(byte);
            if self.tail.len() == 3 {
                let bytes = [self.tail[0], self.tail[1], self.tail[2]];
                self.tail.clear();
                self.emit_triplet(bytes, &mut output);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let mut output = String::new();
        match self.tail.len() {
            1 => {
                let byte = self.tail[0];
                self.emit_symbol(
                    &mut output,
                    encode_digit(self.options.alphabet, byte >> 2),
                );
                self.emit_symbol(
                    &mut output,
                    encode_digit(self.options.alphabet, (byte & 0x03) << 4),
                );
                if self.options.padding != Padding::Forbidden {
                    self.emit_symbol(&mut output, b'=');
                    self.emit_symbol(&mut output, b'=');
                }
            }
            2 => {
                let first = self.tail[0];
                let second = self.tail[1];
                self.emit_symbol(
                    &mut output,
                    encode_digit(self.options.alphabet, first >> 2),
                );
                self.emit_symbol(
                    &mut output,
                    encode_digit(

#[derive(Clone, Copy, PartialEq, Eq)]
enum WrapState {
    Row,
    NeedSeparator { first_offset: usize },
    NeedLf { cr_offset: usize },
    NeedSymbol { first_offset: usize },
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols_seen: 0,
            group: Vec::with_capacity(4),
            pending_second_padding: None,
            terminal_padding: false,
            closed: false,
            sticky: None,
            wrapped_column: 0,
            wrap_state: WrapState::Row,
        })
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn is_ignored_whitespace(&self, byte: u8) -> bool {
        self.options.whitespace == Whitespace::IgnoreAscii
            && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn wrap_byte(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        let (width, ending) = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => (width, ending),
            _ => return Ok(false),
        };

        match self.wrap_state {
            WrapState::Row => {
                if self.wrapped_column == width {
                    let expected = match ending {
                        LineEnding::Lf => b'\n',
                        LineEnding::CrLf => b'\r',
                    };
                    if byte != expected {
                        return Err(Error::BadWrap { offset });
                    }
                    if ending == LineEnding::Lf {
                        self.wrap_state = WrapState::NeedSymbol {
                            first_offset: offset,
                        };
                    } else {
                        self.wrap_state = WrapState::NeedLf { cr_offset: offset };
                    }
                    return Ok(true);
                }
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset });
                }
                self.wrapped_column += 1;
                Ok(false)
            }
            WrapState::NeedLf { cr_offset: _ } => {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                let first_offset = offset - 1;
                self.wrap_state = WrapState::NeedSymbol { first_offset };
                Ok(true)
            }
            WrapState::NeedSymbol { first_offset } => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset });
                }
                self.wrapped_column = 1;
                self.wrap_state = WrapState::Row;
                let _ = first_offset;
                Ok(false)
            }
            WrapState::NeedSeparator { first_offset: _ } => unreachable!(),
        }
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols_seen >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbols_seen += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if self.pending_second_padding.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match self.group.len() {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_second_padding = Some(offset);
                    self.group.push((byte, offset));
                    return Ok(());
                }
                3 => {
                    if self.pending_second_padding.is_some() {
                        self.group.push((byte, offset));

                            return Err(Error::NonCanonical {
                                offset: self.group[1].1,
                            });
                        }
                        output.push((d0 << 2) | (d1 >> 4));
                    } else {
                        let d0 = alphabet_digit(self.options.alphabet, self.group[0].0).unwrap();
                        let d1 = alphabet_digit(self.options.alphabet, self.group[1].0).unwrap();
                        let d2 = alphabet_digit(self.options.alphabet, self.group[2].0).unwrap();
                        if self.options.strict && d2 & 0x03 != 0 {
                            return Err(Error::NonCanonical {
                                offset: self.group[2].1,
                            });
                        }
                        output.push((d0 << 2) | (d1 >> 4));
                        output.push((d1 << 4) | (d2 >> 2));
                    }
                    self.group.clear();
                    self.pending_second_padding = None;
                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => return Err(Error::BadPadding { offset }),
            }
        }

            self.group.push((byte, offset));
            if self.group.len() == 4 {
                let d0 = alphabet_digit(self.options.alphabet, self.group[0].0).unwrap();

                output.push((d1 << 4) | (d2 >> 2));

    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            if self.is_ignored_whitespace(byte) {
                continue;
            }

            if let Whitespace::Wrapped { .. } = self.options.whitespace {
                match self.wrap_byte(byte, offset) {
                    Ok(true) => continue,
                    Ok(false) => {}
                    Err(error) => return self.fail(error),
                }
            }

            if let Err(error) = self.process_symbol(byte, offset, &mut output) {
                return self.fail(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }

        if let WrapState::NeedLf { cr_offset } = self.wrap_state {
            return self.fail(Error::BadWrap { offset: cr_offset });
        }
        if let WrapState::NeedSymbol { first_offset } = self.wrap_state {
            return self.fail(Error::BadWrap {
                offset: first_offset,
            });
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }

        let mut output = Vec::new();
        match self.group.len() {
            0 => {}
            1 => return self.fail(Error::Truncated { offset: self.group[0].1 }),
            2 | 3 if self.options.padding != Padding::Required => {
                let d0 = alphabet_digit(self.options.alphabet, self.group[0].0).unwrap();
                let d1 = alphabet_digit(self.options.alphabet, self.group[1].0).unwrap();
                if self.options.strict && d1 & 0x0f != 0 {
                    return self.fail(Error::NonCanonical {
                        offset: self.group[1].1,
                    });
                }
                output.push((d0 << 2) | (d1 >> 4));
                if self.group.len() == 3 {
                    let d2 = alphabet_digit(self.options.alphabet, self.group[2].0).unwrap();
                    if self.options.strict && d2 & 0x03 != 0 {
                        return self.fail(Error::NonCanonical {
                            offset: self.group[2].1,
                        });
                    }
                    output.push((d1 << 4) | (d2 >> 2));
                }
            }
            _ => return self.fail(Error::Truncated { offset: self.group[0].1 }),
        }

        self.closed = true;
        Ok(output)
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    let mut encoder = Encoder::new(options)?;
    let mut output = encoder.push(input)?;
    output.push_str(&encoder.finish()?);
    Ok(output)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

        for chunk in data[..complete_len].chunks_exact(3) {
            let n = ((chunk[0] as u32) << 16)
                | ((chunk[1] as u32) << 8)
                | chunk[2] as u32;

                alphabet[((n >> 18) & 63) as usize],
                alphabet[((n >> 12) & 63) as usize],
                alphabet[((n >> 6) & 63) as usize],
                alphabet[(n & 63) as usize],
            ];

        }

            0 => {}
            1 => {

struct DecoderState {
    options: Options,
    group: Vec<Entry>,
    absolute: usize,
    symbols_seen: usize,
    terminal_padding: bool,

pub struct Decoder {
    state: DecoderState,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            state: DecoderState {
                options,
                group: Vec::new(),
                absolute: 0,
                symbols_seen: 0,
                terminal_padding: false,
                pending_wrap: None,
                row_symbols: 0,
                after_separator: None,
                finished: false,
                sticky: None,
            },
        })
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.state.sticky = Some(error.clone());
        Err(error)
    }

    fn digit(&self, byte: u8, offset: usize) -> Result<u8, Error> {
        if byte == b'+' || byte == b'/' || byte == b'-' || byte == b'_' {
            let foreign = match self.state.options.alphabet {
                Alphabet::Standard => byte == b'-' || byte == b'_',
                Alphabet::UrlSafe => byte == b'+' || byte == b'/',
            };
            if foreign && self.state.options.strict {
                return Err(Error::MixedAlphabet { offset });
            }
        }
        symbols(self.state.options.alphabet)
            .iter()
            .position(|&b| b == byte)
            .map(|v| v as u8)
            .ok_or(Error::InvalidByte { offset, byte })
    }

    fn complete_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        let g = &self.state.group;
        let (a, b) = (g[0].value, g[1].value);
        if g[2].value == 64 {
            let offset = g[1].offset;
            if self.state.options.strict && b & 15 != 0 {
                return Err(Error::NonCanonical { offset });
            }
            out.push((a << 2) | (b >> 4));
            self.state.terminal_padding = true;
        } else if g[3].value == 64 {
            let c = g[2].value;
            if self.state.options.strict && c & 3 != 0 {
                return Err(Error::NonCanonical { offset: g[2].offset });
            }
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            self.state.terminal_padding = true;
        } else {
            let c = g[2].value;
            let d = g[3].value;
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);
        }
        self.state.group.clear();
        Ok(())
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.state.options.max_symbols {
            if self.state.symbols_seen >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.state.symbols_seen += 1;

        if self.state.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if self.state.group.len() == 2 && self.state.group[1].value == 64 {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.state.group.push(Entry { value: 64, offset });
            self.complete_group(out)?;
            return Ok(());
        }

        let pos = self.state.group.len();
        if byte == b'=' {
            if self.state.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match pos {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.state.group.push(Entry { value: 64, offset });
                    return Ok(());
                }
                3 => {
                    self.state.group.push(Entry { value: 64, offset });
                    self.complete_group(out)?;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = self.digit(byte, offset)?;
        self.state.group.push(Entry { value, offset });
        if self.state.group.len() == 4 {
            self.complete_group(out)?;
        }
        Ok(())
    }

    fn process_wrapped_byte(
        &mut self,
        byte: u8,
        offset: usize,
        out: &mut Vec<u8>,
    ) -> Result<(), Error> {
        let Whitespace::Wrapped { width, ending } = self.state.options.whitespace else {
            return self.process_symbol(byte, offset, out);
        };

        if let Some((start, expecting_lf)) = self.state.pending_wrap {
            if expecting_lf {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                self.state.pending_wrap = None;
                self.state.after_separator = Some(start);
                return Ok(());
            }
            if byte != b'\r' {
                return Err(Error::BadWrap { offset });
            }
            self.state.pending_wrap = Some((start, true));
            return Ok(());
        }

        if let Some(start) = self.state.after_separator {
            if byte == b'\r' || byte == b'\n' || is_ignored(byte) {
                return Err(Error::BadWrap { offset });
            }
            self.state.after_separator = None;
            self.state.row_symbols = 0;
            self.process_symbol(byte, offset, out)?;
            return Ok(());
        }

        if self.state.row_symbols == width {
            match ending {
                LineEnding::Lf if byte == b'\n' => {
                    self.state.after_separator = Some(offset);
                    return Ok(());
                }
                LineEnding::CrLf if byte == b'\r' => {
                    self.state.pending_wrap = Some((offset, false));
                    return Ok(());
                }
                _ => return Err(Error::BadWrap { offset }),
            }
        }

        if byte == b'\r' || byte == b'\n' || is_ignored(byte) {
            return Err(Error::BadWrap { offset });
        }
        self.state.row_symbols += 1;
        self.process_symbol(byte, offset, out)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.state.absolute;
            self.state.absolute += 1;
            let result = match self.state.options.whitespace {
                Whitespace::IgnoreAscii if is_ignored(byte) => Ok(()),
                _ => match self.process_wrapped_byte(byte, offset, &mut out) {
                    Ok(()) => Ok(()),
                    Err(e) => Err(e),
                },
            };
            if let Err(error) = result {
                return self.fail(error);
            }
        }
        Ok(out)
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        if let Some((start, expecting_lf)) = self.state.pending_wrap {
            let offset = if expecting_lf { start } else { start };
            return Err(Error::BadWrap { offset });
        }
        if let Some(start) = self.state.after_separator {
            return Err(Error::BadWrap { offset: start });
        }

        let mut out = Vec::new();
        if self.state.group.len() == 2 && self.state.group[1].value == 64 {
            return Err(Error::BadPadding {
                offset: self.state.group[1].offset,
            });
        }

        let len = self.state.group.len();
        if len == 0 {
            return Ok(out);
        }
        if self.state.options.padding == Padding::Required {
            return Err(Error::Truncated {
                offset: self.state.group[0].offset,
            });
        }
        if len == 1 {
            return Err(Error::Truncated {
                offset: self.state.group[0].offset,
            });
        }

        let a = self.state.group[0].value;
        let b = self.state.group[1].value;
        if len == 2 {
            if self.state.options.strict && b & 15 != 0 {
                return Err(Error::NonCanonical {
                    offset: self.state.group[1].offset,
                });
            }
            out.push((a << 2) | (b >> 4));
        } else if len == 3 {
            let c = self.state.group[2].value;
            if self.state.options.strict && c & 3 != 0 {
                return Err(Error::NonCanonical {
                    offset: self.state.group[2].offset,
                });
            }
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.finished {
            return Err(Error::AlreadyFinished);
        }
        match self.finish_inner() {
            Ok(bytes) => {
                self.state.finished = true;
                Ok(bytes)
            }
            Err(error) => self.fail(error),
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

        52..=61 => b'0' + (value - 52),
        62 => match alphabet {

            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {

            Alphabet::UrlSafe => b'_',
        },

fn lookup(byte: u8, alphabet: Alphabet, strict: bool, offset: usize) -> Result<u8, Error> {
    let value = match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),

        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),

        return Ok(value);

            (alphabet, byte),
            (Alphabet::Standard, b'-' | b'_') | (Alphabet::UrlSafe, b'+' | b'/')

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    closed: bool,
    symbols_in_row: usize,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffered: Vec::new(),
            closed: false,
            symbols_in_row: 0,
        })
    }

    fn emit_symbols(&mut self, symbols: &[u8], output: &mut String) {
        for &symbol in symbols {
            if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
                if self.symbols_in_row == width {
                    match ending {
                        LineEnding::Lf => output.push('\n'),
                        LineEnding::CrLf => output.push_str("\r\n"),
                    }
                    self.symbols_in_row = 0;
                }
                self.symbols_in_row += 1;
            }
            output.push(symbol as char);
        }
    }

    fn encode_full(&mut self, bytes: &[u8], output: &mut String) {
        let symbols = [
            encode_digit(bytes[0] >> 2, self.options.alphabet),
            encode_digit(((bytes[0] & 3) << 4) | (bytes[1] >> 4), self.options.alphabet),
            encode_digit(((bytes[1] & 15) << 2) | (bytes[2] >> 6), self.options.alphabet),
            encode_digit(bytes[2] & 63, self.options.alphabet),
        ];
        self.emit_symbols(&symbols, output);
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut data = Vec::with_capacity(self.buffered.len() + input.len());
        data.extend_from_slice(&self.buffered);
        data.extend_from_slice(input);
        self.buffered.clear();

        let complete_len = data.len() / 3 * 3;
        let mut output = String::new();
        for chunk in data[..complete_len].chunks_exact(3) {
            self.encode_full(chunk, &mut output);
        }
        self.buffered.extend_from_slice(&data[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let mut output = String::new();
        match self.buffered.len() {
            0 => {}
            1 => {
                let byte = self.buffered[0];
                let symbols = [
                    encode_digit(byte >> 2, self.options.alphabet),
                    encode_digit((byte & 3) << 4, self.options.alphabet),
                ];
                self.emit_symbols(&symbols, &mut output);
                if self.options.padding != Padding::Forbidden {
                    self.emit_symbols(b"==", &mut output);
                }
            }
            2 => {
                let a = self.buffered[0];
                let b = self.buffered[1];
                let symbols = [
                    encode_digit(a >> 2, self.options.alphabet),
                    encode_digit(((a & 3) << 4) | (b >> 4), self.options.alphabet),
                    encode_digit((b & 15) << 2, self.options.alphabet),
                ];
                self.emit_symbols(&symbols, &mut output);
                if self.options.padding != Padding::Forbidden {
                    self.emit_symbols(b"=", &mut output);
                }
            }
            _ => unreachable!(),
        }
        self.buffered.clear();
        Ok(output)
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
    value: Option<u8>,
}

pub struct Decoder {
    options: Options,
    position: usize,
    symbols_seen: usize,
    row_symbols: usize,
    framing: FrameState,
    group: Vec<Symbol>,
    pending_pad: Option<usize>,
    terminal_padding: bool,
    closed: bool,
    sticky: Option<Error>,
}

#[derive(Clone, Copy)]
enum FrameState {
    Ready,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            position: 0,
            symbols_seen: 0,
            row_symbols: 0,
            framing: FrameState::Ready,
            group: Vec::with_capacity(4),
            pending_pad: None,
            terminal_padding: false,
            closed: false,
            sticky: None,
        })
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn process_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let g = &self.group;
        let bytes = match g.len() {
            4 if g[2].byte == b'=' && g[3].byte == b'=' => {
                if self.options.strict && g[1].value.unwrap() & 15 != 0 {
                    return self.fail(Error::NonCanonical { offset: g[1].offset });
                }
                self.terminal_padding = true;
                vec![(g[0].value.unwrap() << 2) | (g[1].value.unwrap() >> 4)]
            }
            4 if g[3].byte == b'=' => {
                if self.options.strict && g[2].value.unwrap() & 3 != 0 {
                    return self.fail(Error::NonCanonical { offset: g[2].offset });
                }
                self.terminal_padding = true;
                vec![
                    (g[0].value.unwrap() << 2) | (g[1].value.unwrap() >> 4),
                    (g[1].value.unwrap() << 4) | (g[2].value.unwrap() >> 2),
                ]
            }
            4 => vec![
                (g[0].value.unwrap() << 2) | (g[1].value.unwrap() >> 4),
                (g[1].value.unwrap() << 4) | (g[2].value.unwrap() >> 2),
                (g[2].value.unwrap() << 6) | g[3].value.unwrap(),
            ],
            _ => return Ok(()),
        };
        output.extend(bytes);
        self.group.clear();
        self.pending_pad = None;
        Ok(())
    }

    fn next_wrapped(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        let Whitespace::Wrapped { width, ending } = self.options.whitespace else {
            return Ok(true);
        };

        match self.framing {
            FrameState::NeedLf { cr_offset } => {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                self.framing = FrameState::NeedSymbol {
                    separator_offset: cr_offset,
                };
                return Ok(false);
            }
            FrameState::NeedSymbol { .. } => {
                self.framing = FrameState::Ready;
                self.row_symbols = 0;
                if matches!(byte, b'\r' | b'\n' | b' ' | b'\t' | 11 | 12) {
                    return Err(Error::BadWrap { offset });
                }
                return Ok(true);
            }
            FrameState::Ready => {}
        }

        if self.row_symbols == width {
            match ending {
                LineEnding::Lf if byte == b'\n' => {
                    self.framing = FrameState::NeedSymbol {
                        separator_offset: offset,
                    };
                    Ok(false)
                }
                LineEnding::CrLf if byte == b'\r' => {
                    self.framing = FrameState::NeedLf { cr_offset: offset };
                    Ok(false)
                }
                _ => Err(Error::BadWrap { offset }),
            }
        } else if matches!(byte, b'\r' | b'\n' | b' ' | b'\t' | 11 | 12) {
            Err(Error::BadWrap { offset })
        } else {
            self.row_symbols += 1;
            Ok(true)
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.position;
            self.position += 1;

            if self.options.whitespace == Whitespace::IgnoreAscii && is_ignored(byte) {
                continue;
            }

            let framed_symbol = match self.next_wrapped(byte, offset) {
                Ok(value) => value,
                Err(error) => return self.fail(error),
            };
            if !framed_symbol {
                continue;
            }

            if let Some(limit) = self.options.max_symbols {
                if self.symbols_seen >= limit {
                    return self.fail(Error::Overlong { offset, limit });
                }
            }
            self.symbols_seen += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset });
            }
            if let Some(pad_offset) = self.pending_pad {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset });
                }
                self.group.push(Symbol {
                    byte,
                    offset,
                    value: None,
                });
                self.pending_pad = None;
                if let Err(error) = self.process_group(&mut output) {
                    return Err(error);
                }
                let _ = pad_offset;
                continue;
            }

            let index = self.group.len();
            if byte == b'=' {
                if self.options.padding == Padding::Forbidden {
                    return self.fail(Error::BadPadding { offset });
                }
                match index {
                    0 | 1 => return self.fail(Error::BadPadding { offset }),
                    2 => {
                        self.group.push(Symbol {
                            byte,
                            offset,
                            value: None,
                        });
                        self.pending_pad = Some(offset);
                    }
                    3 => {
                        self.group.push(Symbol {
                            byte,
                            offset,
                            value: None,
                        });
                        if let Err(error) = self.process_group(&mut output) {
                            return Err(error);
                        }
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let value = match lookup(byte, self.options.alphabet, self.options.strict, offset) {
                Ok(value) => value,
                Err(error) => return self.fail(error),
            };
            self.group.push(Symbol {
                byte,
                offset,
                value: Some(value),
            });
            if self.group.len() == 4 {
                if let Err(error) = self.process_group(&mut output) {
                    return Err(error);
                }
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        if let FrameState::NeedLf { cr_offset } = self.framing {
            return self.fail(Error::BadWrap { offset: cr_offset });
        }
        if let FrameState::NeedSymbol { separator_offset } = self.framing {
            return self.fail(Error::BadWrap {
                offset: separator_offset,
            });
        }

        if let Some(offset) = self.pending_pad {
            return self.fail(Error::BadPadding { offset });
        }

        if !self.group.is_empty() {
            if self.options.padding == Padding::Required {
                return self.fail(Error::Truncated {
                    offset: self.group[0].offset,
                });
            }
            match self.group.len() {
                1 => {
                    return self.fail(Error::Truncated {
                        offset: self.group[0].offset,
                    });
                }
                2 => {
                    if self.options.strict && self.group[1].value.unwrap() & 15 != 0 {
                        return self.fail(Error::NonCanonical {
                            offset: self.group[1].offset,
                        });
                    }
                }
                3 => {
                    if self.options.strict && self.group[2].value.unwrap() & 3 != 0 {
                        return self.fail(Error::NonCanonical {
                            offset: self.group[2].offset,
                        });
                    }
                }
                _ => unreachable!(),
            }
        }

        let mut output = Vec::new();
        match self.group.len() {
            2 => {
                output.push((self.group[0].value.unwrap() << 2)
                    | (self.group[1].value.unwrap() >> 4));
            }
            3 => {
                output.push((self.group[0].value.unwrap() << 2)
                    | (self.group[1].value.unwrap() >> 4));
                output.push((self.group[1].value.unwrap() << 4)
                    | (self.group[2].value.unwrap() >> 2));
            }
            _ => {}
        }
        self.group.clear();
        self.closed = true;
        Ok(output)
    }
}
