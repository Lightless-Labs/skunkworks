```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn valid_radix(radix: u32) -> Result<(), Error> {
        if (2..=36).contains(&radix) {
            Ok(())
        } else {
            Err(Error::InvalidRadix { radix })
        }
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::valid_radix(radix)?;

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan every magnitude byte before reporting any malformed separator.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            match Self::digit_value(byte) {
                Some(value) if value < radix => {}
                _ => return Err(Error::InvalidDigit { index }),
            }
        }

        let mut first_separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                first_separator_error.get_or_insert(index);
            }
        }
        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len())
            .find(|&i| bytes[i] != b'_')
            .expect("sign-only input was handled");
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }
        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            let index = (start..bytes.len())
                .find(|&i| {
                    if bytes[i] == b'_' {
                        false
                    } else {
                        seen += 1;
                        seen == excess_number
                    }
                })
                .unwrap_or(first_digit_index);
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut result = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            result.mul_small_assign(radix);
            result.add_small_assign(Self::digit_value(byte).unwrap());
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        Self::valid_radix(radix)?;

        let mut magnitude = self.limbs.clone();
        let mut digits = Vec::new();
        if magnitude.is_empty() {
            digits.push(0);
        } else {
            while !magnitude.is_empty() {
                let rem = Self::div_small_vec(&mut magnitude, radix);
                digits.push(rem as u8);
            }
            digits.reverse();
        }

        let width = min_digits.max(digits.len()).max(1);
        if width > digits.len() {
            let mut padded = vec![0; width - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(match *digit {
                0..=9 => (b'0' + *digit) as char,
                _ => (b'a' + (*digit - 10)) as char,
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (-1, 0 | 1) => Ordering::Less,
            (0 | 1, -1) => Ordering::Greater,
            (0, 0) => Ordering::Equal,
            (1, 1) => Self::cmp_abs(self, rhs),
            (-1, -1) => Self::cmp_abs(rhs, self),
            _ => Ordering::Equal,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative;
            result.normalize();
            result
        } else {
            match Self::cmp_abs(self, rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::sub_abs(self, rhs);
                    result.negative = self.negative;
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self::sub_abs(rhs, self);
                    result.negative = rhs.negative;
                    result.normalize();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            limbs: vec![0; self.limbs.len() + rhs.limbs.len()],
        };

        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = result.limbs[index] as u64
                    + (a as u64) * (b as u64)
                    + carry;
                result.limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry > 0 {
                if index == result.limbs.len() {
                    result.limbs.push(0);
                }
                let value = result.limbs[index] as u64 + carry;
                result.limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                index += 1;
            }
        }

        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (mut quotient, mut remainder) = Self::div_rem_abs(self, rhs);
        quotient.negative = !quotient.is_zero() && (self.negative ^ rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }

        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power > 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = Self::div_rem_abs(&a, &b);
            a = b;
            b = remainder;
        }
        a
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn cmp_abs(a: &Self, b: &Self) -> Ordering {
        match a.limbs.len().cmp(&b.limbs.len()) {
            Ordering::Equal => {
                for (x, y) in a.limbs.iter().rev().zip(b.limbs.iter().rev()) {
                    match x.cmp(y) {
                        Ordering::Equal => {}
                        ordering => return ordering,
                    }
                }
                Ordering::Equal
            }
            ordering => ordering,
        }
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let len = a.limbs.len().max(b.limbs.len());
        let mut limbs = Vec::with_capacity(len + 1);
        let mut carry = 0u64;
        for i in 0..len {
            let x = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let y = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = x + y + carry;
            limbs.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry > 0 {
            limbs.push(carry as u32);
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut limbs = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let x = a.limbs[i] as i64 - borrow;
            let y = b.limbs.get(i).copied().unwrap_or(0) as i64;
            if x < y {
                limbs.push((x + Self::BASE as i64 - y) as u32);
                borrow = 1;
            } else {
                limbs.push((x - y) as u32);
                borrow = 0;
            }
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    fn mul_small_assign(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry > 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_assign(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut index = 0;
        while carry > 0 {
            if index == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[index] as u64 + carry;
            self.limbs[index] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            index += 1;
        }
    }

    fn div_small_vec(limbs: &mut Vec<u32>, divisor: u32) -> u32 {
        let mut remainder = 0u64;
        for limb in limbs.iter_mut().rev() {
            let value = remainder * Self::BASE + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        while limbs.last() == Some(&0) {
            limbs.pop();
        }
        remainder as u32
    }

    fn mul_small_vec(limbs: &mut Vec<u32>, factor: u32) {
        if factor == 0 {
            limbs.clear();
            return;
        }
        let mut carry = 0u64;
        for limb in limbs.iter_mut() {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry > 0 {
            limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_vec(limbs: &mut Vec<u32>, value: u32) {
        let mut carry = value as u64;
        let mut index = 0;
        while carry > 0 {
            if index == limbs.len() {
                limbs.push(0);
            }
            let sum = limbs[index] as u64 + carry;
            limbs[index] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            index += 1;
        }
    }

    // Long division in base BASE. Quotient and remainder are nonnegative.
    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if Self::cmp_abs(a, b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }

        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut remainder = Self::zero();

        for &limb in a.limbs.iter().rev() {
            if !remainder.is_zero() {
                remainder.limbs.insert(0, limb);
            } else if limb != 0 {
                remainder.limbs.push(limb);
            }
            remainder.normalize();

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let mut product = b.limbs.clone();
                Self::mul_small_vec(&mut product, mid as u32);
                let product_big = Self {
                    negative: false,
                    limbs: product,
                };
                match Self::cmp_abs(&product_big, &remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }

            quotient_be.push(best as u32);
            if best > 0 {
                let mut product_limbs = b.limbs.clone();
                Self::mul_small_vec(&mut product_limbs, best as u32);
                let product = Self {
                    negative: false,
                    limbs: product_limbs,
                };
                remainder = Self::sub_abs(&remainder, &product);
            }
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }
}
```
