```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn from_digits(digits: &[u32], radix: u32, negative: bool) -> Self {
        let mut value = Self::zero();
        for &digit in digits {
            value.mul_small_assign(radix);
            value.add_small_assign(digit);
        }
        value.negative = negative && !value.is_zero();
        value
    }

    fn mul_small_assign(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            if factor == 0 {
                self.limbs.clear();
                self.negative = false;
            }
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let product = *limb as u64 * factor as u64 + carry;
            *limb = (product % Self::BASE) as u32;
            carry = product / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_assign(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len().max(b.limbs.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.limbs.len().max(b.limbs.len()) {
            let sum = a.limbs.get(i).copied().unwrap_or(0) as u64
                + b.limbs.get(i).copied().unwrap_or(0) as u64
                + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self {
            negative: false,
            limbs: out,
        }
    }

    // Requires |a| >= |b|.
    fn abs_sub(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let mut value = a.limbs[i] as i64
                - b.limbs.get(i).copied().unwrap_or(0) as i64
                - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u32);
        }
        let mut result = Self {
            negative: false,
            limbs: out,
        };
        result.normalize();
        result
    }

    fn abs(&self) -> Self {
        let mut out = self.clone();
        out.negative = false;
        out
    }

    fn div_rem_small_assign(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let current = rem * Self::BASE + *limb as u64;
            *limb = (current / divisor as u64) as u32;
            rem = current % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    fn abs_mul(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; a.limbs.len() + b.limbs.len()];
        for (i, &x) in a.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.limbs.iter().enumerate() {
                let k = i + j;
                let value = out[k] + x as u64 * y as u64 + carry;
                out[k] = value % Self::BASE;
                carry = value / Self::BASE;
            }
            let mut k = i + b.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let value = out[k] + carry;
                out[k] = value % Self::BASE;
                carry = value / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self {
            negative: false,
            limbs: out.into_iter().map(|x| x as u32).collect(),
        };
        result.normalize();
        result
    }

    // Adds a base-B limb to the low end: value = value * BASE + limb.
    fn shift_base_add(&mut self, limb: u32) {
        if self.is_zero() {
            if limb != 0 {
                self.limbs.push(limb);
            }
            return;
        }
        self.limbs.insert(0, limb);
    }

    fn abs_div_rem(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.abs_cmp(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }

        let mut quotient = vec![0u32; a.limbs.len()];
        let mut remainder = Self::zero();

        for i in (0..a.limbs.len()).rev() {
            remainder.shift_base_add(a.limbs[i]);

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = b.mul_small_abs(mid as u32);
                match product.abs_cmp(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }
            quotient[i] = best as u32;
            if best != 0 {
                let product = b.mul_small_abs(best as u32);
                remainder = Self::abs_sub(&remainder, &product);
            }
        }

        let mut q = Self {
            negative: false,
            limbs: quotient,
        };
        q.normalize();
        remainder.normalize();
        (q, remainder)
    }

    fn mul_small_abs(&self, factor: u32) -> Self {
        let mut out = self.abs();
        out.mul_small_assign(factor);
        out
    }

    fn radix_digits(&self, radix: u32) -> Vec<u32> {
        if self.is_zero() {
            return vec![0];
        }
        let mut value = self.abs();
        let mut digits = Vec::new();
        while !value.is_zero() {
            digits.push(value.div_rem_small_assign(radix));
        }
        digits.reverse();
        digits
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (start, negative) = match bytes[0] {
            b'+' => (1, false),
            b'-' => (1, true),
            _ => (0, false),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && digit_value(byte).is_none_or(|d| d >= radix) {
                invalid = Some(index);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                malformed_separator = Some(i);
                break;
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let mut seen = 0;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == max_digits.saturating_add(1) {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let digits: Vec<u32> = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| digit_value(b).unwrap())
            .collect();
        Ok(Self::from_digits(&digits, radix, negative))
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let digits = self.radix_digits(radix);
        let padding = min_digits.saturating_sub(digits.len());
        let total = padding + digits.len();
        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }

        for i in 0..total {
            if group > 0 && i > 0 && (total - i) % group == 0 {
                out.push('_');
            }
            let digit = if i < padding { 0 } else { digits[i - padding] };
            out.push(digit_char(digit));
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => rhs.abs_cmp(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self::abs_add(self, rhs);
            out.negative = self.negative && !out.is_zero();
            out
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self::abs_sub(self, rhs);
                    out.negative = self.negative;
                    out
                }
                Ordering::Less => {
                    let mut out = Self::abs_sub(rhs, self);
                    out.negative = rhs.negative;
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        let mut negated = rhs.clone();
        negated.negative = !negated.negative && !negated.is_zero();
        self.add(&negated)
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut out = Self::abs_mul(self, rhs);
        out.negative = !out.is_zero() && (self.negative ^ rhs.negative);
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::abs_div_rem(self, rhs);
        quotient.negative = !quotient.is_zero() && (self.negative ^ rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }
        let mut exp = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while exp != 0 {
            if exp & 1 == 1 {
                result = result.mul(&base);
            }
            exp >>= 1;
            if exp != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = Self::abs_div_rem(&a, &b);
            a = b;
            b = remainder;
        }
        a.negative = false;
        a
    }
}

fn digit_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
        b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
        _ => None,
    }
}

fn digit_char(digit: u32) -> char {
    match digit {
        0..=9 => (b'0' + digit as u8) as char,
        _ => (b'a' + (digit - 10) as u8) as char,
    }
}
```
