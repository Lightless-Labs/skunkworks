use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn valid_radix(radix: u32) -> Result<(), Error> {
        if (2..=36).contains(&radix) {
            Ok(())
        } else {
            Err(Error::InvalidRadix { radix })
        }
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::valid_radix(radix)?;

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan every magnitude byte before reporting any malformed separator.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            match Self::digit_value(byte) {
                Some(value) if value < radix => {}
                _ => return Err(Error::InvalidDigit { index }),
            }
        }

        let mut first_separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                first_separator_error.get_or_insert(index);
            }
        }
        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len())
            .find(|&i| bytes[i] != b'_')
            .expect("sign-only input was handled");
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }
        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            let index = (start..bytes.len())
                .find(|&i| {
                    if bytes[i] == b'_' {
                        false
                    } else {
                        seen += 1;
                        seen == excess_number
                    }
                })
                .unwrap_or(first_digit_index);
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut result = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            result.mul_small_assign(radix);
            result.add_small_assign(Self::digit_value(byte).unwrap());
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        Self::valid_radix(radix)?;

        let mut magnitude = self.limbs.clone();
        let mut digits = Vec::new();
        if magnitude.is_empty() {
            digits.push(0);
        } else {
            while !magnitude.is_empty() {
                let rem = Self::div_small_vec(&mut magnitude, radix);
                digits.push(rem as u8);
            }
            digits.reverse();
        }

        let width = min_digits.max(digits.len()).max(1);
        if width > digits.len() {
            let mut padded = vec![0; width - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(match *digit {
                0..=9 => (b'0' + *digit) as char,
                _ => (b'a' + (*digit - 10)) as char,
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (-1, 0 | 1) => Ordering::Less,
            (0 | 1, -1) => Ordering::Greater,
            (0, 0) => Ordering::Equal,
            (1, 1) => Self::cmp_abs(self, rhs),
            (-1, -1) => Self::cmp_abs(rhs, self),
            _ => Ordering::Equal,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative;
            result.normalize();
            result
        } else {
            match Self::cmp_abs(self, rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::sub_abs(self, rhs);
                    result.negative = self.negative;
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self::sub_abs(rhs, self);
                    result.negative = rhs.negative;
                    result.normalize();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            limbs: vec![0; self.limbs.len() + rhs.limbs.len()],
        };

        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = result.limbs[index] as u64
                    + (a as u64) * (b as u64)
                    + carry;
                result.limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry > 0 {
                if index == result.limbs.len() {
                    result.limbs.push(0);
                }
                let value = result.limbs[index] as u64 + carry;
                result.limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                index += 1;
            }
        }

        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (mut quotient, mut remainder) = Self::div_rem_abs(self, rhs);
        quotient.negative = !quotient.is_zero() && (self.negative ^ rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }

        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power > 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = Self::div_rem_abs(&a, &b);
            a = b;
            b = remainder;
        }
        a
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn cmp_abs(a: &Self, b: &Self) -> Ordering {
        match a.limbs.len().cmp(&b.limbs.len()) {
            Ordering::Equal => {
                for (x, y) in a.limbs.iter().rev().zip(b.limbs.iter().rev()) {
                    match x.cmp(y) {
                        Ordering::Equal => {}
                        ordering => return ordering,
                    }
                }
                Ordering::Equal
            }
            ordering => ordering,
        }
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let len = a.limbs.len().max(b.limbs.len());
        let mut limbs = Vec::with_capacity(len + 1);
        let mut carry = 0u64;
        for i in 0..len {
            let x = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let y = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = x + y + carry;
            limbs.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry > 0 {
            limbs.push(carry as u32);
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    // Requires |a| >= |b|.

        let mut limbs = Vec::with_capacity(a.limbs.len());

        for i in 0..a.limbs.len() {
            let x = a.limbs[i] as i64 - borrow;
            let y = b.limbs.get(i).copied().unwrap_or(0) as i64;

                borrow = 1;
            } else {
                limbs.push((x - y) as u32);
                borrow = 0;
            }
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

            return;

            carry = value / Self::BASE;
        }

            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }

            index += 1;
        }
    }

                Self::mul_small_vec(&mut product_limbs, best as u32);
                let product = Self {
                    negative: false,
                    limbs: product_limbs,
                };
                remainder = Self::sub_abs(&remainder, &product);
            }
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            digits: vec![1],
        }
    }

    fn normalize(&mut self) {
        let first = self.digits.iter().position(|&d| d != 0);
        match first {
            Some(i) if i > 0 => {
                self.digits.drain(0..i);
            }
            None => self.digits = vec![0],
            _ => {}
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        self.digits
            .len()
            .cmp(&rhs.digits.len())
            .then_with(|| self.digits.cmp(&rhs.digits))
    }

    fn abs_add(&self, rhs: &Self) -> Vec<u8> {
        let mut out = Vec::new();
        let mut i = self.digits.len();
        let mut j = rhs.digits.len();
        let mut carry = 0u16;
        while i > 0 || j > 0 || carry > 0 {
            let a = if i > 0 {
                i -= 1;
                self.digits[i] as u16
            } else {
                0
            };
            let b = if j > 0 {
                j -= 1;
                rhs.digits[j] as u16
            } else {
                0
            };
            let sum = a + b + carry;
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        out.reverse();
        out
    }

    // Requires |self| >= |rhs|.
    fn abs_sub(&self, rhs: &Self) -> Vec<u8> {
        let mut out = Vec::with_capacity(self.digits.len());
        let mut i = self.digits.len();
        let mut j = rhs.digits.len();
        let mut borrow = 0i16;
        while i > 0 {
            i -= 1;
            let a = self.digits[i] as i16 - borrow;
            let b = if j > 0 {
                j -= 1;
                rhs.digits[j] as i16
            } else {
                0
            };
            if a < b {
                out.push((a + 10 - b) as u8);
                borrow = 1;
            } else {
                out.push((a - b) as u8);
                borrow = 0;
            }
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out.reverse();
        out
    }

    fn abs_mul(&self, rhs: &Self) -> Vec<u8> {
        if self.is_zero() || rhs.is_zero() {
            return vec![0];
        }
        let mut acc = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().rev().enumerate() {
            for (j, &b) in rhs.digits.iter().rev().enumerate() {
                acc[i + j] += a as u32 * b as u32;
            }
        }
        for i in 0..acc.len() - 1 {
            let carry = acc[i] / 10;
            acc[i] %= 10;
            acc[i + 1] += carry;
        }
        while acc.len() > 1 && acc.last() == Some(&0) {
            acc.pop();
        }
        acc.into_iter().rev().map(|x| x as u8).collect()
    }

    fn abs_mul_small(&mut self, m: u8) {
        if m == 0 || self.is_zero() {
            self.digits = vec![0];
            self.negative = false;
            return;
        }
        let mut carry = 0u16;
        for d in self.digits.iter_mut().rev() {
            let n = *d as u16 * m as u16 + carry;
            *d = (n % 10) as u8;
            carry = n / 10;
        }
        while carry > 0 {
            self.digits.insert(0, (carry % 10) as u8);
            carry /= 10;
        }
    }

    fn abs_add_small(&mut self, n: u8) {
        let mut carry = n as u16;
        let mut i = self.digits.len();
        while carry > 0 && i > 0 {
            i -= 1;
            let sum = self.digits[i] as u16 + carry;
            self.digits[i] = (sum % 10) as u8;
            carry = sum / 10;
        }
        if carry > 0 {
            self.digits.insert(0, carry as u8);
        }
    }

    fn abs_div_rem(&self, rhs: &Self) -> (Self, Self) {
        let mut quotient = Vec::with_capacity(self.digits.len());
        let mut rem = Self::zero();
        for &d in &self.digits {
            rem.abs_mul_small(10);
            rem.abs_add_small(d);
            let mut qdigit = 0u8;
            while rem.abs_cmp(rhs) != Ordering::Less {
                rem.digits = rem.abs_sub(rhs);
                qdigit += 1;
            }
            quotient.push(qdigit);
        }
        let mut q = Self {
            negative: false,
            digits: quotient,
        };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    fn abs(&self) -> Self {
        Self {
            negative: false,
            digits: self.digits.clone(),
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let mut values = vec![None; bytes.len()];
        let mut invalid = None;
        for i in start..bytes.len() {
            let b = bytes[i];
            if b == b'_' {
                continue;
            }
            let value = match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a' + 10) as u32),
                b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
                _ => None,
            };
            match value {
                Some(v) if v < radix => values[i] = Some(v as u8),
                _ => {
                    if invalid.is_none() {
                        invalid = Some(i);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || values[i - 1].is_none()
                    || values[i + 1].is_none())
            {
                separator = Some(i);
                break;
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && values[first_digit_index] == Some(0) {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }
        if digit_count > max_digits {
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(max_digits)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut result = Self::zero();
        for i in start..bytes.len() {
            if let Some(d) = values[i] {
                result.abs_mul_small(radix as u8);
                result.abs_add_small(d);
            }
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut n = self.abs();
        let mut output = Vec::new();
        if n.is_zero() {
            output.push(0u8);
        } else {
            while !n.is_zero() {
                let (q, r) = n.div_small(radix as u8);
                output.push(r);
                n = q;
            }
            output.reverse();
        }
        while output.len() < min_digits.max(1) {
            output.insert(0, 0);
        }

        let mut s = String::new();
        if self.negative && !self.is_zero() {
            s.push('-');
        }
        for (i, &d) in output.iter().enumerate() {
            if group > 0 && i > 0 && (output.len() - i) % group == 0 {
                s.push('_');
            }
            s.push(if d < 10 {
                (b'0' + d) as char
            } else {
                (b'a' + d - 10) as char
            });
        }
        Ok(s)
    }

    fn div_small(&self, divisor: u8) -> (Self, u8) {
        let mut quotient = Vec::with_capacity(self.digits.len());
        let mut rem = 0u16;
        for &d in &self.digits {
            let value = rem * 10 + d as u16;
            quotient.push((value / divisor as u16) as u8);
            rem = value % divisor as u16;
        }
        let mut q = Self {
            negative: false,
            digits: quotient,
        };
        q.normalize();
        (q, rem as u8)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = if self.negative == rhs.negative {
            Self {
                negative: self.negative,
                digits: self.abs_add(rhs),
            }
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Greater => Self {
                    negative: self.negative,
                    digits: self.abs_sub(rhs),
                },
                Ordering::Less => Self {
                    negative: rhs.negative,
                    digits: rhs.abs_sub(self),
                },
                Ordering::Equal => Self::zero(),
            }
        };
        result.normalize();
        result
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            digits: self.abs_mul(rhs),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = self.abs_div_rem(rhs);
        q.negative = (self.negative ^ rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e > 0 {
            if e & 1 == 1 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let r = a.div_rem(&b).unwrap().1;
            a = b;
            b = r.abs();
        }
        a.negative = false;
        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn from_digits(digits: &[u32], radix: u32, negative: bool) -> Self {
        let mut value = Self::zero();
        for &digit in digits {
            value.mul_small_assign(radix);
            value.add_small_assign(digit);
        }
        value.negative = negative && !value.is_zero();
        value
    }

    fn mul_small_assign(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            if factor == 0 {
                self.limbs.clear();
                self.negative = false;
            }
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let product = *limb as u64 * factor as u64 + carry;
            *limb = (product % Self::BASE) as u32;
            carry = product / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_assign(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len().max(b.limbs.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.limbs.len().max(b.limbs.len()) {
            let sum = a.limbs.get(i).copied().unwrap_or(0) as u64
                + b.limbs.get(i).copied().unwrap_or(0) as u64
                + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self {
            negative: false,
            limbs: out,
        }
    }

    // Requires |a| >= |b|.
    fn abs_sub(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let mut value = a.limbs[i] as i64
                - b.limbs.get(i).copied().unwrap_or(0) as i64
                - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u32);
        }
        let mut result = Self {
            negative: false,
            limbs: out,
        };
        result.normalize();
        result
    }

    fn abs(&self) -> Self {
        let mut out = self.clone();
        out.negative = false;
        out
    }

    fn div_rem_small_assign(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let current = rem * Self::BASE + *limb as u64;
            *limb = (current / divisor as u64) as u32;
            rem = current % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    fn abs_mul(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; a.limbs.len() + b.limbs.len()];
        for (i, &x) in a.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.limbs.iter().enumerate() {
                let k = i + j;
                let value = out[k] + x as u64 * y as u64 + carry;
                out[k] = value % Self::BASE;
                carry = value / Self::BASE;
            }
            let mut k = i + b.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let value = out[k] + carry;
                out[k] = value % Self::BASE;
                carry = value / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self {
            negative: false,
            limbs: out.into_iter().map(|x| x as u32).collect(),
        };
        result.normalize();
        result
    }

    // Adds a base-B limb to the low end: value = value * BASE + limb.
    fn shift_base_add(&mut self, limb: u32) {
        if self.is_zero() {
            if limb != 0 {
                self.limbs.push(limb);
            }
            return;
        }
        self.limbs.insert(0, limb);
    }

    fn abs_div_rem(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.abs_cmp(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }

        let mut quotient = vec![0u32; a.limbs.len()];
        let mut remainder = Self::zero();

        for i in (0..a.limbs.len()).rev() {
            remainder.shift_base_add(a.limbs[i]);

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = b.mul_small_abs(mid as u32);
                match product.abs_cmp(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }
            quotient[i] = best as u32;
            if best != 0 {
                let product = b.mul_small_abs(best as u32);
                remainder = Self::abs_sub(&remainder, &product);
            }
        }

        let mut q = Self {
            negative: false,
            limbs: quotient,
        };
        q.normalize();
        remainder.normalize();
        (q, remainder)
    }

    fn mul_small_abs(&self, factor: u32) -> Self {
        let mut out = self.abs();
        out.mul_small_assign(factor);
        out
    }

    fn radix_digits(&self, radix: u32) -> Vec<u32> {
        if self.is_zero() {
            return vec![0];
        }
        let mut value = self.abs();
        let mut digits = Vec::new();
        while !value.is_zero() {
            digits.push(value.div_rem_small_assign(radix));
        }
        digits.reverse();
        digits
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (start, negative) = match bytes[0] {
            b'+' => (1, false),
            b'-' => (1, true),
            _ => (0, false),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && digit_value(byte).is_none_or(|d| d >= radix) {
                invalid = Some(index);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                malformed_separator = Some(i);
                break;
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let mut seen = 0;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == max_digits.saturating_add(1) {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let digits: Vec<u32> = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| digit_value(b).unwrap())
            .collect();
        Ok(Self::from_digits(&digits, radix, negative))
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let digits = self.radix_digits(radix);
        let padding = min_digits.saturating_sub(digits.len());
        let total = padding + digits.len();
        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }

        for i in 0..total {
            if group > 0 && i > 0 && (total - i) % group == 0 {
                out.push('_');
            }
            let digit = if i < padding { 0 } else { digits[i - padding] };
            out.push(digit_char(digit));
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => rhs.abs_cmp(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self::abs_add(self, rhs);
            out.negative = self.negative && !out.is_zero();
            out
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self::abs_sub(self, rhs);
                    out.negative = self.negative;
                    out
                }
                Ordering::Less => {
                    let mut out = Self::abs_sub(rhs, self);
                    out.negative = rhs.negative;
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        let mut negated = rhs.clone();
        negated.negative = !negated.negative && !negated.is_zero();
        self.add(&negated)
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut out = Self::abs_mul(self, rhs);
        out.negative = !out.is_zero() && (self.negative ^ rhs.negative);
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::abs_div_rem(self, rhs);
        quotient.negative = !quotient.is_zero() && (self.negative ^ rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }
        let mut exp = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while exp != 0 {
            if exp & 1 == 1 {
                result = result.mul(&base);
            }
            exp >>= 1;
            if exp != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = Self::abs_div_rem(&a, &b);
            a = b;
            b = remainder;
        }
        a.negative = false;
        a
    }
}

fn digit_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
        b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
        _ => None,
    }
}

fn digit_char(digit: u32) -> char {
    match digit {
        0..=9 => (b'0' + digit as u8) as char,
        _ => (b'a' + (digit - 10) as u8) as char,
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

const BASE: u64 = 1_000_000_000;
const POW_LIMIT: u32 = 1024;

impl Big {
    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_values = Vec::new();
        let mut digit_offsets = Vec::new();

        // Invalid bytes take precedence over malformed separators.
        let mut invalid = None;
        for (i, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            match digit_value(byte) {
                Some(v) if v < radix => {
                    digit_values.push(v);
                    digit_offsets.push(i);
                }
                _ => {
                    invalid.get_or_insert(i);
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                return Err(Error::Separator { index: i });
            }
        }

        if digit_values.len() > 1 && digit_values[0] == 0 {
            return Err(Error::LeadingZero {
                index: digit_offsets[0],
            });
        }
        if digit_values.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_offsets[max_digits],
            });
        }

        let mut result = Self::zero();
        for digit in digit_values {
            result.mul_small_assign(radix);
            result.add_small_assign(digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0);
        } else {
            while !n.is_zero() {
                let rem = n.div_small_assign(radix);
                digits.push(rem as u8);
            }
            digits.reverse();
        }

        if digits.len() < min_digits {
            let mut padded = vec![0; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(digit_char(digit as u32));
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (a, b) if a != b => a.cmp(&b),
            (0, 0) => Ordering::Equal,
            (1, 1) => cmp_abs(self, rhs),
            (-1, -1) => cmp_abs(rhs, self),
            _ => Ordering::Equal,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self {
                negative: self.negative,
                limbs: add_abs(&self.limbs, &rhs.limbs),
            };
            out.normalize();
            out
        } else {
            match cmp_abs(self, rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self {
                        negative: self.negative,
                        limbs: sub_abs(&self.limbs, &rhs.limbs),
                    };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self {
                        negative: rhs.negative,
                        limbs: sub_abs(&rhs.limbs, &self.limbs),
                    };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let idx = i + j;
                let total = limbs[idx] as u64 + a as u64 * b as u64 + carry;
                limbs[idx] = (total % BASE) as u32;
                carry = total / BASE;
            }
            let mut idx = i + rhs.limbs.len();
            while carry != 0 {
                let total = limbs[idx] as u64 + carry;
                limbs[idx] = (total % BASE) as u32;
                carry = total / BASE;
                idx += 1;
                if idx == limbs.len() && carry != 0 {
                    limbs.push(0);
                }
            }
        }
        let mut out = Self {
            negative: self.negative != rhs.negative,
            limbs,
        };
        out.normalize();
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let a = self.abs();
        let b = rhs.abs();
        let (q_abs, r_abs) = div_rem_abs(&a, &b);
        let mut quotient = q_abs;
        quotient.negative = !quotient.is_zero() && (self.negative != rhs.negative);
        let mut remainder = r_abs;
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: POW_LIMIT,
            });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e > 0 {
            if e & 1 == 1 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, r) = div_rem_abs(&a, &b);
            a = b;
            b = r;
        }
        a
    }

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn abs(&self) -> Self {
        let mut out = self.clone();
        out.negative = false;
        out
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn mul_small_assign(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let n = *limb as u64 * factor as u64 + carry;
            *limb = (n % BASE) as u32;
            carry = n / BASE;
        }
        if carry != 0 {
            self.limbs.push(carry as u32);
        }
    }

    fn add_small_assign(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let n = self.limbs[i] as u64 + carry;
            self.limbs[i] = (n % BASE) as u32;
            carry = n / BASE;
            i += 1;
        }
    }

    fn div_small_assign(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let n = rem * BASE + *limb as u64;
            *limb = (n / divisor as u64) as u32;
            rem = n % divisor as u64;
        }
        self.normalize();
        rem as u32
    }
}

fn digit_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
        b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
        _ => None,
    }
}

fn digit_char(value: u32) -> char {
    match value {
        0..=9 => (b'0' + value as u8) as char,
        _ => (b'a' + (value - 10) as u8) as char,
    }
}

fn cmp_abs(a: &Big, b: &Big) -> Ordering {
    match a.limbs.len().cmp(&b.limbs.len()) {
        Ordering::Equal => {
            for (x, y) in a.limbs.iter().rev().zip(b.limbs.iter().rev()) {
                match x.cmp(y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u64;
    for i in 0..a.len().max(b.len()) {
        let n = a.get(i).copied().unwrap_or(0) as u64
            + b.get(i).copied().unwrap_or(0) as u64
            + carry;
        out.push((n % BASE) as u32);
        carry = n / BASE;
    }
    if carry != 0 {
        out.push(carry as u32);
    }
    out
}

fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i64;
    for (i, &av) in a.iter().enumerate() {
        let mut n = av as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
        if n < 0 {
            n += BASE as i64;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(n as u32);
    }
    while out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn div_rem_abs(a: &Big, b: &Big) -> (Big, Big) {
    if cmp_abs(a, b) == Ordering::Less {
        return (Big::zero(), a.abs());
    }
    if b.limbs.len() == 1 {
        let mut q = a.abs();
        let rem = q.div_small_assign(b.limbs[0]);
        return (
            q,
            Big {
                negative: false,
                limbs: if rem == 0 { Vec::new() } else { vec![rem] },
            },
        );
    }

    let mut quotient = vec![0u32; a.limbs.len()];
    let mut remainder = Big::zero();

    for i in (0..a.limbs.len()).rev() {
        remainder.limbs.insert(0, a.limbs[i]);
        remainder.normalize();

        let mut low = 0u64;
        let mut high = BASE - 1;
        let mut best = 0u64;
        while low <= high {
            let mid = (low + high) / 2;
            let product = mul_abs_small(b, mid as u32);
            match cmp_abs(&product, &remainder) {
                Ordering::Greater => {
                    if mid == 0 {
                        break;
                    }
                    high = mid - 1;
                }
                _ => {
                    best = mid;
                    low = mid + 1;
                }
            }
        }
        quotient[i] = best as u32;
        if best != 0 {
            remainder.limbs = sub_abs(&remainder.limbs, &mul_abs_small(b, best as u32).limbs);
            remainder.normalize();
        }
    }

    let mut q = Big {
        negative: false,
        limbs: quotient,
    };
    q.normalize();
    (q, remainder)
}

fn mul_abs_small(value: &Big, factor: u32) -> Big {
    if factor == 0 || value.is_zero() {
        return Big::zero();
    }
    let mut out = value.abs();
    out.mul_small_assign(factor);
    out
}
