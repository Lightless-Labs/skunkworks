#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;

    if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = tag_end + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8, radix: u8) {
    let mut carry = addend as u16;
    for digit in value.iter_mut() {
        let n = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (n % radix as u16) as u8;
        carry = n / radix as u16;
    }
    while carry > 0 {
        value.push((carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

fn add_values(left: &[u8], right: &[u8], radix: u8) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for i in 0..left.len().max(right.len()) {
        let sum = left.get(i).copied().unwrap_or(0) as u16
            + right.get(i).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum % radix as u16) as u8);
        carry = sum / radix as u16;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn convert_value(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        multiply_add(&mut value, from, digit, to);
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    value
}

fn format_value(value: &[u8], style: BaseStyle) -> String {
    let mut chars = Vec::with_capacity(value.len().max(style.min_width));
    for &digit in value.iter().rev() {
        let ch = match digit {
            0..=9 => (b'0' + digit) as char,
            _ if style.upper => (b'A' + digit - 10) as char,
            _ => (b'a' + digit - 10) as char,
        };
        chars.push(ch);
    }

    if chars.is_empty() {
        chars.push('0');
    }
    while chars.len() < style.min_width {
        chars.insert(0, '0');
    }

    if style.group == 0 {
        return chars.into_iter().collect();
    }

    let mut output = String::with_capacity(chars.len() + chars.len() / style.group);
    for (index, ch) in chars.iter().enumerate() {
        if index > 0 && (chars.len() - index) % style.group == 0 {
            output.push('_');
        }
        output.push(*ch);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    let value = convert_value(&digits, from, to);
    Ok(format_value(&value, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = convert_value(&left_digits, base, base);
    let right_value = convert_value(&right_digits, base, base);
    let sum = add_values(&left_value, &right_value, base);
    Ok(format_value(&sum, style))
}
