    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if (2..=36).contains(&value) {
        Ok(())
    } else {
        Err(BaseError::BadBase { which, value })
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group <= 8 && style.min_width <= 256 {
        Ok(())
    } else {
        Err(BaseError::BadStyle)
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;

    if let Some(hash_index) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash_index] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash_index + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let mut digits = Vec::new();

    // Invalid digits take precedence over separator and leading-zero errors.
    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push((index, value)),
            _ => return Err(BaseError::InvalidDigit { operand, index }),
        }
    }

    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            let has_digit_before = index > payload_start
                && bytes[index - 1] != b'_'
                && digit_value(bytes[index - 1]).is_some_and(|value| value < radix);
            let has_digit_after = index + 1 < bytes.len()
                && bytes[index + 1] != b'_'
                && digit_value(bytes[index + 1]).is_some_and(|value| value < radix);
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator { operand, index });
            }
        }
    }

    if digits.len() >= 2 && digits[0].1 == 0 {
        return Err(BaseError::LeadingZero { operand, index: payload_start });
    }

    Ok(digits.into_iter().map(|(_, value)| value).collect())
}

// Big unsigned integers are stored as little-endian base-1,000,000,000 limbs.
const LIMB_BASE: u64 = 1_000_000_000;

fn multiply_add(limbs: &mut Vec<u32>, multiplier: u32, addend: u32) {
    let mut carry = addend as u64;
    for limb in limbs.iter_mut() {
        let value = (*limb as u64) * (multiplier as u64) + carry;
        *limb = (value % LIMB_BASE) as u32;
        carry = value / LIMB_BASE;
    }
    while carry != 0 {
        limbs.push((carry % LIMB_BASE) as u32);
        carry /= LIMB_BASE;
    }
}

fn add_limbs(left: &[u32], right: &[u32]) -> Vec<u32> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for index in 0..left.len().max(right.len()) {
        let sum = left.get(index).copied().unwrap_or(0) as u64
            + right.get(index).copied().unwrap_or(0) as u64
            + carry;
        result.push((sum % LIMB_BASE) as u32);
        carry = sum / LIMB_BASE;
    }
    if carry != 0 {
        result.push(carry as u32);
    }
    result
}

fn limbs_to_digits(limbs: &[u32], radix: u8) -> Vec<u8> {
    if limbs.iter().all(|&limb| limb == 0) {
        return vec![0];
    }

    let mut work = limbs.to_vec();
    let mut digits = Vec::new();
    while work.iter().any(|&limb| limb != 0) {
        let mut remainder = 0u64;
        for limb in work.iter_mut().rev() {
            let value = remainder * LIMB_BASE + *limb as u64;
            *limb = (value / radix as u64) as u32;
            remainder = value % radix as u64;
        }
        digits.push(remainder as u8);
        while work.last() == Some(&0) {
            work.pop();
        }
    }
    digits.reverse();
    digits
}

fn render(digits: &[u8], style: BaseStyle) -> String {
    let padding = style.min_width.saturating_sub(digits.len());
    let total_digits = padding + digits.len();
    let separators = if style.group == 0 {
        0
    } else {
        (total_digits - 1) / style.group
    };

    let mut output = String::with_capacity(total_digits + separators);
    let mut digit_index = 0usize;

    for index in 0..total_digits {
        if index != 0
            && style.group != 0
            && (total_digits - index) % style.group == 0
        {
            output.push('_');
        }

        let digit = if index < padding {
            0
        } else {
            let value = digits[digit_index];
            digit_index += 1;
            value
        };

        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(byte as char);
    }

    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let mut limbs = vec![0u32];
    for digit in digits {
        multiply_add(&mut limbs, from as u32, digit as u32);
    }

    Ok(render(&limbs_to_digits(&limbs, to), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_limbs = vec![0u32];
    for digit in left_digits {
        multiply_add(&mut left_limbs, base as u32, digit as u32);
    }

    let mut right_limbs = vec![0u32];
    for digit in right_digits {
        multiply_add(&mut right_limbs, base as u32, digit as u32);
    }

    let sum = add_limbs(&left_limbs, &right_limbs);
    Ok(render(&limbs_to_digits(&sum, base), style))
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for byte in value.iter_mut().rev() {
        let n = (*byte as u16) * (multiplier as u16) + carry;
        *byte = n as u8;
        carry = n >> 8;
    }
    while carry != 0 {
        value.insert(0, carry as u8);
        carry >>= 8;
    }
}

fn add_bigints(left: &[u8], right: &[u8]) -> Vec<u8> {
    let len = left.len().max(right.len());
    let mut out = vec![0u8; len + 1];
    let mut carry = 0u16;

    for i in 0..len {
        let a = left
            .len()
            .checked_sub(i + 1)
            .map_or(0, |idx| left[idx]) as u16;
        let b = right
            .len()
            .checked_sub(i + 1)
            .map_or(0, |idx| right[idx]) as u16;
        let sum = a + b + carry;
        out[len - i] = sum as u8;
        carry = sum >> 8;
    }
    out[0] = carry as u8;
    if out[0] == 0 {
        out.remove(0);
    }
    out
}

fn bigint_to_digits(value: &[u8], base: u8) -> Vec<u8> {
    if value.is_empty() || value.iter().all(|&b| b == 0) {
        return vec![0];
    }

    let mut number = value.to_vec();
    let mut digits = Vec::new();

    while number.iter().any(|&b| b != 0) {
        let mut quotient = Vec::with_capacity(number.len());
        let mut remainder = 0u16;

        for &byte in &number {
            let current = (remainder << 8) | byte as u16;
            let q = (current / base as u16) as u8;
            remainder = current % base as u16;
            if !quotient.is_empty() || q != 0 {
                quotient.push(q);
            }
        }

        digits.push(remainder as u8);
        number = quotient;
    }

    digits.reverse();
    digits
}

fn format_digits(mut digits: Vec<u8>, style: BaseStyle) -> String {
    if digits.len() < style.min_width {
        let mut padded = vec![0; style.min_width - digits.len()];
        padded.append(&mut digits);
        digits = padded;
    }

    let mut chars = Vec::with_capacity(digits.len() + digits.len() / 2);
    for (i, &digit) in digits.iter().enumerate() {

}

    for limb in value.iter_mut().rev() {
        let current = remainder * LIMB_BASE + *limb as u64;
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn render(value: &[u32], radix: u8, style: BaseStyle) -> String {
    let mut number = value.to_vec();
    while number.len() > 1 && number.last() == Some(&0) {
        number.pop();
    }

    let mut digits = Vec::new();
    if number.len() == 1 && number[0] == 0 {
        digits.push(0);
    } else {
        while !(number.len() == 1 && number[0] == 0) {
            digits.push(divide_small(&mut number, radix));
        }
    }

    while digits.len() < style.min_width {
        digits.push(0);
    }
    digits.reverse();

    let mut output = String::new();
    for (index, digit) in digits.iter().copied().enumerate() {
        if style.group != 0 && index != 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(byte as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(render(&value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    let sum = add_values(&left_value, &right_value);
    Ok(render(&sum, base, style))
}

pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;

    if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = tag_end + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8, radix: u8) {
    let mut carry = addend as u16;
    for digit in value.iter_mut() {
        let n = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (n % radix as u16) as u8;
        carry = n / radix as u16;
    }
    while carry > 0 {
        value.push((carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

fn add_values(left: &[u8], right: &[u8], radix: u8) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for i in 0..left.len().max(right.len()) {
        let sum = left.get(i).copied().unwrap_or(0) as u16
            + right.get(i).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum % radix as u16) as u8);
        carry = sum / radix as u16;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn convert_value(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        multiply_add(&mut value, from, digit, to);
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    value
}

fn format_value(value: &[u8], style: BaseStyle) -> String {
    let mut chars = Vec::with_capacity(value.len().max(style.min_width));
    for &digit in value.iter().rev() {
        let ch = match digit {
            0..=9 => (b'0' + digit) as char,
            _ if style.upper => (b'A' + digit - 10) as char,
            _ => (b'a' + digit - 10) as char,
        };
        chars.push(ch);
    }

    if chars.is_empty() {
        chars.push('0');
    }
    while chars.len() < style.min_width {
        chars.insert(0, '0');
    }

    if style.group == 0 {
        return chars.into_iter().collect();
    }

    let mut output = String::with_capacity(chars.len() + chars.len() / style.group);
    for (index, ch) in chars.iter().enumerate() {
        if index > 0 && (chars.len() - index) % style.group == 0 {
            output.push('_');
        }
        output.push(*ch);
    }
    output
}

pub fn convert_base(
    input: &str,
