#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if (2..=36).contains(&value) {
        Ok(())
    } else {
        Err(BaseError::BadBase { which, value })
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group <= 8 && style.min_width <= 256 {
        Ok(())
    } else {
        Err(BaseError::BadStyle)
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;

    if let Some(hash_index) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash_index] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash_index + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let mut digits = Vec::new();

    // Invalid digits take precedence over separator and leading-zero errors.
    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push((index, value)),
            _ => return Err(BaseError::InvalidDigit { operand, index }),
        }
    }

    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            let has_digit_before = index > payload_start
                && bytes[index - 1] != b'_'
                && digit_value(bytes[index - 1]).is_some_and(|value| value < radix);
            let has_digit_after = index + 1 < bytes.len()
                && bytes[index + 1] != b'_'
                && digit_value(bytes[index + 1]).is_some_and(|value| value < radix);
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator { operand, index });
            }
        }
    }

    if digits.len() >= 2 && digits[0].1 == 0 {
        return Err(BaseError::LeadingZero { operand, index: payload_start });
    }

    Ok(digits.into_iter().map(|(_, value)| value).collect())
}

// Big unsigned integers are stored as little-endian base-1,000,000,000 limbs.
const LIMB_BASE: u64 = 1_000_000_000;

fn multiply_add(limbs: &mut Vec<u32>, multiplier: u32, addend: u32) {
    let mut carry = addend as u64;
    for limb in limbs.iter_mut() {
        let value = (*limb as u64) * (multiplier as u64) + carry;
        *limb = (value % LIMB_BASE) as u32;
        carry = value / LIMB_BASE;
    }
    while carry != 0 {
        limbs.push((carry % LIMB_BASE) as u32);
        carry /= LIMB_BASE;
    }
}

fn add_limbs(left: &[u32], right: &[u32]) -> Vec<u32> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for index in 0..left.len().max(right.len()) {
        let sum = left.get(index).copied().unwrap_or(0) as u64
            + right.get(index).copied().unwrap_or(0) as u64
            + carry;
        result.push((sum % LIMB_BASE) as u32);
        carry = sum / LIMB_BASE;
    }
    if carry != 0 {
        result.push(carry as u32);
    }
    result
}

fn limbs_to_digits(limbs: &[u32], radix: u8) -> Vec<u8> {
    if limbs.iter().all(|&limb| limb == 0) {
        return vec![0];
    }

    let mut work = limbs.to_vec();
    let mut digits = Vec::new();
    while work.iter().any(|&limb| limb != 0) {
        let mut remainder = 0u64;
        for limb in work.iter_mut().rev() {
            let value = remainder * LIMB_BASE + *limb as u64;
            *limb = (value / radix as u64) as u32;
            remainder = value % radix as u64;
        }
        digits.push(remainder as u8);
        while work.last() == Some(&0) {
            work.pop();
        }
    }
    digits.reverse();
    digits
}

fn render(digits: &[u8], style: BaseStyle) -> String {
    let padding = style.min_width.saturating_sub(digits.len());
    let total_digits = padding + digits.len();
    let separators = if style.group == 0 {
        0
    } else {
        (total_digits - 1) / style.group
    };

    let mut output = String::with_capacity(total_digits + separators);
    let mut digit_index = 0usize;

    for index in 0..total_digits {
        if index != 0
            && style.group != 0
            && (total_digits - index) % style.group == 0
        {
            output.push('_');
        }

        let digit = if index < padding {
            0
        } else {
            let value = digits[digit_index];
            digit_index += 1;
            value
        };

        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(byte as char);
    }

    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let mut limbs = vec![0u32];
    for digit in digits {
        multiply_add(&mut limbs, from as u32, digit as u32);
    }

    Ok(render(&limbs_to_digits(&limbs, to), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_limbs = vec![0u32];
    for digit in left_digits {
        multiply_add(&mut left_limbs, base as u32, digit as u32);
    }

    let mut right_limbs = vec![0u32];
    for digit in right_digits {
        multiply_add(&mut right_limbs, base as u32, digit as u32);
    }

    let sum = add_limbs(&left_limbs, &right_limbs);
    Ok(render(&limbs_to_digits(&sum, base), style))
}
