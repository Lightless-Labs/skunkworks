#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, Debug, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone, Debug)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    has_wildcard: bool,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(input: &str, base: usize) -> Result<String, usize> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_value(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_value(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut segments = Vec::new();
    let mut seen_names = std::collections::HashSet::<String>::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if end == start {
            if end == bytes.len() - 1 {
                return Ok((segments, true));
            }
            return Err(start);
        }

        let raw = &input[start..end];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen_names.insert(name.to_string()) {
                return Err(start);
            }
            if marker == b'*' && end != bytes.len() {
                return Err(start);
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            });
        } else {
            let value = decode_segment(raw, start)?;
            segments.push(Segment::Static(value));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Ok((segments, true));
        }
    }

    Ok((segments, false))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn encode_segment(value: &str) -> String {
    let mut result = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            result.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(HEX[(b >> 4) as usize] as char);
            result.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    result
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;

        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let route = Route {
            id: self.routes.len(),
            name: name.to_string(),
            method: method.to_string(),
            has_wildcard: segments.iter().any(|s| matches!(s, Segment::Wildcard(_))),
            segments,
            trailing,
        };

        if let Some(existing) = self
            .routes
            .iter()
            .filter(|existing| shape_conflicts(existing, &route, self.policy))
            .map(|existing| existing.id)
            .min()
        {
            return Err(Error::Conflict { with: existing });
        }

        let id = route.id;
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) =
            parse_path(head, false).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>, Vec<u8>)> = Vec::new();

        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }

            let mut params = Vec::new();
            let mut ri = 0;
            let mut pi = 0;
            let mut matched = true;

            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Static(expected) => {
                        if request_segments.get(pi) != Some(&Segment::Static(expected.clone())) {
                            matched = false;
                            break;
                        }
                        ri += 1;
                        pi += 1;
                    }
                    Segment::Param(name) => {
                        match request_segments.get(pi) {
                            Some(Segment::Static(value)) => {
                                params.push((name.clone(), value.clone()));
                                ri += 1;
                                pi += 1;
                            }
                            _ => {
                                matched = false;
                                break;
                            }
                        }
                    }
                    Segment::Wildcard(name) => {
                        let tail = request_segments[pi..]
                            .iter()
                            .map(|segment| match segment {
                                Segment::Static(value) => value.as_str(),
                                _ => "",
                            })
                            .collect::<Vec<_>>()
                            .join("/");
                        if request_segments[pi..]
                            .iter()
                            .any(|segment| !matches!(segment, Segment::Static(_)))
                        {
                            matched = false;
                            break;
                        }
                        params.push((name.clone(), tail));
                        pi = request_segments.len();
                        ri += 1;
                    }
                }
            }

            if matched && pi == request_segments.len() {
                let mut rank = Vec::new();
                for segment in &route.segments {
                    match segment {
                        Segment::Static(_) => rank.push(2),
                        Segment::Param(_) => rank.push(1),
                        Segment::Wildcard(_) => rank.push(0),
                    }
                }
                if !route.has_wildcard {
                    rank.push(3);
                }
                candidates.push((route, params, rank));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut eligible_methods: Vec<String> =
            candidates.iter().map(|(route, _, _)| route.method.clone()).collect();
        eligible_methods.sort();
        eligible_methods.dedup();
        if eligible_methods.iter().any(|m| m == "GET") && !eligible_methods.iter().any(|m| m == "HEAD") {
            eligible_methods.push("HEAD".to_string());
            eligible_methods.sort();
        }

        let head_has_route = method == "HEAD"
            && candidates.iter().any(|(route, _, _)| route.method == "HEAD");
        let wanted = if method == "HEAD" && !head_has_route {
            "GET"
        } else {
            method
        };

        let selected = candidates
            .into_iter()
            .filter(|(route, _, _)| route.method == wanted)
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.0.id.cmp(&a.0.id)));

        let Some((route, params, _)) = selected else {
            return Err(Error::MethodNotAllowed {
                allowed: eligible_methods,
            });
        };

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !head.is_empty()
            && head != "/"
        {
            if route.trailing {
                Some(format!("{}/", head))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::<&str, &str>::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extra: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !required.contains(key))
            .collect();
        extra.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extra.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for segment in &route.segments {
            let name = match segment {
                Segment::Param(name) | Segment::Wildcard(name) => name,
                Segment::Static(_) => continue,
            };
            let value = supplied[name.as_str()];
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid = match segment {
                Segment::Param(_) => {
                    valid_ascii && !value.is_empty() && !value.as_bytes().contains(&b'/')
                }
                Segment::Wildcard(_) => {
                    valid_ascii
                        && (value.is_empty()
                            || value
                                .split('/')
                                .all(|part| !part.is_empty()))
                }
                Segment::Static(_) => true,
            };
            if !valid {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut output = String::from("/");
        let mut first = true;

        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if !first {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(value));
                    first = false;
                }
                Segment::Param(name) => {
                    if !first {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(supplied[name.as_str()]));
                    first = false;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        if !first {
                            output.push('/');
                        }
                        output.push_str(
                            &value
                                .split('/')
                                .map(encode_segment)
                                .collect::<Vec<_>>()
                                .join("/"),
                        );
                        first = false;
                    }
                }
            }
        }

        if first {
            output = "/".to_string();
        } else if route.trailing {
            output.push('/');
        }

        Ok(output)
    }
}
