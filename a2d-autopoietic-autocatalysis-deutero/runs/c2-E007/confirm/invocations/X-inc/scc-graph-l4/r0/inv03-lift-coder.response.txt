use std::cmp::Reverse;
use std::collections::{BTreeMap, BTreeSet, BinaryHeap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EdgeStatus {
    Duplicate,
    Internal,
    Link,
    Merge,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Step {
    pub edge_index: usize,
    pub status: EdgeStatus,
    pub before: (usize, usize),
    pub after: (usize, usize),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub edge_index: usize,
    pub before: Vec<usize>,
    pub after: usize,
    pub members: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Batch {
    pub steps: Vec<Step>,
    pub merges: Vec<Merge>,
    pub graph: Graph,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn validate(n: usize, edges: &[(usize, usize)]) -> Result<(), GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }
    Ok(())
}

fn build_graph(n: usize, edges: &[(usize, usize)]) -> Graph {
    let mut adj = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        adj[from].push(to);
    }

    // Iterative Kosaraju traversal avoids recursion on long paths.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next_index)) = stack.pop() {
            if next_index < adj[v].len() {
                stack.push((v, next_index + 1));
                let w = adj[v][next_index];
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                finish.push(v);
            }
        }
    }

    let mut rev = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        rev[to].push(from);
    }

    let mut assigned = vec![false; n];
    let mut raw_components = Vec::<Vec<usize>>::new();
    for &root in finish.iter().rev() {
        if assigned[root] {
            continue;
        }
        assigned[root] = true;
        let mut members = Vec::new();
        let mut stack = vec![root];
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if !assigned[w] {
                    assigned[w] = true;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    raw_components.sort_by(|a, b| {
        b.len()
            .cmp(&a.len())
            .then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0usize; n];
    for (id, members) in raw_components.iter().enumerate() {
        for &v in members {
            component_of[v] = id;
        }
    }

    let count = raw_components.len();
    let mut cyclic: Vec<bool> = raw_components.iter().map(|c| c.len() >= 2).collect();
    let mut internal_edges = vec![0usize; count];
    let mut arc_map = BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_map.entry((a, b)).or_insert((0, edge_index));
            entry.0 += 1;
            entry.1 = entry.1.min(edge_index);
        }
    }

    let mut arcs: Vec<Arc> = arc_map
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::<usize>::new(); count];
    let mut indegree = vec![0usize; count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    // Heap order: greatest internal edge count, then greatest smallest member.
    let mut available = BinaryHeap::<(usize, usize, Reverse<usize>)>::new();
    for id in 0..count {
        if indegree[id] == 0 {
            available.push((
                internal_edges[id],
                raw_components[id][0],
                Reverse(id),
            ));
        }
    }

    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while let Some((_, _, Reverse(v))) = available.pop() {
        topo.push(v);
        for &w in &outgoing[v] {
            depth[w] = depth[w].max(depth[v] + 1);
            indegree[w] -= 1;
            if indegree[w] == 0 {
                available.push((
                    internal_edges[w],
                    raw_components[w][0],
                    Reverse(w),
                ));
            }
        }
    }

    Graph {
        components: raw_components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    }
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;
    Ok(build_graph(n, edges))
}

fn shortest_lex_route(graph: &Graph, start: usize, target: usize) -> Option<Vec<usize>> {
    if start == target {
        return Some(vec![start]);
    }
    let count = graph.components.len();
    let mut outgoing = vec![Vec::<usize>::new(); count];
    for arc in &graph.arcs {
        outgoing[arc.from].push(arc.to);
    }
    for neighbors in &mut outgoing {
        neighbors.sort_unstable();
        neighbors.dedup();
    }

    let mut distance = vec![usize::MAX; count];
    let mut queue = VecDeque::new();
    distance[target] = 0;
    queue.push_back(target);

    let mut incoming = vec![Vec::<usize>::new(); count];
    for (from, neighbors) in outgoing.iter().enumerate() {
        for &to in neighbors {
            incoming[to].push(from);
        }
    }

    while let Some(v) = queue.pop_front() {
        for &predecessor in &incoming[v] {
            if distance[predecessor] == usize::MAX {
                distance[predecessor] = distance[v] + 1;
                queue.push_back(predecessor);
            }
        }
    }
    if distance[start] == usize::MAX {
        return None;
    }

    let mut route = vec![start];
    let mut current = start;
    while current != target {
        let next = outgoing[current]
            .iter()
            .copied()
            .filter(|&v| distance[v] + 1 == distance[current])
            .min()?;
        route.push(next);
        current = next;
    }
    Some(route)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let n = self.component_of.len();
        if from >= n {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        if let Some(route) = shortest_lex_route(self, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_route(self, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    graph: Graph,
    history: Vec<Merge>,
    pairs: BTreeSet<(usize, usize)>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        let graph = analyze(n, edges)?;
        let retained = edges.to_vec();
        let pairs = retained.iter().copied().collect();
        Ok(Self {
            n,
            edges: retained,
            graph,
            history: Vec::new(),
            pairs,
        })
    }

    pub fn snapshot(&self) -> Graph {
        self.graph.clone()
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, batch_edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + batch_edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(from, to)) in batch_edges.iter().enumerate() {
            if from >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: from,
                });
            }
            if to >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: to,
                });
            }
        }

        let mut steps = Vec::with_capacity(batch_edges.len());
        let mut merges = Vec::new();

        for &(from, to) in batch_edges {
            let edge_index = self.edges.len();
            let before = (
                self.graph.component_of[from],
                self.graph.component_of[to],
            );
            let duplicate = self.pairs.contains(&(from, to));

            self.edges.push((from, to));
            self.pairs.insert((from, to));
            self.graph = build_graph(self.n, &self.edges);

            let after = (
                self.graph.component_of[from],
                self.graph.component_of[to],
            );

            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };

            if status == EdgeStatus::Merge {
                let mut united_ids = BTreeSet::new();
                for vertex in &self.graph.components[after.0] {
                    // The old canonical component ID is represented by the
                    // pre-insertion graph's component_of vector.
                    united_ids.insert(
                        // All vertices of the new component are still present
                        // in the retained graph; recover their previous IDs.
                        // The previous graph is saved below for this purpose.
                        0usize,
                    );
                    let _ = vertex;
                }
                // Reconstruct the previous IDs from the graph before this
                // insertion by temporarily analyzing the retained prefix.
                let previous_graph = build_graph(self.n, &self.edges[..edge_index]);
                for &vertex in &self.graph.components[after.0] {
                    united_ids.insert(previous_graph.component_of[vertex]);
                }
                let before_ids: Vec<usize> = united_ids.into_iter().collect();
                let merge = Merge {
                    edge_index,
                    before: before_ids,
                    after: after.0,
                    members: self.graph.components[after.0].clone(),
                };
                self.history.push(merge.clone());
                merges.push(merge);
            }

            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.snapshot(),
        })
    }
}
