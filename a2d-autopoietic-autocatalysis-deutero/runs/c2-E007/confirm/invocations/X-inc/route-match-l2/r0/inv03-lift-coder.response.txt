```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }

        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(index, ch)| matches!(ch, '?' | '#').then_some(index))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head)?;

        let mut best: Option<(usize, Match)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &request_segments) {
                let candidate = Match {
                    route: route.id,
                    params,
                };

                let replace = match &best {
                    None => true,
                    Some((best_index, _)) => {
                        route.rank > self.routes[*best_index].rank
                    }
                };

                if replace {
                    best = Some((route.id, candidate));
                }
            }
        }

        best.map(|(_, matched)| matched).ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let pieces = split_segments(pattern);
    let mut result = Vec::with_capacity(pieces.len());
    let mut names = Vec::<String>::new();

    for (index, (start, raw)) in pieces.iter().enumerate() {
        if raw.is_empty() {
            if index + 1 == pieces.len() {
                return Err(Error::InvalidPattern {
                    at: pattern.len() - 1,
                });
            }
            return Err(Error::InvalidPattern { at: *start });
        }

        let segment = if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.starts_with('*');
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard && index + 1 != pieces.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_owned());
            if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            }
        } else {
            Segment::Static(decode_segment(raw, *start, true).map_err(|at| {
                Error::InvalidPattern { at }
            })?)
        };
        result.push(segment);
    }

    Ok(result)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let pieces = split_segments(path);
    let mut result = Vec::with_capacity(pieces.len());

    for (index, (start, raw)) in pieces.iter().enumerate() {
        if raw.is_empty() {
            if index + 1 == pieces.len() {
                return Err(Error::InvalidPath {
                    at: path.len() - 1,
                });
            }
            return Err(Error::InvalidPath { at: *start });
        }
        result.push(decode_segment(raw, *start, false).map_err(|at| Error::InvalidPath { at })?);
    }

    Ok(result)
}

fn split_segments(input: &str) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let mut start = 1;
    for (index, byte) in input.bytes().enumerate().skip(1) {
        if byte == b'/' {
            result.push((start, &input[start..index]));
            start = index + 1;
        }
    }
    result.push((start, &input[start..]));
    result
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn decode_segment(raw: &str, base: usize, pattern: bool) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' {
            if index + 2 >= bytes.len() {
                return Err(base + index);
            }
            let Some(high) = uppercase_hex(bytes[index + 1]) else {
                return Err(base + index);
            };
            let Some(low) = uppercase_hex(bytes[index + 2]) else {
                return Err(base + index);
            };
            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + index);
            }
            decoded.push(value);
            index += 3;
        } else {
            if !byte.is_ascii() || !is_unreserved(byte) {
                let _ = pattern;
                return Err(base + index);
            }
            decoded.push(byte);
            index += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segment is ASCII"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    (path_index == path.len()).then_some(params)
}
```
