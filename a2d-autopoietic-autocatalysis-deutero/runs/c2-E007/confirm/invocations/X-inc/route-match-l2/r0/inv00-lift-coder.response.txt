#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let segments = parse_path(head, 0)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(route, &segments) {
                let replace = match &best {
                    None => true,
                    Some((best_route, _)) => {
                        route.rank > best_route.rank
                            || (route.rank == best_route.rank && route.id < best_route.id)
                    }
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        best.map(|(route, params)| Match {
            route: route.id,
            params,
        })
        .ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                if i == bytes.len() {
                    return Err(Error::InvalidPattern { at: i - 1 });
                }
                return Err(Error::InvalidPattern { at: i });
            }
            parts.push((start, i));
            start = i + 1;
        }
        i += 1;
    }

    let mut segments = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (index, &(start, end)) in parts.iter().enumerate() {
        let raw = &pattern[start..end];
        if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.starts_with('*');
            if wildcard && index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }
    }

    Ok(segments)
}

fn parse_path(path: &str, base: usize) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: base });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                if i == bytes.len() {
                    return Err(Error::InvalidPath { at: base + i - 1 });
                }
                return Err(Error::InvalidPath { at: base + i });
            }
            parts.push((start, i));
            start = i + 1;
        }
        i += 1;
    }

    parts
        .into_iter()
        .map(|(start, end)| decode_segment(&path[start..end], base + start, false))
        .collect()
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    matches!(bytes.first(), Some(b'a'..=b'z'))
        && bytes[1..]
            .iter()
            .all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(offset + i));
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return Err(invalid(offset + i));
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return Err(invalid(offset + i));
            };
            let decoded = (high << 4) | low;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(invalid(offset + i));
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(invalid(offset + i));
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), path[path_index..].join("/")));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
