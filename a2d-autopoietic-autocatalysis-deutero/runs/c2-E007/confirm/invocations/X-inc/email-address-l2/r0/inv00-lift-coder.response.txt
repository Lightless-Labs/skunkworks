#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix = if bytes[0] == b'"' {
        let mut i = 1;
        let mut closing = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    closing = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match closing {
            Some(pos) => Some(pos),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(close) = quoted_prefix {
            if i <= close {
                i = close + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let local_bytes = &bytes[..sep];
    let local = if let Some(close) = quoted_prefix {
        if close >= sep {
            // The closing quote is within the local prefix because separators
            // after it are the only ones considered.
            if close + 1 != sep {
                return Err(EmailError::TrailingLocal { at: close + 1 });
            }

            let mut decoded = Vec::new();
            let mut origins = Vec::new();
            let mut j = 1;
            while j < close {
                if bytes[j] == b'\\' {
                    if j + 1 >= close {
                        return Err(EmailError::BadEscape { at: j });
                    }
                    let escaped = bytes[j + 1];
                    if !matches!(escaped, b'"' | b'\\' | b'@') {
                        return Err(EmailError::BadEscape { at: j });
                    }
                    decoded.push(escaped);
                    origins.push(j);
                    j += 2;
                } else {
                    let b = bytes[j];
                    if !(0x20..=0x7e).contains(&b) {
                        return Err(EmailError::QuotedChar { at: j });
                    }
                    decoded.push(b);
                    origins.push(j);
                    j += 1;
                }
            }

            if decoded.is_empty() {
                return Err(EmailError::EmptyLocal { at: close });
            }
            if decoded.len() > 64 {
                return Err(EmailError::LocalTooLong { at: origins[64] });
            }
            String::from_utf8(decoded).expect("quoted ASCII local")
        } else {
            validate_atom(local_bytes)?;
            String::from_utf8(local_bytes.to_vec()).expect("ASCII local")
        }
    } else {
        validate_atom(local_bytes)?;
        String::from_utf8(local_bytes.to_vec()).expect("ASCII local")
    };

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_end = if has_root_dot {
        bytes.len() - 1
    } else {
        bytes.len()
    };
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = domain_start;
    let mut pos = domain_start;
    while pos <= domain_end {
        if pos == domain_end || bytes[pos] == b'.' {
            if label_start == pos {
                let error_at = if pos == domain_end && has_root_dot {
                    domain_end
                } else {
                    pos
                };
                return Err(EmailError::EmptyLabel { at: error_at });
            }
            let label = &bytes[label_start..pos];
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: label_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos - 1 });
            }
            for (offset, &b) in label.iter().enumerate() {
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: label_start + offset,
                    });
                }
            }
            labels.push((label_start, label));
            label_start = pos + 1;
        }
        pos += 1;
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = domain_bytes
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted: quoted_prefix.is_some(),
    })
}

fn validate_atom(local: &[u8]) -> Result<(), EmailError> {
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: i });
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return Err(EmailError::LocalChar { at: i });
        }
    }
    Ok(())
}
