#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_side = |j: usize| {
            j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
        };
        if i == 0 || !valid_side(i - 1) || !valid_side(i + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let (byte_index, ch) = chars[i];
        if ch == '\r' {
            normalized.push(('\n', byte_index));
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 2;
            } else {
                i += 1;
            }
        } else {
            normalized.push((ch, byte_index));
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut cursor = 0;

    while cursor < normalized.len() {
        if normalized[cursor].0 == '\n' {
            let run_start = cursor;
            while cursor < normalized.len() && normalized[cursor].0 == '\n' {
                cursor += 1;
            }
            let run_len = cursor - run_start;
            if run_len >= 2 {
                layout_segment(&normalized[segment_start..run_start], options, &mut output);
                for _ in 1..run_len {
                    output.push(String::new());
                }
                segment_start = cursor;
            }
        } else {
            cursor += 1;
        }
    }
    layout_segment(&normalized[segment_start..], options, &mut output);

    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    text: String,
    pieces: usize,
}

fn layout_segment(segment: &[(char, usize)], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words = Vec::<Word>::new();
    let mut current = Vec::new();

    for &(ch, _) in segment {
        if is_separator(ch) {
            if !current.is_empty() {
                words.push(make_word(&current));
                current.clear();
            }
        } else {
            current.push(ch);
        }
    }
    if !current.is_empty() {
        words.push(make_word(&current));
    }
    if words.is_empty() {
        return;
    }

    let mut lines = Vec::<Line>::new();
    let mut line_number = 0usize;
    let mut line = new_line(line_number, options);
    let mut word_index = 0usize;

    while word_index < words.len() {
        let word = &words[word_index];
        let mut offset = 0usize;

        while offset < word.chars.len() {
            let capacity = options.width - line.indent;
            let line_len = line.text.chars().count();
            let separator_space = if line.pieces > 0 { 1 } else { 0 };
            let room = capacity.saturating_sub(line_len + separator_space);
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room {
                if line.pieces > 0 {
                    line.text.push(' ');
                }
                append_visible(&mut line.text, word, offset, word.chars.len());
                line.pieces += 1;
                offset = word.chars.len();
                continue;
            }

            let usable = word.markers.iter().copied()
                .filter(|&m| m > offset && m < word.chars.len())
                .filter(|&m| m - offset + 1 <= room)
                .collect::<Vec<_>>();

            let chosen = usable.last().copied();
            if line.pieces == 0 {
                if let Some(marker) = chosen {
                    append_visible(&mut line.text, word, offset, marker);
                    line.text.push('-');
                    line.pieces += 1;
                    lines.push(line);
                    line_number += 1;
                    line = new_line(line_number, options);
                    offset = marker + 1;
                    continue;
                }

                if suffix_len > capacity {
                    append_visible(&mut line.text, word, offset, offset + capacity);
                    line.pieces += 1;
                    lines.push(line);
                    line_number += 1;
                    line = new_line(line_number, options);
                    offset += capacity;
                    continue;
                }

                append_visible(&mut line.text, word, offset, word.chars.len());
                line.pieces += 1;
                offset = word.chars.len();
                continue;
            }

            let accepted = chosen.filter(|&marker| {
                let used = line_len + 1 + (marker - offset) + 1;
                let unused = capacity - used;
                let baseline = capacity - line_len;
                unused * unused + options.hyphen_cost < baseline * baseline
            });

            if let Some(marker) = accepted {
                line.text.push(' ');
                append_visible(&mut line.text, word, offset, marker);
                line.text.push('-');
                line.pieces += 1;
                lines.push(line);
                line_number += 1;
                line = new_line(line_number, options);
                offset = marker + 1;
            } else {
                lines.push(line);
                line_number += 1;
                line = new_line(line_number, options);
            }
        }

        word_index += 1;
    }

    if !line.text.is_empty() {
        lines.push(line);
    }

    for i in 0..lines.len() {
        let mut rendered = String::new();
        rendered.extend(std::iter::repeat_n(' ', lines[i].indent));
        rendered.push_str(&lines[i].text);

        let is_nonfinal = i + 1 < lines.len();
        let gaps = lines[i].pieces.saturating_sub(1);
        if options.justify && is_nonfinal && gaps >= 1 {
            let current_width = lines[i].indent + lines[i].text.chars().count();
            let extra = options.width.saturating_sub(current_width);
            let each = extra / gaps;
            let remainder = extra % gaps;
            let start = i % gaps;
            let mut justified = String::new();
            justified.extend(std::iter::repeat_n(' ', lines[i].indent));
            let mut seen_gap = 0usize;
            for ch in lines[i].text.chars() {
                if ch == ' ' {
                    justified.push(' ');
                    justified.extend(std::iter::repeat_n(' ', each));
                    if (seen_gap + gaps - start) % gaps < remainder {
                        justified.push(' ');
                    }
                    seen_gap += 1;
                } else {
                    justified.push(ch);
                }
            }
            rendered = justified;
        }
        output.push(rendered);
    }
}

fn make_word(raw: &[char]) -> Word {
    let mut chars = Vec::new();
    let mut markers = Vec::new();
    for &ch in raw {
        if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }
    Word { chars, markers }
}

fn append_visible(text: &mut String, word: &Word, start: usize, end: usize) {
    for &ch in &word.chars[start..end] {
        text.push(ch);
    }
}

fn new_line(number: usize, options: &WrapOptions) -> Line {
    Line {
        indent: if number == 0 {
            options.first_indent
        } else {
            options.rest_indent
        },
        text: String::new(),
        pieces: 0,
    }
}
