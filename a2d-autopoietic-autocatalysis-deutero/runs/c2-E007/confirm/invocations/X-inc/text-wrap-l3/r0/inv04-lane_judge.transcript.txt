pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    let is_separator = |ch: char| matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}');
    for (i, &(index, ch)) in chars.iter().enumerate() {
        if ch == '~' {
            let valid_side = |side: Option<char>| {
                side.is_some_and(|c| c != '~' && !is_separator(c))
            };
            if !valid_side(i.checked_sub(1).and_then(|j| chars.get(j).map(|x| x.1)))
                || !valid_side(chars.get(i + 1).map(|x| x.1))
            {
                return Err(WrapError::InvalidMarker { index });
            }
        }
    }

    let mut normalized = String::with_capacity(input.len());
    let mut iter = input.chars().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let normalized_chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;

    while i < normalized_chars.len() {
        if normalized_chars[i] == '\n' {
            let run_start = i;

                i += 1;

                segment_start = i;
            }
        } else {
            i += 1;
        }
    }
    layout_segment(&normalized_chars[segment_start..], options, &mut output);

    Ok(output)
}

fn layout_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let is_separator = |ch: char| matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}');
    let mut words: Vec<Piece> = Vec::new();
    let mut at = 0;
    while at < segment.len() {
        while at < segment.len() && is_separator(segment[at]) {
            at += 1;
        }
        if at == segment.len() {
            break;
        }

        let mut markers = Vec::new();
        while at < segment.len() && !is_separator(segment[at]) {
            if segment[at] == '~' {
                markers.push(text.len());
            } else {
                text.push(segment[at]);
            }
            at += 1;
        }
        words.push(Piece { text, markers });
    }

    if words.is_empty() {
        return;
    }

    let mut lines: Vec<Line> = Vec::new();
    let mut current = Line {
        indent: options.first_indent,
        text: Vec::new(),
        gaps: Vec::new(),
        pieces: 0,
    };
    let mut line_index = 0usize;

    for word in &words {
        let mut offset = 0usize;
        while offset < word.text.len() {
            let capacity = options.width - current.indent;
            let available = capacity.saturating_sub(current.text.len());
            let separator_space = usize::from(!current.text.is_empty());
            let suffix_len = word.text.len() - offset;

                    current.text.push(' ');

                current.pieces += 1;
                offset = word.text.len();
                continue;

                .markers
                .iter()
                .copied()

                .collect();

                let prefix_len = marker - offset;
                let unused = available - prefix_len - 1;
                let accept = current.text.is_empty()
                    || unused * unused + options.hyphen_cost
                        < available.saturating_mul(available);

                if accept {
                    if !current.text.is_empty() {
                        current.gaps.push(1);
                        current.text.push(' ');
                    }
                    current.text.extend_from_slice(&word.text[offset..marker]);
                    current.text.push('-');
                    current.pieces += 1;
                    lines.push(current);
                    line_index += 1;
                    current = Line {
                        indent: options.rest_indent,
                        text: Vec::new(),
                        gaps: Vec::new(),
                        pieces: 0,
                    };
                    offset = marker;
                    continue;
                }
            }

            if !current.text.is_empty() {
                lines.push(current);
                line_index += 1;
                current = Line {
                    indent: options.rest_indent,
                    text: Vec::new(),
                    gaps: Vec::new(),
                    pieces: 0,
                };
                continue;
            }

            let capacity = options.width - current.indent;
            if suffix_len > capacity {
                current.text.extend_from_slice(&word.text[offset..offset + capacity]);
                current.pieces += 1;
                lines.push(current);
                line_index += 1;
                current = Line {
                    indent: options.rest_indent,
                    text: Vec::new(),
                    gaps: Vec::new(),
                    pieces: 0,
                };
                offset += capacity;
            } else {
                current.text.extend_from_slice(&word.text[offset..]);
                current.pieces += 1;
                offset = word.text.len();
            }
        }
    }

    if !current.text.is_empty() {
        lines.push(current);
    }

    let final_index = lines.len().saturating_sub(1);
    for (idx, line) in lines.into_iter().enumerate() {
        let content_width = line.text.len();
        let should_justify =
            options.justify && idx != final_index && line.pieces >= 2 && !line.gaps.is_empty();
        let mut rendered = String::new();
        rendered.extend(std::iter::repeat_n(' ', line.indent));

        if should_justify {
            let extra = options.width - line.indent - content_width;
            let gaps = line.gaps.len();
            let each = extra / gaps;
            let remainder = extra % gaps;
            let start = line_index_for_justification(idx, &lines_placeholder());

            let rotation = (idx % gaps) % gaps;

                rendered.push(*ch);

                    let gap_idx = line.text[..=pos].iter().filter(|&&c| c == ' ').count() - 1;
                    let additional = each + usize::from((gap_idx + gaps - rotation) % gaps < remainder);
                    rendered.extend(std::iter::repeat_n(' ', additional));
                }
            }
        } else {
            rendered.extend(line.text);
        }
        output.push(rendered);
    }
}

pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
}

struct Line {
    indent: usize,
    pieces: Vec<Piece>,
    gaps: Vec<usize>,
    content_len: usize,
}

impl Line {
    fn new(indent: usize) -> Self {
        Self {
            indent,
            pieces: Vec::new(),
            gaps: Vec::new(),
            content_len: 0,
        }
    }

    fn is_empty(&self) -> bool {
        self.content_len == 0
    }

    fn push_piece(&mut self, text: String) {
        if !self.pieces.is_empty() {
            self.gaps.push(1);
            self.content_len += 1;
        }
        self.content_len += text.chars().count();
        self.pieces.push(Piece { text });
    }

    fn render(&self, width: usize, justify: bool, final_line: bool, line_index: usize) -> String {
        let mut output = " ".repeat(self.indent);

        if justify && !final_line && self.pieces.len() >= 2 {
            let gap_count = self.gaps.len();
            let extra = width.saturating_sub(self.indent + self.content_len);
            let each = extra / gap_count;
            let remainder = extra % gap_count;
            let start = line_index % gap_count;

            for (i, piece) in self.pieces.iter().enumerate() {
                output.push_str(&piece.text);
                if i < gap_count {
                    let bonus = usize::from((i + gap_count - start) % gap_count < remainder);
                    output.push_str(&" ".repeat(1 + each + bonus));
                }
            }
        } else {
            for (i, piece) in self.pieces.iter().enumerate() {
                if i > 0 {
                    output.push(' ');
                }
                output.push_str(&piece.text);
            }
        }

        output
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &c in segment {
        if is_separator(c) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }

    words
}

fn fragment(word: &Word, start: usize, end: usize) -> String {
    let mut result = String::new();
    for i in start..end {
        result.push(word.chars[i]);
    }
    result
}

fn make_line(indent: usize) -> Line {
    Line::new(indent)
}

fn layout_paragraph(words: &[Word], options: &WrapOptions, output: &mut Vec<String>) {
    if words.is_empty() {
        return;
    }

    let mut lines = Vec::new();
    let mut line_index = 0usize;
    let mut line = make_line(options.first_indent);

    for word in words {
        let mut offset = 0usize;

        while offset < word.chars.len() {
            let capacity = options.width - line.indent;
            let remaining = word.chars.len() - offset;
            let room = capacity.saturating_sub(line.content_len);
            let separator_cost = usize::from(!line.is_empty());

            if remaining + separator_cost <= room {
                line.push_piece(fragment(word, offset, word.chars.len()));
                offset = word.chars.len();
                continue;
            }

            let mut usable = None;
            for &marker in &word.markers {
                if marker <= offset || marker >= word.chars.len() {
                    continue;
                }
                let prefix_len = marker - offset;
                if prefix_len + 1 <= room.saturating_sub(separator_cost) {
                    usable = Some(marker);
                }
            }

            if let Some(marker) = usable {
                let prefix_len = marker - offset;
                let used = line.content_len + separator_cost + prefix_len + 1;
                let unused = capacity.saturating_sub(used);
                let accept = if line.is_empty() {
                    true
                } else {
                    let baseline = capacity.saturating_sub(line.content_len);
                    unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                        < baseline.saturating_mul(baseline)
                };

                if accept {
                    let mut text = fragment(word, offset, marker);
                    text.push('-');
                    line.push_piece(text);
                    lines.push(line);
                    line_index += 1;
                    line = make_line(options.rest_indent);
                    offset = marker;
                    continue;
                }
            }

            if !line.is_empty() {
                lines.push(line);
                line_index += 1;
                line = make_line(options.rest_indent);
                continue;
            }

            let capacity = options.width - line.indent;
            if remaining > capacity {
                let end = offset + capacity;
                line.push_piece(fragment(word, offset, end));
                lines.push(line);
                line_index += 1;
                line = make_line(options.rest_indent);
                offset = end;
            } else {
                line.push_piece(fragment(word, offset, word.chars.len()));
                offset = word.chars.len();
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    let last = lines.len().saturating_sub(1);
    for (i, line) in lines.iter().enumerate() {
        output.push(line.render(options.width, options.justify, i == last, i));
    }

    let _ = line_index;
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }

        let left_ok = i > 0
            && original[i - 1].1 != '~'
            && !is_separator(original[i - 1].1);
        let right_ok = i + 1 < original.len()
            && original[i + 1].1 != '~'
            && !is_separator(original[i + 1].1);

        if !left_ok || !right_ok {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0usize;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let count = i - start;
            if count >= 2 {
                layout_paragraph(&parse_words(&segment), options, &mut output);
                segment.clear();
                for _ in 0..(count - 1) {
                    output.push(String::new());
                }
                continue;
            }
        }

        segment.push(normalized[i]);
        i += 1;
    }

    layout_paragraph(&parse_words(&segment), options, &mut output);
    Ok(output)
}
