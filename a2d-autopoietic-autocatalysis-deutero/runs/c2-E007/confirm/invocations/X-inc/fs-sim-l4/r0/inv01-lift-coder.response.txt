```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_owned()),
            }
        }
        Self::canonical(&parts)
    }

    fn parent_path(path: &str) -> String {
        if path == "/" {
            return "/".to_owned();
        }
        match path.rfind('/') {
            Some(0) => "/".to_owned(),
            Some(i) => path[..i].to_owned(),
            None => "/".to_owned(),
        }
    }

    fn basename(path: &str) -> &str {
        path.rsplit('/').next().unwrap_or("")
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || (path.starts_with(dir)
                && path.as_bytes().get(dir.len()) == Some(&b'/'))
    }

    fn child_paths(&self, dir: &str) -> Vec<String> {
        let prefix = if dir == "/" {
            "/".to_owned()
        } else {
            format!("{}/", dir)
        };
        self.entries
            .keys()
            .filter(|p| *p != dir && p.starts_with(&prefix))
            .filter(|p| {
                let rest = &p[prefix.len()..];
                !rest.is_empty() && !rest.contains('/')
            })
            .cloned()
            .collect()
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_missing_final: bool,
    ) -> Result<String, Error> {
        let mut pending = Self::components(path);
        let mut physical = Vec::<String>::new();
        let mut visited = BTreeSet::<String>::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(Self::canonical(&physical));
            }

            let component = pending.remove(0);
            if component.is_empty() || component == "." {
                continue;
            }
            if component == ".." {
                physical.pop();
                continue;
            }

            let mut next = physical.clone();
            next.push(component);
            let candidate = Self::canonical(&next);
            let final_component = pending.is_empty();

            if let Some(entry) = self.entries.get(&candidate) {
                match entry {
                    Entry::Dir => physical.push(Self::basename(&candidate).to_owned()),
                    Entry::File(_) if !final_component => {
                        return Err(Error::NotDir(candidate));
                    }
                    Entry::File(_) => {
                        physical.push(Self::basename(&candidate).to_owned());
                    }
                    Entry::Link(target) if follow_final || !final_component => {
                        if visited.contains(&candidate) {
                            return Err(Error::Loop(candidate));
                        }
                        if expansions >= 8 {
                            return Err(Error::Depth(candidate));
                        }
                        visited.insert(candidate.clone());
                        expansions += 1;

                        let mut target_parts = if target.starts_with('/') {
                            Vec::new()
                        } else {
                            physical.clone()
                        };
                        for part in target.split('/') {
                            match part {
                                "" | "." => {}
                                ".." => {
                                    target_parts.pop();
                                }
                                other => target_parts.push(other.to_owned()),
                            }
                        }
                        target_parts.extend(pending.into_iter());
                        let expanded = Self::canonical(&target_parts);
                        pending = Self::components(&expanded);
                        physical.clear();
                    }
                    Entry::Link(_) => physical.push(Self::basename(&candidate).to_owned()),
                }
            } else {
                if final_component && create_missing_final {
                    let parent = Self::canonical(&physical);
                    match self.entries.get(&parent) {
                        Some(Entry::Dir) => return Ok(candidate),
                        Some(_) => return Err(Error::NotDir(parent)),
                        None => return Err(Error::Missing(parent)),
                    }
                }
                return Err(Error::Missing(candidate));
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        if path == "/" {
            return Ok(("/".to_owned(), "/".to_owned()));
        }
        let parent = Self::parent_path(path);
        let name = Self::basename(path).to_owned();
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn resolved_child(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn subtree_entries(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| Self::is_within(p, path))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Exists("/".to_owned()));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let dest = Self::resolved_child(&parent, &name);
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => {
                // An un-followed final link is only possible for unusual internal paths.
                Err(Error::Missing(resolved))
            }
            None => {
                let parent = Self::parent_path(&resolved);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {
                        self.entries
                            .insert(resolved, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(parent)),
                    None => Err(Error::Missing(parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut out = self
                    .child_paths(&resolved)
                    .into_iter()
                    .filter_map(|p| {
                        self.entries
                            .get(&p)
                            .map(|entry| (Self::basename(&p).to_owned(), entry.kind()))
                    })
                    .collect::<Vec<_>>();
                out.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(out)
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        let source = Self::normalize(from, &self.cwd);
        Self::validate(to)?;
        let destination = Self::normalize(to, &self.cwd);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&source)?;
        let src = Self::resolved_child(&src_parent, &src_name);
        let src_entry = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(src_entry, Entry::Dir) && Self::is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&destination)?;
        let dst = Self::resolved_child(&dst_parent, &dst_name);

        if matches!(self.entries.get(&dst), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }

        if matches!(src_entry, Entry::Dir) && Self::is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let dst_entry = self.entries.get(&dst).cloned();
        if dst_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }

        if let Some(ref existing) = dst_entry {
            match (&src_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if self.child_paths(&dst).len() > 0 {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moving = self
            .subtree_entries(&src)
            .into_iter()
            .map(|old| {
                let suffix = &old[src.len()..];
                (old, format!("{}{}", dst, suffix))
            })
            .collect::<Vec<_>>();

        if dst_entry.is_some() {
            for old in self.subtree_entries(&dst) {
                self.entries.remove(&old);
            }
        }
        let values = moving
            .iter()
            .map(|(old, _)| (old.clone(), self.entries.get(old).cloned().unwrap()))
            .collect::<Vec<_>>();
        for (old, _) in &moving {
            self.entries.remove(old);
        }
        for ((_, new), (_, entry)) in moving.into_iter().zip(values.into_iter()) {
            self.entries.insert(new, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = Self::resolved_child(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.child_paths(&resolved).is_empty() {
            return Err(Error::NotEmpty(resolved));
        }

        for p in self.subtree_entries(&resolved) {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let canonical_link = Self::normalize(link, &self.cwd);

        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        if canonical_link == "/" {
            return Err(Error::Exists("/".to_owned()));
        }
        let (parent, name) = self.resolve_parent(&canonical_link)?;
        let resolved = Self::resolved_child(&parent, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = Self::resolved_child(&parent, &name);
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let paths = if matches!(entry, Entry::Dir) {
            self.subtree_entries(&resolved)
        } else {
            vec![resolved.clone()]
        };

        let mut sizes = BTreeMap::<String, u64>::new();
        for p in paths {
            let size = match self.entries.get(&p).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p, size);
        }

        let dirs = sizes
            .iter()
            .filter_map(|(p, _)| match self.entries.get(p) {
                Some(Entry::Dir) => Some(p.clone()),
                _ => None,
            })
            .collect::<Vec<_>>();
        for dir in dirs {
            let total = sizes
                .iter()
                .filter(|(p, _)| p.as_str() != dir && Self::is_within(p, &dir))
                .map(|(_, size)| *size)
                .sum::<u64>();
            sizes.insert(dir, total);
        }

        Ok(sizes.into_iter().collect())
    }
}
```
