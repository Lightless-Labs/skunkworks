                    task: task_id,

                    task: task_id,
                    position,
                    value,

                    task: task_id,
                    worker,
                    kind: EventKind::Finish,

                    remaining_predecessors[child] -= 1;
                }
            }

            break;

        let mut next_time: Option<u64> = None;

            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |old| old.min(candidate)));

            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        if let Some(next) = next_time {
            time = next;
        } else if !started_at_time {
            // With a valid DAG and finite releases, this is unreachable, but
            // avoid looping if invariants are ever changed.
            break;
        }
    }

        .into_iter()
        .map(|run| run.expect("every task must be scheduled"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = Vec::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {

            .iter()

            .max()
            .unwrap_or(0);

    let mut worker_busy = vec![false; workers];
    let mut running = Vec::<(u64, usize, usize)>::new(); // finish, task, worker
    let mut events = Vec::with_capacity(n.saturating_mul(2));

    let mut completed_count = 0usize;
    let mut makespan = 0u64;

                completed[task] = true;
                completed_count += 1;
                worker_busy[worker] = false;
                events.push(Event {
                    time: finish,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
                makespan = makespan.max(finish);
                for &child in &children[task] {
                    remaining_preds[child] -= 1;
                }
            }
        }

        // Repeatedly admit one task, reevaluating ranking after every start.
        loop {
            let worker = (0..workers).find(|&w| !worker_busy[w] && worker_free[w] <= now);
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;

                if started[id]
                    || tasks[id].release > now
                    || remaining_preds[id] != 0

            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                // Immediate completion precedes choosing another start.
                completed[task_id] = true;
                completed_count += 1;
                worker_busy[worker] = false;
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
                makespan = makespan.max(now);
                for &child in &children[task_id] {
                    remaining_preds[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change state.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now && next_time.map_or(true, |current| time < current) {
                next_time = Some(time);
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                consider(tasks[id].release);
            }
        }
        for worker in 0..workers {
            if !worker_busy[worker] && worker_free[worker] > now {
                consider(worker_free[worker]);
            }
        }

        // A busy worker's cooldown may expire before its task finishes only
        // when turnaround is zero, in which case it is available at finish.
        if let Some(time) = next_time {
            now = time;
        } else {
            // Every unfinished acyclic task must eventually become startable;
            // reaching here would contradict the validated graph and event rules.
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("validated acyclic tasks are all scheduled"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

use std::cmp::Reverse;
use std::collections::BinaryHeap;

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    let mut ready = BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            ready.push(Reverse(id));
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(Reverse(id)) = ready.pop() {
        topo.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.push(Reverse(child));
            }
        }
    }

    if topo.len() != n {
        let removed: Vec<bool> = {
            let mut removed = vec![false; n];
            for &id in &topo {
                removed[id] = true;
            }
            removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_preds: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();

    // Available workers have reached their cooldown expiry.
    let mut worker_available = vec![0u64; workers];

    // Running completions are ordered by (finish time, task ID, worker ID).
    let mut running: BinaryHeap<Reverse<(u64, usize, usize)>> = BinaryHeap::new();

    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut now = 0u64;
    let mut completion_count = 0usize;
    let mut makespan = 0u64;

    loop {
        // Complete all tasks due now, in ascending task ID.
        while let Some(Reverse((finish, task_id, worker_id))) = running.peek().copied() {
            if finish != now {
                break;
            }
            running.pop();
            completed[task_id] = true;
            completion_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
            for &child in &children[task_id] {
                remaining_preds[child] -= 1;
            }
        }

        if completion_count == n {
            break;
        }

        // Repeatedly select and start one eligible task.
        loop {
            let worker_id = (0..workers).find(|&w| worker_available[w] <= now);
            let Some(worker_id) = worker_id else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || remaining_preds[id] != 0
                {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }
            let Some(task_id) = best else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker_id] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if finish == now {
                completed[task_id] = true;
                completion_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_preds[child] -= 1;
                }
            } else {
                running.push(Reverse((finish, task_id, worker_id)));
            }
        }

        if completion_count == n {
            break;
        }

        // Advance to the earliest event that can change eligibility or worker availability.
        let mut next = running.peek().map(|Reverse((time, _, _))| *time);
        for (id, task) in tasks.iter().enumerate() {
            if !started[id] && task.release > now {
                next = Some(next.map_or(task.release, |t| t.min(task.release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next = Some(next.map_or(available_at, |t| t.min(available_at)));
            }
        }

        // A valid DAG with unfinished tasks always has a future release,
        // completion, or cooldown event.
        now = next.expect("unfinished valid schedule must have a future event");
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task must start"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before all other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    // Deterministic Kahn order: always remove the smallest currently ready ID.
    let mut remaining_indegree = indegree.clone();
    let mut available: Vec<usize> = (0..n)
        .filter(|&id| remaining_indegree[id] == 0)
        .collect();
    let mut kahn_order = Vec::with_capacity(n);

    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, id)| *id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        kahn_order.push(id);

        for &child in &successors[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if kahn_order.len() != n {
        let blocked = (0..n)
            .filter(|&id| remaining_indegree[id] != 0)
            .collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in kahn_order.iter().rev() {
        let max_successor_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available_at = vec![0u64; workers];
    let mut running: Vec<(usize, usize, u64)> = Vec::new(); // (task, worker, finish)
    let mut events = Vec::new();
    let mut completed_count = 0usize;
    let mut time = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, ordered by task ID.
        let mut finished_now: Vec<(usize, usize)> = running
            .iter()
            .filter(|(_, _, finish)| *finish == time)
            .map(|&(task, worker, _)| (task, worker))
            .collect();
        finished_now.sort_by_key(|&(task, _)| task);

        for (task, worker) in finished_now {
            if let Some(pos) = running
                .iter()
                .position(|&(running_task, running_worker, finish)| {
                    running_task == task && running_worker == worker && finish == time
                })
            {
                running.swap_remove(pos);
            }
            completed[task] = true;
            completed_count += 1;
            makespan = makespan.max(time);
            events.push(Event {
                time,
                task,
                worker,
                kind: EventKind::Finish,
            });
        }

        loop {
            let worker = (0..workers).find(|&w| worker_available_at[w] <= time);
            let Some(worker) = worker else {
                break;
            };

            let selected = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= time
                        && tasks[id].deps.iter().all(|&dep| completed[dep])
                })
                .max_by(|&a, &b| {
                    tails[a]
                        .cmp(&tails[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });

            let Some(task_id) = selected else {
                break;
            };

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let worker_available = finish
                .checked_add(turnaround)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available_at[worker] = worker_available;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            events.push(Event {
                time,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if finish == time {
                // Zero-duration tasks complete immediately before the next admission.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(time);
                events.push(Event {
                    time,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((task_id, worker, finish));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for &(_, _, finish) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        time = next_time.expect("unfinished tasks must have a future event");
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task must run exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
