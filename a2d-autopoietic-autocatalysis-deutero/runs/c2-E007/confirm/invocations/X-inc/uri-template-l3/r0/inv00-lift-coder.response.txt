#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let mut content_start = i + 1;
                let operator = if content_start < close {
                    match bytes[content_start] {
                        b'+' => { content_start += 1; Operator::Reserved }
                        b'#' => { content_start += 1; Operator::Fragment }
                        b'.' => { content_start += 1; Operator::Dot }
                        b'/' => { content_start += 1; Operator::Slash }
                        b';' => { content_start += 1; Operator::Semicolon }
                        b'?' => { content_start += 1; Operator::Query }
                        b'&' => { content_start += 1; Operator::Continuation }
                        b'!' | b'=' | b'|' | b'@' => {
                            return Err(Error::UnsupportedOperator { pos: content_start });
                        }
                        _ => Operator::Simple,
                    }
                } else {
                    Operator::Simple
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let name_bytes = &bytes[spec_start..spec_end];
                    for (offset, &b) in name_bytes.iter().enumerate() {
                        let valid = if offset == 0 {
                            b.is_ascii_alphabetic() || b == b'_'
                        } else {
                            b.is_ascii_alphanumeric() || b == b'_'
                        };
                        if !valid {
                            return Err(Error::InvalidName { pos: spec_start + offset });
                        }
                    }

                    let name = &template[spec_start..spec_end];
                    if seen.contains(&name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name);
                    names.push((name.to_owned(), spec_start));

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end: close + 1,
                    operator,
                    names,
                });
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields = Vec::new();
        for (name, _) in &expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
                fields.push((name.as_str(), encode(value, matches!(expression.operator, Operator::Reserved | Operator::Fragment))));
            }
        }

        if !fields.is_empty() {
            match expression.operator {
                Operator::Simple | Operator::Reserved => {
                    join_values(&mut output, &fields, ",", false);
                }
                Operator::Fragment => {
                    output.push('#');
                    join_values(&mut output, &fields, ",", false);
                }
                Operator::Dot => {
                    output.push('.');
                    join_values(&mut output, &fields, ".", false);
                }
                Operator::Slash => {
                    output.push('/');
                    join_values(&mut output, &fields, "/", false);
                }
                Operator::Semicolon => {
                    output.push(';');
                    for (index, (name, value)) in fields.iter().enumerate() {
                        if index > 0 { output.push(';'); }
                        output.push_str(name);
                        if !value.is_empty() {
                            output.push('=');
                            output.push_str(value);
                        }
                    }
                }
                Operator::Query | Operator::Continuation => {
                    output.push(if matches!(expression.operator, Operator::Query) { '?' } else { '&' });
                    for (index, (name, value)) in fields.iter().enumerate() {
                        if index > 0 { output.push('&'); }
                        output.push_str(name);
                        output.push('=');
                        output.push_str(value);
                    }
                }
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn join_values(output: &mut String, fields: &[(&str, String)], separator: &str, _unused: bool) {
    for (index, (_, value)) in fields.iter().enumerate() {
        if index > 0 {
            output.push_str(separator);
        }
        output.push_str(value);
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let reserved_char = reserved && matches!(
            b,
            b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' |
            b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
        );

        if unreserved || reserved_char {
            output.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0F) as usize] as char);
        }
        i += 1;
    }

    output
}
