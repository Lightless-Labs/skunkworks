#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let start = cursor;
                let Some(relative_close) = bytes[start + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = start + 1 + relative_close;

                if let Some(relative_open) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + relative_open,
                    });
                }

                let content_start = start + 1;
                let mut spec_start = content_start;
                let mut operator = None;

                if content_start < end {
                    let ch = bytes[content_start];
                    if matches!(ch, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                        operator = Some(ch as char);
                        spec_start += 1;
                    } else if matches!(ch, b'!' | b'=' | b',' | b'@' | b'|' | b'$') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                }

                if spec_start == end {
                    return Err(Error::EmptyExpression { pos: end });
                }

                let mut variables = Vec::new();
                let mut part_start = spec_start;
                let mut seen: Vec<&str> = Vec::new();

                loop {
                    let part_end = bytes[part_start..end]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| part_start + offset)
                        .unwrap_or(end);

                    if part_start == part_end {
                        return Err(Error::EmptyVariable { pos: part_start });
                    }

                    let name_bytes = &bytes[part_start..part_end];
                    let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
                    let valid_rest = |b: u8| b.is_ascii_alphanumeric() || b == b'_';

                    if !valid_first(name_bytes[0]) {
                        return Err(Error::InvalidName { pos: part_start });
                    }
                    if let Some(offset) = name_bytes[1..].iter().position(|&b| !valid_rest(b)) {
                        return Err(Error::InvalidName {
                            pos: part_start + 1 + offset,
                        });
                    }

                    let name = &template[part_start..part_end];
                    if seen.contains(&name) {
                        return Err(Error::DuplicateVariable { pos: part_start });
                    }
                    seen.push(name);
                    variables.push(name);

                    if part_end == end {
                        break;
                    }
                    part_start = part_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end: end + 1,
                    operator,
                    variables,
                });
                cursor = end + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields = Vec::new();
        for name in expression.variables {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push((name, *value));
            }
        }

        if !fields.is_empty() {
            match expression.operator {
                Some('#') => output.push('#'),
                Some('.') => output.push('.'),
                Some('/') => output.push('/'),
                Some(';') => output.push(';'),
                Some('?') => output.push('?'),
                Some('&') => output.push('&'),
                _ => {}
            }

            let separator = match expression.operator {
                Some('.') => '.',
                Some('/') => '/',
                Some(';') | Some('?') | Some('&') => '&',
                _ => ',',
            };

            for (index, (name, value)) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(separator);
                }

                match expression.operator {
                    Some(';') => {
                        let encoded = encode(value, false);
                        output.push_str(name);
                        if !encoded.is_empty() {
                            output.push('=');
                            output.push_str(&encoded);
                        }
                    }
                    Some('?') | Some('&') => {
                        output.push_str(name);
                        output.push('=');
                        output.push_str(&encode(value, false));
                    }
                    _ => output.push_str(&encode(
                        value,
                        matches!(expression.operator, Some('+') | Some('#')),
                    )),
                }
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved = matches!(
            byte,
            b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
        );

        if unreserved || (allow_reserved && reserved) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}
