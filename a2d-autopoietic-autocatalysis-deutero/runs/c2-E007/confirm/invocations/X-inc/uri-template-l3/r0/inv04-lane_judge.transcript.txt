    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },

    DuplicateVariable { pos: usize },

struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let start = cursor;
                let Some(relative_close) = bytes[start + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = start + 1 + relative_close;

                if let Some(relative_open) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + relative_open,
                    });

                let mut spec_start = content_start;
                let mut operator = None;

                if content_start < end {
                    let ch = bytes[content_start];
                    if matches!(ch, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                        operator = Some(ch as char);
                        spec_start += 1;
                    } else if matches!(ch, b'!' | b'=' | b',' | b'@' | b'|' | b'$') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                }

                    return Err(Error::EmptyExpression { pos: end });

                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| part_start + offset)
                        .unwrap_or(end);

                        return Err(Error::EmptyVariable { pos: part_start });

                    }
                    if let Some(offset) = name_bytes[1..].iter().position(|&b| !valid_rest(b)) {
                        return Err(Error::InvalidName {

                        });

                    part_start = part_end + 1;

                    start,
                    end: end + 1,
                    operator,
                    variables,

        }

                fields.push((name, *value));
            }
        }

        if !fields.is_empty() {
            match expression.operator {
                Some('#') => output.push('#'),
                Some('.') => output.push('.'),
                Some('/') => output.push('/'),
                Some(';') => output.push(';'),
                Some('?') => output.push('?'),
                Some('&') => output.push('&'),
                _ => {}
            }

                Some('.') => '.',
                Some('/') => '/',
                Some(';') | Some('?') | Some('&') => '&',
                _ => ',',
            };

            for (index, (name, value)) in fields.iter().enumerate() {

                }
            }
        }

        literal_start = expression.end;
    }

    Ok(output)

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved = matches!(
            byte,
            b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='

        if unreserved || (allow_reserved && reserved) {
            output.push(byte as char);

            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {

    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let mut content_start = i + 1;
                let operator = if content_start < close {
                    match bytes[content_start] {
                        b'+' => { content_start += 1; Operator::Reserved }
                        b'#' => { content_start += 1; Operator::Fragment }
                        b'.' => { content_start += 1; Operator::Dot }
                        b'/' => { content_start += 1; Operator::Slash }
                        b';' => { content_start += 1; Operator::Semicolon }
                        b'?' => { content_start += 1; Operator::Query }
                        b'&' => { content_start += 1; Operator::Continuation }
                        b'!' | b'=' | b'|' | b'@' => {
                            return Err(Error::UnsupportedOperator { pos: content_start });
                        }
                        _ => Operator::Simple,
                    }
                } else {
                    Operator::Simple
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut seen: Vec<&str> = Vec::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')

                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });

                        };
                        if !valid {
                            return Err(Error::InvalidName { pos: spec_start + offset });
                        }

                    let name = &template[spec_start..spec_end];
                    if seen.contains(&name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name);
                    names.push((name.to_owned(), spec_start));

                    if spec_end == close {
                        break;

                    spec_start = spec_end + 1;
                }

                    start,
                    end: close + 1,
                    operator,
                    names,

            _ => i += 1,

                Operator::Fragment => {
                    output.push('#');
                    join_values(&mut output, &fields, ",", false);
                }
                Operator::Dot => {
                    output.push('.');

                }
                Operator::Slash => {
                    output.push('/');
                    join_values(&mut output, &fields, "/", false);
                }
                Operator::Semicolon => {
                    output.push(';');
                    for (index, (name, value)) in fields.iter().enumerate() {
                        if index > 0 { output.push(';'); }
                        output.push_str(name);

    Ok(output)

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Label,
    Path,
    Matrix,
    Query,
    Continuation,
}

struct Expression {
    operator: Operator,
    variables: Vec<String>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = if let Some(&first) = bytes.first() {
        let recognized = match first {
            b'+' => Some(Operator::Reserved),
            b'#' => Some(Operator::Fragment),
            b'.' => Some(Operator::Label),
            b'/' => Some(Operator::Path),
            b';' => Some(Operator::Matrix),
            b'?' => Some(Operator::Query),
            b'&' => Some(Operator::Continuation),
            _ => None,
        };
        if let Some(operator) = recognized {
            (operator, 1)
        } else {
            (Operator::Simple, 0)
        }
    } else {
        (Operator::Simple, 0)
    };

    let remaining = &content[offset..];
    let remaining_bytes = remaining.as_bytes();
    if remaining_bytes.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut start = 0;

    for end in (0..=remaining_bytes.len())
        .filter(|&index| index == remaining_bytes.len() || remaining_bytes[index] == b',')
    {
        if end == start {
            return Err(Error::EmptyVariable {
                pos: content_pos + offset + start,
            });
        }

        let specifier = &remaining[start..end];
        let specifier_bytes = specifier.as_bytes();
        let mut invalid_at = None;

        if !is_name_start(specifier_bytes[0]) {
            invalid_at = Some(0);
        } else if let Some(index) = specifier_bytes
            .iter()
            .enumerate()
            .skip(1)
            .find_map(|(index, &byte)| (!is_name_continue(byte)).then_some(index))
        {
            invalid_at = Some(index);
        }

        if let Some(index) = invalid_at {
            return Err(Error::InvalidName {
                pos: content_pos + offset + start + index,
            });
        }

        if seen.iter().any(|&name| name == specifier) {
            return Err(Error::DuplicateVariable {
                pos: content_pos + offset + start,
            });
        }

        seen.push(specifier);
        variables.push(specifier.to_owned());
        start = end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn encode_value(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            output.push('%');
            output.push((bytes[index + 1] as char).to_ascii_uppercase());
            output.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = matches!(
            byte,
            b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
                | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
        );

        if unreserved || (reserved && reserved_byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            output.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
        index += 1;
    }

    output
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();
    let reserved = matches!(expression.operator, Operator::Reserved | Operator::Fragment);

    for name in &expression.variables {
        if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
            let encoded = encode_value(value, reserved);
            let field = match expression.operator {
                Operator::Matrix => {
                    if encoded.is_empty() {
                        name.clone()
                    } else {
                        format!("{name}={encoded}")
                    }
                }
                Operator::Query | Operator::Continuation => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Label => (".", "."),
        Operator::Path => ("/", "/"),
        Operator::Matrix => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open_pos = cursor;
                let close_rel = bytes[cursor + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                    .ok_or(Error::Unclosed { pos: open_pos })?;
                let close_pos = cursor + 1 + close_rel;

                if let Some(inner_rel) = bytes[cursor + 1..close_pos]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + inner_rel,
                    });
                }

                let content = &template[cursor + 1..close_pos];
                let expression = parse_expression(content, cursor + 1, close_pos)?;
                expressions.push((open_pos, close_pos + 1, expression));
                cursor = close_pos + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    for (start, end, expression) in expressions {
        output.push_str(&template[literal_start..start]);
        output.push_str(&expand_expression(&expression, vars));
        literal_start = end;
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}

enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Query),
            b'&' => Some(Self::Continuation),
            _ => None,
        }
    }

    fn separator(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => Some(','),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query | Self::Continuation => Some('&'),
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continuation => Some('&'),
            Self::Simple | Self::Reserved => None,
        }
    }

    fn named_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continuation)
    }

    fn semicolon_fields(self) -> bool {
        matches!(self, Self::Semicolon)
    }

    fn allows_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let mut operator = Operator::Simple;

    if content_start < end {
        if let Some(found) = Operator::from_byte(bytes[content_start]) {
            operator = found;
            content_start += 1;
        } else if matches!(bytes[content_start], b'!' | b'=' | b'<' | b'>' | b'|' | b'\\' | b'^' | b'`') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut names = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(end);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let first = bytes[spec_start];
        if !is_name_start(first) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        for pos in spec_start + 1..spec_end {
            if !is_name_continue(bytes[pos]) {
                return Err(Error::InvalidName { pos });
            }
        }

        let name = &template[spec_start..spec_end];
        if !seen.insert(name.to_owned()) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        names.push(name.to_owned());

        if spec_end == end {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, names })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let Some(relative_end) = bytes[pos + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = pos + 1 + relative_end;

                if let Some(offset) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + offset,
                    });
                }

                expressions.push(parse_expression(template, start, end)?);
                pos = end + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            out.push('%');
            out.push(uppercase_hex(bytes[i + 1]) as char);
            out.push(uppercase_hex(bytes[i + 2]) as char);
            i += 3;
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            out.push(byte as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
            i += 1;
        }
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();

    for name in &expression.names {
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, expression.operator.allows_reserved());
            if expression.operator.named_fields() {
                if expression.operator.semicolon_fields() && encoded.is_empty() {
                    fields.push(name.clone());
                } else {
                    fields.push(format!("{name}={encoded}"));
                }
            } else {
                fields.push(encoded);
            }
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut out = String::new();
    if let Some(prefix) = expression.operator.prefix() {
        out.push(prefix);
    }
    let separator = expression.operator.separator().unwrap();
    for (index, field) in fields.iter().enumerate() {
        if index != 0 {
            out.push(separator);
        }
        out.push_str(field);
    }
    out
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut pos = 0;
    let mut expression_index = 0;

    while pos < bytes.len() {
        if bytes[pos] == b'{' {
            let end = bytes[pos + 1..]
                .iter()
                .position(|&byte| byte == b'}')
                .map(|offset| pos + 1 + offset)
                .unwrap();
            output.push_str(&expand_expression(&expressions[expression_index], vars));

                .map(|offset| pos + offset)
                .unwrap_or(bytes.len());
            output.push_str(&template[pos..next]);
            pos = next;
        }
    }

    Ok(output)
}
