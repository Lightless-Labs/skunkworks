pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let task_count = tasks.len();

    // Validate every dependency before constructing the graph or doing scheduling.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    // Build predecessor-to-child edges and indegrees.
    let mut children = vec![Vec::new(); task_count];
    let mut indegree = vec![0usize; task_count];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            children[dependency].push(task_id);
        }
    }

    // Keep available IDs sorted so each iteration removes the smallest current ID.
    let mut available = Vec::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(task_id);
        }
    }

    let mut order = Vec::with_capacity(task_count);
    let mut remaining = indegree;
    while !available.is_empty() {
        let task_id = available.remove(0);
        order.push(task_id);

        for &child in &children[task_id] {
            remaining[child] -= 1;
            if remaining[child] == 0 {
                let position = available.partition_point(|&id| id < child);
                available.insert(position, child);
            }
        }
    }

    if order.len() != task_count {
        let mut removed = vec![false; task_count];
        for &task_id in &order {
            removed[task_id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..task_count).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut times = vec![Timing { start: 0, finish: 0 }; task_count];
    let mut makespan = 0;

    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&dependency| times[dependency].finish)
            .max()
            .unwrap_or(0);

        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;

        times[task_id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}

    // Validate every dependency before doing any scheduling work.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; task_count];
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[dependency] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            seen[dependency] = true;
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; task_count];
    let mut children = vec![Vec::new(); task_count];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            children[dependency].push(task_id);
        }
    }

    // A sorted vector is used as a min-priority queue. Newly unblocked IDs
    // are inserted in order, so each iteration removes the smallest ID.
    let mut available = Vec::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(task_id);
        }
    }

    let mut order = Vec::with_capacity(task_count);
    while !available.is_empty() {
        let task_id = available.remove(0);
        order.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let insertion = available
                    .binary_search(&child)
                    .unwrap_or_else(|index| index);
                available.insert(insertion, child);
            }
        }
    }

    if order.len() != task_count {
        let removed = {
            let mut removed = vec![false; task_count];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        let blocked = (0..task_count)
            .filter(|&task_id| !removed[task_id])
            .collect();
        return Err(Error::Cycle { blocked });
    }

    let mut times = vec![Timing { start: 0, finish: 0 }; task_count];
    let mut makespan = 0u64;

    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&dependency| times[dependency].finish)
            .max()

            .ok_or(Error::Overflow { task: task_id })?;

        times[task_id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let task_count = tasks.len();

    // Validate every dependency before doing graph construction or scheduling.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    // Build predecessor-to-child edges and indegrees.
    let mut children = vec![Vec::new(); task_count];
    let mut indegree = vec![0usize; task_count];

    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available: Vec<usize> = (0..task_count)
        .filter(|&task_id| indegree[task_id] == 0)
        .collect();
    let mut order = Vec::with_capacity(task_count);

    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, &task_id)| task_id)
            .map(|(position, _)| position)
            .expect("available is nonempty");
        let task_id = available.swap_remove(smallest_pos);
        order.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != task_count {
        let blocked = indegree
            .iter()
            .enumerate()
            .filter_map(|(task_id, &degree)| (degree != 0).then_some(task_id))
            .collect();
        return Err(Error::Cycle { blocked });
    }

    let mut starts = vec![0u64; task_count];
    let mut finishes = vec![0u64; task_count];
    let mut makespan = 0u64;

    for &task_id in &order {
        let finish = starts[task_id]
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;
        finishes[task_id] = finish;
        makespan = makespan.max(finish);

        for &child in &children[task_id] {
            starts[child] = starts[child].max(finish);
        }
    }

    let times = (0..task_count)
        .map(|task_id| Timing {
            start: starts[task_id],
            finish: finishes[task_id],
        })
        .collect();

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
