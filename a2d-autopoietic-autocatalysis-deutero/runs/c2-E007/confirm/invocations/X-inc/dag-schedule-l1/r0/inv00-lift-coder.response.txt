#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Plan {
    pub order: Vec<usize>,
    pub times: Vec<Timing>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let task_count = tasks.len();

    // Validate every dependency before doing graph construction or scheduling.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    // Build predecessor-to-child edges and indegrees.
    let mut children = vec![Vec::new(); task_count];
    let mut indegree = vec![0usize; task_count];

    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available: Vec<usize> = (0..task_count)
        .filter(|&task_id| indegree[task_id] == 0)
        .collect();
    let mut order = Vec::with_capacity(task_count);

    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, &task_id)| task_id)
            .map(|(position, _)| position)
            .expect("available is nonempty");
        let task_id = available.swap_remove(smallest_pos);
        order.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != task_count {
        let blocked = indegree
            .iter()
            .enumerate()
            .filter_map(|(task_id, &degree)| (degree != 0).then_some(task_id))
            .collect();
        return Err(Error::Cycle { blocked });
    }

    let mut starts = vec![0u64; task_count];
    let mut finishes = vec![0u64; task_count];
    let mut makespan = 0u64;

    for &task_id in &order {
        let finish = starts[task_id]
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;
        finishes[task_id] = finish;
        makespan = makespan.max(finish);

        for &child in &children[task_id] {
            starts[child] = starts[child].max(finish);
        }
    }

    let times = (0..task_count)
        .map(|task_id| Timing {
            start: starts[task_id],
            finish: finishes[task_id],
        })
        .collect();

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
