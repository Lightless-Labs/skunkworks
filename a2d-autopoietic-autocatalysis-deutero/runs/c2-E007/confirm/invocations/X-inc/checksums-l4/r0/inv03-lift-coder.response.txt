    pub end: usize,

    pub crc32: u32,
    pub adler32: u32,
}

    Overlap { span: usize },

fn crc32_step(mut reg: u32, byte: u8) -> u32 {
    reg ^= byte as u32;
    for _ in 0..8 {
        reg = (reg >> 1) ^ if reg & 1 != 0 { 0xedb88320 } else { 0 };
    }
    reg
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut reg = 0xffff_ffff;
    for &byte in data {
        reg = crc32_step(reg, byte);
    }
    reg ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u64;
    let mut b = 0u64;
    for &byte in data {
        a = (a + byte as u64) % 65521;
        b = (b + a) % 65521;
    }
    ((b as u32) << 16) | a as u32
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())
}

#[derive(Clone, Copy)]
struct Accumulator {
    crc: u32,
    a: u32,
    b: u32,
    f16a: u32,
    f16b: u32,
    f32a: u32,
    f32b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
    crc16_variant: Crc16,
}

impl Accumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
            f16a: 0,
            f16b: 0,
            f32a: 0,
            f32b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            crc16_variant: variant,
        }
    }

    fn seeded(crc32: u32, adler32: u32, variant: Crc16) -> Self {
        let mut acc = Self::new(variant);
        acc.crc = crc32 ^ 0xffff_ffff;
        acc.a = adler32 & 0xffff;
        acc.b = adler32 >> 16;
        acc
    }

    fn push(&mut self, byte: u8) {
        self.crc = crc32_step(self.crc, byte);

        self.a = (self.a + byte as u32) % 65521;
        self.b = (self.b + self.a) % 65521;

        self.f16a = (self.f16a + byte as u32) % 255;
        self.f16b = (self.f16b + self.f16a) % 255;

        if let Some(high) = self.f32_pending.take() {
            let word = ((high as u32) << 8) | byte as u32;
            self.f32a = (self.f32a + word) % 65535;
            self.f32b = (self.f32b + self.f32a) % 65535;
        } else {
            self.f32_pending = Some(byte);
        }

        match self.crc16_variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = self.crc16 & 0x8000 != 0;
                    self.crc16 = self.crc16.wrapping_shl(1);
                    if top {
                        self.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                self.crc16 ^= byte as u16;
                for _ in 0..8 {
                    let low = self.crc16 & 1 != 0;
                    self.crc16 >>= 1;
                    if low {
                        self.crc16 ^= 0x8408;
                    }
                }
            }
        }
    }

    fn finish_f32(&self) -> u32 {
        let (a, b) = if let Some(high) = self.f32_pending {
            let word = (high as u32) << 8;
            let a = (self.f32a + word) % 65535;
            let b = (self.f32b + a) % 65535;
            (a, b)
        } else {
            (self.f32a, self.f32b)
        };
        (b << 16) | a
    }

    fn hashes(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    fn crc16_result(&self) -> u16 {
        match self.crc16_variant {
            Crc16::X25 => self.crc16 ^ 0xffff,
            Crc16::Xmodem | Crc16::False => self.crc16,
        }
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            acc.push(byte);
        }
    }
    Ok(acc.hashes())
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc = crc32_step(self.crc, byte);
            self.a = (self.a + byte as u32) % 65521;
            self.b = (self.b + self.a) % 65521;
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let h = self.checksum();
        format!("c{:08x}:a{:08x}", h.crc32, h.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let mut crc = 0u32;
        let mut adler = 0u32;
        for (offset, slot) in (1..9).zip(0..8) {
            let digit = parse_hex(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            crc |= (digit as u32) << ((7 - slot) * 4);
        }
        for (offset, slot) in (11..19).zip(0..8) {
            let digit = parse_hex(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            adler |= (digit as u32) << ((7 - slot) * 4);
        }
        Self::from_seed(crc, adler)
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut acc = Accumulator::new(Crc16::Xmodem);
    for &byte in data {
        // Only Fletcher-32 state is used here.
        if let Some(high) = acc.f32_pending.take() {
            let word = ((high as u32) << 8) | byte as u32;
            acc.f32a = (acc.f32a + word) % 65535;
            acc.f32b = (acc.f32b + acc.f32a) % 65535;
        } else {
            acc.f32_pending = Some(byte);
        }
    }
    acc.finish_f32()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut f32a = 0u32;
    let mut f32b = 0u32;
    let mut pending = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                let word = ((high as u32) << 8) | byte as u32;
                f32a = (f32a + word) % 65535;
                f32b = (f32b + f32a) % 65535;
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        let word = (high as u32) << 8;
        f32a = (f32a + word) % 65535;
        f32b = (f32b + f32a) % 65535;
    }
    Ok((f32b << 16) | f32a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut acc = Accumulator::new(variant);
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                acc.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = acc.crc16 & 0x8000 != 0;
                    acc.crc16 = acc.crc16.wrapping_shl(1);
                    if top {
                        acc.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                acc.crc16 ^= byte as u16;
                for _ in 0..8 {
                    let low = acc.crc16 & 1 != 0;
                    acc.crc16 >>= 1;
                    if low {
                        acc.crc16 ^= 0x8408;
                    }
                }
            }
        }
    }
    acc.crc16_result()
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(variant);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            acc.push(byte);
        }
    }
    Ok(Multi {
        base: acc.hashes(),
        fletcher16: ((acc.f16b as u16) << 8) | acc.f16a as u16,
        fletcher32: acc.finish_f32(),
        crc16: acc.crc16_result(),
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for n in 0..32 {
        square[n] = gf2_matrix_times(matrix, matrix[n]);
    }
}

fn zero_byte_operator() -> [u32; 32] {
    let mut matrix = [0u32; 32];
    for (i, item) in matrix.iter_mut().enumerate() {
        *item = crc32_step(1u32 << i, 0);
    }
    matrix
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut len = right_len;
    let mut matrix = zero_byte_operator();
    let mut vector = left;
    while len != 0 {
        if len & 1 != 0 {
            vector = gf2_matrix_times(&matrix, vector);
        }
        len >>= 1;
        if len != 0 {
            let mut square = [0u32; 32];
            gf2_matrix_square(&mut square, &matrix);
            matrix = square;
        }
    }
    vector ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn parse_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(Crc16::Xmodem);
    let mut len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        len += selected.len() as u64;
        for &byte in selected {
            acc.push(byte);
        }
    }
    let hashes = acc.hashes();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(
    data: &[u8],
    spans: &[Span],
    token: &str,
) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut parsed_len = 0u64;
    let mut crc = 0u32;
    let mut adler = 0u32;
    for offset in 4..20 {
        let digit = parse_hex(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        parsed_len = (parsed_len << 4) | digit as u64;
    }
    for offset in 25..33 {
        let digit = parse_hex(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        crc = (crc << 4) | digit as u32;
    }
    for offset in 40..48 {
        let digit = parse_hex(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        adler = (adler << 4) | digit as u32;
    }

    let mut acc = Accumulator::new(Crc16::Xmodem);
    let mut selected_len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        selected_len += selected.len() as u64;
        for &byte in selected {
            acc.push(byte);
        }
    }
    if parsed_len != selected_len {
        return Err(VerifyError::Length);
    }
    let actual = acc.hashes();
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
