    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

    end: usize,
    operator: Option<u8>,
    names: Vec<(usize, String)>,

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let start = cursor;
                let Some(relative_end) = bytes[cursor + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = cursor + 1 + relative_end;

                if let Some(relative_open) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + relative_open,
                    });
                }

                let mut content_start = start + 1;
                let operator = if content_start < end
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let op = bytes[content_start];
                    if !matches!(op, b'+' | b'#') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    content_start += 1;
                    Some(op)
                } else {
                    None
                };

                if content_start == end {
                    return Err(Error::EmptyExpression { pos: end });
                }

                let mut names = Vec::new();
                let mut spec_start = content_start;
                let mut seen = std::collections::HashSet::new();

                loop {
                    let spec_end = bytes[spec_start..end]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|relative| spec_start + relative)
                        .unwrap_or(end);

                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    if !is_name_start(spec[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    if let Some(offset) = spec[1..].iter().position(|&b| !is_name_continue(b)) {
                        return Err(Error::InvalidName {
                            pos: spec_start + 1 + offset,
                        });
                    }

                    if !seen.insert(name.clone()) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    names.push((spec_start, name));

                    if spec_end == end {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end,
                    operator,
                    names,
                });
                cursor = end + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

                fields.push(encode_value(value, expression.operator.is_some()));
            }

            if expression.operator == Some(b'#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut encoded = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        let byte = bytes[cursor];

        if byte == b'%' && cursor + 2 < bytes.len() && is_ascii_hex(bytes[cursor + 1])

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
}

struct Expression<'a> {
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

fn parse_expression<'a>(
    template: &'a str,
    content_start: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let content = &template[content_start..close];

    if let Some(relative) = content.find('{') {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let bytes = content.as_bytes();
    let operator = match bytes.first().copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => Some(bytes[0]),
        _ => None,
    };

    if matches!(operator, Some(b'.' | b'/' | b';' | b'?' | b'&')) {
        return Err(Error::UnsupportedOperator {
            pos: content_start,
        });
    }

    let names_start = content_start + usize::from(operator.is_some());
    if names_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let names = &template[names_start..close];
    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut offset = names_start;

    for name in names.split(',') {
        let start = offset;
        if name.is_empty() {
            return Err(Error::EmptyVariable { pos: start });
        }

        let name_bytes = name.as_bytes();
        let valid_initial = |byte: u8| byte.is_ascii_alphabetic() || byte == b'_';
        let valid_rest = |byte: u8| valid_initial(byte) || byte.is_ascii_digit();

        if !valid_initial(name_bytes[0]) {
            return Err(Error::InvalidName { pos: start });
        }

        for (index, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !valid_rest(byte) {
                return Err(Error::InvalidName { pos: start + index });
            }
        }

        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(name);
        variables.push(Variable { name, start });

        offset += name.len() + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {

        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'

    )
}

    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    match value {
        0..=9 => b'0' + value,
        _ => b'A' + (value - 10),
    }
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%'
            && index + 2 < bytes.len()
            && let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
        {
            output.push('%');
            output.push(hex_digit(high) as char);
            output.push(hex_digit(low) as char);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4) as char);
            output.push(hex_digit(byte & 0x0f) as char);
        }
        index += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| *value)
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let Some(relative_open) = template[cursor..].find(['{', '}']) else {
            break;
        };
        let pos = cursor + relative_open;
        match template.as_bytes()[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let content_start = pos + 1;
                let Some(relative_close) = template[content_start..].find('}') else {
                    return Err(Error::Unclosed { pos });
                };
                let close = content_start + relative_close;
                let expression = parse_expression(template, content_start, close)?;
                expressions.push((pos, close + 1, expression));
                cursor = close + 1;
            }
            _ => unreachable!(),
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for (open, end, expression) in expressions {
        output.push_str(&template[cursor..open]);

        let allow_reserved = matches!(expression.operator, Some(b'+' | b'#'));
        let mut fields = Vec::new();
        for variable in expression.variables {
            if let Some(value) = lookup(variable.name, vars) {
                fields.push(encode_value(value, allow_reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        cursor = end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
}

struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'

fn validate_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;
    let operator = if content_start < close {
        match bytes[content_start] {
            b'+' => {
                content_start += 1;
                Operator::Plus
            }
            b'#' => {
                content_start += 1;
                Operator::Hash
            }
            b'.' | b'/' | b';' | b'?' | b'&' => {
                return Err(Error::UnsupportedOperator { pos: content_start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        if !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        for pos in spec_start + 1..spec_end {
            if !is_name_continue(bytes[pos]) {
                return Err(Error::InvalidName { pos });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        names.push((spec_start, spec_end));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        names,
    })
}

fn parse(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open = pos;
                let close = match bytes[open + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close]
                    .iter()
                    .position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                expressions.push(validate_expression(template, open, close)?);
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

                output.push('#');
            }
            output.push_str(&fields.join(separator));
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let content_start = start + 1;
                let close = match bytes[content_start..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(offset) => content_start + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[content_start..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: content_start + offset,
                    });
                }

                let mut body_start = content_start;
                let mut operator = None;
                if body_start < close
                    && matches!(
                        bytes[body_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let op = bytes[body_start] as char;
                    if op != '+' && op != '#' {
                        return Err(Error::UnsupportedOperator { pos: body_start });
                    }
                    operator = Some(op);
                    body_start += 1;
                }

                if body_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut spec_start = body_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&byte| byte == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &template[spec_start..spec_end];
                    let spec_bytes = spec.as_bytes();

                    if !is_name_start(spec_bytes[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    if let Some(offset) = spec_bytes[1..]
                        .iter()
                        .position(|&byte| !is_name_continue(byte))
                    {
                        return Err(Error::InvalidName {
                            pos: spec_start + 1 + offset,
                        });
                    }

                    if seen.iter().any(|name| *name == spec) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(spec);
                    names.push(spec);

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end: close + 1,
                    operator,
                    names,
                });
                pos = close + 1;
            }
            _ => pos += char_len(bytes[pos]),
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields: Vec<&str> = Vec::new();
        for name in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(value);
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            for (index, value) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                encode_value(value, expression.operator.is_some(), &mut output);
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn char_len(byte: u8) -> usize {
    if byte < 0x80 {
        1
    } else if byte < 0xE0 {
        2
    } else if byte < 0xF0 {
        3
    } else {
        4
    }
}

fn encode_value(value: &str, allow_reserved: bool, output: &mut String) {
    let bytes = value.as_bytes();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == b'%' && pos + 2 < bytes.len() && is_hex(bytes[pos + 1]) && is_hex(bytes[pos + 2]) {
            output.push('%');
            output.push((bytes[pos + 1] as char).to_ascii_uppercase());
            output.push((bytes[pos + 2] as char).to_ascii_uppercase());
            pos += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0F));
        }
        pos += 1;
    }
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
