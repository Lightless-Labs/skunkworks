#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut index = start;
    let mut record_index = 0usize;

    #[derive(Clone, Copy)]
    enum State {
        Unquoted,
        Quoted { opening: usize },
        Closed,
    }

    let mut state = State::Unquoted;
    let mut record_started = false;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        if !matches!(state, State::Quoted { .. }) && byte == b'\r' {
            if bytes.get(index + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: index,
                });
            }
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_started = false;
            record_index += 1;
            index += 2;
            state = State::Unquoted;
            continue;
        }

        match state {
            State::Quoted { opening } => {
                if byte == b'"' {
                    if bytes.get(index + 1) == Some(&b'"') {
                        field.push('"');
                        index += 2;
                    } else {
                        state = State::Closed;
                        index += 1;
                    }
                } else {
                    let ch = input[index..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    index += ch.len_utf8();
                }
                let _ = opening;
            }
            State::Unquoted => match byte {
                b'"' => {
                    if field.is_empty() {
                        state = State::Quoted { opening: index };
                        record_started = true;
                        index += 1;
                    } else {
                        return Err(CsvError::UnexpectedQuote {
                            record: record_index,
                            byte: index,
                        });
                    }
                }
                b',' => {
                    record.push(std::mem::take(&mut field));
                    record_started = true;
                    index += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_started = false;
                    record_index += 1;
                    index += 1;
                }
                _ => {
                    let ch = input[index..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    record_started = true;
                    index += ch.len_utf8();
                }
            },
            State::Closed => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    record_started = true;
                    state = State::Unquoted;
                    index += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_started = false;
                    record_index += 1;
                    state = State::Unquoted;
                    index += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            },
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
