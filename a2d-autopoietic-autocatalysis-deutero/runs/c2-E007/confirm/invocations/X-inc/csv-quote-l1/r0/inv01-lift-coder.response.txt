#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut start = 0;

    if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        start = 3;
    }
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut record = 0usize;
    let mut i = start;
    let mut at_field_start = true;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && byte == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if byte == b'"' {
                if bytes.get(i + 1) == Some(&b'"') {
                    field.push('"');
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8 boundary");
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            match byte {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    at_field_start = true;
                    after_quote = false;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    at_field_start = true;
                    after_quote = false;
                    i += 1;
                }
                b'\r' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    at_field_start = true;
                    after_quote = false;
                    i += 2;
                }
                _ => return Err(CsvError::TrailingByte { record, byte: i }),
            }
            continue;
        }

        match byte {
            b'"' if at_field_start => {
                in_quotes = true;
                quote_open = i;
                at_field_start = false;
                i += 1;
            }
            b'"' => return Err(CsvError::UnexpectedQuote { record, byte: i }),
            b',' => {
                fields.push(std::mem::take(&mut field));
                at_field_start = true;
                i += 1;
            }
            b'\n' => {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record += 1;
                at_field_start = true;
                i += 1;
            }
            b'\r' => {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record += 1;
                at_field_start = true;
                i += 2;
            }
            _ => {
                let ch = input[i..].chars().next().expect("valid UTF-8 boundary");
                field.push(ch);
                at_field_start = false;
                i += ch.len_utf8();
            }
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if !at_field_start || after_quote || !fields.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}
