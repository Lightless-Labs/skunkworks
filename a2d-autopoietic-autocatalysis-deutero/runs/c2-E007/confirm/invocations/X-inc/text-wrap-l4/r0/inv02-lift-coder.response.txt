#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker: bool,
    pieces: usize,
    gap: usize,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    let av: Vec<usize> = a.lines.iter().map(|l| l.consumed).collect();
    let bv: Vec<usize> = b.lines.iter().map(|l| l.consumed).collect();
    if av != bv {
        return av > bv;
    }
    let at: Vec<&str> = a.lines.iter().map(|l| l.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|l| l.text.as_str()).collect();
    at < bt
}

fn optimize(words: &[Word], options: &WrapOptions) -> Layout {
    use std::collections::HashMap;

    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        opt: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Layout>,
    ) -> Layout {
        if wi >= words.len() {
            return Layout { cost: 0, marker_lines: 0, lines: Vec::new() };
        }
        if let Some(v) = memo.get(&(wi, off, first)) {
            return v.clone();
        }

        let indent = if first { opt.first_indent } else { opt.rest_indent };
        let cap = opt.width - indent;
        let mut best: Option<Layout> = None;

        let mut consider = |display: String,
                            consumed: usize,
                            next_wi: usize,
                            next_off: usize,
                            marker: bool,
                            raw: bool,
                            pieces: usize| {
            if display.is_empty() {
                return;
            }
            let display_count = display.chars().count();
            let gap = opt.width - indent - display_count;
            let tail = solve(next_wi, next_off, false, words, opt, memo);
            let final_line = tail.lines.is_empty();
            let mut cost = (gap as u64) * (gap as u64);
            if final_line {
                cost /= 4;
            }
            if marker {
                cost += opt.hyphen_cost as u64;
            }
            if raw {
                cost += (2 * opt.width + 1) as u64;
            }
            if !final_line && pieces == 1 {
                cost += 3;
            }
            let mut text = " ".repeat(indent);
            text.push_str(&display);
            let mut lines = Vec::with_capacity(tail.lines.len() + 1);
            lines.push(Line { text, consumed, marker, pieces, gap });
            lines.extend(tail.lines);
            let candidate = Layout {
                cost: cost + tail.cost,
                marker_lines: tail.marker_lines + usize::from(marker),
                lines,
            };
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        };

        let word = &words[wi];
        let remaining = word.chars.len() - off;
        let mut display = String::new();
        let mut consumed = 0usize;
        let mut pieces = 0usize;
        let mut cur_wi = wi;
        let mut cur_off = off;

        loop {
            if cur_wi >= words.len() {
                break;
            }
            let w = &words[cur_wi];
            let start = cur_off;
            let rem = w.chars.len() - start;
            let separator = if pieces == 0 { 0 } else { 1 };
            if display.chars().count() + separator + rem <= cap {
                if separator == 1 {
                    display.push(' ');
                }
                for &c in &w.chars[start..] {
                    display.push(c);
                }
                consumed += rem;
                pieces += 1;
                cur_wi += 1;
                cur_off = 0;
                consider(display.clone(), consumed, cur_wi, 0, false, false, pieces);
                continue;
            }

            let room = cap.saturating_sub(display.chars().count() + separator);
            if room > 0 {
                for &m in w.markers.iter().filter(|&&m| m > start && m < w.chars.len()) {
                    let prefix = m - start;
                    if prefix + 1 <= room {
                        let mut d = display.clone();
                        if separator == 1 {
                            d.push(' ');
                        }
                        d.extend(w.chars[start..m].iter());
                        d.push('-');
                        consider(d, consumed + prefix, cur_wi, m, true, false, pieces + 1);
                    }
                }
            }

            if pieces == 0 && rem > cap {
                let marker_fits = w.markers.iter().any(|&m| {
                    m > start && m < w.chars.len() && (m - start + 1) <= cap
                });
                if !marker_fits {
                    let mut d = String::new();
                    d.extend(w.chars[start..start + cap].iter());
                    let next_off = start + cap;
                    let (nwi, noff) = if next_off == w.chars.len() {
                        (cur_wi + 1, 0)
                    } else {
                        (cur_wi, next_off)
                    };
                    consider(d, cap, nwi, noff, false, true, 1);
                }
            }
            break;
        }

        if let Some(result) = best {
            memo.insert((wi, off, first), result.clone());
            result
        } else {
            // Capacity is always positive; this fallback is reached only for a
            // single scalar that fits on an otherwise empty line.
            let c = word.chars[off];
            let mut next = solve(wi + usize::from(off + 1 == word.chars.len()), if off + 1 == word.chars.len() { 0 } else { off + 1 }, false, words, opt, memo);
            let mut text = " ".repeat(indent);
            text.push(c);
            let gap = opt.width - indent - 1;
            let final_line = next.lines.is_empty();
            let mut cost = (gap as u64) * (gap as u64);
            if final_line { cost /= 4; } else { cost += 3; }
            let mut lines = vec![Line { text, consumed: 1, marker: false, pieces: 1, gap }];
            lines.append(&mut next.lines);
            Layout { cost: cost + next.cost, marker_lines: next.marker_lines, lines }
        }
    }

    let mut memo = HashMap::new();
    solve(0, 0, true, words, options, &mut memo)
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word { chars: Vec::new(), markers: Vec::new() };
    let mut in_word = false;
    for &c in segment {
        let sep = c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}';
        if sep {
            if in_word {
                words.push(current);
                current = Word { chars: Vec::new(), markers: Vec::new() };
                in_word = false;
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
            in_word = true;
        } else {
            current.chars.push(c);
            in_word = true;
        }
    }
    if in_word {
        words.push(current);
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(idx, c) in &chars {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index: idx });
        }
    }

    let is_sep = |c: char| c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}';
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let left_ok = i > 0 && chars[i - 1].1 != '~' && !is_sep(chars[i - 1].1);
            let right_ok = i + 1 < chars.len() && chars[i + 1].1 != '~' && !is_sep(chars[i + 1].1);
            if !left_ok || !right_ok {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut p = 0usize;
    while p <= normalized.len() {
        if p == normalized.len() || normalized[p] != '\n' {
            p += 1;
            continue;
        }
        let run_start = p;
        while p < normalized.len() && normalized[p] == '\n' {
            p += 1;
        }
        let run = p - run_start;
        if run >= 2 {
            let segment = &normalized[start..run_start];
            let words = parse_segment(segment);
            if !words.is_empty() {
                let mut layout = optimize(&words, options);
                if options.justify {
                    let count = layout.lines.len();
                    for (line_index, line) in layout.lines.iter_mut().enumerate() {
                        if line_index + 1 < count && line.pieces >= 2 {
                            let gaps = line.pieces - 1;
                            let extra = line.gap;
                            let each = extra / gaps;
                            let rem = extra % gaps;
                            // Reconstruct using the original line's interword gaps.
                            let indent = if line_index == 0 { options.first_indent } else { options.rest_indent };
                            let body: String = line.text.chars().skip(indent).collect();
                            let parts: Vec<&str> = body.split(' ').collect();
                            let mut rebuilt = String::new();
                            for (j, part) in parts.iter().enumerate() {
                                if j > 0 {
                                    let offset = (j - 1 + gaps - (line_index % gaps)) % gaps;
                                    let add = each + usize::from(offset < rem);
                                    rebuilt.push_str(&" ".repeat(1 + add));
                                }
                                rebuilt.push_str(part);
                            }
                            line.text = " ".repeat(indent) + &rebuilt;
                        }
                    }
                }
                output.extend(layout.lines.into_iter().map(|l| l.text));
            }
            output.extend(std::iter::repeat(String::new()).take(run - 1));
            start = p;
        }
    }
    let words = parse_segment(&normalized[start..]);
    if !words.is_empty() {
        let mut layout = optimize(&words, options);
        if options.justify {
            let count = layout.lines.len();
            for (line_index, line) in layout.lines.iter_mut().enumerate() {
                if line_index + 1 < count && line.pieces >= 2 {
                    let gaps = line.pieces - 1;
                    let each = line.gap / gaps;
                    let rem = line.gap % gaps;
                    let indent = if line_index == 0 { options.first_indent } else { options.rest_indent };
                    let body: String = line.text.chars().skip(indent).collect();
                    let parts: Vec<&str> = body.split(' ').collect();
                    let mut rebuilt = String::new();
                    for (j, part) in parts.iter().enumerate() {
                        if j > 0 {
                            let offset = (j - 1 + gaps - (line_index % gaps)) % gaps;
                            let add = each + usize::from(offset < rem);
                            rebuilt.push_str(&" ".repeat(1 + add));
                        }
                        rebuilt.push_str(part);
                    }
                    line.text = " ".repeat(indent) + &rebuilt;
                }
            }
        }
        output.extend(layout.lines.into_iter().map(|l| l.text));
    }
    Ok(output)
}
