#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
enum Ending {
    Word,
    Marker(usize),
    Raw(usize),
}

#[derive(Clone)]
struct LineChoice {
    next_word: usize,
    next_offset: usize,
    ending: Ending,
    pieces: usize,
    consumed: usize,
    pre_string: String,
}

#[derive(Clone)]
struct Plan {
    cost: u64,
    marker_lines: usize,
    lines: usize,
    consumed: Vec<usize>,
    pre_strings: Vec<String>,
    choices: Vec<LineChoice>,
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn build_line(
    words: &[Word],
    start_word: usize,
    start_offset: usize,
    end_word: usize,
    end_offset: usize,
    ending: &Ending,
    indent: usize,
) -> (String, usize, usize) {
    let mut s = " ".repeat(indent);
    let mut pieces = 0usize;
    let mut displayed = indent;
    for wi in start_word..=end_word {
        let from = if wi == start_word { start_offset } else { 0 };
        let to = if wi == end_word {
            end_offset
        } else {
            words[wi].chars.len()
        };
        if wi > start_word {
            s.push(' ');
            displayed += 1;
        }
        for &c in &words[wi].chars[from..to] {
            s.push(c);
            displayed += 1;
        }
        pieces += 1;
    }
    if matches!(ending, Ending::Marker(_)) {
        s.push('-');
        displayed += 1;
    }
    (s, pieces, displayed)
}

fn solve(
    words: &[Word],
    options: &WrapOptions,
) -> Plan {
    let n = words.len();
    let mut memo: Vec<Vec<Option<Plan>>> = (0..=n)
        .map(|i| {
            if i == n {
                vec![Some(Plan {
                    cost: 0,
                    marker_lines: 0,
                    lines: 0,
                    consumed: Vec::new(),
                    pre_strings: Vec::new(),
                    choices: Vec::new(),
                })]
            } else {
                vec![None; words[i].chars.len() + 1]
            }
        })
        .collect();

    fn better(a: &Plan, b: &Plan) -> bool {
        if a.cost != b.cost {
            return a.cost < b.cost;
        }
        if a.marker_lines != b.marker_lines {
            return a.marker_lines < b.marker_lines;
        }
        if a.lines != b.lines {
            return a.lines < b.lines;
        }
        if a.consumed != b.consumed {
            return a.consumed > b.consumed;
        }
        a.pre_strings < b.pre_strings
    }

    fn recurse(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Vec<Option<Plan>>],
    ) -> Plan {
        if wi == words.len() {
            return Plan {
                cost: 0,
                marker_lines: 0,
                lines: 0,
                consumed: Vec::new(),
                pre_strings: Vec::new(),
                choices: Vec::new(),
            };
        }

        // Memoization is shared between first/rest states only within each call's
        // paragraph; the state has different capacities, so use a local recompute
        // and rely on the bounded input size.
        let indent = if first { options.first_indent } else { options.rest_indent };
        let capacity = options.width - indent;
        let mut best: Option<Plan> = None;
        let mut candidate_ends: Vec<(usize, usize, Ending)> = Vec::new();

        for ew in wi..words.len() {
            let from = if ew == wi { off } else { 0 };
            let len = words[ew].chars.len();
            let separators_count = ew - wi;
            let base = separators_count;
            if base >= capacity {
                break;
            }

            for end in from + 1..=len {
                let text_len = end - from;
                let display_len = base + text_len;
                if display_len > capacity {
                    break;
                }
                if end == len {
                    candidate_ends.push((ew, end, Ending::Word));
                } else {
                    for &m in words[ew].markers.iter().filter(|&&m| m > from && m < end + 1) {
                        let pre = m - from;
                        if base + pre + 1 <= capacity {
                            candidate_ends.push((ew, m, Ending::Marker(m)));
                        }
                    }
                }
            }

            // A completed word may always end a line, including when more words follow.
            if len > from && base + (len - from) <= capacity {
                candidate_ends.push((ew, len, Ending::Word));
            }

            // Raw splitting is only available on an otherwise empty line and only
            // when no marker prefix can fit.
            if ew == wi && base == 0 {
                let remaining = len - from;
                if remaining > capacity {
                    let marker_fits = words[wi].markers.iter().any(|&m| {
                        m > from && m < len && (m - from) + 1 <= capacity
                    });
                    if !marker_fits {
                        candidate_ends.push((wi, from + capacity, Ending::Raw(from + capacity));
                    }
                }
            }
        }

        candidate_ends.sort_by_key(|(ew, end, ending)| {
            (*ew, *end, matches!(ending, Ending::Word) as u8)
        });
        candidate_ends.dedup_by(|a, b| {
            a.0 == b.0 && a.1 == b.1
                && std::mem::discriminant(&a.2) == std::mem::discriminant(&b.2)
        });

        for (ew, end, ending) in candidate_ends {
            let next = match ending {
                Ending::Word => {
                    if end == words[ew].chars.len() {
                        (ew + 1, 0, false)
                    } else {
                        continue;
                    }
                }
                Ending::Marker(_) | Ending::Raw(_) => (ew, end, false),
            };
            let (line, pieces, displayed) =
                build_line(words, wi, off, ew, end, &ending, indent);
            let tail = recurse(next.0, next.1, false, words, options, memo);
            let gap = options.width - displayed;
            let final_line = tail.lines == 0;
            let mut line_cost = if final_line {
                (gap * gap / 4) as u64
            } else {
                (gap * gap) as u64
            };
            match ending {
                Ending::Marker(_) => line_cost += options.hyphen_cost as u64,
                Ending::Raw(_) => line_cost += (2 * options.width + 1) as u64,
                Ending::Word => {}
            }
            if !final_line && pieces == 1 {
                line_cost += 3;
            }

            let choice = LineChoice {
                next_word: next.0,
                next_offset: next.1,
                ending: ending.clone(),
                pieces,
                consumed: match ending {
                    Ending::Word | Ending::Marker(_) => {
                        let mut total = 0;
                        for x in wi..=ew {
                            let a = if x == wi { off } else { 0 };
                            let b = if x == ew { end } else { words[x].chars.len() };
                            total += b - a;
                        }
                        total
                    }
                    Ending::Raw(_) => end - off,
                },
                pre_string: line.clone(),
            };
            let mut plan = Plan {
                cost: line_cost + tail.cost,
                marker_lines: tail.marker_lines + matches!(ending, Ending::Marker(_)) as usize,
                lines: tail.lines + 1,
                consumed: vec![choice.consumed],
                pre_strings: vec![line],
                choices: vec![choice],
            };
            plan.consumed.extend(tail.consumed);
            plan.pre_strings.extend(tail.pre_strings);
            plan.choices.extend(tail.choices);
            if best.as_ref().map_or(true, |b| better(&plan, b)) {
                best = Some(plan);
            }
        }

        best.unwrap_or_else(|| {
            // Capacity is positive; the raw-split rule ensures progress.
            let end = (off + capacity).min(words[wi].chars.len());
            let (line, pieces, displayed) =
                build_line(words, wi, off, wi, end, &Ending::Raw(end), indent);
            let tail = recurse(wi, end, false, words, options, memo);
            let gap = options.width - displayed;
            let cost = (gap * gap) as u64 + (2 * options.width + 1) as u64;
            let choice = LineChoice {
                next_word: wi,
                next_offset: end,
                ending: Ending::Raw(end),
                pieces,
                consumed: end - off,
                pre_string: line.clone(),
            };
            let mut plan = Plan {
                cost: cost + tail.cost,
                marker_lines: tail.marker_lines,
                lines: tail.lines + 1,
                consumed: vec![end - off],
                pre_strings: vec![line],
                choices: vec![choice],
            };
            plan.consumed.extend(tail.consumed);
            plan.pre_strings.extend(tail.pre_strings);
            plan.choices.extend(tail.choices);
            plan
        })
    }

    let _ = &mut memo;
    recurse(0, 0, true, words, options, &mut memo)
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_side = |j: isize| -> bool {
                if j < 0 || j as usize >= chars.len() {
                    return false;
                }
                let c = chars[j as usize].1;
                c != '~' && !separators(c)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 2;
                continue;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut j = 0;
    while j < normalized.len() {
        if normalized[j] == '\n' {
            let start = j;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - start >= 2 {
                layout_segment(&segment, options, &mut output);
                segment.clear();
                for _ in 1..(j - start) {
                    output.push(String::new());
                }
                continue;
            }
            segment.push('\n');
        } else {
            segment.push(normalized[j]);
        }
        j += 1;
    }
    layout_segment(&segment, options, &mut output);
    Ok(output)
}

fn layout_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words = Vec::new();
    let mut current_chars = Vec::new();
    let mut current_markers = Vec::new();

    for &c in segment {
        if separators(c) {
            if !current_chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current_chars),
                    markers: std::mem::take(&mut current_markers),
                });
            }
        } else if c == '~' {
            current_markers.push(current_chars.len());
        } else {
            current_chars.push(c);
        }
    }
    if !current_chars.is_empty() {
        words.push(Word {
            chars: current_chars,
            markers: current_markers,
        });
    }
    if words.is_empty() {
        return;
    }

    let plan = solve(&words, options);
    let mut rendered = Vec::new();
    let mut word = 0usize;
    let mut offset = 0usize;
    for (line_index, choice) in plan.choices.iter().enumerate() {
        let indent = if line_index == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let (mut line, pieces, displayed) = build_line(
            &words,
            word,
            offset,
            match choice.ending {
                Ending::Word => choice.next_word.saturating_sub(1),
                Ending::Marker(_) | Ending::Raw(_) => choice.next_word,
            },
            match choice.ending {
                Ending::Word => {
                    if choice.next_word > word { words[choice.next_word - 1].chars.len() } else { choice.next_offset }
                }
                _ => choice.next_offset,
            },
            &choice.ending,
            indent,
        );

        if options.justify && line_index + 1 < plan.choices.len() && pieces >= 2 {
            let extra = options.width - displayed;
            let gaps = pieces - 1;
            let each = extra / gaps;
            let remainder = extra % gaps;
            let mut rebuilt = " ".repeat(indent);
            let mut wi = word;
            let mut off = offset;
            let mut gap_idx = 0usize;
            let end_word = match choice.ending {
                Ending::Word => choice.next_word - 1,
                _ => choice.next_word,
            };
            for current in wi..=end_word {
                let from = if current == wi { off } else { 0 };
                let to = if current == end_word {
                    match choice.ending {
                        Ending::Word => words[current].chars.len(),
                        _ => choice.next_offset,
                    }
                } else {
                    words[current].chars.len()
                };
                if current > wi {
                    let bonus = if (gap_idx + line_index % gaps) % gaps < remainder { 1 } else { 0 };
                    rebuilt.push_str(&" ".repeat(1 + each + bonus));
                    gap_idx += 1;
                }
                for &c in &words[current].chars[from..to] {
                    rebuilt.push(c);
                }
            }
            if matches!(choice.ending, Ending::Marker(_)) {
                rebuilt.push('-');
            }
            line = rebuilt;
        }
        rendered.push(line);
        word = choice.next_word;
        offset = choice.next_offset;
    }
    output.extend(rendered);
}
