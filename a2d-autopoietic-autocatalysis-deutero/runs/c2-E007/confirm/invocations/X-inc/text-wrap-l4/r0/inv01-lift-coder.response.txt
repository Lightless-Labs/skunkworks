#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

#[derive(Clone)]
struct Line {
    text: String,
    scalar_count: usize,
    pieces: usize,
    marker_ended: bool,
    final_line: bool,
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let av: Vec<usize> = a.lines.iter().map(|line| line.scalar_count).collect();
    let bv: Vec<usize> = b.lines.iter().map(|line| line.scalar_count).collect();
    if av != bv {
        return av > bv;
    }

    let at: Vec<&str> = a.lines.iter().map(|line| line.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|line| line.text.as_str()).collect();
    at < bt
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut Vec<Vec<[Option<Candidate>; 2]>>,
) -> Candidate {
    if wi == words.len() {
        return Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
        };
    }

    let fi = usize::from(first);
    if let Some(result) = &memo[wi][offset][fi] {
        return result.clone();
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let capacity = options.width - indent;
    let mut best: Option<Candidate> = None;

    let mut consider = |display: String,
                        consumed: usize,
                        next_wi: usize,
                        next_offset: usize,
                        marker_ended: bool,
                        raw: bool,
                        pieces: usize| {
        let tail = solve(
            words,
            next_wi,
            next_offset,
            false,
            options,
            memo,
        );
        let final_line = tail.lines.is_empty();
        let used = indent + display.chars().count();
        let gap = options.width - used;
        let mut cost = (gap * gap) as u64;
        if final_line {
            cost /= 4;
        }
        if marker_ended {
            cost += options.hyphen_cost as u64;
        }
        if raw {
            cost += (2 * options.width + 1) as u64;
        }
        if !final_line && pieces == 1 {
            cost += 3;
        }

        let mut text = " ".repeat(indent);
        text.push_str(&display);
        let mut lines = Vec::with_capacity(tail.lines.len() + 1);
        lines.push(Line {
            text,
            scalar_count: consumed,
            pieces,
            marker_ended,
            final_line,
        });
        lines.extend(tail.lines);
        let candidate = Candidate {
            cost: cost + tail.cost,
            marker_lines: usize::from(marker_ended) + tail.marker_lines,
            lines,
        };
        if best.as_ref().is_none_or(|current| better(&candidate, current)) {
            best = Some(candidate);
        }
    };

    let mut display = String::new();
    let mut consumed = 0usize;
    let mut pieces = 0usize;
    let mut current_wi = wi;
    let mut current_offset = offset;

    loop {
        let word = &words[current_wi];
        let remaining = word.chars.len() - current_offset;

        // A marker may end this line at any valid position after the current offset.
        for &marker in &word.markers {
            if marker <= current_offset || marker >= word.chars.len() {
                continue;
            }
            let prefix_len = marker - current_offset;
            let added_space = usize::from(pieces > 0);
            if display.chars().count() + added_space + prefix_len + 1 > capacity {
                continue;
            }
            let mut split_display = display.clone();
            if pieces > 0 {
                split_display.push(' ');
            }
            split_display.extend(word.chars[current_offset..marker].iter());
            split_display.push('-');
            consider(
                split_display,
                consumed + prefix_len,
                current_wi,
                marker,
                true,
                false,
                pieces + 1,
            );
        }

        let added_space = usize::from(pieces > 0);
        let available = capacity.saturating_sub(display.chars().count() + added_space);

        if remaining > available {
            // Raw split is allowed only on an otherwise empty line, and only
            // when no remaining marker prefix fits.
            if pieces == 0
                && display.is_empty()
                && remaining > capacity
                && !word.markers.iter().any(|&marker| {
                    marker > current_offset
                        && marker < word.chars.len()
                        && marker - current_offset + 1 <= capacity
                })
            {
                let mut raw_display = String::new();
                raw_display.extend(
                    word.chars[current_offset..current_offset + capacity]
                        .iter(),
                );
                let next_offset = current_offset + capacity;
                let (next_wi, next_offset) = if next_offset == word.chars.len() {
                    (current_wi + 1, 0)
                } else {
                    (current_wi, next_offset)
                };
                consider(
                    raw_display,
                    capacity,
                    next_wi,
                    next_offset,
                    false,
                    true,
                    1,
                );
            }
            break;
        }

        if pieces > 0 {
            display.push(' ');
        }
        display.extend(word.chars[current_offset..].iter());
        consumed += remaining;
        pieces += 1;
        current_wi += 1;
        current_offset = 0;

        if current_wi == words.len() {
            consider(
                display.clone(),
                consumed,
                current_wi,
                0,
                false,
                false,
                pieces,
            );
            break;
        }

        consider(
            display.clone(),
            consumed,
            current_wi,
            0,
            false,
            false,
            pieces,
        );

        // Continue by attempting to add the next complete word.
        let next_len = words[current_wi].chars.len();
        if display.chars().count() + 1 + next_len > capacity {
            break;
        }
    }

    let result = best.unwrap_or_else(|| {
        // Capacity is at least one; raw splitting guarantees progress.
        let word = &words[wi];
        let mut part = String::new();
        part.extend(word.chars[offset..offset + 1].iter());
        let next_offset = offset + 1;
        let (next_wi, next_offset) = if next_offset == word.chars.len() {
            (wi + 1, 0)
        } else {
            (wi, next_offset)
        };
        let tail = solve(words, next_wi, next_offset, false, options, memo);
        let mut text = " ".repeat(indent);
        text.push_str(&part);
        let mut lines = vec![Line {
            text,
            scalar_count: 1,
            pieces: 1,
            marker_ended: false,
            final_line: tail.lines.is_empty(),
        }];
        lines.extend(tail.lines);
        Candidate {
            cost: tail.cost + (2 * options.width + 1) as u64,
            marker_lines: tail.marker_lines,
            lines,
        }
    });

    memo[wi][offset][fi] = Some(result.clone());
    result
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn paragraph_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars_in_word = Vec::new();
    let mut markers = Vec::new();

    for &c in chars {
        if is_separator(c) {
            if !chars_in_word.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars_in_word),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars_in_word.len());
        } else {
            chars_in_word.push(c);
        }
    }

    if !chars_in_word.is_empty() {
        words.push(Word {
            chars: chars_in_word,
            markers,
        });
    }
    words
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut memo: Vec<Vec<[Option<Candidate>; 2]>> = words
        .iter()
        .map(|word| {
            (0..=word.chars.len())
                .map(|_| [None, None])
                .collect()
        })
        .collect();

    let mut candidate = solve(words, 0, 0, true, options, &mut memo);
    let line_count = candidate.lines.len();

    for (line_index, line) in candidate.lines.iter_mut().enumerate() {
        if !options.justify
            || line.final_line
            || line.pieces < 2
            || line_index + 1 == line_count
        {
            continue;
        }

        let indent = if line_index == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let current_width = line.text.chars().count();
        let extra = options.width - current_width;
        let gaps = line.pieces - 1;
        let each = extra / gaps;
        let remainder = extra % gaps;

        let mut rebuilt = String::new();
        let content = line.text.chars().skip(indent).collect::<String>();
        let pieces: Vec<&str> = content.split(' ').collect();
        rebuilt.push_str(&" ".repeat(indent));
        for (i, piece) in pieces.iter().enumerate() {
            if i > 0 {
                let cyclic = (i - 1 + gaps - (line_index % gaps)) % gaps;
                let add = each + usize::from(cyclic < remainder);
                rebuilt.push_str(&" ".repeat(1 + add));
            }
            rebuilt.push_str(piece);
        }
        line.text = rebuilt;
    }

    candidate.lines.into_iter().map(|line| line.text).collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= chars.len() {
                return false;
            }
            let c = chars[j as usize].1;
            c != '~' && !is_separator(c)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    if normalized.is_empty() {
        return Ok(Vec::new());
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let count = i - start;
            if count >= 2 {
                let words = paragraph_words(&segment);
                result.extend(layout_paragraph(&words, options));
                segment.clear();
                for _ in 1..count {
                    result.push(String::new());
                }
                continue;
            }
            segment.push('\n');
        } else {
            segment.push(normalized[i]);
        }
        i += 1;
    }
    let words = paragraph_words(&segment);
    result.extend(layout_paragraph(&words, options));
    Ok(result)
}
