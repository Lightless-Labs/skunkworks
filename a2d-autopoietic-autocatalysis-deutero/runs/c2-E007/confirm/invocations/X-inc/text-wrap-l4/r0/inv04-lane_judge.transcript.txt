fn plan_cmp(a: &Plan, b: &Plan) -> Ordering {
    a.cost
        .cmp(&b.cost)
        .then_with(|| a.marker_count.cmp(&b.marker_count))
        .then_with(|| a.lines.len().cmp(&b.lines.len()))
        .then_with(|| {
            for (x, y) in a.consumed.iter().zip(&b.consumed) {
                match y.cmp(x) {
                    Ordering::Equal => {}
                    order => return order,
                }
            }
            a.consumed.len().cmp(&b.consumed.len())
        })
        .then_with(|| a.pre_rendered.cmp(&b.pre_rendered))
}

#[allow(clippy::too_many_arguments)]
fn consider_line(
    best: &mut Option<Plan>,
    _state: State,
    next: State,
    words: &[Word],
    opt: &WrapOptions,
    memo: &mut HashMap<State, Option<Plan>>,
    pieces: Vec<Piece>,
    displayed: usize,
    marker_end: bool,
    raw_split: bool,
    consumed: usize,
) {
    let indent = if _state.first {
        opt.first_indent
    } else {
        opt.rest_indent
    };
    let line_width = indent + displayed;
    let gap = opt.width - line_width;
    let suffix = match solve(next, words, opt, memo) {
        Some(p) => p,
        None => return,
    };
    let is_final = suffix.lines.is_empty();
    let mut cost = suffix.cost + if is_final {
        (gap * gap / 4) as u64
    } else {
        (gap * gap) as u64
    };
    if marker_end {
        cost += opt.hyphen_cost as u64;
    }
    if raw_split {
        cost += (2 * opt.width + 1) as u64;
    }
    if !is_final && pieces.len() == 1 {
        cost += 3;
    }

    let mut line = Line {
        pieces,
        consumed,
        gap,
        marker_end,
        raw_split,
        indent,
    };
    let mut candidate = Plan {
        cost,
        marker_count: suffix.marker_count + usize::from(marker_end),
        lines: Vec::with_capacity(suffix.lines.len() + 1),
        consumed: Vec::with_capacity(suffix.consumed.len() + 1),
        pre_rendered: Vec::with_capacity(suffix.pre_rendered.len() + 1),
    };
    candidate.lines.push(line.clone());
    candidate.lines.extend(suffix.lines);
    candidate.consumed.push(consumed);
    candidate.consumed.extend(suffix.consumed);
    candidate.pre_rendered.push(render_pre(&line));
    candidate.pre_rendered.extend(suffix.pre_rendered);

    if best.as_ref().map_or(true, |old| plan_cmp(&candidate, old) == Ordering::Less) {
        *best = Some(candidate);
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn make_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        while i < chars.len() && is_separator(chars[i]) {
            i += 1;
        }
        if i == chars.len() {
            break;
        }
        let mut spelling = Vec::new();
        let mut markers = Vec::new();
        while i < chars.len() && !is_separator(chars[i]) {
            if chars[i] == '~' {
                markers.push(spelling.len());
            } else {
                spelling.push(chars[i]);
            }
            i += 1;
        }
        words.push(Word {
            chars: spelling,
            markers,
        });
    }
    words
}

fn solve_paragraph(words: &[Word], opt: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut memo = HashMap::new();
    let plan = solve(
        State {
            word: 0,
            offset: 0,
            first: true,
        },
        words,
        opt,
        &mut memo,
    )
    .expect("raw splits guarantee a legal layout");

    let mut output = Vec::with_capacity(plan.lines.len());
    for (line_index, line) in plan.lines.iter().enumerate() {
        let mut rendered = " ".repeat(line.indent);
        let gap_count = line.pieces.len().saturating_sub(1);
        let justify = opt.justify
            && line_index + 1 < plan.lines.len()
            && line.pieces.len() >= 2;
        let extra = if justify { line.gap } else { 0 };
        let each = if justify && gap_count > 0 {
            extra / gap_count
        } else {
            0
        };
        let rem = if justify && gap_count > 0 {
            extra % gap_count
        } else {
            0
        };
        for (i, piece) in line.pieces.iter().enumerate() {
            if i > 0 {
                rendered.push(' ');
                if justify {
                    let rotation = line_index % gap_count;
                    let cyclic_index = (i - 1 + gap_count - rotation) % gap_count;
                    rendered.push_str(&" ".repeat(each + usize::from(cyclic_index < rem)));
                }
            }
            rendered.push_str(&piece.text);
        }
        output.push(rendered);
    }
    output
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<char> = input.chars().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    let mut byte_index = 0;
    for &c in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        byte_index += c.len_utf8();
    }

    byte_index = 0;
    let mut marker_error = None;
    for (i, &c) in chars.iter().enumerate() {
        if c == '~' {
            let valid_left = i > 0 && !is_separator(chars[i - 1]) && chars[i - 1] != '~';
            let valid_right =
                i + 1 < chars.len() && !is_separator(chars[i + 1]) && chars[i + 1] != '~';
            if (!valid_left || !valid_right) && marker_error.is_none() {
                marker_error = Some(byte_index);
            }
        }
        byte_index += c.len_utf8();
    }
    if let Some(index) = marker_error {
        return Err(WrapError::InvalidMarker { index });
    }

    // Normalize CRLF pairs to a single LF, and any remaining CR to LF.
    let mut normalized = Vec::with_capacity(chars.len());
    let mut i = 0;
    while i < chars.len() {
        if chars[i] == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1] == '\n' {
                i += 1;
            }
        } else {
            normalized.push(chars[i]);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut pos = 0;
    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let run_start = pos;
            while pos < normalized.len() && normalized[pos] == '\n' {
                pos += 1;
            }
            if pos - run_start >= 2 {
                let words = make_words(&normalized[segment_start..run_start]);
                output.extend(solve_paragraph(&words, options));
                output.extend(std::iter::repeat(String::new()).take(pos - run_start - 1));
                segment_start = pos;
            }
        } else {
            pos += 1;
        }
    }
    let words = make_words(&normalized[segment_start..]);
    output.extend(solve_paragraph(&words, options));
    Ok(output)
}

fn better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    let av: Vec<usize> = a.lines.iter().map(|l| l.consumed).collect();
    let bv: Vec<usize> = b.lines.iter().map(|l| l.consumed).collect();
    if av != bv {
        return av > bv;
    }
    let at: Vec<&str> = a.lines.iter().map(|l| l.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|l| l.text.as_str()).collect();
    at < bt
}

        words: &[Word],
        opt: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Layout>,
    ) -> Layout {
        if wi >= words.len() {
            return Layout { cost: 0, marker_lines: 0, lines: Vec::new() };
        }
        if let Some(v) = memo.get(&(wi, off, first)) {
            return v.clone();
        }

        let indent = if first { opt.first_indent } else { opt.rest_indent };
        let cap = opt.width - indent;
        let mut best: Option<Layout> = None;

        let mut consider = |display: String,
                            consumed: usize,
                            next_wi: usize,
                            next_off: usize,
                            marker: bool,
                            raw: bool,
                            pieces: usize| {
            if display.is_empty() {
                return;
            }
            let display_count = display.chars().count();
            let gap = opt.width - indent - display_count;
            let tail = solve(next_wi, next_off, false, words, opt, memo);
            let final_line = tail.lines.is_empty();
            let mut cost = (gap as u64) * (gap as u64);
            if final_line {
                cost /= 4;
            }
            if marker {
                cost += opt.hyphen_cost as u64;
            }
            if raw {
                cost += (2 * opt.width + 1) as u64;
            }
            if !final_line && pieces == 1 {
                cost += 3;
            }
            let mut text = " ".repeat(indent);
            text.push_str(&display);
            let mut lines = Vec::with_capacity(tail.lines.len() + 1);
            lines.push(Line { text, consumed, marker, pieces, gap });
            lines.extend(tail.lines);
            let candidate = Layout {
                cost: cost + tail.cost,
                marker_lines: tail.marker_lines + usize::from(marker),
                lines,
            };
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        };

        let word = &words[wi];
        let remaining = word.chars.len() - off;
        let mut display = String::new();
        let mut consumed = 0usize;
        let mut pieces = 0usize;
        let mut cur_wi = wi;
        let mut cur_off = off;

        loop {
            if cur_wi >= words.len() {
                break;
            }
            let w = &words[cur_wi];
            let start = cur_off;
            let rem = w.chars.len() - start;
            let separator = if pieces == 0 { 0 } else { 1 };
            if display.chars().count() + separator + rem <= cap {
                if separator == 1 {
                    display.push(' ');
                }
                for &c in &w.chars[start..] {
                    display.push(c);
                }
                consumed += rem;
                pieces += 1;
                cur_wi += 1;
                cur_off = 0;
                consider(display.clone(), consumed, cur_wi, 0, false, false, pieces);
                continue;
            }

            let room = cap.saturating_sub(display.chars().count() + separator);
            if room > 0 {
                for &m in w.markers.iter().filter(|&&m| m > start && m < w.chars.len()) {
                    let prefix = m - start;
                    if prefix + 1 <= room {
                        let mut d = display.clone();
                        if separator == 1 {
                            d.push(' ');
                        }
                        d.extend(w.chars[start..m].iter());
                        d.push('-');
                        consider(d, consumed + prefix, cur_wi, m, true, false, pieces + 1);
                    }
                }
            }

            if pieces == 0 && rem > cap {
                let marker_fits = w.markers.iter().any(|&m| {
                    m > start && m < w.chars.len() && (m - start + 1) <= cap
                });
                if !marker_fits {
                    let mut d = String::new();
                    d.extend(w.chars[start..start + cap].iter());
                    let next_off = start + cap;
                    let (nwi, noff) = if next_off == w.chars.len() {
                        (cur_wi + 1, 0)
                    } else {
                        (cur_wi, next_off)
                    };
                    consider(d, cap, nwi, noff, false, true, 1);
                }
            }
            break;
        }

        if let Some(result) = best {
            memo.insert((wi, off, first), result.clone());
            result
        } else {
            // Capacity is always positive; this fallback is reached only for a
            // single scalar that fits on an otherwise empty line.
            let c = word.chars[off];
            let mut next = solve(wi + usize::from(off + 1 == word.chars.len()), if off + 1 == word.chars.len() { 0 } else { off + 1 }, false, words, opt, memo);
            let mut text = " ".repeat(indent);
            text.push(c);
            let gap = opt.width - indent - 1;
            let final_line = next.lines.is_empty();
            let mut cost = (gap as u64) * (gap as u64);
            if final_line { cost /= 4; } else { cost += 3; }
            let mut lines = vec![Line { text, consumed: 1, marker: false, pieces: 1, gap }];
            lines.append(&mut next.lines);
            Layout { cost: cost + next.cost, marker_lines: next.marker_lines, lines }
        }
    }

    let mut memo = HashMap::new();
    solve(0, 0, true, words, options, &mut memo)
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word { chars: Vec::new(), markers: Vec::new() };
    let mut in_word = false;
    for &c in segment {
        let sep = c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}';
        if sep {
            if in_word {
                words.push(current);
                current = Word { chars: Vec::new(), markers: Vec::new() };
                in_word = false;
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
            in_word = true;
        } else {
            current.chars.push(c);
            in_word = true;
        }
    }
    if in_word {
        words.push(current);
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(idx, c) in &chars {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index: idx });
        }
    }

    let is_sep = |c: char| c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}';
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let left_ok = i > 0 && chars[i - 1].1 != '~' && !is_sep(chars[i - 1].1);
            let right_ok = i + 1 < chars.len() && chars[i + 1].1 != '~' && !is_sep(chars[i + 1].1);
            if !left_ok || !right_ok {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut p = 0usize;
    while p <= normalized.len() {
        if p == normalized.len() || normalized[p] != '\n' {
            p += 1;
            continue;
        }
        let run_start = p;
        while p < normalized.len() && normalized[p] == '\n' {
            p += 1;
        }
        let run = p - run_start;
        if run >= 2 {
            let segment = &normalized[start..run_start];
            let words = parse_segment(segment);
            if !words.is_empty() {
                let mut layout = optimize(&words, options);
                if options.justify {
                    let count = layout.lines.len();
                    for (line_index, line) in layout.lines.iter_mut().enumerate() {
                        if line_index + 1 < count && line.pieces >= 2 {
                            let gaps = line.pieces - 1;
                            let extra = line.gap;
                            let each = extra / gaps;
                            let rem = extra % gaps;
                            // Reconstruct using the original line's interword gaps.
                            let indent = if line_index == 0 { options.first_indent } else { options.rest_indent };
                            let body: String = line.text.chars().skip(indent).collect();
                            let parts: Vec<&str> = body.split(' ').collect();
                            let mut rebuilt = String::new();
                            for (j, part) in parts.iter().enumerate() {
                                if j > 0 {
                                    let offset = (j - 1 + gaps - (line_index % gaps)) % gaps;
                                    let add = each + usize::from(offset < rem);
                                    rebuilt.push_str(&" ".repeat(1 + add));
                                }
                                rebuilt.push_str(part);
                            }
                            line.text = " ".repeat(indent) + &rebuilt;
                        }
                    }
                }
                output.extend(layout.lines.into_iter().map(|l| l.text));
            }
            output.extend(std::iter::repeat(String::new()).take(run - 1));
            start = p;
        }
    }
    let words = parse_segment(&normalized[start..]);
    if !words.is_empty() {
        let mut layout = optimize(&words, options);
        if options.justify {
            let count = layout.lines.len();
            for (line_index, line) in layout.lines.iter_mut().enumerate() {
                if line_index + 1 < count && line.pieces >= 2 {
                    let gaps = line.pieces - 1;
                    let each = line.gap / gaps;
                    let rem = line.gap % gaps;
                    let indent = if line_index == 0 { options.first_indent } else { options.rest_indent };
                    let body: String = line.text.chars().skip(indent).collect();
                    let parts: Vec<&str> = body.split(' ').collect();
                    let mut rebuilt = String::new();
                    for (j, part) in parts.iter().enumerate() {
                        if j > 0 {
                            let offset = (j - 1 + gaps - (line_index % gaps)) % gaps;
                            let add = each + usize::from(offset < rem);
                            rebuilt.push_str(&" ".repeat(1 + add));
                        }
                        rebuilt.push_str(part);
                    }
                    line.text = " ".repeat(indent) + &rebuilt;
                }
            }
        }
        output.extend(layout.lines.into_iter().map(|l| l.text));
    }
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

#[derive(Clone)]
struct Line {
    text: String,
    scalar_count: usize,
    pieces: usize,
    marker_ended: bool,
    final_line: bool,
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let av: Vec<usize> = a.lines.iter().map(|line| line.scalar_count).collect();
    let bv: Vec<usize> = b.lines.iter().map(|line| line.scalar_count).collect();
    if av != bv {
        return av > bv;
    }

    let at: Vec<&str> = a.lines.iter().map(|line| line.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|line| line.text.as_str()).collect();
    at < bt
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut Vec<Vec<[Option<Candidate>; 2]>>,
) -> Candidate {
    if wi == words.len() {
        return Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
        };
    }

    let fi = usize::from(first);
    if let Some(result) = &memo[wi][offset][fi] {
        return result.clone();
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let capacity = options.width - indent;
    let mut best: Option<Candidate> = None;

    let mut consider = |display: String,
                        consumed: usize,
                        next_wi: usize,
                        next_offset: usize,
                        marker_ended: bool,
                        raw: bool,
                        pieces: usize| {
        let tail = solve(
            words,
            next_wi,
            next_offset,
            false,
            options,
            memo,
        );
        let final_line = tail.lines.is_empty();
        let used = indent + display.chars().count();
        let gap = options.width - used;
        let mut cost = (gap * gap) as u64;
        if final_line {
            cost /= 4;
        }
        if marker_ended {
            cost += options.hyphen_cost as u64;
        }
        if raw {
            cost += (2 * options.width + 1) as u64;
        }
        if !final_line && pieces == 1 {
            cost += 3;
        }

        let mut text = " ".repeat(indent);
        text.push_str(&display);
        let mut lines = Vec::with_capacity(tail.lines.len() + 1);
        lines.push(Line {
            text,
            scalar_count: consumed,
            pieces,
            marker_ended,
            final_line,
        });
        lines.extend(tail.lines);
        let candidate = Candidate {
            cost: cost + tail.cost,
            marker_lines: usize::from(marker_ended) + tail.marker_lines,
            lines,
        };
        if best.as_ref().is_none_or(|current| better(&candidate, current)) {
            best = Some(candidate);
        }
    };

    let mut display = String::new();
    let mut consumed = 0usize;
    let mut pieces = 0usize;
    let mut current_wi = wi;
    let mut current_offset = offset;

    loop {
        let word = &words[current_wi];
        let remaining = word.chars.len() - current_offset;

        // A marker may end this line at any valid position after the current offset.
        for &marker in &word.markers {
            if marker <= current_offset || marker >= word.chars.len() {
                continue;
            }
            let prefix_len = marker - current_offset;
            let added_space = usize::from(pieces > 0);
            if display.chars().count() + added_space + prefix_len + 1 > capacity {
                continue;
            }
            let mut split_display = display.clone();
            if pieces > 0 {
                split_display.push(' ');
            }
            split_display.extend(word.chars[current_offset..marker].iter());
            split_display.push('-');
            consider(
                split_display,
                consumed + prefix_len,
                current_wi,
                marker,
                true,
                false,
                pieces + 1,
            );
        }

        let added_space = usize::from(pieces > 0);
        let available = capacity.saturating_sub(display.chars().count() + added_space);

        if remaining > available {
            // Raw split is allowed only on an otherwise empty line, and only
            // when no remaining marker prefix fits.
            if pieces == 0
                && display.is_empty()
                && remaining > capacity
                && !word.markers.iter().any(|&marker| {
                    marker > current_offset
                        && marker < word.chars.len()
                        && marker - current_offset + 1 <= capacity
                })
            {
                let mut raw_display = String::new();
                raw_display.extend(
                    word.chars[current_offset..current_offset + capacity]
                        .iter(),
                );
                let next_offset = current_offset + capacity;
                let (next_wi, next_offset) = if next_offset == word.chars.len() {
                    (current_wi + 1, 0)
                } else {
                    (current_wi, next_offset)
                };
                consider(
                    raw_display,
                    capacity,
                    next_wi,
                    next_offset,
                    false,
                    true,
                    1,
                );
            }
            break;
        }

        if pieces > 0 {
            display.push(' ');
        }
        display.extend(word.chars[current_offset..].iter());
        consumed += remaining;
        pieces += 1;
        current_wi += 1;
        current_offset = 0;

        if current_wi == words.len() {
            consider(
                display.clone(),
                consumed,
                current_wi,
                0,
                false,
                false,
                pieces,
            );
            break;
        }

        consider(
            display.clone(),
            consumed,
            current_wi,
            0,
            false,
            false,
            pieces,
        );

        // Continue by attempting to add the next complete word.
        let next_len = words[current_wi].chars.len();
        if display.chars().count() + 1 + next_len > capacity {
            break;
        }
    }

    let result = best.unwrap_or_else(|| {
        // Capacity is at least one; raw splitting guarantees progress.
        let word = &words[wi];
        let mut part = String::new();
        part.extend(word.chars[offset..offset + 1].iter());
        let next_offset = offset + 1;
        let (next_wi, next_offset) = if next_offset == word.chars.len() {
            (wi + 1, 0)
        } else {
            (wi, next_offset)
        };
        let tail = solve(words, next_wi, next_offset, false, options, memo);
        let mut text = " ".repeat(indent);
        text.push_str(&part);
        let mut lines = vec![Line {
            text,
            scalar_count: 1,
            pieces: 1,
            marker_ended: false,
            final_line: tail.lines.is_empty(),
        }];
        lines.extend(tail.lines);
        Candidate {
            cost: tail.cost + (2 * options.width + 1) as u64,
            marker_lines: tail.marker_lines,
            lines,
        }
    });

    memo[wi][offset][fi] = Some(result.clone());
    result
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn paragraph_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars_in_word = Vec::new();
    let mut markers = Vec::new();

    for &c in chars {
        if is_separator(c) {
            if !chars_in_word.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars_in_word),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars_in_word.len());
        } else {
            chars_in_word.push(c);
        }
    }

    if !chars_in_word.is_empty() {
        words.push(Word {
            chars: chars_in_word,
            markers,
        });
    }
    words
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut memo: Vec<Vec<[Option<Candidate>; 2]>> = words
        .iter()
        .map(|word| {
            (0..=word.chars.len())
                .map(|_| [None, None])
                .collect()
        })
        .collect();

    let mut candidate = solve(words, 0, 0, true, options, &mut memo);
    let line_count = candidate.lines.len();

    for (line_index, line) in candidate.lines.iter_mut().enumerate() {
        if !options.justify
            || line.final_line
            || line.pieces < 2
            || line_index + 1 == line_count
        {
            continue;
        }

        let indent = if line_index == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let current_width = line.text.chars().count();
        let extra = options.width - current_width;
        let gaps = line.pieces - 1;
        let each = extra / gaps;
        let remainder = extra % gaps;

        let mut rebuilt = String::new();
        let content = line.text.chars().skip(indent).collect::<String>();
        let pieces: Vec<&str> = content.split(' ').collect();
        rebuilt.push_str(&" ".repeat(indent));
        for (i, piece) in pieces.iter().enumerate() {
            if i > 0 {
                let cyclic = (i - 1 + gaps - (line_index % gaps)) % gaps;
                let add = each + usize::from(cyclic < remainder);
                rebuilt.push_str(&" ".repeat(1 + add));
            }
            rebuilt.push_str(piece);
        }
        line.text = rebuilt;
    }

    candidate.lines.into_iter().map(|line| line.text).collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= chars.len() {
                return false;
            }
            let c = chars[j as usize].1;
            c != '~' && !is_separator(c)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    if normalized.is_empty() {
        return Ok(Vec::new());
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let count = i - start;
            if count >= 2 {
                let words = paragraph_words(&segment);
                result.extend(layout_paragraph(&words, options));
                segment.clear();
                for _ in 1..count {
                    result.push(String::new());
                }
                continue;
            }
            segment.push('\n');
        } else {
            segment.push(normalized[i]);
        }
        i += 1;
    }
    let words = paragraph_words(&segment);
    result.extend(layout_paragraph(&words, options));
    Ok(result)
}
