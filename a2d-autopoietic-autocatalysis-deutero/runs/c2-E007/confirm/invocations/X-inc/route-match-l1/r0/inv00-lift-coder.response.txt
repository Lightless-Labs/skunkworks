#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
            })
            .collect();
        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != request.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matched = true;
            for (pattern_segment, value) in route.segments.iter().zip(&request) {
                match pattern_segment {
                    Segment::Static(expected) if expected == value => {}
                    Segment::Static(_) => {
                        matched = false;
                        break;
                    }
                    Segment::Param(name) => params.push((name.clone(), value.clone())),
                }
            }
            if !matched {
                continue;
            }

            match best {
                None => best = Some((route_id, params)),
                Some((best_id, _)) => {
                    let best_rank = &self.routes[best_id].rank;
                    if route.rank > *best_rank {
                        best = Some((route_id, params));
                    }
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let body = &input[1..];
    let trailing = body.ends_with('/');
    let content = if trailing { &body[..body.len() - 1] } else { body };

    let mut result = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut offset = 1;

    for raw_segment in content.split('/') {
        if raw_segment.is_empty() {
            return Err(Error::InvalidPattern { at: offset });
        }

        if raw_segment.starts_with(':') {
            let name = &raw_segment[1..];
            let valid = {
                let bytes = name.as_bytes();
                !bytes.is_empty()
                    && bytes[0].is_ascii_lowercase()
                    && bytes[1..]
                        .iter()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
            };
            if !valid || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: offset });
            }
            result.push(Segment::Param(name.to_string()));
        } else if raw_segment.starts_with('*') {
            return Err(Error::InvalidPattern { at: offset });
        } else {
            result.push(Segment::Static(decode_segment(raw_segment, offset, true)?));
        }

        offset += raw_segment.len() + 1;
    }

    if trailing {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }
    Ok(result)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let body = &input[1..];
    let trailing = body.ends_with('/');
    let content = if trailing { &body[..body.len() - 1] } else { body };

    let mut result = Vec::new();
    let mut offset = 1;

    for raw_segment in content.split('/') {
        if raw_segment.is_empty() {
            return Err(Error::InvalidPath { at: offset });
        }
        result.push(decode_segment(raw_segment, offset, false).map_err(|e| match e {
            Error::InvalidPattern { at } => Error::InvalidPath { at },
            _ => unreachable!(),
        })?);
        offset += raw_segment.len() + 1;
    }

    if trailing {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }
    Ok(result)
}

fn decode_segment(input: &str, base: usize, _pattern: bool) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(Error::InvalidPattern { at: base + i });
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            let (Some(hi), Some(lo)) = (hi, lo) else {
                return Err(Error::InvalidPattern { at: base + i });
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(Error::InvalidPattern { at: base + i });
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(Error::InvalidPattern { at: base + i });
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}
