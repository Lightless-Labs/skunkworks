#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Debug)]
struct HunkRow {
    kind: u8,
    payload: String,
    line: usize,
}

#[derive(Debug)]
struct Hunk {
    old_start: i64,
    old_count: usize,
    new_start: i64,
    new_count: usize,
    header_line: usize,
    rows: Vec<HunkRow>,
}

fn parse_canonical_number(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.starts_with('0')) {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let rest = line.strip_prefix("@@ -").ok_or_else(bad)?;
    let (old, rest) = rest.split_once(" +").ok_or_else(bad)?;
    let (new, suffix) = rest.split_once(" @@").ok_or_else(bad)?;
    if suffix.is_empty() == false || old.contains(' ') || new.contains(' ') {
        return Err(bad());
    }
    let (a, b) = old.split_once(',').ok_or_else(bad)?;
    let (c, d) = new.split_once(',').ok_or_else(bad)?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return Err(bad());
    }
    let a = parse_canonical_number(a).ok_or_else(bad)?;
    let b = parse_canonical_number(b).ok_or_else(bad)?;
    let c = parse_canonical_number(c).ok_or_else(bad)?;
    let d = parse_canonical_number(d).ok_or_else(bad)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    Ok((a, b, c, d))
}

fn full_start(start: usize, count: usize) -> i64 {
    if count > 0 {
        (start - 1) as i64
    } else {
        start as i64
    }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    if lines.get(0).copied() != Some("--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if lines.get(1).copied() != Some("+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end = 0i64;
    let mut net = 0i64;

    while i < lines.len() {
        let header_line = i + 1;
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let (a, b, c, d) = parse_hunk_header(lines[i], header_line)?;
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);
        i += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let row = lines[i];
            let bytes = row.as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            if row.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let kind = bytes[0];
            let payload = &row[1..];
            match kind {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                }
                b'-' => old_seen += 1,
                b'+' => new_seen += 1,
                _ => unreachable!(),
            }
            rows.push(HunkRow {
                kind,
                payload: payload.to_owned(),
                line: row_line,
            });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end || new_start != old_start + net {
            return Err(ApplyError::Misordered { line: header_line });
        }
        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        previous_old_end = old_start + b as i64;
        net += d as i64 - b as i64;
        hunks.push(Hunk {
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            header_line,
            rows,
        });
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net = 0i64;
    let mut drift = 0i64;

    for (index, hunk) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < hunk.rows.len() && hunk.rows[lead].kind == b' ' {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < hunk.rows.len()
            && hunk.rows[hunk.rows.len() - 1 - trail].kind == b' '
        {
            trail += 1;
        }

        let expected = hunk.old_start + net + drift;
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let mut best_rank: Option<(usize, i64)> = None;
        let mut best_starts = Vec::<usize>::new();
        let mut selected: Option<(usize, usize, usize)> = None;

        for dropped in 0..=max_l + max_r {
            for left in 0..=max_l {
                if dropped < left {
                    continue;
                }
                let right = dropped - left;
                if right > max_r {
                    continue;
                }
                if left + right != dropped {
                    continue;
                }

                for start in 0..=working.len() {
                    if start + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (start as i64 - expected).abs();
                    if distance > options.max_offset as i64 {
                        continue;
                    }

                    let mut old_payloads = Vec::new();
                    for row in hunk.rows.iter().skip(left).take(hunk.rows.len() - left - right) {
                        if row.kind == b' ' || row.kind == b'-' {
                            old_payloads.push(row.payload.as_str());
                        }
                    }
                    if old_payloads.len() != hunk.old_count - left - right {
                        continue;
                    }
                    if old_payloads
                        .iter()
                        .enumerate()
                        .any(|(offset, payload)| working[start + left + offset] != *payload)
                    {
                        continue;
                    }

                    let rank = (dropped, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            selected = Some((start, left, right));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            selected = Some((start, left, right));
                        }
                        Some(current) if rank == current => {
                            if !best_starts.contains(&start) {
                                best_starts.push(start);
                            }
                            if let Some((chosen_start, chosen_left, _)) = selected {
                                if start == chosen_start && left < chosen_left {
                                    selected = Some((start, left, right));
                                }
                            }
                        }
                        _ => {}
                    }
                }
            }
            if let Some((rank_dropped, _)) = best_rank {
                if rank_dropped < dropped {
                    break;
                }
            }
        }

        let hunk_number = index + 1;
        let Some((start, left, right)) = selected else {
            return Err(ApplyError::NoMatch { hunk: hunk_number });
        };

        best_starts.sort_unstable();
        best_starts.dedup();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_number,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let trim_end = hunk.rows.len() - right;
        let mut replacement = Vec::new();
        for row in hunk.rows.iter().take(trim_end).skip(left) {
            if row.kind == b' ' || row.kind == b'+' {
                replacement.push(row.payload.clone());
            }
        }

        let replace_start = start + left;
        let replace_end = start + hunk.old_count - right;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        drift = start as i64 - (hunk.old_start + net);
        net += hunk.new_count as i64 - hunk.old_count as i64;
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
