#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Row {
    kind: u8,
    payload: String,
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.starts_with('0')) {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let value = s.parse::<usize>().ok()?;
    (value <= 128).then_some(value)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, tail) = rest.split_once(" @@")?;
    if tail.is_empty() == false {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    let a = parse_decimal(a)?;
    let b = parse_decimal(b)?;
    let c = parse_decimal(c)?;
    let d = parse_decimal(d)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn physical_lines(patch: &str) -> Vec<&str> {
    let mut lines: Vec<&str> = patch.split_terminator('\n').collect();
    if patch.is_empty() {
        lines.clear();
    }
    lines
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let mut source_lines: Vec<String> = Vec::new();
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if !source.is_empty() {
        source_lines = source[..source.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect();
        if source_lines.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
    }

    if patch.is_empty() {
        return Ok(source.to_owned());
    }

    let lines = physical_lines(patch);
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    if lines.get(0) != Some(&"--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if lines.get(1) != Some(&"+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    if !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut index = 2;
    let mut previous_old_end: i128 = 0;
    let mut net: i128 = 0;

    while index < lines.len() {
        let header_line = index + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[index]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        index += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while index < lines.len() && !lines[index].starts_with("@@") {
            let row_line = index + 1;
            let bytes = lines[index].as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &lines[index][1..];
            if payload.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let kind = bytes[0];
            if kind == b' ' || kind == b'-' {
                old_seen += 1;
            }
            if kind == b' ' || kind == b'+' {
                new_seen += 1;
            }
            rows.push(Row {
                kind,
                payload: payload.to_owned(),
            });
            index += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_start_i = old_start as i128;
        if old_start_i < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start_i + net;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_end = old_start_i + b as i128;
        net += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });

        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut net_applied: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
        let expected = hunk.old_start as i128 + net_applied + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut matches: Vec<(usize, usize, usize, Vec<String>, Vec<String>)> = Vec::new();

        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > lead + trail {
                    continue;
                }
                let mut old_payloads = Vec::new();
                let mut replacement = Vec::new();
                let mut row_pos = 0usize;
                for row in &hunk.rows {
                    let is_dropped = (row.kind == b' ' && row_pos < l)
                        || (row.kind == b' '
                            && row_pos >= hunk.rows.len().saturating_sub(r));
                    if row.kind == b' ' {
                        row_pos += 1;
                    }
                    if is_dropped {
                        continue;
                    }
                    if row.kind == b' ' || row.kind == b'-' {
                        old_payloads.push(row.payload.clone());
                    }
                    if row.kind == b' ' || row.kind == b'+' {
                        replacement.push(row.payload.clone());
                    }
                }
                let dropped_left = l;
                let dropped_right = r;

                for s in 0..=source_lines.len() {
                    if s + hunk.old_count > source_lines.len() {
                        continue;
                    }
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let start = s + dropped_left;
                    let end = s + hunk.old_count - dropped_right;
                    if end < start || end > source_lines.len() {
                        continue;
                    }
                    if source_lines[start..end] != old_payloads {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            matches.clear();
                            matches.push((s, l, r, old_payloads.clone(), replacement.clone()));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            matches.clear();
                            matches.push((s, l, r, old_payloads.clone(), replacement.clone()));
                        }
                        Some(current) if rank == current => {
                            matches.push((s, l, r, old_payloads.clone(), replacement.clone()));
                        }
                        _ => {}
                    }
                }
            }
        }

        if matches.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }

        matches.sort_by_key(|m| (m.0, m.1));
        let first_start = matches[0].0;
        let mut distinct_starts: Vec<usize> = matches.iter().map(|m| m.0).collect();
        distinct_starts.sort_unstable();
        distinct_starts.dedup();
        if distinct_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: distinct_starts[0],
                second: distinct_starts[1],
            });
        }

        let chosen = matches
            .iter()
            .filter(|m| m.0 == first_start)
            .min_by_key(|m| m.1)
            .unwrap();
        let s = chosen.0;
        let l = chosen.1;
        let r = chosen.2;
        let replacement = &chosen.4;

        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        source_lines.splice(replace_start..replace_end, replacement.iter().cloned());

        if source_lines.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        drift = s as i128 - (hunk.old_start as i128 + net_applied);
        net_applied += hunk.new_count as i128 - hunk.old_count as i128;
    }

    let mut output = String::new();
    for line in source_lines {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
