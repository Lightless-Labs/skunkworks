#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    let base = word.text.len();
    word.text.push_str(value);
    if active_glob {
        for (offset, c) in value.char_indices() {
            if matches!(c, '*' | '?' | '[') {
                word.glob.push(base + offset);
            }
        }
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_literal(&mut word, next, false);
                        }
                        i = after_next;
                    } else {
                        append_literal(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' && input[next_i..].starts_with('(') {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    active = true;
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let escaped = input[after..].chars().next().unwrap();
                                    j = after + escaped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let Some(close_pos) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close_pos];
                        let name_end = contents
                            .char_indices()
                            .find_map(|(idx, ch)| {
                                if is_name_continue(ch) {
                                    None
                                } else {
                                    Some(idx)
                                }
                            })
                            .unwrap_or(contents.len());
                        let name = &contents[..name_end];
                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        let rest = &contents[name_end..];
                        let default = if rest.is_empty() {
                            None
                        } else if let Some(default_text) = rest.strip_prefix(":-") {
                            Some(default_text)
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        };
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = env.iter().find(|(key, _)| *key == name).map(|(_, v)| *v);
                        let selected_default =
                            default.filter(|_| value.is_none_or(|v| v.is_empty()));
                        if let Some(default_text) = selected_default {
                            append_expansion(
                                &mut word,
                                &decode_default(default_text),
                                false,
                            );
                        } else if let Some(value) = value {
                            append_expansion(&mut word, value, false);
                        }
                        i = close_pos + 1;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            if let Some((_, value)) = env.iter().find(|(key, _)| *key == name) {
                                append_expansion(&mut word, value, false);
                            }
                            i = end;
                        } else {
                            append_literal(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
                let _ = open;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        append_literal(&mut word, next, false);
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' && input[next_i..].starts_with('(') {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('{') {
                        active = true;
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let escaped = input[after..].chars().next().unwrap();
                                    j = after + escaped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let Some(close_pos) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close_pos];
                        let name_end = contents
                            .char_indices()
                            .find_map(|(idx, ch)| {
                                if is_name_continue(ch) {
                                    None
                                } else {
                                    Some(idx)
                                }
                            })
                            .unwrap_or(contents.len());
                        let name = &contents[..name_end];
                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        let rest = &contents[name_end..];
                        let default = if rest.is_empty() {
                            None
                        } else if let Some(default_text) = rest.strip_prefix(":-") {
                            Some(default_text)
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        };
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = env.iter().find(|(key, _)| *key == name).map(|(_, v)| *v);
                        let selected_default =
                            default.filter(|_| value.is_none_or(|v| v.is_empty()));
                        if let Some(default_text) = selected_default {
                            append_expansion(
                                &mut word,
                                &decode_default(default_text),
                                true,
                            );
                        } else if let Some(value) = value {
                            append_expansion(&mut word, value, true);
                        }
                        i = close_pos + 1;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            active = true;
                            let name = &input[next_i..end];
                            if let Some((_, value)) = env.iter().find(|(key, _)| *key == name) {
                                append_expansion(&mut word, value, true);
                            }
                            i = end;
                        } else {
                            active = true;
                            append_literal(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        active = true;
                        append_literal(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    active = true;
                    append_literal(&mut word, c, matches!(c, '*' | '?' | '['));
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
