#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append(word: &mut Word, value: &str, active_globs: bool) {
    for c in value.chars() {
        if active_globs && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < chars.len() {
        let (pos, c) = chars[i];

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                i += 1;
                continue;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i += 1;
                    continue;
                }
                if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.text.push(next);
                        }
                        i += 2;
                    } else {
                        word.text.push('\\');
                        i += 1;
                    }
                    continue;
                }
                if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                }
                if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len() && chars[i + 1].1 == '{' {
                        let (end, value) = parse_braced(input, &chars, i, env)?;
                        append(&mut word, &value, false);
                        active = true;
                        i = end;
                        continue;
                    }
                    if i + 1 < chars.len() && is_name_start(chars[i + 1].1) {
                        let (end, name) = parse_name(&chars, i + 1);
                        append(&mut word, lookup(&name, env).unwrap_or(""), false);
                        active = true;
                        i = end;
                        continue;
                    }
                }
                word.text.push(c);
                active = true;
                i += 1;
                continue;
            }
            Quote::None => {}
        }

        if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
            if active {
                words.push(word);
                word = Word {
                    text: String::new(),
                    glob: Vec::new(),
                };
                active = false;
            }
            i += 1;
            continue;
        }

        if c == '\'' {
            quote = Quote::Single(pos);
            active = true;
            i += 1;
            continue;
        }
        if c == '"' {
            quote = Quote::Double(pos);
            active = true;
            i += 1;
            continue;
        }
        if c == '\\' {
            if i + 1 >= chars.len() {
                return Err(SplitError::TrailingEscape { pos });
            }
            let next = chars[i + 1].1;
            active = true;
            if next != '\n' {
                word.text.push(next);
            } else {
                active = false_if_continuation(active, &word);
            }
            i += 2;
            continue;
        }
        if c == '`' {
            return Err(SplitError::CommandSubstitution { pos });
        }
        if c == '$' {
            if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                return Err(SplitError::CommandSubstitution { pos });
            }
            if i + 1 < chars.len() && chars[i + 1].1 == '{' {
                let (end, value) = parse_braced(input, &chars, i, env)?;
                append(&mut word, &value, true);
                active = true;
                i = end;
                continue;
            }
            if i + 1 < chars.len() && is_name_start(chars[i + 1].1) {
                let (end, name) = parse_name(&chars, i + 1);
                append(&mut word, lookup(&name, env).unwrap_or(""), true);
                active = true;
                i = end;
                continue;
            }
        }

        if matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
        active = true;
        i += 1;
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn active_false_if_continuation(active: bool, word: &Word) -> bool {
    active || !word.text.is_empty() || !word.glob.is_empty()
}

fn parse_name(chars: &[(usize, char)], start: usize) -> (usize, String) {
    let mut i = start;
    let mut name = String::new();
    while i < chars.len() && is_name_continue(chars[i].1) {
        name.push(chars[i].1);
        i += 1;
    }
    (i, name)
}

fn parse_braced(
    _input: &str,
    chars: &[(usize, char)],
    dollar_index: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let dollar_pos = chars[dollar_index].0;
    let mut close = None;
    let mut i = dollar_index + 2;

    while i < chars.len() {
        if chars[i].1 == '\\' {
            i += 2;
        } else if chars[i].1 == '}' {
            close = Some(i);
            break;
        } else {
            i += 1;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let content: String = chars[dollar_index + 2..close]
        .iter()
        .map(|(_, c)| *c)
        .collect();

    let content_chars: Vec<char> = content.chars().collect();
    if content_chars.is_empty() || !is_name_start(content_chars[0]) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let mut name_end = 1;
    while name_end < content_chars.len() && is_name_continue(content_chars[name_end]) {
        name_end += 1;
    }
    let name: String = content_chars[..name_end].iter().collect();

    let result = if name_end == content_chars.len() {
        lookup(&name, env).unwrap_or("").to_string()
    } else if name_end + 1 < content_chars.len()
        && content_chars[name_end] == ':'
        && content_chars[name_end + 1] == '-'
    {
        let value = lookup(&name, env).unwrap_or("");
        if !value.is_empty() {
            value.to_string()
        } else {
            decode_default(&content_chars[name_end + 2..].iter().collect::<String>())
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((close + 1, result))
}
