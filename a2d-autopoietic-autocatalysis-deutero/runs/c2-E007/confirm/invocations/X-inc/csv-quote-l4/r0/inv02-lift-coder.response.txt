#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record = 0usize;
    let mut field_start = true;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: pos });
        }

        if !in_quotes && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: pos });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote as char);
                    pos += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    pos += 1;
                }
            } else {
                let ch = input[pos..].chars().next().unwrap();
                field.push(ch);
                pos += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                fields.push(std::mem::take(&mut field));
                field_start = true;
                after_quote = false;
                pos += 1;
            } else if b == b'\n' || b == b'\r' {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record += 1;
                field_start = true;
                after_quote = false;
                pos += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: pos });
            }
            continue;
        }

        if b == delimiter {
            fields.push(std::mem::take(&mut field));
            field_start = true;
            pos += 1;
        } else if b == b'\n' || b == b'\r' {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            record += 1;
            field_start = true;
            pos += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_start {
                in_quotes = true;
                quote_open = pos;
                field_start = false;
                pos += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: pos });
            }
        } else {
            let ch = input[pos..].chars().next().unwrap();
            field.push(ch);
            field_start = false;
            pos += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if !fields.is_empty() || !field.is_empty() || !field_start || after_quote {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn build_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut records = records.into_iter();
    let header = if config.headers {
        records.next()
    } else {
        None
    };

    if let (Some(header), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen: Vec<(&str, usize)> = Vec::new();
        for (second, name) in header.iter().enumerate() {
            if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first: *first,
                    second,
                });
            }
            seen.push((name, second));
        }
    }

    let mut rows = Vec::new();
    for (index, row) in records.enumerate() {
        if let Some(header) = &header {
            let record = index + 1;
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
        rows.push(row);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start])
        && b[start + 1..].iter().all(|x| x.is_ascii_digit())
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    let dot = match b.iter().position(|&x| x == b'.') {
        Some(i) => i,
        None => return false,
    };
    if dot < start || b.len() != dot + 3 {
        return false;
    }
    let whole = &b[start..dot];
    if whole.is_empty()
        || !whole.iter().all(|x| x.is_ascii_digit())
        || (whole.len() > 1 && whole[0] == b'0')
        || (start == 1 && whole == b"0")
    {
        return false;
    }
    b[dot + 1..].iter().all(|x| x.is_ascii_digit())
}

fn eval_cell(
    raw: Option<&String>,
    column: Option<usize>,
    spec: &ColumnSpec,
) -> Result<Value, CellErrorKind> {
    let raw = raw.ok_or(CellErrorKind::MissingCell)?;
    if spec.nullable && raw.is_empty() {
        return Ok(Value::Null);
    }
    if raw.is_empty() && spec.kind != ColumnType::Text {
        return Err(CellErrorKind::Empty);
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(raw.clone())),
        ColumnType::Integer => {
            if !integer_syntax(raw) {
                return Err(CellErrorKind::InvalidInteger);
            }
            raw.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(raw) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = raw.starts_with('-');
            let unsigned = if negative { &raw[1..] } else { raw.as_str() };
            let mut parts = unsigned.split('.');
            let whole = parts.next().unwrap();
            let frac = parts.next().unwrap();
            let magnitude = whole.parse::<i128>().unwrap() * 100
                + frac.parse::<i128>().unwrap();
            let signed = if negative { -magnitude } else { magnitude };
            i64::try_from(signed)
                .map(Value::Decimal)
                .map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match raw.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut out = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };
    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (cell, error_column, forced_error) = match resolved {
                None => (None, None, Some(CellErrorKind::UnknownColumn)),
                Some(i) => (row.get(i), Some(i), None),
            };
            let result = if let Some(kind) = forced_error {
                Err(kind)
            } else {
                eval_cell(cell, error_column, spec)
            };
            match result {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    out.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: error_column,
                        kind,
                    });
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => break,
                        ErrorPolicy::Stop => {
                            out.rows.push(output);
                            break 'rows;
                        }
                    }
                }
            }
        }
        out.rows.push(output);
    }
    out
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn first_duplicate(header: &[String]) -> Option<(String, usize, usize)> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
            return Some((name.clone(), *first, second));
        }
        seen.push((name, second));
    }
    None
}

fn quote_field(field: &str, delimiter: u8, quote: u8, first: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let starts_boundary = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t');
    let ends_boundary = bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let bom = first && field.starts_with('\u{feff}');
    let must_quote = field.is_empty() && sole
        || starts_boundary
        || ends_boundary
        || bom
        || bytes.iter().any(|b| *b == delimiter || *b == quote || *b == b'\r' || *b == b'\n');

    if !must_quote {
        return field.to_string();
    }
    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut originals: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        originals.push(header);
    }
    originals.extend(table.rows.iter());

    let data_record_offset = usize::from(table.header.is_some());
    for (record, row) in originals.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in originals.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: if table.header.is_some() { record } else { record - data_record_offset },
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let (Some(header), DuplicatePolicy::Reject) = (&table.header, table.duplicates) {
            if let Some((name, first, second)) = first_duplicate(header) {
                return Err(WriteError::DuplicateHeader { name, first, second });
            }
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained = Vec::new();
                for (i, name) in header.iter().enumerate() {
                    let retain = match table.duplicates {
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            !header[..i].iter().any(|n| n == name)
                        }
                        DuplicatePolicy::KeepLast => {
                            !header[i + 1..].iter().any(|n| n == name)
                        }
                    };
                    if retain {
                        retained.push((name.clone(), i));
                    }
                }
                retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                records.push(retained.iter().map(|(name, _)| name.clone()).collect());
                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|(_, i)| row[*i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        let cmp = x.as_bytes().cmp(y.as_bytes());
                        if cmp != std::cmp::Ordering::Equal {
                            return cmp;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        let cmp = x.as_bytes().cmp(y.as_bytes());
                        if cmp != std::cmp::Ordering::Equal {
                            return cmp;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                records = rows;
            }
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    for (ri, row) in records.iter().enumerate() {
        if ri > 0 {
            out.push_str(ending);
        }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 {
                out.push(config.delimiter as char);
            }
            out.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                ri == 0 && ci == 0,
                row.len() == 1,
            ));
        }
    }
    if canonical || config.final_terminator {
        out.push_str(ending);
    }
    Ok(out)
}
