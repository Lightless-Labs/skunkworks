#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart { Delimiter, Quote }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy { Reject, KeepFirst, KeepLast }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8, pub quote: u8,
    pub headers: bool, pub duplicates: DuplicatePolicy,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>, pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}
impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => header.iter().position(|s| s == name),
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) { return Err(ConfigPart::Delimiter); }
    if !(0x21..=0x7e).contains(&quote) { return Err(ConfigPart::Quote); }
    if delimiter == quote { return Err(ConfigPart::Quote); }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if i == bytes.len() { return Ok(Vec::new()); }
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0 field start, 1 unquoted, 2 quoted, 3 after quote
    let mut opening = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 { return Err(CsvError::Nul { record: record_index, byte: i }); }

        if state != 2 && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record: record_index, byte: i });
            }
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            state = 0;
            i += 2;
            record_index += 1;
            continue;
        }

        if state == 2 {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    state = 3;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if state == 3 {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                state = 0;
                i += 1;
            } else if b == b'\n' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                state = 0;
                i += 1;
                record_index += 1;
            } else {
                return Err(CsvError::TrailingByte { record: record_index, byte: i });
            }
            continue;
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            state = 0;
            i += 1;
        } else if b == b'\n' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            state = 0;
            i += 1;
            record_index += 1;
        } else if b == quote {
            if state == 0 {
                opening = i;
                state = 2;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            state = 1;
            i += ch.len_utf8();
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: opening });
    }
    if state == 3 || state == 0 || state == 1 || !row.is_empty() {
        row.push(field);
        records.push(row);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() { (None, Vec::new()) }
        else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else { (None, records) };

    if let Some(h) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen = std::collections::HashMap::<&str, usize>::new();
            for (i, name) in h.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader { name: name.clone(), first, second: i });
                }
                seen.insert(name, i);
            }
        }
        for (i, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth { record: i + 1, expected: h.len(), actual: row.len() });
            }
        }
    }
    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}
impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), result: None })
    }
    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() { return Err(CsvError::ReaderFinished); }
        self.input.push_str(chunk);
        Ok(())
    }
    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if self.result.is_none() {
            let parsed = parse_with(&self.input, self.config.delimiter, self.config.quote)
                .and_then(|r| build_table(r, self.config));
            self.result = Some(parsed);
        }
        self.result.as_ref().unwrap().clone()
    }
    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}
pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef { Index(usize), Name(String) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType { Text, Integer, Decimal, Boolean }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef, pub kind: ColumnType, pub nullable: bool,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value { Null, Text(String), Integer(i64), Decimal(i64), Boolean(bool) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn, MissingCell, Empty, InvalidInteger, IntegerOverflow,
    InvalidDecimal, DecimalOverflow, InvalidBoolean,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize, pub spec: usize, pub column: Option<usize>,
    pub kind: CellErrorKind,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy { Collect, FirstPerRow, Stop }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>, pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' { return start == 0 && b.len() == 1; }
    if !(b'1'..=b'9').contains(&b[start]) { return false; }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}
fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    if b.len() < start + 4 { return false; }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(u8::is_ascii_digit) { return false; }
    let int = &b[start..dot];
    if int.is_empty() { return false; }
    if int[0] == b'0' { int.len() == 1 } else {
        (b'1'..=b'9').contains(&int[0]) && int[1..].iter().all(u8::is_ascii_digit)
    }
}
fn evaluate(value: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if value.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(value.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(value) { return Err(CellErrorKind::InvalidInteger); }
            value.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(value) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = value.starts_with('-');
            let body = if neg { &value[1..] } else { value };
            let (whole, frac) = body.split_once('.').unwrap();
            let magnitude = whole.parse::<i128>().unwrap() * 100 + frac.parse::<i128>().unwrap();
            let signed = if neg { -magnitude } else { magnitude };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match value {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        }
    }
}
pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut out = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (ri, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;
        for (si, spec) in columns.iter().enumerate() {
            if row_failed {
                output.push(None);
                continue;
            }
            let col = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let (result, kind) = match col {
                None => (None, Some(CellErrorKind::UnknownColumn)),
                Some(c) => match row.get(c) {
                    None => (None, Some(CellErrorKind::MissingCell)),
                    Some(v) => match evaluate(v, spec) {
                        Ok(x) => (Some(x), None),
                        Err(e) => (None, Some(e)),
                    }
                }
            };
            if let Some(kind) = kind {
                let err_col = if matches!(kind, CellErrorKind::UnknownColumn) { None } else { col };
                out.errors.push(CellError { row: ri, spec: si, column: err_col, kind });
                output.push(None);
                if policy != ErrorPolicy::Collect { row_failed = true; }
                if policy == ErrorPolicy::Stop {
                    out.rows.push(output);
                    return out;
                }
            } else {
                output.push(result);
            }
        }
        out.rows.push(output);
        if policy == ErrorPolicy::FirstPerRow { row_failed = false; }
    }
    out
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8, pub quote: u8, pub line_ending: LineEnding,
    pub final_terminator: bool, pub mode: WriteMode,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut all: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(h) = &table.header { all.push((0, h)); }
    for (i, row) in table.rows.iter().enumerate() {
        all.push((if table.header.is_some() { i + 1 } else { i }, row));
    }
    for (record, row) in &all {
        for (column, cell) in row.iter().enumerate() {
            if let Some(byte) = cell.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record: *record, column, byte });
            }
        }
    }
    if let Some(h) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(WriteError::Width { record: i + 1, expected: h.len(), actual: row.len() });
            }
        }
    }
    for (record, row) in &all {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record: *record }); }
    }

    let mut records: Vec<Vec<String>> = all.iter().map(|(_, r)| (*r).clone()).collect();
    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                let mut seen = std::collections::HashMap::<&str, usize>::new();
                for (i, name) in header.iter().enumerate() {
                    if let Some(&first) = seen.get(name.as_str()) {
                        return Err(WriteError::DuplicateHeader { name: name.clone(), first, second: i });
                    }
                    seen.insert(name, i);
                }
            }
            let mut keep = Vec::<(String, usize)>::new();
            for (i, name) in header.iter().enumerate() {
                let found = keep.iter().position(|(n, _)| n == name);
                match (table.duplicates, found) {
                    (DuplicatePolicy::KeepLast, Some(k)) => keep[k].1 = i,
                    (_, Some(_)) => {}
                    (_, None) => keep.push((name.clone(), i)),
                }
            }
            keep.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
            let new_header: Vec<String> = keep.iter().map(|x| x.0.clone()).collect();
            let mut projected: Vec<Vec<String>> = table.rows.iter().map(|r| keep.iter().map(|x| r[x.1].clone()).collect()).collect();
            projected.sort_by(|a, b| compare_rows(a, b));
            records.clear();
            records.push(new_header);
            records.extend(projected);
        } else {
            records = table.rows.clone();
            records.sort_by(|a, b| compare_rows(a, b));
        }
    }

    if records.is_empty() { return Ok(String::new()); }
    let mut output = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let ending: &[u8] = if canonical || config.line_ending == LineEnding::Lf { b"\n" } else { b"\r\n" };
    for (ri, row) in records.iter().enumerate() {
        if ri > 0 { output.push_str(std::str::from_utf8(ending).unwrap()); }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { output.push(config.delimiter as char); }
            let bytes = field.as_bytes();
            let quote_it = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
                || (ri == 0 && ci == 0 && field.starts_with('\u{feff}'))
                || (row.len() == 1 && field.is_empty());
            if quote_it {
                output.push(config.quote as char);
                for ch in field.chars() {
                    if ch as u32 == config.quote as u32 { output.push(config.quote as char); }
                    output.push(ch);
                }
                output.push(config.quote as char);
            } else {
                output.push_str(field);
            }
        }
    }
    if canonical || config.final_terminator {
        output.push_str(std::str::from_utf8(ending).unwrap());
    }
    Ok(output)
}
fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let ord = x.as_bytes().cmp(y.as_bytes());
        if ord != std::cmp::Ordering::Equal { return ord; }
    }
    a.len().cmp(&b.len())
}
