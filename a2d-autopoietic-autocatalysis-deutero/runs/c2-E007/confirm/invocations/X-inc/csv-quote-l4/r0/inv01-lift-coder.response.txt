#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut record = 0usize;
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;
    let mut touched = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if state != 2 && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        match state {
            0 => {
                if b == quote {
                    quote_start = i;
                    state = 2;
                    touched = true;
                    i += 1;
                } else if b == delimiter {
                    row.push(String::new());
                    touched = true;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    record += 1;
                    touched = false;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    field.push(b as char);
                    state = 1;
                    touched = true;
                    i += 1;
                }
            }
            1 => {
                if b == quote {
                    return Err(CsvError::UnexpectedQuote { record, byte: i });
                } else if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = 0;
                    touched = true;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record += 1;
                    state = 0;
                    touched = false;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            2 => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = 3;
                        i += 1;
                    }
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            3 => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = 0;
                    touched = true;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record += 1;
                    state = 0;
                    touched = false;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte { record, byte: i });
                }
            }
            _ => unreachable!(),
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if touched || !row.is_empty() || !field.is_empty() || state != 0 {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

pub struct TypedResult {

    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if dot + 3 != b.len() || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if dot == start {
        return false;
    }
    if b[start] == b'0' {
        if dot != start + 1 {
            return false;
        }
        return !(start == 1 && b[dot + 1..] == *b"00");
    }
    (b'1'..=b'9').contains(&b[start])
        && b[start + 1..dot].iter().all(u8::is_ascii_digit)
}

fn parse_integer_value(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal_value(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let dot = body.len() - 3;
    let whole = body[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let frac = body[dot + 1..]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let magnitude = whole * 100 + frac;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn convert_cell(
    value: &str,
    kind: ColumnType,
    nullable: bool,
) -> Result<Value, CellErrorKind> {
    if value.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match kind {
        ColumnType::Text => Ok(Value::Text(value.to_string())),
        ColumnType::Integer => parse_integer_value(value).map(Value::Integer),
        ColumnType::Decimal => parse_decimal_value(value).map(Value::Decimal),
        ColumnType::Boolean => match value {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                output.push(None);
                continue;
            }
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, cell) = match resolved {
                None => (None, None),
                Some(i) => (Some(i), row.get(i).map(String::as_str)),
            };
            let converted = match (column, cell) {
                (None, _) => Err(CellErrorKind::UnknownColumn),
                (Some(_), None) => Err(CellErrorKind::MissingCell),
                (Some(_), Some(value)) => convert_cell(value, spec.kind, spec.nullable),
            };
            match converted {
                Ok(value) => output.push(Some(value)),
                Err(kind) => {
                    output.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow | ErrorPolicy::Stop => row_failed = true,
                    }
                }
            }
        }
        result.rows.push(output);
        if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
            break;
        }
    }

    result
}

fn duplicate_header(header: &[String]) -> Option<(String, usize, usize)> {
    for second in 0..header.len() {
        for first in 0..second {
            if header[first] == header[second] {
                return Some((header[second].clone(), first, second));
            }
        }
    }
    None
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;
    let mut records = parse_dialect(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some((name, first, second)) = duplicate_header(h) {
                return Err(CsvError::DuplicateHeader {
                    name,
                    first,
                    second,
                });
            }
        }
        for (index, row) in records.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if self.cached.is_none() {
            self.cached = Some(parse_table(&self.input, self.config));
        }
        self.cached.as_ref().unwrap().clone()
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push((0, header));
    }
    for (i, row) in table.rows.iter().enumerate() {
        original_records.push((i + usize::from(table.header.is_some()), row));
    }

    for (record, row) in &original_records {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: *record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in &original_records {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record: *record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some((name, first, second)) = duplicate_header(header) {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();

    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: Vec<usize> = Vec::new();
                for i in 0..header.len() {
                    let earlier = header[..i].iter().any(|x| x == &header[i]);
                    if table.duplicates == DuplicatePolicy::KeepLast
                        && header[i + 1..].iter().any(|x| x == &header[i])
                    {
                        continue;
                    }
                    if table.duplicates != DuplicatePolicy::KeepLast && earlier {
                        continue;
                    }
                    retained.push(i);
                }
                retained.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));

                output_records.push(retained.iter().map(|&i| header[i].clone()).collect());
                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|&i| row[i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        let ord = x.as_bytes().cmp(y.as_bytes());
                        if ord != std::cmp::Ordering::Equal {
                            return ord;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                output_records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        let ord = x.as_bytes().cmp(y.as_bytes());
                        if ord != std::cmp::Ordering::Equal {
                            return ord;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                output_records = rows;
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let (delimiter, quote) = (config.delimiter, config.quote);
    let mut encoded_records = Vec::with_capacity(output_records.len());
    for (record_index, row) in output_records.iter().enumerate() {
        let mut encoded_row = Vec::with_capacity(row.len());
        for (column_index, field) in row.iter().enumerate() {
            let bytes = field.as_bytes();
            let has_bom_prefix = bytes.starts_with(&[0xef, 0xbb, 0xbf]);
            let sole_empty = row.len() == 1 && field.is_empty();
            let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
            let must_quote = bytes.contains(&delimiter)
                || bytes.contains(&quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || edge_space
                || (record_index == 0 && column_index == 0 && has_bom_prefix)
                || sole_empty;

            if must_quote {
                let mut encoded = String::new();
                encoded.push(quote as char);
                for ch in field.chars() {
                    if ch as u32 == quote as u32 {
                        encoded.push(quote as char);
                        encoded.push(quote as char);
                    } else {
                        encoded.push(ch);
                    }
                }
                encoded.push(quote as char);
                encoded_row.push(encoded);
            } else {
                encoded_row.push(field.clone());
            }
        }
        encoded_records.push(encoded_row.join(&(delimiter as char).to_string()));
    }

    let line_ending = match config.mode {
        WriteMode::Canonical => "\n",
        WriteMode::Preserve => match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        },
    };
    let mut output = encoded_records.join(line_ending);
    let add_final = match config.mode {
        WriteMode::Canonical => true,
        WriteMode::Preserve => config.final_terminator,
    };
    if add_final {
        output.push_str(line_ending);
    }
    Ok(output)
}
