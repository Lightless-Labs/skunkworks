use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        6 => ParseError::EmptyItem { field, offset },
        _ => ParseError::InvalidCharacter { field, offset },
    }
}

fn parse_number(bytes: &[u8], field: Field, offset: usize, step: bool) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(err(field, offset, 1));
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| err(field, offset, 2))?;
    }
    let (min, max) = bounds(field);
    if step {
        let width = max - min + 1;
        if value == 0 || value > width {
            return Err(err(field, offset, 5));
        }
    } else if value < min || value > max {
        return Err(err(field, offset, 3));
    }
    Ok(value)
}

fn named_value(bytes: &[u8], field: Field) -> Option<u32> {
    if field != Field::Month && field != Field::DayOfWeek {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    let names: &[&[u8]] = if field == Field::Month {
        &[
            b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN", b"JUL", b"AUG", b"SEP", b"OCT",
            b"NOV", b"DEC",
        ]
    } else {
        &[b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT"]
    };
    names
        .iter()
        .position(|n| *n == upper.as_slice())
        .map(|i| if field == Field::Month { i as u32 + 1 } else { i as u32 })
}

fn parse_atom(
    bytes: &[u8],
    field: Field,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(err(field, offset, 0));
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        return parse_number(bytes, field, offset, step);
    }
    if !step {
        if let Some(v) = named_value(bytes, field) {
            return Ok(v);
        }
    }
    Err(err(field, offset, 0))
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<bool>, bool), ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(err(field, i, 7));
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; (max + 1) as usize];
    let mut unrestricted = false;

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];
        if item.is_empty() {
            return Err(err(field, item_start, 6));
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();
        if slash_positions.len() > 1 {
            return Err(err(field, item_start, 0));
        }
        let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
        } else {
            (item, None, 0)
        };

        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        let base_kind = if base == b"*" {
            0
        } else if hyphens.is_empty() && !base.is_empty() {
            1
        } else if hyphens.len() == 1
            && hyphens[0] > 0
            && hyphens[0] + 1 < base.len()
            && base[0] != b'*'
            && base[hyphens[0] + 1] != b'*'
        {
            2
        } else {
            return Err(err(field, item_start, 0));
        };

        if step_bytes.is_some_and(|s| s.is_empty()) {
            return Err(err(field, item_start, 0));
        }

        let (start, end) = match base_kind {
            0 => (min, max),
            1 => {
                let value = parse_atom(base, field, item_start, false)?;
                (value, value)
            }
            _ => {
                let h = hyphens[0];
                let left = parse_atom(&base[..h], field, item_start, false)?;
                let right = parse_atom(&base[h + 1..], field, item_start + h + 1, false)?;
                if left > right {
                    return Err(err(field, item_start + h + 1, 4));
                }
                (left, right)
            }
        };

        let step = if let Some(step_bytes) = step_bytes {
            Some(parse_atom(step_bytes, field, step_offset, true)?)
        } else {
            None
        };

        if field == Field::DayOfMonth && item == b"*" {
            unrestricted = true;
        }

        if field == Field::DayOfWeek && item == b"*" {
            unrestricted = true;
        }

        if let Some(step) = step {
            let mut v = start;
            while v <= end {
                selected[v as usize] = true;
                match v.checked_add(step) {
                    Some(next) => v = next,
                    None => break,
                }
            }
        } else {
            for v in start..=end {
                selected[v as usize] = true;
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok((selected, unrestricted))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.starts_with('@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut parsed: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    let mut dom_unrestricted = false;
    let mut dow_unrestricted = false;

    for (i, (&field, part)) in FIELDS.iter().zip(parts.iter()).enumerate() {
        let (values, unrestricted) = parse_field(part, field)?;
        parsed[i] = values;
        if field == Field::DayOfMonth {
            dom_unrestricted = unrestricted;
        } else if field == Field::DayOfWeek {
            dow_unrestricted = unrestricted;
        }
    }

    Ok(Cron {
        values: parsed,
        day_of_month_unrestricted: dom_unrestricted,
        day_of_week_unrestricted: dow_unrestricted,
    })
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_days(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_days(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += month_days(at.year, month) as u32;
    }
    days += (at.day - 1) as u32;
    ((6 + days) % 7) as u8
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, max) = bounds(field);
        self.values[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(v, &yes)| {
                if yes && (v as u32) >= min && (v as u32) <= max {
                    Some(v as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let values = &self.values;
        if !values[0][at.minute as usize]
            || !values[1][at.hour as usize]
            || !values[3][at.month as usize]
        {
            return false;
        }

        let dom = values[2][at.day as usize];
        let dow = values[4][weekday(at) as usize];
        match (self.day_of_month_unrestricted, self.day_of_week_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_days(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn first_or_last(
        &self,
        base: CivilDateTime,
        inclusive: bool,
        forward: bool,
    ) -> Option<CivilDateTime> {
        let minutes = self.allowed(Field::Minute);
        let hours = self.allowed(Field::Hour);
        if minutes.is_empty() || hours.is_empty() {
            return None;
        }

        let mut date = CivilDateTime {
            hour: 0,
            minute: 0,
            ..base
        };

        loop {
            let date_only = CivilDateTime { hour: 0, minute: 0, ..date };
            if self.matches_date(date_only) {
                if forward {
                    for &hour in &hours {
                        for &minute in &minutes {
                            let candidate = CivilDateTime { hour, minute, ..date };
                            if (candidate > base || (inclusive && candidate == base))
                                && self.matches_valid(candidate)
                            {
                                return Some(candidate);
                            }
                        }
                    }
                } else {
                    for &hour in hours.iter().rev() {
                        for &minute in minutes.iter().rev() {
                            let candidate = CivilDateTime { hour, minute, ..date };
                            if (candidate < base || (inclusive && candidate == base))
                                && self.matches_valid(candidate)
                            {
                                return Some(candidate);
                            }
                        }
                    }
                }
            }

            if forward {
                if date.year == 2399 && date.month == 12 && date.day == 31 {
                    break;
                }
                if date.day < month_days(date.year, date.month) {
                    date.day += 1;
                } else if date.month < 12 {
                    date.month += 1;
                    date.day = 1;
                } else {
                    date.year += 1;
                    date.month = 1;
                    date.day = 1;
                }
            } else {
                if date.year == 2000 && date.month == 1 && date.day == 1 {
                    break;
                }
                if date.day > 1 {
                    date.day -= 1;
                } else if date.month > 1 {
                    date.month -= 1;
                    date.day = month_days(date.year, date.month);
                } else {
                    date.year -= 1;
                    date.month = 12;
                    date.day = 31;
                }
            }
        }
        None
    }

    fn matches_date(&self, at: CivilDateTime) -> bool {
        let values = &self.values;
        if !values[3][at.month as usize] {
            return false;
        }
        let dom = values[2][at.day as usize];
        let dow = values[4][weekday(at) as usize];
        match (self.day_of_month_unrestricted, self.day_of_week_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_days(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let base = CivilDateTime { hour: 0, minute: 0, ..after };
        Ok(self.first_or_last(base, false, true).filter(|candidate| *candidate > after))
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let base = before;
        Ok(self.first_or_last(base, false, false))
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        if start == end {
            return Ok(FireWindow { fires: Vec::new(), truncated: false });
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(next) = self.first_or_last(cursor, fires.is_empty(), true) {
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow { fires, truncated: true });
            }
            fires.push(next);
            cursor = next;
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let mut out = Vec::new();
        for field in FIELDS {
            let vals = self.allowed(field);
            let (min, max) = bounds(field);
            let full = vals.len() == (max - min + 1) as usize;
            let text = if field == Field::DayOfMonth && self.day_of_month_unrestricted {
                "*".to_string()
            } else if field == Field::DayOfWeek && self.day_of_week_unrestricted {
                "*".to_string()
            } else if full {
                if matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                    "*/1".to_string()
                } else {
                    "*".to_string()
                }
            } else {
                vals.iter().map(u8::to_string).collect::<Vec<_>>().join(",")
            };
            out.push(text);
        }
        out.join(" ")
    }
}
