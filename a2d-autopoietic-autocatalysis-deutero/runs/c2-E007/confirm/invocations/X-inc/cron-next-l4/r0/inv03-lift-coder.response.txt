#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    day_unrestricted: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn slot(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn allowed_name(field: Field, s: &str) -> Option<u32> {
    let upper = s.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn numeric(s: &str, field: Field, offset: usize, step: bool) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    let (min, max) = bounds(field);
    if step {
        if n == 0 || n > max - min + 1 {
            return Err(parse_error(field, offset, 5));
        }
    } else if n < min || n > max {
        return Err(parse_error(field, offset, 3));
    }
    Ok(n)
}

fn atom(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if s.as_bytes().iter().all(u8::is_ascii_digit) && !s.is_empty() {
        numeric(s, field, offset, false)
    } else if let Some(n) = allowed_name(field, s) {
        Ok(n)
    } else {
        Err(parse_error(field, offset, 0))
    }
}

fn parse_field(
    text: &str,
    field: Field,
    is_day: Option<usize>,
) -> Result<(Vec<bool>, bool), ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; (max + 1) as usize];
    let unrestricted = is_day.is_some() && text == "*";

    let mut start = 0usize;
    loop {
        let end = text[start..]
            .find(',')
            .map(|p| start + p)
            .unwrap_or(text.len());
        if start == end {
            return Err(ParseError::EmptyItem { field, offset: start });
        }
        let item = &text[start..end];
        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(parse_error(field, start, 0));
        }
        let (base, step_text) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((pos + 1, &item[pos + 1..])))
        } else {
            (item, None)
        };
        if base.is_empty() || step_text.map_or(false, |(_, s)| s.is_empty()) {
            return Err(parse_error(field, start, 0));
        }
        let hyphens = base.bytes().filter(|&b| b == b'-').count();
        if hyphens > 1 || base == "*" && hyphens != 0 {
            return Err(parse_error(field, start, 0));
        }
        let (left, right, left_off, right_off) = if let Some(pos) = base.find('-') {
            let l = &base[..pos];
            let r = &base[pos + 1..];
            if l.is_empty() || r.is_empty() || l == "*" || r == "*" {
                return Err(parse_error(field, start, 0));
            }
            (l, Some(r), start, start + pos + 1)
        } else {
            (base, None, start)
        };
        if left != "*" && atom(left, field, left_off).is_err()
            || right.map_or(false, |r| atom(r, field, right_off).is_err())
        {
            return Err(parse_error(field, start, 0));
        }

        let (lo, hi) = if left == "*" {
            (min, max)
        } else {
            let a = atom(left, field, left_off)?;
            let b = if let Some(r) = right {
                atom(r, field, right_off)?
            } else {
                a
            };
            if let Some(r) = right {
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: right_off,
                    });
                }
            }
            (a, b)
        };

        let step = if let Some((step_off, s)) = step_text {
            numeric(s, field, start + step_off, true)?
        } else {
            1
        };
        let effective_hi = if step_text.is_some() && right.is_none() && left != "*" {
            max
        } else {
            hi
        };
        let stepped = step_text.is_some();
        let mut value = lo;
        loop {
            selected[value as usize] = true;
            if !stepped || value > effective_hi.saturating_sub(step) {
                break;
            }
            value += step;
            if value > effective_hi {
                break;
            }
        }

        if end == text.len() {
            break;
        }
        start = end + 1;
        if start > text.len() {
            break;
        }
    }
    Ok((selected, unrestricted))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => Some("0 * * * *"),
            "@daily" => Some("0 0 * * *"),
            "@weekly" => Some("0 0 * * 0"),
            "@monthly" => Some("0 0 1 * *"),
            "@yearly" => Some("0 0 1 1 *"),
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        None
    };
    let text = expanded.unwrap_or(input);
    let pieces: Vec<&str> = text.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }
    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (_, max) = bounds(FIELDS[i]);
        vec![false; (max + 1) as usize]
    });
    let mut day_unrestricted = [false; 2];
    for i in 0..5 {
        let day_index = match i {
            2 => Some(0),
            4 => Some(1),
            _ => None,
        };
        let (set, unrestricted) = parse_field(pieces[i], FIELDS[i], day_index)?;
        sets[i] = set;
        if let Some(j) = day_index {
            day_unrestricted[j] = unrestricted;
        }
    }
    Ok(Cron {
        sets,
        day_unrestricted,
    })
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_days(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_days(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut w = 6u8;
    let mut y = 2000u16;
    let mut m = 1u8;
    let mut d = 1u8;
    while (y, m, d) < (year, month, day) {
        d += 1;
        if d > month_days(y, m) {
            d = 1;
            m += 1;
            if m > 12 {
                m = 1;
                y += 1;
            }
        }
        w = (w + 1) % 7;
    }
    w
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, max) = bounds(field);
        (min..=max)
            .filter(|&v| self.sets[slot(field)][v as usize])
            .map(|v| v as u8)
            .collect()
    }

    fn date_matches(&self, year: u16, month: u8, day: u8) -> bool {
        let dom = self.sets[2][day as usize];
        let dow = self.sets[4][weekday(year, month, day) as usize];
        match (self.day_unrestricted[0], self.day_unrestricted[1]) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if day == month_days(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.sets[0][at.minute as usize]
            && self.sets[1][at.hour as usize]
            && self.sets[3][at.month as usize]
            && self.date_matches(at.year, at.month, at.day)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut at = after;
        loop {
            if at == CivilDateTime {
                year: 2399,
                month: 12,
                day: 31,
                hour: 23,
                minute: 59,
            } {
                return Ok(None);
            }
            at = increment_minute(at);
            if self.matches_valid(at) {
                return Ok(Some(at));
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut at = before;
        loop {
            if at == CivilDateTime {
                year: 2000,
                month: 1,
                day: 1,
                hour: 0,
                minute: 0,
            } {
                return Ok(None);
            }
            at = decrement_minute(at);
            if self.matches_valid(at) {
                return Ok(Some(at));
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut at = start;
        while at < end {
            if self.matches_valid(at) {
                if fires.len() == cap {
                    return Ok(FireWindow {
                        fires,
                        truncated: true,
                    });
                }
                fires.push(at);
            }
            if at == CivilDateTime {
                year: 2399,
                month: 12,
                day: 31,
                hour: 23,
                minute: 59,
            } {
                break;
            }
            at = increment_minute(at);
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut parts = Vec::new();
        for field in FIELDS {
            let (min, max) = bounds(field);
            let i = slot(field);
            let full = (min..=max).all(|v| self.sets[i][v as usize]);
            let s = if full {
                match field {
                    Field::DayOfMonth if !self.day_unrestricted[0] => "*/1".to_string(),
                    Field::DayOfWeek if !self.day_unrestricted[1] => "*/1".to_string(),
                    _ => "*".to_string(),
                }
            } else {
                (min..=max)
                    .filter(|&v| self.sets[i][v as usize])
                    .map(|v| v.to_string())
                    .collect::<Vec<_>>()
                    .join(",")
            };
            parts.push(s);
        }
        parts.join(" ")
    }
}

fn increment_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
        } else {
            at.hour = 0;
            at.day += 1;
            if at.day > month_days(at.year, at.month) {
                at.day = 1;
                at.month += 1;
                if at.month > 12 {
                    at.month = 1;
                    at.year += 1;
                }
            }
        }
    }
    at
}

fn decrement_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute > 0 {
        at.minute -= 1;
    } else {
        at.minute = 59;
        if at.hour > 0 {
            at.hour -= 1;
        } else {
            at.hour = 23;
            if at.day > 1 {
                at.day -= 1;
            } else {
                if at.month > 1 {
                    at.month -= 1;
                } else {
                    at.month = 12;
                    at.year -= 1;
                }
                at.day = month_days(at.year, at.month);
            }
        }
    }
    at
}
