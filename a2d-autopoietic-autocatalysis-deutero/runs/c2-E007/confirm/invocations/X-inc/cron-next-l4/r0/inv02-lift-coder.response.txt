#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn month_name(s: &str) -> Option<u8> {
    const NAMES: [&str; 12] = [
        "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
    ];
    NAMES
        .iter()
        .position(|name| s.eq_ignore_ascii_case(name))
        .map(|i| (i + 1) as u8)
}

fn weekday_name(s: &str) -> Option<u8> {
    const NAMES: [&str; 7] = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];
    NAMES
        .iter()
        .position(|name| s.eq_ignore_ascii_case(name))
        .map(|i| i as u8)
}

fn atom_value(field: Field, bytes: &[u8], start: usize, end: usize, item_start: usize) -> Result<u8, ParseError> {
    let s = std::str::from_utf8(&bytes[start..end]).unwrap();
    if s.bytes().all(|b| b.is_ascii_digit()) {
        if s.len() > 1 && s.as_bytes()[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset: start });
        }
        let mut value: u32 = 0;
        for b in s.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset: start })?;
        }
        let (min, max) = domain(field);
        if value < min as u32 || value > max as u32 {
            return Err(ParseError::OutOfRange { field, offset: start });
        }
        return Ok(value as u8);
    }

    match field {
        Field::Month => month_name(s).ok_or(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        }),
        Field::DayOfWeek => weekday_name(s).ok_or(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        }),
        _ => Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        }),
    }
}

fn parse_step(field: Field, bytes: &[u8], start: usize, end: usize) -> Result<u8, ParseError> {
    let s = &bytes[start..end];
    if s.is_empty() || !s.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }
    if s.len() > 1 && s[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }
    let mut value: u32 = 0;
    for &b in s {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    let (min, max) = domain(field);
    let width = max as u32 - min as u32 + 1;
    if value == 0 || value > width {
        return Err(ParseError::BadStep { field, offset: start });
    }
    Ok(value as u8)
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let mut selected = vec![false; max as usize + 1];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base_end, step) = if let Some(&slash) = slash_positions.first() {
            let step_start = item_start + slash + 1;
            let step = parse_step(field, bytes, step_start, item_end)?;
            (item_start + slash, Some(step))
        } else {
            (item_end, None)
        };

        if base_end == item_start {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let base_bytes = &bytes[item_start..base_end];
        let is_star = base_bytes == b"*";
        let hyphens: Vec<usize> = base_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if is_star && hyphens.is_empty() {
            let step = step.unwrap_or(1);
            let mut value = min;
            while value <= max {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        } else if !is_star && hyphens.is_empty() {
            let start = atom_value(field, bytes, item_start, base_end, item_start)?;
            let end = if step.is_some() { max } else { start };
            let step = step.unwrap_or(1);
            let mut value = start;
            while value <= end {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        } else if !is_star && hyphens.len() == 1 {
            let hyphen = item_start + hyphens[0];
            if hyphen == item_start || hyphen + 1 == base_end {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            let start = atom_value(field, bytes, item_start, hyphen, item_start)?;
            let end = atom_value(field, bytes, hyphen + 1, base_end, item_start)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: hyphen + 1,
                });
            }
            let step = step.unwrap_or(1);
            let mut value = start;
            while value <= end {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expansion = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => Some("0 * * * *"),
            "@daily" => Some("0 0 * * *"),
            "@weekly" => Some("0 0 * * 0"),
            "@monthly" => Some("0 0 1 * *"),
            "@yearly" => Some("0 0 1 1 *"),
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        None
    };
    let source = expansion.unwrap_or(input);
    let parts: Vec<&str> = source.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, field) in FIELDS.iter().copied().enumerate() {
        selected[i] = parse_field(field, parts[i])?;
    }

    Ok(Cron {
        selected,
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut day = date_ordinal(after.year, after.month, after.day);
        let last = date_ordinal(2399, 12, 31);
        while day <= last {
            let (year, month, date) = ordinal_date(day);
            if self.date_matches(year, month, date) {
                let hours = self.allowed(Field::Hour);
                let minutes = self.allowed(Field::Minute);
                for hour in hours {
                    for &minute in &minutes {
                        let candidate = CivilDateTime {
                            year,
                            month,
                            day: date,
                            hour,
                            minute,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            day += 1;
        }
        Ok(None)
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut day = date_ordinal(before.year, before.month, before.day);
        loop {
            let (year, month, date) = ordinal_date(day);
            if self.date_matches(year, month, date) {
                let hours = self.allowed(Field::Hour);
                let minutes = self.allowed(Field::Minute);
                for &hour in hours.iter().rev() {
                    for &minute in minutes.iter().rev() {
                        let candidate = CivilDateTime {
                            year,
                            month,
                            day: date,
                            hour,
                            minute,
                        };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if day == date_ordinal(2000, 1, 1) {
                break;
            }
            day -= 1;
        }
        Ok(None)
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        if start == end {
            return Ok(FireWindow {
                fires: Vec::new(),
                truncated: false,
            });
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(next) = self.next_fire(cursor).map_err(|_| WindowError::ReversedWindow)? {
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(next);
            cursor = next;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for field in FIELDS {
            let i = field_index(field);
            let (min, max) = domain(field);
            let is_day = matches!(field, Field::DayOfMonth | Field::DayOfWeek);
            let unrestricted = match field {
                Field::DayOfMonth => self.unrestricted_days[0],
                Field::DayOfWeek => self.unrestricted_days[1],
                _ => false,
            };
            if unrestricted {
                fields.push("*".to_string());
                continue;
            }
            let allowed = self.allowed(field);
            let full = allowed.len() == (max as usize - min as usize + 1);
            if full {
                fields.push(if is_day {
                    "*/1".to_string()
                } else {
                    "*".to_string()
                });
            } else {
                fields.push(
                    allowed
                        .iter()
                        .map(u8::to_string)
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        fields.join(" ")
    }

    fn date_matches(&self, year: u16, month: u8, day: u8) -> bool {
        if !self.selected[field_index(Field::Month)][month as usize] {
            return false;
        }

        let dom = self.selected[field_index(Field::DayOfMonth)][day as usize];
        let dow = self.selected[field_index(Field::DayOfWeek)]
            [weekday(year, month, day) as usize];

        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if day == days_in_month(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.selected[field_index(Field::Minute)][at.minute as usize]
            && self.selected[field_index(Field::Hour)][at.hour as usize]
            && self.date_matches(at.year, at.month, at.day)
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    ((date_ordinal(year, month, day) - date_ordinal(2000, 1, 1) + 6) % 7) as u8
}

fn date_ordinal(year: u16, month: u8, day: u8) -> i32 {
    let mut count = 0;
    for y in 2000..year {
        count += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        count += days_in_month(year, m) as i32;
    }
    count + day as i32 - 1
}

fn ordinal_date(mut ordinal: i32) -> (u16, u8, u8) {
    let mut year = 2000u16;
    loop {
        let days = if leap_year(year) { 366 } else { 365 };
        if ordinal < days {
            break;
        }
        ordinal -= days;
        year += 1;
    }
    let mut month = 1u8;
    loop {
        let days = days_in_month(year, month) as i32;
        if ordinal < days {
            break;
        }
        ordinal -= days;
        month += 1;
    }
    (year, month, ordinal as u8 + 1)
}
