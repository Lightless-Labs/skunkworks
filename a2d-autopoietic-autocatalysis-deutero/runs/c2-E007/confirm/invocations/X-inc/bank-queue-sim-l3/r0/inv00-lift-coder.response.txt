#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: u64,
}

struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| b.started.cmp(&a.started))
        .then_with(|| {
            if a.started {
                a.pause_time.cmp(&b.pause_time)
                    .then_with(|| a.customer.id.cmp(&b.customer.id))
            } else {
                a.customer.arrival.cmp(&b.customer.arrival)
                    .then_with(|| b.customer.id.cmp(&a.customer.id))
            }
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiters: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut time = arrivals[0].arrival;

    while next_arrival < arrivals.len() || running.is_some() || !waiters.is_empty() {
        if let Some(active) = running.as_ref() {
            if active.finish == time {
                let done = running.take().unwrap();
                events.push(Event {
                    time,
                    customer: done.customer.id,
                    kind: Kind::Finish,
                });
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiters.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: 0,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                waiters.sort_by(waiter_cmp);
                if waiters
                    .first()
                    .is_some_and(|w| w.customer.priority < active.customer.priority)
                {
                    let by = waiters[0].customer.id;
                    let done = running.take().unwrap();
                    let remaining = done.finish - time;
                    events.push(Event {
                        time,
                        customer: done.customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiters.push(Waiting {
                        customer: done.customer,
                        remaining,
                        started: true,
                        pause_time: time,
                    });
                }
            }
        }

        if running.is_none() {
            loop {
                if waiters.is_empty() {
                    break;
                }
                waiters.sort_by(waiter_cmp);
                let waiting = waiters.remove(0);
                let customer = waiting.customer;
                let remaining = waiting.remaining;
                let cost = if last_dispatched.is_some_and(|id| id != customer.id) {
                    switch_cost
                } else {
                    0
                };
                let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time,
                })?;
                let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time,
                })?;
                last_dispatched = Some(customer.id);
                let kind = if waiting.started {
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    }
                };
                events.push(Event {
                    time,
                    customer: customer.id,
                    kind,
                });
                running = Some(Running {
                    customer: customer.clone(),
                    remaining,
                    work_at,
                    finish,
                });

                if finish == time {
                    running = None;
                    events.push(Event {
                        time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    continue;
                }
                break;
            }
        }

        if next_arrival >= arrivals.len() && running.is_none() && waiters.is_empty() {
            break;
        }

        let mut next_time = arrivals.get(next_arrival).map(|c| c.arrival);
        if let Some(active) = running.as_ref() {
            for candidate in [Some(active.finish), (active.work_at > time).then_some(active.work_at)]
                .into_iter()
                .flatten()
            {
                if candidate > time && next_time.is_none_or(|n| candidate < n) {
                    next_time = Some(candidate);
                }
            }
        }

        match next_time {
            Some(t) => time = t,
            None => break,
        }
    }

    Ok(events)
}
