```rust
use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[0] == *effective.last().unwrap() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0_i128;
    if points.is_empty() {
        return sum;
    }
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn abs_i128_u128(value: i128) -> u128 {
    if value < 0 {
        (-value) as u128
    } else {
        value as u128
    }
}

fn rounded_sqrt(n: u128) -> u128 {
    let mut lo = 0_u128;
    let mut hi = 1_u128;
    while hi <= n / hi {
        hi *= 2;
    }
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let lower_sq = lo * lo;
    let upper = lo + 1;
    let upper_sq = upper * upper;
    if n - lower_sq >= upper_sq - n {
        upper
    } else {
        lo
    }
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let scaled_sq = ((dx * dx + dy * dy) as u128) * 1_000_000;
    rounded_sqrt(scaled_sq) as u64
}

fn edge_tie_key(a: Point, b: Point) -> (Point, Point) {
    if a <= b {
        (a, b)
    } else {
        (b, a)
    }
}

fn measures_for(points: &[Point]) -> Measures {
    let signed_area2 = shoelace(points);
    let orientation = match signed_area2.cmp(&0) {
        Ordering::Less => Orientation::Clockwise,
        Ordering::Equal => Orientation::Degenerate,
        Ordering::Greater => Orientation::CounterClockwise,
    };

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..points.len() {
        let length = edge_length_milli(points[i], points[(i + 1) % points.len()]);
        edge_milli.push(length);
        perimeter_milli += length;

        match longest_edge {
            None => longest_edge = Some(i),
            Some(best) => {
                if length > edge_milli[best]
                    || (length == edge_milli[best]
                        && (edge_tie_key(points[i], points[(i + 1) % points.len()])
                            < edge_tie_key(points[best], points[(best + 1) % points.len()])
                            || (edge_tie_key(
                                points[i],
                                points[(i + 1) % points.len()],
                            ) == edge_tie_key(
                                points[best],
                                points[(best + 1) % points.len()],
                            ) && i < best)))
                {
                    longest_edge = Some(i);
                }
            }
        }
    }

    Measures {
        signed_area2,
        area2: abs_i128_u128(signed_area2),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let effective = prepare(points)?;
    Ok(measures_for(&effective))
}

fn on_segment(a: Point, b: Point, p: Point) -> bool {
    cross(a, b, p) == 0
        && p.x >= a.x.min(b.x)
        && p.x <= a.x.max(b.x)
        && p.y >= a.y.min(b.y)
        && p.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &q) in queries.iter().enumerate() {
        if !in_range(q) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..polygon.len() {
            if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let determinant = cross(a, b, q);
                if (b.y > a.y && determinant > 0) || (b.y < a.y && determinant < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(p, _)| p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    let mut hull: Vec<(Point, usize)> = if unique.len() <= 1 {
        unique
    } else {
        let all_collinear = unique
            .iter()
            .all(|&(p, _)| cross(unique[0].0, unique[unique.len() - 1].0, p) == 0);

        if all_collinear {
            match mode {
                Collinear::Include => unique,
                Collinear::Exclude => vec![unique[0], unique[unique.len() - 1]],
            }
        } else {
            let mut lower = Vec::new();
            for &item in &unique {
                while lower.len() >= 2 {
                    let n = lower.len();
                    let turn = cross(lower[n - 2].0, lower[n - 1].0, item.0);
                    if turn > 0 || (mode == Collinear::Include && turn == 0) {
                        break;
                    }
                    lower.pop();
                }
                lower.push(item);
            }

            let mut upper = Vec::new();
            for &item in unique.iter().rev() {
                while upper.len() >= 2 {
                    let n = upper.len();
                    let turn = cross(upper[n - 2].0, upper[n - 1].0, item.0);
                    if turn > 0 || (mode == Collinear::Include && turn == 0) {
                        break;
                    }
                    upper.pop();
                }
                upper.push(item);
            }

            lower.pop();
            upper.pop();
            lower.extend(upper);
            lower
        }
    };

    if hull.len() > 1 {
        let minimum_pos = hull
            .iter()
            .enumerate()
            .min_by_key(|(_, (point, _))| *point)
            .unwrap()
            .0;
        hull.rotate_left(minimum_pos);
    }

    let vertices: Vec<Point> = hull.iter().map(|&(p, _)| p).collect();
    let source_indices: Vec<usize> = hull.iter().map(|&(_, i)| i).collect();
    let area2 = abs_i128_u128(shoelace(&vertices));
    let perimeter_milli = measures_for(&vertices).perimeter_milli;

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);

    if ab_c == 0 && on_segment(a, b, c) {
        return true;
    }
    if ab_d == 0 && on_segment(a, b, d) {
        return true;
    }
    if cd_a == 0 && on_segment(c, d, a) {
        return true;
    }
    if cd_b == 0 && on_segment(c, d, b) {
        return true;
    }
    (ab_c < 0) != (ab_d < 0) && (cd_a < 0) != (cd_b < 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let polygon = prepare(points)?;
    let n = polygon.len();

    let mut duplicates = Vec::new();
    for i in 0..n {
        for j in i + 1..n {
            if polygon[i] == polygon[j] {
                duplicates.push((i, j));
            }
        }
    }
    if let Some(&(first, second)) = duplicates.iter().min() {
        return Err(GeomError::Duplicate { first, second });
    }

    let signed_area2 = shoelace(&polygon);
    if signed_area2 == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = polygon[(i + n - 1) % n];
        let current = polygon[i];
        let next = polygon[(i + 1) % n];
        if cross(prev, current, next) == 0 {
            let ax = prev.x as i128 - current.x as i128;
            let ay = prev.y as i128 - current.y as i128;
            let bx = next.x as i128 - current.x as i128;
            let by = next.y as i128 - current.y as i128;
            if ax * bx + ay * by > 0 {
                return Err(GeomError::AdjacentOverlap { vertex: i });
            }
        }
    }

    for i in 0..n {
        for j in i + 1..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                polygon[i],
                polygon[(i + 1) % n],
                polygon[j],
                polygon[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let minimum_index = (0..n).min_by_key(|&i| polygon[i]).unwrap();
    let mut order = Vec::with_capacity(n);
    if signed_area2 > 0 {
        for offset in 0..n {
            order.push((minimum_index + offset) % n);
        }
    } else {
        for offset in 0..n {
            order.push((minimum_index + n - offset) % n);
        }
    }

    let vertices: Vec<Point> = order.iter().map(|&i| polygon[i]).collect();
    let source_indices = order;
    let measures = measures_for(&vertices);

    Ok(Validated {
        vertices,
        source_indices,
        measures,
    })
}
```
