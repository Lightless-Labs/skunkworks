#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut arrival_index = 0;
    let mut current_time = 0u64;
    let mut finish: Option<u64> = None;
    let mut previous_finish: Option<u64> = None;
    let mut completed = 0usize;

    while completed < customers.len() {
        let next_arrival = arrivals.get(arrival_index).map(|c| c.arrival);
        let next_time = match (finish, next_arrival) {
            (Some(f), Some(a)) => f.min(a),
            (Some(f), None) => f,
            (None, Some(a)) => a,
            (None, None) => break,
        };
        current_time = next_time;

        if finish == Some(current_time) {
            let customer_id = match events.last() {
                Some(Event {
                    customer,
                    kind: Kind::Start { .. },
                    ..
                }) => *customer,
                _ => {
                    // The active customer's ID is tracked by the latest start.
                    events
                        .iter()
                        .rev()
                        .find_map(|event| {
                            if matches!(event.kind, Kind::Start { .. }) {
                                Some(event.customer)
                            } else {
                                None
                            }
                        })
                        .expect("scheduled finish must have a matching start")
                }
            };
            events.push(Event {
                time: current_time,
                customer: customer_id,
                kind: Kind::Finish,
            });
            finish = None;
            previous_finish = Some(current_time);
            completed += 1;
        }

        while arrival_index < arrivals.len()
            && arrivals[arrival_index].arrival == current_time
        {
            let customer = arrivals[arrival_index];
            events.push(Event {
                time: current_time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            arrival_index += 1;
        }

        if finish.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = previous_finish
                    .map(|last_finish| current_time > last_finish)
                    .unwrap_or(true);

                let service_finish = current_time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    },
                )?;
                let customer_finish = service_finish
                    .checked_add(if wake { wake_cost } else { 0 })
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    })?;

                events.push(Event {
                    time: current_time,
                    customer: customer.id,
                    kind: Kind::Start {
                        finish: customer_finish,
                        wake,
                    },
                });

                if customer_finish == current_time {
                    events.push(Event {
                        time: current_time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(current_time);
                    completed += 1;

                    while arrival_index < arrivals.len()
                        && arrivals[arrival_index].arrival == current_time
                    {
                        let arriving = arrivals[arrival_index];
                        events.push(Event {
                            time: current_time,
                            customer: arriving.id,
                            kind: Kind::Arrive,
                        });
                        waiting.push_back(arriving);
                        arrival_index += 1;
                    }
                } else {
                    finish = Some(customer_finish);
                }
            }
        }
    }

    Ok(events)
}
