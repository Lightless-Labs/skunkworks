fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    number_kind: ErrorKind,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(error(field, offset, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in bytes {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or_else(|| error(field, offset, ErrorKind::NumberOverflow))?;
    }

    match number_kind {
        ErrorKind::OutOfRange => {
            let (min, max) = bounds(field);
            if value < min || value > max {
                return Err(error(field, offset, ErrorKind::OutOfRange));
            }
        }
        ErrorKind::BadStep => {
            let (_, max) = bounds(field);
            if value == 0 || value > max + 1 {
                return Err(error(field, offset, ErrorKind::BadStep));
            }
        }
        _ => {}
    }

    Ok(value)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (_, max) = domain(fields[i]);
        vec![false; max as usize + 1]
    });
    let mut day_star = [false; 2];

    for (index, (&field, text)) in fields.iter().zip(parts.iter()).enumerate() {
        if text.as_bytes().iter().any(|b| {
            !b.is_ascii_digit() && !matches!(*b, b'*' | b'-' | b'/' | b',')
        }) {
            let offset = text
                .as_bytes()
                .iter()
                .position(|b| {
                    !b.is_ascii_digit() && !matches!(*b, b'*' | b'-' | b'/' | b',')
                })
                .unwrap();
            return Err(error(field, offset, ErrorKind::InvalidCharacter));
        }

        if index == 2 {
            day_star[0] = *text == "*";
        } else if index == 4 {
            day_star[1] = *text == "*";
        }

        let (min, max) = domain(field);
        let width = (max - min + 1) as u32;
        let mut item_start = 0usize;

        for item in text.split(',') {
            if item.is_empty() {
                return Err(error(field, item_start, ErrorKind::EmptyItem));
            }
            parse_item(
                item,
                item_start,
                field,
                min,
                max,
                width,
                &mut selected[index],
            )?;
            item_start += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        day_star,
    })
}

fn parse_item(
    item: &str,
    item_offset: usize,
    field: Field,
    min: u8,
    max: u8,
    width: u32,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(error(field, item_offset, ErrorKind::InvalidSyntax));
    }

    let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(slash + 1))
    } else {
        (item, None)
    };

    if base_end.is_empty() {
        return Err(error(field, item_offset, ErrorKind::InvalidSyntax));
    }

    let (base_kind, first_range, second_range) = if base_end == "*" {
        (BaseKind::Star, None, None)
    } else {
        let hyphens: Vec<usize> = base_end
            .as_bytes()
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        match hyphens.as_slice() {
            [] if base_end.as_bytes().iter().all(u8::is_ascii_digit) => {
                (BaseKind::Number, Some((0, base_end.len())), None)
            }
            [hyphen]
                if *hyphen > 0
                    && *hyphen + 1 < base_end.len()
                    && base_end.as_bytes()[..*hyphen].iter().all(u8::is_ascii_digit)
                    && base_end.as_bytes()[hyphen + 1..]
                        .iter()
                        .all(u8::is_ascii_digit) =>
            {
                (
                    BaseKind::Range,
                    Some((0, *hyphen)),
                    Some((*hyphen + 1, base_end.len())),
                )
            }
            _ => return Err(error(field, item_offset, ErrorKind::InvalidSyntax)),
        }
    };

    if let Some(start) = step_start {
        if start >= item.len() || !item.as_bytes()[start..].iter().all(u8::is_ascii_digit) {
            return Err(error(field, item_offset, ErrorKind::InvalidSyntax));
        }
    }

    let mut start_value = min as u32;
    let mut end_value = max as u32;

    if let Some((begin, end)) = first_range {
        start_value = parse_number(
            &base_end[begin..end],
            item_offset + begin,
            field,
            min as u32,
            max as u32,
            ErrorKind::OutOfRange,
        )?;
        end_value = if let Some((begin, end)) = second_range {
            parse_number(
                &base_end[begin..end],
                item_offset + begin,
                field,
                min as u32,
                max as u32,
                ErrorKind::OutOfRange,
            )?
        } else {
            start_value
        };

        if second_range.is_some() && start_value > end_value {
            let (begin, _) = second_range.unwrap();
            return Err(error(
                field,
                item_offset + begin,
                ErrorKind::ReversedRange,
            ));
        }
    }

    let step = if let Some(step_offset) = step_start {
        parse_number(
            &item[step_offset..],
            item_offset + step_offset,
            field,
            1,
            width,
            ErrorKind::BadStep,
        )?
    } else {
        1
    };

    match base_kind {
        BaseKind::Number if step_start.is_none() => {
            selected[start_value as usize] = true;
        }
        BaseKind::Star | BaseKind::Range | BaseKind::Number => {
            let mut value = start_value;
            while value <= end_value {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        }
    }

    Ok(())
}

fn parse_number(
    number: &str,
    offset: usize,
    field: Field,
    lower: u32,
    upper: u32,
    bounds_error: ErrorKind,
) -> Result<u32, ParseError> {
    let bytes = number.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(error(field, offset, ErrorKind::InvalidSyntax));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(error(field, offset, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| error(field, offset, ErrorKind::NumberOverflow))?;
    }

    if value < lower || value > upper {
        return Err(error(field, offset, bounds_error));
    }
    Ok(value)
}

    step: bool,
) -> Result<u32, ParseError> {
    if end == start {
        return Err(ParseError::InvalidSyntax { field, offset: start });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    let (min, max, width) = field_limits(field);
    if step {
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else if value < min || value > max {
        return Err(ParseError::OutOfRange { field, offset: start });
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; (field_limits(field).1 + 1) as usize];
    let mut item_start = 0usize;
    let mut day_star = false;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_end == item_start {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
            (slash, Some(slash + 1))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        enum Base {
            Star,
            Single(usize, usize),
            Range(usize, usize, usize, usize),
        }

        let parsed_base = if star {
            Base::Star
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            Base::Single(0, base.len())
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen == 0
                || hyphen + 1 >= base.len()
                || !base[..hyphen].iter().all(u8::is_ascii_digit)
                || !base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Base::Range(0, hyphen, hyphen + 1, base.len())
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        let step = if let Some(step_offset) = step_start {
            if step_offset >= item.len()
                || !item[step_offset..].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Some(parse_number(
                item,
                step_offset,
                item.len(),
                field,
                true,
            )?)
        } else {
            None
        };

        let (mut start, mut end) = match parsed_base {
            Base::Star => {
                if field == Field::DayOfMonth && item_start == 0 && item == b"*" {
                    day_star = true;
                }
                field_limits(field).pipe(|(min, max, _)| (min, max))
            }
            Base::Single(a, b) => {
                let value = parse_number(base, a, b, field, false)?;
                let max = field_limits(field).1;
                (value, if step.is_some() { max } else { value })
            }
            Base::Range(a, b, c, d) => {
                let first = parse_number(base, a, b, field, false)?;
                let second = parse_number(base, c, d, field, false)?;
                if first > second {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + c,
                    });
                }
                (first, second)
            }
        };

        if step.is_none() && matches!(parsed_base, Base::Star) {
            let (min, max, _) = field_limits(field);
            start = min;
            end = max;
        }

        let stride = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok((
        selected
            .into_iter()
            .enumerate()
            .filter_map(|(value, yes)| yes.then_some(value as u8))
            .collect(),
        day_star,
    ))
}

trait Pipe: Sized {
    fn pipe<R>(self, f: impl FnOnce(Self) -> R) -> R {
        f(self)
    }
}
impl<T> Pipe for T {}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed = Vec::with_capacity(5);
    let mut day_stars = [false; 2];

    for (index, (&text, &field)) in parts.iter().zip(fields.iter()).enumerate() {
        let (values, is_star) = parse_field(text, field)?;
        if field == Field::DayOfMonth {
            day_stars[0] = is_star;
        } else if field == Field::DayOfWeek {
            day_stars[1] = is_star;
        }
        parsed.push(values);
        let _ = index;
    }

    let values: [Vec<u8>; 5] = parsed
        .try_into()
        .expect("exactly five cron fields were parsed");

    Ok(Cron { values, day_stars })
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_domain(field);
    let width = maximum - minimum + 1;
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&byte| byte == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &byte)| (byte == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(invalid_syntax(field, item_start));
        }

        let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
            (slash, Some(item_start + slash + 1))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let (base_kind, start_value, end_value, end_offset) = if base == b"*" {
            (0u8, minimum, maximum, item_start)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &byte)| (byte == b'-').then_some(i))
                .collect();

            if hyphens.len() > 1 || base.is_empty() {
                return Err(invalid_syntax(field, item_start));
            }

            if let Some(&hyphen) = hyphens.first() {
                if hyphen == 0 || hyphen + 1 == base.len() {
                    return Err(invalid_syntax(field, item_start));
                }
                let start = parse_number(base, 0, hyphen, field)?;
                let end = parse_number(base, hyphen + 1, base.len(), field)?;

                if start < minimum || start > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start,
                    });
                }
                if end < minimum || end > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
                (2u8, start, end, item_start + hyphen + 1)
            } else {
                let start = parse_number(base, 0, base.len(), field)?;
                if start < minimum || start > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start,
                    });
                }
                (1u8, start, start, item_start)
            }
        };

        let step = if let Some(step_start) = step_start {
            let local_start = step_start - item_start;
            if local_start >= item.len() {
                return Err(invalid_syntax(field, item_start));
            }
            let value = parse_number(item, local_start, item.len(), field)
                .map_err(|error| match error {
                    ParseError::LeadingZero { .. } => ParseError::LeadingZero {
                        field,
                        offset: step_start,
                    },
                    ParseError::NumberOverflow { .. } => ParseError::NumberOverflow {
                        field,
                        offset: step_start,
                    },
                    other => other,
                })?;
            if value == 0 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_start,
                });
            }
            Some(value)
        } else {
            None
        };

        let (start, end) = match (base_kind, step) {
            (0, Some(_)) => (minimum, maximum),
            (1, Some(_)) => (start_value, maximum),
            (_, _) => (start_value, end_value),
        };
        let stride = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .into_iter()
        .enumerate()
        .filter_map(|(value, is_selected)| is_selected.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
