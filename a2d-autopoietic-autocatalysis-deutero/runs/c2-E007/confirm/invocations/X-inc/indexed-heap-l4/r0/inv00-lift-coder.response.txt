```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    capacity: Option<usize>,
    next_age: u128,
}

fn rank_cmp(a: &Entry, b: &Entry) -> std::cmp::Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

fn validate_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();
    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }
    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }
    Ok(input[start..end].to_ascii_lowercase())
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::with_capacity_inner(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_capacity_inner(Some(capacity))
    }

    fn with_capacity_inner(capacity: Option<usize>) -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            capacity,
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn next_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut order: Vec<usize> = (0..self.entries.len()).collect();
            order.sort_by_key(|&i| self.entries[i].age);
            for (age, index) in order.into_iter().enumerate() {
                self.entries[index].age = age as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    fn reindex_from(&mut self, start: usize) {
        for i in start..self.entries.len() {
            self.indices.insert(self.entries[i].item.key.clone(), i);
        }
    }

    fn remove_at(&mut self, index: usize) -> Entry {
        let removed = self.entries.remove(index);
        self.indices.remove(&removed.item.key);
        self.reindex_from(index);
        removed
    }

    fn worst_index(&self) -> Option<usize> {
        self.entries
            .iter()
            .enumerate()
            .max_by(|(_, a), (_, b)| rank_cmp(a, b))
            .map(|(i, _)| i)
    }

    fn ordered_indices(&self) -> Vec<usize> {
        let mut indices: Vec<usize> = (0..self.entries.len()).collect();
        indices.sort_by(|&a, &b| rank_cmp(&self.entries[a], &self.entries[b]));
        indices
    }

    fn trim_to_capacity(&mut self) -> Vec<Item> {
        let Some(capacity) = self.capacity else {
            return Vec::new();
        };
        let mut evicted = Vec::new();
        while self.entries.len() > capacity {
            let index = self.worst_index().expect("nonempty while over capacity");
            evicted.push(self.remove_at(index).item);
        }
        evicted
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<Admission, Error> {
        let key = validate_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Item {
            key: key.clone(),
            priority,
            value: value.to_owned(),
        };
        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed));
        }

        if let Some(capacity) = self.capacity {
            if self.entries.len() >= capacity {
                let worst = self.worst_index().expect("positive full capacity is nonempty");
                let proposed_entry = Entry {
                    item: proposed.clone(),
                    age: self.next_age,
                };
                if rank_cmp(&proposed_entry, &self.entries[worst])
                    != std::cmp::Ordering::Less
                {
                    return Ok(Admission::Discarded(proposed));
                }
                let evicted = self.remove_at(worst).item;
                let age = self.next_age();
                let index = self.entries.len();
                self.entries.push(Entry {
                    item: proposed,
                    age,
                });
                self.indices.insert(key, index);
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.next_age();
        let index = self.entries.len();
        self.entries.push(Entry {
            item: proposed,
            age,
        });
        self.indices.insert(key, index);
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.ordered_indices()
            .first()
            .map(|&i| self.entries[i].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let index = self.ordered_indices().first().copied()?;
        Some(self.remove_at(index).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = validate_key(key)?;
        Ok(self
            .indices
            .get(&key)
            .map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(std::mem::replace(
            &mut self.entries[index].item.value,
            value.to_owned(),
        ))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.remove_at(index).item)
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = self.trim_to_capacity();
        evicted.reverse();
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.ordered_indices()
            .into_iter()
            .map(|i| self.entries[i].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order = other.ordered_indices();
        if policy == CollisionPolicy::Reject {
            for &i in &source_order {
                let key = &other.entries[i].item.key;
                if self.indices.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let source_items: Vec<Item> = source_order
            .into_iter()
            .map(|i| other.entries[i].item.clone())
            .collect();
        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for item in source_items {
            if let Some(&index) = self.indices.get(&item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = item.priority;
                        self.entries[index].item.value = item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.next_age();
                let index = self.entries.len();
                self.entries.push(Entry {
                    item: item.clone(),
                    age,
                });
                self.indices.insert(item.key, index);
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.indices.clear();
        report.evicted = self.trim_to_capacity();
        report.evicted.reverse();
        Ok(report)
    }
}
```
