#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_group(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        output.push_str(std::str::from_utf8(&encode_group(chunk[0], chunk[1], chunk[2])).unwrap());
    }

    match chunks.remainder() {
        [] => {}
        [a] => {
            let group = encode_group(*a, 0, 0);
            output.push(group[0] as char);
            output.push(group[1] as char);
            output.push_str("==");
        }
        [a, b] => {
            let group = encode_group(*a, *b, 0);
            output.push(group[0] as char);
            output.push(group[1] as char);
            output.push(group[2] as char);
            output.push('=');
        }
        _ => unreachable!(),
    }

    output
}

#[derive(Clone, Copy)]
struct Symbol {
    value: u8,
    offset: usize,
}

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn decode_full(group: [Symbol; 4]) -> [u8; 3] {
    [
        (group[0].value << 2) | (group[1].value >> 4),
        (group[1].value << 4) | (group[2].value >> 2),
        (group[2].value << 6) | group[3].value,
    ]
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::with_capacity((self.tail_len + input.len()) / 3 * 4);
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                let group = encode_group(self.tail[0], self.tail[1], input[index - 1]);
                output.push_str(std::str::from_utf8(&group).unwrap());
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            let group = encode_group(input[index], input[index + 1], input[index + 2]);
            output.push_str(std::str::from_utf8(&group).unwrap());
            index += 3;
        }

        let remainder = &input[index..];
        self.tail_len = remainder.len();
        self.tail[..self.tail_len].copy_from_slice(remainder);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            0 => {}
            1 => {
                let group = encode_group(self.tail[0], 0, 0);
                output.push(group[0] as char);
                output.push(group[1] as char);
                output.push_str("==");
            }
            2 => {
                let group = encode_group(self.tail[0], self.tail[1], 0);
                output.push(group[0] as char);
                output.push(group[1] as char);
                output.push(group[2] as char);
                output.push('=');
            }
            _ => unreachable!(),
        }
        self.tail_len = 0;
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    group: Vec<Symbol>,
    next_offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            next_offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset });
            }

            if self.pending_second_padding.is_some() && byte != b'=' {
                return self.fail(Error::BadPadding { offset });
            }

            let position = self.group.len();
            if byte == b'=' {
                if position <= 1 {
                    return self.fail(Error::BadPadding { offset });
                }
                if position == 2 {
                    self.pending_second_padding = Some(offset);
                    self.group.push(Symbol { value: 0, offset });
                    continue;
                }
                if position == 3 {
                    self.group.push(Symbol { value: 0, offset });
                    let decoded = [
                        (self.group[0].value << 2) | (self.group[1].value >> 4),
                    ];
                    output.extend_from_slice(&decoded);
                    self.group.clear();
                    self.pending_second_padding = None;
                    self.terminal_padding = true;
                    continue;
                }
            }

            let Some(value) = digit(byte) else {
                return self.fail(Error::InvalidByte { offset, byte });
            };

            if position >= 4 {
                return self.fail(Error::BadPadding { offset });
            }

            self.group.push(Symbol { value, offset });

            if self.group.len() == 4 {
                output.extend_from_slice(&decode_full([
                    self.group[0],
                    self.group[1],
                    self.group[2],
                    self.group[3],
                ]));
                self.group.clear();
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }

        if !self.group.is_empty() {
            return self.fail(Error::Truncated {
                offset: self.group[0].offset,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
