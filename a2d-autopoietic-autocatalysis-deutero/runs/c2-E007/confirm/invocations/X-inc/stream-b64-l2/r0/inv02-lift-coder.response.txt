#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8]) -> String {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    let mut out = String::with_capacity(4);
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        output.push_str(&encode_group(chunk));
    }

    let tail = chunks.remainder();
    if !tail.is_empty() {
        output.push_str(&encode_group(tail));
    }

    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::with_capacity(2),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if !self.tail.is_empty() {
            while self.tail.len() < 3 && index < input.len() {
                self.tail.push(input[index]);
                index += 1;
            }
            if self.tail.len() == 3 {
                output.push_str(&encode_group(&self.tail));
                self.tail.clear();
            }
        }

        while index + 3 <= input.len() {
            output.push_str(&encode_group(&input[index..index + 3]));
            index += 3;
        }

        self.tail.extend_from_slice(&input[index..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        if self.tail.is_empty() {
            Ok(String::new())
        } else {
            let output = encode_group(&self.tail);
            self.tail.clear();
            Ok(output)
        }
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    group: Vec<(u8, usize)>,
    offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    fn process_byte(&mut self, byte: u8, at: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return self.fail(Error::BadPadding { offset: at });
        }

        if self.pending_second_padding.is_some() && byte != b'=' {
            return self.fail(Error::BadPadding { offset: at });
        }

        let position = self.group.len();

        if byte == b'=' {
            match position {
                0 | 1 => return self.fail(Error::BadPadding { offset: at }),
                2 => {
                    self.pending_second_padding = Some(at);
                    self.group.push((byte, at));
                    return Ok(());
                }
                3 => {
                    if self.pending_second_padding.is_none() {
                        self.terminal_padding = true;
                    }
                    self.group.push((byte, at));
                }
                _ => unreachable!(),
            }
        } else {
            let Some(value) = digit(byte) else {
                return self.fail(Error::InvalidByte { offset: at, byte });
            };
            self.group.push((value, at));
        }

        if self.group.len() == 4 {
            let a = self.group[0].0;
            let b = self.group[1].0;

            if self.group[2].0 == b'=' {
                output.push((a << 2) | (b >> 4));
                self.terminal_padding = true;
            } else if self.group[3].0 == b'=' {
                let c = self.group[2].0;
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                self.terminal_padding = true;
            } else {
                let c = self.group[2].0;
                let d = self.group[3].0;
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
            }

            self.group.clear();
            self.pending_second_padding = None;
        }

        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let at = self.offset;
            self.offset += 1;
            if let Err(error) = self.process_byte(byte, at, &mut output) {
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            return self.fail(Error::Truncated {
                offset: self.group[0].1,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
