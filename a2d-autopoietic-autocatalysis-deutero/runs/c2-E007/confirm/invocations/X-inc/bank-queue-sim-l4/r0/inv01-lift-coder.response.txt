#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let other_half = denominator - r;
    Some(if r > other_half || (r == other_half && q % 2 == 1) {
        q + 1
    } else {
        q
    })
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    accepted: bool,
    started: bool,
    queued: bool,
    entry: u64,
    waited: u64,
    last_pause: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    protected: bool,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn better_waiter(a: usize, b: usize, states: &[CustomerState]) -> bool {
    let x = &states[a];
    let y = &states[b];
    let xp = x.customer.priority;
    let yp = y.customer.priority;
    if xp != yp {
        return xp < yp;
    }
    match (x.last_pause, y.last_pause) {
        (Some(tx), Some(ty)) if tx != ty => tx < ty,
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (Some(_), Some(_)) => x.customer.id < y.customer.id,
        (None, None) if x.customer.arrival != y.customer.arrival => {
            x.customer.arrival < y.customer.arrival
        }
        (None, None) => x.customer.id > y.customer.id,
    }
}

fn best_waiter(states: &[CustomerState]) -> Option<usize> {
    let mut best = None;
    for (i, s) in states.iter().enumerate() {
        if s.queued && best.map_or(true, |b| better_waiter(i, b, states)) {
            best = Some(i);
        }
    }
    best
}

fn consume_wait(state: &mut CustomerState, now: u64, wait_ticks: &mut u128) {
    let segment = now - state.entry;
    state.waited = state.waited.saturating_add(segment);
    *wait_ticks += segment as u128;
    if let Some(deadline) = state.deadline {
        state.deadline = Some(deadline - segment);
    }
}

fn expire_waiters(
    states: &mut [CustomerState],
    now: u64,
    events: &mut Vec<Event>,
    stats: &mut Stats,
    wait_ticks: &mut u128,
) {
    let mut expired: Vec<usize> = states
        .iter()
        .enumerate()
        .filter_map(|(i, s)| {
            (s.queued && s.deadline.map_or(false, |d| d <= now)).then_some(i)
        })
        .collect();
    expired.sort_by_key(|&i| (states[i].deadline.unwrap(), states[i].customer.id));
    for i in expired {
        let s = &mut states[i];
        if !s.queued {
            continue;
        }
        consume_wait(s, now, wait_ticks);
        s.queued = false;
        emit(events, now, s.customer.id, Kind::Renege { waited: s.waited });
        stats.reneged += 1;
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if customers[..i].iter().any(|x| x.id == c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index: i, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index: i, id: c.id });
            }
        }
    }

    let mut states: Vec<CustomerState> = customers
        .iter()
        .cloned()
        .map(|c| CustomerState {
            deadline: c.patience.and_then(|p| c.arrival.checked_add(p)),
            remaining: c.service,
            customer: c,
            accepted: false,
            started: false,
            queued: false,
            entry: 0,
            waited: 0,
            last_pause: None,
        })
        .collect();
    let mut cohort: Vec<usize> = (0..states.len()).collect();
    cohort.sort_by_key(|&i| (
        states[i].customer.arrival,
        states[i].customer.priority,
        std::cmp::Reverse(states[i].customer.id),
    ));

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0, balked: 0, reneged: 0, completed: 0, started: 0,
        preemptions: 0, elapsed: 0, busy_ticks: 0, wait_ticks: 0,
        turnaround_ticks: 0, mean_wait_milli: 0, mean_turnaround_milli: 0,
        utilization_bps: 0, completion_order: Vec::new(),
    };
    let mut wait_ticks = 0u128;
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut now = cohort.first().map_or(0, |&i| states[i].customer.arrival);
    let mut first_dispatch = true;

    loop {
        // A: finish takes precedence over every other event at this tick.
        if active.as_ref().map_or(false, |a| a.finish == now) {
            let a = active.take().unwrap();
            let s = &states[a.index];
            emit(&mut events, now, s.customer.id, Kind::Finish);
            stats.completed += 1;
            stats.turnaround_ticks += (now - s.customer.arrival) as u128;
            stats.completion_order.push(s.customer.id);
        }

        // B: expire those already waiting.
        expire_waiters(&mut states, now, &mut events, &mut stats, &mut wait_ticks);

        // C: process the complete arrival cohort.
        while next_arrival < cohort.len()
            && states[cohort[next_arrival]].customer.arrival == now
        {
            let i = cohort[next_arrival];
            next_arrival += 1;
            let id = states[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);
            let waiting = states.iter().filter(|s| s.queued).count();
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting >= limit {
                emit(&mut events, now, id, Kind::Balk);
                stats.balked += 1;
            } else {
                let s = &mut states[i];
                s.accepted = true;
                s.queued = true;
                s.entry = now;
                stats.accepted += 1;
            }
        }

        // D: preempt only after protection ends, before re-enqueueing the current customer.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&states) {
                    if states[w].customer.priority < states[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let remaining = a.finish - now;
                        let by = states[w].customer.id;
                        let id = states[a.index].customer.id;
                        emit(&mut events, now, id, Kind::Preempt { remaining, by });
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.work_at.min(now)) as u128;
                        let s = &mut states[a.index];
                        s.remaining = remaining;
                        s.queued = true;
                        s.entry = now;
                        s.last_pause = Some(now);
                        if let Some(budget) = s.customer.patience {
                            let consumed = s.waited;
                            let spent_before = s.customer.patience.unwrap().saturating_sub(budget);
                            let _ = (consumed, spent_before);
                            // Remaining budget is tracked by moving the current deadline.
                            let rem = s.deadline.map(|d| d.saturating_sub(now)).unwrap_or(budget);
                            s.deadline = Some(now.checked_add(rem).ok_or(Error::TimeOverflow {
                                customer: id, time: now, stage: OverflowStage::Patience
                            })?);
                        }
                    }
                }
            }
        }

        // E: dispatch repeatedly to handle inline completions.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&states) else { break };
            {
                let s = &mut states[i];
                s.queued = false;
                consume_wait(s, now, &mut wait_ticks);
            }
            let id = states[i].customer.id;
            let remaining = states[i].remaining;
            let cost = if first_dispatch || last_dispatched == Some(id) { 0 } else { config.switch_cost };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::WorkAt
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::Finish
            })?;
            stats.busy_ticks += (finish - now) as u128;
            let was_started = states[i].started;
            if was_started {
                emit(&mut events, now, id, Kind::Resume { remaining, work_at, finish });
            } else {
                emit(&mut events, now, id, Kind::Start { remaining, work_at, finish });
                states[i].started = true;
                stats.started += 1;
            }
            first_dispatch = false;
            last_dispatched = Some(id);
            active = Some(Active { index: i, work_at, finish, protected: work_at > now });
            if finish == now {
                let _a = active.take().unwrap();
                emit(&mut events, now, id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[i].customer.arrival) as u128;
                stats.completion_order.push(id);
                continue;
            }
            break;
        }

        // F: newly queued zero-budget and otherwise due customers expire.
        expire_waiters(&mut states, now, &mut events, &mut stats, &mut wait_ticks);

        if stats.completed + stats.reneged + stats.balked == customers.len() {
            break;
        }

        let mut next: Option<u64> = None;
        if next_arrival < cohort.len() {
            next = Some(states[cohort[next_arrival]].customer.arrival);
        }
        if let Some(a) = &active {
            for t in [Some(a.work_at).filter(|&t| t > now), Some(a.finish).filter(|&t| t > now)]
                .into_iter().flatten()
            {
                next = Some(next.map_or(t, |n| n.min(t)));
            }
        }
        for s in &states {
            if s.queued {
                if let Some(d) = s.deadline.filter(|&d| d > now) {
                    next = Some(next.map_or(d, |n| n.min(d)));
                }
            }
        }
        let Some(t) = next else { break };
        now = t;
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(1000u128 * wait_ticks, stats.accepted as u128).unwrap()
    };
    stats.wait_ticks = wait_ticks;
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(1000u128 * stats.turnaround_ticks, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(10000u128 * stats.busy_ticks, stats.elapsed as u128).unwrap() as u32
    };
    Ok(Report { events, stats })
}
