use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;

    Some(match remainder.cmp(&complement) {
        Ordering::Less => quotient,
        Ordering::Greater => quotient + 1,
        Ordering::Equal => quotient + (quotient & 1),
    })
}

#[derive(Clone)]
struct State {
    customer: Customer,
    remaining: u64,
    accepted: bool,
    terminal: bool,
    ever_started: bool,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiting_cmp(a: &State, b: &State) -> Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(pa), Some(pb)) => pa.cmp(&pb).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => b
                .customer
                .arrival
                .cmp(&a.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter(states: &[State]) -> Option<usize> {
    states
        .iter()
        .enumerate()
        .filter(|(_, s)| s.accepted && !s.terminal && s.entry != u64::MAX - 1)
        .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
        .map(|(i, _)| i)
}

fn is_waiting(s: &State) -> bool {
    s.accepted && !s.terminal && s.entry != u64::MAX - 1
}

fn expire_waiters(
    states: &mut [State],
    events: &mut Vec<Event>,
    now: u64,
    stats: &mut Stats,
) {
    let mut expired: Vec<usize> = states
        .iter()
        .enumerate()
        .filter(|(_, s)| is_waiting(s) && s.deadline.is_some_and(|d| d <= now))
        .map(|(i, _)| i)
        .collect();

    expired.sort_by(|&a, &b| {
        states[a]
            .deadline
            .cmp(&states[b].deadline)
            .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
    });

    for i in expired {
        let segment = now.saturating_sub(states[i].entry);
        states[i].waited = states[i].waited.saturating_add(segment);
        stats.wait_ticks += segment as u128;
        states[i].entry = u64::MAX - 1;
        states[i].terminal = true;
        stats.reneged += 1;
        emit(
            events,
            now,
            states[i].customer.id,
            Kind::Renege {
                waited: states[i].waited,
            },
        );
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|prior| prior.id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: c.id,
                });
            }
        }
    }

    let mut states: Vec<State> = customers
        .iter()
        .map(|c| State {
            customer: c.clone(),
            remaining: c.service,
            accepted: false,
            terminal: false,
            ever_started: false,
            entry: u64::MAX - 1,
            deadline: c.patience.and_then(|p| c.arrival.checked_add(p)),
            waited: 0,
            pause_time: None,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        states[a]
            .customer
            .arrival
            .cmp(&states[b].customer.arrival)
            .then_with(|| states[a].customer.priority.cmp(&states[b].customer.priority))
            .then_with(|| states[b].customer.id.cmp(&states[a].customer.id))
    });

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut events = Vec::new();
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut protection_ended = true;
    let mut first_dispatch = true;

    while stats.completed + stats.reneged + stats.balked < customers.len() {
        let next_time = {
            let mut t = u64::MAX;
            if next_arrival < arrivals.len() {
                t = t.min(states[arrivals[next_arrival]].customer.arrival);
            }
            if let Some(a) = &active {
                t = t.min(a.finish);
                if !protection_ended {
                    t = t.min(a.work_at);
                }
            }
            for s in &states {
                if is_waiting(s) {
                    if let Some(d) = s.deadline {
                        t = t.min(d);
                    }
                }
            }
            t
        };
        if next_time == u64::MAX {
            break;
        }
        now = next_time;

        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            stats.busy_ticks += (now - a.dispatch) as u128;
            let id = states[a.index].customer.id;
            states[a.index].terminal = true;
            stats.completed += 1;
            stats.turnaround_ticks += (now - states[a.index].customer.arrival) as u128;
            stats.completion_order.push(id);
            emit(&mut events, now, id, Kind::Finish);
            protection_ended = true;
        }

        expire_waiters(&mut states, &mut events, now, &mut stats);

        while next_arrival < arrivals.len()
            && states[arrivals[next_arrival]].customer.arrival == now
        {
            let i = arrivals[next_arrival];
            next_arrival += 1;
            let id = states[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);

            let waiting_count = states.iter().filter(|s| is_waiting(s)).count();
            let idle_bonus = usize::from(active.is_none());
            if waiting_count >= config.waiting_capacity + idle_bonus {
                stats.balked += 1;
                states[i].terminal = true;
                emit(&mut events, now, id, Kind::Balk);
                continue;
            }

            stats.accepted += 1;
            states[i].accepted = true;
            states[i].entry = now;
            states[i].deadline = states[i]
                .customer
                .patience
                .and_then(|p| states[i].customer.arrival.checked_add(p));
        }

        if let Some(a) = active.as_ref() {
            if protection_ended || now >= a.work_at {
                protection_ended = true;
                if let Some(w) = best_waiter(&states) {
                    if states[w].customer.priority < states[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let id = states[a.index].customer.id;
                        let remaining = a.finish - now;
                        emit(
                            &mut events,
                            now,
                            id,
                            Kind::Preempt {
                                remaining,
                                by: states[w].customer.id,
                            },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch) as u128;

                        states[a.index].remaining = remaining;
                        states[a.index].entry = now;
                        states[a.index].pause_time = Some(now);
                        if states[a.index].customer.patience.is_some() {
                            let budget = states[a.index].deadline.unwrap().saturating_sub(a.dispatch);
                            states[a.index].deadline = Some(
                                now.checked_add(budget).ok_or(Error::TimeOverflow {
                                    customer: id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                })?,
                            );
                        }
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&states) else {
                break;
            };

            let waited = now - states[i].entry;
            states[i].waited = states[i].waited.saturating_add(waited);
            stats.wait_ticks += waited as u128;
            if let Some(deadline) = states[i].deadline {
                let budget_left = deadline.saturating_sub(states[i].entry);
                states[i].deadline = Some(now.saturating_add(budget_left));
            }

            let id = states[i].customer.id;
            let switch = if first_dispatch {
                0
            } else if last_dispatched == Some(id) {
                0
            } else {
                config.switch_cost
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[i].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let kind = if states[i].ever_started {
                Kind::Resume {
                    remaining: states[i].remaining,
                    work_at,
                    finish,
                }
            } else {
                stats.started += 1;
                Kind::Start {
                    remaining: states[i].remaining,
                    work_at,
                    finish,
                }
            };
            states[i].ever_started = true;
            states[i].entry = u64::MAX - 1;
            states[i].pause_time = None;
            active = Some(Active {
                index: i,
                work_at,
                finish,
                dispatch: now,
            });
            protection_ended = switch == 0;
            last_dispatched = Some(id);
            first_dispatch = false;
            emit(&mut events, now, id, kind);

            if finish == now {
                let a = active.take().unwrap();
                states[i].terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[i].customer.arrival) as u128;
                stats.completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                protection_ended = true;
                continue;
            }
            break;
        }

        expire_waiters(&mut states, &mut events, now, &mut stats);
        stats.elapsed = stats.elapsed.max(events.last().map_or(0, |e| e.time));
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = rounded_ratio(stats.wait_ticks.saturating_mul(1000), stats.accepted as u128).unwrap_or(0);
    stats.mean_turnaround_milli = rounded_ratio(
        stats.turnaround_ticks.saturating_mul(1000),
        stats.completed as u128,
    )
    .unwrap_or(0);
    stats.utilization_bps = rounded_ratio(stats.busy_ticks.saturating_mul(10000), stats.elapsed as u128)
        .unwrap_or(0)
        .min(u32::MAX as u128) as u32;

    Ok(Report { events, stats })
}
