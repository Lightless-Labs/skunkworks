use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|part| *part == "." || *part == "..")
}

fn format_path(parts: &[String], absolute: bool, dir: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if dir && result != "/" && result != "." && !result.ends_with('/') {
        result.push('/');
    }
    result
}

fn reduce(
    initial: &mut Vec<String>,
    raw: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let mut pos = 0;
    for part in raw.split('/') {
        let len = part.len();
        if !part.is_empty() {
            match part {
                "." => {}
                ".." => {
                    if initial.last().is_some_and(|last| last != "..") {
                        initial.pop();
                    } else if absolute {
                        return Err(PathError::Escape {
                            at: offset_base + pos,
                        });
                    } else {
                        initial.push("..".to_owned());
                    }
                }
                _ => initial.push(part.to_owned()),
            }
        }
        pos += len + 1;
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let dir = directory_intent(path);
    let mut parts = Vec::new();
    reduce(&mut parts, path, 0, absolute)?;
    Ok(format_path(&parts, absolute, dir))
}

fn normalize_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(components(&normalized)
        .into_iter()
        .map(str::to_owned)
        .collect())
}

fn validate_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = normalize_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(home) => Some(normalize_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

fn tilde_start<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, Vec<String>, &'a str, usize), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((
                false,
                home.to_vec(),
                &path[consumed..],
                consumed,
            ));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if path.starts_with('/') {
        Ok((true, Vec::new(), path, 0))
    } else {
        Ok((false, Vec::new(), path, 0))
    }
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_raw(path)?;

    let dir = directory_intent(path);
    let (absolute, mut parts, request, consumed) =
        tilde_start(path, home_parts.as_deref())?;

    if !absolute && !path.starts_with('~') {
        parts = cwd_parts;
    }

    reduce(
        &mut parts,
        request,
        consumed,
        true,
    )?;
    Ok(format_path(&parts, true, dir))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut by_path = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();

        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = by_path.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        by_path.insert(
            link.path.clone(),
            (index, link.target.clone()),
        );
    }
    Ok(by_path)
}

#[derive(Clone)]
struct WorkItem {
    raw: String,
    pos: usize,
    offset_base: usize,
    target_link: Option<usize>,
}

fn stack_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn resolve_links_validated(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
    table: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let dir = directory_intent(path);

    let (absolute, mut stack, request, consumed) =
        tilde_start(path, home_parts)?;
    if !absolute && !path.starts_with('~') {
        stack = cwd_parts.to_vec();
    }

    let mut work = vec![WorkItem {
        raw: request.to_owned(),
        pos: 0,
        offset_base: consumed,
        target_link: None,
    }];
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(item) = work.last_mut() {
        if item.pos >= item.raw.len() {
            let done = work.pop().unwrap();
            if let Some(index) = done.target_link {
                if active.last() == Some(&index) {
                    active.pop();
                } else if let Some(position) = active.iter().rposition(|&v| v == index) {
                    active.remove(position);
                }
            }
            continue;
        }

        let bytes = item.raw.as_bytes();
        let start = item.pos;
        while item.pos < bytes.len() && bytes[item.pos] == b'/' {
            item.pos += 1;
        }
        if item.pos >= bytes.len() {
            continue;
        }
        let component_start = item.pos;
        while item.pos < bytes.len() && bytes[item.pos] != b'/' {
            item.pos += 1;
        }
        let component_end = item.pos;
        let part = item.raw[component_start..component_end].to_owned();
        let at = item.offset_base + component_start;
        let _ = start;

        match part.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(part);
                let key = stack_path(&stack);
                if let Some((index, target)) = table.get(&key) {
                    let index = *index;
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    expansions += 1;
                    active.push(index);

                    stack.pop();
                    let target_absolute = target.starts_with('/');
                    if target_absolute {
                        stack.clear();
                    }

                    work.push(WorkItem {
                        raw: target.clone(),
                        pos: 0,
                        offset_base: 0,
                        target_link: Some(index),
                    });
                }
            }
        }
    }

    Ok(format_path(&stack, true, dir))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    let table = validate_links(links)?;
    resolve_links_validated(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        &table,
        max_expansions,
    )
}

fn canonical_parts(path: &str) -> Vec<String> {
    components(path).into_iter().map(str::to_owned).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();

    let mut result = Vec::new();
    result.extend((common..base.len()).map(|_| "..".to_owned()));
    result.extend(target[common..].iter().cloned());
    if result.is_empty() {
        ".".to_owned()
    } else {
        result.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = canonical_parts(&resolved_base);
    let target_parts = canonical_parts(&resolved_target);
    let target_dir = resolved_target.ends_with('/');

    let table = validate_links(links)?;
    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let probe_parts = canonical_parts(&probe);
        if probe_parts.len() > target_parts.len()
            || probe_parts
                .iter()
                .zip(target_parts.iter())
                .any(|(a, b)| a != b)
        {
            continue;
        }

        let mut candidate = route(&base_parts, &canonical_parts(&link.path));
        let suffix = &target_parts[probe_parts.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_dir && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if canonical_parts(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    let _ = table;
    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
