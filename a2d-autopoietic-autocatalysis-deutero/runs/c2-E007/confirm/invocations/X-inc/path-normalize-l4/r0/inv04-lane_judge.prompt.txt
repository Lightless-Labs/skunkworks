use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|part| *part == "." || *part == "..")
}

fn format_path(parts: &[String], absolute: bool, dir: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if dir && result != "/" && result != "." && !result.ends_with('/') {
        result.push('/');
    }
    result
}

fn reduce(
    initial: &mut Vec<String>,
    raw: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let mut pos = 0;
    for part in raw.split('/') {
        let len = part.len();
        if !part.is_empty() {
            match part {
                "." => {}
                ".." => {
                    if initial.last().is_some_and(|last| last != "..") {
                        initial.pop();
                    } else if absolute {
                        return Err(PathError::Escape {
                            at: offset_base + pos,
                        });
                    } else {
                        initial.push("..".to_owned());
                    }
                }
                _ => initial.push(part.to_owned()),
            }
        }
        pos += len + 1;
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let dir = directory_intent(path);
    let mut parts = Vec::new();
    reduce(&mut parts, path, 0, absolute)?;
    Ok(format_path(&parts, absolute, dir))
}

fn normalize_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(components(&normalized)
        .into_iter()
        .map(str::to_owned)
        .collect())
}

fn validate_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = normalize_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(home) => Some(normalize_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

fn tilde_start<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, Vec<String>, &'a str, usize), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((
                false,
                home.to_vec(),
                &path[consumed..],
                consumed,
            ));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if path.starts_with('/') {
        Ok((true, Vec::new(), path, 0))
    } else {
        Ok((false, Vec::new(), path, 0))
    }
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_raw(path)?;

    let dir = directory_intent(path);
    let (absolute, mut parts, request, consumed) =
        tilde_start(path, home_parts.as_deref())?;

    if !absolute && !path.starts_with('~') {
        parts = cwd_parts;
    }

    reduce(
        &mut parts,
        request,
        consumed,
        true,
    )?;
    Ok(format_path(&parts, true, dir))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut by_path = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();

        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = by_path.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        by_path.insert(
            link.path.clone(),
            (index, link.target.clone()),
        );
    }
    Ok(by_path)
}

#[derive(Clone)]
struct WorkItem {
    raw: String,
    pos: usize,
    offset_base: usize,
    target_link: Option<usize>,
}

fn stack_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn resolve_links_validated(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
    table: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let dir = directory_intent(path);

    let (absolute, mut stack, request, consumed) =
        tilde_start(path, home_parts)?;
    if !absolute && !path.starts_with('~') {
        stack = cwd_parts.to_vec();
    }

    let mut work = vec![WorkItem {
        raw: request.to_owned(),
        pos: 0,
        offset_base: consumed,
        target_link: None,
    }];
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(item) = work.last_mut() {
        if item.pos >= item.raw.len() {
            let done = work.pop().unwrap();
            if let Some(index) = done.target_link {
                if active.last() == Some(&index) {
                    active.pop();
                } else if let Some(position) = active.iter().rposition(|&v| v == index) {
                    active.remove(position);
                }
            }
            continue;
        }

        let bytes = item.raw.as_bytes();
        let start = item.pos;
        while item.pos < bytes.len() && bytes[item.pos] == b'/' {
            item.pos += 1;
        }
        if item.pos >= bytes.len() {
            continue;
        }
        let component_start = item.pos;
        while item.pos < bytes.len() && bytes[item.pos] != b'/' {
            item.pos += 1;
        }
        let component_end = item.pos;
        let part = item.raw[component_start..component_end].to_owned();
        let at = item.offset_base + component_start;
        let _ = start;

        match part.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(part);
                let key = stack_path(&stack);
                if let Some((index, target)) = table.get(&key) {
                    let index = *index;
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    expansions += 1;
                    active.push(index);

                    stack.pop();
                    let target_absolute = target.starts_with('/');
                    if target_absolute {
                        stack.clear();
                    }

                    work.push(WorkItem {
                        raw: target.clone(),
                        pos: 0,
                        offset_base: 0,
                        target_link: Some(index),
                    });
                }
            }
        }
    }

    Ok(format_path(&stack, true, dir))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    let table = validate_links(links)?;
    resolve_links_validated(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        &table,
        max_expansions,
    )
}

fn canonical_parts(path: &str) -> Vec<String> {
    components(path).into_iter().map(str::to_owned).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();

    let mut result = Vec::new();
    result.extend((common..base.len()).map(|_| "..".to_owned()));
    result.extend(target[common..].iter().cloned());
    if result.is_empty() {
        ".".to_owned()
    } else {
        result.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = canonical_parts(&resolved_base);
    let target_parts = canonical_parts(&resolved_target);
    let target_dir = resolved_target.ends_with('/');

    let table = validate_links(links)?;
    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let probe_parts = canonical_parts(&probe);
        if probe_parts.len() > target_parts.len()
            || probe_parts
                .iter()
                .zip(target_parts.iter())
                .any(|(a, b)| a != b)
        {
            continue;
        }

        let mut candidate = route(&base_parts, &canonical_parts(&link.path));
        let suffix = &target_parts[probe_parts.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_dir && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if canonical_parts(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    let _ = table;
    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}

            i += 1;
        }
        result.push((path[start..i].to_owned(), base_offset + start));
    }
    result
}

fn reduce_plain(
    stack: &mut Vec<String>,
    parts: &[(String, usize)],
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: *at });
                } else {
                    stack.push(part.clone());
                }
            }
            _ => stack.push(part.clone()),
        }
    }
    Ok(())
}

fn render_relative(parts: &[String], directory: bool) -> String {
    let mut result = if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };
    if directory && result != "." {
        result.push('/');
    }
    result
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    components(path, 0)
        .last()
        .is_some_and(|(part, _)| part == "." || part == "..")
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let mut stack = Vec::new();
    reduce_plain(&mut stack, &components(path, 0), absolute)?;

    let mut result = if absolute {
        if stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else {
        render_relative(&stack, false)
    };

    let intent = directory_intent(path);
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

fn validate_context(path: &str) -> Result<String, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    normalize(path).map_err(|_| ())
}

fn prepare_request(path: &str) -> Result<(bool, usize, bool), PathError> {
    validate_raw(path)?;
    let tilde = path.starts_with('~');
    let offset = if tilde {
        match path.as_bytes().get(1) {
            None => 1,
            Some(b'/') => 2,
            Some(_) => return Err(PathError::UnsupportedTilde { at: 1 }),
        }
    } else if path.starts_with('/') {
        0
    } else {
        0
    };
    Ok((tilde, offset, directory_intent(path)))
}

fn absolute_parts(path: &str) -> Vec<String> {
    components(path, 0).into_iter().map(|(p, _)| p).collect()
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home = match home {
        Some(h) => Some(validate_context(h).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    let (tilde, offset, intent) = prepare_request(path)?;
    let mut stack = if tilde {
        match home.as_deref() {
            Some(h) => absolute_parts(h),
            None => return Err(PathError::MissingHome),
        }
    } else if path.starts_with('/') {
        Vec::new()
    } else {
        absolute_parts(&cwd)
    };

    let remainder = &path[offset..];
    let parts = components(remainder, offset);
    reduce_plain(&mut stack, &parts, true)?;

    let mut result = if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    };
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut keys = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path).is_ok_and(|normalized| normalized == link.path);
        let target_valid = validate_raw(&link.target).is_ok();
        if !path_valid || !target_valid {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = keys.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        keys.insert(link.path.clone(), index);
    }
    Ok(keys)
}

struct LinkWalker<'a> {
    links: &'a [Link],
    keys: &'a HashMap<String, usize>,
    stack: Vec<String>,
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl LinkWalker<'_> {
    fn current_path(&self) -> String {
        if self.stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", self.stack.join("/"))
        }
    }

    fn process_parts(
        &mut self,
        parts: &[(String, usize)],
        absolute: bool,
    ) -> Result<(), PathError> {
        for (part, at) in parts {
            match part.as_str() {
                "." => {}
                ".." => {
                    if self.stack.pop().is_none() {
                        return Err(PathError::Escape { at: *at });
                    }
                }
                _ => {
                    self.stack.push(part.clone());
                    let key = self.current_path();
                    if let Some(&index) = self.keys.get(&key) {
                        if self.active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if self.expansions >= self.max_expansions {
                            return Err(PathError::Limit { index });
                        }
                        self.expansions += 1;
                        self.active.push(index);

                        self.stack.pop();
                        let target = &self.links[index].target;
                        let target_absolute = target.starts_with('/');
                        if target_absolute {
                            self.stack.clear();
                        }
                        let target_parts = components(target, 0);
                        let outcome = self.process_parts(&target_parts, true);

                        self.active.pop();
                        outcome?;
                    }
                }
            }
        }
        let _ = absolute;
        Ok(())
    }
}

fn resolve_links_internal(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home = match home {
        Some(h) => Some(validate_context(h).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    let keys = validate_links(links)?;
    let (tilde, offset, intent) = prepare_request(path)?;

    let request_absolute = !tilde && path.starts_with('/');
    let mut walker = LinkWalker {
        links,
        keys: &keys,
        stack: Vec::new(),
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    if !request_absolute {
        let starting = if tilde {
            match home.as_deref() {
                Some(h) => h,
                None => return Err(PathError::MissingHome),
            }
        } else {
            &cwd
        };
        let start_parts = components(starting, 0);
        walker.process_parts(&start_parts, true)?;
    }

    let request_parts = components(&path[offset..], offset);
    walker.process_parts(&request_parts, true)?;

    let mut result = walker.current_path();
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_internal(path, cwd, home, links, max_expansions)
}

fn route(base: &[String], target: &[String]) -> Vec<String> {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut result = vec!["..".to_owned(); base.len() - common];
    result.extend_from_slice(&target[common..]);
    result
}

fn join_candidate(parts: &[String]) -> String {
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

fn make_candidate(mut parts: Vec<String>, directory: bool) -> String {
    let mut candidate = join_candidate(&parts);
    if candidate.starts_with('~') {
        candidate.insert_str(0, "./");
    }
    if directory && candidate != "." {
        candidate.push('/');
    }
    parts.clear();
    candidate
}

fn is_component_prefix(prefix: &[String], full: &[String]) -> bool {
    prefix.len() <= full.len() && prefix.iter().zip(full).all(|(a, b)| a == b)
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = absolute_parts(&resolved_base);
    let target_parts = absolute_parts(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![make_candidate(
        route(&base_parts, &target_parts),
        target_directory,
    )];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let alias_parts = absolute_parts(&probe);
        if !is_component_prefix(&alias_parts, &target_parts) {
            continue;
        }

        let mut shortcut = route(&base_parts, &absolute_parts(&link.path));
        let suffix = &target_parts[alias_parts.len()..];
        if !suffix.is_empty() {
            if shortcut.as_slice() == ["."] {
                shortcut.clear();
            }
            shortcut.extend_from_slice(suffix);
        }
        candidates.push(make_candidate(shortcut, target_directory));
    }

    let mut valid = Vec::new();
    for candidate in candidates {
        let Ok(resolved) =
            resolve_links(&candidate, &resolved_base, home, links, max_expansions)
        else {
            continue;
        };
        if absolute_parts(&resolved) == target_parts {
            valid.push(candidate);
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_owned(), start));
    }
    result
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(c, _)| c == "." || c == "..")
}

fn render_absolute(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return "/".to_owned();
    }
    let mut out = format!("/{}", stack.join("/"));
    if intent {
        out.push('/');
    }
    out
}

fn render_normalized(stack: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return render_absolute(stack, intent);
    }
    if stack.is_empty() {
        return ".".to_owned();
    }
    let mut out = stack.join("/");
    if intent {
        out.push('/');
    }
    out
}

fn reduce_plain(
    path: &str,
    absolute: bool,
    initial: Vec<String>,
    offset_adjust: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    for (part, at) in components(path) {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|c| c != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: at + offset_adjust,
                    });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let stack = reduce_plain(path, absolute, Vec::new(), 0)?;
    Ok(render_normalized(&stack, absolute, intent))
}

fn validate_context<'a>(
    cwd: &'a str,
    home: Option<&'a str>,
) -> Result<(String, Option<String>), PathError> {
    let cwd_norm = if cwd.starts_with('/') {
        normalize(cwd).map_err(|_| PathError::InvalidCwd)?
    } else {
        return Err(PathError::InvalidCwd);
    };
    let home_norm = match home {
        Some(h) => {
            if !h.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(normalize(h).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };
    Ok((cwd_norm, home_norm))
}

fn expand_request<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, Vec<(String, usize)>, Option<String>), PathError> {
    validate_raw(path)?;
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home.ok_or(PathError::MissingHome)?.to_owned();
        let remainder_start = if bytes.len() > 1 { 2 } else { 1 };
        let remainder = &path[remainder_start..];
        let parts = components(remainder)
            .into_iter()
            .map(|(p, at)| (p, at + remainder_start))
            .collect();
        Ok((false, parts, Some(home)))
    } else {
        Ok((path.starts_with('/'), components(path), None))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (cwd_norm, home_norm) = validate_context(cwd, home)?;
    let (absolute, parts, tilde_home) = expand_request(path, home_norm.as_deref())?;
    let intent = directory_intent(path);
    let start = if absolute {
        Vec::new()
    } else if let Some(h) = tilde_home {
        components(&h).into_iter().map(|(c, _)| c).collect()
    } else {
        components(&cwd_norm).into_iter().map(|(c, _)| c).collect()
    };
    let mut stack = start;
    for (part, at) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(render_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut indices = HashMap::new();
    for (i, link) in links.iter().enumerate() {
        let path_ok = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path).is_ok_and(|n| n == link.path);
        if !path_ok || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index: i });
        }
        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink { first, second: i });
        }
        indices.insert(link.path.clone(), i);
    }
    Ok(indices)
}

fn stack_key(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn traverse_parts(
    parts: &[(String, usize)],
    stack: &mut Vec<String>,
    links: &[Link],
    index: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
    absolute_parent_escape: bool,
) -> Result<(), PathError> {
    for (part, at) in parts {
        match part.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    if absolute_parent_escape {
                        return Err(PathError::Escape { at: *at });
                    }
                    stack.push("..".to_owned());
                }
                continue;
            }
            _ => {}
        }

        stack.push(part.clone());
        let key = stack_key(stack);
        if let Some(&link_index) = index.get(&key) {
            if active.contains(&link_index) {
                return Err(PathError::Loop { index: link_index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index: link_index });
            }
            *expansions += 1;
            active.push(link_index);

            let target = &links[link_index].target;
            if target.starts_with('/') {
                stack.clear();
            } else {
                stack.pop();
            }

            let target_parts = components(target);
            traverse_parts(
                &target_parts,
                stack,
                links,
                index,
                active,
                expansions,
                max_expansions,
                true,
            )?;
            active.pop();
        }
    }
    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_norm, home_norm) = validate_context(cwd, home)?;
    let index = validate_links(links)?;
    let (absolute, parts, tilde_home) = expand_request(path, home_norm.as_deref())?;
    let intent = directory_intent(path);

    let mut stack = Vec::new();
    let context = if absolute {
        None
    } else if let Some(h) = tilde_home {
        Some(h)
    } else {
        Some(cwd_norm)
    };

    let mut active = Vec::new();
    let mut expansions = 0;
    if let Some(ctx) = context {
        let ctx_parts = components(&ctx);
        traverse_parts(
            &ctx_parts,
            &mut stack,
            links,
            &index,
            &mut active,
            &mut expansions,
            max_expansions,
            true,
        )?;
    }
    traverse_parts(
        &parts,
        &mut stack,
        links,
        &index,
        &mut active,
        &mut expansions,
        max_expansions,
        true,
    )?;
    Ok(render_absolute(&stack, intent))
}

fn absolute_components(path: &str) -> Vec<String> {
    components(path).into_iter().map(|(c, _)| c).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

fn candidate_with_suffix(mut head: String, suffix: &[String]) -> String {
    if suffix.is_empty() {
        return head;
    }
    if head == "." {
        head = suffix.join("/");
    } else {
        head.push('/');
        head.push_str(&suffix.join("/"));
    }
    head
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let b = absolute_components(&resolved_base);
    let t = absolute_components(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&b, &t)];
    for link in links {
        if let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let a = absolute_components(&probe);
            if a.len() <= t.len() && a.iter().zip(&t).all(|(x, y)| x == y) {
                let suffix = &t[a.len()..];
                let key_components = absolute_components(&link.path);
                let c = candidate_with_suffix(route(&b, &key_components), suffix);
                candidates.push(c);
            }
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }
        if let Ok(resolved) =
            resolve_links(&candidate, &resolved_base, home, links, max_expansions)
        {
            if absolute_components(&resolved) == t {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
