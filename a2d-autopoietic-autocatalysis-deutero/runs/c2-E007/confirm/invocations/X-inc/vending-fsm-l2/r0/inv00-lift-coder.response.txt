#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::new();

        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|byte| byte.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }

            if seen.iter().any(|id: &String| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }

        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }

        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match &self.selected {
            Some(id) => id.clone(),
            None => return Err(Error::NoSelection),
        };

        let product = self
            .products
            .iter()
            .find(|product| product.id == selected_id)
            .expect("selected product is always in the catalog");
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change_counts = if amount == 0 {
            [0; 4]
        } else {
            let greedy = greedy_change(amount, &self.tubes);
            if greedy.1 == 0 {
                greedy.0
            } else {
                match optimal_change(amount, &self.tubes) {
                    Some(counts) => counts,
                    None => return Err(Error::NoChange { amount }),
                }
            }
        };

        let mut proposed_tubes = self.tubes;
        for index in 0..4 {
            proposed_tubes[index] -= change_counts[index];
        }
        for &coin in &self.escrow {
            let index = denomination_index(coin).expect("escrow contains accepted coins");
            proposed_tubes[index] += 1;
        }

        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        let mut change = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..change_counts[index] {
                change.push(DENOMINATIONS[index]);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt {
            id: selected_id,
            price,
            change,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(refunded)
    }
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&denomination| denomination == coin)
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> ([u16; 4], u16) {
    let mut remaining = amount;
    let mut counts = [0; 4];

    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index];
        let take = tubes[index].min(remaining / denomination);
        counts[index] = take;
        remaining -= take * denomination;
    }

    (counts, remaining)
}

fn optimal_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    fn search(
        index: isize,
        remaining: u16,
        tubes: &[u16; 4],
        counts: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_coins: &mut u32,
    ) {
        if remaining == 0 {
            let total: u32 = counts.iter().map(|&count| u32::from(count)).sum();
            let is_better = if total < *best_coins {
                true
            } else if total > *best_coins {
                false
            } else {
                match best {
                    None => true,
                    Some(existing) => {
                        for i in (0..4).rev() {
                            if counts[i] != existing[i] {
                                return_if_better(counts[i] > existing[i], best, best_coins, counts, total);
                                return;
                            }
                        }
                        false
                    }
                }
            };

            if is_better {
                *best = Some(*counts);
                *best_coins = total;
            }
            return;
        }

        if index < 0 {
            return;
        }

        let i = index as usize;
        let denomination = DENOMINATIONS[i];
        let max_take = tubes[i].min(remaining / denomination);

        for take in (0..=max_take).rev() {
            let next_total: u32 = counts
                .iter()
                .map(|&count| u32::from(count))
                .sum::<u32>()
                + u32::from(take);

            if next_total > *best_coins {
                continue;
            }

            counts[i] = take;
            search(
                index - 1,
                remaining - take * denomination,
                tubes,
                counts,
                best,
                best_coins,
            );
        }

        counts[i] = 0;
    }

    fn return_if_better(
        better: bool,
        best: &mut Option<[u16; 4]>,
        best_coins: &mut u32,
        counts: &[u16; 4],
        total: u32,
    ) {
        if better {
            *best = Some(*counts);
            *best_coins = total;
        }
    }

    let mut best = None;
    let mut best_coins = u32::MAX;
    let mut counts = [0; 4];
    search(3, amount, tubes, &mut counts, &mut best, &mut best_coins);
    best
}
