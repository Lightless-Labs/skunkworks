use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let Some(_) = DENOMINATIONS.iter().position(|&denomination| denomination == coin)
        else {
            return Err(Error::BadCoin { coin });
        };

        if self.credit + u32_to_u16_saturating(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let product = self
            .products
            .iter()
            .find(|product| &product.id == selected_id)
            .expect("selected product is always in the catalog");

        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let amount = self.credit - product.price;
        let change = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for (index, &denomination) in DENOMINATIONS.iter().enumerate() {
            let used = change
                .iter()
                .filter(|&&coin| coin == denomination)
                .count() as u16;
            proposed_tubes[index] -= used;
        }

        for &coin in &self.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow only contains accepted denominations");
            if proposed_tubes[index] == 100 {
                return Err(Error::TubeLimit { coin });
            }
            proposed_tubes[index] += 1;
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price: product.price,
            change,
        };

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(coins)
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0_u16; 4];
    let mut remaining = amount;
    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index];
        let count = tubes[index].min(remaining / denomination);
        greedy_counts[index] = count;
        remaining -= count * denomination;
    }

    if remaining == 0 {
        return Some(counts_to_descending_coins(greedy_counts));
    }

    let mut best: Option<([u16; 4], u16)> = None;
    search_change(
        3,
        amount,
        tubes,
        [0; 4],
        0,
        &mut best,
    );

    best.map(|(counts, _)| counts_to_descending_coins(counts))
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: [u16; 4],
    total_coins: u16,
    best: &mut Option<([u16; 4], u16)>,
) {
    if index < 0 {
        if remaining == 0 {
            let replace = match best {
                None => true,
                Some((best_counts, best_total)) => {
                    total_coins < *best_total
                        || (total_coins == *best_total
                            && descending_lex_greater(&counts, best_counts))
                }
            };
            if replace {
                *best = Some((counts, total_coins));
            }
        }
        return;
    }

    let index = index as usize;
    let denomination = DENOMINATIONS[index];
    let max_count = tubes[index].min(remaining / denomination);

    for count in 0..=max_count {
        let next_total = total_coins + count;
        if best
            .as_ref()
            .is_some_and(|(_, best_total)| next_total > *best_total)
        {
            continue;
        }

        let mut next_counts = counts;
        next_counts[index] = count;
        search_change(
            index as isize - 1,
            remaining - count * denomination,
            tubes,
            next_counts,
            next_total,
            best,
        );
    }
}

fn descending_lex_greater(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for index in (0..4).rev() {
        if left[index] != right[index] {
            return left[index] > right[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

fn u32_to_u16_saturating(value: u16) -> u16 {
    value
}

#[cfg(test)]
mod tests {
    use super::*;

    fn pub_product(id: &str, price: u16) -> Product {
        Product {
            id: id.into(),
            price,
        }
    }

    fn pub_new(price: u16, tubes: [u16; 4]) -> Machine {
        Machine::new(
            vec![pub_product("tea", price), pub_product("juice", 7)],
            tubes,
        )
        .unwrap()
    }

    #[test]
    fn public_exact_sale() {
        let mut m = pub_new(6, [20, 10, 10, 10]);
        m.select("tea").unwrap();
        m.insert(3).unwrap();
        m.insert(3).unwrap();
        assert_eq!(
            m.vend().unwrap(),
            Receipt {
                id: "tea".into(),
                price: 6,
                change: vec![]
            }
        );
        assert_eq!(m.state().credit, 0);
        assert_eq!(m.state().selected, None);
    }

    #[test]
    fn public_refund_and_selection() {
        let mut m = pub_new(6, [20, 10, 10, 10]);
        m.select("juice").unwrap();
        m.insert(1).unwrap();
        m.insert(4).unwrap();
        assert_eq!(m.refund().unwrap(), vec![4, 1]);
        assert_eq!(m.state().escrow, Vec::<u16>::new());
    }

    #[test]
    fn public_change() {
        let mut m = pub_new(6, [20, 10, 10, 10]);
        m.insert(10).unwrap();
        m.select("tea").unwrap();
        assert_eq!(m.vend().unwrap().change, vec![4]);
        assert_eq!(m.state().tubes, [20, 10, 9, 11]);
    }
}
