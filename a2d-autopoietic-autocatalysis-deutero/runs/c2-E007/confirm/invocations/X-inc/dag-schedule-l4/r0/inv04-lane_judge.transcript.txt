pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency in the required order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resources task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            children[dependency].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest available task ID.
    let mut available = std::collections::BTreeSet::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(id) = available.pop_first() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Bottom levels, computed in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    );
    let mut used = vec![0u64; capacity.len()];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];

    // Worker availability times; initially all workers are free at time 0.
    let mut worker_free = vec![0u64; worker_count];

    // Running entries are (finish time, task ID, worker ID).
    let mut running: Vec<(u64, usize, usize)> = Vec::new();
    let mut current_time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now in ascending task ID.
        let mut ending: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|entry| entry.0 == current_time)
            .collect();
        ending.sort_by_key(|entry| entry.1);

        for (finish, id, worker) in ending {
            running.retain(|entry| entry.1 != id);
            completed[id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);

            for (resource, &need) in tasks[id].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }

            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                current_time,
                id,
                worker,
                resource_snapshot(&used)
            ));

            for &child in &children[id] {
                remaining_predecessors[child] -= 1;
            }
        }

        // Keep admitting one task at a time, reevaluating after each start.
        loop {
            let chosen = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= current_time
                        && remaining_predecessors[id] == 0
                        && tasks[id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(r, &need)| {
                                u64::from(need) <= u64::from(capacity[r]) - used[r]
                            })
                })
                .max_by(|&a, &b| {
                    tail[a]
                        .cmp(&tail[b])
                        .then(tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });

            let Some(id) = chosen else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (id % worker_count + offset) % worker_count)
                .find(|&w| worker_free[w] <= current_time);

            let Some(worker) = worker else {
                break;
            };

            let finish = current_time
                .checked_add(tasks[id].duration)
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Finish,
                })?;
            let turnaround = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Turnaround,
                })?;

            started[id] = true;
            worker_free[worker] = turnaround;
            runs[id] = Some(Run {
                start: current_time,
                finish,
                worker,
            });

            for (resource, &need) in tasks[id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }

            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                current_time,
                id,
                worker,
                resource_snapshot(&used)
            ));

            running.push((finish, id, worker));

            // Zero-duration tasks complete immediately before another start.
            if finish == current_time {
                running.retain(|entry| entry.1 != id);
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);

                for (resource, &need) in tasks[id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }

                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    current_time,
                    id,
                    worker,
                    resource_snapshot(&used)
                ));

                for &child in &children[id] {
                    remaining_predecessors[child] -= 1;
                }
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest future event that may change eligibility.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > current_time {
                next_time = Some(next_time.map_or(finish, |old| old.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > current_time {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |old| old.min(release)));
            }
        }
        for &free_at in &worker_free {
            if free_at > current_time {
                next_time = Some(next_time.map_or(free_at, |old| old.min(free_at)));
            }
        }

        // An unfinished acyclic schedule always has a future release, finish,
        // or cooldown event that can make progress.
        current_time = next_time.expect("unfinished schedule must have a future event");
    }

    trace.push_str(&format!("makespan={}\n", makespan));

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (r, &need) in task.needs.iter().enumerate() {
            if need > capacity[r] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource: r,
                    need,
                    capacity: capacity[r],
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &parent in &task.deps {
            children[parent].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest available ID.
    let mut available: Vec<usize> = (0..n).filter(|&i| indegree[i] == 0).collect();
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(index, _)| index)
            .unwrap();
        let id = available.swap_remove(smallest);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }
    if order.len() != n {
        let removed = {
            let mut v = vec![false; n];
            for &id in &order {
                v[id] = true;
            }
            v
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&i| !removed[i]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut worker_available = vec![0u64; cooldowns.len()];
    let mut running: Vec<Running> = Vec::new();
    let mut used = vec![0u128; capacity.len()];
    let mut trace = format!("workers={};resources={}\n", cooldowns.len(), capacity.len());
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(i, entry)| (entry.finish == now).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| running[i].task);
        for index in finishing.into_iter().rev() {
            let entry = running.remove(index);
            let task_id = entry.task;
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(entry.finish);
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] -= need as u128;
            }
            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }
            trace.push_str(&format!("{}|end|{}|{}|", now, task_id, entry.worker));
            append_usage(&mut trace, &used);
            trace.push('\n');
        }

        // Admit one task at a time, reevaluating eligibility and priority.
        let mut admitted_any = false;
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_deps[id] == 0
                        && fits(&tasks[id], capacity, &used)
                })
                .collect();
            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let mut selected = None;
            for task_id in candidates {
                let worker_count = cooldowns.len();
                for offset in 0..worker_count {
                    let worker = (task_id % worker_count + offset) % worker_count;
                    if worker_available[worker] <= now {
                        selected = Some((task_id, worker));
                        break;
                    }
                }
                if selected.is_some() {
                    break;
                }
            }

            let Some((task_id, worker)) = selected else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow { task: task_id, stage: Stage::Finish })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow { task: task_id, stage: Stage::Turnaround })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += need as u128;
            }
            runs[task_id] = Some(Run { start: now, finish, worker });
            trace.push_str(&format!("{}|start|{}|{}|", now, task_id, worker));
            append_usage(&mut trace, &used);
            trace.push('\n');

            if finish == now {
                // Immediate completion of a zero-duration task.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= need as u128;
                }
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
                trace.push_str(&format!("{}|end|{}|{}|", now, task_id, worker));
                append_usage(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running { task: task_id, worker, finish });
            }
            admitted_any = true;
        }

        if completed_count == n {
            break;
        }

        if admitted_any {
            // The loop has reached a point where no more starts are possible.
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };
        for entry in &running {
            consider(entry.finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }
        let Some(next) = next_time else {
            // This can only arise if no work remains; guard against accidental looping.
            break;
        };
        now = next;
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation must precede all other checks.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successors and indegrees. Dependency duplicates have already
    // been rejected, so each edge is represented once.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }
    for children in &mut successors {
        children.sort_unstable();
    }

    // Kahn traversal always removes the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    let mut kahn_order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(position, _)| position)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        kahn_order.push(id);

        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if kahn_order.len() != n {
        let mut removed = vec![false; n];
        for &id in &kahn_order {
            removed[id] = true;
        }
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &id in kahn_order.iter().rev() {
        let max_successor_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        n
    ];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut worker_available = vec![0u64; worker_count];
    let mut used_resources = vec![0u64; capacity.len()];
    let mut running = Vec::<Running>::new();

    let mut trace = String::new();
    trace.push_str("workers=");
    trace.push_str(&worker_count.to_string());
    trace.push_str(";resources=");
    trace.push_str(&capacity.len().to_string());
    trace.push('\n');

    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now in ascending task ID order.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(position, entry)| {
                if entry.finish == now {
                    Some(position)
                } else {
                    None
                }
            })
            .collect();
        finishing.sort_unstable_by_key(|&position| running[position].task);

        for position in finishing.into_iter().rev() {
            let entry = running.swap_remove(position);
            let task_id = entry.task;

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used_resources[resource] -= u64::from(need);
            }

            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(entry.finish);
            append_event(
                &mut trace,
                now,
                "end",
                task_id,
                entry.worker,
                &used_resources,
            );

            for &child in &successors[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        let mut started_at_this_time = false;

        loop {
            let mut best_task: Option<usize> = None;

            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > now
                    || remaining_predecessors[task_id] != 0
                {
                    continue;
                }

                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(resource, &need)| {
                        u64::from(need)
                            <= u64::from(capacity[resource]) - used_resources[resource]
                    });
                if !fits {
                    continue;
                }

                let is_better = match best_task {
                    None => true,
                    Some(best) => {
                        tails[task_id] > tails[best]
                            || (tails[task_id] == tails[best]
                                && (tasks[task_id].priority > tasks[best].priority
                                    || (tasks[task_id].priority == tasks[best].priority
                                        && task_id < best)))
                    }
                };
                if is_better {
                    best_task = Some(task_id);
                }
            }

            let Some(task_id) = best_task else {
                break;
            };

            let mut chosen_worker = None;
            for offset in 0..worker_count {
                let worker = (task_id % worker_count + offset) % worker_count;
                if worker_available[worker] <= now {
                    chosen_worker = Some(worker);
                    break;
                }
            }

            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;

            let turnaround = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            runs[task_id] = Run {
                start: now,
                finish,
                worker,
            };
            worker_available[worker] = turnaround;

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used_resources[resource] += u64::from(need);
            }

            append_event(
                &mut trace,
                now,
                "start",
                task_id,
                worker,
                &used_resources,
            );

            if tasks[task_id].duration == 0 {
                // Zero-duration completions are immediate, before another
                // task is selected at this same time.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used_resources[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                append_event(
                    &mut trace,
                    now,
                    "end",
                    task_id,
                    worker,
                    &used_resources,
                );
                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    finish,
                    worker,
                });
            }

            started_at_this_time = true;
        }

        if completed_count == n {
            break;
        }

        if started_at_this_time {
            // A start may have unblocked zero-duration work or exhausted
            // current admission opportunities. Re-enter the event loop only
            // after all same-time admissions have been considered.
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now && next_time.map_or(true, |current| candidate < current) {
                next_time = Some(candidate);
            }
        };

        for entry in &running {
            consider(entry.finish);
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                consider(tasks[task_id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        if let Some(next) = next_time {
            now = next;
        } else {
            // With an acyclic graph, every unstarted task has a pending
            // predecessor completion, release, or worker event. This branch
            // is only reachable for the empty schedule, which exits above.
            break;
        }
    }

    trace.push_str("makespan=");
    trace.push_str(&makespan.to_string());
    trace.push('\n');

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependencies in the required order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resources task by task, completing each task's checks before
    // moving to the next.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build the graph and perform deterministic Kahn traversal using a
    // min-heap encoded with Reverse.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    let mut ready = std::collections::BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            ready.push(std::cmp::Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = ready.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut set = vec![false; n];
            for &id in &order {
                set[id] = true;
            }
            set
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u64; capacity.len()];
    let mut trace = String::new();
    use std::fmt::Write as _;
    let _ = writeln!(
        trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    );

    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Finish all tasks at this instant in ascending task ID.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == now)
            .collect();
        finishing.sort_by_key(|entry| entry.task);

        for entry in finishing {
            running.retain(|other| other.task != entry.task);
            completed[entry.task] = true;
            completed_count += 1;
            makespan = makespan.max(now);
            for (resource, &need) in tasks[entry.task].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            write!(
                trace,
                "{}|end|{}|{}|",
                now, entry.task, entry.worker
            )
            .unwrap();
            append_used(&mut trace, &used);
            trace.push('\n');
        }

        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && tasks[id].deps.iter().all(|&dep| completed[dep])
                        && tasks[id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(resource, &need)| {
                                u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                            })
                })
                .collect();

            if candidates.is_empty() {
                break;
            }

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });
            let task_id = candidates[0];

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(
                Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                },
            )?;
            let available = finish.checked_add(cooldowns[worker]).ok_or(
                Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                },
            )?;

            worker_available[worker] = available;
            started[task_id] = true;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            running.push(Running {
                task: task_id,
                worker,
                finish,
            });

            write!(trace, "{}|start|{}|{}|", now, task_id, worker).unwrap();
            append_used(&mut trace, &used);
            trace.push('\n');

            if finish == now {
                // Complete zero-duration tasks immediately, before another
                // task is selected at the same instant.
                running.retain(|entry| entry.task != task_id);
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                write!(trace, "{}|end|{}|{}|", now, task_id, worker).unwrap();
                append_used(&mut trace, &used);
                trace.push('\n');
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_event: Option<u64> = None;
        for entry in &running {
            if entry.finish > now {
                next_event = Some(next_event.map_or(entry.finish, |t| t.min(entry.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next_event = Some(next_event.map_or(tasks[id].release, |t| t.min(tasks[id].release)));
            }
        }
        for &available in &worker_available {
            if available > now {
                next_event = Some(next_event.map_or(available, |t| t.min(available)));
            }
        }

        // With a valid acyclic graph and finite releases, an unstarted task
        // or running task always supplies a future event.
        if let Some(next) = next_event {
            now = next;
        } else {
            break;
        }
    }

    let mut runs: Vec<Run> = runs.into_iter().map(Option::unwrap).collect();
    // The runs were stored by task ID, so the vector is already in required order.
    let _ = &mut runs;

    let _ = writeln!(trace, "makespan={}", makespan);
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn append_used(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            use std::fmt::Write as _;
            write!(trace, "{}", value).unwrap();
        }
    }
}
