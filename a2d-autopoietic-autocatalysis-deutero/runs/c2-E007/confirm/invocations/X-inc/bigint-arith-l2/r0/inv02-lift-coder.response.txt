```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let index = bytes[start..]
                .iter()
                .enumerate()
                .filter(|(_, &b)| b != b'_')
                .nth(excess_number)
                .map(|(offset, _)| start + offset)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits: Vec<u8> = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| b - b'0')
            .collect();
        normalize(&mut digits);

        Ok(Self {
            negative: negative && !is_zero(&digits),
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = magnitude_len.max(min_digits);
        let mut out = String::new();

        if self.negative {
            out.push('-');
        }

        let padding = width - magnitude_len;
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                out.push('_');
            }
            if i < padding {
                out.push('0');
            } else {
                let digit = self.digits[i - padding];
                out.push((b'0' + digit) as char);
            }
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = add_magnitude(&self.digits, &rhs.digits);
            Self {
                negative: self.negative && !is_zero(&digits),
                digits,
            }
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => {
                    let digits = sub_magnitude(&self.digits, &rhs.digits);
                    Self {
                        negative: self.negative && !is_zero(&digits),
                        digits,
                    }
                }
                Ordering::Less => {
                    let digits = sub_magnitude(&rhs.digits, &self.digits);
                    Self {
                        negative: rhs.negative && !is_zero(&digits),
                        digits,
                    }
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        Self {
            negative: !self.negative && !is_zero(&self.digits),
            digits: self.digits.clone(),
        }
    }

    pub fn signum(&self) -> i8 {
        if is_zero(&self.digits) {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if is_zero(&self.digits) || is_zero(&rhs.digits) {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().rev().enumerate() {
            for (j, &b) in rhs.digits.iter().rev().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }

        for i in 0..accum.len() - 1 {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }

        let mut digits: Vec<u8> = accum.into_iter().rev().map(|n| n as u8).collect();
        normalize(&mut digits);

        Self {
            negative: self.negative != rhs.negative,
            digits,
        }
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }
}

fn normalize(digits: &mut Vec<u8>) {
    let first_nonzero = digits.iter().position(|&d| d != 0);
    match first_nonzero {
        Some(index) => {
            if index > 0 {
                digits.drain(0..index);
            }
        }
        None => {
            digits.clear();
            digits.push(0);
        }
    }
}

fn is_zero(digits: &[u8]) -> bool {
    digits.len() == 1 && digits[0] == 0
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => a.cmp(b),
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    let mut ai = a.len();
    let mut bi = b.len();

    while ai > 0 || bi > 0 || carry > 0 {
        let av = if ai > 0 {
            ai -= 1;
            a[ai]
        } else {
            0
        };
        let bv = if bi > 0 {
            bi -= 1;
            b[bi]
        } else {
            0
        };
        let sum = av + bv + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }

    result.reverse();
    normalize(&mut result);
    result
}

fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_magnitude(a, b) != Ordering::Less);

    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i16;
    let mut ai = a.len();
    let mut bi = b.len();

    while ai > 0 {
        ai -= 1;
        let av = i16::from(a[ai]) - borrow;
        let bv = if bi > 0 {
            bi -= 1;
            i16::from(b[bi])
        } else {
            0
        };

        if av < bv {
            result.push((av + 10 - bv) as u8);
            borrow = 1;
        } else {
            result.push((av - bv) as u8);
            borrow = 0;
        }
    }

    result.reverse();
    normalize(&mut result);
    result
}
```
