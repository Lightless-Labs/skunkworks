#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    level: usize,
    burst_pos: usize,
    remaining: u32,
    completion: Option<u64>,
}

#[derive(Clone, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        return 0;
    }
    let numerator = total * 100;
    (numerator + count as u64 / 2) / count as u64
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: job.arrival as u64,
            bursts: job.bursts.clone(),
            level: job.level,
            burst_pos: 0,
            remaining: job.bursts[0],
            completion: None,
        })
        .collect();

    let mut events = Vec::new();
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            level: job.level,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut event_pos = 0usize;
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let admit_through = |limit: u64,
                         events: &[Event],
                         event_pos: &mut usize,
                         queues: &mut [std::collections::VecDeque<usize>]| {
        while *event_pos < events.len() && events[*event_pos].time <= limit {
            let event = &events[*event_pos];
            queues[event.level].push_back(event.index);
            *event_pos += 1;
        }
    };

    loop {
        admit_through(time, &events, &mut event_pos, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_pos >= events.len() {
                break;
            }
            let next_time = events[event_pos].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let process_index = queues[level].pop_front().unwrap();
        let id = processes[process_index].id;
        if last_dispatch_id.is_some_and(|last_id| last_id != id) {
            time += switch_cost as u64;
            admit_through(time, &events, &mut event_pos, &mut queues);
        }

        let start = time;
        let quantum = quanta[level];
        let consumed = processes[process_index].remaining.min(quantum);
        time += consumed as u64;
        let end = time;

        processes[process_index].remaining -= consumed;
        let burst_finished = processes[process_index].remaining == 0;
        let final_burst = burst_finished
            && processes[process_index].burst_pos + 1 == processes[process_index].bursts.len();
        runs.push(Run {
            id,
            start,
            end,
            finished: final_burst,
            level,
        });

        admit_through(end, &events, &mut event_pos, &mut queues);

        if final_burst {
            processes[process_index].completion = Some(end);
        } else if burst_finished {
            let io_delay = processes[process_index].bursts[processes[process_index].burst_pos + 1];
            processes[process_index].burst_pos += 2;
            processes[process_index].remaining = processes[process_index].bursts[processes[process_index].burst_pos];
            let wake_level = level.saturating_sub(1);
            let wake_event = Event {
                time: end + io_delay as u64,
                kind: 1,
                index: process_index,
                level: wake_level,
            };
            let insert_at = events[event_pos..]
                .partition_point(|event| (event.time, event.kind, event.index) <= (wake_event.time, wake_event.kind, wake_event.index))
                + event_pos;
            events.insert(insert_at, wake_event);
            admit_through(end, &events, &mut event_pos, &mut queues);
        } else {
            processes[process_index].level = (level + 1).min(quanta.len() - 1);
            queues[processes[process_index].level].push_back(process_index);
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;
    let mut end_time = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = processes[index].completion.unwrap();
        let turnaround = completion - job.arrival as u64;
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&duration| duration as u64)
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&duration| duration as u64)
            .sum();
        let waiting = turnaround - cpu - io;

        total_waiting += waiting;
        total_turnaround += turnaround;
        end_time = end_time.max(completion);
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end: end_time,
    })
}
