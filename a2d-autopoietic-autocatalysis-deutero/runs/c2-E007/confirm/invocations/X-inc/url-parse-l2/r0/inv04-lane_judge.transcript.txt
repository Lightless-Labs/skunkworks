pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };

        if safe {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| auth_start + n)
        .unwrap_or(bytes.len());

        .bytes()
        .enumerate()

        .collect();

    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + at_positions[1],
        });
    }

    let (userinfo, host_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_start..];
    let absolute_host_start = auth_start + host_start;

    let (host, port, ipv6) = if host_port.starts_with('[') {
        let close = match host_port.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(n) => n + 1,
            None => return Err(ParseError::InvalidIpv6 { at: absolute_host_start }),
        };

        let interior = &host_port[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: absolute_host_start });

        let suffix = &host_port[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], absolute_host_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start + close + 1,
            });
        };

        (interior.to_ascii_lowercase(), parsed_port, true)
    } else {
        let split = host_port.find(':');
        let (host_text, port_text) = match split {
            Some(i) => (&host_port[..i], Some(&host_port[i + 1..])),
            None => (host_port, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: absolute_host_start });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: absolute_host_start + i,
                });
            }
        }

        let parsed_port = match (split, port_text) {
            (Some(i), Some(text)) => Some(parse_port(text, absolute_host_start + i + 1)?),
            _ => None,
        };
        (host_text.to_ascii_lowercase(), parsed_port, false)
    };

    let remainder = &input[auth_end..];
    let (before_fragment, fragment) = match remainder.find('#') {
        Some(i) => (&remainder[..i], Some(&remainder[i + 1..])),
        None => (remainder, None),
    };

    let (path_query, query) = match before_fragment.find('?') {
        Some(i) => (&before_fragment[..i], Some(&before_fragment[i + 1..])),
        None => (before_fragment, None),
    };

    let path = if path_query.is_empty() {
        "/".to_owned()
    } else {
        path_query.to_owned()
    };

    let mut url = Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    };

    if let Some(ref value) = url.userinfo {
        validate_component(Component::UserInfo, value, auth_start)?;
    }
    if !path_query.is_empty() {
        validate_component(Component::Path, &url.path, auth_end)?;
    }
    if let Some(value) = query {
        validate_component(Component::Query, value, auth_end + path_query.len() + 1)?;
    }
    if let Some(value) = fragment {
        let fragment_offset = auth_end + remainder.find('#').unwrap_or(0) + 1;
        validate_component(Component::Fragment, value, fragment_offset)?;
    }

    Ok(url)
}

fn validate_component(component: Component, value: &str, base: usize) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

fn parse_port(text: &str, start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }

    for (i, b) in text.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }

    let bytes = text.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn hex_value(b: u8) -> Option<u8> {

        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(err) => {
            let offset = err.valid_up_to();
            let original_at = origins.get(offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| authority_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let mut at_sign = None;
    for (i, b) in authority.bytes().enumerate() {
        if b == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + i,
                });
            }
            at_sign = Some(i);
        }
    }

    let (userinfo, hostport_start) = if let Some(i) = at_sign {
        if i == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + i,
            });
        }
        (Some(input[authority_start..authority_start + i].to_owned()), authority_start + i + 1)
    } else {
        (None, authority_start)
    };

    let hostport = &input[hostport_start..authority_end];
    let hostport_bytes = hostport.as_bytes();
    let mut ipv6 = false;
    let (host, port, port_colon) = if hostport_bytes.first() == Some(&b'[') {
        let close = hostport_bytes.iter().position(|&b| b == b']');
        let close = match close {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..close];
        if std::net::Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        ipv6 = true;
        let suffix_start = close + 1;
        let suffix = &hostport[suffix_start..];
        let (port, colon_at) = if suffix.is_empty() {
            (None, None)
        } else if suffix.as_bytes()[0] == b':' {
            let absolute_colon = hostport_start + suffix_start;
            (Some(parse_port(&suffix[1..], absolute_colon)?), Some(absolute_colon))
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + suffix_start,
            });
        };
        (interior.to_ascii_lowercase(), port, colon_at)
    } else {
        let colon_in_hostport = hostport_bytes.iter().position(|&b| b == b':');
        let (host_part, port, colon_at) = if let Some(i) = colon_in_hostport {
            let absolute_colon = hostport_start + i;
            (&hostport[..i], Some(parse_port(&hostport[i + 1..], absolute_colon)?), Some(absolute_colon))
        } else {
            (hostport, None, None)
        };

        if host_part.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host_part.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }
        (host_part.to_ascii_lowercase(), port, colon_at)
    };
    let _ = port_colon;

    let fragment_pos = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|n| authority_end + n);
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());

    let query_pos = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|n| authority_end + n);

    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path = if authority_end == path_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_pos.map(|at| input[at + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|at| input[at + 1..].to_owned());

    if let Some(ref value) = userinfo {
        validate_component(Component::UserInfo, value, authority_start)?;
    }
    if path != "/" || path_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let Some(at) = query_pos {
        validate_component(Component::Query, query.as_deref().unwrap_or(""), at + 1)?;
    }
    if let Some(at) = fragment_pos {
        validate_component(Component::Fragment, fragment.as_deref().unwrap_or(""), at + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },

    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },

    UserInfo,
    Path,
    Query,
    Fragment,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = hex_value(bytes[i + 1]);
            let low = hex_value(bytes[i + 2]);
            match (high, low) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = offsets.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for at in 1..colon {
        let b = bytes[at];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = (authority_start..bytes.len())
        .find(|&i| matches!(bytes[i], b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = &bytes[authority_start..authority_end];

    let mut at_positions = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i));
    let first_at = at_positions.next();
    if let Some(second_at) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, host_start_rel) = if let Some(user_at) = first_at {
        if user_at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + user_at,
            });
        }
        (Some(authority[..user_at].to_owned()), user_at + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_start_rel..];
    let host_port_start = authority_start + host_start_rel;

    let (host, port, ipv6) = if host_port.as_bytes().first() == Some(&b'[') {
        let close_rel = host_port.as_bytes()[1..]
            .iter()
            .position(|&b| b == b']')
            .map(|i| i + 1);
        let close_rel = match close_rel {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_port_start }),
        };
        let interior = &host_port[1..close_rel];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_port_start });
        }

        let suffix = &host_port[close_rel + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(&suffix[1..], host_port_start + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_port_start + close_rel + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port, true)
    } else {
        let host_port_bytes = host_port.as_bytes();
        let port_colon = host_port_bytes.iter().position(|&b| b == b':');
        let (host_text, parsed_port) = if let Some(port_rel) = port_colon {
            (
                &host_port[..port_rel],
                Some(parse_port(
                    &host_port[port_rel + 1..],
                    host_port_start + port_rel + 1,
                )?),
            )
        } else {
            (host_port, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_port_start });
        }
        for (rel, &b) in host_text.as_bytes().iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_port_start + rel,
                });
            }
        }
        (host_text.to_ascii_lowercase(), parsed_port, false)
    };

    let tail = &input[authority_end..];
    let tail_bytes = tail.as_bytes();
    let fragment_rel = tail_bytes.iter().position(|&b| b == b'#');
    let before_fragment_end = fragment_rel.unwrap_or(tail.len());
    let before_fragment = &tail[..before_fragment_end];

    let query_rel = before_fragment.as_bytes().iter().position(|&b| b == b'?');
    let (path_text, query, query_base) = if let Some(q) = query_rel {
        (
            &before_fragment[..q],
            Some(before_fragment[q + 1..].to_owned()),
            authority_end + q + 1,
        )
    } else {
        (before_fragment, None, 0)
    };

    let fragment = fragment_rel.map(|f| tail[f + 1..].to_owned());
    let fragment_base = fragment_rel.map(|f| authority_end + f + 1).unwrap_or(0);
    let path_base = authority_end;

    if let Some(ref value) = userinfo {
        validate_component(
            Component::UserInfo,
            value,
            authority_start,
        )?;
    }
    if !path_text.is_empty() {
        validate_component(Component::Path, path_text, path_base)?;
    }
    if let Some(ref value) = query {
        validate_component(Component::Query, value, query_base)?;
    }
    if let Some(ref value) = fragment {
        validate_component(Component::Fragment, value, fragment_base)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path: if path_text.is_empty() {
            "/".to_owned()
        } else {
            path_text.to_owned()
        },
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(text: &str, start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }

    for (rel, &byte) in text.as_bytes().iter().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + rel });
        }
    }

    let bytes = text.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for &byte in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
        {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
    }

    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

            (
                &hostport[..i],
                Some(parse_port(&hostport[i + 1..], hostport_start + i + 1)?),
            )
        } else {
            (hostport, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }
        (host_text.to_ascii_lowercase(), port, false)
    };

    let fragment_delim = input[authority_end..].find('#').map(|i| authority_end + i);
    let before_fragment_end = fragment_delim.unwrap_or(input.len());
    let query_delim = input[authority_end..before_fragment_end]
        .find('?')
        .map(|i| authority_end + i);

    let path_end = query_delim.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_delim.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_delim.map(|f| input[f + 1..].to_owned());

    if let Some(ref userinfo_text) = userinfo {
        validate_component(
            Component::UserInfo,
            userinfo_text,
            authority_start,
        )?;
    }
    if path_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let Some(q) = query_delim {
        validate_component(
            Component::Query,
            query.as_deref().unwrap_or(""),
            q + 1,
        )?;
    }
    if let Some(f) = fragment_delim {
        validate_component(
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            f + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}
