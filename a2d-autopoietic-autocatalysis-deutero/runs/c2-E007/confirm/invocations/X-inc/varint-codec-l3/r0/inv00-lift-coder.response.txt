#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Message {
    pub fields: Vec<Field>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Field {
    UInt { tag: u32, value: u64 },
    Bytes { tag: u32, data: Vec<u8> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
}

fn encode_varint(mut value: u64, output: &mut Vec<u8>) {
    while value >= 0x80 {
        output.push((value as u8 & 0x7f) | 0x80);
        value >>= 7;
    }
    output.push(value as u8);
}

fn decode_varint(
    input: &[u8],
    mut cursor: usize,
    end: usize,
    start: usize,
) -> Result<(u64, usize), Error> {
    let mut value = 0u64;

    for index in 0..10 {
        if cursor >= end {
            return Err(Error::Truncated { offset: end, start });
        }

        let byte = input[cursor];
        let payload = (byte & 0x7f) as u64;

        if index == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow {
                offset: cursor,
                start,
            });
        }

        value |= payload << (index * 7);
        cursor += 1;

        if byte & 0x80 == 0 {
            return Ok((value, cursor));
        }
    }

    unreachable!()
}

pub fn encode_message(message: &Message) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous: Option<(u32, bool)> = None;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, is_bytes) = match field {
            Field::UInt { tag, .. } => (*tag, false),
            Field::Bytes { tag, .. } => (*tag, true),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag {
                offset: index,
                tag: tag as u64,
            });
        }

        if let Some((previous_tag, previous_bytes)) = previous {
            if tag < previous_tag {
                return Err(Error::OutOfOrder {
                    offset: index,
                    previous: previous_tag,
                    tag,
                });
            }
            if tag == previous_tag && !(previous_bytes && is_bytes) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        match field {
            Field::UInt { value, .. } => {
                encode_varint((tag as u64) << 3, &mut body);
                encode_varint(*value, &mut body);
            }
            Field::Bytes { data, .. } => {
                encode_varint(((tag as u64) << 3) | 2, &mut body);
                encode_varint(data.len() as u64, &mut body);
                body.extend_from_slice(data);
            }
        }

        previous = Some((tag, is_bytes));
    }

    let mut output = Vec::new();
    encode_varint(body.len() as u64, &mut output);
    output.extend_from_slice(&body);
    Ok(output)
}

pub fn decode_message(input: &[u8], start: usize) -> Result<(Message, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let (body_length, body_start) = decode_varint(input, start, input.len(), start)?;
    let remaining = input.len() - body_start;
    if body_length > remaining as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: start,
            length: body_length,
        });
    }

    let body_len = body_length as usize;
    let body_end = body_start + body_len;
    let mut cursor = body_start;
    let mut fields = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    while cursor < body_end {
        let key_start = cursor;
        let (key, next_cursor) = decode_varint(input, cursor, body_end, key_start)?;
        cursor = next_cursor;

        let tag_u64 = key >> 3;
        if !(1..=255).contains(&tag_u64) {
            return Err(Error::InvalidTag {
                offset: key_start,
                tag: tag_u64,
            });
        }
        let tag = tag_u64 as u32;
        let wire = (key & 7) as u8;

        if !matches!(wire, 0 | 1 | 2 | 5) {
            return Err(Error::InvalidWire {
                offset: key_start,
                wire,
            });
        }

        if let Some((previous_tag, previous_wire)) = previous {
            if tag < previous_tag {
                return Err(Error::OutOfOrder {
                    offset: key_start,
                    previous: previous_tag,
                    tag,
                });
            }
            if tag == previous_tag && !(previous_wire == 2 && wire == 2) {
                return Err(Error::Duplicate {
                    offset: key_start,
                    tag,
                });
            }
        }

        match wire {
            0 => {
                let payload_start = cursor;
                let (value, next) = decode_varint(input, cursor, body_end, payload_start)?;
                cursor = next;
                if tag <= 15 {
                    fields.push(Field::UInt { tag, value });
                }
            }
            1 | 5 => {
                let length = if wire == 1 { 8usize } else { 4usize };
                if body_end - cursor < length {
                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: length as u64,
                    });
                }
                cursor += length;
            }
            2 => {
                let length_start = cursor;
                let (data_length, next) =
                    decode_varint(input, cursor, body_end, length_start)?;
                cursor = next;

                let data_remaining = body_end - cursor;
                if data_length > data_remaining as u64 {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length: data_length,
                    });
                }

                let data_len = data_length as usize;
                if tag <= 15 {
                    if let Some(Field::Bytes {
                        tag: last_tag,
                        data,
                    }) = fields.last_mut()
                    {
                        if *last_tag == tag {
                            data.extend_from_slice(&input[cursor..cursor + data_len]);
                        } else {
                            fields.push(Field::Bytes {
                                tag,
                                data: input[cursor..cursor + data_len].to_vec(),
                            });
                        }
                    } else {
                        fields.push(Field::Bytes {
                            tag,
                            data: input[cursor..cursor + data_len].to_vec(),
                        });
                    }
                }
                cursor += data_len;
            }
            _ => unreachable!(),
        }

        previous = Some((tag, wire));
    }

    Ok((Message { fields }, body_end))
}
