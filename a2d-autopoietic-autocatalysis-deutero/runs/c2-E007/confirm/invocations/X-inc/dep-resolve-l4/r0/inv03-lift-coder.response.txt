fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b as char, '.' | '_' | '-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn evidence_lines(evidence: &[Evidence]) -> String {
    let mut out = String::new();
    for e in evidence_sorted(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}{}",
            encode(package),
            version_text(*selected),
            encode(feature),
            format!("\n  F:{} via {}", encode(feature), path.join(" -> "))
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    requested: BTreeSet<String>,
    activated: BTreeSet<String>,
    path: Option<Vec<String>>,
}

use std::collections::{BTreeMap, BTreeSet};

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, BTreeMap<String, Version>)>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn release(&self, package: &str, version: Version) -> Option<&'a Release> {
        self.package(package)?
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn platform_matches(&self, p: Platform) -> bool {
        matches!(
            (p, self.target),
            (Platform::Any, _)
                | (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
    }

    fn dependency_active(&self, dep: &Dependency, source_features: &BTreeSet<String>) -> bool {
        self.platform_matches(dep.platform)
            && dep
                .gate
                .as_ref()
                .map_or(true, |gate| source_features.contains(gate))
    }

    fn dep_token(
        source: &str,
        version: Version,
        index: usize,
        dep: &Dependency,
    ) -> String {
        let mut fs = dep.features.clone();
        fs.sort();
        fs.dedup();
        let gate = dep.gate.as_deref().unwrap_or("*");
        let platform = match dep.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        };
        format!(
            "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
            source,
            version_text(version),
            index,
            dep.package,
            constraint_text(dep.constraint),
            fs.join(","),
            gate,
            platform
        )
    }

    fn root_token(i: usize, root: &Requirement) -> String {
        let mut fs = root.features.clone();
        fs.sort();
        fs.dedup();
        format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            root.package,
            constraint_text(root.constraint),
            fs.join(",")
        )
    }

    fn closure(&self, selected: &BTreeMap<String, Version>) -> BTreeMap<String, Active> {
        let mut active: BTreeMap<String, Active> = BTreeMap::new();
        for (i, root) in self.roots.iter().enumerate() {
            let entry = active.entry(root.package.clone()).or_default();
            entry.constraints.push(root.constraint);
            entry.requested.extend(root.features.iter().cloned());
            entry.origins.push(Origin {
                constraint: root.constraint,
                path: vec![Self::root_token(i, root)],
            });
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = active.keys().cloned().collect();
            for name in names {
                let Some(version) = selected.get(&name).copied() else {
                    continue;
                };
                let Some(release) = self.release(&name, version) else {
                    continue;
                };

                let requested = active[&name].requested.clone();
                let mut activated = requested.clone();
                let mut queue: Vec<String> = requested.into_iter().collect();
                while let Some(feature_name) = queue.pop() {
                    if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                        for enabled in &feature.enables {
                            if activated.insert(enabled.clone()) {
                                queue.push(enabled.clone());
                            }
                        }
                    }
                }

                {
                    let entry = active.get_mut(&name).unwrap();
                    if entry.activated != activated {
                        entry.activated = activated.clone();
                        changed = true;
                    }
                }

                for (j, dep) in release.deps.iter().enumerate() {
                    if !self.dependency_active(dep, &activated) {
                        continue;
                    }
                    let source_path = active[&name]
                        .path
                        .clone()
                        .unwrap_or_default();
                    let mut path = source_path;
                    path.push(Self::dep_token(&name, version, j, dep));

                    let entry = active.entry(dep.package.clone()).or_default();
                    if !entry.constraints.contains(&dep.constraint) {
                        entry.constraints.push(dep.constraint);
                        changed = true;
                    }
                    if !entry.origins.iter().any(|o| {
                        o.constraint == dep.constraint && o.path == path
                    }) {
                        entry.origins.push(Origin {
                            constraint: dep.constraint,
                            path,
                        });
                        changed = true;
                    }
                    for f in &dep.features {
                        if entry.requested.insert(f.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            let paths = self.canonical_paths(&active, selected);
            for (name, path) in paths {
                let entry = active.get_mut(&name).unwrap();
                if entry.path.as_ref() != Some(&path) {
                    entry.path = Some(path);
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        active
    }

    fn canonical_paths(
        &self,
        active: &BTreeMap<String, Active>,
        selected: &BTreeMap<String, Version>,
    ) -> BTreeMap<String, Vec<String>> {
        let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
        for (i, root) in self.roots.iter().enumerate() {
            let candidate = vec![Self::root_token(i, root)];
            let replace = paths
                .get(&root.package)
                .map_or(true, |old| (candidate.len(), &candidate) < (old.len(), old));
            if replace {
                paths.insert(root.package.clone(), candidate);
            }
        }

        loop {
            let mut changed = false;
            let current: Vec<(String, Vec<String>)> =
                paths.iter().map(|(n, p)| (n.clone(), p.clone())).collect();
            for (source, source_path) in current {
                let Some(version) = selected.get(&source).copied() else {
                    continue;
                };
                let Some(release) = self.release(&source, version) else {
                    continue;
                };
                let features = active
                    .get(&source)
                    .map(|a| &a.activated)
                    .cloned()
                    .unwrap_or_default();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !self.dependency_active(dep, &features) {
                        continue;
                    }
                    let mut candidate = source_path.clone();
                    candidate.push(Self::dep_token(&source, version, j, dep));
                    let replace = paths
                        .get(&dep.package)
                        .map_or(true, |old| (candidate.len(), &candidate) < (old.len(), old));
                    if replace {
                        paths.insert(dep.package.clone(), candidate);
                        changed = true;
                    }
                }
            }
            if !changed {
                break;
            }
        }
        paths.retain(|name, _| active.contains_key(name));
        paths
    }

    fn canonical_active(&self, active: &mut BTreeMap<String, Active>) {
        for a in active.values_mut() {
            a.constraints = canonical_constraints(a.constraints.clone());
            a.origins.sort_by(|x, y| {
                x.constraint
                    .cmp(&y.constraint)
                    .then_with(|| x.path.cmp(&y.path))
            });
            a.origins.dedup_by(|x, y| x.constraint == y.constraint && x.path == y.path);
        }
    }

    fn inspect(
        &self,
        selected: &BTreeMap<String, Version>,
        active: &BTreeMap<String, Active>,
    ) -> Option<ResolveError> {
        for name in active.keys() {
            if self.package(name).is_none() {
                let evidence = active[name]
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                return Some(ResolveError::Missing {
                    package: name.clone(),
                    evidence,
                });
            }
        }

        for (name, version) in selected {
            if let Some(a) = active.get(name) {
                if !a.constraints.iter().all(|c| matches(*version, *c)) {
                    let evidence = a
                        .origins
                        .iter()
                        .map(|o| Evidence {
                            constraint: o.constraint,
                            path: o.path.clone(),
                        })
                        .collect();
                    return Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: a.constraints.clone(),
                        evidence,
                    });
                }
            }
        }

        for (name, a) in active {
            if selected.contains_key(name) {
                continue;
            }
            let has_candidate = self.package(name).is_some_and(|p| {
                p.releases.iter().any(|r| {
                    a.constraints.iter().all(|c| matches(r.version, *c))
                })
            });
            if !has_candidate {
                let evidence = a
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: a.constraints.clone(),
                    evidence,
                });
            }
        }

        for (name, version) in selected {
            let Some(a) = active.get(name) else {
                continue;
            };
            let Some(release) = self.release(name, *version) else {
                continue;
            };
            for feature in &a.activated {
                if !release.features.iter().any(|f| f.name == *feature) {
                    return Some(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: a.path.clone().unwrap_or_default(),
                    });
                }
            }
        }
        None
    }

    fn solve(&mut self, selected: &mut BTreeMap<String, Version>) {
        let mut active = self.closure(selected);
        self.canonical_active(&mut active);
        if let Some(error) = self.inspect(selected, &active) {
            if !matches!(error, ResolveError::Limit { .. }) && self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return;
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(package_name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| selected.get(*name) != Some(*version))
                .count();
            let replace = self.best.as_ref().map_or(true, |(best_cost, best)| {
                if cost != *best_cost {
                    cost < *best_cost
                } else {
                    self.vector_cmp(selected, best).is_gt()
                }
            });
            if replace {
                self.best = Some((cost, selected.clone()));
            }
            return;
        };

        let constraints = active[&package_name].constraints.clone();
        let package = self.package(&package_name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.get(&package_name) {
            if let Some(i) = candidates.iter().position(|v| v == lock) {
                let locked = candidates.remove(i);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                self.first_error = Some(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
                return;
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), candidate);
            self.solve(selected);
            selected.remove(&package_name);
            if matches!(self.first_error, Some(ResolveError::Limit { .. })) {
                return;
            }
        }
    }

    fn vector_cmp(
        &self,
        a: &BTreeMap<String, Version>,
        b: &BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        for name in &self.catalog_names {
            let ord = match (a.get(name), b.get(name)) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(x), Some(y)) => x.cmp(y),
            };
            if !ord.is_eq() {
                return ord;
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.enable[{}]", base, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut feature_names = BTreeSet::new();
            for feature in &release.features {
                if !feature_names.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut lock_map = BTreeMap::new();
    for lock in locks {
        lock_map.insert(lock.package.clone(), lock.version);
    }
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        roots,
        locks: lock_map.clone(),
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        catalog_names,
    };

    let mut selected = BTreeMap::new();
    search.solve(&mut selected);
    if let Some(ResolveError::Limit { attempts }) = search.first_error {
        return Err(ResolveError::Limit { attempts });
    }
    let Some((_, best)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec: Vec<Selection> = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut active = search.closure(&best);
    search.canonical_active(&mut active);
    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(best.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .package(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !active.get(&package).is_some_and(|a| {
                    a.constraints.iter().all(|c| matches(old, *c))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
