fn prepare_polygon(points: &[Point]) -> Result<&[Point], GeomError> {

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_sqrt(n: u128) -> u128 {
    let mut lo = 0u128;
    let mut hi = 1u128;
    while hi * hi < n {
        hi *= 2;
    }
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid * mid < n {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let lo_sq = lo * lo;
    let hi_sq = hi * hi;
    if n - lo_sq < hi_sq - n {
        lo
    } else {
        hi
    }
}

fn edge_millimeters(a: Point, b: Point) -> u64 {
    rounded_sqrt(squared_distance(a, b) * 1_000_000) as u64
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    if points.len() >= 2 {
        for i in 0..points.len() {
            let a = points[i];
            let b = points[(i + 1) % points.len()];
            sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
        }
    }
    sum
}

fn perimeter(points: &[Point]) -> (Vec<u64>, u64) {
    if points.is_empty() {
        return (Vec::new(), 0);
    }
    let mut lengths = Vec::with_capacity(points.len());
    let mut total = 0u64;
    for i in 0..points.len() {
        let length = edge_millimeters(points[i], points[(i + 1) % points.len()]);
        lengths.push(length);
        total += length;
    }
    (lengths, total)
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare_polygon(points)?;
    let signed_area2 = shoelace(points);
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };
    let (edge_milli, perimeter_milli) = perimeter(points);

    let mut longest_edge = None;
    for i in 0..points.len() {
        let Some(current) = longest_edge else {
            longest_edge = Some(i);
            continue;
        };
        if edge_milli[i] > edge_milli[current] {
            longest_edge = Some(i);
        } else if edge_milli[i] == edge_milli[current] {
            let edge_key = |index: usize| {
                let a = points[index];
                let b = points[(index + 1) % points.len()];
                if a <= b { (a, b) } else { (b, a) }
            };
            if edge_key(i) < edge_key(current)
                || (edge_key(i) == edge_key(current) && i < current)
            {
                longest_edge = Some(i);
            }
        }
    }

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        for i in 0..polygon.len() {
            if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], query) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > query.y) != (b.y > query.y) {
                let side = cross(a, b, query);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(p, _)| *p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    let mut hull: Vec<(Point, usize)>;
    if unique.len() <= 1 {
        hull = unique;
    } else {
        let all_collinear = unique
            .iter()
            .skip(2)
            .all(|&(point, _)| cross(unique[0].0, unique[1].0, point) == 0);

        if all_collinear {
            hull = match mode {
                Collinear::Include => unique,
                Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
            };
        } else {
            let mut lower: Vec<(Point, usize)> = Vec::new();
            for &item in &unique {
                while lower.len() >= 2 {
                    let n = lower.len();
                    let turn = cross(lower[n - 2].0, lower[n - 1].0, item.0);
                    if if mode == Collinear::Include {
                        turn < 0
                    } else {
                        turn <= 0
                    } {
                        lower.pop();
                    } else {
                        break;
                    }
                }
                lower.push(item);
            }

            let mut upper: Vec<(Point, usize)> = Vec::new();
            for &item in unique.iter().rev() {
                while upper.len() >= 2 {
                    let n = upper.len();
                    let turn = cross(upper[n - 2].0, upper[n - 1].0, item.0);
                    if if mode == Collinear::Include {
                        turn < 0
                    } else {
                        turn <= 0
                    } {
                        upper.pop();
                    } else {
                        break;
                    }
                }
                upper.push(item);
            }
            lower.pop();
            upper.pop();
            lower.extend(upper);
            hull = lower;
        }
    }

    let vertices: Vec<Point> = hull.iter().map(|&(point, _)| point).collect();
    let source_indices: Vec<usize> = hull.iter().map(|&(_, index)| index).collect();
    let signed_area2 = shoelace(&vertices);
    let (_, perimeter_milli) = perimeter(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: signed_area2.unsigned_abs(),
        perimeter_milli,
    })
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_POINTS {

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let n = squared_distance(a, b) * 1_000_000;
    let mut lo = 0u128;
    let mut hi = 3_000_000_000u128;
    while lo < hi {
        let mid = lo + (hi - lo + 1) / 2;
        if mid * mid <= n {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    let remainder = n - lo * lo;
    if remainder >= lo * 2 + 1 {
        (lo + 1) as u64
    } else {
        lo as u64
    }
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn perimeter_and_edges(points: &[Point]) -> (Vec<u64>, u64) {
    if points.is_empty() {
        return (Vec::new(), 0);

    for i in 0..points.len() {
        let length = rounded_edge_milli(points[i], points[(i + 1) % points.len()]);
        edges.push(length);
        sum += length;
    }
    (edges, sum)
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare_polygon(points)?;
    let signed = signed_area2(&points);
    let (edge_milli, perimeter_milli) = perimeter_and_edges(&points);

    let orientation = if signed > 0 {
        Orientation::CounterClockwise
    } else if signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let longest_edge = (0..edge_milli.len()).max_by(|&a, &b| {
        edge_milli[a].cmp(&edge_milli[b]).then_with(|| {
            let ea = (
                points[a].min(points[(a + 1) % points.len()]),
                points[a].max(points[(a + 1) % points.len()]),

                points[b].min(points[(b + 1) % points.len()]),
                points[b].max(points[(b + 1) % points.len()]),

        })

        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;

    if queries.len() > MAX_POINTS {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];

            if on_segment(a, b, q) {
                boundary = Some(i);
                break;
            }

            if (a.y > q.y) != (b.y > q.y) {
                let side = cross(a, b, q);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }

        results.push(match boundary {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }

    Ok(results)
}

fn point_cloud(points: &[Point]) -> Result<Vec<(Point, usize)>, GeomError> {
    if points.len() > MAX_POINTS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(existing, _)| *existing == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);
    Ok(unique)
}

fn is_collinear(points: &[(Point, usize)]) -> bool {
    points.len() < 3 || points[2..].iter().all(|&(p, _)| cross(points[0].0, points[1].0, p) == 0)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    let unique = point_cloud(points)?;

    let hull_points: Vec<(Point, usize)> = if unique.len() <= 1 {
        unique
    } else if is_collinear(&unique) {
        match mode {
            Collinear::Include => unique,
            Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
        }
    } else {
        let mut lower: Vec<(Point, usize)> = Vec::new();
        for &point in &unique {
            while lower.len() >= 2 {
                let n = lower.len();
                let turn = cross(lower[n - 2].0, lower[n - 1].0, point.0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                lower.pop();
            }
            lower.push(point);
        }

        let mut upper: Vec<(Point, usize)> = Vec::new();
        for &point in unique.iter().rev() {
            while upper.len() >= 2 {
                let n = upper.len();
                let turn = cross(upper[n - 2].0, upper[n - 1].0, point.0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                upper.pop();
            }
            upper.push(point);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_points.iter().map(|&(point, _)| point).collect();
    let source_indices: Vec<usize> = hull_points.iter().map(|&(_, index)| index).collect();
    let signed = signed_area2(&vertices);
    let (_, perimeter_milli) = perimeter_and_edges(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: signed.unsigned_abs(),
        perimeter_milli,
    })
}
