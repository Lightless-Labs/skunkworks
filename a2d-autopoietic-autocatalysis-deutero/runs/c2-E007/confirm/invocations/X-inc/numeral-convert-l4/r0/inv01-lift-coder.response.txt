#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const ONES: [&str; 20] = [
    "", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten",
    "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "quintillion",
    "quadrillion",
    "trillion",
    "billion",
    "million",
    "thousand",
    "",
];

fn encode_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(ONES[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            encode_group(remainder, out);
        }
    } else if n >= 20 {
        out.push_str(TENS[(n / 10) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(ONES[(n % 10) as usize]);
        }
    } else if n != 0 {
        out.push_str(ONES[n as usize]);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in groups.iter_mut().rev() {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let has_higher = groups[..6].iter().any(|&g| g != 0);
    let units_needs_and = has_higher && groups[6] > 0 && groups[6] < 100;

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut emitted = false;
    for (index, &group) in groups.iter().enumerate() {
        if group == 0 {
            continue;
        }

        if emitted {
            if index == 6 && units_needs_and {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }

        encode_group(group, &mut out);
        if !SCALES[index].is_empty() {
            out.push(' ');
            out.push_str(SCALES[index]);
        }
        emitted = true;
    }

    Ok(out)
}

    match word {
        "zero" => Some((0, WordKind::Number)),
        "one" => Some((1, WordKind::Number)),
        "two" => Some((2, WordKind::Number)),
        "three" => Some((3, WordKind::Number)),
        "four" => Some((4, WordKind::Number)),
        "five" => Some((5, WordKind::Number)),
        "six" => Some((6, WordKind::Number)),
        "seven" => Some((7, WordKind::Number)),
        "eight" => Some((8, WordKind::Number)),
        "nine" => Some((9, WordKind::Number)),
        "ten" => Some((10, WordKind::Number)),
        "eleven" => Some((11, WordKind::Number)),
        "twelve" => Some((12, WordKind::Number)),
        "thirteen" => Some((13, WordKind::Number)),
        "fourteen" => Some((14, WordKind::Number)),
        "fifteen" => Some((15, WordKind::Number)),
        "sixteen" => Some((16, WordKind::Number)),
        "seventeen" => Some((17, WordKind::Number)),
        "eighteen" => Some((18, WordKind::Number)),
        "nineteen" => Some((19, WordKind::Number)),
        "twenty" => Some((20, WordKind::Number)),
        "thirty" => Some((30, WordKind::Number)),
        "forty" => Some((40, WordKind::Number)),
        "fifty" => Some((50, WordKind::Number)),
        "sixty" => Some((60, WordKind::Number)),
        "seventy" => Some((70, WordKind::Number)),
        "eighty" => Some((80, WordKind::Number)),
        "ninety" => Some((90, WordKind::Number)),
        "hundred" => Some((100, WordKind::Hundred)),
        "thousand" => Some((1_000, WordKind::Scale)),
        "million" => Some((1_000_000, WordKind::Scale)),
        "billion" => Some((1_000_000_000, WordKind::Scale)),

        "quadrillion" => Some((1_000_000_000_000_000, WordKind::Scale)),
        "quintillion" => Some((1_000_000_000_000_000_000, WordKind::Scale)),
        "and" => Some((0, WordKind::And)),
        "minus" => Some((0, WordKind::Minus)),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut parsed = Vec::with_capacity(tokens.len());
    for &(token, index) in &tokens {
        let Some((value, kind)) = word_value(token) else {
            return Err(WordsError::InvalidWord { index });
        };
        parsed.push((value, kind, index));
    }

    let negative = matches!(parsed.first(), Some((_, WordKind::Minus, _)));
    for (position, &(_, kind, index)) in parsed.iter().enumerate() {
        if matches!(kind, WordKind::Minus) && position != 0 {
            return Err(WordsError::InvalidWord { index });
        }
    }

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    let skip = usize::from(negative);

    for &(value, kind, _) in parsed.iter().skip(skip) {
        match kind {
            WordKind::Number => group += value,
            WordKind::Hundred => group *= 100,
            WordKind::Scale => {
                total += group * value;
                group = 0;
            }
            WordKind::And | WordKind::Minus => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let result = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(result)?;
    if canonical.as_bytes() != bytes {
        let mismatch = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(result)
}
