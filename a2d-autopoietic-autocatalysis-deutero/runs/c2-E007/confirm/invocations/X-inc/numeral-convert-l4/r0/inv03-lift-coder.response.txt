#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const ONES: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
    (1, ""),
];

fn append_small(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(ONES[n as usize]);
    } else {
        let tens = n / 10;
        let units = n % 10;
        out.push_str(TENS[(tens - 2) as usize]);
        if units != 0 {
            out.push('-');
            out.push_str(ONES[units as usize]);
        }
    }
}

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(ONES[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_small(remainder, out);
        }
    } else {
        append_small(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut groups: Vec<(u16, &str)> = Vec::new();
    let mut remaining = magnitude;
    while remaining != 0 {
        groups.push(((remaining % 1000) as u16, ""));
        remaining /= 1000;
    }

    // Reverse the collected groups into high-to-low order and attach scales.
    groups.reverse();
    let scale_names = [
        "", "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
    ];
    let group_count = groups.len();
    let mut emitted = false;

    for (index, (group, _)) in groups.into_iter().enumerate() {
        if group == 0 {
            continue;
        }

        let scale_index = group_count - 1 - index;
        let scale = scale_names[scale_index];
        let is_units_group = scale_index == 0;
        let cross_and = is_units_group && emitted && group < 100;

        if emitted {
            result.push(' ');
        }
        if cross_and {
            result.push_str("and ");
        }
        append_group(group, &mut result);
        if !scale.is_empty() {
            result.push(' ');
            result.push_str(scale);
        }
        emitted = true;
    }

    Ok(result)
}

#[derive(Clone, Copy)]
enum Word {
    Number(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

fn lookup_word(word: &str) -> Option<Word> {
    if word == "zero" {
        return Some(Word::Number(0));
    }
    for (i, spelling) in ONES.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(Word::Number(i as u128));
        }
    }
    for (i, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Number(((i + 2) * 10) as u128));
        }
    }
    match word {
        "hundred" => Some(Word::Hundred),
        "thousand" => Some(Word::Scale(1_000)),
        "million" => Some(Word::Scale(1_000_000)),
        "billion" => Some(Word::Scale(1_000_000_000)),
        "trillion" => Some(Word::Scale(1_000_000_000_000)),
        "quadrillion" => Some(Word::Scale(1_000_000_000_000_000)),
        "quintillion" => Some(Word::Scale(1_000_000_000_000_000_000)),
        "and" => Some(Word::And),
        "minus" => Some(Word::Minus),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut words = Vec::with_capacity(tokens.len());
    for &(token, index) in &tokens {
        match lookup_word(token) {
            Some(Word::Minus) if index != 0 => {
                return Err(WordsError::InvalidWord { index });
            }
            Some(word) => words.push(word),
            None => return Err(WordsError::InvalidWord { index }),
        }
    }

    let negative = matches!(words.first(), Some(Word::Minus));
    let arithmetic = if negative { &words[1..] } else { &words[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for word in arithmetic {
        match *word {
            Word::Number(value) => group += value,
            Word::Hundred => group *= 100,
            Word::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
            Word::And | Word::Minus => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != input.as_bytes() {
        let canonical_bytes = canonical.as_bytes();
        let common = canonical_bytes.len().min(bytes.len());
        let index = (0..common)
            .find(|&i| canonical_bytes[i] != bytes[i])
            .unwrap_or(common);
        return Err(WordsError::NonCanonical { index });
    }
    Ok(value)
}
