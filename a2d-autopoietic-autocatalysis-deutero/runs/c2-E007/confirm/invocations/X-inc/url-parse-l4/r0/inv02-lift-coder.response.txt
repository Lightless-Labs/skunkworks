use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

pub enum PathError {

    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

#[derive(Debug, Clone, PartialEq, Eq)]

    Base(ParseError),
    Reference(ParseError),

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn scan_ascii_url_bytes(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        if bytes[index] == b'%' {
            if index + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: index });
            }
            let Some(high) = hex_value(bytes[index + 1]) else {
                return Err(CodecError::BadEscape { at: index });
            };
            let Some(low) = hex_value(bytes[index + 2]) else {
                return Err(CodecError::BadEscape { at: index });
            };
            decoded.push((high << 4) | low);
            origins.push(index);
            index += 3;
        } else {
            decoded.push(bytes[index]);
            origins.push(index);
            index += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            let original_at = origins.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

fn component_byte_is_safe(component: Component, byte: u8) -> bool {
    matches!(byte, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for byte in text.bytes() {
        if component_byte_is_safe(component, byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn parse_port(port_text: &str, port_at: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_at });
    }

    if let Some(offset) = port_text.bytes().position(|byte| !byte.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: port_at + offset });
    }

    if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_at });
    }

    let mut value: u32 = 0;
    for byte in port_text.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|current| current.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: port_at }),
        };
    }

    if !(1..=65_535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_at });
    }
    Ok(value as u16)
}

fn validate_url_component(component: Component, value: &str, base: usize) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_ascii_url_bytes(input)?;

    let bytes = input.as_bytes();
    let Some(colon) = bytes.iter().position(|&byte| byte == b':') else {
        return Err(ParseError::InvalidScheme { at: input.len() });
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let scheme = input[..colon].to_ascii_lowercase();
    let authority_start = colon + 3;
    let authority_end = input[authority_start..]
        .bytes()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let mut at_positions = authority.match_indices('@').map(|(index, _)| index);
    let first_at = at_positions.next();
    if let Some(second_at) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }
    if first_at == Some(0) {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start,
        });
    }

    let (userinfo, host_start, host_port) = if let Some(at) = first_at {
        (
            Some(authority[..at].to_owned()),
            authority_start + at + 1,
            &authority[at + 1..],
        )
    } else {
        (None, authority_start, authority)
    };

    let mut ipv6 = false;
    let (host, port) = if host_port.starts_with('[') {
        let Some(close_rel) = host_port[1..].find(']') else {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        };
        let close = 1 + close_rel;
        let interior = &host_port[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }

        let suffix = &host_port[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if let Some(port_text) = suffix.strip_prefix(':') {
            Some(parse_port(port_text, host_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), port)
    } else {
        let host_colon = host_port.find(':');
        let (host_text, port) = if let Some(colon_in_host) = host_colon {
            let port_text = &host_port[colon_in_host + 1..];
            (
                &host_port[..colon_in_host],
                Some(parse_port(port_text, host_start + colon_in_host + 1)?),
            )
        } else {
            (host_port, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), port)
    };

    let remainder = &input[authority_end..];
    let fragment_rel = remainder.find('#');
    let before_fragment_end = fragment_rel.unwrap_or(remainder.len());
    let before_fragment = &remainder[..before_fragment_end];
    let query_rel = before_fragment.find('?');

    let path_end = query_rel.unwrap_or(before_fragment.len());
    let raw_path = &before_fragment[..path_end];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    let query = query_rel.map(|query_at| before_fragment[query_at + 1..].to_owned());
    let fragment = fragment_rel.map(|fragment_at| remainder[fragment_at + 1..].to_owned());

    if let Some(raw_userinfo) = userinfo.as_deref() {
        validate_url_component(Component::UserInfo, raw_userinfo, authority_start)?;
    }
    if !raw_path.is_empty() {
        validate_url_component(Component::Path, raw_path, authority_end)?;
    }
    if let Some(query_at) = query_rel {
        validate_url_component(
            Component::Query,
            query.as_deref().unwrap_or(""),
            authority_end + query_at + 1,
        )?;
    }
    if let Some(fragment_at) = fragment_rel {
        validate_url_component(
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            authority_end + fragment_at + 1,
        )?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn codec_error_to_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_error_to_path)?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut final_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(codec_error_to_path)?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if index + 1 == segments.len() {
            final_navigation = is_dot || is_dotdot;
        }

        if is_dot {
            continue;
        } else if is_dotdot {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if final_navigation {
        stack.push("");
    }
    let joined = stack.join("/");
    if joined.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{joined}"))
    }
}

fn construct_url(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut output = String::new();
    output.push_str(&url.scheme);
    output.push_str("://");
    if let Some(userinfo) = &url.userinfo {
        output.push_str(userinfo);
        output.push('@');
    }
    if url.ipv6 {
        output.push('[');
        output.push_str(&url.host);
        output.push(']');
    } else {
        output.push_str(&url.host);
    }
    if let Some(port) = url.port {
        output.push(':');
        output.push_str(&port.to_string());
    }
    output.push_str(path);
    if let Some(query) = query {
        output.push('?');
        output.push_str(query);
    }
    if let Some(fragment) = fragment {
        output.push('#');
        output.push_str(fragment);
    }
    output
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base_path =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_ascii_url_bytes(reference).map_err(ResolveError::Reference)?;

    let first_structural = reference
        .bytes()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_structural].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let fragment_at = reference.find('#');
    let before_fragment_end = fragment_at.unwrap_or(reference.len());
    let before_fragment = &reference[..before_fragment_end];
    let query_at = before_fragment.find('?');
    let reference_path_end = query_at.unwrap_or(before_fragment.len());
    let reference_path = &before_fragment[..reference_path_end];
    let reference_query = query_at.map(|at| &before_fragment[at + 1..]);
    let reference_fragment = fragment_at.map(|at| &reference[at + 1..]);

    let selected_path = if reference_path.is_empty() {
        normalized_base_path.clone()
    } else if reference_path.starts_with('/') {
        reference_path.to_owned()
    } else {
        let slash = normalized_base_path.rfind('/').unwrap_or(0);
        format!("{}{}", &normalized_base_path[..=slash], reference_path)
    };

    let selected_query = if reference_path.is_empty() && query_at.is_none() {
        base_url.query.as_deref()
    } else {
        reference_query
    };

    let constructed = construct_url(
        &base_url,
        &selected_path,
        selected_query,
        reference_fragment,
    );
    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn decode_form_component(input: &str) -> Result<String, CodecError> {
    let replaced = input.replace('+', " ");
    decode_component(Component::Query, &replaced)
}

fn encode_form_component(input: &str) -> String {
    let mut output = String::with_capacity(input.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for byte in input.bytes() {
        if matches!(byte, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else if byte == b' ' {
            output.push('+');
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn codec_error_to_query(error: CodecError, base: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();

    for (field_start, field) in input.split('&').scan(0usize, |offset, field| {
        let start = *offset;
        *offset += field.len() + 1;
        Some((start, field))
    }) {
        if field.is_empty() {
            continue;
        }

        let equals_at = field.find('=');
        let key_raw = equals_at.map(|at| &field[..at]).unwrap_or(field);
        let value_raw = equals_at.map(|at| &field[at + 1..]).unwrap_or("");

        let key = decode_form_component(key_raw)
            .map_err(|error| codec_error_to_query(error, field_start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }

        let value_base = field_start + equals_at.map(|at| at + 1).unwrap_or(field.len());
        let value = decode_form_component(value_raw)
            .map_err(|error| codec_error_to_query(error, value_base))?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| existing == &key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();

    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, accumulated)) = merged.iter_mut().find(|(existing, _)| existing == key) {
            accumulated.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = encode_form_component(&key);
        for value in values {
            fields.push(format!("{encoded_key}={}", encode_form_component(&value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for byte in text.bytes() {
        if matches!(byte, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let normalized_path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = String::new();
    output.push_str(&url.scheme);
    output.push_str("://");

    if let Some(userinfo) = url.userinfo.as_deref() {
        let decoded =
            decode_component(Component::UserInfo, userinfo).expect("parse_url validated userinfo");
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }

    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host).expect("parse_url validated IPv6 host");
        output.push('[');
        output.push_str(&address.to_string());
        output.push(']');
    } else {
        output.push_str(&url.host);
    }

    let omit_port = matches!(
        (url.scheme.as_str(), url.port),
        ("http", Some(80)) | ("https", Some(443))
    );
    if let Some(port) = url.port.filter(|_| !omit_port) {
        output.push(':');
        output.push_str(&port.to_string());
    }

    let encoded_path = normalized_path
        .split('/')
        .map(|segment| {
            let decoded =
                decode_component(Component::Path, segment).expect("normalized path is valid");
            encode_unreserved(&decoded)
        })
        .collect::<Vec<_>>()
        .join("/");
    output.push_str(&encoded_path);

    if let Some(query_map) = query_map {
        output.push('?');
        output.push_str(&serialise_query(&query_map));
    }
    if let Some(fragment) = url.fragment.as_deref() {
        let decoded =
            decode_component(Component::Fragment, fragment).expect("parse_url validated fragment");
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}
