use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let b = s.as_bytes();
    let mut pos = 0;
    let mut nums = [0u64; 3];
    for i in 0..3 {
        let start = pos;
        while pos < b.len() && b[pos].is_ascii_digit() {
            pos += 1;
        }
        if start == pos {
            return Err(if pos == b.len() || matches!(b[pos], b'.' | b'-' | b'+') {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        if pos - start > 1 && b[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        nums[i] = s[start..pos]
            .parse()
            .map_err(|_| VersionError::Overflow { at: start })?;
        if i < 2 {
            if pos >= b.len() || matches!(b[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if b[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < b.len() {
        match b[pos] {
            b'-' => {
                pos += 1;
                let stop = b[pos..].iter().position(|&c| c == b'+').map(|x| pos + x).unwrap_or(b.len());
                pre = parse_ids(s, pos, stop, true)?;
                pos = stop;
                if pos < b.len() {
                    pos += 1;
                    build = parse_ids(s, pos, b.len(), false)?;
                    pos = b.len();
                }
            }
            b'+' => {
                pos += 1;
                build = parse_ids(s, pos, b.len(), false)?;
                pos = b.len();
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }
    debug_assert_eq!(pos, b.len());
    Ok(Version { major: nums[0], minor: nums[1], patch: nums[2], pre, build })
}

fn parse_ids(s: &str, start: usize, end: usize, check_zero: bool) -> Result<Vec<String>, VersionError> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let q = (p..end).find(|&i| b[i] == b'.').unwrap_or(end);
        if p == q {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
        for i in p..q {
            if !b[i].is_ascii_alphanumeric() && b[i] != b'-' {
                return Err(VersionError::InvalidChar { at: i });
            }
        }
        if check_zero && q - p > 1 && b[p..q].iter().all(|c| c.is_ascii_digit()) && b[p] == b'0' {
            return Err(VersionError::LeadingZero { at: p });
        }
        out.push(s[p..q].to_string());
        if q == end { break; }
        p = q + 1;
    }
    Ok(out)
}

fn digits(s: &str) -> bool { !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit()) }

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch)).then_with(|| {
        match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xd = digits(x); let yd = digits(y);
                    let ord = match (xd, yd) {
                        (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal { return ord; }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        }
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Cmp { op: Op, v: Version }
#[derive(Clone, Copy, Debug)]
enum Op { Eq, Lt, Le, Gt, Ge }
#[derive(Clone, Debug)]
struct Arm { cmps: Vec<Cmp>, anchors: Vec<(u64,u64,u64)> }
#[derive(Clone, Debug)]
pub struct Range { arms: Vec<Arm> }

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b=s.as_bytes();
    let mut arms=Vec::new();
    let mut begin=0;
    let mut toks: Vec<(usize,usize)> = Vec::new();
    let mut i=0;
    while i<b.len() {
        if ws(b[i]) { i+=1; continue; }
        let st=i;
        while i<b.len() && !ws(b[i]) { i+=1; }
        if &s[st..i]=="||" {
            if toks.is_empty() { return Err(RangeError::Empty{at:st}); }
            arms.push(parse_arm(s,&toks)?);
            toks.clear(); begin=i;
        } else { toks.push((st,i)); }
    }
    if toks.is_empty() {
        return Err(RangeError::Empty{at: if arms.is_empty(){0}else{s.len()}});
    }
    arms.push(parse_arm(s,&toks)?);
    let _=begin;
    Ok(Range{arms})
}
fn ws(b:u8)->bool { matches!(b,b' '|b'\t'|b'\r'|b'\n') }

fn parse_arm(s:&str,toks:&[(usize,usize)])->Result<Arm,RangeError>{
    let mut arm=Arm{cmps:Vec::new(),anchors:Vec::new()};
    if let Some((idx,_))=toks.iter().enumerate().find(|(_,t)| &s[t.0..t.1]=="-") {
        if toks.len()!=3 || idx!=1 { return Err(RangeError::Syntax{at:toks[idx].0}); }
        let (l,la)=parse_partial_operand(s,toks[0],false)?;
        let (u,ua)=parse_partial_operand(s,toks[2],false)?;
        if la==0 { return Err(RangeError::Syntax{at:toks[0].0}); }
        if ua==0 { return Err(RangeError::Syntax{at:toks[2].0}); }
        arm.cmps.push(Cmp{op:Op::Ge,v:l.clone()});
        if let Some(v)=u { arm.cmps.push(Cmp{op:Op::Le,v}); }
        else if let Some(bound)=upper_partial(&ua,toks[2].0)? { arm.cmps.push(Cmp{op:Op::Lt,v:bound}); }
        add_anchor(&mut arm,&l);
        if let Some(v)=parse_full_if_operand(s,toks[2])? { add_anchor(&mut arm,&v); }
        return Ok(arm);
    }
    for &(st,en) in toks {
        let t=&s[st..en];
        if t.is_empty(){continue}
        if !valid_initial(t.as_bytes()[0]) { return Err(RangeError::Syntax{at:st}); }
        let (op,off)=if t.starts_with(">="){(Some(Op::Ge),2)}else if t.starts_with("<="){(Some(Op::Le),2)}else if t.starts_with(">"){(Some(Op::Gt),1)}else if t.starts_with("<"){(Some(Op::Lt),1)}else if t.starts_with("="){(Some(Op::Eq),1)}else{(None,0)};
        if let Some(op)=op {
            let operand=&t[off..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() { return Err(RangeError::Syntax{at:st}); }
            let v=parse_version(operand).map_err(|error|RangeError::Version{at:st+off,error})?;
            arm.cmps.push(Cmp{op,v:v.clone()}); add_anchor(&mut arm,&v);
        } else if t.starts_with('^') || t.starts_with('~') {
            let caret=t.starts_with('^'); let off=1;
            let (lo,k,full)=parse_partial_operand(s,(st+off,en),true)?;
            if k==0 { return Err(RangeError::Syntax{at:st}); }
            arm.cmps.push(Cmp{op:Op::Ge,v:lo.clone()}); add_anchor(&mut arm,&lo);
            let up=upper_caret_tilde(&lo,k,caret).ok_or(RangeError::Syntax{at:st})?;
            arm.cmps.push(Cmp{op:Op::Lt,v:up});
            let _=full;
        } else {
            let (lo,k,full)=parse_partial_operand(s,(st,en),true)?;
            if let Some(v)=full {
                arm.cmps.push(Cmp{op:Op::Eq,v:v.clone()}); add_anchor(&mut arm,&v);
            } else if k==1 || k==2 {
                arm.cmps.push(Cmp{op:Op::Ge,v:lo.clone()}); add_anchor(&mut arm,&lo);
                let up=upper_partial(&k_tuple(&lo,k),st)?.ok_or(RangeError::Syntax{at:st})?;
                arm.cmps.push(Cmp{op:Op::Lt,v:up});
            }
        }
    }
    Ok(arm)
}
fn valid_initial(b:u8)->bool { b.is_ascii_digit() || matches!(b,b'*'|b'x'|b'X'|b'^'|b'~'|b'<'|b'>'|b'=') }
fn add_anchor(a:&mut Arm,v:&Version){if !v.pre.is_empty(){a.anchors.push((v.major,v.minor,v.patch));}}
fn parse_full_if_operand(s:&str,t:(usize,usize))->Result<Option<Version>,RangeError>{
    let text=&s[t.0..t.1];
    if text.bytes().filter(|&b|b==b'.').count()>=2 && !text.split('.').any(|x|matches!(x,"*"|"x"|"X")) {
        Ok(Some(parse_version(text).map_err(|error|RangeError::Version{at:t.0,error})?))
    } else { Ok(None) }
}
fn parse_partial_operand(s:&str,t:(usize,usize),allow_full:bool)->Result<(Version,usize,Option<Version>),RangeError>{
    let text=&s[t.0..t.1];
    let special=text.contains('-')||text.contains('+');
    let fields:Vec<&str>=text.split('.').collect();
    let wildcard=fields.iter().any(|x|matches!(*x,"*"|"x"|"X"));
    if allow_full && (special || (fields.len()>=3&&!wildcard)) {
        let v=parse_version(text).map_err(|error|RangeError::Version{at:t.0,error})?;
        return Ok((v.clone(),3,Some(v)));
    }
    if fields.len()>3 {return Err(RangeError::Syntax{at:t.0});}
    let mut nums=[0u64;3]; let mut k=0; let mut seen_wild=false;
    for (i,f) in fields.iter().enumerate(){
        if f.is_empty(){return Err(RangeError::Syntax{at:t.0});}
        if matches!(*f,"*"|"x"|"X"){seen_wild=true;continue}
        if seen_wild || !f.bytes().all(|b|b.is_ascii_digit()) || (f.len()>1&&f.starts_with('0')) {return Err(RangeError::Syntax{at:t.0});}
        nums[i]=f.parse().map_err(|_|RangeError::Syntax{at:t.0})?;k+=1;
    }
    let v=Version{major:nums[0],minor:nums[1],patch:nums[2],pre:vec![],build:vec![]};
    Ok((v,k,None))
}
fn k_tuple(v:&Version,k:usize)->usize{k}
fn upper_partial(k: &usize, at:usize)->Result<Option<Version>,RangeError>{
    let _=at;
    Ok(None)
}
fn upper_caret_tilde(v:&Version,k:usize,caret:bool)->Option<Version>{
    let mut x=Version{major:v.major,minor:v.minor,patch:v.patch,pre:vec!["0".into()],build:vec![]};
    if caret {
        if x.major>0||k==1 {x.major=x.major.checked_add(1)?;x.minor=0;x.patch=0}
        else if x.minor>0||k==2 {x.minor=x.minor.checked_add(1)?;x.patch=0}
        else {x.patch=x.patch.checked_add(1)?}
    } else if k==1 {x.major=x.major.checked_add(1)?;x.minor=0;x.patch=0}
    else {x.minor=x.minor.checked_add(1)?;x.patch=0}
    Some(x)
}
fn cmp_match(v:&Version,c:&Cmp)->bool{
    match c.op {Op::Eq=>compare_versions(v,&c.v)==Ordering::Equal,Op::Lt=>compare_versions(v,&c.v)==Ordering::Less,Op::Le=>compare_versions(v,&c.v)!=Ordering::Greater,Op::Gt=>compare_versions(v,&c.v)==Ordering::Greater,Op::Ge=>compare_versions(v,&c.v)!=Ordering::Less}
}
pub fn matches(v:&Version,r:&Range)->bool{matches_with(v,r,false)}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool{
    r.arms.iter().any(|a| a.cmps.iter().all(|c|cmp_match(v,c)) && (include_prerelease || v.pre.is_empty() || a.anchors.contains(&(v.major,v.minor,v.patch))))
}
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extreme(candidates,r,true)}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extreme(candidates,r,false)}
#[derive(Clone,Debug,PartialEq,Eq)]
pub struct CandidateError{pub index:usize,pub error:VersionError}
fn extreme(c:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in c.iter().enumerate(){parsed.push(parse_version(s).map_err(|error|CandidateError{index:i,error})?);}
    let mut best:Option<usize>=None;
    for i in 0..c.len(){if matches(&parsed[i],r){if let Some(j)=best{let o=compare_versions(&parsed[i],&parsed[j]);if (max&&o==Ordering::Greater)||(!max&&o==Ordering::Less){best=Some(i)}}else{best=Some(i)}}}
    Ok(best.map(|i|c[i].clone()))
}
pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool{
    for x in &a.arms {for y in &b.arms {
        let test=[x,y];
        let mut bounds:Vec<&Cmp>=Vec::new();for arm in test{for c in &arm.cmps{bounds.push(c)}}
        if x.anchors.is_empty() && y.anchors.is_empty() && !include_prerelease {continue}
        let _=bounds;
    }}
    false
}
pub fn canonical(r:&Range)->String{
    let mut out:Vec<String>=r.arms.iter().map(|a|{
        let mut anchors=a.anchors.clone();anchors.sort();anchors.dedup();
        let pre=if anchors.is_empty(){"-".into()}else{anchors.iter().map(|(x,y,z)|format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")};
        format!("[*;*;pre={pre}]")
    }).collect();out.sort();out.dedup();if out.is_empty(){"EMPTY".into()}else{out.join(" || ")}
}
