#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    modifier: Modifier,
    pos: usize,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone)]
struct Expression {
    operator: char,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn syntax_error(template: &str) -> Result<Vec<(char, Vec<Specifier>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = i + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }
                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: open + 1 + offset });
                }

                let mut content_start = open + 1;
                let mut operator = '\0';
                if content_start < close && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' | b'!' | b'@' | b'=' | b',' | b':' | b'%' | b' ' | b'\\' | b'-' | b'(' | b')' | b'[' | b']' | b'~' | b'$' | b'\'' | b'*') {
                    let b = bytes[content_start];
                    if matches!(b, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                        operator = b as char;
                        content_start += 1;
                    } else if !matches!(b, b'!' | b'@' | b'=' | b',' | b':' | b'%' | b' ' | b'\\' | b'-' | b'(' | b')' | b'[' | b']' | b'~' | b'$' | b'\'' | b'*') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut specs = Vec::new();
                let mut start = content_start;
                let mut seen = Vec::<String>::new();
                while start <= close {
                    let mut end = start;
                    while end < close && bytes[end] != b',' {
                        end += 1;
                    }
                    if end == start {
                        return Err(Error::EmptyVariable { pos: start });
                    }

                    let part = &template[start..end];
                    let split = part.find(['*', ':']).unwrap_or(part.len());
                    let name_part = &part[..split];
                    if name_part.is_empty() {
                        return Err(Error::InvalidName { pos: start });
                    }

                    let name_bytes = name_part.as_bytes();
                    if !is_name_start(name_bytes[0]) {
                        return Err(Error::InvalidName { pos: start });
                    }
                    for (offset, &b) in name_bytes.iter().enumerate().skip(1) {
                        if !is_name_continue(b) {
                            return Err(Error::InvalidName { pos: start + offset });
                        }
                    }

                    let modifier = if split == part.len() {
                        Modifier::None
                    } else {
                        let suffix = &part[split..];
                        let modifier_pos = start + split;
                        if suffix == "*" {
                            Modifier::Explode
                        } else if let Some(digits) = suffix.strip_prefix(':') {
                            let bs = digits.as_bytes();
                            if bs.is_empty()
                                || bs.len() > 4
                                || bs[0] == b'0'
                                || !bs.iter().all(u8::is_ascii_digit)
                            {
                                return Err(Error::InvalidModifier { pos: modifier_pos });
                            }
                            let n = digits.parse::<usize>().unwrap_or(0);
                            if !(1..=9999).contains(&n) {
                                return Err(Error::InvalidModifier { pos: modifier_pos });
                            }
                            Modifier::Prefix(n)
                        } else {
                            return Err(Error::InvalidModifier { pos: modifier_pos });
                        }
                    };

                    let name = name_part.to_owned();
                    if seen.iter().any(|previous| previous == &name) {
                        return Err(Error::DuplicateVariable { pos: start });
                    }
                    seen.push(name.clone());
                    specs.push(Specifier {
                        name,
                        modifier,
                        pos: start,
                    });

                    if end == close {
                        break;
                    }
                    start = end + 1;
                }

                expressions.push((operator, specs));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(b, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'=');
        if unreserved || reserved_byte {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: char) -> String {
    if operator == ';' && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(operator: char, specs: &[Specifier], vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(operator, '+' | '#');
    let separator = match operator {
        '.' => '.',
        '/' => '/',
        ';' => ';',
        '?' | '&' => '&',
        _ => ',',
    };

    let mut fields = Vec::<String>::new();
    for spec in specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        match value {
            Value::Scalar(s) => {
                let source = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                fields.push(encode(&source, reserved));
            }
            Value::List(items) => {
                if matches!(spec.modifier, Modifier::Prefix(_)) {
                    return Err(Error::TypeMismatch { pos: spec.pos });
                }
                if items.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(operator, ';' | '?' | '&') {
                            named_field(&spec.name, encoded, operator)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items.iter().map(|item| encode(item, reserved)).collect::<Vec<_>>().join(",");
                    fields.push(if matches!(operator, ';' | '?' | '&') {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if matches!(spec.modifier, Modifier::Prefix(_)) {
                    return Err(Error::TypeMismatch { pos: spec.pos });
                }
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(key, _)| key == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }
                if entries.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, val) in entries {
                        let k = encode(key, reserved);
                        let v = encode(val, reserved);
                        let field = if operator == ';' && v.is_empty() {
                            k
                        } else {
                            format!("{k}={v}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(val, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(operator, ';' | '?' | '&') {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let body = fields.join(&separator.to_string());
    let prefix = match operator {
        '#' => "#",
        '.' => ".",
        '/' => "/",
        ';' => ";",
        '?' => "?",
        '&' => "&",
        _ => "",
    };
    Ok(format!("{prefix}{body}"))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = syntax_error(template)?;
    let mut output = String::new();
    let mut cursor = 0;
    let bytes = template.as_bytes();
    let mut expr_index = 0;

    while cursor < bytes.len() {
        if bytes[cursor] == b'{' {
            let mut close = cursor + 1;
            while close < bytes.len() && bytes[close] != b'}' {
                close += 1;
            }
            output.push_str(&template[cursor..cursor]); // keep boundary handling byte based
            let (operator, specs) = &expressions[expr_index];
            output.push_str(&expand_expression(*operator, specs, vars)?);
            expr_index += 1;
            cursor = close + 1;
        } else {
            let start = cursor;
            while cursor < bytes.len() && bytes[cursor] != b'{' && bytes[cursor] != b'}' {
                cursor += 1;
            }
            output.push_str(&template[start..cursor]);
        }
    }

    Ok(output)
}
