#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Plus),
            b'#' => Some(Self::Hash),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Question),
            b'&' => Some(Self::Ampersand),
            _ => None,
        }
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Plus | Self::Hash)
    }

    fn separator(self) -> char {
        match self {
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Question | Self::Ampersand => '&',
            Self::Simple | Self::Plus | Self::Hash => ',',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Hash => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Question => Some('?'),
            Self::Ampersand => Some('&'),
            Self::Simple | Self::Plus => None,
        }
    }

    fn named_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Question | Self::Ampersand)
    }
}

#[derive(Debug)]
struct Specifier {
    name: String,
    start: usize,
    explode: bool,
    prefix: Option<usize>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn validate_expression(content: &str, content_pos: usize) -> Result<(Operator, Vec<Specifier>), Error> {
    let bytes = content.as_bytes();
    let mut operator = Operator::Simple;
    let mut offset = 0;

    if let Some(&first) = bytes.first() {
        if let Some(op) = Operator::from_byte(first) {
            operator = op;
            offset = 1;
        }
    }

    if offset == bytes.len() {
        return Err(Error::EmptyExpression {
            pos: content_pos + bytes.len(),
        });
    }

    let remaining = &content[offset..];
    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut segment_start = 0;

    loop {
        let comma = remaining[segment_start..]
            .find(',')
            .map(|n| segment_start + n);
        let segment_end = comma.unwrap_or(remaining.len());
        let spec = &remaining[segment_start..segment_end];
        let spec_pos = content_pos + offset + segment_start;

        if spec.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let spec_bytes = spec.as_bytes();
        let modifier_at = spec_bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_at.unwrap_or(spec_bytes.len());

        if name_end == 0 {
            return Err(Error::InvalidName { pos: spec_pos });
        }

        if !is_name_start(spec_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_pos });
        }

        for i in 1..name_end {
            if !is_name_continue(spec_bytes[i]) {
                return Err(Error::InvalidName {
                    pos: spec_pos + i,
                });
            }
        }

        let name = &spec[..name_end];
        let mut explode = false;
        let mut prefix = None;

        if let Some(modifier_index) = modifier_at {
            let suffix = &spec[modifier_index..];
            let suffix_bytes = suffix.as_bytes();

            if suffix_bytes == b"*" {
                explode = true;
            } else if suffix_bytes.first() == Some(&b':')
                && (2..=5).contains(&suffix_bytes.len())
                && suffix_bytes[1].is_ascii_digit()
                && (b'1'..=b'9').contains(&suffix_bytes[1])
                && suffix_bytes[2..].iter().all(u8::is_ascii_digit)
            {
                let mut value = 0usize;
                for &digit in &suffix_bytes[1..] {
                    value = value * 10 + usize::from(digit - b'0');
                }
                if value <= 9999 {
                    prefix = Some(value);
                } else {
                    return Err(Error::InvalidModifier {
                        pos: spec_pos + modifier_index,
                    });
                }
            } else {
                return Err(Error::InvalidModifier {
                    pos: spec_pos + modifier_index,
                });
            }
        }

        if !seen.insert(name.to_owned()) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }

        specs.push(Specifier {
            name: name.to_owned(),
            start: spec_pos,
            explode,
            prefix,
        });

        match comma {
            Some(index) => segment_start = index + 1,
            None => break,
        }
    }

    Ok((operator, specs))
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn uppercase_hex(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%'
            && i + 2 < bytes.len()
            && is_hex(bytes[i + 1])
            && is_hex(bytes[i + 2])
        {
            output.push('%');
            output.push(char::from(uppercase_hex(bytes[i + 1])));
            output.push(char::from(uppercase_hex(bytes[i + 2])));
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_char = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_char {
            output.push(char::from(byte));
        } else {
            output.push('%');
            output.push(char::from(b"0123456789ABCDEF"[(byte >> 4) as usize]));
            output.push(char::from(b"0123456789ABCDEF"[(byte & 0x0f) as usize]));
        }
        i += 1;
    }

    output
}

fn select_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn encode_named(name: &str, value: &str, operator: Operator, reserved: bool) -> String {
    let encoded = encode(value, reserved);
    if operator == Operator::Semicolon && encoded.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={encoded}")
    }
}

fn expand_expression(
    operator: Operator,
    specs: &[Specifier],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let reserved = operator.reserved();
    let mut fields = Vec::new();

    for spec in specs {
        let Some(value) = select_value(&spec.name, vars) else {
            continue;
        };

        if spec.prefix.is_some() && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(text) => {
                let selected = if let Some(limit) = spec.prefix {
                    text.chars().take(limit).collect::<String>()
                } else {
                    text.clone()
                };
                if operator.named_fields() {
                    fields.push(encode_named(&spec.name, &selected, operator, reserved));
                } else {
                    fields.push(encode(&selected, reserved));
                }
            }
            Value::List(items) => {
                if spec.explode {
                    for item in items {
                        if operator.named_fields() {
                            fields.push(encode_named(&spec.name, item, operator, reserved));
                        } else {
                            fields.push(encode(item, reserved));
                        }
                    }
                } else if !items.is_empty() {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if operator.named_fields() {
                        fields.push(encode_named(&spec.name, &joined, operator, reserved));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }

                if spec.explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        if operator == Operator::Semicolon && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else if !entries.is_empty() {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    if operator.named_fields() {
                        fields.push(encode_named(&spec.name, &joined, operator, reserved));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[cursor + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => cursor + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[cursor + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + offset,
                    });
                }

                let content = &template[cursor + 1..close];
                let (operator, specs) = validate_expression(content, cursor + 1)?;
                let expansion = expand_expression(operator, &specs, vars)?;
                output.push_str(&expansion);
                cursor = close + 1;
            }
            _ => {
                let next = template[cursor..]
                    .find(['{', '}'])
                    .map(|offset| cursor + offset)
                    .unwrap_or(bytes.len());
                output.push_str(&template[cursor..next]);
                cursor = next;
            }
        }
    }

    Ok(output)
}
