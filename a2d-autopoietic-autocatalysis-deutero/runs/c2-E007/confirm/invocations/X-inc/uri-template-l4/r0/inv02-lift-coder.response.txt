#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semi,
    Question,
    Amp,
}

impl Operator {
    fn from_byte(b: u8) -> Option<Self> {
        match b {
            b'+' => Some(Self::Plus),
            b'#' => Some(Self::Hash),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semi),
            b'?' => Some(Self::Question),
            b'&' => Some(Self::Amp),
            _ => None,
        }
    }

    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Plus | Self::Hash => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semi => ';',
            Self::Question | Self::Amp => '&',
        }
    }

    fn prefix(self) -> &'static str {
        match self {
            Self::Simple | Self::Plus => "",
            Self::Hash => "#",
            Self::Dot => ".",
            Self::Slash => "/",
            Self::Semi => ";",
            Self::Question => "?",
            Self::Amp => "&",
        }
    }

    fn named(self) -> bool {
        matches!(self, Self::Semi | Self::Question | Self::Amp)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Plus | Self::Hash)
    }
}

#[derive(Clone)]
struct Spec {
    name: String,
    start: usize,
    explode: bool,
    prefix: Option<usize>,
}

struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    specs: Vec<Spec>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let mut operator = Operator::Simple;

    if content_start < end {
        if let Some(op) = Operator::from_byte(bytes[content_start]) {
            operator = op;
            content_start += 1;
        } else if matches!(bytes[content_start], b'=' | b'!' | b'@' | b'|' | b';' | b'?' | b'&') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut part_start = content_start;

    loop {
        let part_end = bytes[part_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| part_start + offset)
            .unwrap_or(end);

        if part_start == part_end {
            return Err(Error::EmptyVariable { pos: part_start });
        }

        let modifier_pos = (part_start..part_end)
            .find(|&i| matches!(bytes[i], b'*' | b':'))
            .unwrap_or(part_end);

        if modifier_pos == part_start {
            return Err(Error::InvalidName { pos: part_start });
        }

        for i in part_start..modifier_pos {
            let valid = if i == part_start {
                is_name_start(bytes[i])
            } else {
                is_name_continue(bytes[i])
            };
            if !valid {
                return Err(Error::InvalidName { pos: i });
            }
        }

        let suffix = &bytes[modifier_pos..part_end];
        let (explode, prefix) = if suffix.is_empty() {
            (false, None)
        } else if suffix == b"*" {
            (true, None)
        } else if suffix.first() == Some(&b':') {
            let digits = &suffix[1..];
            let valid = (1..=4).contains(&digits.len())
                && digits[0].is_ascii_digit()
                && digits[0] != b'0'
                && digits.iter().all(u8::is_ascii_digit);
            if !valid {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            let n = digits.iter().fold(0usize, |acc, digit| {
                acc * 10 + (digit - b'0') as usize
            });
            if !(1..=9999).contains(&n) {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            (false, Some(n))
        } else {
            return Err(Error::InvalidModifier { pos: modifier_pos });
        };

        let name = template[part_start..modifier_pos].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: part_start });
        }
        specs.push(Spec {
            name,
            start: part_start,
            explode,
            prefix,
        });

        if part_end == end {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        specs,
    })
}

fn parse_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = bytes[i + 1..]
                    .iter()
                    .position(|&b| b == b'}')
                    .map(|offset| i + 1 + offset);
                let Some(end) = close else {
                    return Err(Error::Unclosed { pos: start });
                };
                if let Some(offset) = bytes[i + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }
                expressions.push(parse_expression(template, start, end)?);
                i = end + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'('
            | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(hi), Some(lo)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(HEX[hi as usize] as char);
                out.push(HEX[lo as usize] as char);
                i += 3;
                continue;
            }
        }
        if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semi && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.operator;
    let mut fields = Vec::new();

    for spec in &expr.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        match value {
            Value::Scalar(s) => {
                let value = if let Some(n) = spec.prefix {
                    s.chars().take(n).collect::<String>()
                } else {
                    s.clone()
                };
                let encoded = encode(&value, op.reserved());
                fields.push(if op.named() {
                    named_field(&spec.name, encoded, op)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if spec.prefix.is_some() {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
                if items.is_empty() {
                    continue;
                }
                if spec.explode {
                    for item in items {
                        let encoded = encode(item, op.reserved());
                        fields.push(if op.named() {
                            named_field(&spec.name, encoded, op)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, op.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if op.named() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if spec.prefix.is_some() {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if entries.is_empty() {
                    continue;
                }
                if spec.explode {
                    for (key, val) in entries {
                        let encoded_key = encode(key, op.reserved());
                        let encoded_val = encode(val, op.reserved());
                        let field = if op == Operator::Semi && encoded_val.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_val}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, op.reserved()));
                        components.push(encode(val, op.reserved()));
                    }
                    let joined = components.join(",");
                    fields.push(if op.named() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::from(op.prefix());
    output.push_str(&fields.join(&op.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expr in expressions {
        output.push_str(&template[cursor..expr.start]);
        output.push_str(&expand_expression(&expr, vars)?);
        cursor = expr.end + 1;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
