#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

struct Expression<'a> {
    operator: Operator,
    specifiers: Vec<Specifier<'a>>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression<'a>(template: &'a str, content_start: usize, close: usize) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut start = content_start;

    let operator = if start < close {
        match bytes[start] {
            b'+' => {
                start += 1;
                Operator::Plus
            }
            b'#' => {
                start += 1;
                Operator::Hash
            }
            b'.' => {
                start += 1;
                Operator::Dot
            }
            b'/' => {
                start += 1;
                Operator::Slash
            }
            b';' => {
                start += 1;
                Operator::Semicolon
            }
            b'?' => {
                start += 1;
                Operator::Question
            }
            b'&' => {
                start += 1;
                Operator::Ampersand
            }
            b'=' | b',' | b'!' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specifiers = Vec::new();
    let mut seen = Vec::<&str>::new();
    let mut spec_start = start;

    loop {
        let mut spec_end = spec_start;
        while spec_end < close && bytes[spec_end] != b',' {
            spec_end += 1;
        }

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let part = &template[spec_start..spec_end];
        let modifier_offset = part.find(['*', ':']).unwrap_or(part.len());
        let name = &part[..modifier_offset];
        let suffix = &part[modifier_offset..];

        if name.is_empty() {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (i, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName { pos: spec_start + i });
            }
        }

        let modifier = if suffix.is_empty() {
            Modifier::None
        } else if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let valid = !digits.is_empty()
                && digits.len() <= 4
                && digits.as_bytes()[0].is_ascii_digit()
                && digits.as_bytes()[0] != b'0'
                && digits.bytes().all(|b| b.is_ascii_digit());
            if !valid {
                return Err(Error::InvalidModifier {
                    pos: spec_start + modifier_offset,
                });
            }
            let n = digits.bytes().fold(0usize, |acc, b| acc * 10 + (b - b'0') as usize);
            if !(1..=9999).contains(&n) {
                return Err(Error::InvalidModifier {
                    pos: spec_start + modifier_offset,
                });
            }
            Modifier::Prefix(n)
        } else {
            return Err(Error::InvalidModifier {
                pos: spec_start + modifier_offset,
            });
        };

        if seen.iter().any(|seen_name| *seen_name == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        specifiers.push(Specifier {
            name,
            modifier,
            pos: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = i + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }
                if let Some(inner) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner,
                    });
                }

                let expression = parse_expression(template, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(input: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            out.push(byte as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
            i += 1;
        }
    }

    out
}

fn separator(operator: Operator) -> char {
    match operator {
        Operator::Simple | Operator::Plus | Operator::Hash => ',',
        Operator::Dot => '.',
        Operator::Slash => '/',
        Operator::Semicolon => ';',
        Operator::Question | Operator::Ampersand => '&',
    }
}

fn prefix(operator: Operator) -> Option<char> {
    match operator {
        Operator::Hash => Some('#'),
        Operator::Dot => Some('.'),
        Operator::Slash => Some('/'),
        Operator::Semicolon => Some(';'),
        Operator::Question => Some('?'),
        Operator::Ampersand => Some('&'),
        Operator::Simple | Operator::Plus => None,
    }
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
            format!("{name}={value}")
        }
        _ => value,
    }
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Operator::Plus | Operator::Hash);
    let mut fields = Vec::<String>::new();

    for specifier in &expression.specifiers {
        let Some(value) = selected_value(specifier.name, vars) else {
            continue;
        };

        if matches!(specifier.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: specifier.pos });
        }

        let explode = matches!(specifier.modifier, Modifier::Explode);
        match value {
            Value::Scalar(scalar) => {
                let value = match specifier.modifier {
                    Modifier::Prefix(n) => scalar.chars().take(n).collect::<String>(),
                    _ => scalar.clone(),
                };
                fields.push(named_field(
                    specifier.name,
                    encode(&value, allow_reserved),
                    operator,
                ));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if explode {
                    for item in items {
                        fields.push(named_field(
                            specifier.name,
                            encode(item, allow_reserved),
                            operator,
                        ));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(named_field(specifier.name, joined, operator));
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: specifier.pos });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(value, allow_reserved);
                        let field = if matches!(operator, Operator::Semicolon)
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, allow_reserved));
                        components.push(encode(value, allow_reserved));
                    }
                    let joined = components.join(",");
                    fields.push(named_field(specifier.name, joined, operator));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    if let Some(prefix) = prefix(operator) {
        out.push(prefix);
    }
    let joiner = separator(operator);
    for (index, field) in fields.iter().enumerate() {
        if index != 0 {
            out.push(joiner);
        }
        out.push_str(field);
    }
    Ok(out)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut out = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        out.push_str(&template[cursor..start]);
        out.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
