use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;

    fn parse_core(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
        let start = *pos;
        while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
            *pos += 1;
        }

        if *pos == start {
            if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        let digits = &bytes[start..*pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in digits {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        Ok(value)
    }

    let major = parse_core(bytes, &mut pos)?;
    for _ in 0..1 {
        if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        if bytes[pos] != b'.' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
    }

    let minor = parse_core(bytes, &mut pos)?;
    if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::MissingCore { at: pos });
    }
    if bytes[pos] != b'.' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core(bytes, &mut pos)?;

    if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    fn parse_identifiers(
        bytes: &[u8],
        start: usize,
        end: usize,
        is_pre: bool,
    ) -> Result<Vec<String>, VersionError> {
        if start == end {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let mut result = Vec::new();
        let mut ident_start = start;

        for i in start..=end {
            if i == end || bytes[i] == b'.' {
                if i == ident_start {
                    return Err(VersionError::EmptyIdentifier { at: ident_start });
                }

                let ident = &bytes[ident_start..i];
                for (offset, &byte) in ident.iter().enumerate() {
                    if !(byte.is_ascii_alphanumeric() || byte == b'-') {
                        return Err(VersionError::InvalidChar {
                            at: ident_start + offset,
                        });
                    }
                }

                if is_pre
                    && ident.len() > 1
                    && ident[0] == b'0'
                    && ident.iter().all(u8::is_ascii_digit)
                {
                    return Err(VersionError::LeadingZero { at: ident_start });
                }

                result.push(String::from_utf8(ident.to_vec()).unwrap());
                ident_start = i + 1;
            }
        }

        Ok(result)
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|n| pre_start + n)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(bytes, pre_start, plus, true)?;
        pos = plus;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(bytes, build_start, bytes.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (left, right) in a.pre.iter().zip(&b.pre) {
                    let ordering = compare_identifiers(left, right);
                    if ordering != Ordering::Equal {
                        return ordering;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut comparators = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        while pos < bytes.len() && matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (operator, operand_offset) = if token.starts_with("<=") {
            (Operator::LessEqual, 2)
        } else if token.starts_with(">=") {
            (Operator::GreaterEqual, 2)
        } else if token.starts_with('<') {
            (Operator::Less, 1)
        } else if token.starts_with('>') {
            (Operator::Greater, 1)
        } else if token.starts_with('=') {
            (Operator::Equal, 1)
        } else {
            (Operator::Equal, 0)
        };

        if operand_offset >= token.len()
            || !token.as_bytes()[operand_offset].is_ascii_digit()
        {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand_start = token_start + operand_offset;
        let version = parse_version(&s[operand_start..token_end]).map_err(|error| {
            RangeError::Version {
                at: operand_start,
                error,
            }
        })?;

        comparators.push(Comparator { operator, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && comparator.version.major == v.major
                    && comparator.version.minor == v.minor
                    && comparator.version.patch == v.patch
            })
        {
            return false;
        }

        arm.iter().all(|comparator| {
            let ordering = compare_versions(v, &comparator.version);
            match comparator.operator {
                Operator::Equal => ordering == Ordering::Equal,
                Operator::Less => ordering == Ordering::Less,
                Operator::LessEqual => ordering != Ordering::Greater,
                Operator::Greater => ordering == Ordering::Greater,
                Operator::GreaterEqual => ordering != Ordering::Less,
            }
        })
    })
}
