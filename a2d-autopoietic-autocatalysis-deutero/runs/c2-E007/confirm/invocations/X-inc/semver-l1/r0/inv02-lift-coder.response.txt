use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
        return Err(VersionError::MissingCore { at: *pos });
    }
    if !bytes[*pos].is_ascii_digit() {
        return Err(VersionError::InvalidChar { at: *pos });
    }

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in digits {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    validate_pre_numbers: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut identifiers = Vec::new();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let identifier = &s[id_start..i];
        if validate_pre_numbers
            && identifier.bytes().all(|b| b.is_ascii_digit())
            && identifier.len() > 1
            && identifier.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        identifiers.push(identifier.to_owned());
        if i == end {
            break;
        }
        id_start = i + 1;
    }

    Ok(identifiers)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if is_core_terminator(s.as_bytes().get(pos).copied()) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if is_core_terminator(s.as_bytes().get(pos).copied()) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    let bytes = s.as_bytes();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let pre_end = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map_or(bytes.len(), |offset| pre_start + offset);
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn compare_pre_identifier(a: &str, b: &str) -> Ordering {
    match (is_numeric_identifier(a), is_numeric_identifier(b)) {
        (true, true) => compare_numeric_identifiers(a, b),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (left, right) in a.pre.iter().zip(&b.pre) {
                    let order = compare_pre_identifier(left, right);
                    if order != Ordering::Equal {
                        return order;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        let (operator, operator_len) = match bytes[pos] {
            b'=' => (Operator::Equal, 1),
            b'<' if bytes.get(pos + 1) == Some(&b'=') => (Operator::LessEqual, 2),
            b'<' => (Operator::Less, 1),
            b'>' if bytes.get(pos + 1) == Some(&b'=') => (Operator::GreaterEqual, 2),
            b'>' => (Operator::Greater, 1),
            _ => (Operator::Equal, 0),
        };

        let operand_start = pos + operator_len;
        if operand_start >= bytes.len() || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let mut token_end = operand_start;
        while token_end < bytes.len() && !is_range_whitespace(bytes[token_end]) {
            token_end += 1;
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { operator, version });
        pos = token_end;
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { comparators })
}

fn comparator_matches(candidate: &Version, comparator: &Comparator) -> bool {
    let order = compare_versions(candidate, &comparator.version);
    match comparator.operator {
        Operator::Equal => order == Ordering::Equal,
        Operator::Less => order == Ordering::Less,
        Operator::LessEqual => order != Ordering::Greater,
        Operator::Greater => order == Ordering::Greater,
        Operator::GreaterEqual => order != Ordering::Less,
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let admits_prerelease = r.comparators.iter().any(|comparator| {
        !comparator.version.pre.is_empty()
            && (comparator.version.major, comparator.version.minor, comparator.version.patch)
                == (v.major, v.minor, v.patch)
    });

    if !v.pre.is_empty() && !admits_prerelease {
        return false;
    }

    r.comparators
        .iter()
        .all(|comparator| comparator_matches(v, comparator))
}
