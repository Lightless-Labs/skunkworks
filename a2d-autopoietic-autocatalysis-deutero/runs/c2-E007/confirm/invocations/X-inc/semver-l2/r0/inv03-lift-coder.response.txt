use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn missing_byte(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while matches!(byte(s, *pos), Some(b'0'..=b'9')) {
        *pos += 1;
    }
    if *pos == start {
        return match byte(s, start) {
            b if missing_byte(b) => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn valid_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let mut out = Vec::new();
    let mut ident_start = start;
    let mut i = start;
    while i <= end {
        if i == end || byte(s, i) == Some(b'.') {
            if i == ident_start {
                return Err(VersionError::EmptyIdentifier { at: ident_start });
            }
            let ident = &s[ident_start..i];
            if is_pre && ident.bytes().all(|b| b.is_ascii_digit())
                && ident.len() > 1
                && ident.as_bytes()[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
            out.push(ident.to_string());
            ident_start = i + 1;
        } else if !valid_ident_char(byte(s, i).unwrap()) {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    for _ in 0..2 {
        if byte(s, pos) != Some(b'.') {
            return match byte(s, pos) {
                b if missing_byte(b) => Err(VersionError::MissingCore { at: pos }),
                _ => Err(VersionError::InvalidChar { at: pos }),
            };
        }
        pos += 1;
        if _ == 0 {
            // Parsed below after confirming the required separator.
        }
    }
    // The loop above only checks separators; parse fields in a dedicated pass.
    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    if byte(s, pos) != Some(b'.') {
        return match byte(s, pos) {
            b if missing_byte(b) => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let minor = parse_core_field(s, &mut pos)?;
    if byte(s, pos) != Some(b'.') {
        return match byte(s, pos) {
            b if missing_byte(b) => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let patch = parse_core_field(s, &mut pos)?;
    match byte(s, pos) {
        None => Ok(Version { major, minor, patch, pre: Vec::new(), build: Vec::new() }),
        Some(b'-' | b'+') => {}
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if byte(s, pos) == Some(b'-') {
        let pre_start = pos + 1;
        let plus = s[pre_start..].find('+').map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }
    if byte(s, pos) == Some(b'+') {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let cmp = match (xn, yn) {
            (true, true) => compare_numeric_str(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if cmp != Ordering::Equal {
            return cmp;
        }
    }
    a.len().cmp(&b.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tuple_version(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version { major, minor, patch, pre, build: Vec::new() }
}

fn increment_version(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = tuple_version(v.major, v.minor, v.patch, vec!["0".to_string()]);
    match field {
        0 => {
            out.major = out.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            out.minor = 0;
            out.patch = 0;
        }
        1 => {
            out.minor = out.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            out.patch = 0;
        }
        _ => out.patch = out.patch.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    Ok(out)
}

#[derive(Clone)]
struct Partial {
    version: Version,
    k: usize,
    full: bool,
    wildcard_only: bool,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_sign = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let is_full = has_sign || (fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")));
    if is_full {
        return parse_version(s)
            .map(|v| Partial { version: v, k: 3, full: true, wildcard_only: false })
            .map_err(|error| RangeError::Version { at, error });
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.bytes().all(|b| b.is_ascii_digit()) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    let k = nums.len();
    let wildcard_only = k == 0;
    let major = *nums.get(0).unwrap_or(&0);
    let minor = *nums.get(1).unwrap_or(&0);
    let patch = *nums.get(2).unwrap_or(&0);
    Ok(Partial { version: tuple_version(major, minor, patch, Vec::new()), k, full: false, wildcard_only })
}

fn push_cmp(out: &mut Vec<Comparator>, op: Op, v: Version, anchor: bool) {
    out.push(Comparator { op, version: v, anchor });
}

fn expand_bare(p: Partial, at: usize, out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    if p.full {
        let anchor = !p.version.pre.is_empty();
        push_cmp(out, Op::Eq, p.version, anchor);
    } else if p.k == 0 {
    } else {
        push_cmp(out, Op::Ge, p.version.clone(), false);
        if p.k <= 2 {
            let field = if p.k == 1 { 0 } else { 1 };
            push_cmp(out, Op::Lt, increment_version(&p.version, field, at)?, false);
        }
    }
    Ok(())
}

fn token_starting_char(b: u8) -> bool {
    b.is_ascii_digit() || matches!(b, b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'=')
}

fn parse_arm(tokens: &[(usize, &str)], out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    if let Some((idx, _)) = tokens.iter().enumerate().find(|(_, (_, t))| *t == "-") {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: tokens[idx].0 });
        }
        let (lo_at, lo_s) = tokens[0];
        let (hi_at, hi_s) = tokens[2];
        let lo = parse_partial(lo_s, lo_at)?;
        let hi = parse_partial(hi_s, hi_at)?;
        if lo.k == 0 { return Err(RangeError::Syntax { at: lo_at }); }
        if hi.k == 0 { return Err(RangeError::Syntax { at: hi_at }); }
        push_cmp(out, Op::Ge, lo.version.clone(), !lo.version.pre.is_empty());
        if hi.full {
            push_cmp(out, Op::Le, hi.version.clone(), !hi.version.pre.is_empty());
        } else {
            let upper = match hi.k {
                1 => increment_version(&hi.version, 0, hi_at)?,
                2 => increment_version(&hi.version, 1, hi_at)?,
                _ => hi.version.clone(),
            };
            if hi.k == 3 {
                push_cmp(out, Op::Le, upper, false);
            } else {
                push_cmp(out, Op::Lt, upper, false);
            }
        }
        return Ok(());
    }

    for &(at, token) in tokens {
        if !token.as_bytes().first().copied().map(token_starting_char).unwrap_or(false) {
            return Err(RangeError::Syntax { at });
        }
        if let Some(rest) = token.strip_prefix('^').or_else(|| token.strip_prefix('~')) {
            let caret = token.starts_with('^');
            let op_at = at + 1;
            let p = parse_partial(rest, op_at)?;
            if p.k == 0 { return Err(RangeError::Syntax { at }); }
            let anchor = !p.version.pre.is_empty();
            push_cmp(out, Op::Ge, p.version.clone(), anchor);
            let field = if caret {
                if p.version.major > 0 || p.k == 1 { 0 }
                else if p.version.minor > 0 || p.k == 2 { 1 }
                else { 2 }
            } else if p.k == 1 { 0 } else { 1 };
            push_cmp(out, Op::Lt, increment_version(&p.version, field, at)?, false);
        } else if token.starts_with(['<', '>', '=']) {
            let (op, n) = if token.starts_with("<=") { (Op::Le, 2) }
                else if token.starts_with(">=") { (Op::Ge, 2) }
                else if token.starts_with('<') { (Op::Lt, 1) }
                else if token.starts_with('>') { (Op::Gt, 1) }
                else { (Op::Eq, 1) };
            let operand = &token[n..];
            let operand_at = at + n;
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = parse_version(operand)
                .map_err(|error| RangeError::Version { at: operand_at, error })?;
            push_cmp(out, op, v.clone(), !v.pre.is_empty());
        } else {
            let p = parse_partial(token, at)?;
            expand_bare(p, at, out)?;
        }
    }
    Ok(())
}

fn parse_tokens(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let mut tokens = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && ws(s.as_bytes()[i]) { i += 1; }
        if i >= end { break; }
        let token_start = i;
        while i < end && !ws(s.as_bytes()[i]) { i += 1; }
        tokens.push((token_start, &s[token_start..i]));
    }
    tokens
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut first = 0;
    while first < bytes.len() && ws(bytes[first]) { first += 1; }
    if first == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    while i + 1 < bytes.len() {
        if bytes[i] == b'|' && bytes[i + 1] == b'|' {
            let before_ws = i > 0 && ws(bytes[i - 1]);
            let after = i + 2;
            let after_ws = after < bytes.len() && ws(bytes[after]);
            if before_ws && after_ws {
                let tokens = parse_tokens(s, arm_start, i);
                if tokens.is_empty() {
                    return Err(RangeError::Empty { at: i });
                }
                let mut comparators = Vec::new();
                parse_arm(&tokens, &mut comparators)?;
                arms.push(comparators);
                arm_start = after;
                while arm_start < bytes.len() && ws(bytes[arm_start]) { arm_start += 1; }
                i = arm_start;
                continue;
            }
        }
        i += 1;
    }
    let tokens = parse_tokens(s, arm_start, bytes.len());
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: bytes.len() });
    }
    let mut comparators = Vec::new();
    parse_arm(&tokens, &mut comparators)?;
    arms.push(comparators);
    Ok(Range { arms })
}

fn satisfies(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &[Comparator]) -> bool {
    if !v.pre.is_empty()
        && !arm.iter().any(|c| c.anchor && c.version.major == v.major && c.version.minor == v.minor && c.version.patch == v.patch)
    {
        return false;
    }
    arm.iter().all(|c| satisfies(v, c))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        match best {
            None => best = Some(i),
            Some(j) => {
                let cmp = compare_versions(v, &parsed[j]);
                if (want_max && cmp == Ordering::Greater) || (!want_max && cmp == Ordering::Less) {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
