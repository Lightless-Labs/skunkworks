use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(match bytes.get(*pos).copied() {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: *pos },
            _ => VersionError::InvalidChar { at: *pos },
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &b in &bytes[start..*pos] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn valid_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;
    let mut i = start;
    loop {
        while i < end && bytes[i] != b'.' {
            if !valid_ident_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let part = &s[ident_start..i];
        if is_pre
            && part.len() > 1
            && part.as_bytes().iter().all(|b| is_digit(*b))
            && part.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(part.to_string());
        if i == end {
            break;
        }
        i += 1;
        ident_start = i;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(match s.as_bytes().get(pos).copied() {
                None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }
        pos += 1;
        if _ == 0 { /* placeholder */ }
    }
    // Parse minor and patch explicitly to preserve left-to-right field errors.
    let mut p = 0;
    let major = parse_core_field(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(match s.as_bytes().get(p).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p },
            _ => VersionError::InvalidChar { at: p },
        });
    }
    p += 1;
    let minor = parse_core_field(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(match s.as_bytes().get(p).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p },
            _ => VersionError::InvalidChar { at: p },
        });
    }
    p += 1;
    let patch = parse_core_field(s, &mut p)?;
    if p < s.len() && s.as_bytes()[p] != b'-' && s.as_bytes()[p] != b'+' {
        return Err(VersionError::InvalidChar { at: p });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let start = p + 1;
        let plus = s[start..].find('+').map(|x| start + x);
        let end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, start, end, true)?;
        p = end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let start = p + 1;
        build = parse_identifiers(s, start, s.len(), false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    match (numeric_identifier(a), numeric_identifier(b)) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let cmp = compare_identifier(x, y);
                    if cmp != Ordering::Equal {
                        return cmp;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    at: usize,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str, base: usize) -> Vec<Token> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !ws(bytes[i]) {
            i += 1;
        }
        out.push(Token { text: s[start..i].to_string(), at: base + start });
    }
    out
}

fn version_operand(text: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(text).map_err(|error| RangeError::Version { at, error })
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator { op, version, anchor });
}

fn accepts_cmp(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn upper_version(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: vec!["0".into()], build: Vec::new() }
}

fn partial(text: &str, at: usize) -> Result<(Vec<Option<u64>>, bool), RangeError> {
    let fields: Vec<&str> = text.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut wildcard = false;
    let mut nums = Vec::new();
    for f in fields {
        if f == "*" || f == "x" || f == "X" {
            wildcard = true;
            nums.push(None);
        } else if wildcard || !f.as_bytes().iter().all(|b| is_digit(*b)) || (f.len() > 1 && f.starts_with('0')) {
            return Err(RangeError::Syntax { at });
        } else {
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(Some(n));
        }
    }
    Ok((nums, wildcard))
}

fn full_or_partial(text: &str, at: usize) -> Result<(Version, usize, bool), RangeError> {
    let fields: Vec<&str> = text.split('.').collect();
    let is_full = text.contains(['-', '+'])
        || (fields.len() >= 3 && !fields.iter().any(|f| *f == "*" || *f == "x" || *f == "X"));
    if is_full {
        let v = version_operand(text, at)?;
        return Ok((v, 3, true));
    }
    let (nums, _) = partial(text, at)?;
    let k = nums.iter().take_while(|x| x.is_some()).count();
    let mut vals = [0u64; 3];
    for i in 0..k {
        vals[i] = nums[i].unwrap();
    }
    Ok((Version { major: vals[0], minor: vals[1], patch: vals[2], pre: Vec::new(), build: Vec::new() }, k, false))
}

fn increment_bound(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut major = v.major;
    let mut minor = v.minor;
    let mut patch = v.patch;
    match field {
        0 => { major = major.checked_add(1).ok_or(RangeError::Syntax { at })?; minor = 0; patch = 0; }
        1 => { minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?; patch = 0; }
        _ => { patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?; }
    }
    Ok(upper_version(major, minor, patch))
}

fn parse_arm(tokens: &[Token], arm: &mut Arm) -> Result<(), RangeError> {
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    if let Some(minus) = tokens.iter().position(|t| t.text == "-") {
        if tokens.len() != 3 || minus != 1 {
            return Err(RangeError::Syntax { at: tokens[minus].at });
        }
        let (lo, lk, lf) = full_or_partial(&tokens[0].text, tokens[0].at)?;
        let (hi, hk, hf) = full_or_partial(&tokens[2].text, tokens[2].at)?;
        if lk == 0 || hk == 0 {
            return Err(RangeError::Syntax { at: if lk == 0 { tokens[0].at } else { tokens[2].at } });
        }
        add_cmp(arm, Op::Ge, lo.clone(), true);
        if hf {
            add_cmp(arm, Op::Le, hi, true);
        } else {
            let ub = increment_bound(&hi, if hk == 1 { 0 } else { 1 }, tokens[2].at)?;
            add_cmp(arm, Op::Lt, ub, false);
        }
        return Ok(());
    }

    for token in tokens {
        let t = token.text.as_str();
        let (op, operand, offset) = if t.starts_with(">=") {
            (Some(Op::Ge), &t[2..], 2)
        } else if t.starts_with("<=") {
            (Some(Op::Le), &t[2..], 2)
        } else if t.starts_with('>') {
            (Some(Op::Gt), &t[1..], 1)
        } else if t.starts_with('<') {
            (Some(Op::Lt), &t[1..], 1)
        } else if t.starts_with('=') {
            (Some(Op::Eq), &t[1..], 1)
        } else { (None, t, 0) };

        if let Some(op) = op {
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: token.at });
            }
            let v = version_operand(operand, token.at + offset)?;
            add_cmp(arm, op, v, true);
            continue;
        }

        if !(t.as_bytes()[0].is_ascii_digit() || matches!(t.as_bytes()[0], b'*' | b'x' | b'X' | b'^' | b'~')) {
            return Err(RangeError::Syntax { at: token.at });
        }
        if t.starts_with('^') || t.starts_with('~') {
            let opchar = t.as_bytes()[0];
            let operand = &t[1..];
            let at = token.at + 1;
            let (lo, k, full) = full_or_partial(operand, at)?;
            if k == 0 { return Err(RangeError::Syntax { at: token.at }); }
            add_cmp(arm, Op::Ge, lo.clone(), true);
            let field = if opchar == b'^' {
                if lo.major > 0 || k == 1 { 0 } else if lo.minor > 0 || k == 2 { 1 } else { 2 }
            } else if k == 1 { 0 } else { 1 };
            let ub = increment_bound(&lo, field, token.at)?;
            add_cmp(arm, Op::Lt, ub, false);
            let _ = full;
            continue;
        }
        let (v, k, full) = full_or_partial(t, token.at)?;
        if full {
            add_cmp(arm, Op::Eq, v, true);
        } else if k == 0 {
            // Wildcards match any stable release.
        } else {
            add_cmp(arm, Op::Ge, v.clone(), false);
            let ub = increment_bound(&v, if k == 1 { 0 } else { 1 }, token.at)?;
            add_cmp(arm, Op::Lt, ub, false);
        }
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s, 0);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if i == start {
                return Err(RangeError::Empty { at: tokens[i].at });
            }
            let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
            parse_arm(&tokens[start..i], &mut arm)?;
            arms.push(arm);
            start = i + 1;
        }
        i += 1;
    }
    if start == tokens.len() {
        let at = s.len();
        return Err(RangeError::Empty { at });
    }
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    parse_arm(&tokens[start..], &mut arm)?;
    arms.push(arm);
    let _ = bytes;
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.iter().any(|&(a, b, c)| (a, b, c) == (v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.comparators.iter().all(|c| accepts_cmp(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, text) in candidates.iter().enumerate() {
        let version = parse_version(text).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }
    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(version, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
