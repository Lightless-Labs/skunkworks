#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    occupied_until: Option<(u64, u32)>,
}

fn next_event_time(
    now: u64,
    customers: &[Customer],
    arrived: &[bool],
    servers: &[ServerState],
) -> Option<u64> {
    let next_arrival = customers
        .iter()
        .enumerate()
        .filter(|(i, _)| !arrived[*i])
        .map(|(_, c)| c.arrival)
        .min();

    let next_open = servers
        .iter()
        .filter(|s| !s.opened)
        .map(|s| s.server.opens)
        .min();

    let next_finish = servers
        .iter()
        .filter_map(|s| s.occupied_until.map(|(time, _)| time))
        .min();

    [next_arrival, next_open, next_finish]
        .into_iter()
        .flatten()
        .filter(|&time| time > now)
        .min()
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied_until: None,
        })
        .collect();

    let mut arrived = vec![false; customers.len()];
    let mut next_arrival = 0usize;
    let mut waiting: Vec<usize> = Vec::new();
    let mut events = Vec::new();

    let mut now = arrival_order
        .first()
        .map(|&i| customers[i].arrival)
        .into_iter()
        .chain(servers.iter().map(|s| s.opens))
        .min()
        .unwrap_or(0);

    loop {
        let mut finishing: Vec<(usize, u32, u64)> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                state
                    .occupied_until
                    .filter(|(time, _)| *time == now)
                    .map(|(time, customer)| (i, customer, time))
            })
            .collect();
        finishing.sort_by_key(|(i, _, _)| states[*i].server.id);
        for (i, customer, time) in finishing {
            let server = states[i].server.id;
            events.push(Event::Finish {
                time,
                customer,
                server,
            });
            states[i].occupied_until = None;
            states[i].idle_since = time;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter(|(_, state)| !state.opened && state.server.opens == now)
            .map(|(i, _)| i)
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[i].server.id,
            });
        }

        while next_arrival < arrival_order.len()
            && customers[arrival_order[next_arrival]].arrival == now
        {
            let index = arrival_order[next_arrival];
            let customer = &customers[index];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            arrived[index] = true;
            waiting.push(index);
            next_arrival += 1;
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index].server;

            let Some(after_service) = now.checked_add(customer.service) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };
            let Some(finish) = after_service.checked_add(server.overhead) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
                states[server_index].occupied_until = None;
            } else {
                states[server_index].occupied_until = Some((finish, customer.id));
            }
        }

        if next_arrival == customers.len()
            && states.iter().all(|state| state.opened)
            && waiting.is_empty()
            && states.iter().all(|state| state.occupied_until.is_none())
        {
            break;
        }

        let Some(next) = next_event_time(now, customers, &arrived, &states) else {
            break;
        };
        now = next;
    }

    Ok(events)
}
