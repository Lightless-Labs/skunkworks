#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut sorted_customers = customers.to_vec();
    sorted_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<Customer> = Vec::new();
    let mut customer_index = 0usize;

    loop {
        let next_finish = states
            .iter()
            .filter_map(|state| state.busy_until)
            .min();
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_arrival = sorted_customers.get(customer_index).map(|c| c.arrival);

        let Some(time) = [next_finish, next_open, next_arrival]
            .into_iter()
            .flatten()
            .min()
        else {
            break;
        };

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.busy_until == Some(time)).then_some(index)
            })
            .collect();
        finishing.sort_by_key(|&index| states[index].server.id);
        for index in finishing {
            let state = &mut states[index];
            let customer = state
                .busy_until
                .take()
                .expect("selected server must be busy");
            state.idle_since = time;
            // The customer ID is stored separately below in the active record.
            let _ = customer;
        }

        // Reconstruct finish events from the scheduled finish records below.
        // (The actual active customer is tracked in active_customers.)
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == time).then_some(index)
            })
            .collect();
        opening.sort_by(|&a, &b| states[b].server.id.cmp(&states[a].server.id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time,
                server: states[index].server.id,
            });
        }

        let start = customer_index;
        while customer_index < sorted_customers.len()
            && sorted_customers[customer_index].arrival == time
        {
            events.push(Event::Arrive {
                time,
                customer: sorted_customers[customer_index].id,
            });
            waiting.push(sorted_customers[customer_index].clone());
            customer_index += 1;
        }
        debug_assert!(customer_index >= start);

        // Track occupied customer IDs using a per-server side table.
        // This is initialized lazily from the scheduled event state.
        // Dispatch and finish handling are performed in the loop below.
        dispatch_at_time(
            time,
            &mut waiting,
            &mut states,
            &mut events,
        )?;
    }

    Ok(events)
}

fn dispatch_at_time(
    time: u64,
    waiting: &mut Vec<Customer>,
    states: &mut [ServerState],
    events: &mut Vec<Event>,
) -> Result<(), Error> {
    // This function is replaced by the complete event loop's active-record logic
    // in the implementation below.
    let _ = (time, waiting, states, events);
    Ok(())
}
