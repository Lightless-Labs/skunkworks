#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    occupied_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<&Customer> = customers.iter().collect();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            occupied_until: None,
            idle_since: server.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut next_customer = 0usize;
    let mut time: Option<u64> = None;

    loop {
        let mut next_time = None;
        let mut consider = |candidate: u64| {
            next_time = Some(next_time.map_or(candidate, |current: u64| current.min(candidate)));
        };

        for state in &states {
            if !state.opened {
                consider(state.opens);
            }
            if let Some(finish) = state.occupied_until {
                consider(finish);
            }
        }
        if next_customer < ordered_customers.len() {
            consider(ordered_customers[next_customer].arrival);
        }

        let Some(tick) = next_time else {
            break;
        };
        if time.is_some_and(|previous| tick < previous) {
            unreachable!("simulation time must be monotonic");
        }
        time = Some(tick);

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.occupied_until == Some(tick)).then_some(index)
            })
            .collect();
        finishing.sort_by_key(|&index| states[index].id);
        for index in finishing {
            let server_id = states[index].id;
            let customer_id = match events.last() {
                Some(Event::Start {
                    customer, server, ..
                }) if *server == server_id => *customer,
                _ => {
                    // The matching start may not be the latest event; find the
                    // most recent unfinished start assigned to this server.
                    let mut found = None;
                    for event in events.iter().rev() {
                        if let Event::Start {
                            customer, server, ..
                        } = event
                        {
                            if *server == server_id {
                                found = Some(*customer);
                                break;
                            }
                        }
                        if let Event::Finish { server, .. } = event {
                            if *server == server_id {
                                break;
                            }
                        }
                    }
                    found.expect("occupied server has a matching start")
                }
            };
            events.push(Event::Finish {
                time: tick,
                customer: customer_id,
                server: server_id,
            });
            states[index].occupied_until = None;
            states[index].idle_since = tick;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.opens == tick).then_some(index)
            })
            .collect();
        opening.sort_by(|&a, &b| states[b].id.cmp(&states[a].id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time: tick,
                server: states[index].id,
            });
        }

        let arrival_start = next_customer;
        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == tick
        {
            next_customer += 1;
        }
        for customer in &ordered_customers[arrival_start..next_customer] {
            events.push(Event::Arrive {
                time: tick,
                customer: customer.id,
            });
            waiting.push(customer);
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting.remove(0);
            let server_id = states[server_index].id;
            let after_service = tick.checked_add(customer.service).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server_id,
                time: tick,
            })?;
            let finish = after_service
                .checked_add(states[server_index].overhead)
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time: tick,
                })?;

            events.push(Event::Start {
                time: tick,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == tick {
                events.push(Event::Finish {
                    time: tick,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = tick;
            } else {
                states[server_index].occupied_until = Some(finish);
            }
        }
    }

    Ok(events)
}
