        .iter()
        .enumerate()
        .filter(|(i, _)| !arrived[*i])

        .min();

        .flatten()
        .filter(|&time| time > now)

            unreachable!("simulation time must be monotonic");
        }
        time = Some(tick);

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.occupied_until == Some(tick)).then_some(index)
            })
            .collect();
        finishing.sort_by_key(|&index| states[index].id);
        for index in finishing {
            let server_id = states[index].id;
            let customer_id = match events.last() {
                Some(Event::Start {
                    customer, server, ..
                }) if *server == server_id => *customer,
                _ => {
                    // The matching start may not be the latest event; find the
                    // most recent unfinished start assigned to this server.
                    let mut found = None;
                    for event in events.iter().rev() {
                        if let Event::Start {
                            customer, server, ..
                        } = event
                        {
                            if *server == server_id {
                                found = Some(*customer);
                                break;
                            }
                        }
                        if let Event::Finish { server, .. } = event {
                            if *server == server_id {
                                break;
                            }
                        }
                    }
                    found.expect("occupied server has a matching start")
                }
            };
            events.push(Event::Finish {
                time: tick,
                customer: customer_id,
                server: server_id,
            });
            states[index].occupied_until = None;
            states[index].idle_since = tick;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.opens == tick).then_some(index)
            })
            .collect();
        opening.sort_by(|&a, &b| states[b].id.cmp(&states[a].id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time: tick,
                server: states[index].id,
            });
        }

        let arrival_start = next_customer;
        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == tick
        {
            next_customer += 1;
        }
        for customer in &ordered_customers[arrival_start..next_customer] {
            events.push(Event::Arrive {
                time: tick,
                customer: customer.id,
            });
            waiting.push(customer);
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting.remove(0);
            let server_id = states[server_index].id;
            let after_service = tick.checked_add(customer.service).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server_id,
                time: tick,
            })?;
            let finish = after_service
                .checked_add(states[server_index].overhead)
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time: tick,
                })?;

            events.push(Event::Start {
                time: tick,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == tick {
                events.push(Event::Finish {
                    time: tick,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = tick;
            } else {
                states[server_index].occupied_until = Some(finish);
            }
        }
    }

    Ok(events)
}
