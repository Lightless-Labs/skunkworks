```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut event_order: Vec<usize> = (0..jobs.len()).collect();
    event_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut ready: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completion_times: Vec<Option<u64>> = vec![None; jobs.len()];

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut next_event = 0usize;
    let mut last_dispatched_id: Option<u32> = None;
    let mut completed_count = 0usize;

    fn admit_events(
        jobs: &[Job],
        event_order: &[usize],
        next_event: &mut usize,
        time: u64,
        ready: &mut [std::collections::VecDeque<usize>],
        completion_times: &mut [Option<u64>],
        completed_count: &mut usize,
    ) {
        while *next_event < event_order.len() {
            let index = event_order[*next_event];
            let job = &jobs[index];
            if u64::from(job.arrival) > time {
                break;
            }

            *next_event += 1;
            if job.ticks == 0 {
                completion_times[index] = Some(u64::from(job.arrival));
                *completed_count += 1;
            } else {
                ready[job.level].push_back(index);
            }
        }
    }

    while completed_count < jobs.len() {
        admit_events(
            jobs,
            &event_order,
            &mut next_event,
            time,
            &mut ready,
            &mut completion_times,
            &mut completed_count,
        );

        let selected_level = ready.iter().position(|queue| !queue.is_empty());

        let Some(selected_level) = selected_level else {
            if next_event >= event_order.len() {
                break;
            }

            let next_time = u64::from(jobs[event_order[next_event]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let job_index = ready[selected_level]
            .pop_front()
            .expect("selected queue must be nonempty");
        let job_id = jobs[job_index].id;

        if last_dispatched_id.is_some_and(|last_id| last_id != job_id) {
            time += u64::from(switch_cost);
        }

        admit_events(
            jobs,
            &event_order,
            &mut next_event,
            time,
            &mut ready,
            &mut completion_times,
            &mut completed_count,
        );

        let start = time;
        let consumed = remaining[job_index].min(quanta[selected_level]);
        time += u64::from(consumed);
        remaining[job_index] -= consumed;
        let finished = remaining[job_index] == 0;

        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished,
            level: selected_level,
        });

        admit_events(
            jobs,
            &event_order,
            &mut next_event,
            time,
            &mut ready,
            &mut completion_times,
            &mut completed_count,
        );

        if finished {
            completion_times[job_index] = Some(time);
            completed_count += 1;
        } else {
            levels[job_index] = (selected_level + 1).min(quanta.len() - 1);
            ready[levels[job_index]].push_back(job_index);
        }

        last_dispatched_id = Some(job_id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            (
                job.id,
                completion_times[index].expect("every job must complete"),
            )
        })
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
```
