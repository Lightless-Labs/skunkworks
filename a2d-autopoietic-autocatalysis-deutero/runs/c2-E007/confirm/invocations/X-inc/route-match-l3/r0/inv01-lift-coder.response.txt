#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn name_valid(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1usize;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|p| start + p)
            .unwrap_or(end);
        if next == start {
            return Err(start);
        }
        let raw = &input[start..next];

        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !name_valid(name) || !names.insert(name.to_owned()) {
                return Err(start);
            }
            if marker == b'*' && next != end {
                return Err(start);
            }
            if marker == b':' {
                segments.push(Segment::Param(name.to_owned()));
            } else {
                segments.push(Segment::Wildcard(name.to_owned()));
            }
        } else {
            let decoded = decode_segment(raw, start)?;
            segments.push(Segment::Static(decoded));
        }
        start = next + 1;
    }

    Ok((segments, trailing))
}

fn parse_request(input: &str) -> Result<ParsedPath, usize> {
    let head_end = input
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(input.len());
    let head = &input[..head_end];
    let (parts, trailing) = parse_path(head, false)?;
    let segments = parts
        .into_iter()
        .map(|s| match s {
            Segment::Static(v) => v,
            _ => unreachable!(),
        })
        .collect();
    Ok(ParsedPath { segments, trailing })
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;

        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let parsed = parse_request(path).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_has_route = method == "HEAD"
            && candidates
                .iter()
                .any(|(route, _)| route.method == "HEAD");
        let effective_method = if method == "HEAD" && !head_has_route {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == effective_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                a.rank
                    .cmp(&b.rank)
                    .then_with(|| b.id.cmp(&a.id))
            })
            .expect("eligible is nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && winner.trailing != parsed.trailing
            && path_head(path).len() > 1
        {
            let mut head = path_head(path).to_owned();
            if winner.trailing {
                head.push('/');
            } else if head.ends_with('/') {
                head.pop();
            }
            Some(head)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params,
            redirect,
        })
    }
}

fn path_head(path: &str) -> &str {
    let end = path
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(path.len());
    &path[..end]
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    let fixed_len = if wildcard {
        route.segments.len().checked_sub(1)?
    } else {
        route.segments.len()
    };
    if (!wildcard && request.len() != fixed_len) || (wildcard && request.len() < fixed_len) {
        return None;
    }

    let mut params = Vec::new();
    for (i, segment) in route.segments.iter().take(fixed_len).enumerate() {
        match segment {
            Segment::Static(value) if value != &request[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), request[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(Segment::Wildcard(name)) = route.segments.last() {
        params.push((name.clone(), request[fixed_len..].join("/")));
    }
    Some(params)
}
