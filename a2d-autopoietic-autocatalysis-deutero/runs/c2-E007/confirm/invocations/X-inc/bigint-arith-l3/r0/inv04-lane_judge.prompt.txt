    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' || bytes[0] == b'-' {
            1
        } else {
            0
        };
        if start == bytes.len() && start == 1 {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;

                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                malformed_separator = Some(i);
                break;
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });

                    first_digit = Some((i, b));

                return Err(Error::LeadingZero { index });

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for (&x, &y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(&y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()).saturating_add(1));
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        out.push(carry);
    }
    out
}

fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_magnitude(a, b) != Ordering::Less);
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(digit as u8);
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = vec![0u8; a.len().saturating_add(b.len()).saturating_add(1)];
    for (i, &x) in a.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &y) in b.iter().enumerate() {
            let k = i + j;
            let total = out[k] as u16 + x as u16 * y as u16 + carry;
            out[k] = (total % 10) as u8;
            carry = total / 10;
        }
        let mut k = i + b.len();
        while carry != 0 {
            let total = out[k] as u16 + carry;
            out[k] = (total % 10) as u8;
            carry = total / 10;
            k += 1;
        }
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_magnitude_digit(a: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }
    let mut out = Vec::with_capacity(a.len().saturating_add(1));
    let mut carry = 0u16;
    for &x in a {
        let total = x as u16 * digit as u16 + carry;
        out.push((total % 10) as u8);
        carry = total / 10;
    }
    while carry != 0 {
        out.push((carry % 10) as u8);
        carry /= 10;
    }
    out
}

fn div_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0u8];

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        trim_magnitude(&mut remainder);

        let mut qdigit = 0u8;
        for candidate in (1..=9).rev() {
            let product = mul_magnitude_digit(divisor, candidate);
            if cmp_magnitude(&product, &remainder) != Ordering::Greater {
                qdigit = candidate;
                remainder = sub_magnitude(&remainder, &product);
                break;
            }
        }
        quotient_be.push(qdigit);
    }

    while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
        quotient_be.remove(0);
    }
    quotient_be.reverse();
    trim_magnitude(&mut remainder);
    (quotient_be, remainder)
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
}

            _ => (false, 0),

                    first_digit = Some((i, b));
                }

            return Err(Error::InvalidDigit { index });
        }

        let (quotient, remainder) = div_rem_magnitude(&self.digits, &rhs.digits);
        Ok((
            Self::from_parts(self.negative ^ rhs.negative, quotient),
            Self::from_parts(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let mut value = Self {
            negative,
            digits: normalize(digits),
        };
        if value.is_zero() {
            value.negative = false;
        }
        value
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }
}

fn normalize(mut digits: Vec<u8>) -> Vec<u8> {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        vec![0]
    } else {
        digits
    }
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for i in (0..a.len()).rev() {
                match a[i].cmp(&b[i]) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        result.push(carry);
    }
    result
}

// Requires a >= b.
fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(digit as u8);
    }
    normalize(result)
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = vec![0u8; a.len() + b.len()];
    for (i, &ad) in a.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &bd) in b.iter().enumerate() {
            let k = i + j;
            let value = result[k] as u16 + ad as u16 * bd as u16 + carry;
            result[k] = (value % 10) as u8;
            carry = value / 10;
        }
        let mut k = i + b.len();
        while carry > 0 {
            let value = result[k] as u16 + carry;
            result[k] = (value % 10) as u8;
            carry = value / 10;
            k += 1;
        }
    }
    normalize(result)
}

fn mul_magnitude_digit(a: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }
    let mut result = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u16;
    for &value in a {
        let product = value as u16 * digit as u16 + carry;
        result.push((product % 10) as u8);
        carry = product / 10;
    }
    while carry > 0 {
        result.push((carry % 10) as u8);
        carry /= 10;
    }
    normalize(result)
}

// Magnitudes are little-endian.
fn div_rem_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_magnitude(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0u8];

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        remainder = normalize(remainder);

        let mut qdigit = 0u8;
        while qdigit < 9
            && cmp_magnitude(&remainder, &mul_magnitude_digit(divisor, qdigit + 1))
                != Ordering::Less
        {
            qdigit += 1;
        }
        if qdigit != 0 {
            remainder = sub_magnitude(
                &remainder,
                &mul_magnitude_digit(divisor, qdigit),
            );
        }
        quotient_be.push(qdigit);
    }

    quotient_be.reverse();
    (normalize(quotient_be), normalize(remainder))
}

use std::cmp::Ordering;

const BASE: u64 = 1_000_000_000;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    limbs: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over all separator and digit-count errors.
        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_bad_separator = None;

        for i in start..bytes.len() {
            if bytes[i].is_ascii_digit() {
                digit_count += 1;
                first_digit.get_or_insert(i);
            } else if bytes[i] == b'_' {
                let left_is_digit = i > start && bytes[i - 1].is_ascii_digit();
                let right_is_digit = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !(left_is_digit && right_is_digit) {
                    first_bad_separator.get_or_insert(i);
                }
            }
        }

        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        let first_digit = first_digit.expect("nonempty magnitude after sign");
        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits.saturating_add(1);
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i].is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
            // For usize::MAX, digit_count cannot exceed the limit.
        }

        let mut limbs = Vec::new();
        let mut chunk = 0u32;
        let mut chunk_digits = 0u32;
        for &byte in &bytes[start..] {
            if byte.is_ascii_digit() {
                chunk = chunk * 10 + u32::from(byte - b'0');
                chunk_digits += 1;
                if chunk_digits == 9 {
                    limbs.push(chunk);
                    chunk = 0;
                    chunk_digits = 0;
                }
            }
        }
        if chunk_digits != 0 {
            limbs.push(chunk);
        }

        let mut result = Self { negative, limbs };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut magnitude = self.magnitude_string();
        if magnitude.len() < min_digits {
            let mut padded = String::with_capacity(min_digits);
            padded.extend(std::iter::repeat_n('0', min_digits - magnitude.len()));
            padded.push_str(&magnitude);
            magnitude = padded;
        }

        if group > 0 && magnitude.len() > group {
            let first_group = {
                let rem = magnitude.len() % group;
                if rem == 0 { group } else { rem }
            };
            let separators = (magnitude.len() - 1) / group;
            let mut grouped = String::with_capacity(magnitude.len() + separators);
            grouped.push_str(&magnitude[..first_group]);
            let mut i = first_group;
            while i < magnitude.len() {
                grouped.push('_');
                grouped.push_str(&magnitude[i..i + group]);
                i += group;
            }
            magnitude = grouped;
        }

        if self.negative {
            let mut out = String::with_capacity(magnitude.len() + 1);
            out.push('-');
            out.push_str(&magnitude);
            out
        } else {
            magnitude
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.limbs, &rhs.limbs),
            (true, true) => cmp_magnitude(&rhs.limbs, &self.limbs),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self {
                negative: self.negative,
                limbs: add_magnitude(&self.limbs, &rhs.limbs),
            };
            result.normalize();
            result
        } else {
            match cmp_magnitude(&self.limbs, &rhs.limbs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        limbs: sub_magnitude(&self.limbs, &rhs.limbs),
                    };
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        limbs: sub_magnitude(&rhs.limbs, &self.limbs),
                    };
                    result.normalize();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let value = u64::from(limbs[k]) + u64::from(a) * u64::from(b) + carry;
                limbs[k] = (value % BASE) as u32;
                carry = value / BASE;
            }

            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == limbs.len() {
                    limbs.push(0);
                }
                let value = u64::from(limbs[k]) + carry;
                limbs[k] = (value % BASE) as u32;
                carry = value / BASE;
                k += 1;
            }
        }

        let mut result = Self {
            negative: self.negative != rhs.negative,
            limbs,
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        if self.is_zero() {
            return Ok((Self::zero(), Self::zero()));
        }

        match cmp_magnitude(&self.limbs, &rhs.limbs) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                return Ok((
                    Self {
                        negative: self.negative != rhs.negative,
                        limbs: vec![1],
                    },
                    Self::zero(),
                ));
            }
            Ordering::Greater => {}
        }

        let (quotient_limbs, remainder_limbs) =
            div_magnitude(&self.limbs, &rhs.limbs);

        let mut quotient = Self {
            negative: self.negative != rhs.negative,
            limbs: quotient_limbs,
        };
        quotient.normalize();

        let mut remainder = Self {
            negative: self.negative,
            limbs: remainder_limbs,
        };
        remainder.normalize();

        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn magnitude_string(&self) -> String {
        let Some((&last, rest)) = self.limbs.split_last() else {
            return "0".to_owned();
        };

        let mut out = last.to_string();
        for limb in rest.iter().rev() {
            use std::fmt::Write;
            write!(&mut out, "{limb:09}").expect("writing to String cannot fail");
        }
        out
    }
}

fn cmp_magnitude(lhs: &[u32], rhs: &[u32]) -> Ordering {
    match lhs.len().cmp(&rhs.len()) {
        Ordering::Equal => {
            for (a, b) in lhs.iter().rev().zip(rhs.iter().rev()) {
                match a.cmp(b) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(lhs: &[u32], rhs: &[u32]) -> Vec<u32> {
    let n = lhs.len().max(rhs.len());
    let mut out = Vec::with_capacity(n + 1);
    let mut carry = 0u64;

    for i in 0..n {
        let a = lhs.get(i).copied().unwrap_or(0) as u64;
        let b = rhs.get(i).copied().unwrap_or(0) as u64;
        let sum = a + b + carry;
        out.push((sum % BASE) as u32);
        carry = sum / BASE;
    }
    if carry != 0 {
        out.push(carry as u32);
    }
    out
}

fn sub_magnitude(lhs: &[u32], rhs: &[u32]) -> Vec<u32> {
    debug_assert!(cmp_magnitude(lhs, rhs) != Ordering::Less);

    let mut out = Vec::with_capacity(lhs.len());
    let mut borrow = 0i64;
    for i in 0..lhs.len() {
        let a = i64::from(lhs[i]);
        let b = i64::from(rhs.get(i).copied().unwrap_or(0));
        let mut value = a - b - borrow;
        if value < 0 {
            value += BASE as i64;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(value as u32);
    }

    while out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_magnitude_small(value: &[u32], factor: u32) -> Vec<u32> {
    if factor == 0 || value.is_empty() {
        return Vec::new();
    }
    let mut out = Vec::with_capacity(value.len() + 1);
    let mut carry = 0u64;
    for &limb in value {
        let product = u64::from(limb) * u64::from(factor) + carry;
        out.push((product % BASE) as u32);
        carry = product / BASE;
    }
    if carry != 0 {
        out.push(carry as u32);
    }
    out
}

fn sub_magnitude_in_place(lhs: &mut Vec<u32>, rhs: &[u32]) {
    debug_assert!(cmp_magnitude(lhs, rhs) != Ordering::Less);
    let mut borrow = 0i64;
    for i in 0..lhs.len() {
        let a = i64::from(lhs[i]);
        let b = i64::from(rhs.get(i).copied().unwrap_or(0));
        let mut value = a - b - borrow;
        if value < 0 {
            value += BASE as i64;
            borrow = 1;
        } else {
            borrow = 0;
        }
        lhs[i] = value as u32;
    }
    while lhs.last() == Some(&0) {
        lhs.pop();
    }
}

fn div_magnitude(dividend: &[u32], divisor: &[u32]) -> (Vec<u32>, Vec<u32>) {
    let mut quotient = vec![0u32; dividend.len()];
    let mut remainder = Vec::<u32>::new();

    for i in (0..dividend.len()).rev() {
        if !remainder.is_empty() {
            remainder.insert(0, dividend[i]);
        } else if dividend[i] != 0 {
            remainder.push(dividend[i]);
        }

        while remainder.last() == Some(&0) {
            remainder.pop();
        }

        let mut low = 0u32;
        let mut high = (BASE - 1) as u32;
        let mut best = 0u32;

        while low <= high {
            let mid = low + (high - low) / 2;
            let product = mul_magnitude_small(divisor, mid);
            match cmp_magnitude(&product, &remainder) {
                Ordering::Greater => {
                    if mid == 0 {
                        break;
                    }
                    high = mid - 1;
                }
                _ => {
                    best = mid;
                    if mid == (BASE - 1) as u32 {
                        break;
                    }
                    low = mid + 1;
                }
            }
        }

        quotient[i] = best;
        if best != 0 {
            let product = mul_magnitude_small(divisor, best);
            sub_magnitude_in_place(&mut remainder, &product);
        }
    }

    while quotient.last() == Some(&0) {
        quotient.pop();
    }
    (quotient, remainder)
}
