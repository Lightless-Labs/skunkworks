use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' || bytes[0] == b'-' {
            1
        } else {
            0
        };
        if start == bytes.len() && start == 1 {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                malformed_separator = Some(i);
                break;
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        let mut first_digit = None;
        let mut digit_count = 0usize;
        let mut excess_index = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b.is_ascii_digit() {
                digit_count += 1;
                if first_digit.is_none() {
                    first_digit = Some((i, b));
                }
                if digit_count == max_digits.saturating_add(1) && max_digits != usize::MAX {
                    excess_index = Some(i);
                }
            }
        }

        if digit_count > 1 {
            if let Some((index, b'0')) = first_digit {
                return Err(Error::LeadingZero { index });
            }
        }

        if digit_count > max_digits {
            let index = if max_digits == usize::MAX {
                // This branch cannot be reached for an addressable input.
                bytes.len()
            } else {
                excess_index.unwrap_or_else(|| {
                    bytes
                        .iter()
                        .enumerate()
                        .skip(start)
                        .filter(|(_, b)| b.is_ascii_digit())
                        .nth(max_digits)
                        .map(|(i, _)| i)
                        .unwrap_or(bytes.len())
                })
            };
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count.max(1));
        for &b in bytes[start..].iter().rev() {
            if b.is_ascii_digit() {
                digits.push(b - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        Ok(Self::canonical(
            start == 1 && bytes[0] == b'-',
            digits,
        ))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let mut magnitude = String::with_capacity(width.saturating_add(width / 2));
        for _ in self.digits.len()..width {
            magnitude.push('0');
        }
        for &digit in self.digits.iter().rev() {
            magnitude.push((b'0' + digit) as char);
        }

        let mut result = String::with_capacity(
            magnitude
                .len()
                .saturating_add(if group > 0 { magnitude.len() / group } else { 0 })
                .saturating_add(usize::from(self.negative)),
        );
        if self.negative {
            result.push('-');
        }
        if group == 0 {
            result.push_str(&magnitude);
        } else {
            let first_group = {
                let rem = magnitude.len() % group;
                if rem == 0 { group } else { rem }
            };
            result.push_str(&magnitude[..first_group]);
            let mut offset = first_group;
            while offset < magnitude.len() {
                result.push('_');
                let end = offset.saturating_add(group).min(magnitude.len());
                result.push_str(&magnitude[offset..end]);
                offset = end;
            }
        }
        result
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::canonical(
                self.negative,
                add_magnitude(&self.digits, &rhs.digits),
            )
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => Self::canonical(
                    self.negative,
                    sub_magnitude(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::canonical(
                    rhs.negative,
                    sub_magnitude(&rhs.digits, &self.digits),
                ),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::canonical(
            self.negative ^ rhs.negative,
            mul_magnitude(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (quotient, remainder) = div_magnitude(&self.digits, &rhs.digits);
        Ok((
            Self::canonical(self.negative ^ rhs.negative, quotient),
            Self::canonical(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn canonical(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for (&x, &y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(&y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()).saturating_add(1));
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        out.push(carry);
    }
    out
}

fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_magnitude(a, b) != Ordering::Less);
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(digit as u8);
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = vec![0u8; a.len().saturating_add(b.len()).saturating_add(1)];
    for (i, &x) in a.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &y) in b.iter().enumerate() {
            let k = i + j;
            let total = out[k] as u16 + x as u16 * y as u16 + carry;
            out[k] = (total % 10) as u8;
            carry = total / 10;
        }
        let mut k = i + b.len();
        while carry != 0 {
            let total = out[k] as u16 + carry;
            out[k] = (total % 10) as u8;
            carry = total / 10;
            k += 1;
        }
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_magnitude_digit(a: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }
    let mut out = Vec::with_capacity(a.len().saturating_add(1));
    let mut carry = 0u16;
    for &x in a {
        let total = x as u16 * digit as u16 + carry;
        out.push((total % 10) as u8);
        carry = total / 10;
    }
    while carry != 0 {
        out.push((carry % 10) as u8);
        carry /= 10;
    }
    out
}

fn div_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0u8];

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        trim_magnitude(&mut remainder);

        let mut qdigit = 0u8;
        for candidate in (1..=9).rev() {
            let product = mul_magnitude_digit(divisor, candidate);
            if cmp_magnitude(&product, &remainder) != Ordering::Greater {
                qdigit = candidate;
                remainder = sub_magnitude(&remainder, &product);
                break;
            }
        }
        quotient_be.push(qdigit);
    }

    while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
        quotient_be.remove(0);
    }
    quotient_be.reverse();
    trim_magnitude(&mut remainder);
    (quotient_be, remainder)
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
}
