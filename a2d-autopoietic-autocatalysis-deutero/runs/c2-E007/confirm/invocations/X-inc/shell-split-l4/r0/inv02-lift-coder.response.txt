#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn push_literal(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn push_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        push_literal(word, c, active_glob);
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            push_literal(&mut word, next, false);
                        }
                        i = after_next;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' && after < input.len() {
                                let skipped = input[after..].chars().next().unwrap();
                                j = after + skipped.len_utf8();
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(idx) = contents.find(":-") {
                            (&contents[..idx], Some(&contents[idx + 2..]))
                        } else {
                            (contents, None)
                        };
                        let mut name_chars = name.chars();
                        let valid_name = name_chars.next().is_some_and(is_name_start)
                            && name_chars.all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if let Some(value) = lookup(name, env) {
                            if let Some(default_text) = default {
                                if value.is_empty() {
                                    let decoded = decode_default(default_text);
                                    push_expansion(&mut word, &decoded, false);
                                } else {
                                    push_expansion(&mut word, value, false);
                                }
                            } else {
                                push_expansion(&mut word, value, false);
                            }
                        } else if let Some(default_text) = default {
                            let decoded = decode_default(default_text);
                            push_expansion(&mut word, &decoded, false);
                        }
                        active = true;
                        i = close + 1;
                    } else {
                        let after_dollar = next_i;
                        if after_dollar < input.len() {
                            let first = input[after_dollar..].chars().next().unwrap();
                            if is_name_start(first) {
                                let mut end = after_dollar + first.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if is_name_continue(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[after_dollar..end];
                                if let Some(value) = lookup(name, env) {
                                    push_expansion(&mut word, value, false);
                                }
                                active = true;
                                i = end;
                            } else {
                                word.text.push('$');
                                active = true;
                                i = next_i;
                            }
                        } else {
                            word.text.push('$');
                            active = true;
                            i = next_i;
                        }
                    }
                } else {
                    word.text.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        push_literal(&mut word, next, false);
                        active = true;
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' && after < input.len() {
                                let skipped = input[after..].chars().next().unwrap();
                                j = after + skipped.len_utf8();
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(idx) = contents.find(":-") {
                            (&contents[..idx], Some(&contents[idx + 2..]))
                        } else {
                            (contents, None)
                        };
                        let mut name_chars = name.chars();
                        let valid_name = name_chars.next().is_some_and(is_name_start)
                            && name_chars.all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if let Some(value) = lookup(name, env) {
                            if let Some(default_text) = default {
                                if value.is_empty() {
                                    let decoded = decode_default(default_text);
                                    push_expansion(&mut word, &decoded, true);
                                } else {
                                    push_expansion(&mut word, value, true);
                                }
                            } else {
                                push_expansion(&mut word, value, true);
                            }
                        } else if let Some(default_text) = default {
                            let decoded = decode_default(default_text);
                            push_expansion(&mut word, &decoded, true);
                        }
                        active = true;
                        i = close + 1;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            if let Some(value) = lookup(name, env) {
                                push_expansion(&mut word, value, true);
                            }
                            active = true;
                            i = end;
                        } else {
                            word.text.push('$');
                            active = true;
                            i = next_i;
                        }
                    } else {
                        word.text.push('$');
                        active = true;
                        i = next_i;
                    }
                } else {
                    push_literal(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();
        if c == '\\' && next_i < text.len() {
            let next = text[next_i..].chars().next().unwrap();
            out.push(next);
            i = next_i + next.len_utf8();
        } else {
            out.push(c);
            i = next_i;
        }
    }
    out
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%') {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        let mut best: Option<(usize, String)> = None;
        for (priority, candidate) in candidates.into_iter().enumerate() {
            if let Some(candidate) = candidate {
                let len = candidate.len();
                if best.as_ref().is_none_or(|(best_len, _)| len < *best_len) {
                    best = Some((len, candidate));
                } else if best.as_ref().is_some_and(|(best_len, _)| len == *best_len)
                    && priority < 3
                {
                    // Candidates are visited in tie priority order.
                }
            }
        }

        let result = if let Some((_, candidate)) = best {
            candidate
        } else {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push('\'');
                    fallback.push('\\');
                    fallback.push('\'');
                    fallback.push('\'');
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            fallback
        };
        encoded.push(result);
    }

    Ok(encoded.join(" "))
}
