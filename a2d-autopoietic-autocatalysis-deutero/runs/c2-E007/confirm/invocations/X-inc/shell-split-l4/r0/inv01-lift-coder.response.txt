#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            }
        } else {
            result.push(c);
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                    i = next_i;
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_literal(&mut word, next, false);
                        }
                        i = after_next;
                    } else {
                        append_literal(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut cursor = content_start;
                        let mut closing = None;
                        while cursor < input.len() {
                            let ch = input[cursor..].chars().next().unwrap();
                            let after = cursor + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let escaped = input[after..].chars().next().unwrap();
                                    cursor = after + escaped.len_utf8();
                                } else {
                                    cursor = after;
                                }
                            } else if ch == '}' {
                                closing = Some(cursor);
                                break;
                            } else {
                                cursor = after;
                            }
                        }
                        let close = match closing {
                            Some(pos) => pos,
                            None => return Err(SplitError::UnterminatedExpansion { pos: dollar }),
                        };
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(split_at) = contents.find(":-") {
                            (&contents[..split_at], Some(&contents[split_at + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty()
                            || !name.chars().next().map(is_name_start).unwrap_or(false)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if default.is_none() && contents.contains('\\') {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if let Some(default_text) = default {
                            if contents[..name.len()] != *name {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            let selected = match env_value(name, env) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(default_text),
                            };
                            append_expansion(&mut word, &selected, false);
                        } else {
                            append_expansion(
                                &mut word,
                                env_value(name, env).unwrap_or(""),
                                false,
                            );
                        }
                        active = true;
                        i = close + 1;
                    } else if let Some(name_start) = input[next_i..].chars().next() {
                        if is_name_start(name_start) {
                            let mut end = next_i;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_char(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            append_expansion(
                                &mut word,
                                env_value(name, env).unwrap_or(""),
                                false,
                            );
                            active = true;
                            i = end;
                        } else {
                            append_literal(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        append_literal(&mut word, next, false);
                        active = true;
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut cursor = content_start;
                        let mut closing = None;
                        while cursor < input.len() {
                            let ch = input[cursor..].chars().next().unwrap();
                            let after = cursor + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let escaped = input[after..].chars().next().unwrap();
                                    cursor = after + escaped.len_utf8();
                                } else {
                                    cursor = after;
                                }
                            } else if ch == '}' {
                                closing = Some(cursor);
                                break;
                            } else {
                                cursor = after;
                            }
                        }
                        let close = match closing {
                            Some(pos) => pos,
                            None => return Err(SplitError::UnterminatedExpansion { pos: dollar }),
                        };
                        let contents = &input[content_start..close];

                        let (name, default) = if let Some(split_at) = contents.find(":-") {
                            (&contents[..split_at], Some(&contents[split_at + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty()
                            || !name.chars().next().map(is_name_start).unwrap_or(false)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if default.is_none() && contents.contains('\\') {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let selected = if let Some(default_text) = default {
                            match env_value(name, env) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(default_text),
                            }
                        } else {
                            env_value(name, env).unwrap_or("").to_string()
                        };
                        append_expansion(&mut word, &selected, true);
                        active = true;
                        i = close + 1;
                    } else if let Some(name_start) = input[next_i..].chars().next() {
                        if is_name_start(name_start) {
                            let mut end = next_i;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_char(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            append_expansion(
                                &mut word,
                                env_value(name, env).unwrap_or(""),
                                true,
                            );
                            active = true;
                            i = end;
                        } else {
                            append_literal(&mut word, '$', true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut word, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }
    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut result = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates: Vec<(usize, usize, String)> = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate.len(), 0, candidate));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate.len(), 1, candidate));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate.len(), 2, candidate));
        }

        let chosen = if candidates.is_empty() {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push('\'');
                    fallback.push('\\');
                    fallback.push('\'');
                    fallback.push('\'');
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            fallback
        } else {
            candidates.sort_by_key(|(length, priority, _)| (*length, *priority));
            candidates.remove(0).2
        };
        encoded.push(chosen);
    }

    Ok(encoded.join(" "))
}
