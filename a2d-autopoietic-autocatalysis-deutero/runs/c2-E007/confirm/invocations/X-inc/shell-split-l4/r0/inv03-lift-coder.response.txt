use std::fmt::Write;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append(word: &mut Word, text: &str, wildcard_active: bool) {
    for c in text.chars() {
        if wildcard_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let mut cursor = dollar + 2;
    let mut closing = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += 1;
            if cursor < input.len() {
                cursor += input[cursor..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            closing = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[dollar + 2..close];
    let chars: Vec<(usize, char)> = contents.char_indices().collect();

    if chars.is_empty() || !is_name_start(chars[0].1) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let mut name_end = 0;
    while name_end < chars.len() && is_name_char(chars[name_end].1) {
        name_end += 1;
    }
    let name = &contents[..chars[name_end - 1].0 + chars[name_end - 1].1.len_utf8()];

    let selected = if name_end == chars.len() {
        lookup(env, name).unwrap_or("")
    } else {
        let op_start = chars[name_end].0;
        if !contents[op_start..].starts_with(":-") {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        let default_start = op_start + 2;
        let value = lookup(env, name).unwrap_or("");
        if !value.is_empty() {
            value.to_string()
        } else {
            let default = &contents[default_start..];
            let mut decoded = String::new();
            let mut i = 0;
            while i < default.len() {
                let c = default[i..].chars().next().unwrap();
                if c == '\\' {
                    i += 1;
                    if i < default.len() {
                        let escaped = default[i..].chars().next().unwrap();
                        decoded.push(escaped);
                        i += escaped.len_utf8();
                    }
                } else {
                    decoded.push(c);
                    i += c.len_utf8();
                }
            }
            decoded
        }
    };

    Ok((close + 1, selected))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let len = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                    i += len;
                } else {
                    append(&mut word, &input[i..i + len], false);
                    i += len;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i += len;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append(&mut word, &input[i..i + next.len_utf8()], false);
                        }
                        i += next.len_utf8();
                    } else {
                        append(&mut word, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_i = i + 1;
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, value) = parse_braced(input, i, env)?;
                        active = true;
                        append(&mut word, &value, false);
                        i = end;
                    } else if next_i < input.len()
                        && is_name_start(input[next_i..].chars().next().unwrap())
                    {
                        let mut end = next_i;
                        while end < input.len() && is_name_char(input[end..].chars().next().unwrap()) {
                            end += input[end..].chars().next().unwrap().len_utf8();
                        }
                        let value = lookup(env, &input[next_i..end]).unwrap_or("");
                        active = true;
                        append(&mut word, value, false);
                        i = end;
                    } else {
                        append(&mut word, "$", false);
                        i += 1;
                    }
                } else {
                    append(&mut word, &input[i..i + len], false);
                    i += len;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += len;
                } else if c == '\'' || c == '"' {
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_start = i;
                    active = true;
                    i += len;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        active = true;
                        append(&mut word, &input[i..i + next.len_utf8()], false);
                    }
                    i += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_i = i + 1;
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, value) = parse_braced(input, i, env)?;
                        active = true;
                        append(&mut word, &value, true);
                        i = end;
                    } else if next_i < input.len()
                        && is_name_start(input[next_i..].chars().next().unwrap())
                    {
                        let mut end = next_i;
                        while end < input.len() && is_name_char(input[end..].chars().next().unwrap()) {
                            end += input[end..].chars().next().unwrap().len_utf8();
                        }
                        let value = lookup(env, &input[next_i..end]).unwrap_or("");
                        active = true;
                        append(&mut word, value, true);
                        i = end;
                    } else {
                        active = true;
                        append(&mut word, "$", true);
                        i += 1;
                    }
                } else {
                    active = true;
                    append(&mut word, &input[i..i + len], true);
                    i += len;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut result = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

fn fallback_candidate(word: &str) -> String {
    let mut result = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            result.push('\'');
            result.push('\\');
            result.push('\'');
            result.push('\'');
        } else {
            result.push(c);
        }
    }
    result.push('\'');
    result
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate, 0));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate, 1));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate, 2));
        }

        let best = if candidates.is_empty() {
            fallback_candidate(word)
        } else {
            candidates
                .into_iter()
                .min_by_key(|(candidate, priority)| (candidate.len(), *priority))
                .unwrap()
                .0
        };
        encoded.push(best);
    }

    let mut output = String::new();
    for (index, word) in encoded.iter().enumerate() {
        if index > 0 {
            output.push(' ');
        }
        output.push_str(word);
    }
    Ok(output)
}
