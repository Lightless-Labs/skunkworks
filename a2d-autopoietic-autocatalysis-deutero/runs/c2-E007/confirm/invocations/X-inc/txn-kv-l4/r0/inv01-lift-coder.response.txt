```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Entry {
    value: String,
    deadline: Option<u64>,
}

#[derive(Clone)]
enum VersionData {
    Value(Entry),
    Tombstone,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    data: VersionData,
}

#[derive(Clone)]
struct Write {
    data: VersionData,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn validate_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn selected<'a>(&'a self, key: &str, tx: Option<u64>) -> Option<&'a VersionData> {
        let versions = self.data.get(key)?;
        let selected = match tx {
            Some(id) => {
                let revision = self.transactions.get(&id)?.begin_revision;
                versions.iter().rev().find(|v| v.revision <= revision)
            }
            None => versions.last(),
        }?;
        Some(&selected.data)
    }

    fn visible_entry(&self, key: &str, tx: Option<u64>, now: u64) -> Option<&Entry> {
        let data = if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(write) = transaction.writes.get(key) {
                &write.data
            } else {
                self.selected(key, tx)?
            }
        } else {
            self.selected(key, None)?
        };

        match data {
            VersionData::Value(entry)
                if entry.deadline.map_or(true, |deadline| deadline > now) =>
            {
                Some(entry)
            }
            _ => None,
        }
    }

    fn committed_live_entries(&self, now: u64) -> BTreeMap<String, Entry> {
        let mut result = BTreeMap::new();
        for (key, versions) in &self.data {
            if let Some(Version {
                data: VersionData::Value(entry),
                ..
            }) = versions.last()
            {
                if entry.deadline.map_or(true, |deadline| deadline > now) {
                    result.insert(key.clone(), entry.clone());
                }
            }
        }
        result
    }

    fn matching_keys(
        &self,
        tx: Option<u64>,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        after: Option<&str>,
    ) -> Vec<(String, String)> {
        let mut result = Vec::new();
        for key in self.data.keys() {
            if key.as_str() < start
                || end.map_or(false, |bound| key.as_str() >= bound)
                || prefix.map_or(false, |p| !key.starts_with(p))
                || after.map_or(false, |last| key.as_str() <= last)
            {
                continue;
            }
            if let Some(entry) = self.visible_entry(key, tx, now) {
                result.push((key.clone(), entry.value.clone()));
            }
        }
        result
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;

        if !transaction.writes.is_empty() {
            for key in transaction.writes.keys() {
                if self
                    .data
                    .get(key)
                    .and_then(|versions| versions.last())
                    .map_or(false, |version| version.revision > transaction.begin_revision)
                {
                    return Err(Error::Conflict {
                        tx: id,
                        key: key.clone(),
                    });
                }
            }

            let writes = transaction
                .writes
                .iter()
                .map(|(key, write)| (key.clone(), write.data.clone()))
                .collect::<Vec<_>>();
            self.revision += 1;
            let revision = self.revision;
            for (key, data) in writes {
                self.data
                    .entry(key)
                    .or_default()
                    .push(Version { revision, data });
            }
            self.transactions.remove(&id);
            Ok(revision)
        } else {
            self.transactions.remove(&id);
            Ok(self.revision)
        }
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let data = match ttl {
            Some(0) => VersionData::Tombstone,
            Some(duration) => VersionData::Value(Entry {
                value: value.to_string(),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            }),
            None => VersionData::Value(Entry {
                value: value.to_string(),
                deadline: None,
            }),
        };

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), Write { data });
        } else {
            self.revision += 1;
            self.data.entry(key.to_string()).or_default().push(Version {
                revision: self.revision,
                data,
            });
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let existed = self.visible_entry(key, tx, now).is_some();
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(
                key.to_string(),
                Write {
                    data: VersionData::Tombstone,
                },
            );
        } else {
            self.revision += 1;
            self.data
                .entry(key.to_string())
                .or_default()
                .push(Version {
                    revision: self.revision,
                    data: VersionData::Tombstone,
                });
        }
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self
            .visible_entry(key, tx, now)
            .map(|entry| entry.value.clone()))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        let mut count = 0;
        for key in self.data.keys() {
            if self
                .visible_entry(key, tx, now)
                .map_or(false, |entry| entry.value == value)
            {
                count += 1;
            }
        }
        if let Some(id) = tx {
            for (key, write) in &self.transactions.get(&id).unwrap().writes {
                if !self.data.contains_key(key)
                    && matches!(&write.data, VersionData::Value(entry) if entry.deadline.map_or(true, |deadline| deadline > now) && entry.value == value)
                {
                    count += 1;
                }
            }
        }
        Ok(count)
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.map_or(false, |bound| bound <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(cursor) = cursor {
            if cursor.start != start
                || cursor.end.as_deref() != end
                || cursor.prefix.as_deref() != prefix
                || cursor.offset != offset
                || cursor.last.is_empty()
                || cursor.last < start
                || end.map_or(false, |bound| cursor.last.as_str() >= bound)
                || prefix.map_or(false, |p| !cursor.last.starts_with(p))
            {
                return Err(Error::CursorMismatch);
            }
        }
        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut matches = self.matching_keys(
            tx,
            now,
            start,
            end,
            prefix,
            cursor.map(|cursor| cursor.last.as_str()),
        );
        if cursor.is_none() {
            let discard = offset.min(matches.len());
            matches.drain(..discard);
        }
        let take = limit.min(matches.len());
        let has_more = take < matches.len();
        let items = matches.into_iter().take(take).collect::<Vec<_>>();
        let next = if has_more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.map_or(false, |bound| bound <= start) {
            return Err(Error::InvalidRange);
        }

        let matches = self.matching_keys(tx, now, start, end, prefix, None);
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in matches {
            let number = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum
                .checked_add(number)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let saved = self.committed_live_entries(now);
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        validate_name(name)?;
        let saved = self
            .snapshots
            .get(name)
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let copied_live = saved
            .iter()
            .filter(|(_, entry)| entry.deadline.map_or(true, |deadline| deadline > now))
            .map(|(key, entry)| (key.clone(), entry.clone()))
            .collect::<BTreeMap<_, _>>();
        let current_live = self.committed_live_entries(now);
        let union = current_live
            .keys()
            .chain(copied_live.keys())
            .cloned()
            .collect::<BTreeSet<_>>();
        if union.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in union {
            let data = match copied_live.get(&key) {
                Some(entry) => VersionData::Value(entry.clone()),
                None => VersionData::Tombstone,
            };
            self.data
                .entry(key)
                .or_default()
                .push(Version { revision, data });
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        validate_name(name)?;
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for transaction in self.transactions.values() {
                if let Some(index) = versions
                    .iter()
                    .rposition(|version| version.revision <= transaction.begin_revision)
                {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self
            .data
            .values()
            .flat_map(|versions| versions.iter())
            .filter(|version| matches!(version.data, VersionData::Tombstone))
            .count();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys: self.committed_live_entries(now).len(),
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self
                .transactions
                .values()
                .filter(|transaction| transaction.read_only)
                .count(),
            staged_writes: self
                .transactions
                .values()
                .map(|transaction| transaction.writes.len())
                .sum(),
            snapshots: self.snapshots.len(),
        })
    }
}

fn validate_name(name: &str) -> Result<(), Error> {
    let bytes = name.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        let valid = if index == 0 {
            byte.is_ascii_lowercase()
        } else {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_'
        };
        if !valid {
            return Err(Error::NameInvalid { at: index });
        }
    }
    Ok(())
}

fn parse_canonical_i64(value: &str) -> Option<i64> {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return None;
    }

    let digits = if bytes[0] == b'-' {
        if bytes.len() < 2 {
            return None;
        }
        &bytes[1..]
    } else {
        bytes
    };

    if !digits.iter().all(u8::is_ascii_digit) {
        return None;
    }
    if digits.len() > 1 && digits[0] == b'0' {
        return None;
    }
    if bytes[0] == b'-' && digits == b"0" {
        return None;
    }
    if bytes[0] != b'-' && digits == b"0" {
        return Some(0);
    }
    value.parse::<i64>().ok()
}
```
