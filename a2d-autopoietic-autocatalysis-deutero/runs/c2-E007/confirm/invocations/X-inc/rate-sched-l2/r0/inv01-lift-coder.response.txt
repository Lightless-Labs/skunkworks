#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    eligible_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved_ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(text, 0, split, 60, &mut minutes)?;
        parse_field(text, split + 1, text.len(), 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = start;

    for i in start..=end {
        if i == end || bytes[i] == b',' {
            if i == item_start {
                return Err(CronError::Syntax(item_start));
            }
            parse_item(text, item_start, i, limit, selected)?;
            item_start = i + 1;
        }
    }

    Ok(())
}

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut slash = None;

    for i in start..end {
        if bytes[i] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(i));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(end);
    let has_step = slash.is_some();

    if let Some(slash_pos) = slash {
        if bytes[start] != b'*' && !bytes[start..base_end].contains(&b'-') {
            return Err(CronError::Syntax(slash_pos));
        }
    }

    let (base_start, base_finish) = if bytes[start] == b'*' {
        if start + 1 != base_end {
            return Err(CronError::Syntax(start + 1));
        }
        (0usize, limit - 1)
    } else if let Some(dash_rel) = bytes[start..base_end].iter().position(|&b| b == b'-') {
        let dash = start + dash_rel;
        let left = parse_number(text, start, dash, limit)?;
        let right = parse_number(text, dash + 1, base_end, limit)?;
        if left > right {
            return Err(CronError::Descending(dash));
        }
        (left, right)
    } else {
        let n = parse_number(text, start, base_end, limit)?;
        (n, n)
    };

    let step = if let Some(slash_pos) = slash {
        parse_step(text, slash_pos + 1, end, limit)?
    } else {
        1
    };

    let mut value = base_start;
    while value <= base_finish {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    let _ = has_step;
    Ok(())
}

fn parse_number(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
) -> Result<usize, CronError> {
    let bytes = text.as_bytes();

    if start == end {
        return Err(CronError::Syntax(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for i in start..end {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((bytes[i] - b'0') as usize))
        {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
        if value >= limit {
            return Err(CronError::Bounds(start));
        }
    }

    Ok(value)
}

fn parse_step(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
) -> Result<usize, CronError> {
    let bytes = text.as_bytes();

    if start == end {
        return Err(CronError::Syntax(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for i in start..end {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((bytes[i] - b'0') as usize))
        {
            Some(v) => v,
            None => return Err(CronError::Step(start)),
        };
        if value > limit {
            return Err(CronError::Step(start));
        }
    }

    if value == 0 || value > limit {
        return Err(CronError::Step(start));
    }

    Ok(value)
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved_ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved_ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let eligible_after = self.last_poll;
        let order = self.jobs.len();

        self.reserved_ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
            eligible_after,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let interval_start = self.last_poll;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let lower_exclusive = match (interval_start, job.eligible_after) {
                (Some(a), Some(b)) => Some(a.max(b)),
                (Some(a), None) => Some(a),
                (None, Some(b)) => Some(b),
                (None, None) => None,
            };

            let first_minute = match lower_exclusive {
                Some(ms) => ms / 60_000 + 1,
                None => 0,
            };
            let last_minute = now / 60_000;

            if first_minute > last_minute {
                continue;
            }

            let mut matching = Vec::new();
            for minute in first_minute..=last_minute {
                if job.cron.matches(minute) {
                    matching.push(minute);
                }
            }

            match job.policy {
                Policy::All => {
                    for minute in matching {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Once => {
                    if let Some(&minute) = matching.last() {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        if minute >= first_minute && minute <= last_minute && job.cron.matches(minute)
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}
