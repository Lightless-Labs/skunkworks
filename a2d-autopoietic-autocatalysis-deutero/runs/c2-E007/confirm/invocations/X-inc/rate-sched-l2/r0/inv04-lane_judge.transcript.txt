pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 {
            return Err(CronError::Layout(split));
        }
        if split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(bytes, 0, split, 60, &mut minutes)?;
        parse_field(bytes, split + 1, bytes.len(), 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    bytes: &[u8],
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut item_start = start;
    loop {
        let item_end = bytes[item_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| item_start + offset)
            .unwrap_or(end);

        if item_start == item_end {
            return Err(CronError::Syntax(item_start));
        }

        parse_item(bytes, item_start, item_end, limit, selected)?;

        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(())
}

fn parse_item(
    bytes: &[u8],
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let slashes: Vec<usize> = bytes[start..end]
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(start + i))
        .collect();

    if slashes.len() > 1 {
        return Err(CronError::Syntax(slashes[1]));
    }
    let slash = slashes.first().copied();

    let base_end = slash.unwrap_or(end);
    let step_start = slash.map(|p| p + 1);

    let dash = bytes[start..base_end]
        .iter()
        .position(|&b| b == b'-')
        .map(|offset| start + offset);

    let (range_start, range_end, is_star) = if bytes.get(start) == Some(&b'*') {
        if start + 1 != base_end {
            return Err(CronError::Syntax(start + 1));
        }
        (0usize, limit - 1, true)
    } else if let Some(dash_pos) = dash {
        let left = parse_number(bytes, start, dash_pos, limit)?;
        let right = parse_number(bytes, dash_pos + 1, base_end, limit)?;
        if left > right {
            return Err(CronError::Descending(dash_pos));
        }
        (left, right, true)
    } else {
        if let Some(slash_pos) = slash {
            return Err(CronError::Syntax(slash_pos));
        }
        let n = parse_number(bytes, start, base_end, limit)?;
        (n, n, false)
    };

    let step = if let Some(step_start) = step_start {
        if !is_star && dash.is_none() {
            return Err(CronError::Syntax(slash.unwrap()));
        }
        parse_step(bytes, step_start, end, limit)?
    } else {
        1
    };

    let mut n = range_start;
    while n <= range_end {
        selected[n] = true;
        match n.checked_add(step) {
            Some(next) if next > n => n = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_step(bytes: &[u8], start: usize, end: usize, limit: usize) -> Result<usize, CronError> {
    let n = parse_number_unbounded(bytes, start, end, limit, true)?;
    if n == 0 || n > limit {
        return Err(CronError::Step(start));
    }
    Ok(n)
}

fn parse_number(bytes: &[u8], start: usize, end: usize, limit: usize) -> Result<usize, CronError> {
    parse_number_unbounded(bytes, start, end, limit, false)
}

fn parse_number_unbounded(
    bytes: &[u8],
    start: usize,
    end: usize,
    limit: usize,
    is_step: bool,
) -> Result<usize, CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for &b in &bytes[start..end] {
        let digit = (b - b'0') as usize;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => {
                return if is_step {
                    Err(CronError::Step(start))
                } else {
                    Err(CronError::Bounds(start))
                };
            }
        };
    }

    if value >= limit {
        return if is_step {
            Err(CronError::Step(start))
        } else {
            Err(CronError::Bounds(start))
        };
    }
    Ok(value)
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        let owned_id = id.to_owned();
        self.reserved.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if first_minute <= last_minute {
            for job in &self.jobs {
                let mut matching = Vec::new();
                let mut minute = first_minute;
                while minute <= last_minute {
                    if job.cron.matches(minute) {
                        matching.push(minute);
                    }
                    minute += 1;
                }

                match job.policy {
                    Policy::All => {
                        for m in matching {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: m * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                    Policy::Once => {
                        if let Some(&m) = matching.last() {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: m * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0 {
                            let m = now / 60_000;
                            if job.cron.matches(m) && m >= first_minute {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: now,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}

fn parse_field(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],

        }
    }

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut slash = None;

    for i in start..end {
        if bytes[i] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(i));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(end);
    let has_step = slash.is_some();

    if let Some(slash_pos) = slash {
        if bytes[start] != b'*' && !bytes[start..base_end].contains(&b'-') {
            return Err(CronError::Syntax(slash_pos));
        }
    }

    let (base_start, base_finish) = if bytes[start] == b'*' {
        if start + 1 != base_end {
            return Err(CronError::Syntax(start + 1));
        }
        (0usize, limit - 1)
    } else if let Some(dash_rel) = bytes[start..base_end].iter().position(|&b| b == b'-') {
        let dash = start + dash_rel;
        let left = parse_number(text, start, dash, limit)?;
        let right = parse_number(text, dash + 1, base_end, limit)?;
        if left > right {
            return Err(CronError::Descending(dash));
        }
        (left, right)
    } else {
        let n = parse_number(text, start, base_end, limit)?;
        (n, n)
    };

    let step = if let Some(slash_pos) = slash {
        parse_step(text, slash_pos + 1, end, limit)?
    } else {
        1
    };

    let mut value = base_start;
    while value <= base_finish {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    let _ = has_step;
    Ok(())
}

fn parse_number(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
) -> Result<usize, CronError> {

    if start == end {
        return Err(CronError::Syntax(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {

        }
    }

}

        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let eligible_after = self.last_poll;
        let order = self.jobs.len();

        self.reserved_ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
            eligible_after,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let interval_start = self.last_poll;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let lower_exclusive = match (interval_start, job.eligible_after) {
                (Some(a), Some(b)) => Some(a.max(b)),
                (Some(a), None) => Some(a),
                (None, Some(b)) => Some(b),
                (None, None) => None,
            };

            let first_minute = match lower_exclusive {
                Some(ms) => ms / 60_000 + 1,
                None => 0,
            };
            let last_minute = now / 60_000;

            if first_minute > last_minute {
                continue;
            }

            let mut matching = Vec::new();
            for minute in first_minute..=last_minute {
                if job.cron.matches(minute) {
                    matching.push(minute);
                }
            }

            match job.policy {
                Policy::All => {
                    for minute in matching {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Once => {
                    if let Some(&minute) = matching.last() {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        if minute >= first_minute && minute <= last_minute && job.cron.matches(minute)
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}

pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(&text[..split], 0, 60, &mut minutes)?;
        parse_field(&text[split + 1..], split + 1, 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .find(',')
            .map_or(field.len(), |i| start + i);
        if start == end {
            return Err(CronError::Syntax(offset + start));
        }
        parse_item(&field[start..end], offset + start, limit, selected)?;
        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();

    let slash = bytes.iter().position(|&b| b == b'/');
    if let Some(s) = slash {
        if let Some(second) = bytes[s + 1..].iter().position(|&b| b == b'/') {
            return Err(CronError::Syntax(offset + s + 1 + second));
        }
    }

    let base_end = slash.unwrap_or(bytes.len());
    let base = &item[..base_end];
    let dash = base.as_bytes().iter().position(|&b| b == b'-');

    if slash.is_some() && base != "*" && dash.is_none() {
        return Err(CronError::Syntax(offset + base_end));
    }

    let (range_start, range_end) = if base == "*" {
        (0, limit - 1)
    } else if let Some(d) = dash {
        let left = parse_number(&base[..d], offset, limit)?;
        let right = parse_number(&base[d + 1..], offset + d + 1, limit)?;
        if left > right {
            return Err(CronError::Descending(offset + d));
        }
        (left, right)
    } else {
        let n = parse_number(base, offset, limit)?;
        (n, n)
    };

    let step = if let Some(s) = slash {
        parse_step(&item[s + 1..], offset + s + 1, limit)?
    } else {
        1
    };

    let mut value = range_start;
    loop {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next <= range_end => value = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_number(fragment: &str, offset: usize, limit: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();
    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as usize))
        {
            Some(n) => n,
            None => return Err(CronError::Bounds(offset)),
        };
    }
    if value >= limit {
        return Err(CronError::Bounds(offset));
    }
    Ok(value)
}

fn parse_step(fragment: &str, offset: usize, limit: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();
    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as usize))
        {
            Some(n) => n,
            None => return Err(CronError::Step(offset)),
        };
    }
    if value == 0 || value > limit {
        return Err(CronError::Step(offset));
    }
    Ok(value)
}

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')

            policy,
            insertion_order: order,
        });

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let start_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let end_minute = now / 60_000;
        let now_is_boundary = now % 60_000 == 0;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            match job.policy {
                Policy::All => {
                    if start_minute <= end_minute {
                        for minute in start_minute..=end_minute {

                                        id: job.id.clone(),

                                    job.insertion_order,

                            }
                        }

                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.insertion_order,
                            ));
                        }
                    }
                }
                Policy::Skip => {
                    if now_is_boundary {
                        let minute = now / 60_000;
                        if minute >= start_minute && job.cron.matches(minute) {

                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.insertion_order,
