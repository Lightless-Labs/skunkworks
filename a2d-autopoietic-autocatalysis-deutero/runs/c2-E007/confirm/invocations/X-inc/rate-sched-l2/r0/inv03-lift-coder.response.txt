#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct FieldSpec {
    min: usize,
    max: usize,
}

#[derive(Clone, Copy)]
struct Item {
    start: usize,
    end: usize,
    step: usize,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(
            bytes,
            0,
            split,
            FieldSpec { min: 0, max: 59 },
            &mut minutes,
        )?;
        parse_field(
            bytes,
            split + 1,
            bytes.len(),
            FieldSpec { min: 0, max: 23 },
            &mut hours,
        )?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    bytes: &[u8],
    field_start: usize,
    field_end: usize,
    spec: FieldSpec,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut item_start = field_start;

    loop {
        let item_end = bytes[item_start..field_end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| item_start + offset)
            .unwrap_or(field_end);

        if item_start == item_end {
            return Err(CronError::Syntax(item_start));
        }

        parse_item(bytes, item_start, item_end, spec, selected)?;

        if item_end == field_end {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

fn parse_item(
    bytes: &[u8],
    start: usize,
    end: usize,
    spec: FieldSpec,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut slash = None;
    for i in start..end {
        if bytes[i] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(i));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(end);
    let step_start = slash.map(|i| i + 1);

    let mut base_start = start;
    let mut base_stop = base_end;
    let is_star = base_start < base_stop && bytes[base_start] == b'*';

    if is_star {
        base_start += 1;
        if base_start != base_stop {
            return Err(CronError::Syntax(base_start));
        }
    } else {
        let dash = (base_start..base_stop).find(|&i| bytes[i] == b'-');
        if let Some(dash) = dash {
            if step_start.is_none() {
                // A range without a slash remains valid.
            }
            let left = parse_number(bytes, base_start, dash, spec)?;
            let right = parse_number(bytes, dash + 1, base_stop, spec)?;
            if left > right {
                return Err(CronError::Descending(dash));
            }
            base_start = left;
            base_stop = right;
        } else {
            if step_start.is_some() {
                return Err(CronError::Syntax(slash.unwrap()));
            }
            let value = parse_number(bytes, base_start, base_stop, spec)?;
            base_start = value;
            base_stop = value;
        }
    }

    let step = if let Some(step_start) = step_start {
        let step = parse_step(bytes, step_start, end, spec)?;
        if step == 0 || step > spec.max + 1 {
            return Err(CronError::Step(step_start));
        }
        step
    } else {
        1
    };

    let mut value = base_start;
    while value <= base_stop {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    spec: FieldSpec,
) -> Result<usize, CronError> {
    validate_digits(bytes, start, end)?;
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for &b in &bytes[start..end] {
        value = value
            .saturating_mul(10)
            .saturating_add((b - b'0') as usize);
        if value > spec.max {
            return Err(CronError::Bounds(start));
        }
    }

    if value < spec.min || value > spec.max {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

fn parse_step(
    bytes: &[u8],
    start: usize,
    end: usize,
    spec: FieldSpec,
) -> Result<usize, CronError> {
    validate_digits(bytes, start, end)?;
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for &b in &bytes[start..end] {
        value = value
            .saturating_mul(10)
            .saturating_add((b - b'0') as usize);
        if value > spec.max + 1 {
            return Err(CronError::Step(start));
        }
    }
    Ok(value)
}

fn validate_digits(bytes: &[u8], start: usize, end: usize) -> Result<(), CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    Ok(())
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved_ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved_ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved_ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        let id = id.to_owned();
        self.reserved_ids.push(id.clone());
        self.jobs.push(Job {
            id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let first_minute = match lower_exclusive {
            Some(last) => last / 60_000 + 1,
            None => 0,
        };
        let last_minute = now / 60_000;

        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let matching: Vec<u64> = if first_minute > last_minute {
                Vec::new()
            } else {
                (first_minute..=last_minute)
                    .filter(|&minute| job.cron.matches(minute))
                    .collect()
            };

            match job.policy {
                Policy::All => {
                    for minute in matching {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Once => {
                    if let Some(minute) = matching.last() {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: *minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        if matching.last() == Some(&minute) {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}
