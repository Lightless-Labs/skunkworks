pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {

        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative Kosaraju traversal avoids recursion on long paths.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((vertex, next)) = stack.last_mut() {
            if *next < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next];
                *next += 1;
                if !seen[neighbor] {
                    seen[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (vertex, _) = stack.pop().unwrap();
                finish.push(vertex);
            }
        }
    }

    let mut raw_components = Vec::new();
    let mut assigned = vec![false; n];
    for &root in finish.iter().rev() {
        if assigned[root] {
            continue;
        }
        assigned[root] = true;
        let mut component = Vec::new();
        let mut stack = vec![root];
        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &neighbor in &reverse[vertex] {
                if !assigned[neighbor] {
                    assigned[neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }
        component.sort_unstable();
        raw_components.push(component);
    }

    raw_components.sort_by(|a, b| {
        b.len()
            .cmp(&a.len())
            .then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0; n];
    for (id, members) in components.iter().enumerate() {
        for &vertex in members {
            component_of[vertex] = id;
        }
    }

    let count = components.len();

    let mut internal_edges = vec![0usize; count];
    let mut arc_map = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_map.entry((a, b)).or_insert((0, edge_index));
            entry.0 += 1;
            entry.1 = entry.1.min(edge_index);
        }
    }

    let mut arcs: Vec<Arc> = arc_map
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    let mut predecessors = vec![Vec::new(); count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        predecessors[arc.to].push(arc.from);
        indegree[arc.to] += 1;
    }

    // Repeatedly choose the best currently available component.
    let mut emitted = vec![false; count];
    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];

        let next = (0..count)
            .filter(|&id| !emitted[id] && indegree[id] == 0)
            .max_by(|&a, &b| {

            indegree[target] -= 1;
        }
    }

        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut adjacency = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adjacency[arc.from].push(arc.to);
        }
        for neighbors in &mut adjacency {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        if let Some(route) = shortest_lex_route(&adjacency, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_route(&adjacency, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn shortest_lex_route(adjacency: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let mut distance = vec![usize::MAX; adjacency.len()];
    let mut queue = std::collections::VecDeque::new();
    distance[goal] = 0;
    queue.push_back(goal);

    let mut reverse_adjacency = vec![Vec::new(); adjacency.len()];
    for (from, neighbors) in adjacency.iter().enumerate() {
        for &to in neighbors {
            reverse_adjacency[to].push(from);
        }
    }
    while let Some(vertex) = queue.pop_front() {
        for &previous in &reverse_adjacency[vertex] {
            if distance[previous] == usize::MAX {
                distance[previous] = distance[vertex] + 1;
                queue.push_back(previous);
            }
        }
    }
    if distance[start] == usize::MAX {
        return None;
    }

    let mut route = vec![start];
    let mut current = start;
    while current != goal {
        let next = adjacency[current]
            .iter()
            .copied()
            .filter(|&candidate| distance[candidate] + 1 == distance[current])
            .min()?;
        route.push(next);
        current = next;
    }
    Some(route)
}

fn strongly_connected_components(adjacency: &[Vec<usize>]) -> Vec<Vec<usize>> {
    let n = adjacency.len();
    let mut reverse = vec![Vec::new(); n];
    for (from, neighbors) in adjacency.iter().enumerate() {
        for &to in neighbors {
            reverse[to].push(from);
        }
    }

    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);

    for start in 0..n {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.last_mut() {
            if *next_index < adjacency[*vertex].len() {
                let next = adjacency[*vertex][*next_index];
                *next_index += 1;
                if !visited[next] {
                    visited[next] = true;
                    stack.push((next, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                finish_order.push(finished);
            }
        }
    }

    visited.fill(false);
    let mut components = Vec::new();

    for &start in finish_order.iter().rev() {
        if visited[start] {
            continue;
        }
        visited[start] = true;
        let mut component = Vec::new();
        let mut stack = vec![start];

        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &next in &reverse[vertex] {
                if !visited[next] {
                    visited[next] = true;
                    stack.push(next);
                }
            }
        }

        component.sort_unstable();
        components.push(component);
    }

    components
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {

        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
    }

    let mut components = strongly_connected_components(&adjacency);
    components.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));

    let mut component_of = vec![0usize; n];
    for (component_id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
    }

    let component_count = components.len();
    let mut cyclic: Vec<bool> = components.iter().map(|c| c.len() >= 2).collect();
    let mut internal_edges = vec![0usize; component_count];
    let mut arc_data: Vec<Vec<Option<(usize, usize)>>> =
        vec![vec![None; component_count]; component_count];

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let source = component_of[from];
        let target = component_of[to];
        if source == target {
            internal_edges[source] += 1;
            if from == to {
                cyclic[source] = true;
            }
        } else {
            match &mut arc_data[source][target] {
                Some((multiplicity, first_edge)) => {
                    *multiplicity += 1;
                    *first_edge = (*first_edge).min(edge_index);
                }
                slot @ None => *slot = Some((1, edge_index)),
            }
        }
    }

    let mut arcs = Vec::new();
    for from in 0..component_count {
        for to in 0..component_count {
            if let Some((multiplicity, first_edge)) = arc_data[from][to] {
                arcs.push(Arc {
                    from,
                    to,
                    multiplicity,
                    first_edge,
                });
            }
        }
    }
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); component_count];
    let mut indegree = vec![0usize; component_count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut available = Vec::new();
    for component_id in 0..component_count {
        if indegree[component_id] == 0 {
            available.push(component_id);
        }
    }

    let mut topo = Vec::with_capacity(component_count);
    let mut depth = vec![0usize; component_count];

    while !available.is_empty() {
        let best_position = (0..available.len())
            .max_by(|&left, &right| {
                let left_id = available[left];
                let right_id = available[right];
                internal_edges[left_id]
                    .cmp(&internal_edges[right_id])
                    .then_with(|| {
                        components[left_id][0].cmp(&components[right_id][0])
                    })
            })
            .unwrap();

        topo.push(component_id);

            depth[target] = depth[target].max(depth[component_id] + 1);
            indegree[target] -= 1;
            if indegree[target] == 0 {
                available.push(target);
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,

        let count = self.components.len();
        let mut outgoing = vec![Vec::new(); count];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }
        for neighbors in &mut outgoing {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        let mut parent = vec![None; count];
        let mut visited = vec![false; count];
        let mut queue = std::collections::VecDeque::new();
        visited[start] = true;
        queue.push_back(start);

        while let Some(vertex) = queue.pop_front() {
            if vertex == goal {
                break;
            }
            for &next in &outgoing[vertex] {
                if !visited[next] {
                    visited[next] = true;
                    parent[next] = Some(vertex);
                    queue.push_back(next);
                }
            }
        }

        if !visited[goal] {
            return None;
        }

        let mut route = vec![goal];
        let mut current = goal;
        while current != start {
            current = parent[current]?;
            route.push(current);
        }
        route.reverse();
        Some(route)
    }

    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];

        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        if let Some(route) = self.shortest_route(a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }

        if let Some(mut route) = self.shortest_route(b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }

        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn finish_order(adjacency: &[Vec<usize>]) -> Vec<usize> {
    let n = adjacency.len();
    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);

    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((v, next)) = stack.last_mut() {
            if *next < adjacency[*v].len() {
                let w = adjacency[*v][*next];
                *next += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                let (v, _) = stack.pop().unwrap();
                order.push(v);
            }
        }
    }
    order
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    let order = finish_order(&adjacency);
    let mut raw_of = vec![usize::MAX; n];
    let mut raw_components = Vec::<Vec<usize>>::new();

    for &start in order.iter().rev() {
        if raw_of[start] != usize::MAX {
            continue;
        }
        let id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        raw_of[start] = id;
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &reverse[v] {
                if raw_of[w] == usize::MAX {
                    raw_of[w] = id;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    let mut raw_ids: Vec<usize> = (0..raw_components.len()).collect();
    raw_ids.sort_by(|&a, &b| {
        raw_components[b]
            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
    });

    let mut remap = vec![0usize; raw_components.len()];
    let mut components = Vec::with_capacity(raw_components.len());
    for (id, raw) in raw_ids.into_iter().enumerate() {
        remap[raw] = id;
        components.push(raw_components[raw].clone());
    }

    let component_of: Vec<usize> = raw_of.iter().map(|&raw| remap[raw]).collect();
    let count = components.len();
    let mut cyclic: Vec<bool> = components.iter().map(|c| c.len() > 1).collect();
    let mut internal_edges = vec![0usize; count];
    let mut arc_pairs = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_pairs.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs: Vec<Arc> = arc_pairs
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = std::collections::BinaryHeap::<(usize, usize, usize)>::new();
    for id in 0..count {
        if indegree[id] == 0 {
            ready.push((
                internal_edges[id],
                components[id][0],
                id,
            ));
        }
    }

    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while let Some((_, _, v)) = ready.pop() {
        topo.push(v);
        for &w in &outgoing[v] {
            depth[w] = depth[w].max(depth[v] + 1);
            indegree[w] -= 1;
            if indegree[w] == 0 {
                ready.push((internal_edges[w], components[w][0], w));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    fn shortest_route(&self, start: usize, goal: usize) -> Option<Vec<usize>> {
        let n = self.components.len();
        let mut outgoing = vec![Vec::new(); n];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }
        for neighbors in &mut outgoing {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        let mut distance = vec![usize::MAX; n];
        let mut queue = std::collections::VecDeque::new();
        distance[goal] = 0;
        queue.push_back(goal);

        let mut incoming = vec![Vec::new(); n];
        for arc in &self.arcs {
            incoming[arc.to].push(arc.from);
        }
        while let Some(v) = queue.pop_front() {
            for &p in &incoming[v] {
                if distance[p] == usize::MAX {
                    distance[p] = distance[v] + 1;
                    queue.push_back(p);
                }
            }
        }

        if distance[start] == usize::MAX {
            return None;
        }
        let mut route = vec![start];
        let mut current = start;
        while current != goal {
            let next = *outgoing[current]
                .iter()
                .filter(|&&v| distance[v] + 1 == distance[current])
                .min()?;
            route.push(next);
            current = next;
        }
        Some(route)
    }

    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            return if self.cyclic[a] {
                Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                })
            } else {
                Ok(Query {
                    kind: Relation::Trivial,
                    route: vec![a],
                })
            };
        }

        if let Some(route) = self.shortest_route(a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = self.shortest_route(b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    // Iterative Kosaraju avoids recursion on long paths.
    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((v, next)) = stack.pop() {
            if next < adj[v].len() {
                stack.push((v, next + 1));
                let w = adj[v][next];
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                finish.push(v);
            }
        }
    }

    seen.fill(false);
    let mut raw_components = Vec::<Vec<usize>>::new();
    for &start in finish.iter().rev() {
        if seen[start] {
            continue;
        }
        let mut component = Vec::new();
        let mut stack = vec![start];
        seen[start] = true;
        while let Some(v) = stack.pop() {
            component.push(v);
            for &w in &rev[v] {
                if !seen[w] {
                    seen[w] = true;
                    stack.push(w);
                }
            }
        }
        component.sort_unstable();
        raw_components.push(component);
    }

    raw_components.sort_by(|a, b| {
        b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0usize; n];
    for (cid, component) in raw_components.iter().enumerate() {
        for &v in component {
            component_of[v] = cid;
        }
    }

    let count = raw_components.len();
    let mut cyclic = vec![false; count];
    let mut internal_edges = vec![0usize; count];
    let mut arc_pairs = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_pairs.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            if i < entry.1 {
                entry.1 = i;
            }
        }
    }
    for (cid, component) in raw_components.iter().enumerate() {
        if component.len() > 1 {
            cyclic[cid] = true;
        }
    }

    let mut arcs: Vec<Arc> = arc_pairs
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); count];
    let mut incoming = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        incoming[arc.to].push(arc.from);
        indegree[arc.to] += 1;
    }

    // Priority Kahn: greater internal edge count, then greater minimum vertex.
    let mut available = Vec::<usize>::new();
    for cid in 0..count {
        if indegree[cid] == 0 {
            available.push(cid);
        }
    }
    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while !available.is_empty() {
        let best_pos = (0..available.len())
            .max_by(|&i, &j| {
                let a = available[i];
                let b = available[j];
                internal_edges[a]
                    .cmp(&internal_edges[b])
                    .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
            })
            .unwrap();
        let cid = available.swap_remove(best_pos);
        topo.push(cid);
        for &next in &outgoing[cid] {
            depth[next] = depth[next].max(depth[cid] + 1);
            indegree[next] -= 1;
            if indegree[next] == 0 {
                available.push(next);
            }
        }
    }

    Ok(Graph {
        components: raw_components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let n = self.component_of.len();
        if from >= n {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let ncomp = self.components.len();
        let mut outgoing = vec![Vec::new(); ncomp];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }
        for neighbors in &mut outgoing {
            neighbors.sort_unstable();
        }

        fn shortest_lex(start: usize, target: usize, adj: &[Vec<usize>]) -> Option<Vec<usize>> {
            let mut parent = vec![usize::MAX; adj.len()];
            let mut seen = vec![false; adj.len()];
            let mut queue = std::collections::VecDeque::new();
            seen[start] = true;
            queue.push_back(start);
            while let Some(v) = queue.pop_front() {
                if v == target {
                    break;
                }
                for &w in &adj[v] {
                    if !seen[w] {
                        seen[w] = true;
                        parent[w] = v;
                        queue.push_back(w);
                    }
                }
            }
            if !seen[target] {
                return None;
            }
            let mut route = Vec::new();
            let mut cur = target;
            route.push(cur);
            while cur != start {
                cur = parent[cur];
                route.push(cur);
            }
            route.reverse();
            Some(route)
        }

        if let Some(route) = shortest_lex(a, b, &outgoing) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex(b, a, &outgoing) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}
