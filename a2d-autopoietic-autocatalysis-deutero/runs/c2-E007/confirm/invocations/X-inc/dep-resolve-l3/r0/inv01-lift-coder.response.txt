#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn percent_encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            out.push(byte as char);
        } else {
            out.push_str(&format!("%{:02X}", byte));
        }
    }
    out
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", percent_encode(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", percent_encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            percent_encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            percent_encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

#[derive(Clone)]
struct SearchResult {
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

struct Search<'a> {
    catalog: &'a [Package],
    package_indices: std::collections::BTreeMap<String, usize>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn active_constraints(
        &self,
        selected: &std::collections::BTreeMap<String, Version>,
    ) -> std::collections::BTreeMap<String, Vec<Constraint>> {
        let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
        for root in self.roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in selected {
            if let Some(&idx) = self.package_indices.get(name) {
                if let Some(release) = self.catalog[idx]
                    .releases
                    .iter()
                    .find(|release| release.version == *version)
                {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }
        active
    }

    fn inspect(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<SearchResult, ResolveError> {
        let active = self.active_constraints(selected);

        if let Some(package) = active
            .keys()
            .find(|name| !self.package_indices.contains_key(*name))
        {
            return Ok(SearchResult {
                best: None,
                first_error: Some(ResolveError::Missing {
                    package: package.clone(),
                }),
            });
        }

        for (package, version) in selected.iter() {
            if let Some(requirements) = active.get(package) {
                if !satisfies(*version, requirements) {
                    return Ok(SearchResult {
                        best: None,
                        first_error: Some(ResolveError::Conflict {
                            package: package.clone(),
                            selected: *version,
                            requirements: requirements.clone(),
                        }),
                    });
                }
            }
        }

        for (package, requirements) in &active {
            if !selected.contains_key(package) {
                let idx = self.package_indices[package];
                if !self.catalog[idx]
                    .releases
                    .iter()
                    .any(|release| satisfies(release.version, requirements))
                {
                    return Ok(SearchResult {
                        best: None,
                        first_error: Some(ResolveError::NoMatch {
                            package: package.clone(),
                            requirements: requirements.clone(),
                        }),
                    });
                }
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name));
        let Some(package) = next.cloned() else {
            let selections = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect::<Vec<_>>();
            let lock_map = self
                .locks
                .iter()
                .map(|lock| (lock.package.as_str(), lock.version))
                .collect::<std::collections::BTreeMap<_, _>>();
            let cost = lock_map
                .iter()
                .filter(|(name, version)| selected.get(**name) != Some(**version))
                .count();
            return Ok(SearchResult {
                best: Some((cost, selections)),
                first_error: None,
            });
        };

        let idx = self.package_indices[&package];
        let requirements = &active[&package];
        let mut candidates = self.catalog[idx]
            .releases
            .iter()
            .filter(|release| satisfies(release.version, requirements))
            .map(|release| release.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == package) {
            if let Some(pos) = candidates.iter().position(|version| *version == lock.version) {
                let version = candidates.remove(pos);
                candidates.insert(0, version);
            }
        }

        let mut best: Option<(usize, Vec<Selection>)> = None;
        let mut first_error = None;

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let child = self.inspect(selected);
            selected.remove(&package);

            let child = child?;
            if let Some(solution) = child.best {
                let replace = best.as_ref().is_none_or(|existing| {
                    solution.0 < existing.0
                        || (solution.0 == existing.0
                            && self.vector_cmp(&solution.1, &existing.1).is_gt())
                });
                if replace {
                    best = Some(solution);
                }
            } else if first_error.is_none() {
                first_error = child.first_error;
            }
        }

        Ok(SearchResult { best, first_error })
    }

    fn vector_cmp(&self, a: &[Selection], b: &[Selection]) -> std::cmp::Ordering {
        let amap = a
            .iter()
            .map(|selection| (selection.package.as_str(), selection.version))
            .collect::<std::collections::BTreeMap<_, _>>();
        let bmap = b
            .iter()
            .map(|selection| (selection.package.as_str(), selection.version))
            .collect::<std::collections::BTreeMap<_, _>>();
        for name in &self.catalog_names {
            match (amap.get(name.as_str()), bmap.get(name.as_str())) {
                (Some(av), Some(bv)) => {
                    let ordering = av.cmp(bv);
                    if !ordering.is_eq() {
                        return ordering;
                    }
                }
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (None, None) => {}
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_indices = catalog
        .iter()
        .enumerate()
        .map(|(i, package)| (package.name.clone(), i))
        .collect::<std::collections::BTreeMap<_, _>>();
    let mut catalog_names = catalog
        .iter()
        .map(|package| package.name.clone())
        .collect::<Vec<_>>();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        package_indices,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        catalog_names,
    };
    let mut selected = std::collections::BTreeMap::new();
    let result = search.inspect(&mut selected)?;

    let Some((_, selections)) = result.best else {
        return Err(result.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let final_selected = selections
        .iter()
        .map(|selection| (selection.package.clone(), selection.version))
        .collect::<std::collections::BTreeMap<_, _>>();
    let final_constraints = search.active_constraints(&final_selected);
    let lock_map = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect::<std::collections::BTreeMap<_, _>>();

    let mut names = std::collections::BTreeSet::new();
    names.extend(lock_map.keys().cloned());
    names.extend(final_selected.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = final_selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = package_indices
                    .get(&package)
                    .is_some_and(|&idx| catalog[idx].releases.iter().any(|r| r.version == locked));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(&package)
                    .is_some_and(|constraints| !satisfies(locked, constraints))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selections,
        changes,
    })
}
