#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        let next_i = i + ch.len_utf8();

        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                i = next_i;
            }
            Quote::Double(open) => {
                match ch {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if matches!(escaped, '"' | '\\' | '$') {
                            word.push(escaped);
                            i = after;
                        } else if escaped == '\n' {
                            i = after;
                        } else {
                            word.push('\\');
                            i = next_i;
                        }
                    }
                    '$' => {
                        let (consumed, value) = expand(input, i, env)?;
                        if consumed == 0 {
                            word.push('$');
                            i = next_i;
                        } else {
                            word.push_str(&value);
                            active = true;
                            i += consumed;
                        }
                    }
                    _ => {
                        word.push(ch);
                        i = next_i;
                    }
                }
                let _ = open;
            }
            Quote::None => {
                match ch {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(std::mem::take(&mut word));
                            active = false;
                        }
                        i = next_i;
                    }
                    '\'' => {
                        quote = Quote::Single(i);
                        active = true;
                        i = next_i;
                    }
                    '"' => {
                        quote = Quote::Double(i);
                        active = true;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if escaped != '\n' {
                            word.push(escaped);
                            active = true;
                        }
                        i = after;
                    }
                    '$' => {
                        let (consumed, value) = expand(input, i, env)?;
                        if consumed == 0 {
                            word.push('$');
                            active = true;
                            i = next_i;
                        } else {
                            word.push_str(&value);
                            active = true;
                            i += consumed;
                        }
                    }
                    _ => {
                        word.push(ch);
                        active = true;
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((0, String::new()));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let mut close = None;

        while j < input.len() {
            let ch = input[j..].chars().next().unwrap();
            let next = j + ch.len_utf8();
            if ch == '\\' {
                if next < input.len() {
                    let escaped = input[next..].chars().next().unwrap();
                    j = next + escaped.len_utf8();
                } else {
                    j = next;
                }
            } else if ch == '}' {
                close = Some(j);
                break;
            } else {
                j = next;
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .take_while(|(idx, ch)| {
                if *idx == 0 {
                    is_name_start(*ch)
                } else {
                    is_name_continue(*ch)
                }
            })
            .map(|(idx, ch)| idx + ch.len_utf8())
            .last()
            .unwrap_or(0);

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let rest = &contents[name_end..];
        let value = if rest.is_empty() {
            lookup(env, name).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            match lookup(env, name) {
                Some(v) if !v.is_empty() => v.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((close + 1 - dollar, value))
    } else if is_name_start(first) {
        let mut end = after_dollar;
        while end < input.len() {
            let ch = input[end..].chars().next().unwrap();
            if is_name_continue(ch) {
                end += ch.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((end - dollar, lookup(env, name).unwrap_or("").to_string()))
    } else {
        Ok((0, String::new()))
    }
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_continue(ch: char) -> bool {
    ch.is_ascii_alphanumeric() || ch == '_'
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let ch = text[i..].chars().next().unwrap();
        if ch == '\\' {
            let next_i = i + 1;
            if next_i < text.len() {
                let escaped = text[next_i..].chars().next().unwrap();
                result.push(escaped);
                i = next_i + escaped.len_utf8();
            } else {
                result.push('\\');
                i += 1;
            }
        } else {
            result.push(ch);
            i += ch.len_utf8();
        }
    }
    result
}
