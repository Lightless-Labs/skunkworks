#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    let rest = &input[after_dollar..];

    if rest.starts_with('{') {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let mut name_end = 0;
        let mut chars = contents.char_indices();
        if let Some((_, first)) = chars.next() {
            if is_name_start(first) {
                name_end = first.len_utf8();
                for (index, c) in chars {
                    if is_name_continue(c) {
                        name_end = index + c.len_utf8();
                    } else {
                        break;
                    }
                }
            }
        }

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let suffix = &contents[name_end..];
        let value = if suffix.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default_text) = suffix.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default_text),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((close + 1, value))
    } else {
        let mut cursor = after_dollar;
        let Some(first) = input[cursor..].chars().next() else {
            return Ok((after_dollar, "$".to_string()));
        };
        if !is_name_start(first) {
            return Ok((after_dollar, "$".to_string()));
        }

        cursor += first.len_utf8();
        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if is_name_continue(c) {
                cursor += c.len_utf8();
            } else {
                break;
            }
        }

        let name = &input[after_dollar..cursor];
        Ok((cursor, lookup(name, env).unwrap_or("").to_string()))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                cursor = next_cursor;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if c == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    let after_next = next_cursor + next.len_utf8();
                    if next == '"' || next == '\\' || next == '$' {
                        word.push(next);
                        cursor = after_next;
                    } else if next == '\n' {
                        cursor = after_next;
                    } else {
                        word.push('\\');
                        cursor = next_cursor;
                    }
                } else if c == '$' {
                    let (end, value) = expansion(input, cursor, env)?;
                    if end == cursor + 1 && value == "$" {
                        word.push('$');
                    } else {
                        word.push_str(&value);
                        active = true;
                    }
                    cursor = end;
                } else {
                    word.push(c);
                    cursor = next_cursor;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next_cursor;
                } else if c == '\'' {
                    quote = Quote::Single(cursor);
                    active = true;
                    cursor = next_cursor;
                } else if c == '"' {
                    quote = Quote::Double(cursor);
                    active = true;
                    cursor = next_cursor;
                } else if c == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    let after_next = next_cursor + next.len_utf8();
                    if next == '\n' {
                        cursor = after_next;
                    } else {
                        word.push(next);
                        active = true;
                        cursor = after_next;
                    }
                } else if c == '$' {
                    let (end, value) = expansion(input, cursor, env)?;
                    if end == cursor + 1 && value == "$" {
                        word.push('$');
                        active = true;
                    } else {
                        word.push_str(&value);
                        active = true;
                    }
                    cursor = end;
                } else {
                    word.push(c);
                    active = true;
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
