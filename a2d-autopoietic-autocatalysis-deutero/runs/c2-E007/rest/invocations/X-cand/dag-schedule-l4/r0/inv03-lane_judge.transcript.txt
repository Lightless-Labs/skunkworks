    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,

#[derive(Debug, Clone, PartialEq, Eq)]

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate all dependency entries before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shape and demand in task, then resource, order.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Deterministic Kahn order: repeatedly remove the smallest available ID.
    let mut available = std::collections::BinaryHeap::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut seen = vec![false; n];
            for &task_id in &order {
                seen[task_id] = true;
            }
            seen
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_free = vec![0u64; worker_count];
    let mut used = vec![0u64; capacity.len()];
    let mut running = Vec::<(usize, usize, u64)>::new(); // (task, worker, finish)
    let mut trace = String::new();
    use std::fmt::Write as _;
    writeln!(
        &mut trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    )
    .unwrap();

    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every task ending now, in ascending task ID.
        running.sort_by_key(|&(task_id, _, finish)| (finish, task_id));
        let mut kept = Vec::with_capacity(running.len());
        for (task_id, worker, finish) in running.drain(..) {
            if finish == time {
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                write!(
                    &mut trace,
                    "{}|end|{}|{}|",
                    time, task_id, worker
                )
                .unwrap();
                write_resource_snapshot(&mut trace, &used);
                trace.push('\n');
            } else {
                kept.push((task_id, worker, finish));
            }
        }
        running = kept;

        // Start one task at a time, reevaluating after each admission.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > time
                    || !tasks[task_id].deps.iter().all(|&dep| completed[dep])
                {
                    continue;
                }

                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r]);
                if !fits {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[task_id] > tails[current]
                            || (tails[task_id] == tails[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            // Pick the first available worker in the task's cyclic preference order.
            let mut selected_worker = None;
            for offset in 0..worker_count {
                let worker = (task_id % worker_count + offset) % worker_count;
                if worker_free[worker] <= time {
                    selected_worker = Some(worker);
                    break;
                }
            }
            let Some(worker) = selected_worker else {
                break;
            };

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_free[worker] = available_at;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            write!(
                &mut trace,
                "{}|start|{}|{}|",
                time, task_id, worker
            )
            .unwrap();
            write_resource_snapshot(&mut trace, &used);
            trace.push('\n');

            if finish == time {
                // Immediate completion of a newly started zero-duration task.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                write!(
                    &mut trace,
                    "{}|end|{}|{}|",
                    time, task_id, worker
                )
                .unwrap();
                write_resource_snapshot(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push((task_id, worker, finish));
            }
        }

        if completed_count == n {
            break;
        }

        // If nothing can start, advance to the earliest event that may change
        // eligibility, worker availability, or resource availability.
        let mut next_time: Option<u64> = None;
        for &(_, _, finish) in &running {
            if finish > time {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > time {
                let release = tasks[task_id].release;
                next_time = Some(next_time.map_or(release, |t| t.min(release)));
            }
        }
        for &free_at in &worker_free {
            if free_at > time {
                next_time = Some(next_time.map_or(free_at, |t| t.min(free_at)));
            }
        }

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    writeln!(&mut trace, "makespan={}", makespan).unwrap();

    Ok(Schedule {
        runs: runs
            .into_iter()
            .map(|run| run.expect("all tasks are started"))
            .collect(),
        makespan,
        trace,
    })
}

fn write_resource_snapshot(output: &mut String, used: &[u64]) {
    if used.is_empty() {
        output.push('-');
    } else {
        for (index, amount) in used.iter().enumerate() {
            if index != 0 {
                output.push(',');
            }
            use std::fmt::Write as _;
            write!(output, "{}", amount).unwrap();
        }
    }
}

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Kahn order with the smallest available ID first.
    let mut available = std::collections::BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom-level tails in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u64; resource_count];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut trace = format!("workers={worker_count};resources={resource_count}\n");
    let mut time = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks finishing now, in task ID order.
        let mut finished_now: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for entry in running.drain(..) {
            if entry.0 == time {
                finished_now.push(entry);
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;
        finished_now.sort_by_key(|entry| entry.1);

        for &(finish, id, worker) in &finished_now {
            for r in 0..resource_count {
                used[r] -= u64::from(tasks[id].needs[r]);
            }
            completed[id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            for &child in &children[id] {
                remaining_predecessors[child] -= 1;
            }
            append_event(&mut trace, time, "end", id, worker, &used);
        }

        // Start tasks one at a time, reconsidering eligibility after each start.
        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > time
                    || remaining_predecessors[id] != 0
                    || !fits(&tasks[id].needs, &used, capacity)
                {
                    continue;
                }
                if best.map_or(true, |current| {
                    tails[id] > tails[current]
                        || (tails[id] == tails[current]
                            && (tasks[id].priority > tasks[current].priority
                                || (tasks[id].priority == tasks[current].priority
                                    && id < current)))
                }) {
                    best = Some(id);
                }
            }

            let Some(id) = best else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= time)
                .expect("a worker is available when an admission is made");

            let finish = time.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            worker_available[worker] = available_at;
            for r in 0..resource_count {
                used[r] += u64::from(tasks[id].needs[r]);
            }
            runs[id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            append_event(&mut trace, time, "start", id, worker, &used);

            if finish == time {
                // Immediate completion for a zero-duration task.
                for r in 0..resource_count {
                    used[r] -= u64::from(tasks[id].needs[r]);
                }
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for &child in &children[id] {
                    remaining_predecessors[child] -= 1;
                }
                append_event(&mut trace, time, "end", id, worker, &used);
            } else {
                running.push((finish, id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change eligibility.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > time {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > time {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |current| current.min(release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > time {
                next_time = Some(next_time.map_or(available_at, |current| current.min(available_at)));
            }
        }

        time = next_time.expect("unfinished tasks always have a future event");
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs
        .iter()
        .zip(used.iter().zip(capacity.iter()))
        .all(|(&need, (&currently_used, &limit))| {
            u64::from(need) <= u64::from(limit) - currently_used
        })
}

fn append_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    trace.push_str(&format!("{time}|{kind}|{task}|{worker}|"));
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            trace.push_str(&value.to_string());
        }
    }
    trace.push('\n');
}

    pub deps: Vec<usize>,
    pub release: u64,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,

    },
    ResourceExceeded {
        task: usize,

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate all dependency entries before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resources task by task, preserving the specified error precedence.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in task.needs.iter().zip(capacity).enumerate() {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest available ID chosen at every step.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    let mut topo = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| *id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        topo.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if topo.len() != n {
        let removed: Vec<bool> = {
            let mut flags = vec![false; n];
            for &id in &topo {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom-level tails in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    if n == 0 {
        let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
        trace.push_str("makespan=0\n");
        return Ok(Schedule {
            runs: Vec::new(),
            makespan: 0,
            trace,
        });
    }

    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut used = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut completed_count = 0usize;
    let mut makespan = 0u64;
    let mut now = 0u64;

    loop {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        finishing.sort_by_key(|(_, task, _)| *task);

        if !finishing.is_empty() {
            running.retain(|(finish, _, _)| *finish != now);

            for (finish, task_id, worker) in finishing {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);

                for r in 0..capacity.len() {
                    used[r] -= u64::from(tasks[task_id].needs[r]);
                }

                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }

                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    format_usage(&used)
                ));
            }
        }

        if completed_count == n {
            break;
        }

        // Start tasks one at a time, reevaluating eligibility and ranking each time.
        let mut started_any = false;
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_predecessors[id] == 0
                        && fits(&tasks[id].needs, &used, capacity)
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now)
                .expect("a start is attempted only when an available worker exists");

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let next_worker_available = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = next_worker_available;

            for r in 0..capacity.len() {
                used[r] += u64::from(tasks[task_id].needs[r]);
            }

            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                format_usage(&used)
            ));

            if finish == now {
                // Immediate completion of a zero-duration task.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);

                for r in 0..capacity.len() {
                    used[r] -= u64::from(tasks[task_id].needs[r]);
                }

                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }

                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    format_usage(&used)
                ));
            } else {
                running.push((finish, task_id, worker));
            }

            started_any = true;
        }

        if completed_count == n {
            break;
        }

        // With no further starts possible at this time, advance to the next event.
        let mut next_time: Option<u64> = None;

        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |current| current.min(release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |current| current.min(available_at)));
            }
        }

        if let Some(next) = next_time {
            now = next;
        } else if !started_any {
            // In an acyclic graph with valid resources, this should be unreachable.
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task in an acyclic schedule must start"))
        .collect();

    trace.push_str(&format!("makespan={makespan}\n"));

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs
        .iter()
        .zip(used)
        .zip(capacity)
        .all(|((&need, &in_use), &cap)| u64::from(need) <= u64::from(cap) - in_use)
}

fn format_usage(used: &[u64]) -> String {
    if used.is_empty() {
        "-".to_string()
    } else {
        used.iter()
            .map(u64::to_string)
            .collect::<Vec<_>>()
            .join(",")
    }
}
