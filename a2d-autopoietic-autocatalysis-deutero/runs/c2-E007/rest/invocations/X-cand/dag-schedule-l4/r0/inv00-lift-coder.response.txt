#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before proceeding to other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Kahn order with the smallest available ID first.
    let mut available = std::collections::BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom-level tails in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u64; resource_count];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut trace = format!("workers={worker_count};resources={resource_count}\n");
    let mut time = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks finishing now, in task ID order.
        let mut finished_now: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for entry in running.drain(..) {
            if entry.0 == time {
                finished_now.push(entry);
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;
        finished_now.sort_by_key(|entry| entry.1);

        for &(finish, id, worker) in &finished_now {
            for r in 0..resource_count {
                used[r] -= u64::from(tasks[id].needs[r]);
            }
            completed[id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            for &child in &children[id] {
                remaining_predecessors[child] -= 1;
            }
            append_event(&mut trace, time, "end", id, worker, &used);
        }

        // Start tasks one at a time, reconsidering eligibility after each start.
        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > time
                    || remaining_predecessors[id] != 0
                    || !fits(&tasks[id].needs, &used, capacity)
                {
                    continue;
                }
                if best.map_or(true, |current| {
                    tails[id] > tails[current]
                        || (tails[id] == tails[current]
                            && (tasks[id].priority > tasks[current].priority
                                || (tasks[id].priority == tasks[current].priority
                                    && id < current)))
                }) {
                    best = Some(id);
                }
            }

            let Some(id) = best else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= time)
                .expect("a worker is available when an admission is made");

            let finish = time.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            worker_available[worker] = available_at;
            for r in 0..resource_count {
                used[r] += u64::from(tasks[id].needs[r]);
            }
            runs[id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            append_event(&mut trace, time, "start", id, worker, &used);

            if finish == time {
                // Immediate completion for a zero-duration task.
                for r in 0..resource_count {
                    used[r] -= u64::from(tasks[id].needs[r]);
                }
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for &child in &children[id] {
                    remaining_predecessors[child] -= 1;
                }
                append_event(&mut trace, time, "end", id, worker, &used);
            } else {
                running.push((finish, id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change eligibility.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > time {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > time {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |current| current.min(release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > time {
                next_time = Some(next_time.map_or(available_at, |current| current.min(available_at)));
            }
        }

        time = next_time.expect("unfinished tasks always have a future event");
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs
        .iter()
        .zip(used.iter().zip(capacity.iter()))
        .all(|(&need, (&currently_used, &limit))| {
            u64::from(need) <= u64::from(limit) - currently_used
        })
}

fn append_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    trace.push_str(&format!("{time}|{kind}|{task}|{worker}|"));
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            trace.push_str(&value.to_string());
        }
    }
    trace.push('\n');
}
