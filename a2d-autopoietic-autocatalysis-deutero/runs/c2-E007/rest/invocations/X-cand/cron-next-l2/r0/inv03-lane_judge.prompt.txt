fn parse_field(text: &str, field: Field, selected: &mut [bool]) -> Result<(), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !matches!(b, b'0'..=b'9' | b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let d = domain(field);
    let mut item_start = 0;

    for item in text.split(',') {
        let item_bytes = item.as_bytes();
        if item_bytes.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base, step_part) = if let Some(&slash) = slash_positions.first() {
            (&item_bytes[..slash], Some(&item_bytes[slash + 1..]))
        } else {
            (item_bytes, None)
        };

        let base_kind = if base == b"*" {
            Some((None, None))
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();
            match hyphens.as_slice() {
                [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                    Some((Some((0, base.len())), None))
                }
                [h] if *h > 0
                    && *h + 1 < base.len()
                    && base[..*h].iter().all(u8::is_ascii_digit)
                    && base[*h + 1..].iter().all(u8::is_ascii_digit) =>
                {
                    Some((Some((0, *h)), Some((*h + 1, base.len()))))
                }
                _ => None,
            }
        };

        let Some((single_part, end_part)) = base_kind else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        if step_part.is_some_and(|step| step.is_empty() || !step.iter().all(u8::is_ascii_digit)) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let mut start_value;
        let mut end_value;
        let is_star = base == b"*";
        let is_range = end_part.is_some();

        if is_star {
            start_value = d.min;
            end_value = d.max;
        } else if let Some((a, b)) = single_part {
            let value = parse_number(&base[a..b], item_start + a, field)?;
            if value < d.min || value > d.max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + a,
                });
            }
            start_value = value;
            end_value = value;
        } else {
            unreachable!();
        }

            let (start_a, start_b) = single_part.unwrap();
            start_value = parse_number(&base[start_a..start_b], item_start + start_a, field)?;
            if start_value < d.min || start_value > d.max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + start_a,
                });
            }

            end_value = parse_number(&base[a..b], item_start + a, field)?;
            if end_value < d.min || end_value > d.max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + a,
                });
            }

            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + a,
                });
            }
        }

            let step_offset = item_start + base.len() + 1;
            let value = parse_number(step_bytes, step_offset, field)?;
            if value == 0 || value > d.width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            value
        } else {
            1
        };

            selected[start_value as usize] = true;

                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        }

        item_start += item_bytes.len() + 1;
    }

    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut sets = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 32],
        vec![false; 13],
        vec![false; 7],
    ];

    for i in 0..5 {
        parse_field(parts[i], fields[i], &mut sets[i])?;
    }

    Ok(Cron {
        sets,
        day_of_month_is_exact_star: parts[2] == "*",
        day_of_week_is_exact_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (index, min, max) = match field {
            Field::Minute => (0, 0, 59),
            Field::Hour => (1, 0, 23),
            Field::DayOfMonth => (2, 1, 31),
            Field::Month => (3, 1, 12),
            Field::DayOfWeek => (4, 0, 6),
        };
        (min..=max)
            .filter(|&value| self.sets[index][value as usize])
            .map(|value| value as u8)
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.sets[0][at.minute as usize]
            && self.sets[1][at.hour as usize]
            && self.sets[2][at.day as usize]
            && self.sets[3][at.month as usize]
            && self.sets[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.minute == 59 {
                current.minute = 0;
                if current.hour == 23 {
                    current.hour = 0;
                    if current.day == days_in_month(current.year, current.month) {
                        current.day = 1;
                        if current.month == 12 {
                            if current.year == 2399 {
                                return Ok(None);
                            }
                            current.month = 1;
                            current.year += 1;
                        } else {
                            current.month += 1;
                        }
                    } else {
                        current.day += 1;
                    }
                } else {
                    current.hour += 1;
                }
            } else {
                current.minute += 1;
            }

            if self.matches(current)? {
                return Ok(Some(current));
            }
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut elapsed = 0u32;
    for y in 2000..year {
        elapsed += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        elapsed += days_in_month(year, m) as u32;
    }
    elapsed += day as u32 - 1;
    ((6 + elapsed) % 7) as u8
}

        };

        let base = &ib[..base_end];
        let star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        let range = if star {
            if !hyphens.is_empty() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            None
        } else if hyphens.is_empty() {
            if base.is_empty() || !base.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            None
        } else {
            let h = hyphens[0];
            if hyphens.len() != 1
                || h == 0
                || h + 1 == base.len()
                || !base[..h].iter().all(u8::is_ascii_digit)
                || !base[h + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Some(h)
        };

        let (start_value, end_value, start_offset, end_offset) = if star {
            (min, max, item_start, item_start)
        } else if let Some(h) = range {
            let a = parse_number(ib, 0, h, field, false)?;
            let b = parse_number(ib, h + 1, base_end, field, false)?;
            if a > b {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + h + 1,
                });
            }
            (a, b, item_start, item_start + h + 1)
        } else {
            let a = parse_number(ib, 0, base_end, field, false)?;
            (a, if step.is_some() { max } else { a }, item_start, item_start)
        };

        let stride = if let Some((s, e)) = step {
            parse_number(ib, s, e, field, true)?
        } else {
            1
        };

        let _ = (start_offset, end_offset);
        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, field) in FIELDS.iter().enumerate() {
        selected[i] = parse_field(*field, pieces[i])?;
    }

    Ok(Cron {
        selected,
        day_of_month_is_star: pieces[2] == "*",
        day_of_week_is_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| if yes { Some(i as u8) } else { None })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut at = after;
        loop {
            if at.year == 2399 && at.month == 12 && at.day == 31 && at.hour == 23 && at.minute == 59 {
                return Ok(None);
            }
            at = next_minute(at);
            if self.matches(at)? {
                return Ok(Some(at));
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_days(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_days(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_days(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn next_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < month_days(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (minimum, maximum) = bounds(field);
        let width = maximum - minimum + 1;
        if value < 1 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (minimum, maximum) = bounds(field);
        if value < minimum || value > maximum {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }
    Ok(value)
}

fn set_range(
    selected: &mut [bool],
    start: u32,
    end: u32,
    step: u32,
    minimum: u32,
) {
    let mut value = start;
    loop {
        selected[(value - minimum) as usize] = true;
        if end - value < step {
            break;
        }
        value += step;
    }
}

fn parse_field(text: &str, field: Field, selected: &mut [bool]) -> Result<(), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
            (slash, Some(slash + 1))
        } else {
            (item.len(), None)
        };
        let base = &item[..base_end];

        let (minimum, maximum) = bounds(field);
        let (start_value, end_value, start_offset, end_offset) = if base == b"*" {
            (minimum, maximum, None, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
                .collect();

            if hyphens.len() > 1
                || base.is_empty()
                || (hyphens.len() == 1
                    && (hyphens[0] == 0 || hyphens[0] + 1 == base.len()))
                || base.iter().any(|&b| !b.is_ascii_digit() && b != b'-')
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }

            if let Some(&hyphen) = hyphens.first() {
                let left = parse_number(
                    base,
                    0,
                    hyphen,
                    field,
                    false,
                )?;
                let right = parse_number(
                    base,
                    hyphen + 1,
                    base.len(),
                    field,
                    false,
                )?;
                if left > right {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
                (
                    left,
                    right,
                    Some(item_start),
                    Some(item_start + hyphen + 1),
                )
            } else {
                let value = parse_number(base, 0, base.len(), field, false)?;
                (value, if step_start.is_some() { maximum } else { value }, Some(item_start), None)
            }
        };

        let step = if let Some(step_start) = step_start {
            if step_start == item.len() || item[step_start..].iter().any(|&b| !b.is_ascii_digit()) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            parse_number(
                item,
                step_start,
                item.len(),
                field,
                true,
            )?
        } else {
            1
        };

        let _ = (start_offset, end_offset);
        set_range(
            selected,
            start_value,
            end_value,
            step,
            minimum,
        );

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected: [Vec<bool>; 5] = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 31],
        vec![false; 12],
        vec![false; 7],
    ];

    for i in 0..5 {
        parse_field(pieces[i], FIELDS[i], &mut selected[i])?;
    }

    Ok(Cron {
        selected,
        day_of_month_is_star: pieces[2] == "*",
        day_of_week_is_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (minimum, _) = bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| {
                if yes {
                    Some((i as u32 + minimum) as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let (minimum, maximum) = bounds(field);
        let value = value as u32;
        value >= minimum
            && value <= maximum
            && self.selected[field_index(field)][(value - minimum) as usize]
    }

    fn day_matches(&self, date: CivilDateTime) -> bool {
        let weekday = weekday(date.year, date.month, date.day);
        self.contains(Field::DayOfMonth, date.day)
            && self.contains(Field::DayOfWeek, weekday)
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.day_matches(at)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            advance_minute(&mut current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut offset = 0u32;
    for y in 2000..year {
        offset += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        offset += days_in_month(year, m) as u32;
    }
    offset += day as u32 - 1;
    ((offset + 6) % 7) as u8
}

fn advance_minute(value: &mut CivilDateTime) {
    if value.minute < 59 {
        value.minute += 1;
        return;
    }
    value.minute = 0;

    if value.hour < 23 {
        value.hour += 1;
        return;
    }
    value.hour = 0;

    if value.day < days_in_month(value.year, value.month) {
        value.day += 1;
        return;
    }
    value.day = 1;

    if value.month < 12 {
        value.month += 1;
        return;
    }
    value.month = 1;
    value.year += 1;
}
