#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Parameter(String),
}

#[derive(Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Parameter(_) => 1,
            })
            .collect();

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head, false)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != request_segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_segment, request_segment) in
                route.segments.iter().zip(request_segments.iter())
            {
                match pattern_segment {
                    Segment::Static(value) if value == request_segment => {}
                    Segment::Static(_) => {
                        matches = false;
                        break;
                    }
                    Segment::Parameter(name) => {
                        params.push((name.clone(), request_segment.clone()));
                    }
                }
            }

            if matches {
                let replace = match best.as_ref() {
                    None => true,
                    Some((best_id, _)) => {
                        route.rank > self.routes[*best_id].rank
                    }
                };
                if replace {
                    best = Some((route_id, params));
                }
            }
        }

        best.map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    let raw_segments = parse_raw_segments(pattern, true)?;
    let mut result = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();

    for (start, raw) in raw_segments {
        if raw.starts_with(':') {
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(Segment::Parameter(name.to_owned()));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            result.push(Segment::Static(decode_segment(raw, start, true)?));
        }
    }

    Ok(result)
}

fn parse_path(path: &str, pattern: bool) -> Result<Vec<String>, Error> {
    let raw_segments = parse_raw_segments(path, pattern)?;
    raw_segments
        .into_iter()
        .map(|(start, raw)| decode_segment(raw, start, pattern))
        .collect()
}

fn parse_raw_segments<'a>(
    input: &'a str,
    pattern: bool,
) -> Result<Vec<(usize, &'a str)>, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                // Validate preceding segments before rejecting the trailing slash.
                for (segment_start, segment) in &result {
                    if pattern && (segment.starts_with(':') || segment.starts_with('*')) {
                        continue;
                    }
                    decode_segment(segment, *segment_start, pattern)?;
                }
                if start == i {
                    return Err(invalid(start));
                }
                return Err(invalid(i));
            }
            if start == i {
                return Err(invalid(start));
            }
            result.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(invalid(bytes.len() - 1));
    }
    result.push((start, &input[start..]));
    Ok(result)
}

fn decode_segment(raw: &str, start: usize, pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(start + i));
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return Err(invalid(start + i));
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return Err(invalid(start + i));
            };
            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(start + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid(start + i));
        }
    }

    // Valid escapes decode only to ASCII bytes.
    Ok(String::from_utf8(decoded).expect("decoded segment is ASCII"))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}
