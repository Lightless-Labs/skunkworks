#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Clone)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let sep = spaces[0];
        if sep == 0 || sep + 1 == bytes.len() {
            return Err(CronError::Layout(sep));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(&bytes[..sep], 0, 59, &mut minutes)?;
        parse_field(&bytes[sep + 1..], sep + 1, 23, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn syntax_at(offset: usize) -> CronError {
    CronError::Syntax(offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    max: u8,
) -> Result<u8, CronError> {
    if start == end {
        return Err(syntax_at(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(syntax_at(i));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value: u16 = 0;
    for &b in &bytes[start..end] {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u16);
        if value > max as u16 {
            return Err(CronError::Bounds(start));
        }
    }
    Ok(value as u8)
}

fn parse_field(
    field: &[u8],
    offset: usize,
    max: u8,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut item_start = 0;
    loop {
        let item_end = field[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(field.len());

        if item_start == item_end {
            return Err(syntax_at(offset + item_start));
        }

        parse_item(
            field,
            item_start,
            item_end,
            offset,
            max,
            selected,
        )?;

        if item_end == field.len() {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(())
}

fn parse_item(
    field: &[u8],
    start: usize,
    end: usize,
    offset: usize,
    max: u8,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut slash = None;
    for i in start..end {
        if field[i] == b'/' {
            if slash.is_some() {
                return Err(syntax_at(offset + i));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(end);
    let step_start = slash.map(|i| i + 1);

    let mut dash = None;
    for i in start..base_end {
        if field[i] == b'-' {
            dash = Some(i);
            break;
        }
    }

    let is_star = base_end - start == 1 && field[start] == b'*';
    let is_range = dash.is_some();

    if step_start.is_some() && !is_star && !is_range {
        return Err(syntax_at(offset + slash.unwrap()));
    }

    let (lo, hi) = if is_star {
        (0, max)
    } else if let Some(dash_pos) = dash {
        let lo = parse_number(field, start, dash_pos, max)?;
        let hi = parse_number(field, dash_pos + 1, base_end, max)?;
        if lo > hi {
            return Err(CronError::Descending(offset + dash_pos));
        }
        (lo, hi)
    } else {
        let n = parse_number(field, start, base_end, max)?;
        (n, n)
    };

    let step = if let Some(step_start) = step_start {
        let value = parse_number(field, step_start, end, u8::MAX)?;
        if value == 0 || value > if max == 59 { 60 } else { 24 } {
            return Err(CronError::Step(offset + step_start));
        }
        value as usize
    } else {
        1
    };

    let mut n = lo as usize;
    while n <= hi as usize {
        selected[n] = true;
        n += step;
    }

    Ok(())
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    first_minute: u64,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: std::collections::HashSet<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: std::collections::HashSet::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.contains(id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let first_minute = self
            .last_poll
            .map(|t| t / 60_000 + 1)
            .unwrap_or(0);

        self.ids.insert(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
            first_minute,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let upper_minute = now / 60_000;
        let lower_exclusive = self.last_poll.map(|t| t / 60_000);
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let start = lower_exclusive
                .map(|m| m.saturating_add(1))
                .unwrap_or(0)
                .max(job.first_minute);
            if start > upper_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    for minute in start..=upper_minute {
                        if job.cron.matches(minute) {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    if let Some(minute) = (start..=upper_minute)
                        .rev()
                        .find(|&minute| job.cron.matches(minute))
                    {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0
                        && now / 60_000 >= start
                        && job.cron.matches(upper_minute)
                    {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}
