#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|item| item == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|item| item == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') {
        '\u{feff}'.len_utf8()
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut opening_quote = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8 boundary");
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            if after_quote || !in_quotes {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                field_started = false;
                after_quote = false;
                record_index += 1;
                i += 2;
                continue;
            }
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                i += 1;
                continue;
            }
            if b == b'\n' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                field_started = false;
                after_quote = false;
                record_index += 1;
                i += 1;
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            field_started = false;
            record_index += 1;
            i += 1;
            continue;
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
            continue;
        }

        if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
            in_quotes = true;
            opening_quote = i;
            field_started = true;
            i += 1;
            continue;
        }

        let ch = input[i..].chars().next().expect("valid UTF-8 boundary");
        field.push(ch);
        field_started = true;
        i += ch.len_utf8();
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if after_quote || field_started || !field.is_empty() || !record.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut records = records.into_iter();
    let header = if config.headers {
        records.next()
    } else {
        None
    };

    if let (Some(header_values), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen: std::collections::HashMap<&str, usize> =
            std::collections::HashMap::new();
        for (second, name) in header_values.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name.as_str(), second);
        }
    }

    let mut rows = Vec::new();
    for (offset, row) in records.enumerate() {
        if let Some(ref header_values) = header {
            if row.len() != header_values.len() {
                let record = offset + 1;
                return Err(CsvError::HeaderWidth {
                    record,
                    expected: header_values.len(),
                    actual: row.len(),
                });
            }
        }
        rows.push(row);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }

        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub enum ColumnRef {

    Name(String),

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {

    pub kind: ColumnType,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' {
        &bytes[1..]
    } else {
        bytes
    };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    (b'1'..=b'9').contains(&digits[0]) && digits[1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let body = if bytes.first() == Some(&b'-') {
        &bytes[1..]
    } else {
        bytes
    };
    let Some(dot) = body.iter().position(|&b| b == b'.') else {
        return false;
    };
    if body[dot + 1..].len() != 2
        || !body[dot + 1..].iter().all(u8::is_ascii_digit)
        || body[..dot].is_empty()
    {
        return false;
    }
    let integral = &body[..dot];
    let integral_ok = integral == b"0"
        || (!integral.is_empty()
            && (b'1'..=b'9').contains(&integral[0])
            && integral[1..].iter().all(u8::is_ascii_digit));
    if !integral_ok {
        return false;
    }
    !(bytes.first() == Some(&b'-') && integral == b"0" && body[dot + 1..] == *b"00")
}

fn parse_decimal_value(s: &str) -> Option<i64> {
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let dot = body.find('.')?;
    let integral = body[..dot].parse::<i128>().ok()?;
    let fractional = body[dot + 1..].parse::<i128>().ok()?;
    let magnitude = integral.checked_mul(100)?.checked_add(fractional)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).ok()
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        for (spec_index, spec) in columns.iter().enumerate() {
            let column_index = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let Some(index) = column_index else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                output.push(None);
                match policy {
                    ErrorPolicy::Collect => continue,
                    ErrorPolicy::FirstPerRow => {
                        output.resize(columns.len(), None);
                        result.rows.push(output);
                        continue 'rows;
                    }
                    ErrorPolicy::Stop => {
                        output.resize(columns.len(), None);
                        result.rows.push(output);
                        break 'rows;
                    }
                }
            };

            let Some(cell) = row.get(index) else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(index),
                    kind: CellErrorKind::MissingCell,
                });
                output.push(None);
                match policy {
                    ErrorPolicy::Collect => continue,
                    ErrorPolicy::FirstPerRow => {
                        output.resize(columns.len(), None);
                        result.rows.push(output);
                        continue 'rows;
                    }
                    ErrorPolicy::Stop => {
                        output.resize(columns.len(), None);
                        result.rows.push(output);
                        break 'rows;
                    }
                }
            };

            let evaluated: Result<Value, CellErrorKind> = if cell.is_empty() {
                if spec.nullable {
                    Ok(Value::Null)
                } else if spec.kind == ColumnType::Text {
                    Ok(Value::Text(String::new()))
                } else {
                    Err(CellErrorKind::Empty)
                }
            } else {
                match spec.kind {
                    ColumnType::Text => Ok(Value::Text(cell.clone())),
                    ColumnType::Integer => {
                        if !integer_syntax(cell) {
                            Err(CellErrorKind::InvalidInteger)
                        } else {
                            match cell.parse::<i64>() {
                                Ok(value) => Ok(Value::Integer(value)),
                                Err(_) => Err(CellErrorKind::IntegerOverflow),
                            }
                        }
                    }
                    ColumnType::Decimal => {
                        if !decimal_syntax(cell) {
                            Err(CellErrorKind::InvalidDecimal)
                        } else {
                            parse_decimal_value(cell)
                                .map(Value::Decimal)
                                .ok_or(CellErrorKind::DecimalOverflow)
                        }
                    }
                    ColumnType::Boolean => match cell.as_str() {
                        "yes" | "1" => Ok(Value::Boolean(true)),
                        "no" | "0" => Ok(Value::Boolean(false)),
                        _ => Err(CellErrorKind::InvalidBoolean),
                    },
                }
            };

            match evaluated {
                Ok(value) => output.push(Some(value)),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind,
                    });
                    output.push(None);
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => {
                            output.resize(columns.len(), None);
                            result.rows.push(output);
                            continue 'rows;
                        }
                        ErrorPolicy::Stop => {
                            output.resize(columns.len(), None);
                            result.rows.push(output);
                            break 'rows;
                        }
                    }
                }
            }
        }
        result.rows.push(output);
    }

    result
}
