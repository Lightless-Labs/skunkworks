#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    const fn zero() -> Self {
        Self { num: 0, den: 1 }
    }

    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self::zero());
        }

        let (num, den) = if den < 0 {
            (
                num.checked_neg().ok_or(RationalError::Overflow)?,
                den.checked_neg().ok_or(RationalError::Overflow)?,
            )
        } else {
            (num, den)
        };

        let gcd = gcd_u128(num.unsigned_abs(), den as u128);
        let reduced_num = num / gcd as i128;
        let reduced_den = den / gcd as i128;

        if reduced_num < i64::MIN as i128
            || reduced_num > i64::MAX as i128
            || reduced_den < 1
            || reduced_den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: reduced_num as i64,
            den: reduced_den as i64,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        Self::from_i128(a * d + c * b, b * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        Self::from_i128(a * d - c * b, b * d)

        Self::from_i128(
            self.num as i128 * rhs.num as i128,

            return Err(RationalError::DivideByZero);

            self.den as i128 * rhs.num as i128,

    }

        let parse_err = |at, kind| RationalError::Parse { at, kind };

            i = 1;
        }

        let num_start = i;

            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

            }

        }

        }

        }

            return Err(parse_err(0, ParseKind::NonCanonical));
        }

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn parse_magnitude(digits: &[u8]) -> u128 {
    digits.iter().fold(0u128, |value, &digit| {
        value * 10 + (digit - b'0') as u128
    })
}

        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduction = eliminate(augmented, n)?;
        let rank = reduction.pivots.len();
        if rank < n {
            return Err(MatrixError::Singular { rank });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for row in 0..n {
            for col in 0..n {
                inverse_data.push(reduction.matrix.data[row * width + n + col]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }
}

fn eliminate(mut matrix: Matrix, pivot_cols: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let cols = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for candidate_col in 0..pivot_cols {
        if p == rows {
            break;
        }

        let mut chosen: Option<usize> = None;
        for row in p..rows {
            let value = matrix.data[row * cols + candidate_col];
            if value.num == 0 {
                continue;
            }
            let key = (value.den as u64, value.num.unsigned_abs(), row);
            if chosen.map_or(true, |old| {
                let old_value = matrix.data[old * cols + candidate_col];
                key < (
                    old_value.den as u64,
                    old_value.num.unsigned_abs(),
                    old,
                )
            }) {
                chosen = Some(row);
            }
        }

        let Some(q) = chosen else {
            continue;
        };

        if q != p {
            for col in 0..cols {
                matrix.data.swap(p * cols + col, q * cols + col);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + candidate_col];
        for col in 0..cols {
            let index = p * cols + col;
            matrix.data[index] = matrix.data[index].div(&pivot).map_err(|source| {
                MatrixError::Arithmetic {
                    row: p,
                    col,
                    step: p,
                    source,
                }
            })?;
        }

        for row in 0..rows {
            if row == p {
                continue;
            }
            let factor = matrix.data[row * cols + candidate_col];
            if factor.num == 0 {
                continue;
            }

            for col in 0..cols {
                let product = factor
                    .mul(&matrix.data[p * cols + col])
                    .map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: p,
                        source,
                    })?;
                let index = row * cols + col;
                matrix.data[index] =
                    matrix.data[index]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col,
                            step: p,
                            source,
                        })?;
            }
        }

        pivots.push(candidate_col);
        p += 1;
    }

    Ok(Reduction {
        matrix,
        pivots,
        swaps,
    })
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128
        || n > i64::MAX as i128
        || d < 1
        || d > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

fn rational_from_i128_wide(num: i128, den: i128) -> Result<Rational, RationalError> {

        return Err(RationalError::ZeroDenominator);

        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        rational_from_i128_wide(
            self.num as i128 * a + rhs.num as i128 * b,
            self.den as i128 * a,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        rational_from_i128_wide(
            self.num as i128 * a - rhs.num as i128 * b,
            self.den as i128 * a,
        )
    }

        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128) as i128;
        rational_from_i128_wide(

        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        rational_from_i128_wide(
            (self.num as i128 / g1) * (rhs.den as i128 / g2),
            (self.den as i128 / g2) * (rhs.num as i128 / g1),
        )
    }

        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational { num: -self.num, den: self.den })

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn perr(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(perr(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(perr(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(perr(i, ParseKind::InvalidByte));
            }
            i += 1;
            let ds = i;
            den_start = Some(ds);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == ds {
                return Err(perr(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(perr(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let digits_start = if negative { 1 } else { 0 };
        if num_end - digits_start >= 2 && bytes[digits_start] == b'0' {
            return Err(perr(digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_end == 2 && bytes[1] == b'0' {
            return Err(perr(0, ParseKind::NonCanonical));
        }
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(perr(ds + 1, ParseKind::NonCanonical));
            }
        }

        let mut nmag: u128 = 0;
        for &b in &bytes[digits_start..num_end] {
            nmag = nmag.saturating_mul(10).saturating_add((b - b'0') as u128);
            let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
            if nmag > limit {
                return Err(perr(0, ParseKind::OutOfRange));
            }
        }

        let mut dmag: u128 = 1;
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            dmag = 0;
            for &b in &bytes[ds..de] {
                dmag = dmag.saturating_mul(10).saturating_add((b - b'0') as u128);
                if dmag > i64::MAX as u128 {
                    return Err(perr(ds, ParseKind::OutOfRange));
                }
            }
            if dmag == 0 {
                return Err(perr(ds, ParseKind::ZeroDenominator));
            }
            if dmag == 1 {
                return Err(perr(ds - 1, ParseKind::NonCanonical));
            }
            if nmag == 0 {
                return Err(perr(ds - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(nmag, dmag) > 1 {
                return Err(perr(ds - 1, ParseKind::NonCanonical));
            }
        }

        let n = if negative {
            if nmag == 1u128 << 63 { i64::MIN } else { -(nmag as i64) }
        } else {
            nmag as i64
        };
        Ok(Rational { num: n, den: dmag as i64 })
    }
}

pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let idx = row.checked_mul(self.cols)
            .and_then(|x| x.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data.get(idx).copied().ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Matrix { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Matrix { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, pivot_cols: usize) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_cols {
            if p == matrix.rows { break; }
            let mut best: Option<usize> = None;
            for r in p..matrix.rows {
                let v = matrix.data[r * matrix.cols + col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), r);
                if best.map_or(true, |br| {
                    let b = matrix.data[br * matrix.cols + col];
                    key < (b.den, b.num.unsigned_abs(), br)
                }) {
                    best = Some(r);
                }
            }
            let Some(q) = best else { continue };
            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..matrix.cols {
                let idx = p * matrix.cols + c;
                matrix.data[idx] = matrix.data[idx].div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            }
            for r in 0..matrix.rows {
                if r == p { continue; }
                let factor = matrix.data[r * matrix.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..matrix.cols {
                    let piv = matrix.data[p * matrix.cols + c];
                    let product = factor.mul(&piv)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                    let idx = r * matrix.cols + c;
                    matrix.data[idx] = matrix.data[idx].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let augmented = Matrix { rows: n, cols: width, data };
        let red = augmented.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut right = Vec::with_capacity(total / 2);
        for r in 0..n {
            right.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Matrix { rows: n, cols: n, data: right })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

            if i < bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let num_digits = &bytes[num_start..num_end];

            if de - ds >= 2 && bytes[ds] == b'0' {

            }

    }

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn parse_bounded_magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &byte in digits {
        let digit = (byte - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[row * self.cols + k];
                    let right = rhs.data[k * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };

        for row in 0..n {
            data.extend_from_slice(&self.data[row * n..(row + 1) * n]);
            for col in 0..n {
                data.push(if row == col { one } else { zero });
            }
        }

        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduction = eliminate(augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for row in 0..n {
            inverse_data.extend_from_slice(
                &reduction.matrix.data[row * width + n..row * width + width],
            );
        }

        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }
}

fn eliminate(mut matrix: Matrix, pivot_limit: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let cols = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for pivot_col in 0..pivot_limit {
        if p == rows {
            break;
        }

        let mut chosen: Option<usize> = None;
        for row in p..rows {
            let value = matrix.data[row * cols + pivot_col];
            if value.num == 0 {
                continue;
            }
            let key = (value.den, value.num.unsigned_abs(), row);
            if chosen
                .map(|current| {
                    let other = matrix.data[current * cols + pivot_col];
                    key < (other.den, other.num.unsigned_abs(), current)
                })
                .unwrap_or(true)
            {
                chosen = Some(row);
            }
        }

        let Some(q) = chosen else {
            continue;
        };

        if q != p {
            for col in 0..cols {
                matrix.data.swap(p * cols + col, q * cols + col);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + pivot_col];
        for col in 0..cols {
            let index = p * cols + col;
            matrix.data[index] = matrix.data[index].div(&pivot).map_err(|source| {
                MatrixError::Arithmetic {
                    row: p,
                    col,
                    step: p,
                    source,
                }
            })?;
        }

        for row in 0..rows {
            if row == p {
                continue;
            }
            let factor = matrix.data[row * cols + pivot_col];
            if factor.num == 0 {
                continue;
            }
            for col in 0..cols {
                let product = factor
                    .mul(&matrix.data[p * cols + col])
                    .map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: p,
                        source,
                    })?;
                let index = row * cols + col;
                matrix.data[index] = matrix.data[index].sub(&product).map_err(|source| {
                    MatrixError::Arithmetic {
                        row,
                        col,
                        step: p,
                        source,
                    }
                })?;
            }
        }

        pivots.push(pivot_col);
        p += 1;
    }

    Ok(Reduction {
        matrix,
        pivots,
        swaps,
    })
}
