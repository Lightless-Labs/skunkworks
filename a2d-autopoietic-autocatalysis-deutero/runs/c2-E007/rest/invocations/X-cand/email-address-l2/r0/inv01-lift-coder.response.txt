#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_end = if bytes.first() == Some(&b'"') {
        let mut i = 1;
        let mut end = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    end = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match end {
            Some(end) => Some(end),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, quoted) = if bytes[0] == b'"' {
        let end = quoted_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            if bytes[j] == b'\\' {
                let escaped = bytes.get(j + 1).copied();
                match escaped {
                    Some(b'"' | b'\\' | b'@') => {
                        decoded.push(escaped.unwrap());
                        origins.push(j);
                        j += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: j }),
                }
            } else {
                let b = bytes[j];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: j });
                }
                decoded.push(b);
                origins.push(j);
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }

        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let local_bytes = &bytes[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0
                    || j + 1 == local_bytes.len()
                    || local_bytes[j - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: j });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: j });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let removed_root_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for pos in domain_start..domain_end {
        if bytes[pos] == b'.' {
            labels.push((label_start, pos));
            label_start = pos + 1;
        }
    }
    labels.push((label_start, domain_end));

    for &(start, end) in &labels {
        if start == end {
            let at = if start == domain_end {
                removed_root_dot.unwrap_or(domain_end)
            } else {
                start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }

        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: end - 1 });
        }

        for pos in start..end {
            let b = bytes[pos];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = String::from_utf8(domain_bytes.to_vec())
        .unwrap()
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
