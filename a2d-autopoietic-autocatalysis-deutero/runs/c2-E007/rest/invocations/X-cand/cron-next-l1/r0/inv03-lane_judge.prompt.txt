        ErrorKind::EmptyItem => ParseError::EmptyItem { field, offset },
        ErrorKind::InvalidSyntax => ParseError::InvalidSyntax { field, offset },
        ErrorKind::LeadingZero => ParseError::LeadingZero { field, offset },
        ErrorKind::NumberOverflow => ParseError::NumberOverflow { field, offset },
        ErrorKind::OutOfRange => ParseError::OutOfRange { field, offset },
        ErrorKind::ReversedRange => ParseError::ReversedRange { field, offset },

#[derive(Clone, Copy)]
enum ErrorKind {
    InvalidCharacter,
    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,
    OutOfRange,
    ReversedRange,
    BadStep,
}

fn parse_number(
    bytes: &[u8],
    offset: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(make_error(field, offset, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| make_error(field, offset, ErrorKind::NumberOverflow))?;
    }

    if step {
        let (_, max) = field_bounds(field);
        let width = (max as u32) + 1 - (field_bounds(field).0 as u32);
        if value == 0 || value > width {
            return Err(make_error(field, offset, ErrorKind::BadStep));
        }
    } else {
        let (min, max) = field_bounds(field);
        if value < min as u32 || value > max as u32 {
            return Err(make_error(field, offset, ErrorKind::OutOfRange));
        }
    }

    Ok(value)
}

fn parse_field(field_text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = field_text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(make_error(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let (domain_min, domain_max) = field_bounds(field);
    let mut selected = vec![false; (domain_max as usize) + 1];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(make_error(field, item_start, ErrorKind::EmptyItem));
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(make_error(field, item_start, ErrorKind::InvalidSyntax));
        }

        let (base, step_part, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
        } else {
            (item, None, 0)
        };

        let (base_kind, start_part, end_part, end_offset) = if base == b"*" {
            (BaseKind::Star, None, None, 0)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();

            if hyphens.len() > 1 {
                return Err(make_error(field, item_start, ErrorKind::InvalidSyntax));
            }

            if let Some(&hyphen) = hyphens.first() {
                let left = &base[..hyphen];
                let right = &base[hyphen + 1..];
                if left.is_empty() || right.is_empty() {
                    return Err(make_error(field, item_start, ErrorKind::InvalidSyntax));
                }
                (BaseKind::Range, Some(left), Some(right), item_start + hyphen + 1)
            } else {
                if base.is_empty() || !base.iter().all(u8::is_ascii_digit) {
                    return Err(make_error(field, item_start, ErrorKind::InvalidSyntax));
                }
                (BaseKind::Number, Some(base), None, 0)
            }
        };

        if base_kind == BaseKind::Star && step_part.is_none() {
            for value in domain_min..=domain_max {
                selected[value as usize] = true;
            }
        } else {
            let (start, end) = match base_kind {
                BaseKind::Star => (domain_min as u32, domain_max as u32),
                BaseKind::Number => {
                    let part = start_part.unwrap();
                    let start = parse_number(
                        part,
                        item_start,
                        field,
                        false,
                    )?;
                    let end = if step_part.is_some() {
                        domain_max as u32
                    } else {
                        start
                    };
                    (start, end)
                }
                BaseKind::Range => {
                    let left = start_part.unwrap();
                    let right = end_part.unwrap();
                    let start = parse_number(left, item_start, field, false)?;
                    let end = parse_number(right, end_offset, field, false)?;
                    if start > end {
                        return Err(make_error(field, end_offset, ErrorKind::ReversedRange));
                    }
                    (start, end)
                }
            };

            let step = if let Some(part) = step_part {
                if part.is_empty() || !part.iter().all(u8::is_ascii_digit) {
                    return Err(make_error(field, item_start, ErrorKind::InvalidSyntax));
                }
                parse_number(part, step_offset, field, true)?
            } else {
                1
            };

            let mut value = start;
            while value <= end {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
        .collect())
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum BaseKind {
    Star,
    Number,
    Range,
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_stars = [false; 2];

    for i in 0..5 {
        if i == 2 {
            day_stars[0] = pieces[i].as_bytes() == b"*";
        } else if i == 4 {
            day_stars[1] = pieces[i].as_bytes() == b"*";
        }
        selected[i] = parse_field(pieces[i], fields[i])?;
    }

    Ok(Cron {
        selected,
        day_stars,
    })
}

        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(error(field, start, 0));
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(field, start, 1));
    }

    let mut value: u32 = 0;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| error(field, start, 2))?;
    }

    if step {
        let (_, max) = domain(field);
        if value == 0 || value > max as u32 + 1 {
            return Err(error(field, start, 5));
        }
    } else {
        let (min, max) = domain(field);
        if value < min as u32 || value > max as u32 {
            return Err(error(field, start, 3));
        }
    }

    Ok(value)
}

fn parse_field(
    text: &str,
    field: Field,
    selected: &mut [bool],
) -> Result<bool, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(error(field, offset, 7));
        }
    }

    let exact_star = text == "*";
    let (domain_min, domain_max) = domain(field);

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(error(field, item_start, 6));
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();

        let (base_end, step_start) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some(slash + 1)),
            _ => return Err(error(field, item_start, 0)),
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        enum Base {
            Star,
            Single(usize, usize),
            Range(usize, usize, usize, usize),
        }

        let parsed_base = if base == b"*" {
            Base::Star
        } else {
            match hyphens.as_slice() {
                [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                    Base::Single(0, base.len())
                }
                [hyphen]
                    if *hyphen > 0
                        && *hyphen + 1 < base.len()
                        && base[..*hyphen].iter().all(u8::is_ascii_digit)
                        && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
                {
                    Base::Range(0, *hyphen, *hyphen + 1, base.len())
                }
                _ => return Err(error(field, item_start, 0)),
            }
        };

        let (start_value, end_value, numeric_range) = match parsed_base {
            Base::Star => (domain_min as u32, domain_max as u32, false),
            Base::Single(start, end) => {
                let value = parse_number(base, start, end, field, false)?;
                (value, if step_start.is_some() { domain_max as u32 } else { value }, false)
            }
            Base::Range(start, end, end_start, end_end) => {
                let first = parse_number(base, start, end, field, false)?;
                let last = parse_number(base, end_start, end_end, field, false)?;
                if first > last {
                    return Err(error(field, item_start + end_start, 4));
                }
                (first, last, true)
            }
        };

        let step = if let Some(start) = step_start {
            let end = item.len();
            if start == end || !item[start..].iter().all(u8::is_ascii_digit) {
                return Err(error(field, item_start, 0));
            }
            parse_number(item, start, end, field, true)?
        } else {
            1
        };

        let _ = numeric_range;
        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(exact_star)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let field_order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| vec![false; 60]);
    let mut day_star = [false; 2];

    for i in 0..5 {
        let field = field_order[i];
        let (min, max) = domain(field);
        let mut values = vec![false; max as usize + 1];
        let exact_star = parse_field(fields[i], field, &mut values)?;
        for value in min..=max {
            selected[i][value as usize] = values[value as usize];
        }
        if i == 2 {
            day_star[0] = exact_star;
        } else if i == 4 {
            day_star[1] = exact_star;
        }
    }

    Ok(Cron { selected, day_star })
}

    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end <= start {
        return Err(error(field, start, ErrorKind::InvalidSyntax));
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(field, start, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(digit))
            .ok_or_else(|| error(field, start, ErrorKind::NumberOverflow))?;
    }

    let (min, max) = field_info(field);
    if step {
        let width = max - min + 1;
        if value == 0 || value > width {
            return Err(error(field, start, ErrorKind::BadStep));
        }
    } else if value < min || value > max {
        return Err(error(field, start, ErrorKind::OutOfRange));
    }

    Ok(value)
}

fn parse_field(field: Field, text: &str) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(error(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let (min, max) = field_info(field);
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(error(field, item_start, ErrorKind::EmptyItem));
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        }

        let (base_end, step_range) = if let Some(&slash) = slash_positions.first() {
            (slash, Some((slash + 1, item.len())))
        } else {
            (item.len(), None)
        };

        let hyphens: Vec<usize> = item[..base_end]
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let is_star = base_end == 1 && item[0] == b'*';
        let (range_start, range_end) = if is_star {
            if !hyphens.is_empty() {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
            (None, None)
        } else if hyphens.is_empty() {
            if item[..base_end].is_empty()
                || !item[..base_end].iter().all(u8::is_ascii_digit)
            {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
            (Some((0, base_end)), None)
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen == 0
                || hyphen + 1 == base_end
                || !item[..hyphen].iter().all(u8::is_ascii_digit)
                || !item[hyphen + 1..base_end].iter().all(u8::is_ascii_digit)
            {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
            (Some((0, hyphen)), Some((hyphen + 1, base_end)))
        } else {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        };

        if let Some((step_start, step_end)) = step_range {
            if step_start == step_end
                || !item[step_start..step_end].iter().all(u8::is_ascii_digit)
            {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
        }

        let mut start_value = min;
        let mut end_value = max;

        if let Some((start, end)) = range_start {
            start_value = parse_number(bytes, item_start + start, item_start + end, field, false)?;
            if let Some((end_start, end_end)) = range_end {
                end_value =
                    parse_number(bytes, item_start + end_start, item_start + end_end, field, false)?;
                if start_value > end_value {
                    return Err(error(
                        field,
                        item_start + end_start,
                        ErrorKind::ReversedRange,
                    ));
                }
            } else if step_range.is_none() {
                end_value = start_value;
            }
        }

        let step = if let Some((step_start, step_end)) = step_range {
            parse_number(
                bytes,
                item_start + step_start,
                item_start + step_end,
                field,
                true,
            )?
        } else {
            1
        };

        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &present)| present.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed = Vec::with_capacity(5);

    for (field, text) in field_kinds.into_iter().zip(fields.iter().copied()) {
        parsed.push(parse_field(field, text)?);
    }

    Ok(Cron {
        selected: [
            parsed.remove(0),
            parsed.remove(0),
            parsed.remove(0),
            parsed.remove(0),
            parsed.remove(0),
        ],
        day_of_month_is_star: fields[2] == "*",
        day_of_week_is_star: fields[4] == "*",
    })
}
