            Some(Ticket {
                id: self.jobs[index].id.clone(),
                attempt: entry.attempts,

            None

        Ok(result)

            return Err(Error::NotRunning);

        self.record_time(now);
        Ok(())

                    changed = true;
                }
            }

                break;
            }
        }

            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;
        Ok(self.states[index].state.clone())
    }

fn valid_id(id: &str) -> bool {
    (1..=16).contains(&id.len())
        && id
            .bytes()

}

fn effective_priority(base: u8, ready_at: u64, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return base;
    }
    let increase = (now - ready_at) / aging_ms;
    (base as u64 + increase).min(9) as u8

    BadPriority(usize),
    BadAttempts(usize),

pub struct Runner {
    jobs: Vec<JobData>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    has_cycle: bool,

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut seen = std::collections::HashSet::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if !seen.insert(job.id.as_str()) {
                return Err(Error::Duplicate(i));
            }
        }

        let ids: std::collections::HashMap<&str, usize> = jobs
            .iter()
            .enumerate()
            .map(|(i, job)| (job.id.as_str(), i))
            .collect();

        let mut converted = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.into_iter().enumerate() {
            let mut dep_seen = std::collections::HashSet::new();
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !dep_seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dep.as_str()) {
                    Some(&index) => dependencies.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            converted.push(JobData {
                id: job.id,
                priority: job.priority,
                dependencies,
                max_attempts: job.max_attempts,
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            });
        }

        if contains_cycle(&converted) {
            return Err(Error::Cycle);
        }

        Ok(Self {
            jobs: converted,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.jobs.iter().any(|job| {
            matches!(job.state, State::Waiting { .. } | State::Running { .. })
        }) && self.has_cycle
        {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.state = State::Waiting { ready_at: now };
            job.attempts = 0;
        }
        self.has_cycle = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, job) in self.jobs.iter().enumerate() {
            let ready_at = match job.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !job.dependencies.iter().all(|&dep| {
                matches!(self.jobs[dep].state, State::Succeeded)
            }) {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (job.priority as u64 + age).min(9) as u8
            };
            if best.map_or(true, |(best_i, best_priority)| {
                effective > best_priority || (effective == best_priority && i < best_i)
            }) {
                best = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = best else {
            return Ok(None);
        };

        let job = &mut self.jobs[index];
        job.attempts += 1;
        job.state = State::Running {
            attempt: job.attempts,
        };
        Ok(Some(Ticket {
            id: job.id.clone(),
            attempt: job.attempts,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;
        if !matches!(self.jobs[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        let next_state = if success {
            State::Succeeded
        } else if self.jobs[index].attempts == self.jobs[index].max_attempts {
            State::Failed
        } else {
            let delay = retry_delay(
                self.retry_base,
                self.retry_cap,
                self.jobs[index].attempts,
            );
            State::Waiting {
                ready_at: now.saturating_add(delay),
            }
        };

        self.jobs[index].state = next_state;
        if matches!(self.jobs[index].state, State::Failed) {
            self.propagate_blocked();
        }
        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        self.jobs
            .iter()
            .find(|job| job.id == id)
            .map(|job| job.state.clone())
            .ok_or(Error::Unknown)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn propagate_blocked(&mut self) {
        loop {
            let newly_blocked: Vec<usize> = self
                .jobs
                .iter()
                .enumerate()
                .filter_map(|(i, job)| {
                    if !matches!(job.state, State::Waiting { .. }) {
                        return None;
                    }
                    job.dependencies
                        .iter()
                        .any(|&dep| {
                            matches!(
                                self.jobs[dep].state,
                                State::Failed | State::Blocked
                            )
                        })
                        .then_some(i)
                })
                .collect();

            if newly_blocked.is_empty() {
                break;
            }
            for index in newly_blocked {
                self.jobs[index].state = State::Blocked;
            }
        }
    }
}
