```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobState {
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<Job>,
    dependencies: Vec<Vec<usize>>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    states: Vec<JobState>,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            let valid_id = (1..=16).contains(&job.id.len())
                && job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
            if !valid_id {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.as_str(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dependency.as_str()) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(resolved);
        }

        fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
            fn visit(node: usize, dependencies: &[Vec<usize>], marks: &mut [u8]) -> bool {
                if marks[node] == 1 {
                    return true;
                }
                if marks[node] == 2 {
                    return false;
                }
                marks[node] = 1;
                for &dep in &dependencies[node] {
                    if visit(dep, dependencies, marks) {
                        return true;
                    }
                }
                marks[node] = 2;
                false
            }

            let mut marks = vec![0; dependencies.len()];
            (0..dependencies.len()).any(|node| visit(node, dependencies, &mut marks))
        }

        if has_cycle(&dependencies) {
            return Err(Error::Cycle);
        }

        let states = jobs
            .iter()
            .map(|_| JobState {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            dependencies,
            aging_ms,
            retry_base,
            retry_cap,
            states,
            last_time: 0,
            cycle_started: false,
        })
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn record_time(&mut self, now: u64) {
        self.last_time = now;
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.cycle_started
            && self.states.iter().any(|entry| {
                matches!(entry.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for entry in &mut self.states {
            entry.state = State::Waiting { ready_at: now };
            entry.attempts = 0;
        }
        self.cycle_started = true;
        self.record_time(now);
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (index, job) in self.jobs.iter().enumerate() {
            let ready_at = match self.states[index].state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !self.dependencies[index]
                .iter()
                .all(|&dep| matches!(self.states[dep].state, State::Succeeded))
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.priority
            } else {
                let aging = (now - ready_at) / self.aging_ms;
                (u64::from(job.priority) + aging).min(9) as u8
            };

            if best.map_or(true, |(_, priority)| effective > priority) {
                best = Some((index, effective));
            }
        }

        let result = if let Some((index, _)) = best {
            let entry = &mut self.states[index];
            entry.attempts += 1;
            entry.state = State::Running {
                attempt: entry.attempts,
            };
            Some(Ticket {
                id: self.jobs[index].id.clone(),
                attempt: entry.attempts,
            })
        } else {
            None
        };

        self.record_time(now);
        Ok(result)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;

        if !matches!(self.states[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.states[index].state = State::Succeeded;
        } else if self.states[index].attempts == self.jobs[index].max_attempts {
            self.states[index].state = State::Failed;
        } else {
            let attempt = self.states[index].attempts;
            let mut delay = self.retry_base;
            for _ in 1..attempt {
                delay = delay.saturating_mul(2).min(self.retry_cap);
            }
            delay = delay.min(self.retry_cap);
            self.states[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.propagate_blocked();
        self.record_time(now);
        Ok(())
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for index in 0..self.jobs.len() {
                if !matches!(self.states[index].state, State::Waiting { .. }) {
                    continue;
                }
                if self.dependencies[index].iter().any(|&dep| {
                    matches!(
                        self.states[dep].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    self.states[index].state = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;
        Ok(self.states[index].state.clone())
    }
}
```
