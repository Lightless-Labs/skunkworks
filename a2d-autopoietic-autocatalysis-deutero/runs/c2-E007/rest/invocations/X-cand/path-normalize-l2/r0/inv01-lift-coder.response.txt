#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(String, usize)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let at = *offset;
            *offset += part.len() + 1;
            Some((part, at))
        })
        .filter(|(part, _)| !part.is_empty())
        .map(|(part, at)| (part.to_owned(), at))
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(part, _)| part == "." || part == "..")
}

fn reduce(
    stack: &mut Vec<String>,
    comps: &[(String, usize)],
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in comps {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: *at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(part.clone()),
        }
    }
    Ok(())
}

fn format_result(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let comps = components(path);
    let mut stack = Vec::new();
    reduce(&mut stack, &comps, absolute)?;

    Ok(format_result(&stack, absolute, intent))
}

fn normalized_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return Err(());
    }

    let comps = components(path);
    let mut stack = Vec::new();
    reduce(&mut stack, &comps, true).map_err(|_| ())?;
    Ok(stack)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(normalized_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);

    let (absolute, mut stack, remainder, offset_adjustment) = if path.starts_with('/') {
        (true, Vec::new(), path, 0usize)
    } else if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let base = home_components.ok_or(PathError::MissingHome)?;
        let remainder = if bytes.len() > 1 { &path[2..] } else { "" };
        (true, base, remainder, 2usize)
    } else {
        (true, cwd_components, path, 0usize)
    };

    let comps: Vec<(String, usize)> = components(remainder)
        .into_iter()
        .map(|(part, at)| (part, at + offset_adjustment))
        .collect();
    reduce(&mut stack, &comps, absolute)?;

    Ok(format_result(&stack, true, intent))
}
