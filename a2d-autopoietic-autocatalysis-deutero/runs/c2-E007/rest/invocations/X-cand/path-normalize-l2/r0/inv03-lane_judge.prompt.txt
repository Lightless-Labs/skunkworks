#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .filter(|(_, part)| !part.is_empty())
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, part)| *part == "." || *part == "..")
}

fn append_component(stack: &mut Vec<String>, part: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    match part {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_owned());
            }
        }
        _ => stack.push(part.to_owned()),
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, dir_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        return "/".to_owned();
    }
    if dir_intent && result != "." && result != "/" {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let dir_intent = directory_intent(path);
    let mut stack = Vec::new();

    for (at, part) in components(path) {
        append_component(&mut stack, part, absolute, at)?;
    }

    Ok(render(&stack, absolute, dir_intent))
}

fn validated_absolute(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized)
        .into_iter()
        .map(|(_, part)| part.to_owned())
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validated_absolute(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validated_absolute(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let expanded = if tilde {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            true
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        false
    };

    let (mut stack, start) = if expanded {
        (
            home_components.clone().ok_or(PathError::MissingHome)?,
            if bytes.len() > 1 { 2 } else { 1 },
        )
    } else if bytes.first() == Some(&b'/') {
        (Vec::new(), 0)
    } else {
        (cwd_components, 0)
    };

    let dir_intent = directory_intent(path);
    let remainder = &path[start..];

    for (offset, part) in components(remainder) {
        let at = start + offset;
        append_component(&mut stack, part, true, at)?;
    }

    Ok(render(&stack, true, dir_intent))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> impl Iterator<Item = (usize, &str)> {
    path.match_indices('/')
        .map(|(at, _)| (at, ""))
        .chain(std::iter::once((path.len(), "")))
        .scan(0usize, move |start, (end, _)| {
            let result = (*start, &path[*start..end]);
            *start = end + 1;
            Some(result)
        })
}

fn reduce(
    path: &str,
    absolute: bool,
    initial: Vec<String>,
    offset: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;

    for (start, component) in components(path) {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at: offset + start });
            } else {
                stack.push("..".to_owned());
            }
        } else {
            stack.push(component.to_owned());
        }
    }

    Ok(stack)
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        return result;
    }
    if !absolute && stack.is_empty() {
        return result;
    }
    if intent {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let stack = reduce(path, absolute, Vec::new(), 0)?;
    Ok(render(&stack, absolute, directory_intent(path)))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .trim_start_matches('/')
        .trim_end_matches('/')
        .split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .map(str::to_owned)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd)?;

    let home_components = match home {
        Some(home_path) => {
            if !home_path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(validate_context(home_path).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate_raw(path)?;

    let (absolute, initial, remainder, offset) = if path.starts_with('~') {
        if path.len() > 1 && !path[1..].starts_with('/') {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let initial = home_components.ok_or(PathError::MissingHome)?;
        let remainder = if path == "~" {
            ""
        } else {
            &path[2..]
        };
        (true, initial, remainder, if path == "~" { path.len() } else { 2 })
    } else if path.starts_with('/') {
        (true, Vec::new(), path, 0)
    } else {
        (true, cwd_components, path, 0)
    };

    let stack = reduce(remainder, absolute, initial, offset)?;
    Ok(render(&stack, true, directory_intent(path)))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(String, usize)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let at = *offset;
            *offset += part.len() + 1;
            Some((part, at))
        })
        .filter(|(part, _)| !part.is_empty())
        .map(|(part, at)| (part.to_owned(), at))
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(part, _)| part == "." || part == "..")
}

fn reduce(
    stack: &mut Vec<String>,
    comps: &[(String, usize)],
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in comps {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: *at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(part.clone()),
        }
    }
    Ok(())
}

fn format_result(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let comps = components(path);
    let mut stack = Vec::new();
    reduce(&mut stack, &comps, absolute)?;

    Ok(format_result(&stack, absolute, intent))
}

fn normalized_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return Err(());
    }

    let comps = components(path);
    let mut stack = Vec::new();
    reduce(&mut stack, &comps, true).map_err(|_| ())?;
    Ok(stack)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(normalized_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);

    let (absolute, mut stack, remainder, offset_adjustment) = if path.starts_with('/') {
        (true, Vec::new(), path, 0usize)
    } else if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let base = home_components.ok_or(PathError::MissingHome)?;
        let remainder = if bytes.len() > 1 { &path[2..] } else { "" };
        (true, base, remainder, 2usize)
    } else {
        (true, cwd_components, path, 0usize)
    };

    let comps: Vec<(String, usize)> = components(remainder)
        .into_iter()
        .map(|(part, at)| (part, at + offset_adjustment))
        .collect();
    reduce(&mut stack, &comps, absolute)?;

    Ok(format_result(&stack, true, intent))
}
