#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut mode = Mode::Bare;
    let mut iter = input.char_indices().peekable();

    while let Some((pos, c)) = iter.next() {
        match mode {
            Mode::Single(_) => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    current.text.push(c);
                }
            }
            Mode::Double(_) => {
                match c {
                    '"' => mode = Mode::Bare,
                    '\\' => {
                        let Some(&(next_pos, next)) = iter.peek() else {
                            return Err(SplitError::TrailingEscape { pos });
                        };
                        if matches!(next, '"' | '\\' | '$' | '\n') {
                            iter.next();
                            if next != '\n' {
                                append_literal(&mut current, next, false);
                            }
                        } else {
                            current.text.push('\\');
                            let _ = next_pos;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos }),
                    '$' => {
                        if let Some(&(next_pos, next)) = iter.peek() {
                            if next == '(' {
                                return Err(SplitError::CommandSubstitution { pos });
                            }
                            if is_name_start(next) {
                                active = true;
                                let mut name = String::new();
                                while let Some(&(_, nc)) = iter.peek() {
                                    if is_name_continue(nc) {
                                        name.push(nc);
                                        iter.next();
                                    } else {
                                        break;
                                    }
                                }
                                append_expansion(
                                    &mut current,
                                    lookup(env, &name).unwrap_or(""),
                                    false,
                                );
                                let _ = next_pos;
                            } else if next == '{' {
                                active = true;
                                iter.next();
                                let (value, consumed_end) =
                                    parse_braced(input, pos, env, &mut iter)?;
                                append_expansion(&mut current, &value, false);
                                let _ = consumed_end;
                            } else {
                                current.text.push('$');
                            }
                        } else {
                            current.text.push('$');
                        }
                    }
                    _ => current.text.push(c),
                }
            }
            Mode::Bare => {
                match c {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(current);
                            current = Word {
                                text: String::new(),
                                glob: Vec::new(),
                            };
                            active = false;
                        }
                    }
                    '\'' => {
                        active = true;
                        mode = Mode::Single(pos);
                    }
                    '"' => {
                        active = true;
                        mode = Mode::Double(pos);
                    }
                    '\\' => {
                        let Some(&(_, next)) = iter.peek() else {
                            return Err(SplitError::TrailingEscape { pos });
                        };
                        iter.next();
                        if next == '\n' {
                            continue;
                        }
                        active = true;
                        append_literal(&mut current, next, false);
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos }),
                    '$' => {
                        if let Some(&(_, next)) = iter.peek() {
                            if next == '(' {
                                return Err(SplitError::CommandSubstitution { pos });
                            }
                            if is_name_start(next) {
                                active = true;
                                let mut name = String::new();
                                while let Some(&(_, nc)) = iter.peek() {
                                    if is_name_continue(nc) {
                                        name.push(nc);
                                        iter.next();
                                    } else {
                                        break;
                                    }
                                }
                                append_expansion(
                                    &mut current,
                                    lookup(env, &name).unwrap_or(""),
                                    true,
                                );
                            } else if next == '{' {
                                active = true;
                                iter.next();
                                let (value, _) = parse_braced(input, pos, env, &mut iter)?;
                                append_expansion(&mut current, &value, true);
                            } else {
                                active = true;
                                current.text.push('$');
                            }
                        } else {
                            active = true;
                            current.text.push('$');
                        }
                    }
                    _ => {
                        active = true;
                        append_literal(&mut current, c, true);
                    }
                }
            }
        }
    }

    match mode {
        Mode::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Mode::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Mode::Bare => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn parse_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
    iter: &mut std::iter::Peekable<std::str::CharIndices<'_>>,
) -> Result<(String, usize), SplitError> {
    let content_start = iter.peek().map(|(p, _)| *p).unwrap_or(input.len());
    let mut scan = content_start;
    let mut close = None;

    while scan < input.len() {
        let c = input[scan..].chars().next().unwrap();
        let next_pos = scan + c.len_utf8();
        if c == '\\' {
            if next_pos < input.len() {
                let escaped = input[next_pos..].chars().next().unwrap();
                scan = next_pos + escaped.len_utf8();
            } else {
                scan = next_pos;
            }
        } else if c == '}' {
            close = Some(scan);
            break;
        } else {
            scan = next_pos;
        }
    }

    let Some(close_pos) = close else {
        return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
    };

    let contents = &input[content_start..close_pos];
    let chars: Vec<(usize, char)> = contents.char_indices().collect();
    let mut name_end = 0;
    if chars.first().is_some_and(|(_, c)| is_name_start(*c)) {
        name_end = chars[0].1.len_utf8();
        for (_, c) in chars.iter().skip(1) {
            if is_name_continue(*c) {
                name_end += c.len_utf8();
            } else {
                break;
            }
        }
    }

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];
    let value = if rest.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    let consumed_end = close_pos + 1;
    while iter.peek().is_some_and(|(p, _)| *p < consumed_end) {
        iter.next();
    }
    Ok((value, consumed_end))
}
