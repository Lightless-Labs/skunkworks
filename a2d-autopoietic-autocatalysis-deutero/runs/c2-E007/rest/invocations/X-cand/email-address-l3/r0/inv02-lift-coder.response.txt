#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold(bytes: &[u8], i: usize) -> bool {
    i + 2 < bytes.len()
        && bytes[i] == b'\r'
        && bytes[i + 1] == b'\n'
        && is_hws(bytes[i + 2])
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::with_capacity(bytes.len());
    let mut positions = Vec::with_capacity(bytes.len());
    let mut runs: Vec<Run> = Vec::new();
    let mut i = 0;
    let mut protected = false;
    let mut quote_closed = false;

    while i < bytes.len() {
        if protected {
            let b = bytes[i];
            clean.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    protected = false;
                    quote_closed = true;
                }
                i += 1;
            }
            continue;
        }

        if bytes[i] == b'"' && clean.is_empty() && !quote_closed {
            protected = true;
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        let start = i;
        let mut consumed_cfws = false;

        loop {
            if i >= bytes.len() {
                break;
            }
            if is_hws(bytes[i]) {
                consumed_cfws = true;
                while i < bytes.len() && is_hws(bytes[i]) {
                    i += 1;
                }
                continue;
            }
            if bytes[i] == b'\r' || bytes[i] == b'\n' {
                if !is_fold(bytes, i) {
                    return Err(EmailError::BadFold { at: i });
                }
                consumed_cfws = true;
                i += 2;
                while i < bytes.len() && is_hws(bytes[i]) {
                    i += 1;
                }
                continue;
            }
            if bytes[i] == b'(' {
                consumed_cfws = true;
                let outer = i;
                let mut stack = [0usize; 3];
                let mut depth = 1usize;
                stack[0] = i;
                i += 1;

                while i < bytes.len() {
                    let b = bytes[i];
                    if b == b'\\' {
                        let slash = i;
                        if i + 1 >= bytes.len()
                            || !(0x20..=0x7e).contains(&bytes[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                        continue;
                    }
                    if b == b'\r' || b == b'\n' {
                        if !is_fold(bytes, i) {
                            return Err(EmailError::BadFold { at: i });
                        }
                        i += 2;
                        while i < bytes.len() && is_hws(bytes[i]) {
                            i += 1;
                        }
                        continue;
                    }
                    if b == b'(' {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        stack[depth] = i;
                        depth += 1;
                        i += 1;
                        continue;
                    }
                    if b == b')' {
                        depth -= 1;
                        i += 1;
                        if depth == 0 {
                            break;
                        }
                        continue;
                    }
                    if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                }
                if depth != 0 {
                    return Err(EmailError::UnclosedComment { at: outer });
                }
                continue;
            }
            break;
        }

        if consumed_cfws {
            runs.push(Run { start, end: i });
        }

        if i < bytes.len() {
            if bytes[i] == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
        }
    }

    let n = clean.len();
    for run in &runs {
        let left_at = run.start > 0 && bytes[run.start - 1] == b'@';
        let right_at = run.end < bytes.len() && bytes[run.end] == b'@';
        let at_boundary = run.start == 0 || run.end == bytes.len();
        if !at_boundary && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if n == 0 {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1;
        while j < n {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: positions[0],
            });
        }
    }

    let mut ats = Vec::new();
    let mut j = 0;
    while j < n {
        if let Some(end) = quote_end {
            if j <= end {
                j = end + 1;
                continue;
            }
        }
        if clean[j] == b'@' {
            ats.push(j);
        }
        j += 1;
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: positions[ats[1]],
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[sep],
        });
    }

    let mut local = String::new();
    let quoted;
    if clean[0] == b'"' {
        let end = quote_end.expect("quoted prefix must have a closing quote");
        if end + 1 != sep {
            let at = if end + 1 < n {
                positions[end + 1]
            } else {
                input.len()
            };
            return Err(EmailError::TrailingLocal { at });
        }
        quoted = true;
        let mut decoded: Vec<(u8, usize)> = Vec::new();
        let mut k = 1;
        while k < end {
            let b = clean[k];
            if b == b'\\' {
                if k + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: positions[k],
                    });
                }
                let escaped = clean[k + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: positions[k],
                    });
                }
                decoded.push((escaped, positions[k]));
                k += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: positions[k],
                    });
                }
                decoded.push((b, positions[k]));
                k += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[end],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded[64].1,
            });
        }
        for (b, _) in decoded {
            local.push(b as char);
        }
    } else {
        quoted = false;
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for k in 0..sep {
            let b = clean[k];
            if b == b'.' {
                if k == 0 || k + 1 == sep || clean[k - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: positions[k],
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: positions[k],
                });
            }
            local.push(b as char);
        }
    }

    let domain_start = sep + 1;
    if domain_start >= n {
        return Err(EmailError::EmptyDomain {
            at: input.len(),
        });
    }

    let mut domain_end = n;
    let mut root_dot = None;
    if clean[domain_end - 1] == b'.' {
        root_dot = Some(domain_end - 1);
        domain_end -= 1;
    }
    if domain_start >= domain_end {
        return Err(EmailError::EmptyDomain {
            at: positions[domain_start.min(n - 1)],
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for k in domain_start..domain_end {
        if clean[k] == b'.' {
            if label_start == k {
                return Err(EmailError::EmptyLabel {
                    at: positions[k],
                });
            }
            labels.push((label_start, k));
            label_start = k + 1;
        }
    }
    if label_start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: root_dot.map(|p| positions[p]).unwrap_or(input.len()),
        });
    }
    labels.push((label_start, domain_end));

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[start],
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for k in start..end {
            let b = clean[k];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: positions[k],
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: positions[domain_start],
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|&b| (b as char).to_ascii_lowercase())
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
