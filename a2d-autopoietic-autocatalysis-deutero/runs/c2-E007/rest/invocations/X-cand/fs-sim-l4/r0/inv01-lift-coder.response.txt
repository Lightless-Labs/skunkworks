```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        }
    }

    fn make_path(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let absolute = path.starts_with('/');
        let mut parts = if absolute {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }
        Self::make_path(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        let idx = path.rfind('/').unwrap_or(0);
        if idx == 0 {
            "/".to_string()
        } else {
            path[..idx].to_string()
        }
    }

    fn basename(path: &str) -> String {
        path.rsplit('/').next().unwrap_or("").to_string()
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn is_descendant_or_same(path: &str, ancestor: &str) -> bool {
        if ancestor == "/" {
            return path.starts_with('/');
        }
        path == ancestor
            || path
                .strip_prefix(ancestor)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                *candidate != path
                    && Self::parent(candidate) == path
                    && path != candidate.as_str()
            })
            .cloned()
            .collect()
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        allow_missing_final: bool,
    ) -> Result<String, Error> {
        let mut todo = Self::components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if todo.is_empty() {
                return Ok(Self::make_path(&resolved));
            }

            let name = todo.remove(0);
            let candidate = Self::join(&Self::make_path(&resolved), &name);
            let should_follow = follow_final || !todo.is_empty();

            let Some(entry) = self.entries.get(&candidate) else {
                if allow_missing_final && todo.is_empty() {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            };

            if let Entry::Link(target) = entry {
                if should_follow {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;

                    let link_parent = Self::parent(&candidate);
                    let mut target_path = if target.starts_with('/') {
                        Self::normalize(target, "/")
                    } else {
                        Self::normalize(target, &link_parent)
                    };
                    if !todo.is_empty() {
                        let suffix = todo.iter().map(String::as_str).collect::<Vec<_>>().join("/");
                        if suffix.is_empty() {
                            target_path = Self::normalize(&target_path, "/");
                        } else {
                            target_path = Self::normalize(&format!("{}/{}", target_path, suffix), "/");
                        }
                    }
                    todo = Self::components(&target_path);
                    resolved.clear();
                    continue;
                }
                resolved.push(name);
                continue;
            }

            if !todo.is_empty() && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
            resolved.push(name);
        }
    }

    fn resolve_parent(&self, canonical: &str) -> Result<String, Error> {
        let parent = Self::parent(canonical);
        self.resolve(&parent, true, false)
    }

    fn check_parent_dir(&self, parent: &str) -> Result<(), Error> {
        match self.entries.get(parent) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(parent.to_string())),
            None => Err(Error::Missing(parent.to_string())),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let parent = Self::resolve_parent(&canonical)?;
        self.check_parent_dir(&parent)?;
        let destination = Self::join(&parent, &Self::basename(&canonical));
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, true)?;
        if matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::IsDir(resolved));
        }
        let parent = Self::parent(&resolved);
        self.check_parent_dir(&parent)?;
        self.entries.insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        let mut result = Vec::new();
        for child in self.children(&resolved) {
            let kind = match self.entries.get(&child) {
                Some(Entry::Dir) => Kind::Dir,
                Some(Entry::File(_)) => Kind::File,
                Some(Entry::Link(_)) => Kind::Link,
                None => continue,
            };
            result.push((Self::basename(&child), kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        let source_canonical = Self::normalize(from, &self.cwd);
        Self::validate(to)?;
        let destination_canonical = Self::normalize(to, &self.cwd);

        if source_canonical == "/" || destination_canonical == "/" {
            return Err(Error::Root);
        }

        let source_parent = self.resolve_parent(&source_canonical)?;
        self.check_parent_dir(&source_parent)?;
        let source = Self::join(&source_parent, &Self::basename(&source_canonical));
        let Some(source_entry) = self.entries.get(&source).cloned() else {
            return Err(Error::Missing(source));
        };
        if matches!(source_entry, Entry::Dir)
            && Self::is_descendant_or_same(&self.cwd, &source)
        {
            return Err(Error::Busy(source));
        }

        let destination_parent = self.resolve_parent(&destination_canonical)?;
        self.check_parent_dir(&destination_parent)?;
        let destination = Self::join(
            &destination_parent,
            &Self::basename(&destination_canonical),
        );

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_descendant_or_same(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir)
            && Self::is_descendant_or_same(&destination, &source)
        {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref entry) = destination_entry {
            let src_dir = matches!(source_entry, Entry::Dir);
            let dst_dir = matches!(entry, Entry::Dir);
            if src_dir && !dst_dir {
                return Err(Error::NotDir(destination));
            }
            if !src_dir && dst_dir {
                return Err(Error::IsDir(destination));
            }
            if src_dir && dst_dir && !self.children(&destination).is_empty() {
                return Err(Error::NotEmpty(destination));
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_descendant_or_same(path, &source))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(Entry::Dir) = destination_entry {
            self.entries.remove(&destination);
        }
        for (path, _) in &moved {
            self.entries.remove(path);
        }
        for (path, entry) in moved {
            let suffix = if path == source {
                String::new()
            } else {
                path[source.len()..].to_string()
            };
            let new_path = format!("{}{}", destination, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }
        let parent = self.resolve_parent(&canonical)?;
        self.check_parent_dir(&parent)?;
        let resolved = Self::join(&parent, &Self::basename(&canonical));
        let Some(entry) = self.entries.get(&resolved) else {
            return Err(Error::Missing(resolved));
        };
        if matches!(entry, Entry::Dir) && Self::is_descendant_or_same(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        let descendants = self.children(&resolved);
        if matches!(entry, Entry::Dir) && !recursive && !descendants.is_empty() {
            return Err(Error::NotEmpty(resolved));
        }
        if recursive && matches!(entry, Entry::Dir) {
            let doomed: Vec<String> = self
                .entries
                .keys()
                .filter(|candidate| Self::is_descendant_or_same(candidate, &resolved))
                .cloned()
                .collect();
            for candidate in doomed {
                self.entries.remove(&candidate);
            }
        } else {
            self.entries.remove(&resolved);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let canonical = Self::normalize(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let parent = self.resolve_parent(&canonical)?;
        self.check_parent_dir(&parent)?;
        let resolved = Self::join(&parent, &Self::basename(&canonical));
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries
            .insert(resolved, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        let Some(entry) = self.entries.get(&resolved) else {
            return Err(Error::Missing(resolved));
        };

        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::is_descendant_or_same(candidate, &resolved))
            .cloned()
            .collect();
        paths.sort();

        let mut result = Vec::new();
        for current in paths {
            let size = match self.entries.get(&current) {
                Some(Entry::File(data)) => data.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                Some(Entry::Dir) => self
                    .entries
                    .iter()
                    .filter(|(candidate, _)| {
                        Self::is_descendant_or_same(candidate, &current) && *candidate != &current
                    })
                    .map(|(_, child_entry)| match child_entry {
                        Entry::File(data) => data.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
                None => 0,
            };
            result.push((current, size));
        }

        if result.is_empty() {
            let size = match entry {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            result.push((resolved, size));
        }
        Ok(result)
    }
}
```
