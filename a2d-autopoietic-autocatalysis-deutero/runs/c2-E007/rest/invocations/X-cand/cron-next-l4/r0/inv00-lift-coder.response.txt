use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];
const MINS: [u8; 5] = [0, 0, 1, 1, 0];
const MAXS: [u8; 5] = [59, 23, 31, 12, 6];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn name_value(field: Field, text: &str) -> Option<u8> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT",
            "NOV", "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    names
        .iter()
        .position(|n| n.eq_ignore_ascii_case(text))
        .map(|i| if field == Field::Month { (i + 1) as u8 } else { i as u8 })
}

fn atom_valid(field: Field, text: &str) -> bool {
    !text.is_empty()
        && (text.bytes().all(|b| b.is_ascii_digit()) || name_value(field, text).is_some())
}

fn parse_u32(field: Field, text: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_err(field, offset, 1));
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_err(field, offset, 2))?;
    }
    Ok(value)
}

fn parse_atom(field: Field, text: &str, offset: usize) -> Result<u8, ParseError> {
    if let Some(v) = name_value(field, text) {
        return Ok(v);
    }
    let value = parse_u32(field, text, offset)?;
    if value < MINS[field_index(field)] as u32 || value > MAXS[field_index(field)] as u32 {
        return Err(parse_err(field, offset, 3));
    }
    Ok(value as u8)
}

fn parse_step(field: Field, text: &str, offset: usize) -> Result<u32, ParseError> {
    let value = parse_u32(field, text, offset)?;
    let width = (MAXS[field_index(field)] - MINS[field_index(field)] + 1) as u32;
    if value == 0 || value > width {
        return Err(parse_err(field, offset, 5));
    }
    Ok(value)
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    for (offset, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; (MAXS[field_index(field)] + 1) as usize];
    let mut start_offset = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: start_offset,
            });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(parse_err(field, start_offset, 0));
        }
        let (base, step_text) = match item.split_once('/') {
            Some((base, step)) => (base, Some(step)),
            None => (item, None),
        };
        if base.is_empty() || step_text == Some("") {
            return Err(parse_err(field, start_offset, 0));
        }

        let (base_start, base_end, start_num_offset, end_num_offset) = if base == "*" {
            (None, None, start_offset, start_offset)
        } else {
            let hyphens = base.bytes().filter(|&b| b == b'-').count();
            if hyphens > 1 {
                return Err(parse_err(field, start_offset, 0));
            }
            if let Some((left, right)) = base.split_once('-') {
                if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                    return Err(parse_err(field, start_offset, 0));
                }
                if !atom_valid(field, left) || !atom_valid(field, right) {
                    return Err(parse_err(field, start_offset, 0));
                }
                let right_offset = start_offset + left.len() + 1;
                (
                    Some(left),
                    Some(right),
                    start_offset,
                    right_offset,
                )
            } else {
                if !atom_valid(field, base) {
                    return Err(parse_err(field, start_offset, 0));
                }
                (Some(base), None, start_offset, start_offset)
            }
        };

        let step_offset = step_text.map(|_| start_offset + base.len() + 1);
        let step = match (step_text, step_offset) {
            (Some(s), Some(o)) => Some(parse_step(field, s, o)?),
            _ => None,
        };

        let (lo, hi) = if base == "*" {
            (MINS[field_index(field)], MAXS[field_index(field)])
        } else {
            let first_text = base_start.unwrap();
            let lo = parse_atom(field, first_text, start_num_offset)?;
            let hi = if let Some(last_text) = base_end {
                let end = parse_atom(field, last_text, end_num_offset)?;
                if lo > end {
                    return Err(parse_err(field, end_num_offset, 4));
                }
                end
            } else if step.is_some() {
                MAXS[field_index(field)]
            } else {
                lo
            };
            (lo, hi)
        };

        if let Some(step) = step {
            let mut value = lo as u32;
            while value <= hi as u32 {
                selected[value as usize] = true;
                value += step;
            }
        } else {
            for value in lo..=hi {
                selected[value as usize] = true;
            }
        }
        start_offset += item.len() + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let source = if input.starts_with('@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let parts: Vec<&str> = source.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        selected[i] = parse_field(FIELDS[i], parts[i])?;
    }
    Ok(Cron {
        selected,
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = field_index(field);
        self.selected[i]
            .iter()
            .enumerate()
            .filter_map(|(v, yes)| if *yes { Some(v as u8) } else { None })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.selected[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn day_matches(&self, at: CivilDateTime, weekday: u8) -> bool {
        let dom_unrestricted = self.unrestricted_days[0];
        let dow_unrestricted = self.unrestricted_days[1];
        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday);

        match (dom_unrestricted, dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if at.day == days_in_month(at.year, at.month) => dom && dow,
            (false, false) => dom || dow,
        }
    }

    fn time_month_matches(&self, at: CivilDateTime) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.time_month_matches(at) && self.day_matches(at, weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut day = date_ordinal(after.year, after.month, after.day);
        let last_day = date_ordinal(2399, 12, 31);
        while day <= last_day {
            let (year, month, date) = ordinal_date(day);
            if self.contains(Field::Month, month) {
                let wd = weekday(year, month, date);
                let probe = CivilDateTime { year, month, day: date, hour: 0, minute: 0 };
                if self.day_matches(probe, wd) {
                    for h in 0..24 {
                        if !self.contains(Field::Hour, h) {
                            continue;
                        }
                        for m in 0..60 {
                            if !self.contains(Field::Minute, m) {
                                continue;
                            }
                            let candidate = CivilDateTime { year, month, day: date, hour: h, minute: m };
                            if candidate > after {
                                return Ok(Some(candidate));
                            }
                        }
                    }
                }
            }
            day += 1;
        }
        Ok(None)
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut day = date_ordinal(before.year, before.month, before.day);
        let first_day = date_ordinal(2000, 1, 1);
        loop {
            let (year, month, date) = ordinal_date(day);
            if self.contains(Field::Month, month) {
                let wd = weekday(year, month, date);
                let probe = CivilDateTime { year, month, day: date, hour: 0, minute: 0 };
                if self.day_matches(probe, wd) {
                    for h in (0..24).rev() {
                        if !self.contains(Field::Hour, h) {
                            continue;
                        }
                        for m in (0..60).rev() {
                            if !self.contains(Field::Minute, m) {
                                continue;
                            }
                            let candidate = CivilDateTime { year, month, day: date, hour: h, minute: m };
                            if candidate < before {
                                return Ok(Some(candidate));
                            }
                        }
                    }
                }
            }
            if day == first_day {
                break;
            }
            day -= 1;
        }
        Ok(None)
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        if start == end {
            return Ok(FireWindow { fires: Vec::new(), truncated: false });
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        loop {
            let Some(next) = self.next_fire(cursor).map_err(|_| unreachable!())? else {
                break;
            };
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow { fires, truncated: true });
            }
            fires.push(next);
            cursor = next;
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for i in 0..5 {
            let field = FIELDS[i];
            let min = MINS[i];
            let max = MAXS[i];
            let full = (min..=max).all(|v| self.selected[i][v as usize]);
            let rendered = if i == 2 && self.unrestricted_days[0] {
                "*".to_string()
            } else if i == 4 && self.unrestricted_days[1] {
                "*".to_string()
            } else if full && matches!(field, Field::Minute | Field::Hour | Field::Month) {
                "*".to_string()
            } else if full && matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                "*/1".to_string()
            } else {
                self.allowed(field)
                    .into_iter()
                    .filter(|v| *v >= min && *v <= max)
                    .map(|v| v.to_string())
                    .collect::<Vec<_>>()
                    .join(",")
            };
            fields.push(rendered);
        }
        fields.join(" ")
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    ((date_ordinal(year, month, day) + 6) % 7) as u8
}

fn date_ordinal(year: u16, month: u8, day: u8) -> i64 {
    let y = year as i64;
    let mut days = (y - 2000) * 365;
    let mut yy = 2000;
    while yy < year {
        if is_leap(yy) {
            days += 1;
        }
        yy += 1;
    }
    for m in 1..month {
        days += days_in_month(year, m) as i64;
    }
    days + (day as i64 - 1)
}

fn ordinal_date(mut ordinal: i64) -> (u16, u8, u8) {
    let mut year = 2000u16;
    loop {
        let year_days = if is_leap(year) { 366 } else { 365 };
        if ordinal < year_days {
            break;
        }
        ordinal -= year_days;
        year += 1;
    }
    let mut month = 1u8;
    loop {
        let month_days = days_in_month(year, month) as i64;
        if ordinal < month_days {
            break;
        }
        ordinal -= month_days;
        month += 1;
    }
    (year, month, ordinal as u8 + 1)
}
