#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn name_value(field: Field, text: &str) -> Option<u32> {
    names(field)?.iter().position(|name| name.eq_ignore_ascii_case(text)).map(|n| {
        if field == Field::Month {
            n as u32 + 1
        } else {
            n as u32
        }
    })
}

#[derive(Clone, Copy)]
enum Atom {
    Number(u32),
    Name(u32),
}

impl Atom {
    fn value(self) -> u32 {
        match self {
            Atom::Number(v) | Atom::Name(v) => v,
        }
    }

    fn is_number(self) -> bool {
        matches!(self, Atom::Number(_))
    }
}

fn syntax_atom(s: &str, field: Field) -> Option<Atom> {
    if !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit()) {
        return Some(Atom::Number(0));
    }
    name_value(field, s).map(Atom::Name)
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (min, max) = bounds(field);
    if value < min || value > max {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(value)
}

fn parse_step(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (min, max) = bounds(field);
    let width = max - min + 1;
    if value < 1 || value > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field, index: usize) -> Result<Vec<bool>, ParseError> {
    for (offset, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (base, step_text) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((&item[pos + 1..], item_start + pos + 1)))
        } else {
            (item, None)
        };
        if base.is_empty() || step_text.is_some_and(|(s, _)| s.is_empty()) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let is_star = base == "*";
        let hyphens = base.bytes().filter(|&b| b == b'-').count();
        if !is_star && hyphens > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (left_text, right_text) = if is_star {
            (None, None)
        } else if let Some(pos) = base.find('-') {
            (&base[..pos], Some(&base[pos + 1..]))
        } else {
            (base, None)
        };

        let mut start_offset = item_start;
        let mut end_offset = 0usize;
        let left_atom = if let Some(left) = left_text {
            syntax_atom(left, field).ok_or(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            })?
            .into()
        } else {
            None
        };
        let right_atom = if let Some(right) = right_text {
            let pos = base.find('-').unwrap();
            end_offset = item_start + pos + 1;
            Some(syntax_atom(right, field).ok_or(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            })?)
        } else {
            None
        };

        if let Some((step, step_offset)) = step_text {
            if !step.bytes().all(|b| b.is_ascii_digit()) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }
        if is_star && step_text.is_none() {
            for v in min..=max {
                selected[v as usize] = true;
            }
            item_start += item.len() + 1;
            continue;
        }

        let start = if is_star {
            min
        } else {
            let atom = left_atom.unwrap();
            if atom.is_number() {
                parse_number(&base[..base.find('-').unwrap_or(base.len())], field, start_offset)?
            } else {
                atom.value()
            }
        };

        let end = if let Some(atom) = right_atom {
            if atom.is_number() {
                let pos = base.find('-').unwrap();
                parse_number(&base[pos + 1..], field, end_offset)?
            } else {
                atom.value()
            }
        } else if is_star || step_text.is_some() {
            max
        } else {
            start
        };

        if !is_star && left_atom.unwrap().is_number() {
            // parse_number has already validated the start
        }
        if let Some(atom) = right_atom {
            if atom.is_number() {
                // parse_number has already validated the end
            }
        }

        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: if right_text.is_some() { end_offset } else { item_start },
            });
        }

        let step = if let Some((step_text, step_offset)) = step_text {
            parse_step(step_text, field, step_offset)?
        } else {
            1
        };

        let mut v = start;
        while v <= end {
            selected[v as usize] = true;
            match v.checked_add(step) {
                Some(next) => v = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }

    let _ = index;
    Ok(selected)
}

impl From<Atom> for Option<Atom> {
    fn from(value: Atom) -> Self {
        Some(value)
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let pieces: Vec<&str> = expanded.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        selected[i] = parse_field(pieces[i], FIELDS[i], i)?;
    }

    Ok(Cron {
        selected,
        unrestricted_dom: pieces[2] == "*",
        unrestricted_dow: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.selected[i]
            .iter()
            .enumerate()
            .filter_map(|(v, &yes)| if yes { Some(v as u8) } else { None })
            .collect()
    }

    fn contains(&self, index: usize, value: u8) -> bool {
        self.selected[index]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn day_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.contains(2, at.day);
        let dow = self.contains(4, weekday(at.year, at.month, at.day));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                let last = at.day == month_length(at.year, at.month);
                if last {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.contains(0, at.minute)
            && self.contains(1, at.hour)
            && self.contains(3, at.month)
            && self.day_matches(at)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        while let Some(next) = increment_minute(current) {
            current = next;
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
        Ok(None)
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut current = before;
        while let Some(prev) = decrement_minute(current) {
            current = prev;
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
        Ok(None)
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut current = start;
        while current < end {
            if self.matches_valid(current) {
                if fires.len() == cap {
                    return Ok(FireWindow {
                        fires,
                        truncated: true,
                    });
                }
                fires.push(current);
            }
            match increment_minute(current) {
                Some(next) => current = next,
                None => break,
            }
        }

        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut parts = Vec::with_capacity(5);
        for i in 0..5 {
            let values: Vec<usize> = self.selected[i]
                .iter()
                .enumerate()
                .filter_map(|(v, &yes)| if yes { Some(v) } else { None })
                .collect();
            let field = FIELDS[i];
            let (min, max) = bounds(field);
            let full = values.len() == (max - min + 1) as usize
                && values.first().copied() == Some(min as usize)
                && values.last().copied() == Some(max as usize);
            if i == 2 && self.unrestricted_dom || i == 4 && self.unrestricted_dow {
                parts.push("*".to_string());
            } else if full && matches!(field, Field::Minute | Field::Hour | Field::Month) {
                parts.push("*".to_string());
            } else if full && matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                parts.push("*/1".to_string());
            } else {
                parts.push(
                    values
                        .iter()
                        .map(|v| v.to_string())
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        parts.join(" ")
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn increment_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        at.minute += 1;
        return Some(at);
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return Some(at);
    }
    at.hour = 0;
    if at.day < month_length(at.year, at.month) {
        at.day += 1;
        return Some(at);
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return Some(at);
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return Some(at);
    }
    None
}

fn decrement_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute > 0 {
        at.minute -= 1;
        return Some(at);
    }
    at.minute = 59;
    if at.hour > 0 {
        at.hour -= 1;
        return Some(at);
    }
    at.hour = 23;
    if at.day > 1 {
        at.day -= 1;
        return Some(at);
    }
    if at.month > 1 {
        at.month -= 1;
    } else {
        if at.year <= 2000 {
            return None;
        }
        at.year -= 1;
        at.month = 12;
    }
    at.day = month_length(at.year, at.month);
    Some(at)
}
