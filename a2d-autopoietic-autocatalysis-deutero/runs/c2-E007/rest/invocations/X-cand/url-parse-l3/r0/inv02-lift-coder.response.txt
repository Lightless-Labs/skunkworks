use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn scan_url_bytes(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let bad_at = error.valid_up_to();
            let original_at = origins.get(bad_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_url_component(
    component: Component,
    text: &str,
    input_offset: usize,
) -> Result<(), ParseError> {
    match decode_component(component, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape {
            at: input_offset + at,
        }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 {
            at: input_offset + at,
        }),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_url_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_relative) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_start = authority_start + host_start_relative;
    let remaining = &authority[host_start_relative..];
    let remaining_bytes = remaining.as_bytes();

    let (host, port, ipv6) = if remaining_bytes.first() == Some(&b'[') {
        let closing = remaining_bytes.iter().position(|&b| b == b']');
        let closing = match closing {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &remaining[1..closing];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }

        let suffix = &remaining[closing + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(&suffix[1..], host_start + closing + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + closing + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let host_port_colon = remaining_bytes.iter().position(|&b| b == b':');
        let (host_raw, port) = if let Some(i) = host_port_colon {
            (
                &remaining[..i],
                Some(parse_port(&remaining[i + 1..], host_start + i + 1)?),
            )
        } else {
            (remaining, None)
        };

        if host_raw.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, byte) in host_raw.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + i,
                });
            }
        }
        (host_raw.to_ascii_lowercase(), port, false)
    };

    let fragment_delimiter = input[authority_end..]
        .find('#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_delimiter.unwrap_or(input.len());
    let query_delimiter = input[authority_end..before_fragment_end]
        .find('?')
        .map(|offset| authority_end + offset);

    let path_end = query_delimiter.unwrap_or(before_fragment_end);
    let path = if authority_end == path_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_delimiter.map(|at| {
        let end = fragment_delimiter.unwrap_or(input.len());
        input[at + 1..end].to_owned()
    });
    let fragment = fragment_delimiter.map(|at| input[at + 1..].to_owned());

    if let Some(userinfo_text) = &userinfo {
        validate_url_component(Component::UserInfo, userinfo_text, authority_start)?;
    }
    if path != "/" || authority_end != path_end {
        validate_url_component(Component::Path, &path, authority_end)?;
    }
    if let (Some(at), Some(query_text)) = (query_delimiter, &query) {
        validate_url_component(Component::Query, query_text, at + 1)?;
    }
    if let (Some(at), Some(fragment_text)) = (fragment_delimiter, &fragment) {
        validate_url_component(Component::Fragment, fragment_text, at + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (offset, byte) in port_text.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }
    if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port_text.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|number| number.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).expect("path was validated");
        if decoded == "." {
            terminal_navigation = index + 1 == segments.len();
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            terminal_navigation = index + 1 == segments.len();
        } else {
            stack.push(segment);
            terminal_navigation = false;
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        let mut result = String::from("/");
        result.push_str(&stack.join("/"));
        Ok(result)
    }
}

fn path_error_to_resolve(error: PathError) -> ResolveError {
    ResolveError::Path(error)
}

fn normalize_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(path_error_to_resolve)?;
    Ok(url)
}

fn reference_global_scan(reference: &str) -> Result<(), ParseError> {
    scan_url_bytes(reference)
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_at = reference.find('#');
    let before_fragment = fragment_at.map(|at| &reference[..at]).unwrap_or(reference);
    let fragment = fragment_at.map(|at| &reference[at + 1..]);
    let query_at = before_fragment.find('?');
    let path = query_at
        .map(|at| &before_fragment[..at])
        .unwrap_or(before_fragment);
    let query = query_at.map(|at| &before_fragment[at + 1..]);
    (path, query, fragment)
}

fn assemble_url(
    scheme: &str,
    userinfo: Option<&str>,
    host: &str,
    ipv6: bool,
    port: Option<u16>,
    path: &str,
    query: Option<&str>,
    fragment: Option<&str>,
) -> String {
    let mut result = String::new();
    result.push_str(scheme);
    result.push_str("://");
    if let Some(userinfo) = userinfo {
        result.push_str(userinfo);
        result.push('@');
    }
    if ipv6 {
        result.push('[');
        result.push_str(host);
        result.push(']');
    } else {
        result.push_str(host);
    }
    if let Some(port) = port {
        result.push(':');
        result.push_str(&port.to_string());
    }
    result.push_str(path);
    if let Some(query) = query {
        result.push('?');
        result.push_str(query);
    }
    if let Some(fragment) = fragment {
        result.push('#');
        result.push_str(fragment);
    }
    result
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalize_url_path(base_url)?;
    reference_global_scan(reference).map_err(ResolveError::Reference)?;

    if reference.is_empty() {
        let mut result = base_url;
        result.fragment = None;
        return Ok(result);
    }

    let first_delimiter = reference
        .bytes()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference.as_bytes()[..first_delimiter].contains(&b':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);
    let selected_path;
    let selected_query;

    if reference_path.is_empty() {
        selected_path = base_url.path.clone();
        selected_query = if reference_query.is_none() {
            base_url.query.as_deref()
        } else {
            reference_query
        };
    } else {
        selected_query = reference_query;
        if reference_path.starts_with('/') {
            selected_path = reference_path.to_owned();
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            let mut merged = base_url.path[..=slash].to_owned();
            merged.push_str(reference_path);
            selected_path = merged;
        }
    }

    let constructed = assemble_url(
        &base_url.scheme,
        base_url.userinfo.as_deref(),
        &base_url.host,
        base_url.ipv6,
        base_url.port,
        &selected_path,
        selected_query,
        reference_fragment,
    );
    let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalize_url_path(url)
}
