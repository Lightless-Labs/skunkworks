use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    let mut records = Vec::new();
    if start == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut field_started = false;
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: closed quote
    let mut opening_quote = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != 2 && b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
        }

        if state == 2 {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                state = 3;
                i += 1;
                continue;
            }
            field.push(b as char);
            i += 1;
            continue;
        }

        if state == 3 {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                field_started = false;
                state = 0;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                field_started = false;
                state = 0;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            state = 0;
            i += 1;
            continue;
        }
        if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            field_started = false;
            state = 0;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }
        if b == quote {
            if state == 0 && !field_started {
                state = 2;
                opening_quote = i;
                field_started = true;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote {
                record: record_index,
                byte: i,
            });
        }

        field.push(b as char);
        field_started = true;
        state = 1;
        i += 1;
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if !record.is_empty() || field_started || !field.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

fn duplicate_header_error(
    header: &[String],
) -> Option<CsvError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name.as_str(), second);
    }
    None
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;
    let mut records = parse_with_dialect(input, config.delimiter, config.quote)?;

    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header_error(header) {
                return Err(error);
            }
        }

        for (i, row) in records.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let digits = if b.first() == Some(&b'-') {
        &b[1..]
    } else {
        b
    };
    let Some(dot) = digits.iter().position(|&b| b == b'.') else {
        return false;
    };
    if digits[dot + 1..].len() != 2
        || !digits[dot + 1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    let whole = &digits[..dot];
    if whole == b"0" {
        return b.first() != Some(&b'-') || digits[dot + 1..] != *b"00";
    }
    !whole.is_empty()
        && whole[0].is_ascii_digit()
        && whole[0] != b'0'
        && whole.iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let mut parts = unsigned.split('.');
    let whole = parts.next().unwrap_or("");
    let fraction = parts.next().unwrap_or("");
    let magnitude_text = format!("{}{}", whole, fraction);
    let magnitude = magnitude_text
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut out = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut values = Vec::with_capacity(columns.len());
        let mut had_error = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let Some(column_index) = resolved else {
                values.push(None);
                out.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                had_error = true;
                if policy != ErrorPolicy::Collect {
                    break;
                }
                continue;
            };

            let Some(cell) = row.get(column_index) else {
                values.push(None);
                out.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::MissingCell,
                });
                had_error = true;
                if policy != ErrorPolicy::Collect {
                    break;
                }
                continue;
            };

            if cell.is_empty() {
                if spec.nullable {
                    values.push(Some(Value::Null));
                    continue;
                }
                if spec.kind != ColumnType::Text {
                    values.push(None);
                    out.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind: CellErrorKind::Empty,
                    });
                    had_error = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                    continue;
                }
            }

            let converted = match spec.kind {
                ColumnType::Text => Ok(Value::Text(cell.clone())),
                ColumnType::Integer => parse_integer(cell).map(Value::Integer),
                ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
                ColumnType::Boolean => match cell.as_str() {
                    "yes" | "1" => Ok(Value::Boolean(true)),
                    "no" | "0" => Ok(Value::Boolean(false)),
                    _ => Err(CellErrorKind::InvalidBoolean),
                },
            };

            match converted {
                Ok(value) => values.push(Some(value)),
                Err(kind) => {
                    values.push(None);
                    out.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    had_error = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }

        out.rows.push(values);
        if had_error {
            match policy {
                ErrorPolicy::Collect | ErrorPolicy::FirstPerRow => {}
                ErrorPolicy::Stop => break 'rows,
            }
        }
    }

    out
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns {
        record: usize,
    },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn all_original_records<'a>(table: &'a Table) -> Vec<(usize, &'a Vec<String>)> {
    let mut result = Vec::new();
    if let Some(header) = &table.header {
        result.push((0, header));
    }
    let base = usize::from(table.header.is_some());
    for (i, row) in table.rows.iter().enumerate() {
        result.push((base + i, row));
    }
    result
}

fn compare_rows(a: &Vec<String>, b: &Vec<String>) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let order = x.as_bytes().cmp(y.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

fn quote_field(field: &str, delimiter: u8, quote: u8, first: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let starts_bom = first && field.starts_with('\u{feff}');
    let must_quote = field.as_bytes().contains(&delimiter)
        || field.as_bytes().contains(&quote)
        || field.as_bytes().contains(&b'\r')
        || field.as_bytes().contains(&b'\n')
        || edge_space
        || starts_bom
        || (sole && field.is_empty());

    if !must_quote {
        return field.to_string();
    }

    let mut result = String::new();
    result.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            result.push(quote as char);
            result.push(quote as char);
        } else {
            result.push(ch);
        }
    }
    result.push(quote as char);
    result
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let original_records = all_original_records(table);

    for (record_index, row) in &original_records {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: *record_index,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, row) in &original_records {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: *record_index,
            });
        }
    }

    let mut output_header: Option<Vec<String>> = table.header.clone();
    let mut output_rows = table.rows.clone();

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                let mut seen: HashMap<&str, usize> = HashMap::new();
                for (second, name) in header.iter().enumerate() {
                    if let Some(&first) = seen.get(name.as_str()) {
                        return Err(WriteError::DuplicateHeader {
                            name: name.clone(),
                            first,
                            second,
                        });
                    }
                    seen.insert(name.as_str(), second);
                }
            }

            let mut chosen: HashMap<&str, usize> = HashMap::new();
            for (i, name) in header.iter().enumerate() {
                match table.duplicates {
                    DuplicatePolicy::KeepLast => {
                        chosen.insert(name.as_str(), i);
                    }
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        chosen.entry(name.as_str()).or_insert(i);
                    }
                }
            }
            let mut indexes: Vec<usize> = chosen.into_values().collect();
            indexes.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));

            output_header = Some(indexes.iter().map(|&i| header[i].clone()).collect());
            output_rows = table
                .rows
                .iter()
                .map(|row| indexes.iter().map(|&i| row[i].clone()).collect())
                .collect();
        }
        output_rows.sort_by(compare_rows);
    }

    let mut records = Vec::new();
    if let Some(header) = output_header {
        records.push(header);
    }
    records.extend(output_rows);

    if records.is_empty() {
        return Ok(String::new());
    }

    let (line_ending, final_terminator) = match config.mode {
        WriteMode::Preserve => (
            match config.line_ending {
                LineEnding::Lf => "\n",
                LineEnding::CrLf => "\r\n",
            },
            config.final_terminator,
        ),
        WriteMode::Canonical => ("\n", true),
    };

    let mut serialized = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            serialized.push_str(line_ending);
        }
        let sole = row.len() == 1;
        for (column, field) in row.iter().enumerate() {
            if column > 0 {
                serialized.push(config.delimiter as char);
            }
            let first_field_first_record = record_index == 0 && column == 0;
            serialized.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                first_field_first_record,
                sole,
            ));
        }
    }
    if final_terminator {
        serialized.push_str(line_ending);
    }
    Ok(serialized)
}
