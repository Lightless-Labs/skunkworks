pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|item| item == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|item| item == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quoted_open = 0usize;
    let mut record_index = 0usize;
    let mut record_started = false;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if in_quotes {
            if b == quote {
                if pos + 1 < bytes.len() && bytes[pos + 1] == quote {
                    field.push(quote as char);
                    pos += 2;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                pos += 1;
                continue;
            }
            if b < 0x80 {
                field.push(b as char);
                pos += 1;
            } else {
                let ch = input[pos..].chars().next().unwrap();
                field.push(ch);
                pos += ch.len_utf8();
            }
            continue;
        }

        if b == b'\r' {
            if pos + 1 >= bytes.len() || bytes[pos + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: pos,
                });
            }
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            pos += 2;
            after_quote = false;
            record_started = false;
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                pos += 1;
                after_quote = false;
                record_started = true;
                continue;
            }
            if b == b'\n' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_index += 1;
                pos += 1;
                after_quote = false;
                record_started = false;
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: pos,
            });
        }

        if b == b'\n' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            pos += 1;
            record_started = false;
            continue;
        }
        if b == delimiter {
            row.push(std::mem::take(&mut field));
            pos += 1;
            record_started = true;
            continue;
        }
        if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quoted_open = pos;
                record_started = true;
                pos += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote {
                record: record_index,
                byte: pos,
            });
        }
        if b < 0x80 {
            field.push(b as char);
            pos += 1;
        } else {
            let ch = input[pos..].chars().next().unwrap();
            field.push(ch);
            pos += ch.len_utf8();
        }
        record_started = true;
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quoted_open,
        });
    }

    if record_started || after_quote || !row.is_empty() || !field.is_empty() {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn table_from_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote).map_err(|component| {
        CsvError::InvalidConfig { component }
    })?;
    let mut records = parse_with_dialect(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref names) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..names.len() {
                if let Some(first) = names[..second].iter().position(|n| n == &names[second]) {
                    return Err(CsvError::DuplicateHeader {
                        name: names[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }
        for (i, row) in records.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = table_from_csv(&self.input, self.config);
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }
    if bytes[start] == b'0' {
        return start + 1 == bytes.len() && start == 0;
    }
    (b'1'..=b'9').contains(&bytes[start])
        && bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let start = usize::from(bytes.first() == Some(&b'-'));
    if start >= bytes.len() {
        return false;
    }
    let dot = match bytes[start..].iter().position(|&b| b == b'.') {
        Some(i) => start + i,
        None => return false,
    };
    if bytes.len() != dot + 3 {
        return false;
    }
    if !bytes[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if dot == start {
        return false;
    }
    if bytes[start] == b'0' {
        dot == start + 1
    } else {
        (b'1'..=b'9').contains(&bytes[start])
            && bytes[start + 1..dot].iter().all(u8::is_ascii_digit)
    }
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) || s == "-0.00" {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let bytes = s.as_bytes();
    let negative = bytes[0] == b'-';
    let start = usize::from(negative);
    let dot = bytes.iter().position(|&b| b == b'.').unwrap();

    // Syntax has already been checked. Accumulate against the appropriate
    // magnitude limit so arbitrarily long inputs return DecimalOverflow
    // without overflowing an intermediate integer.
    let limit: u64 = if negative {
        (i64::MAX as u64) + 1
    } else {
        i64::MAX as u64
    };
    let mut magnitude = 0u64;

    for &byte in &bytes[start..dot] {
        let digit = (byte - b'0') as u64;
        magnitude = magnitude
            .checked_mul(10)
            .and_then(|value| value.checked_add(digit))
            .filter(|&value| value <= limit / 100)
            .ok_or(CellErrorKind::DecimalOverflow)?;
    }

    let fraction = ((bytes[dot + 1] - b'0') as u64) * 10
        + (bytes[dot + 2] - b'0') as u64;
    magnitude = magnitude
        .checked_mul(100)
        .and_then(|value| value.checked_add(fraction))
        .filter(|&value| value <= limit)
        .ok_or(CellErrorKind::DecimalOverflow)?;

    if negative {
        if magnitude == (i64::MAX as u64) + 1 {
            Ok(i64::MIN)
        } else {
            Ok(-(magnitude as i64))
        }
    } else {
        Ok(magnitude as i64)
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut row_failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed && policy == ErrorPolicy::FirstPerRow {
                out.push(None);
                continue;
            }
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (value, error) = match resolved {
                None => (
                    None,
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    }),
                ),
                Some(column) => match row.get(column) {
                    None => (
                        None,
                        Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(column),
                            kind: CellErrorKind::MissingCell,
                        }),
                    ),
                    Some(cell) if cell.is_empty() && spec.nullable => {
                        (Some(Value::Null), None)
                    }
                    Some(cell) if cell.is_empty() && spec.kind != ColumnType::Text => (
                        None,
                        Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(column),
                            kind: CellErrorKind::Empty,
                        }),
                    ),
                    Some(cell) => {
                        let parsed = match spec.kind {
                            ColumnType::Text => Ok(Value::Text(cell.clone())),
                            ColumnType::Integer => parse_integer(cell).map(Value::Integer),
                            ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
                            ColumnType::Boolean => match cell.as_str() {
                                "yes" | "1" => Ok(Value::Boolean(true)),
                                "no" | "0" => Ok(Value::Boolean(false)),
                                _ => Err(CellErrorKind::InvalidBoolean),
                            },
                        };
                        match parsed {
                            Ok(v) => (Some(v), None),
                            Err(kind) => (
                                None,
                                Some(CellError {
                                    row: row_index,
                                    spec: spec_index,
                                    column: Some(column),
                                    kind,
                                }),
                            ),
                        }
                    }
                },
            };

            out.push(value);
            if let Some(error) = error {
                result.errors.push(error);
                row_failed = true;
                if policy == ErrorPolicy::Stop {
                    out.extend(std::iter::repeat(None).take(columns.len() - out.len()));
                    result.rows.push(out);
                    return result;
                }
            }
        }
        result.rows.push(out);
    }

    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn check_writer_table(table: &Table) -> Result<(), WriteError> {
    let mut records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        records.push(header);
    }
    records.extend(table.rows.iter());

    for (record_index, record) in records.iter().enumerate() {
        for (column_index, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    Ok(())
}

fn encode_field(
    out: &mut String,
    field: &str,
    first_field_first_record: bool,
    sole_empty: bool,
    delimiter: u8,
    quote: u8,
) {
    let bytes = field.as_bytes();
    let edge_space = bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t');
    let bom_data = first_field_first_record && field.starts_with('\u{feff}');
    let needs_quotes = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || edge_space
        || bom_data
        || (sole_empty && field.is_empty());

    if !needs_quotes {
        out.push_str(field);
        return;
    }

    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;
    check_writer_table(table)?;

    if config.mode == WriteMode::Canonical {
        if table.header.is_some() && table.duplicates == DuplicatePolicy::Reject {
            let header = table.header.as_ref().unwrap();
            for second in 0..header.len() {
                if let Some(first) = header[..second].iter().position(|v| v == &header[second]) {
                    return Err(WriteError::DuplicateHeader {
                        name: header[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }
    }

    let mut header_storage: Option<Vec<String>> = None;
    let mut rows_storage: Vec<Vec<String>> = Vec::new();

    if config.mode == WriteMode::Preserve {
        if let Some(header) = &table.header {
            header_storage = Some(header.clone());
        }
        rows_storage = table.rows.clone();
    } else if let Some(header) = &table.header {
        let mut retained: Vec<usize> = Vec::new();
        match table.duplicates {
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                for i in 0..header.len() {
                    if !header[..i].iter().any(|name| name == &header[i]) {
                        retained.push(i);
                    }
                }
            }
            DuplicatePolicy::KeepLast => {
                for i in 0..header.len() {
                    if !header[i + 1..].iter().any(|name| name == &header[i]) {
                        retained.push(i);
                    }
                }
            }
        }
        retained.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
        header_storage = Some(retained.iter().map(|&i| header[i].clone()).collect());
        rows_storage = table
            .rows
            .iter()
            .map(|row| retained.iter().map(|&i| row[i].clone()).collect())
            .collect();
        rows_storage.sort_by(|a, b| compare_rows(a, b));
    } else {
        rows_storage = table.rows.clone();
        rows_storage.sort_by(|a, b| compare_rows(a, b));
    }

    let mut records: Vec<&[String]> = Vec::new();
    if let Some(header) = &header_storage {
        records.push(header.as_slice());
    }
    for row in &rows_storage {
        records.push(row.as_slice());
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let terminator = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            out.push_str(terminator);
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            encode_field(
                &mut out,
                field,
                record_index == 0 && column_index == 0,
                record.len() == 1,
                config.delimiter,
                config.quote,
            );
        }
    }

    let add_final = canonical || config.final_terminator;
    if add_final {
        out.push_str(terminator);
    }

    Ok(out)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}
