```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                let component = &path[start..i];
                if component.is_empty() || component == "." || component == ".." {
                    let offset = if component.is_empty() {
                        i.saturating_sub(1)
                    } else {
                        start
                    };
                    return Err(Error::InvalidPath(offset));
                }
                components.push(component);
                start = i + 1;
            }
        }

        Ok(components)
    }

    fn canonical(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            let mut result = String::new();
            for component in components {
                result.push('/');
                result.push_str(component);
            }
            result
        }
    }

    fn get<'a>(root: &'a Entry, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = root;
        let mut prefix = Vec::new();

        for (index, component) in components.iter().enumerate() {
            let path = Self::canonical(&prefix);
            let children = match current {
                Entry::Dir(children) => children,
                Entry::File(_) => return Err(Error::NotDir(path)),
            };
            prefix.push(component);
            current = children
                .get(*component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;

            if index + 1 < components.len() && matches!(current, Entry::File(_)) {
                return Err(Error::NotDir(Self::canonical(&prefix)));
            }
        }
        Ok(current)
    }

    fn get_mut<'a>(root: &'a mut Entry, components: &[&str]) -> Result<&'a mut Entry, Error> {
        let mut current = root;
        let mut prefix = Vec::new();

        for (index, component) in components.iter().enumerate() {
            let path = Self::canonical(&prefix);
            let children = match current {
                Entry::Dir(children) => children,
                Entry::File(_) => return Err(Error::NotDir(path)),
            };
            prefix.push(component);
            current = children
                .get_mut(*component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;

            if index + 1 < components.len() && matches!(current, Entry::File(_)) {
                return Err(Error::NotDir(Self::canonical(&prefix)));
            }
        }
        Ok(current)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let (parent, name) = components.split_at(components.len() - 1);
        let parent_path = Self::canonical(parent);
        let parent_entry = Self::get_mut(&mut self.root, parent)?;
        let children = match parent_entry {
            Entry::Dir(children) => children,
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
        };

        let canonical_path = Self::canonical(&components);
        if children.contains_key(name[0]) {
            return Err(Error::Exists(canonical_path));
        }
        children.insert(name[0].to_owned(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let (parent, name) = components.split_at(components.len() - 1);
        let parent_path = Self::canonical(parent);
        let parent_entry = Self::get_mut(&mut self.root, parent)?;
        let children = match parent_entry {
            Entry::Dir(children) => children,
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
        };

        match children.get_mut(name[0]) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(Self::canonical(&components))),
            Some(Entry::File(bytes)) => {
                bytes.clear();
                bytes.extend_from_slice(data);
                Ok(())
            }
            None => {
                children.insert(name[0].to_owned(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        match Self::get(&self.root, &components)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical(&components))),
            Entry::File(bytes) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        match Self::get(&self.root, &components)? {
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
