use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{self:?}")
    }
}

impl std::error::Error for Error {}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn digit(byte: u8, alphabet: Alphabet, strict: bool, offset: usize) -> Result<u8, Error> {
    let value = match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    };

    if let Some(value) = value {
        return Ok(value);
    }

    let foreign = match alphabet {
        Alphabet::Standard => matches!(byte, b'-' | b'_'),
        Alphabet::UrlSafe => matches!(byte, b'+' | b'/'),
    };

    if foreign && strict {
        Err(Error::MixedAlphabet { offset })
    } else {
        Err(Error::InvalidByte { offset, byte })
    }
}

fn encode_digit(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn append_wrapped_symbol(
    out: &mut String,
    symbol: u8,
    options: Options,
    column: &mut usize,
) {
    if let Whitespace::Wrapped { width, ending } = options.whitespace {
        if *column == width {
            match ending {
                LineEnding::Lf => out.push('\n'),
                LineEnding::CrLf => out.push_str("\r\n"),
            }
            *column = 0;
        }
        out.push(symbol as char);
        *column += 1;
    } else {
        out.push(symbol as char);
    }
}

fn append_encoded_group(
    out: &mut String,
    bytes: &[u8],
    options: Options,
    column: &mut usize,
    final_group: bool,
) {
    let b0 = bytes[0];
    let b1 = *bytes.get(1).unwrap_or(&0);
    let b2 = *bytes.get(2).unwrap_or(&0);

    let digits = [
        b0 >> 2,
        ((b0 & 3) << 4) | (b1 >> 4),
        ((b1 & 15) << 2) | (b2 >> 6),
        b2 & 63,
    ];

    let count = if bytes.len() == 1 { 2 } else if bytes.len() == 2 { 3 } else { 4 };
    for value in digits.iter().take(count) {
        append_wrapped_symbol(out, encode_digit(*value, options.alphabet), options, column);
    }

    if final_group && options.padding != Padding::Forbidden {
        for _ in count..4 {
            append_wrapped_symbol(out, b'=', options, column);
        }
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut out = encoder.push(input)?;
    out.push_str(&encoder.finish()?);
    Ok(out)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    column: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            pending: Vec::new(),
            column: 0,
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut data = std::mem::take(&mut self.pending);
        data.extend_from_slice(input);
        let complete = data.len() / 3 * 3;
        let mut out = String::new();
        for chunk in data[..complete].chunks_exact(3) {
            append_encoded_group(&mut out, chunk, self.options, &mut self.column, false);
        }
        self.pending.extend_from_slice(&data[complete..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        if !self.pending.is_empty() {
            append_encoded_group(
                &mut out,
                &self.pending,
                self.options,
                &mut self.column,
                true,
            );
        }
        self.pending.clear();
        self.finished = true;
        Ok(out)
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
    value: Option<u8>,
}

pub struct Decoder {
    options: Options,
    group: Vec<Symbol>,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    physical_offset: usize,
    symbol_count: usize,
    wrapped_column: usize,
    wrap_state: WrapState,
    finished: bool,
    sticky_error: Option<Error>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum WrapState {
    Row,
    NeedLf { cr_offset: usize, first_offset: usize },
    NeedSymbol { first_offset: usize },
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            pending_second_padding: None,
            physical_offset: 0,
            symbol_count: 0,
            wrapped_column: 0,
            wrap_state: WrapState::Row,
            finished: false,
            sticky_error: None,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.physical_offset;
            self.physical_offset += 1;

            if let Err(error) = self.process_byte(byte, offset, &mut out) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(out)
    }

    fn process_byte(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => {}
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Ok(());
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.wrap_state {
                    WrapState::NeedLf { cr_offset, first_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        let _ = cr_offset;
                        self.wrap_state = WrapState::NeedSymbol { first_offset };
                        return Ok(());
                    }
                    WrapState::NeedSymbol { first_offset } => {
                        self.wrap_state = WrapState::Row;
                        if byte == b'\r' || byte == b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrapped_column = 0;
                        let _ = first_offset;
                    }
                    WrapState::Row => {
                        if self.wrapped_column == width {
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.wrap_state = WrapState::NeedSymbol {
                                        first_offset: offset,
                                    };
                                    return Ok(());
                                }
                                LineEnding::Lf => return Err(Error::BadWrap { offset }),
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.wrap_state = WrapState::NeedLf {
                                        cr_offset: offset,
                                        first_offset: offset,
                                    };
                                    return Ok(());
                                }
                                LineEnding::CrLf => return Err(Error::BadWrap { offset }),
                            }
                        }
                        if byte == b'\r' || byte == b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                    }
                }
            }
        }

        if let Some(limit) = self.options.max_symbols {
            if self.symbol_count >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbol_count += 1;

        if let Whitespace::Wrapped { width, .. } = self.options.whitespace {
            self.wrapped_column += 1;
            if self.wrapped_column > width {
                return Err(Error::BadWrap { offset });
            }
        }

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.pending_second_padding.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        let pos = self.group.len();
        if pos <= 1 && byte == b'=' {
            return Err(Error::BadPadding { offset });
        }

        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if pos == 2 {
                self.pending_second_padding = Some(offset);
                self.group.push(Symbol { byte, offset, value: None });
                return Ok(());
            }
            if pos == 3 {
                self.group.push(Symbol { byte, offset, value: None });
                let first = self.group[0].value.unwrap();
                let second = self.group[1].value.unwrap();
                if self.options.strict && second & 15 != 0 {
                    return Err(Error::NonCanonical {
                        offset: self.group[1].offset,
                    });
                }
                out.push((first << 2) | (second >> 4));
                self.group.clear();
                self.pending_second_padding = None;
                self.terminal_padding = true;
                return Ok(());
            }
            return Err(Error::BadPadding { offset });
        }

        let value = digit(byte, self.options.alphabet, self.options.strict, offset)?;
        self.group.push(Symbol {
            byte,
            offset,
            value: Some(value),
        });

        if self.group.len() == 4 {
            let a = self.group[0].value.unwrap();
            let b = self.group[1].value.unwrap();
            let c = self.group[2].value.unwrap();
            let d = self.group[3].value.unwrap();
            out.extend_from_slice(&[
                (a << 2) | (b >> 4),
                (b << 4) | (c >> 2),
                (c << 6) | d,
            ]);
            self.group.clear();
        }
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let result = self.finish_inner();
        match result {
            Ok(bytes) => {
                self.finished = true;
                Ok(bytes)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        if let Whitespace::Wrapped { ending, .. } = self.options.whitespace {
            match self.wrap_state {
                WrapState::NeedLf { cr_offset, .. } => {
                    return Err(Error::BadWrap { offset: cr_offset });
                }
                WrapState::NeedSymbol { first_offset } => {
                    return Err(Error::BadWrap { offset: first_offset });
                }
                WrapState::Row => {}
            }
            let _ = ending;
        }

        if let Some(offset) = self.pending_second_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group.is_empty() {
            return Ok(Vec::new());
        }

        if self.options.padding == Padding::Required {
            return Err(Error::Truncated {
                offset: self.group[0].offset,
            });
        }

        match self.group.len() {
            1 => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
            2 => {
                let a = self.group[0].value.unwrap();
                let b = self.group[1].value.unwrap();
                if self.options.strict && b & 15 != 0 {
                    return Err(Error::NonCanonical {
                        offset: self.group[1].offset,
                    });
                }
                Ok(vec![(a << 2) | (b >> 4)])
            }
            3 => {
                let a = self.group[0].value.unwrap();
                let b = self.group[1].value.unwrap();
                let c = self.group[2].value.unwrap();
                if self.options.strict && c & 3 != 0 {
                    return Err(Error::NonCanonical {
                        offset: self.group[2].offset,
                    });
                }
                Ok(vec![(a << 2) | (b >> 4), (b << 4) | (c >> 2)])
            }
            _ => unreachable!(),
        }
    }
}
