#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn digit(byte: u8, alphabet: Alphabet, strict: bool, offset: usize) -> Result<u8, Error> {
    match byte {
        b'A'..=b'Z' => Ok(byte - b'A'),
        b'a'..=b'z' => Ok(byte - b'a' + 26),
        b'0'..=b'9' => Ok(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Ok(62),
        b'/' if alphabet == Alphabet::Standard => Ok(63),
        b'-' if alphabet == Alphabet::UrlSafe => Ok(62),
        b'_' if alphabet == Alphabet::UrlSafe => Ok(63),
        b'+' | b'/' | b'-' | b'_' if strict => Err(Error::MixedAlphabet { offset }),
        _ => Err(Error::InvalidByte { offset, byte }),
    }
}

fn symbol(value: u8, alphabet: Alphabet) -> u8 {
    const STANDARD: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const URL: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
    match alphabet {
        Alphabet::Standard => STANDARD[value as usize],
        Alphabet::UrlSafe => URL[value as usize],
    }
}

struct WrapEncoder {
    width: usize,
    ending: LineEnding,
    column: usize,
}

impl WrapEncoder {
    fn separator_if_needed(&mut self, out: &mut String) {
        if self.column == self.width {
            match self.ending {
                LineEnding::Lf => out.push('\n'),
                LineEnding::CrLf => out.push_str("\r\n"),
            }
            self.column = 0;
        }
    }

    fn push_byte(&mut self, out: &mut String, byte: u8) {
        self.separator_if_needed(out);
        out.push(byte as char);
        self.column += 1;
    }
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    wrap: Option<WrapEncoder>,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        let wrap = match options.whitespace {
            Whitespace::Wrapped { width, ending } => Some(WrapEncoder {
                width,
                ending,
                column: 0,
            }),
            _ => None,
        };
        Ok(Self {
            options,
            tail: Vec::new(),
            wrap,
            closed: false,
        })
    }

    fn emit(&mut self, bytes: &[u8], out: &mut String) {
        if let Some(wrap) = self.wrap.as_mut() {
            wrap.push_byte(out, bytes[0]);
            wrap.push_byte(out, bytes[1]);
            wrap.push_byte(out, bytes[2]);
            wrap.push_byte(out, bytes[3]);
        } else {
            for &byte in bytes {
                out.push(byte as char);
            }
        }
    }

    fn encode_group(&mut self, a: u8, b: u8, c: u8, out: &mut String) {
        let alphabet = self.options.alphabet;
        let encoded = [
            symbol(a >> 2, alphabet),
            symbol(((a & 0x03) << 4) | (b >> 4), alphabet),
            symbol(((b & 0x0f) << 2) | (c >> 6), alphabet),
            symbol(c & 0x3f, alphabet),
        ];
        self.emit(&encoded, out);
    }

    fn encode_tail(&mut self, out: &mut String) {
        let alphabet = self.options.alphabet;
        match self.tail.len() {
            0 => {}
            1 => {
                let a = self.tail[0];
                let encoded = [
                    symbol(a >> 2, alphabet),
                    symbol((a & 0x03) << 4, alphabet),
                ];
                self.emit(&[encoded[0], encoded[1], b'=', b'='], out);
                if self.options.padding == Padding::Forbidden {
                    // Forbidden padding is omitted from the output.
                    // The two padding bytes above are removed below.
                    for _ in 0..2 {
                        out.pop();
                        if let Some(wrap) = self.wrap.as_mut() {
                            wrap.column -= 1;
                        }
                    }
                }
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                let encoded = [
                    symbol(a >> 2, alphabet),
                    symbol(((a & 0x03) << 4) | (b >> 4), alphabet),
                    symbol((b & 0x0f) << 2, alphabet),
                ];
                self.emit(&[encoded[0], encoded[1], encoded[2], b'='], out);
                if self.options.padding == Padding::Forbidden {
                    out.pop();
                    if let Some(wrap) = self.wrap.as_mut() {
                        wrap.column -= 1;
                    }
                }
            }
            _ => unreachable!(),
        }
        self.tail.clear();
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        let mut index = 0;

        if self.tail.len() == 1 && !input.is_empty() {
            self.tail.push(input[0]);
            index += 1;
            if self.tail.len() == 2 && index < input.len() {
                let a = self.tail[0];
                let b = self.tail[1];
                let c = input[index];
                index += 1;
                self.encode_group(a, b, c, &mut out);
                self.tail.clear();
            }
        }

        while index + 3 <= input.len() {
            self.encode_group(input[index], input[index + 1], input[index + 2], &mut out);
            index += 3;
        }

        self.tail.extend_from_slice(&input[index..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let mut out = String::new();
        self.encode_tail(&mut out);
        Ok(out)
    }
}

#[derive(Clone, Copy)]
enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { first_offset: usize },
}

pub struct Decoder {
    options: Options,
    buffer: Vec<(u8, usize)>,
    offset: usize,
    symbols: usize,
    completed_padding: bool,
    pending_padding: Option<usize>,
    frame_state: FrameState,
    row_symbols: usize,
    closed: bool,
    sticky: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffer: Vec::with_capacity(4),
            offset: 0,
            symbols: 0,
            completed_padding: false,
            pending_padding: None,
            frame_state: FrameState::Normal,
            row_symbols: 0,
            closed: false,
            sticky: None,
        })
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn decode_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        let group = &self.buffer;
        let vals: Vec<Option<u8>> = group
            .iter()
            .map(|&(byte, offset)| {
                if byte == b'=' {
                    Ok(None)
                } else {
                    digit(byte, self.options.alphabet, self.options.strict, offset).map(Some)
                }
            })
            .collect::<Result<_, _>>()?;

        let v0 = vals[0].unwrap();
        let v1 = vals[1].unwrap();
        if vals[2].is_none() {
            let offset = group[2].1;
            if vals[3].is_none() {
                if self.options.strict && v1 & 0x0f != 0 {
                    return self.fail(Error::NonCanonical {
                        offset: group[1].1,
                    });
                }
                out.push((v0 << 2) | (v1 >> 4));
                self.completed_padding = true;
                return Ok(());
            }
            return self.fail(Error::BadPadding { offset });
        }

        let v2 = vals[2].unwrap();
        if vals[3].is_none() {
            if self.options.strict && v2 & 0x03 != 0 {
                return self.fail(Error::NonCanonical {
                    offset: group[2].1,
                });
            }
            out.push((v0 << 2) | (v1 >> 4));
            out.push((v1 << 4) | (v2 >> 2));
            self.completed_padding = true;
            return Ok(());
        }

        let v3 = vals[3].unwrap();
        out.push((v0 << 2) | (v1 >> 4));
        out.push((v1 << 4) | (v2 >> 2));
        out.push((v2 << 6) | v3);
        Ok(())
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols >= limit {
                return self.fail(Error::Overlong { offset, limit });
            }
        }
        self.symbols += 1;

        if self.completed_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if let Some(pending) = self.pending_padding {
            if byte != b'=' {
                return self.fail(Error::BadPadding { offset });
            }
            self.pending_padding = None;
            self.buffer.push((byte, offset));
            self.decode_group(out)?;
            self.buffer.clear();
            return Ok(());
        }

        match self.buffer.len() {
            0 | 1 if byte == b'=' => {
                return self.fail(Error::BadPadding { offset });
            }
            2 if byte == b'=' => {
                if self.options.padding == Padding::Forbidden {
                    return self.fail(Error::BadPadding { offset });
                }
                self.pending_padding = Some(offset);
                self.buffer.push((byte, offset));
                return Ok(());
            }
            3 if byte == b'=' => {
                if self.options.padding == Padding::Forbidden {
                    return self.fail(Error::BadPadding { offset });
                }
                self.buffer.push((byte, offset));
                self.decode_group(out)?;
                self.buffer.clear();
                return Ok(());
            }
            _ => {}
        }

        let value = match digit(byte, self.options.alphabet, self.options.strict, offset) {
            Ok(v) => v,
            Err(error) => return self.fail(error),
        };
        let _ = value;
        self.buffer.push((byte, offset));
        if self.buffer.len() == 4 {
            self.decode_group(out)?;
            self.buffer.clear();
        }
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::IgnoreAscii if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) => return Ok(()),
            Whitespace::Wrapped { width, ending } => {
                match self.frame_state {
                    FrameState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return self.fail(Error::BadWrap { offset });
                        }
                        self.frame_state = FrameState::NeedSymbol {
                            first_offset: cr_offset,
                        };
                        return Ok(());
                    }
                    FrameState::NeedSymbol { first_offset } => {
                        self.frame_state = FrameState::Normal;
                        self.row_symbols = 0;
                        if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                            return self.fail(Error::BadWrap { offset });
                        }
                        let _ = first_offset;
                    }
                    FrameState::Normal => {}
                }

                if self.row_symbols == width {
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.frame_state = FrameState::NeedSymbol {
                                first_offset: offset,
                            };
                            return Ok(());
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.frame_state = FrameState::NeedLf { cr_offset: offset };
                            return Ok(());
                        }
                        _ => return self.fail(Error::BadWrap { offset }),
                    }
                }

                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return self.fail(Error::BadWrap { offset });
                }
                self.row_symbols += 1;
                self.process_symbol(byte, offset, out)
            }
            _ => self.process_symbol(byte, offset, out),
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = self.sticky.clone() {
            return Err(error);
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }
        let mut out = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut out) {
                return Err(error);
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = self.sticky.clone() {
            return Err(error);
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedLf { cr_offset } => {
                    return self.fail(Error::BadWrap { offset: cr_offset });
                }
                FrameState::NeedSymbol { first_offset } => {
                    return self.fail(Error::BadWrap {
                        offset: first_offset,
                    });
                }
                FrameState::Normal => {}
            }
        }

        if let Some(offset) = self.pending_padding {
            return self.fail(Error::BadPadding { offset });
        }

        match self.buffer.len() {
            0 => {}
            1 => {
                return self.fail(Error::Truncated {
                    offset: self.buffer[0].1,
                });
            }
            2 if self.options.padding != Padding::Required => {
                let a = digit(
                    self.buffer[0].0,
                    self.options.alphabet,
                    self.options.strict,
                    self.buffer[0].1,
                )
                .map_err(|e| e)?;
                let b = digit(
                    self.buffer[1].0,
                    self.options.alphabet,
                    self.options.strict,
                    self.buffer[1].1,
                )
                .map_err(|e| e)?;
                if self.options.strict && b & 0x0f != 0 {
                    return self.fail(Error::NonCanonical {
                        offset: self.buffer[1].1,
                    });
                }
                out.push((a << 2) | (b >> 4));
            }
            2 => {
                return self.fail(Error::Truncated {
                    offset: self.buffer[0].1,
                });
            }
            3 if self.options.padding != Padding::Required => {
                let a = digit(
                    self.buffer[0].0,
                    self.options.alphabet,
                    self.options.strict,
                    self.buffer[0].1,
                )
                .map_err(|e| e)?;
                let b = digit(
                    self.buffer[1].0,
                    self.options.alphabet,
                    self.options.strict,
                    self.buffer[1].1,
                )
                .map_err(|e| e)?;
                let c = digit(
                    self.buffer[2].0,
                    self.options.alphabet,
                    self.options.strict,
                    self.buffer[2].1,
                )
                .map_err(|e| e)?;
                if self.options.strict && c & 0x03 != 0 {
                    return self.fail(Error::NonCanonical {
                        offset: self.buffer[2].1,
                    });
                }
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
            }
            3 => {
                return self.fail(Error::Truncated {
                    offset: self.buffer[0].1,
                });
            }
            _ => unreachable!(),
        }

        self.closed = true;
        Ok(out)
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}
