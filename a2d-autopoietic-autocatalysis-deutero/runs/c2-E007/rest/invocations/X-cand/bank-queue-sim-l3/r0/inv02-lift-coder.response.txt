#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    remaining: u64,
    pause_time: Option<u64>,
}

struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(at), Some(bt)) => at.cmp(&bt).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn take_best(waiting: &mut Vec<Job>) -> Option<Job> {
    let idx = waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiter_cmp(a, b))
        .map(|(idx, _)| idx)?;
    Some(waiting.swap_remove(idx))
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut ids = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;
    let mut now = arrivals[0].arrival;

    while finished_count < customers.len() {
        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let active = running.take().unwrap();
                events.push(Event {
                    time: now,
                    customer: active.job.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                let best_priority = waiting
                    .iter()
                    .map(|job| job.customer.priority)
                    .min();
                if let Some(priority) = best_priority {
                    if priority < active.job.customer.priority {
                        let by = waiting
                            .iter()
                            .filter(|job| job.customer.priority == priority)
                            .min_by(|a, b| waiter_cmp(a, b))
                            .unwrap()
                            .customer
                            .id;
                        let mut active = running.take().unwrap();
                        let remaining = active.finish - now;
                        events.push(Event {
                            time: now,
                            customer: active.job.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        active.job.remaining = remaining;
                        active.job.pause_time = Some(now);
                        waiting.push(active.job);
                    }
                }
            }
        }

        while running.is_none() {
            let Some(mut job) = take_best(&mut waiting) else {
                break;
            };

            let cost = if last_dispatched == Some(job.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            job.started = true;
            job.pause_time = None;
            last_dispatched = Some(job.customer.id);
            events.push(Event {
                time: now,
                customer: job.customer.id,
                kind,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: job.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            } else {
                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            if active.finish > now {
                next_time = Some(next_time.map_or(active.finish, |t| t.min(active.finish)));
            }
            if active.work_at > now {
                next_time = Some(next_time.map_or(active.work_at, |t| t.min(active.work_at)));
            }
        }

        if let Some(next) = next_time {
            now = next;
        } else {
            break;
        }
    }

    Ok(events)
}
