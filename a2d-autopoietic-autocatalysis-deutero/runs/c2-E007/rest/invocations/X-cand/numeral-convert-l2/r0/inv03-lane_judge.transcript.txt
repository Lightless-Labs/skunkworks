        b'0'..=b'9' => Some(byte - b'0'),

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {

        hash + 1

        0

        return Err(BaseError::Empty { operand });

            operand,
            index: payload_start,

fn multiply_add(limbs: &mut Vec<u32>, multiplier: u32, addend: u32) {
    let mut carry = addend as u64;

        let value = *limb as u64 * multiplier as u64 + carry;

        carry = value / LIMB_BASE;

}

fn render_value(mut limbs: Vec<u32>, radix: u8, style: BaseStyle) -> String {

        value.pop();

        return vec![0];

    let mut output = Vec::new();

        let mut remainder = 0u32;

    output.reverse();
    output

fn render(mut value: Vec<u8>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if value.is_empty() {
        digits.push(b'0');
    } else {
        while !value.is_empty() {
            let digit = divide_big(&mut value, radix);
            let byte = if digit < 10 {
                b'0' + digit
            } else if style.upper {
                b'A' + digit - 10
            } else {
                b'a' + digit - 10
            };
            digits.push(byte);
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let mut padded = vec![b'0'; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    if style.group == 0 {
        return String::from_utf8(digits).unwrap();
    }

    let separators = (digits.len() - 1) / style.group;
    let mut output = Vec::with_capacity(digits.len() + separators);
    for (i, &digit) in digits.iter().enumerate() {
        if i > 0 && (digits.len() - i) % style.group == 0 {
            output.push(b'_');
        }
        output.push(digit);
    }
    String::from_utf8(output).unwrap()
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let mut value = Vec::new();
    for digit in digits {
        multiply_add(&mut value, from, digit);
    }
    Ok(render(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_value = Vec::new();
    for digit in left_digits {
        multiply_add(&mut left_value, base, digit);
    }
    let mut right_value = Vec::new();
    for digit in right_digits {
        multiply_add(&mut right_value, base, digit);
    }

    Ok(render(add_big(&left_value, &right_value), base, style))
}
