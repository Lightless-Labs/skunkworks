#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let (payload_start, payload) = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        (hash + 1, &bytes[hash + 1..])
    } else {
        (0, bytes)
    };

    if payload.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut digits = Vec::new();
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in value.iter_mut() {
        let n = (*limb as u16) * (radix as u16) + carry;
        *limb = (n & 0xff) as u8;
        carry = n >> 8;
    }
    while carry != 0 {
        value.push((carry & 0xff) as u8);
        carry >>= 8;
    }
}

fn add_big(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for i in 0..left.len().max(right.len()) {
        let sum = left.get(i).copied().unwrap_or(0) as u16
            + right.get(i).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum & 0xff) as u8);
        carry = sum >> 8;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn divide_big(value: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for limb in value.iter_mut().rev() {
        let current = (remainder << 8) | *limb as u16;
        *limb = (current / divisor as u16) as u8;
        remainder = current % divisor as u16;
    }
    while value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn render(mut value: Vec<u8>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if value.is_empty() {
        digits.push(b'0');
    } else {
        while !value.is_empty() {
            let digit = divide_big(&mut value, radix);
            let byte = if digit < 10 {
                b'0' + digit
            } else if style.upper {
                b'A' + digit - 10
            } else {
                b'a' + digit - 10
            };
            digits.push(byte);
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let mut padded = vec![b'0'; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    if style.group == 0 {
        return String::from_utf8(digits).unwrap();
    }

    let separators = (digits.len() - 1) / style.group;
    let mut output = Vec::with_capacity(digits.len() + separators);
    for (i, &digit) in digits.iter().enumerate() {
        if i > 0 && (digits.len() - i) % style.group == 0 {
            output.push(b'_');
        }
        output.push(digit);
    }
    String::from_utf8(output).unwrap()
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let mut value = Vec::new();
    for digit in digits {
        multiply_add(&mut value, from, digit);
    }
    Ok(render(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_value = Vec::new();
    for digit in left_digits {
        multiply_add(&mut left_value, base, digit);
    }
    let mut right_value = Vec::new();
    for digit in right_digits {
        multiply_add(&mut right_value, base, digit);
    }

    Ok(render(add_big(&left_value, &right_value), base, style))
}
