#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if bytes[..hash] != *expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        if !matches!(digit_value(byte), Some(value) if value < radix) {
            return Err(BaseError::InvalidDigit {
                operand,
                index: payload_start + offset,
            });
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && digit_value(payload[offset - 1]).is_some();
            let has_digit_after =
                offset + 1 < payload.len() && digit_value(payload[offset + 1]).is_some();
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let digit_count = payload.iter().filter(|&&b| b != b'_').count();
    if digit_count >= 2 {
        let first_digit = payload.iter().find(|&&b| b != b'_').copied().unwrap();
        if digit_value(first_digit) == Some(0) {
            return Err(BaseError::LeadingZero {
                operand,
                index: payload_start,
            });
        }
    }

    Ok(payload
        .iter()
        .filter_map(|&byte| if byte == b'_' { None } else { digit_value(byte) })
        .collect())
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u32;
    for digit in value.iter_mut() {
        let next = (*digit as u32) * (multiplier as u32) + carry;
        *digit = (next & 0xff) as u8;
        carry = next >> 8;
    }
    while carry != 0 {
        value.push((carry & 0xff) as u8);
        carry >>= 8;
    }
}

fn to_radix(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        multiply_add(&mut value, from, digit);
    }

    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }

    if value.len() == 1 && value[0] == 0 {
        return vec![0];
    }

    let mut output = Vec::new();
    while value.iter().any(|&byte| byte != 0) {
        let mut remainder = 0u32;
        for byte in value.iter_mut().rev() {
            let current = (remainder << 8) | (*byte as u32);
            *byte = (current / to as u32) as u8;
            remainder = current % to as u32;
        }
        output.push(remainder as u8);
        while value.len() > 1 && value.last() == Some(&0) {
            value.pop();
        }
    }
    output.reverse();
    output
}

fn add_numbers(left: &[u8], right: &[u8], radix: u8) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    let len = left.len().max(right.len());

    for index in 0..len {
        let a = left.get(index).copied().unwrap_or(0) as u16;
        let b = right.get(index).copied().unwrap_or(0) as u16;
        let sum = a + b + carry;
        result.push((sum % radix as u16) as u8);
        carry = sum / radix as u16;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn format_result(digits: &[u8], style: BaseStyle) -> String {
    let mut padded = vec![0u8; style.min_width.saturating_sub(digits.len())];
    padded.extend_from_slice(digits);

    let mut output = String::new();
    for (index, &digit) in padded.iter().enumerate() {
        if style.group != 0 && index > 0 && (padded.len() - index) % style.group == 0 {
            output.push('_');
        }

        let character = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(character as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let converted = to_radix(&digits, from, to);
    Ok(format_result(&converted, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let sum = add_numbers(&left_digits, &right_digits, base);
    Ok(format_result(&sum, style))
}
