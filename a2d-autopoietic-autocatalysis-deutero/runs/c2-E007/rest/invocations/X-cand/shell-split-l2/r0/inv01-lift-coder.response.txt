#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        let next_i = i + ch.len_utf8();

        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                match ch {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if matches!(escaped, '"' | '\\' | '$') {
                            word.push(escaped);
                            i = after;
                        } else if escaped == '\n' {
                            i = after;
                        } else {
                            word.push('\\');
                            i = next_i;
                        }
                    }
                    '$' => {
                        let (value, after) = expand(input, i, env)?;
                        word.push_str(&value);
                        active = true;
                        i = after;
                    }
                    _ => {
                        word.push(ch);
                        i = next_i;
                    }
                }
            }
            Quote::None => {
                match ch {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(std::mem::take(&mut word));
                            active = false;
                        }
                        i = next_i;
                    }
                    '\'' => {
                        active = true;
                        quote = Quote::Single(i);
                        i = next_i;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double(i);
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if escaped != '\n' {
                            word.push(escaped);
                            active = true;
                        }
                        i = after;
                    }
                    '$' => {
                        let (value, after) = expand(input, i, env)?;
                        word.push_str(&value);
                        active = true;
                        i = after;
                    }
                    _ => {
                        word.push(ch);
                        active = true;
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok(("$".to_string(), after_dollar));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut i = content_start;
        let mut close = None;

        while i < input.len() {
            let ch = input[i..].chars().next().unwrap();
            let next = i + ch.len_utf8();
            if ch == '\\' {
                if next < input.len() {
                    let escaped = input[next..].chars().next().unwrap();
                    i = next + escaped.len_utf8();
                } else {
                    i = next;
                }
            } else if ch == '}' {
                close = Some(i);
                break;
            } else {
                i = next;
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .find_map(|(idx, ch)| {
                let valid = if idx == 0 {
                    is_name_start(ch)
                } else {
                    is_name_continue(ch)
                };
                if valid { None } else { Some(idx) }
            })
            .unwrap_or(contents.len());

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let remainder = &contents[name_end..];
        let selected = if remainder.is_empty() {
            lookup(name, env).unwrap_or("")
        } else if let Some(default) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
            .leak()
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((selected.to_string(), close + 1));
    }

    if !is_name_start(first) {
        return Ok(("$".to_string(), after_dollar));
    }

    let mut end = after_dollar + first.len_utf8();
    while end < input.len() {
        let ch = input[end..].chars().next().unwrap();
        if is_name_continue(ch) {
            end += ch.len_utf8();
        } else {
            break;
        }
    }
    let name = &input[after_dollar..end];
    Ok((lookup(name, env).unwrap_or("").to_string(), end))
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let ch = text[i..].chars().next().unwrap();
        let next = i + ch.len_utf8();
        if ch == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            result.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            result.push(ch);
            i = next;
        }
    }
    result
}
