#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn next_char(input: &str, pos: usize) -> Option<(char, usize)> {
    input[pos..]
        .chars()
        .next()
        .map(|c| (c, pos + c.len_utf8()))
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut pos = 0;

    while pos < text.len() {
        let (c, next) = next_char(text, pos).unwrap();
        if c == '\\' {
            if let Some((escaped, after)) = next_char(text, next) {
                result.push(escaped);
                pos = after;
            } else {
                result.push('\\');
                pos = next;
            }
        } else {
            result.push(c);
            pos = next;
        }
    }

    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut pos = 0;

    while pos < input.len() {
        let (c, next) = next_char(input, pos).unwrap();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                pos = next;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next;
                } else if c == '\\' {
                    let slash_pos = pos;
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let (escaped, after) = next_char(input, next).unwrap();
                    if matches!(escaped, '"' | '\\' | '$') {
                        word.push(escaped);
                        pos = after;
                    } else if escaped == '\n' {
                        pos = after;
                    } else {
                        word.push('\\');
                        pos = next;
                    }
                } else if c == '$' {
                    if let Some((expanded, after)) = parse_expansion(input, pos, env)? {
                        active = true;
                        word.push_str(&expanded);
                        pos = after;
                    } else {
                        word.push('$');
                        pos = next;
                    }
                } else {
                    word.push(c);
                    pos = next;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    pos = next;
                } else if c == '\'' {
                    quote = Quote::Single(pos);
                    active = true;
                    pos = next;
                } else if c == '"' {
                    quote = Quote::Double(pos);
                    active = true;
                    pos = next;
                } else if c == '\\' {
                    let slash_pos = pos;
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let (escaped, after) = next_char(input, next).unwrap();
                    if escaped == '\n' {
                        pos = after;
                    } else {
                        active = true;
                        word.push(escaped);
                        pos = after;
                    }
                } else if c == '$' {
                    if let Some((expanded, after)) = parse_expansion(input, pos, env)? {
                        active = true;
                        word.push_str(&expanded);
                        pos = after;
                    } else {
                        active = true;
                        word.push('$');
                        pos = next;
                    }
                } else {
                    active = true;
                    word.push(c);
                    pos = next;
                }
            }
        }
    }

    match quote {
        Quote::Single(open_pos) => {
            return Err(SplitError::UnterminatedSingle { pos: open_pos });
        }
        Quote::Double(open_pos) => {
            return Err(SplitError::UnterminatedDouble { pos: open_pos });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn parse_expansion(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<Option<(String, usize)>, SplitError> {
    let after_dollar = dollar_pos + 1;
    if after_dollar >= input.len() {
        return Ok(None);
    }

    let (first, first_end) = next_char(input, after_dollar).unwrap();
    if first == '{' {
        let content_start = first_end;
        let mut scan = content_start;

        while scan < input.len() {
            let (c, next) = next_char(input, scan).unwrap();
            if c == '\\' {
                if next < input.len() {
                    let (_, after_escaped) = next_char(input, next).unwrap();
                    scan = after_escaped;
                } else {
                    scan = next;
                }
            } else if c == '}' {
                let content = &input[content_start..scan];
                let name_end_rel = content
                    .char_indices()
                    .take_while(|&(_, c)| is_name_continue(c))
                    .map(|(i, c)| i + c.len_utf8())
                    .last()
                    .unwrap_or(0);

                if name_end_rel == 0 {
                    return Err(SplitError::InvalidExpansion { pos: dollar_pos });
                }

                let name = &content[..name_end_rel];
                let rest = &content[name_end_rel..];

                let value = if rest.is_empty() {
                    lookup(env, name).unwrap_or("").to_string()
                } else if let Some(default) = rest.strip_prefix(":-") {
                    match lookup(env, name) {
                        Some(value) if !value.is_empty() => value.to_string(),
                        _ => decode_default(default),
                    }
                } else {
                    return Err(SplitError::InvalidExpansion { pos: dollar_pos });
                };

                return Ok(Some((value, next)));
            } else {
                scan = next;
            }
        }

        return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
    }

    if !is_name_start(first) {
        return Ok(None);
    }

    let mut end = first_end;
    while end < input.len() {
        let (c, next) = next_char(input, end).unwrap();
        if is_name_continue(c) {
            end = next;
        } else {
            break;
        }
    }

    Ok(Some((
        lookup(env, &input[after_dollar..end])
            .unwrap_or("")
            .to_string(),
        end,
    )))
}
