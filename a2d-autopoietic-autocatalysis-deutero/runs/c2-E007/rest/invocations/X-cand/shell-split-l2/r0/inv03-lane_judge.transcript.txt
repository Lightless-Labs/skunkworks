fn is_separator(c: char) -> bool {

fn next_char(input: &str, pos: usize) -> Option<(char, usize)> {
    input[pos..]

        .next()

        }

fn parse_expansion(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<Option<(String, usize)>, SplitError> {
    let after_dollar = dollar_pos + 1;
    if after_dollar >= input.len() {
        return Ok(None);
    }

    let (first, first_end) = next_char(input, after_dollar).unwrap();
    if first == '{' {
        let content_start = first_end;
        let mut scan = content_start;

        while scan < input.len() {
            let (c, next) = next_char(input, scan).unwrap();
            if c == '\\' {
                if next < input.len() {
                    let (_, after_escaped) = next_char(input, next).unwrap();
                    scan = after_escaped;
                } else {
                    scan = next;
                }
            } else if c == '}' {
                let content = &input[content_start..scan];
                let name_end_rel = content
                    .char_indices()
                    .take_while(|&(_, c)| is_name_continue(c))
                    .map(|(i, c)| i + c.len_utf8())
                    .last()
                    .unwrap_or(0);

                if name_end_rel == 0 {
                    return Err(SplitError::InvalidExpansion { pos: dollar_pos });
                }

                let name = &content[..name_end_rel];
                let rest = &content[name_end_rel..];

                let value = if rest.is_empty() {
                    lookup(env, name).unwrap_or("").to_string()
                } else if let Some(default) = rest.strip_prefix(":-") {
                    match lookup(env, name) {
                        Some(value) if !value.is_empty() => value.to_string(),
                        _ => decode_default(default),
                    }
                } else {
                    return Err(SplitError::InvalidExpansion { pos: dollar_pos });
                };

                return Ok(Some((value, next)));
            } else {
                scan = next;
            }
        }

        return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
    }

    if !is_name_start(first) {
        return Ok(None);
    }

    let mut end = first_end;
    while end < input.len() {
        let (c, next) = next_char(input, end).unwrap();
        if is_name_continue(c) {
            end = next;
        } else {
            break;
        }
    }

    Ok(Some((
        lookup(env, &input[after_dollar..end])
            .unwrap_or("")
            .to_string(),
        end,
    )))
}

    while let Some(c) = chars.next() {
        if c == '\\' {

            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

fn parse_name(input: &str, start: usize) -> (&str, usize) {
    let mut end = start;
    for c in input[start..].chars() {
        if is_name_continue(c) {
            end += c.len_utf8();
        } else {
            break;
        }
    }
    (&input[start..end], end)
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;
    let mut close = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let Some(close_pos) = close else {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    };

    let contents = &input[content_start..close_pos];
    let mut name_end = 0;
    for c in contents.chars() {
        if is_name_continue(c) {
            name_end += c.len_utf8();
        } else {
            break;
        }
    }
    let name = &contents[..name_end];
    if name.is_empty()
        || !is_name_start(name.chars().next().unwrap())
        || !contents[name_end..].is_empty() && !contents[name_end..].starts_with(":-")
    {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let suffix = &contents[name_end..];
    let value = if suffix.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else {
        let current = lookup(env, name).unwrap_or("");
        if !current.is_empty() {
            current.to_string()
        } else {
            decode_default(&suffix[2..])
        }
    };

    Ok((value, close_pos + 1))
}

                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if escaped != '\n' {
                            word.push(escaped);
                            active = true;
                        }
                        i = after;
                    }
                    '$' => {
                        let (value, after) = expand(input, i, env)?;
                        word.push_str(&value);
                        active = true;
                        i = after;
                    }
                    _ => {
                        word.push(ch);
                        active = true;
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok(("$".to_string(), after_dollar));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut i = content_start;
        let mut close = None;

        while i < input.len() {
            let ch = input[i..].chars().next().unwrap();
            let next = i + ch.len_utf8();
            if ch == '\\' {
                if next < input.len() {
                    let escaped = input[next..].chars().next().unwrap();
                    i = next + escaped.len_utf8();
                } else {
                    i = next;
                }
            } else if ch == '}' {
                close = Some(i);
                break;
            } else {
                i = next;
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .find_map(|(idx, ch)| {
                let valid = if idx == 0 {
                    is_name_start(ch)
                } else {
                    is_name_continue(ch)
                };
                if valid { None } else { Some(idx) }
            })
            .unwrap_or(contents.len());

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let remainder = &contents[name_end..];
        let selected = if remainder.is_empty() {
            lookup(name, env).unwrap_or("")
        } else if let Some(default) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
            .leak()
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((selected.to_string(), close + 1));
    }

    if !is_name_start(first) {
        return Ok(("$".to_string(), after_dollar));
    }

    let mut end = after_dollar + first.len_utf8();
    while end < input.len() {
        let ch = input[end..].chars().next().unwrap();
        if is_name_continue(ch) {
            end += ch.len_utf8();
        } else {
            break;
        }
    }
    let name = &input[after_dollar..end];
    Ok((lookup(name, env).unwrap_or("").to_string(), end))
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let ch = text[i..].chars().next().unwrap();
        let next = i + ch.len_utf8();
        if ch == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            result.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            result.push(ch);
            i = next;
        }
    }
    result
}
