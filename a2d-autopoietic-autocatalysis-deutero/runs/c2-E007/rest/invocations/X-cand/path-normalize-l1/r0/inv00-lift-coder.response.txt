#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    let bytes = path.as_bytes();
    let absolute = bytes[0] == b'/';
    let directory_intent = bytes[bytes.len() - 1] == b'/';

    let mut components: Vec<(usize, &str)> = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        let component = &path[start..i];

        if component == "." || component == ".." {
            // The final nonempty component controls directory intent.
            // Track it separately below.
            if component == ".." {
                match components.last() {
                    Some((_, previous)) if *previous != ".." => {
                        components.pop();
                    }
                    _ if absolute => return Err(PathError::Escape { at: start }),
                    _ => components.push((start, component)),
                }
            }
        } else {
            components.push((start, component));
        }
    }

    let last_nonempty = path
        .split('/')
        .filter(|component| !component.is_empty())
        .last();
    let directory_intent =
        directory_intent || matches!(last_nonempty, Some(".") | Some(".."));

    let joined = components
        .iter()
        .map(|(_, component)| *component)
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
