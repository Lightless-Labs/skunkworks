#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();

    if let Some(at) = bytes.iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = bytes.iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = bytes
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = bytes[0] == b'/';
    let mut directory_intent = bytes[bytes.len() - 1] == b'/';
    let mut components: Vec<&str> = Vec::new();
    let mut start = 0;

    while start < path.len() {
        while start < path.len() && bytes[start] == b'/' {
            start += 1;
        }
        if start == path.len() {
            break;
        }

        let end = path[start..]
            .find('/')
            .map_or(path.len(), |offset| start + offset);
        let component = &path[start..end];

        if end == path.len() {
            directory_intent |= component == "." || component == "..";
        }

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|&name| name != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push("..");
                }
            }
            _ => components.push(component),
        }

        start = end;
    }

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    for (index, component) in components.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
