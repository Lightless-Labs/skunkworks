#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    // Compare remainder with denominator - remainder rather than doubling it.
    let complement = denominator - remainder;
    if remainder > complement || (remainder == complement && quotient & 1 == 1) {
        Some(quotient + 1)
    } else {
        Some(quotient)
    }
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    queue_since: u64,
    waited: u64,
    consumed_wait: u64,
    pause_time: Option<u64>,
    ever_started: bool,
    terminal: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiter_better(a: usize, b: usize, states: &[CustomerState]) -> bool {
    let x = &states[a];
    let y = &states[b];
    if x.customer.priority != y.customer.priority {
        return x.customer.priority < y.customer.priority;
    }
    match (x.pause_time, y.pause_time) {
        (Some(px), Some(py)) => {
            px < py || (px == py && x.customer.id < y.customer.id)
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            x.customer.arrival < y.customer.arrival
                || (x.customer.arrival == y.customer.arrival
                    && x.customer.id > y.customer.id)
        }
    }
}

fn best_waiter(waiting: &[usize], states: &[CustomerState]) -> Option<usize> {
    let mut best = None;
    for &i in waiting {
        if states[i].terminal {
            continue;
        }
        if best.map_or(true, |b| waiter_better(i, b, states)) {
            best = Some(i);
        }
    }
    best
}

fn consume_wait(state: &mut CustomerState, time: u64) -> u64 {
    let segment = time - state.queue_since;
    state.waited = state.waited.saturating_add(segment);
    state.consumed_wait = state.consumed_wait.saturating_add(segment);
    segment
}

fn expiration_order(waiting: &[usize], states: &[CustomerState], time: u64) -> Vec<usize> {
    let mut expired: Vec<usize> = waiting
        .iter()
        .copied()
        .filter(|&i| {
            !states[i].terminal && states[i].deadline.map_or(false, |d| d <= time)
        })
        .collect();
    expired.sort_by(|&a, &b| {
        states[a]
            .deadline
            .cmp(&states[b].deadline)
            .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
    });
    expired
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut seen: Vec<u32> = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        seen.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        let deadline = match c.patience {
            Some(p) => Some(
                c.arrival
                    .checked_add(p)
                    .ok_or(Error::DeadlineOverflow { index, id: c.id })?,
            ),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let n = customers.len();
    let mut states: Vec<CustomerState> = customers
        .iter()
        .enumerate()
        .map(|(i, c)| CustomerState {
            customer: c.clone(),
            deadline: initial_deadlines[i],
            remaining: c.service,
            queue_since: c.arrival,
            waited: 0,
            consumed_wait: 0,
            pause_time: None,
            ever_started: false,
            terminal: false,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..n).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    while stats.completed + stats.reneged + stats.balked < n {
        let mut next = None;
        let mut consider = |t: u64| {
            if t >= now && next.map_or(true, |old| t < old) {
                next = Some(t);
            }
        };

        if arrival_pos < n {
            consider(customers[arrivals[arrival_pos]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if !states[i].terminal {
                if let Some(d) = states[i].deadline {
                    consider(d);
                }
            }
        }

        let t = next.expect("unfinished customers must have a pending event");
        now = t;

        // Phase A: finish at this time.
        if active.as_ref().map_or(false, |a| a.finish == now) {
            let a = active.take().unwrap();
            let id = states[a.index].customer.id;
            stats.busy_ticks += (now - a.dispatch_time) as u128;
            stats.completed += 1;
            stats.turnaround_ticks +=
                (now - states[a.index].customer.arrival) as u128;
            stats.completion_order.push(id);
            states[a.index].terminal = true;
            emit(&mut events, now, id, Kind::Finish);
        }

        // Phase B: expire those already waiting before this cohort.
        let expired = expiration_order(&waiting, &states, now);
        for i in expired {
            waiting.retain(|&j| j != i);
            let segment = consume_wait(&mut states[i], now);
            stats.wait_ticks += segment as u128;
            stats.reneged += 1;
            states[i].terminal = true;
            emit(
                &mut events,
                now,
                states[i].customer.id,
                Kind::Renege {
                    waited: states[i].waited,
                },
            );
        }

        // Phase C: arrivals; the admission gate is held for the full cohort.
        while arrival_pos < n && customers[arrivals[arrival_pos]].arrival == now {
            let i = arrivals[arrival_pos];
            arrival_pos += 1;
            let c = &states[i].customer;
            emit(&mut events, now, c.id, Kind::Arrive);

            let server_idle = active.is_none();
            let limit = config.waiting_capacity + usize::from(server_idle);
            if waiting.len() >= limit {
                stats.balked += 1;
                states[i].terminal = true;
                emit(&mut events, now, c.id, Kind::Balk);
            } else {
                stats.accepted += 1;
                states[i].queue_since = now;
                waiting.push(i);
            }
        }

        // Phase D: once switching protection ends, a strictly superior waiter
        // can preempt the active customer.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, &states) {
                    if states[w].customer.priority < states[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let remaining = a.finish - now;
                        let paused_id = states[a.index].customer.id;
                        let by = states[w].customer.id;
                        stats.busy_ticks += (now - a.dispatch_time) as u128;
                        stats.preemptions += 1;
                        emit(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt { remaining, by },
                        );

                        states[a.index].remaining = remaining;
                        states[a.index].queue_since = now;
                        states[a.index].pause_time = Some(now);
                        if let Some(budget) = states[a.index].customer.patience {
                            let remaining_budget =
                                budget.saturating_sub(states[a.index].consumed_wait);
                            states[a.index].deadline = Some(now.checked_add(remaining_budget).ok_or(
                                Error::TimeOverflow {
                                    customer: paused_id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?);
                        }
                        waiting.push(a.index);
                    }
                }
            }
        }

        // Phase E: dispatch repeatedly while zero-time completions leave the
        // server idle.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, &states) else {
                break;
            };
            waiting.retain(|&j| j != i);
            let segment = consume_wait(&mut states[i], now);
            stats.wait_ticks += segment as u128;

            let id = states[i].customer.id;
            let switch = if last_dispatch.map_or(false, |last| last != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[i].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let remaining = states[i].remaining;
            if states[i].ever_started {
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                states[i].ever_started = true;
                stats.started += 1;
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    },
                );
            }

            last_dispatch = Some(id);
            active = Some(Active {
                index: i,
                work_at,
                finish,
                dispatch_time: now,
            });

            if finish != now {
                break;
            }

            // Inline zero-duration finish. Phase A is not rerun.
            active = None;
            stats.completed += 1;
            stats.turnaround_ticks +=
                (now - states[i].customer.arrival) as u128;
            stats.completion_order.push(id);
            states[i].terminal = true;
            emit(&mut events, now, id, Kind::Finish);
        }

        // Phase F: expire newly queued zero-budget or paused zero-budget
        // customers after dispatching.
        let expired = expiration_order(&waiting, &states, now);
        for i in expired {
            waiting.retain(|&j| j != i);
            let segment = consume_wait(&mut states[i], now);
            stats.wait_ticks += segment as u128;
            stats.reneged += 1;
            states[i].terminal = true;
            emit(
                &mut events,
                now,
                states[i].customer.id,
                Kind::Renege {
                    waited: states[i].waited,
                },
            );
        }

        if let Some(last) = events.last() {
            stats.elapsed = last.time;
        }
    }

    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(
            stats.turnaround_ticks * 1000,
            stats.completed as u128,
        )
        .unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10_000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
