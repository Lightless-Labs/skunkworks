#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))
}

#[derive(Clone)]
struct CustomerState {
    input: Customer,
    deadline: Option<u64>,
    remaining: u64,
    accepted: bool,
    started: bool,
    terminal: bool,
    total_waited: u64,
    queue_since: u64,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Running {
    idx: usize,
    dispatch: u64,
    work_at: u64,
    finish: u64,
    remaining: u64,
    protected: bool,
}

fn better(a: usize, b: usize, states: &[CustomerState]) -> bool {
    let x = &states[a];
    let y = &states[b];

    if x.input.priority != y.input.priority {
        return x.input.priority < y.input.priority;
    }

    match (x.pause_time, y.pause_time) {
        (Some(px), Some(py)) => {
            if px != py {
                px < py
            } else {
                x.input.id < y.input.id
            }
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            if x.input.arrival != y.input.arrival {
                x.input.arrival < y.input.arrival
            } else {
                x.input.id > y.input.id
            }
        }
    }
}

fn best_waiter(queue: &[usize], states: &[CustomerState]) -> Option<usize> {
    let mut best = None;
    for &idx in queue {
        if best.map_or(true, |b| better(idx, b, states)) {
            best = Some(idx);
        }
    }
    best
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);

        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }

        let deadline = match c.patience {
            Some(p) => Some(
                c.arrival
                    .checked_add(p)
                    .ok_or(Error::DeadlineOverflow { index, id: c.id })?,
            ),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut states: Vec<CustomerState> = customers
        .iter()
        .zip(initial_deadlines)
        .map(|(c, deadline)| CustomerState {
            input: c.clone(),
            deadline,
            remaining: c.service,
            accepted: false,
            started: false,
            terminal: false,
            total_waited: 0,
            queue_since: c.arrival,
            pause_time: None,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        states[a]
            .input
            .arrival
            .cmp(&states[b].input.arrival)
            .then_with(|| states[a].input.priority.cmp(&states[b].input.priority))
            .then_with(|| states[b].input.id.cmp(&states[a].input.id))
    });

    let mut events = Vec::new();
    let mut queue: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut has_dispatched = false;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    while next_arrival < arrivals.len() || running.is_some() || !queue.is_empty() {
        if running.is_none() && queue.is_empty() {
            if next_arrival >= arrivals.len() {
                break;
            }
            now = now.max(states[arrivals[next_arrival]].input.arrival);
        }

        // Phase A: finish the active customer at this tick.
        if let Some(r) = running.as_ref() {
            if r.finish == now {
                let r = running.take().unwrap();
                let id = states[r.idx].input.id;
                busy_ticks += (now - r.dispatch) as u128;
                states[r.idx].terminal = true;
                completed += 1;
                turnaround_ticks += (now - states[r.idx].input.arrival) as u128;
                completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
            }
        }

        // Phase B: expire existing waiters.
        let mut expiring: Vec<usize> = queue
            .iter()
            .copied()
            .filter(|&idx| states[idx].deadline.map_or(false, |d| d <= now))
            .collect();
        expiring.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].input.id.cmp(&states[b].input.id))
        });
        for idx in expiring {
            if let Some(pos) = queue.iter().position(|&x| x == idx) {
                queue.remove(pos);
                let waited = states[idx].total_waited + (now - states[idx].queue_since);
                wait_ticks += (now - states[idx].queue_since) as u128;
                states[idx].total_waited = waited;
                states[idx].terminal = true;
                reneged += 1;
                emit(
                    &mut events,
                    now,
                    states[idx].input.id,
                    Kind::Renege { waited },
                );
            }
        }

        // Phase C: process the whole arrival cohort.
        while next_arrival < arrivals.len()
            && states[arrivals[next_arrival]].input.arrival == now
        {
            let idx = arrivals[next_arrival];
            next_arrival += 1;
            let id = states[idx].input.id;
            emit(&mut events, now, id, Kind::Arrive);

            let gate = config.waiting_capacity + usize::from(running.is_none());
            if queue.len() < gate {
                states[idx].accepted = true;
                states[idx].queue_since = now;
                queue.push(idx);
                accepted += 1;
            } else {
                balked += 1;
                states[idx].terminal = true;
                emit(&mut events, now, id, Kind::Balk);
            }
        }

        // Phase D: preempt only after protection ends.
        if let Some(r) = running.as_ref() {
            if now >= r.work_at {
                if let Some(w) = best_waiter(&queue, &states) {
                    if states[w].input.priority < states[r.idx].input.priority {
                        let r = running.take().unwrap();
                        let remaining = r.finish - now;
                        let paused_id = states[r.idx].input.id;
                        let by = states[w].input.id;
                        busy_ticks += (now - r.dispatch) as u128;
                        emit(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt { remaining, by },
                        );
                        preemptions += 1;

                        let new_deadline = match states[r.idx].input.patience {
                            Some(budget) => Some(
                                now.checked_add(budget).ok_or(Error::TimeOverflow {
                                    customer: paused_id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                })?,
                            ),
                            None => None,
                        };
                        states[r.idx].remaining = remaining;
                        states[r.idx].queue_since = now;
                        states[r.idx].pause_time = Some(now);
                        states[r.idx].deadline = new_deadline;
                        queue.push(r.idx);
                    }
                }
            }
        }

        // Phase E: dispatch until the server is occupied or no waiter remains.
        loop {
            if running.is_some() {
                break;
            }
            let Some(idx) = best_waiter(&queue, &states) else {
                break;
            };
            let pos = queue.iter().position(|&x| x == idx).unwrap();
            queue.remove(pos);

            let waited = now - states[idx].queue_since;
            wait_ticks += waited as u128;
            states[idx].total_waited += waited;
            if let Some(deadline) = states[idx].deadline {
                let remaining_budget = deadline.saturating_sub(states[idx].queue_since);
                let _ = remaining_budget;
                states[idx].deadline = None;
            }

            let id = states[idx].input.id;
            let switching = if !has_dispatched {
                0
            } else if last_dispatch == Some(id) {
                0
            } else {
                config.switch_cost
            };

            let work_at = now.checked_add(switching).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[idx].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let is_resume = states[idx].started;
            if !is_resume {
                states[idx].started = true;
                started += 1;
            }
            emit(
                &mut events,
                now,
                id,
                if is_resume {
                    Kind::Resume {
                        remaining: states[idx].remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: states[idx].remaining,
                        work_at,
                        finish,
                    }
                },
            );

            has_dispatched = true;
            last_dispatch = Some(id);
            running = Some(Running {
                idx,
                dispatch: now,
                work_at,
                finish,
                remaining: states[idx].remaining,
                protected: work_at > now,
            });

            if finish == now {
                let _ = running.take();
                busy_ticks += 0;
                states[idx].terminal = true;
                completed += 1;
                turnaround_ticks += (now - states[idx].input.arrival) as u128;
                completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                continue;
            }
        }

        // Phase F: expire customers newly queued during C or D.
        let mut expiring: Vec<usize> = queue
            .iter()
            .copied()
            .filter(|&idx| states[idx].deadline.map_or(false, |d| d <= now))
            .collect();
        expiring.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].input.id.cmp(&states[b].input.id))
        });
        for idx in expiring {
            if let Some(pos) = queue.iter().position(|&x| x == idx) {
                queue.remove(pos);
                let waited = states[idx].total_waited + (now - states[idx].queue_since);
                wait_ticks += (now - states[idx].queue_since) as u128;
                states[idx].total_waited = waited;
                states[idx].terminal = true;
                reneged += 1;
                emit(
                    &mut events,
                    now,
                    states[idx].input.id,
                    Kind::Renege { waited },
                );
            }
        }

        // Jump to the next relevant time.
        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(states[arrivals[next_arrival]].input.arrival);
        }
        if let Some(r) = running.as_ref() {
            for t in [Some(r.work_at), Some(r.finish)].into_iter().flatten() {
                if t > now {
                    next_time = Some(next_time.map_or(t, |x| x.min(t)));
                }
            }
        }
        for &idx in &queue {
            if let Some(d) = states[idx].deadline {
                if d > now {
                    next_time = Some(next_time.map_or(d, |x| x.min(d)));
                }
            }
        }

        match next_time {
            Some(t) if t > now => now = t,
            _ if running.is_none() && queue.is_empty() && next_arrival < arrivals.len() => {
                now = states[arrivals[next_arrival]].input.arrival;
            }
            _ => {
                if running.is_none() && queue.is_empty() && next_arrival >= arrivals.len() {
                    break;
                }
                // A saturated u64 boundary can still be processed once; all
                // remaining actionable events must occur at this same tick.
                if now == u64::MAX {
                    break;
                }
            }
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}
