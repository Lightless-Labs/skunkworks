```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }

    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let divisor = gcd(n.unsigned_abs(), d as u128) as i128;
    n /= divisor;
    d /= divisor;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_parts(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_parts(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_parts(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_parts(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        from_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        use ParseKind::*;

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: Empty });
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
            if i == bytes.len() || !bytes[i].is_ascii_digit() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: InvalidByte,
                });
            }
        } else if !bytes[i].is_ascii_digit() {
            return Err(RationalError::Parse {
                at: i,
                kind: InvalidByte,
            });
        }

        let num_start = if bytes[0] == b'-' { 1 } else { 0 };
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        let num_end = i;

        let mut slash = None;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse {
                    at: i,
                    kind: InvalidByte,
                });
            }
            slash = Some(i);
            i += 1;
            if i == bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: InvalidByte,
                });
            }
            if !bytes[i].is_ascii_digit() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: InvalidByte,
                });
            }
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: InvalidByte,
                });
            }
        }

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: NonCanonical,
            });
        }
        if bytes[0] == b'-' && num_end - num_start == 1 && bytes[num_start] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: NonCanonical,
            });
        }
        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(RationalError::Parse {
                    at: start + 1,
                    kind: NonCanonical,
                });
            }
        }

        let negative = bytes[0] == b'-';
        let mut nmag: u128 = 0;
        for &b in &bytes[num_start..num_end] {
            nmag = nmag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
        }
        let nlimit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        if nmag > nlimit {
            return Err(RationalError::Parse {
                at: 0,
                kind: OutOfRange,
            });
        }
        let num = if negative {
            if nmag == 1u128 << 63 {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        if let (Some(start), Some(end), Some(slash_at)) = (den_start, den_end, slash) {
            let mut dmag: u128 = 0;
            for &b in &bytes[start..end] {
                dmag = dmag
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
            }
            if dmag > i64::MAX as u128 {
                return Err(RationalError::Parse {
                    at: start,
                    kind: OutOfRange,
                });
            }
            if dmag == 0 {
                return Err(RationalError::Parse {
                    at: start,
                    kind: ZeroDenominator,
                });
            }
            if dmag == 1 {
                return Err(RationalError::Parse {
                    at: slash_at,
                    kind: NonCanonical,
                });
            }
            if num == 0 {
                return Err(RationalError::Parse {
                    at: slash_at,
                    kind: NonCanonical,
                });
            }
            if gcd(nmag, dmag) > 1 {
                return Err(RationalError::Parse {
                    at: slash_at,
                    kind: NonCanonical,
                });
            }
            Ok(Rational {
                num,
                den: dmag as i64,
            })
        } else {
            Ok(Rational { num, den: 1 })
        }
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
