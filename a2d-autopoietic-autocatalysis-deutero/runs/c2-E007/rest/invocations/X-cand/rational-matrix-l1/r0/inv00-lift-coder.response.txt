```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

fn from_i128_pair(num: i128, den: i128) -> Result<Rational, RationalError> {
    from_i128(num, den)
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let n = self.num as i128 * b + rhs.num as i128 * a;
        let d = a * rhs.den as i128;
        from_i128_pair(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let n = self.num as i128 * b - rhs.num as i128 * a;
        let d = a * rhs.den as i128;
        from_i128_pair(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs(), self.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.num as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.den as i128 / g1);
        from_i128_pair(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.num.unsigned_abs()) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.den as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.num as i128 / g1);
        from_i128_pair(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let len = bytes.len();
        let parse_error = |at, kind| RationalError::Parse { at, kind };
        if len == 0 {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let num_start = i;
        while i < len && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_error(i, if i == len { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < len && bytes[i] == b'/' {
            i += 1;
            den_start = Some(i);
            while i < len && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
        }
        if i < len {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let num_digits = &bytes[num_start..num_end];
        let num_token_start = if negative { 1 } else { 0 };
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_error(num_token_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(parse_error(ds + 1, ParseKind::NonCanonical));
            }
        }

        let num_mag = parse_u128_digits(num_digits);
        let num_limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag > num_limit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let den_mag = if let (Some(ds), Some(de)) = (den_start, den_end) {
            let v = parse_u128_digits(&bytes[ds..de]);
            if v > i64::MAX as u128 {
                return Err(parse_error(ds, ParseKind::OutOfRange));
            }
            v
        } else {
            1
        };

        if den_start.is_some() && den_mag == 0 {
            return Err(parse_error(den_start.unwrap(), ParseKind::ZeroDenominator));
        }

        if let (Some(ds), _) = (den_start, den_end) {
            if den_mag == 1 {
                return Err(parse_error(ds - 1, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_error(ds - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(parse_error(ds - 1, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        Ok(Rational {
            num: signed_num,
            den: den_mag as i64,
        })
    }
}

fn parse_u128_digits(digits: &[u8]) -> u128 {
    let mut value = 0u128;
    for &b in digits {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u128);
    }
    value
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
