#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty",
    "ninety",
];

const SCALES: [&str; 7] = [
    "", "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn spell_under_thousand(n: u16) -> String {
    let mut out = String::new();
    let hundreds = n / 100;
    let remainder = n % 100;

    if hundreds != 0 {
        out.push_str(SMALL[hundreds as usize]);
        out.push_str(" hundred");
        if remainder != 0 {
            out.push_str(" and ");
        }
    }

    if remainder >= 20 {
        out.push_str(TENS[(remainder / 10) as usize]);
        let units = remainder % 10;
        if units != 0 {
            out.push('-');
            out.push_str(SMALL[units as usize]);
        }
    } else if remainder != 0 {
        out.push_str(SMALL[remainder as usize]);
    }

    out
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut parts: Vec<String> = Vec::new();
    for i in (1..7).rev() {
        if groups[i] != 0 {
            parts.push(format!("{} {}", spell_under_thousand(groups[i]), SCALES[i]));
        }
    }

    let units = groups[0];
    if units != 0 {
        let mut unit_spelling = spell_under_thousand(units);
        if units < 100 && !parts.is_empty() {
            if let Some(last) = parts.last_mut() {
                last.push_str(" and ");
            }
            parts.push(unit_spelling);
        } else {
            parts.push(std::mem::take(&mut unit_spelling));
        }
    }

    let mut result = parts.join(" ");
    if value < 0 {
        result.insert_str(0, "minus ");
    }
    Ok(result)
}

fn word_value(word: &str) -> Option<(u128, WordKind)> {
    for (i, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some((i as u128, WordKind::Add));
        }
    }
    for (i, spelling) in TENS.iter().enumerate().skip(2) {
        if word == *spelling {
            return Some(((i * 10) as u128, WordKind::Add));
        }
    }
    match word {
        "zero" => Some((0, WordKind::Add)),
        "hundred" => Some((100, WordKind::Hundred)),
        "thousand" => Some((1_000, WordKind::Scale)),
        "million" => Some((1_000_000, WordKind::Scale)),
        "billion" => Some((1_000_000_000, WordKind::Scale)),
        "trillion" => Some((1_000_000_000_000, WordKind::Scale)),
        "quadrillion" => Some((1_000_000_000_000_000, WordKind::Scale)),
        "quintillion" => Some((1_000_000_000_000_000_000, WordKind::Scale)),
        "and" => Some((0, WordKind::And)),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum WordKind {
    Add,
    Hundred,
    Scale,
    And,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let tokens: Vec<(&str, usize)> = input
        .split(|c| c == ' ' || c == '-')
        .scan(0usize, |offset, token| {
            let start = *offset;
            *offset += token.len() + 1;
            Some((token, start))
        })
        .collect();

    for (i, (token, index)) in tokens.iter().enumerate() {
        if *token == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: *index });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().map(|(token, _)| *token == "minus").unwrap_or(false);
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for (token, _) in arithmetic_tokens {
        let (value, kind) = word_value(token).expect("dictionary scan validated token");
        match kind {
            WordKind::Add => group += value,
            WordKind::Hundred => group *= 100,
            WordKind::Scale => {
                total += group * value;
                group = 0;
            }
            WordKind::And => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != bytes {
        let mismatch = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(signed)
}
