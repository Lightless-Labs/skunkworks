#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    MissingCheck { offset: usize },
    InvalidByte { offset: usize, byte: u8 },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    CheckMismatch {
        offset: usize,
        expected: u8,
        found: u8,
    },
}

const DATA_ALPHABET: &[u8; 32] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
const CHECK_ALPHABET: &[u8; 37] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";

pub fn encode(input: &[u8], checked: bool) -> String {
    let mut output = String::new();
    let mut remainder = 0u8;
    let mut accumulator = 0u16;
    let mut bit_count = 0u8;

    for &byte in input {
        accumulator = (accumulator << 8) | byte as u16;
        bit_count += 8;

        while bit_count >= 5 {
            bit_count -= 5;
            let value = ((accumulator >> bit_count) & 0x1f) as u8;
            output.push(DATA_ALPHABET[value as usize] as char);
            remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
        }
    }

    if bit_count != 0 {
        let value = ((accumulator << (5 - bit_count)) & 0x1f) as u8;
        output.push(DATA_ALPHABET[value as usize] as char);
        remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
    }

    if checked {
        output.push(CHECK_ALPHABET[remainder as usize] as char);
    }

    output
}

fn data_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' | b'a'..=b'z' => {
            let upper = byte.to_ascii_uppercase();
            match upper {
                b'I' | b'L' => Some(1),
                b'O' => Some(0),
                b'U' => None,
                _ => DATA_ALPHABET
                    .iter()
                    .position(|&symbol| symbol == upper)
                    .map(|value| value as u8),
            }
        }
        _ => None,
    }
}

fn check_value(byte: u8) -> Option<u8> {
    match byte {
        b'*' => Some(32),
        b'~' => Some(33),
        b'$' => Some(34),
        b'=' => Some(35),
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' | b'a'..=b'z' => {
            let upper = byte.to_ascii_uppercase();
            if upper == b'U' {
                Some(36)
            } else {
                data_value(upper)
            }
        }
        _ => None,
    }
}

pub fn decode(input: &str, checked: bool) -> Result<Vec<u8>, DecodeError> {
    let significant: Vec<(usize, u8)> = input
        .as_bytes()
        .iter()
        .copied()
        .enumerate()
        .filter(|&(_, byte)| byte != b'-')
        .collect();

    if checked && significant.is_empty() {
        return Err(DecodeError::MissingCheck {
            offset: input.len(),
        });
    }

    let data_len = if checked {
        significant.len() - 1
    } else {
        significant.len()
    };

    let mut values = Vec::with_capacity(data_len);
    for &(offset, byte) in &significant[..data_len] {
        match data_value(byte) {
            Some(value) => values.push((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    }

    let check = if checked {
        let (offset, byte) = significant[data_len];
        match check_value(byte) {
            Some(value) => Some((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    } else {
        None
    };

    let residue = data_len % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength {
            offset: input.len(),
        });
    }

    let unused_bits = match residue {
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => 0,
    };

    if unused_bits != 0 {
        let (offset, value) = values[data_len - 1];
        let mask = (1u8 << unused_bits) - 1;
        if value & mask != 0 {
            return Err(DecodeError::TrailingBits { offset });
        }
    }

    let mut output = Vec::with_capacity(data_len * 5 / 8);
    let mut accumulator = 0u16;
    let mut bit_count = 0u8;
    let mut remainder = 0u8;

    for &(_, value) in &values {
        remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
        accumulator = (accumulator << 5) | value as u16;
        bit_count += 5;

        while bit_count >= 8 {
            bit_count -= 8;
            output.push((accumulator >> bit_count) as u8);
        }
    }

    if let Some((offset, found)) = check {
        if found != remainder {
            return Err(DecodeError::CheckMismatch {
                offset,
                expected: remainder,
                found,
            });
        }
    }

    Ok(output)
}
