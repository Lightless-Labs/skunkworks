#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        output.push(ALPHABET[(a >> 2) as usize] as char);
        output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
        output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
        output.push(ALPHABET[(c & 0x3f) as usize] as char);
    }

    match chunks.remainder() {
        [a] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push_str("==");
        }
        [a, b] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        _ => {}
    }

    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut output = Vec::new();
    let mut group: Vec<(u8, usize)> = Vec::with_capacity(4);
    let mut pending_padding: Option<usize> = None;
    let mut terminal_padding = false;

    for (offset, &byte) in input.iter().enumerate() {
        if terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if let Some(padding_offset) = pending_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }

            let first = digit(group[0].0).expect("group digit was validated");
            let second = digit(group[1].0).expect("group digit was validated");
            output.push((first << 2) | (second >> 4));
            pending_padding = None;
            group.clear();
            terminal_padding = true;
            let _ = padding_offset;
            continue;
        }

        if byte == b'=' {
            match group.len() {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => pending_padding = Some(offset),
                3 => {
                    let a = digit(group[0].0).expect("group digit was validated");
                    let b = digit(group[1].0).expect("group digit was validated");
                    let c = digit(group[2].0).expect("group digit was validated");
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                    group.clear();
                    terminal_padding = true;
                }
                _ => unreachable!(),
            }
            continue;
        }

        let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
        group.push((byte, offset));

        if group.len() == 4 {
            let a = digit(group[0].0).expect("group digit was validated");
            let b = digit(group[1].0).expect("group digit was validated");
            let c = digit(group[2].0).expect("group digit was validated");
            let d = value;
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
            group.clear();
        }
    }

    if let Some(offset) = pending_padding {
        return Err(Error::BadPadding { offset });
    }
    if !group.is_empty() {
        return Err(Error::Truncated {
            offset: group[0].1,
        });
    }

    Ok(output)
}
