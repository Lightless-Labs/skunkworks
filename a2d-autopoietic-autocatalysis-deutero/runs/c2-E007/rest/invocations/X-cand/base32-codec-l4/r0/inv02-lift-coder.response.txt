#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

pub struct Decoder {
    config: Config,
    accumulator: u8,
    pending_bits: u8,
    data_count: usize,
    total_consumed: usize,
    last_data_offset: Option<usize>,
    last_symbol_value: u8,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    closed: bool,
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut all = Vec::with_capacity(self.pending.len() + input.len());
        all.extend_from_slice(&self.pending);
        all.extend_from_slice(input);

        let complete_len = (all.len() / 5) * 5;
        let mut output = String::new();
        let alphabet = alphabet_bytes(self.config.alphabet);

        for chunk in all[..complete_len].chunks_exact(5) {
            let b0 = chunk[0] as u64;
            let b1 = chunk[1] as u64;
            let b2 = chunk[2] as u64;
            let b3 = chunk[3] as u64;
            let b4 = chunk[4] as u64;
            let bits = (b0 << 32) | (b1 << 24) | (b2 << 16) | (b3 << 8) | b4;

            for i in 0..8 {
                let shift = 35 - i * 5;
                output.push(alphabet[((bits >> shift) & 31) as usize] as char);
            }
        }

        self.pending.clear();
        self.pending.extend_from_slice(&all[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        let alphabet = alphabet_bytes(self.config.alphabet);

        if !self.pending.is_empty() {
            let mut accumulator: u64 = 0;
            for &byte in &self.pending {
                accumulator = (accumulator << 8) | byte as u64;
            }

            let bit_count = self.pending.len() * 8;
            let symbol_count = (bit_count + 4) / 5;
            let shifted = accumulator << (40 - bit_count);

            for i in 0..symbol_count {
                let shift = 35 - i * 5;
                output.push(alphabet[((shifted >> shift) & 31) as usize] as char);
            }

            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            accumulator: 0,
            pending_bits: 0,
            data_count: 0,
            total_consumed: 0,
            last_data_offset: None,
            last_symbol_value: 0,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: 0,
            closed: false,
        }
    }

    fn symbol_value(alphabet: Alphabet, byte: u8) -> Option<u8> {
        alphabet_bytes(alphabet)
            .iter()
            .position(|&symbol| symbol == byte)
            .map(|value| value as u8)
    }

    fn expected_padding_for_residue(residue: usize) -> Option<usize> {
        match residue {
            0 => Some(0),
            2 => Some(6),
            4 => Some(4),
            5 => Some(3),
            7 => Some(1),
            _ => None,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &original in input.as_bytes() {
            let offset = next.total_consumed;
            let upper = original.to_ascii_uppercase();

            if original == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }
                if next.first_padding_offset.is_none() {
                    let residue = next.data_count % 8;
                    let Some(expected) = Self::expected_padding_for_residue(residue) else {
                        return Err(StreamError::Padding { offset });
                    };
                    if residue == 0 {
                        return Err(StreamError::Padding { offset });
                    }
                    next.first_padding_offset = Some(offset);
                    next.expected_padding = expected;
                }

                if next.padding_count >= next.expected_padding {
                    return Err(StreamError::Padding { offset });
                }
                next.padding_count += 1;
                next.total_consumed += 1;
                continue;
            }

            let Some(value) = Self::symbol_value(next.config.alphabet, upper) else {
                return Err(StreamError::InvalidByte {
                    offset,
                    byte: original,
                });
            };

            if original.is_ascii_lowercase() && next.config.canonical {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: upper,
                });
            }

            if next.first_padding_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.accumulator = (next.accumulator << 5) | value;
            next.pending_bits += 5;
            while next.pending_bits >= 8 {
                next.pending_bits -= 8;
                output.push((next.accumulator >> next.pending_bits) as u8);
                if next.pending_bits == 0 {
                    next.accumulator = 0;
                } else {
                    next.accumulator &= (1u8 << next.pending_bits) - 1;
                }
            }

            next.data_count += 1;
            next.last_data_offset = Some(offset);
            next.last_symbol_value = value;
            next.total_consumed += 1;
        }

        self.apply_state(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let Some(expected) = Self::expected_padding_for_residue(residue) else {
            return Err(StreamError::InvalidLength {
                offset: self.total_consumed,
            });
        };

        if self.config.padding == Padding::Required && self.padding_count != expected {
            return Err(StreamError::Padding {
                offset: self
                    .first_padding_offset
                    .unwrap_or(self.total_consumed),
            });
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };

        if unused_bits != 0 {
            let mask = (1u8 << unused_bits) - 1;
            if self.last_symbol_value & mask != 0 {
                return Err(StreamError::TrailingBits {
                    offset: self.last_data_offset.unwrap_or(self.total_consumed),
                });
            }
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            accumulator: self.accumulator,
            pending_bits: self.pending_bits,
            data_count: self.data_count,
            total_consumed: self.total_consumed,
            last_data_offset: self.last_data_offset,
            last_symbol_value: self.last_symbol_value,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }

    fn apply_state(&mut self, state: Self) {
        self.accumulator = state.accumulator;
        self.pending_bits = state.pending_bits;
        self.data_count = state.data_count;
        self.total_consumed = state.total_consumed;
        self.last_data_offset = state.last_data_offset;
        self.last_symbol_value = state.last_symbol_value;
        self.first_padding_offset = state.first_padding_offset;
        self.padding_count = state.padding_count;
        self.expected_padding = state.expected_padding;
        self.closed = state.closed;
    }
}
