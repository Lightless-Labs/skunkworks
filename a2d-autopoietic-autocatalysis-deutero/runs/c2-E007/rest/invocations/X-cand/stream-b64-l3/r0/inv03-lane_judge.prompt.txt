fn encode_digit(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn decode_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_triplet(bytes: &[u8], alphabet: Alphabet, padding: Padding) -> Vec<u8> {
    let mut out = Vec::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(encode_digit(a >> 2, alphabet));
    out.push(encode_digit(((a & 0x03) << 4) | (b >> 4), alphabet));

    if bytes.len() > 1 {
        out.push(encode_digit(((b & 0x0f) << 2) | (c >> 6), alphabet));
    } else if padding != Padding::Forbidden {
        out.push(b'=');
    }

    if bytes.len() > 2 {
        out.push(encode_digit(c & 0x3f, alphabet));
    } else if padding != Padding::Forbidden {
        out.push(b'=');
    }

    out
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    symbols_in_row: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            symbols_in_row: 0,
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = std::mem::take(&mut self.tail);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut symbols = Vec::with_capacity(complete_len / 3 * 4);
        for chunk in data[..complete_len].chunks_exact(3) {
            symbols.extend(encode_triplet(chunk, self.options.alphabet, self.options.padding));
        }
        self.tail.extend_from_slice(&data[complete_len..]);

        let output = wrap_symbols(&symbols, self.options, &mut self.symbols_in_row);
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let symbols = if self.tail.is_empty() {
            Vec::new()
        } else {
            encode_triplet(&self.tail, self.options.alphabet, self.options.padding)
        };
        self.tail.clear();

        let output = wrap_symbols(&symbols, self.options, &mut self.symbols_in_row);
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }
}

    options: Options,
    group: Vec<Symbol>,
    terminal_padding: bool,
    physical_offset: usize,
    symbols_in_row: usize,
    wrap_state: WrapState,
    finished: bool,

}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            physical_offset: 0,
            symbols_in_row: 0,
            wrap_state: WrapState::Row,
            finished: false,
            sticky_error: None,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.physical_offset;
            self.physical_offset += 1;

            if let Err(error) = self.consume(byte, offset, &mut output) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    fn consume(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::IgnoreAscii
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) =>
            {
                return Ok(());
            }

                    WrapState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }

                            separator_offset: cr_offset,
                        };
                        return Ok(());
                    }
                    WrapState::NeedSymbol { .. } => {
                        self.wrap_state = WrapState::Row;
                    }
                    WrapState::Row => {}

                        LineEnding::Lf if byte == b'\n' => {

                                separator_offset: offset,
                            };
                            return Ok(());

                            return Ok(());

                                return self.fail(Error::BadWrap { offset });
                            }
                            let result = self.scan_symbol(byte, offset, &mut out);
                            if let Err(error) = result {
                                return self.fail(error);
                            }
                            self.row_symbols += 1;
                        }
                        FrameState::Row => {
                            if byte == b'\r' || byte == b'\n' {
                                return self.fail(Error::BadWrap { offset });
                            }
                            if self.row_symbols == width {
                                self.frame_state = FrameState::NeedSeparator { first_offset: offset };
                                match ending {
                                    LineEnding::Lf if byte == b'\n' => {
                                        self.frame_state = FrameState::NeedSymbol { first_offset: offset };
                                        continue;
                                    }
                                    LineEnding::CrLf if byte == b'\r' => {
                                        self.frame_state = FrameState::NeedLf { cr_offset: offset };
                                        continue;
                                    }
                                    _ => return self.fail(Error::BadWrap { offset }),
                                }
                            }
                            let result = self.scan_symbol(byte, offset, &mut out);
                            if let Err(error) = result {
                                return self.fail(error);
                            }
                            self.row_symbols += 1;
                            if self.row_symbols == width {
                                self.frame_state = FrameState::NeedSeparator { first_offset: offset + 1 };
                            }
                        }
                    }
                }
                Whitespace::None | Whitespace::IgnoreAscii => {
                    if let Err(error) = self.scan_symbol(byte, offset, &mut out) {
                        return self.fail(error);
                    }
                }
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedSeparator { .. } => {}
                FrameState::NeedLf { cr_offset } => {
                    return self.fail(Error::BadWrap { offset: cr_offset });
                }
                FrameState::NeedSymbol { first_offset } => {
                    return self.fail(Error::BadWrap { offset: first_offset });
                }
                FrameState::Row => {}
            }
        }

        if self.group.len() == 3 && self.group[2].0 == b'=' {
            return self.fail(Error::BadPadding {
                offset: self.group[2].1,
            });
        }

        let mut out = Vec::new();
        match self.group.len() {
            0 => {}
            1 => {
                return self.fail(Error::Truncated {
                    offset: self.group[0].1,
                });
            }
            2 | 3 => match self.options.padding {
                Padding::Required => {
                    return self.fail(Error::Truncated {
                        offset: self.group[0].1,
                    });
                }
                Padding::Forbidden | Padding::Optional => {
                    let a = digit(self.group[0].0, self.options.alphabet).unwrap();
                    let b = digit(self.group[1].0, self.options.alphabet).unwrap();
                    out.push((a << 2) | (b >> 4));
                    if self.group.len() == 3 {
                        let c = digit(self.group[2].0, self.options.alphabet).unwrap();
                        out.push((b << 4) | (c >> 2));
                    }
                }
            },
            _ => unreachable!(),
        }

        self.group.clear();
        self.finished = true;
        Ok(out)
    }
