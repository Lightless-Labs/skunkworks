            break;

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in the required order before other checks.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal, always taking the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut removed = vec![false; n];
            for &id in &order {
                removed[id] = true;
            }
            removed
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut worker_free_at = vec![0u64; workers];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finished_now = Vec::<(usize, usize)>::new();
        let mut still_running = Vec::with_capacity(running.len());
        for (finish, task_id, worker_id) in running.drain(..) {
            if finish == now {
                finished_now.push((task_id, worker_id));
            } else {
                still_running.push((finish, task_id, worker_id));
            }
        }
        running = still_running;
        finished_now.sort_by_key(|&(task_id, _)| task_id);

        for (task_id, worker_id) in finished_now {
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(now);
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });

        loop {
            let worker_id = match (0..workers).find(|&w| worker_free_at[w] <= now) {
                Some(w) => w,
                None => break,
            };

            let selected = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && tasks[id].deps.iter().all(|&dep| completed[dep])
                })
                .max_by(|&a, &b| {
                    tail[a]
                        .cmp(&tail[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });

            let task_id = match selected {
                Some(id) => id,
                None => break,
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;

                task: task_id,
                stage: Stage::Turnaround,
            })?;

            worker_free_at[worker_id] = worker_available;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,

                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,

                completed[task_id] = true;

                    time: now,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,

        // Advance to the earliest strictly later relevant event.
        let next_finish = running.iter().map(|&(finish, _, _)| finish).min();
        let next_release = (0..n)
            .filter(|&id| !started[id])
            .map(|id| tasks[id].release)
            .filter(|&release| release > now)
            .min();
        let next_worker = worker_free_at
            .iter()
            .copied()
            .filter(|&time| time > now)
            .min();

        now = [next_finish, next_release, next_worker]
            .into_iter()
            .flatten()
            .min()
            .expect("unfinished tasks must have a future scheduling event");
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before any other validation or computation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and the indegrees used by Kahn traversal.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Use a min-heap so the smallest currently available task is removed first.
    let mut available = std::collections::BinaryHeap::<std::cmp::Reverse<usize>>::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &task_id in &order {
            removed[task_id] = true;
        }
        for (task_id, was_removed) in removed.into_iter().enumerate() {
            if !was_removed {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_successor_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut worker_available_at = vec![0u64; workers];
    let mut events = Vec::new();
    let mut completed_predecessors = vec![0usize; n];
    let mut ready = std::collections::BinaryHeap::<(u64, i32, std::cmp::Reverse<usize>)>::new();
    let mut time = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    // Tasks with no dependencies begin in the ready heap; release eligibility is
    // checked against the current simulation time when selecting each task.
    for task_id in 0..n {
        if tasks[task_id].deps.is_empty() {
            ready.push((
                tails[task_id],
                tasks[task_id].priority,
                std::cmp::Reverse(task_id),
            ));
        }
    }

    while completed_count < n {
        // Complete all running tasks finishing now, in ascending task ID.
        running.sort_unstable_by_key(|&(finish, task_id, _)| (finish, task_id));
        let mut completed_now = Vec::new();
        while running.first().is_some_and(|&(finish, _, _)| finish == time) {
            let (_, task_id, worker) = running.remove(0);
            completed_now.push((task_id, worker));
        }

        for (task_id, worker) in completed_now {
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(time);
            events.push(Event {
                time,
                task: task_id,
                worker,
                kind: EventKind::Finish,
            });

            for &child in &successors[task_id] {
                completed_predecessors[child] += 1;
                if completed_predecessors[child] == tasks[child].deps.len() {
                    ready.push((
                        tails[child],
                        tasks[child].priority,
                        std::cmp::Reverse(child),
                    ));
                }
            }
        }

        loop {
            // Find the lowest-ID worker available now.
            let worker = (0..workers).find(|&w| worker_available_at[w] <= time);

            // Select the highest-ranked eligible task, temporarily skipping tasks
            // whose release time has not arrived yet.
            let mut deferred = Vec::new();
            let selected = loop {
                let Some(entry) = ready.pop() else {
                    break None;
                };
                let task_id = entry.2 .0;
                if !started[task_id] && tasks[task_id].release <= time {
                    break Some((task_id, entry));
                }
                deferred.push(entry);
            };
            for entry in deferred {
                ready.push(entry);
            }

            let (Some(worker), Some((task_id, _entry))) = (worker, selected) else {
                // Restore a selected task if there is no worker available.
                if let Some((task_id, entry)) = selected {
                    let _ = task_id;
                    ready.push(entry);
                }
                break;
            };

            started[task_id] = true;
            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(turnaround)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            worker_available_at[worker] = available_at;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            events.push(Event {
                time,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                // Immediate completion precedes the next admission at this tick.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(time);
                events.push(Event {
                    time,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &successors[task_id] {
                    completed_predecessors[child] += 1;
                    if completed_predecessors[child] == tasks[child].deps.len() {
                        ready.push((
                            tails[child],
                            tasks[child].priority,
                            std::cmp::Reverse(child),
                        ));
                    }
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time && next_time.map_or(true, |current| candidate < current) {
                next_time = Some(candidate);
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > time {
                consider(tasks[task_id].release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        if let Some(next) = next_time {
            time = next;
        } else {
            // With valid dependencies and all events accounted for, unfinished
            // tasks must have a future release, finish, or worker availability.
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("acyclic tasks are all scheduled"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate all dependency entries before any further validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = Vec::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut children = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest available ID.
    let mut available: Vec<usize> = (0..n).filter(|&i| indegree[i] == 0).collect();
    let mut kahn_order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, &id)| id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        kahn_order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if kahn_order.len() != n {
        let blocked = indegree
            .iter()
            .enumerate()
            .filter_map(|(id, &degree)| if degree > 0 { Some(id) } else { None })
            .collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in kahn_order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // finish, task, worker
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, ordered by task ID.
        let mut finished_now: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for item in running.drain(..) {
            if item.0 == now {
                finished_now.push(item);
            } else {
                still_running.push(item);
            }
        }
        running = still_running;
        finished_now.sort_by_key(|&(_, task, _)| task);
        for &(finish, task, worker) in &finished_now {
            completed[task] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task,
                worker,
                kind: EventKind::Finish,
            });
        }

        // Repeatedly start one eligible task, reevaluating after each start.
        loop {
            let worker = (0..workers).find(|&w| worker_available[w] <= now);
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if !tasks[id].deps.iter().all(|&dep| completed[dep]) {
                    continue;
                }
                match best {
                    None => best = Some(id),
                    Some(current) => {
                        let candidate_key =
                            (tails[id], tasks[id].priority, std::cmp::Reverse(id));
                        let current_key = (
                            tails[current],
                            tasks[current].priority,
                            std::cmp::Reverse(current),
                        );
                        if candidate_key > current_key {
                            best = Some(id);
                        }
                    }
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if finish == now {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change eligibility.
        let next_finish = running.iter().map(|&(finish, _, _)| finish).min();
        let next_release = (0..n)
            .filter(|&id| !started[id] && tasks[id].release > now)
            .map(|id| tasks[id].release)
            .min();
        let next_worker = worker_available
            .iter()
            .copied()
            .filter(|&time| time > now)
            .min();

        let next_time = [next_finish, next_release, next_worker]
            .into_iter()
            .flatten()
            .min();

        match next_time {
            Some(time) => now = time,
            None => {
                // With an acyclic graph and validated times, progress must be possible.
                // This branch is a defensive guard against any unexpected deadlock.
                break;
            }
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task must be scheduled"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
