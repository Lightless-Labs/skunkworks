#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn alphabet_byte(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 if alphabet == Alphabet::Standard => b'+',
        63 if alphabet == Alphabet::Standard => b'/',
        62 => b'-',
        _ => b'_',
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    options: Options,
    buffer: [u8; 2],
    buffered: usize,
    symbols_in_row: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffer: [0; 2],
            buffered: 0,
            symbols_in_row: 0,
            finished: false,
        })
    }

    fn emit_symbol(&mut self, output: &mut String, value: u8) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            if self.symbols_in_row == width {
                match ending {
                    LineEnding::Lf => output.push('\n'),
                    LineEnding::CrLf => output.push_str("\r\n"),
                }
                self.symbols_in_row = 0;
            }
        }
        output.push(alphabet_byte(value, self.options.alphabet) as char);
        self.symbols_in_row += 1;
    }

    fn emit_triplet(&mut self, output: &mut String, bytes: [u8; 3]) {
        self.emit_symbol(output, bytes[0] >> 2);
        self.emit_symbol(output, ((bytes[0] & 0x03) << 4) | (bytes[1] >> 4));
        self.emit_symbol(output, ((bytes[1] & 0x0f) << 2) | (bytes[2] >> 6));
        self.emit_symbol(output, bytes[2] & 0x3f);
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.buffered > 0 {
            while self.buffered < 2 && index < input.len() {
                self.buffer[self.buffered] = input[index];
                self.buffered += 1;
                index += 1;
            }
            if self.buffered == 2 && index < input.len() {
                self.emit_triplet(
                    &mut output,
                    [self.buffer[0], self.buffer[1], input[index]],
                );
                self.buffered = 0;
                index += 1;
            }
        }

        while index + 3 <= input.len() {
            self.emit_triplet(
                &mut output,
                [input[index], input[index + 1], input[index + 2]],
            );
            index += 3;
        }

        while index < input.len() {
            self.buffer[self.buffered] = input[index];
            self.buffered += 1;
            index += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.buffered {
            0 => {}
            1 => {
                let byte = self.buffer[0];
                self.emit_symbol(&mut output, byte >> 2);
                self.emit_symbol(&mut output, (byte & 0x03) << 4);
                if self.options.padding != Padding::Forbidden {
                    self.emit_symbol(&mut output, 64);
                    self.emit_symbol(&mut output, 64);
                }
            }
            2 => {
                let first = self.buffer[0];
                let second = self.buffer[1];
                self.emit_symbol(&mut output, first >> 2);
                self.emit_symbol(&mut output, ((first & 0x03) << 4) | (second >> 4));
                self.emit_symbol(&mut output, (second & 0x0f) << 2);
                if self.options.padding != Padding::Forbidden {
                    self.emit_symbol(&mut output, 64);
                }
            }
            _ => unreachable!(),
        }
        self.buffered = 0;
        Ok(output)
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

pub struct Decoder {
    options: Options,
    group: [Symbol; 4],
    group_len: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    physical_offset: usize,
    row_symbols: usize,
    framing: FramingState,
    finished: bool,
    sticky_error: Option<Error>,
}

#[derive(Clone, Copy)]
enum FramingState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: [Symbol { byte: 0, offset: 0 }; 4],
            group_len: 0,
            pending_second_padding: None,
            terminal_padding: false,
            physical_offset: 0,
            row_symbols: 0,
            framing: FramingState::Normal,
            finished: false,
            sticky_error: None,
        })
    }

    fn record_error(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn output_group(&mut self, output: &mut Vec<u8>) {
        let a = alphabet_value(self.group[0].byte, self.options.alphabet).unwrap();
        let b = alphabet_value(self.group[1].byte, self.options.alphabet).unwrap();

        output.push((a << 2) | (b >> 4));

        if self.group[2].byte != b'=' {
            let c = alphabet_value(self.group[2].byte, self.options.alphabet).unwrap();
            output.push((b << 4) | (c >> 2));

            if self.group[3].byte != b'=' {
                let d = alphabet_value(self.group[3].byte, self.options.alphabet).unwrap();
                output.push((c << 6) | d);
            } else {
                self.terminal_padding = true;
            }
        } else {
            self.terminal_padding = true;
        }

        if self.group[3].byte == b'=' {
            self.terminal_padding = true;
        }
        self.group_len = 0;
        self.pending_second_padding = None;
    }

    fn accept_symbol(&mut self, symbol: Symbol, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset: symbol.offset });
        }

        if let Some(pending_offset) = self.pending_second_padding {
            if symbol.byte != b'=' {
                return Err(Error::BadPadding { offset: symbol.offset });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: symbol.offset });
            }
            self.group[3] = symbol;
            self.group_len = 4;
            self.pending_second_padding = None;
            self.output_group(output);
            let _ = pending_offset;
            return Ok(());
        }

        if symbol.byte == b'=' {
            match self.group_len {
                0 | 1 => return Err(Error::BadPadding { offset: symbol.offset }),
                2 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset: symbol.offset });
                    }
                    self.group[2] = symbol;
                    self.group_len = 3;
                    self.pending_second_padding = Some(symbol.offset);
                    return Ok(());
                }
                3 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset: symbol.offset });
                    }
                    self.group[3] = symbol;
                    self.group_len = 4;
                    self.output_group(output);
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = alphabet_value(symbol.byte, self.options.alphabet);
        if value.is_none() {
            return Err(Error::InvalidByte {
                offset: symbol.offset,
                byte: symbol.byte,
            });
        }

        if self.group_len == 3 && self.group[2].byte == b'=' {
            return Err(Error::BadPadding { offset: symbol.offset });
        }

        self.group[self.group_len] = symbol;
        self.group_len += 1;
        if self.group_len == 4 {
            self.output_group(output);
        }
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => {
                self.accept_symbol(Symbol { byte, offset }, output)
            }
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.accept_symbol(Symbol { byte, offset }, output)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.framing {
                    FramingState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.framing = FramingState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        return Ok(());
                    }
                    FramingState::NeedSymbol { .. } => {
                        self.framing = FramingState::Normal;
                    }
                    FramingState::Normal => {}
                }

                if self.row_symbols == width {
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.row_symbols = 0;
                            self.framing = FramingState::NeedSymbol { separator_offset: offset };
                            return Ok(());
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.row_symbols = 0;
                            self.framing = FramingState::NeedLf { cr_offset: offset };
                            return Ok(());
                        }
                        _ => return Err(Error::BadWrap { offset }),
                    }
                }

                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset });
                }

                self.row_symbols += 1;
                self.accept_symbol(Symbol { byte, offset }, output)
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.physical_offset;
            self.physical_offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut output) {
                return Err(self.record_error(error));
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.framing {
                FramingState::NeedLf { cr_offset } => {
                    let error = Error::BadWrap { offset: cr_offset };
                    return Err(self.record_error(error));
                }
                FramingState::NeedSymbol { separator_offset } => {
                    let error = Error::BadWrap { offset: separator_offset };
                    return Err(self.record_error(error));
                }
                FramingState::Normal => {}
            }
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            return Err(self.record_error(error));
        }

        let mut output = Vec::new();
        match self.group_len {
            0 => {}
            1 => {
                let error = Error::Truncated {
                    offset: self.group[0].offset,
                };
                return Err(self.record_error(error));
            }
            2 | 3 if self.options.padding != Padding::Required => {
                let a = alphabet_value(self.group[0].byte, self.options.alphabet).unwrap();
                let b = alphabet_value(self.group[1].byte, self.options.alphabet).unwrap();
                output.push((a << 2) | (b >> 4));
                if self.group_len == 3 {
                    let c = alphabet_value(self.group[2].byte, self.options.alphabet).unwrap();
                    output.push((b << 4) | (c >> 2));
                }
            }
            _ => {
                let error = Error::Truncated {
                    offset: self.group[0].offset,
                };
                return Err(self.record_error(error));
            }
        }

        self.finished = true;
        Ok(output)
    }
}
