#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize, usize)>, // cleaned insertion index, original start, original end
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Cleaned {
        bytes: Vec::with_capacity(input.len()),
        positions: Vec::with_capacity(input.len()),
        runs: Vec::new(),
    };
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut run_start: Option<usize> = None;
    let mut run_clean_index = 0usize;

    while i < input.len() {
        let b = input[i];

        if depth == 0 && out.bytes.is_empty() && b == b'"' {
            if let Some(start) = run_start.take() {
                out.runs.push((run_clean_index, start, i));
            }
            let quote_start = i;
            out.bytes.push(b);
            out.positions.push(i);
            i += 1;
            while i < input.len() {
                let c = input[i];
                out.bytes.push(c);
                out.positions.push(i);
                i += 1;
                if c == b'\\' {
                    if i < input.len() {
                        out.bytes.push(input[i]);
                        out.positions.push(i);
                        i += 1;
                    }
                } else if c == b'"' {
                    break;
                }
            }
            let _ = quote_start;
            continue;
        }

        if depth == 0 {
            if is_hws(b) || b == b'\r' || b == b'\n' || b == b'(' {
                if run_start.is_none() {
                    run_start = Some(i);
                    run_clean_index = out.bytes.len();
                }
                if is_hws(b) {
                    i += 1;
                    continue;
                }
                if b == b'\r' || b == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                depth = 1;
                outer_open = i;
                i += 1;
                continue;
            }
            if b == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            if let Some(start) = run_start.take() {
                out.runs.push((run_clean_index, start, i));
            }
            out.bytes.push(b);
            out.positions.push(i);
            i += 1;
            continue;
        }

        // Inside a comment.
        if b == b'\\' {
            let Some(&next) = input.get(i + 1) else {
                return Err(EmailError::CommentEscape { at: i });
            };
            if !(0x20..=0x7e).contains(&next) {
                return Err(EmailError::CommentEscape { at: i });
            }
            i += 2;
            continue;
        }
        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }
        if b == b'(' {
            if depth == 3 {
                return Err(EmailError::CommentDepth { at: i });
            }
            depth += 1;
            i += 1;
            continue;
        }
        if b == b')' {
            depth -= 1;
            i += 1;
            if depth == 0 {
                continue;
            }
            continue;
        }
        if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
            return Err(EmailError::CommentChar { at: i });
        }
        i += 1;
    }

    if depth != 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        out.runs.push((run_clean_index, start, input.len()));
    }
    Ok(out)
}

fn position(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn is_atext(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes[0] == b'.' || bytes[bytes.len() - 1] == b'.' {
        return false;
    }
    let mut previous_dot = false;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else {
            if !is_atext(b) {
                return false;
            }
            previous_dot = false;
        }
    }
    true
}

fn parse_with_policy(input: &str, policy: Option<&Policy>) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let n = cleaned.bytes.len();

    for &(insert, start, end) in &cleaned.runs {
        let at_start = insert == 0;
        let at_end = insert == n;
        let adjacent_at = cleaned.bytes.get(insert) == Some(&b'@')
            || insert.checked_sub(1).and_then(|x| cleaned.bytes.get(x)) == Some(&b'@');
        if !(at_start || at_end || adjacent_at) {
            return Err(EmailError::CfwsBoundary { at: start.min(end) });
        }
        let _ = start;
    }

    if n == 0 {
        return Err(EmailError::EmptyInput);
    }

    let bytes = &cleaned.bytes;
    let mut protected_end = 0usize;
    if bytes[0] == b'"' {
        let mut i = 1;
        let mut close = None;
        while i < n {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        let Some(close) = close else {
            return Err(EmailError::UnclosedQuote {
                at: position(&cleaned, 0, raw.len()),
            });
        };
        protected_end = close + 1;
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| {
            if b == b'@' && !(i < protected_end) {
                Some(i)
            } else {
                None
            }
        })
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: position(&cleaned, separators[1], raw.len()),
        });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: position(&cleaned, sep, raw.len()),
        });
    }

    let local_raw = &bytes[..sep];
    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let quoted;

    if local_raw[0] == b'"' {
        let close = protected_end - 1;
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: position(&cleaned, close + 1, raw.len()),
            });
        }
        quoted = true;
        let mut i = 1;
        while i < close {
            let b = local_raw[i];
            if b == b'\\' {
                let Some(&escaped) = local_raw.get(i + 1).filter(|_| i + 1 < close) else {
                    return Err(EmailError::BadEscape {
                        at: position(&cleaned, i, raw.len()),
                    });
                };
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: position(&cleaned, i, raw.len()),
                    });
                }
                local.push(escaped);
                local_origins.push(position(&cleaned, i, raw.len()));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: position(&cleaned, i, raw.len()),
                    });
                }
                local.push(b);
                local_origins.push(position(&cleaned, i, raw.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: position(&cleaned, close, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        quoted = false;
        if local_raw.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: position(&cleaned, 64, raw.len()),
            });
        }
        let mut prev_dot = false;
        for (i, &b) in local_raw.iter().enumerate() {
            let at = position(&cleaned, i, raw.len());
            if b == b'.' {
                if i == 0 || i + 1 == local_raw.len() || prev_dot {
                    return Err(EmailError::LocalDot { at });
                }
                prev_dot = true;
            } else {
                if !is_atext(b) {
                    return Err(EmailError::LocalChar { at });
                }
                prev_dot = false;
            }
            local.push(b);
            local_origins.push(at);
        }
    }

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let mut domain_end = bytes.len();
    let root_pos = if raw_domain.last() == Some(&b'.') {
        domain_end -= 1;
        Some(position(&cleaned, domain_end, raw.len()))
    } else {
        None
    };
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: position(&cleaned, domain_start, raw.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: position(&cleaned, domain_start + 253, raw.len()),
        });
    }

    let domain = &bytes[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0usize;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            if label_start == i {
                let at = if i == domain.len() {
                    root_pos.unwrap_or_else(|| position(&cleaned, domain_start + i, raw.len()))
                } else {
                    position(&cleaned, domain_start + i, raw.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = i - label_start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: position(&cleaned, domain_start + label_start + 63, raw.len()),
                });
            }
            let first = domain[label_start];
            let last = domain[i - 1];
            if first == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: position(&cleaned, domain_start + label_start, raw.len()),
                });
            }
            if last == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: position(&cleaned, domain_start + i - 1, raw.len()),
                });
            }
            for j in label_start..i {
                let c = domain[j];
                if !(c.is_ascii_alphanumeric() || c == b'-') {
                    return Err(EmailError::DomainChar {
                        at: position(&cleaned, domain_start + j, raw.len()),
                    });
                }
            }
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    if labels.len() < 2 && !policy.is_some_and(|p| p.allow_single_label) {
        return Err(EmailError::DomainNeedsDot {
            at: position(&cleaned, domain_start, raw.len()),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if domain[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: position(&cleaned, domain_start + last_start, raw.len()),
        });
    }

    let mut domain_string = String::from_utf8(domain.to_vec()).unwrap();
    domain_string.make_ascii_lowercase();
    let mut address = Address {
        local: String::from_utf8(local.clone()).unwrap(),
        domain: domain_string,
        quoted,
    };

    if let Some(policy) = policy {
        if policy.local_case == LocalCase::Lower {
            for b in &mut local {
                b.make_ascii_lowercase();
            }
            address.local = String::from_utf8(local.clone()).unwrap();
        }

        if policy.strip_subaddress {
            if let Some(plus) = local.iter().position(|&b| b == b'+') {
                if plus == 0 {
                    return Err(EmailError::EmptyNormalizedLocal {
                        at: local_origins[0],
                    });
                }
                local.truncate(plus);
                local_origins.truncate(plus);
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyNormalizedLocal {
                at: local_origins.first().copied().unwrap_or(position(&cleaned, 0, raw.len())),
            });
        }
        address.local = String::from_utf8(local.clone()).unwrap();
        let safe = valid_dot_atom(&local);
        if !safe {
            address.quoted = true;
        } else if policy.unquote_safe {
            address.quoted = false;
        }
    }

    Ok((address, local_origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, None).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, Some(policy)).map(|(address, _)| address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut seen: std::collections::HashMap<(String, String), usize> =
        std::collections::HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    result.duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }

    result
}
