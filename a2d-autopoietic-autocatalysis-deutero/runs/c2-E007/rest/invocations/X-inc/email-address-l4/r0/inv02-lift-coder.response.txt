#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_impl(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, local_origins) = parse_impl(input, policy.allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.find('+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: local_origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }

    if !is_dot_atom(address.local.as_bytes()) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    addresses.push(address);
                    source_indices.push(index);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

fn is_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atext(byte) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn is_atext(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
}

fn parse_impl(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, origins, runs) = cleanup(raw)?;
    for (start, end, original_start) in runs {
        let at_left = start > 0 && clean[start - 1] == b'@';
        let at_right = end < clean.len() && clean[end] == b'@';
        if start != 0 && end != clean.len() && !at_left && !at_right {
            return Err(EmailError::CfwsBoundary { at: original_start });
        }
    }
    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: origins[0] });
        }
    }

    let mut separators = Vec::new();
    for (i, &byte) in clean.iter().enumerate() {
        if byte == b'@' && !quoted_end.map_or(false, |end| i <= end) {
            separators.push(i);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: origins[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: origins[sep] });
    }

    let local_bytes = &clean[..sep];
    let (local, quoted, local_origins) = if local_bytes[0] != b'"' {
        validate_atom(local_bytes, &origins[..sep])?;
        (
            String::from_utf8(local_bytes.to_vec()).unwrap(),
            false,
            origins[..sep].to_vec(),
        )
    } else {
        let end = quoted_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: origins[end + 1],
            });
        }
        let mut decoded = Vec::new();
        let mut decoded_origins = Vec::new();
        let mut i = 1;
        while i < end {
            let byte = clean[i];
            if byte == b'\\' {
                if i + 1 >= end
                    || !matches!(clean[i + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape { at: origins[i] });
                }
                decoded.push(clean[i + 1]);
                decoded_origins.push(origins[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&byte) {
                    return Err(EmailError::QuotedChar { at: origins[i] });
                }
                decoded.push(byte);
                decoded_origins.push(origins[i]);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: origins[end] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded_origins[64],
            });
        }
        (
            String::from_utf8(decoded).unwrap(),
            true,
            decoded_origins,
        )
    };

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut removed_root = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_root = Some(origins[domain_end]);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: origins.get(domain_start).copied().unwrap_or(input.len()),
        });
    }

    let domain_bytes = &clean[domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: origins[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for i in domain_start..domain_end {
        if clean[i] == b'.' {
            if i == label_start {
                return Err(EmailError::EmptyLabel { at: origins[i] });
            }
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }
    if label_start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: removed_root.unwrap_or_else(|| {
                origins.get(domain_end).copied().unwrap_or(input.len())
            }),
        });
    }
    labels.push((label_start, domain_end));

    for &(start, end) in &labels {
        let length = end - start;
        if length > 63 {
            return Err(EmailError::LabelTooLong {
                at: origins[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: origins[start] });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: origins[end - 1] });
        }
        for i in start..end {
            let b = clean[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: origins[i] });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: origins[domain_start],
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: origins[last_start],
        });
    }

    let mut domain = String::from_utf8(domain_bytes.to_vec()).unwrap();
    domain.make_ascii_lowercase();
    Ok((
        Address {
            local,
            domain,
            quoted,
        },
        local_origins,
    ))
}

fn validate_atom(bytes: &[u8], origins: &[usize]) -> Result<(), EmailError> {
    if bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: origins[64] });
    }
    for i in 0..bytes.len() {
        let byte = bytes[i];
        if byte == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: origins[i] });
            }
        } else if !is_atext(byte) {
            return Err(EmailError::LocalChar { at: origins[i] });
        }
    }
    Ok(())
}

fn cleanup(raw: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<(usize, usize, usize)>), EmailError> {
    let mut clean = Vec::new();
    let mut origins = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_protected = false;
    let mut quote_closed = false;
    let mut in_run = false;
    let mut run_start_clean = 0;
    let mut run_original = 0;

    while i < raw.len() {
        let b = raw[i];

        if quote_protected && !quote_closed {
            clean.push(b);
            origins.push(i);
            if b == b'\\' {
                i += 1;
                if i < raw.len() {
                    clean.push(raw[i]);
                    origins.push(i);
                    i += 1;
                }
                continue;
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if !quote_protected && clean.is_empty() && b == b'"' {
            quote_protected = true;
            clean.push(b);
            origins.push(i);
            i += 1;
            continue;
        }

        let mut consumed = 0usize;
        if b == b' ' || b == b'\t' {
            consumed = 1;
        } else if b == b'\r' || b == b'\n' {
            if b == b'\r'
                && i + 2 < raw.len()
                && raw[i + 1] == b'\n'
                && (raw[i + 2] == b' ' || raw[i + 2] == b'\t')
            {
                consumed = 3;
                while i + consumed < raw.len()
                    && (raw[i + consumed] == b' ' || raw[i + consumed] == b'\t')
                {
                    consumed += 1;
                }
            } else {
                return Err(EmailError::BadFold { at: i });
            }
        } else if b == b'(' {
            let outer = i;
            let mut depth = 0usize;
            let mut j = i;
            while j < raw.len() {
                let c = raw[j];
                if c == b'\\' {
                    if j + 1 >= raw.len()
                        || !(0x20..=0x7e).contains(&raw[j + 1])
                    {
                        return Err(EmailError::CommentEscape { at: j });
                    }
                    j += 2;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && j + 2 < raw.len()
                        && raw[j + 1] == b'\n'
                        && (raw[j + 2] == b' ' || raw[j + 2] == b'\t')
                    {
                        j += 3;
                        while j < raw.len() && (raw[j] == b' ' || raw[j] == b'\t') {
                            j += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: j });
                }
                if c == b'(' {
                    depth += 1;
                    if depth > 3 {
                        return Err(EmailError::CommentDepth { at: j });
                    }
                    j += 1;
                    continue;
                }
                if c == b')' {
                    depth -= 1;
                    j += 1;
                    if depth == 0 {
                        break;
                    }
                    continue;
                }
                if c == b'\t' || (0x20..=0x7e).contains(&c) {
                    j += 1;
                    continue;
                }
                return Err(EmailError::CommentChar { at: j });
            }
            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            consumed = j - i;
        } else if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if consumed > 0 {
            if !in_run {
                in_run = true;
                run_start_clean = clean.len();
                run_original = i;
            }
            i += consumed;
            continue;
        }

        if in_run {
            runs.push((run_start_clean, clean.len(), run_original));
            in_run = false;
        }
        clean.push(b);
        origins.push(i);
        i += 1;
    }

    if in_run {
        runs.push((run_start_clean, clean.len(), run_original));
    }
    Ok((clean, origins, runs))
}
