#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut bytes = std::mem::take(&mut self.pending);
        bytes.extend_from_slice(input);

        let complete_len = (bytes.len() / 5) * 5;
        let mut output = encode_symbols(&bytes[..complete_len], self.config.alphabet);
        self.pending.extend_from_slice(&bytes[complete_len..]);
        Ok(std::mem::take(&mut output))
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = encode_symbols(&self.pending, self.config.alphabet);
        if !self.pending.is_empty() && self.config.padding == Padding::Required {
            while output.len() % 8 != 0 {
                output.push('=');
            }
        }
        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_symbols(bytes: &[u8], alphabet: Alphabet) -> String {
    let table = alphabet_bytes(alphabet);
    let mut output = String::new();
    let mut accumulator: u32 = 0;
    let mut bits = 0u32;

    for &byte in bytes {
        accumulator = (accumulator << 8) | u32::from(byte);
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            output.push(table[((accumulator >> bits) & 31) as usize] as char);
        }
    }

    if bits > 0 {
        output.push(table[((accumulator << (5 - bits)) & 31) as usize] as char);
    }

    output
}

pub struct Decoder {
    config: Config,
    bit_buffer: u32,
    bit_count: u32,
    data_count: usize,
    last_data_offset: Option<usize>,
    last_symbol: u8,
    padding_count: usize,
    first_padding_offset: Option<usize>,
    expected_padding: usize,
    position: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            bit_buffer: 0,
            bit_count: 0,
            data_count: 0,
            last_data_offset: None,
            last_symbol: 0,
            padding_count: 0,
            first_padding_offset: None,
            expected_padding: 0,
            position: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut trial = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = trial.position;
            let symbol = recognize(byte, trial.config.alphabet);

            match symbol {
                Symbol::Invalid => {
                    return Err(StreamError::InvalidByte { offset, byte });
                }
                Symbol::Data(value, uppercase) => {
                    if trial.config.canonical && byte != uppercase {
                        return Err(StreamError::NonCanonical {
                            offset,
                            expected: uppercase,
                        });
                    }
                    if trial.padding_count > 0 {
                        return Err(StreamError::Padding { offset });
                    }
                    trial.data_count = trial.data_count.saturating_add(1);
                    trial.last_data_offset = Some(offset);
                    trial.last_symbol = value;
                    trial.bit_buffer = (trial.bit_buffer << 5) | u32::from(value);
                    trial.bit_count += 5;
                    while trial.bit_count >= 8 {
                        trial.bit_count -= 8;
                        output.push(((trial.bit_buffer >> trial.bit_count) & 0xff) as u8);
                    }
                    if trial.bit_count == 0 {
                        trial.bit_buffer = 0;
                    } else {
                        trial.bit_buffer &= (1u32 << trial.bit_count) - 1;
                    }
                }
                Symbol::Padding => {
                    if trial.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }
                    if trial.padding_count == 0 {
                        let residue = trial.data_count % 8;
                        trial.expected_padding = match residue {
                            2 => 6,
                            4 => 4,
                            5 => 3,
                            7 => 1,
                            _ => return Err(StreamError::Padding { offset }),
                        };
                        trial.first_padding_offset = Some(offset);
                    }
                    if trial.padding_count >= trial.expected_padding {
                        return Err(StreamError::Padding { offset });
                    }
                    trial.padding_count += 1;
                }
            }

            trial.position = trial.position.saturating_add(1);
        }

        self.copy_from(trial);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits > 0 && (self.last_symbol & ((1u8 << unused_bits) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            data_count: self.data_count,
            last_data_offset: self.last_data_offset,
            last_symbol: self.last_symbol,
            padding_count: self.padding_count,
            first_padding_offset: self.first_padding_offset,
            expected_padding: self.expected_padding,
            position: self.position,
            closed: self.closed,
        }
    }

    fn copy_from(&mut self, other: Self) {
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.data_count = other.data_count;
        self.last_data_offset = other.last_data_offset;
        self.last_symbol = other.last_symbol;
        self.padding_count = other.padding_count;
        self.first_padding_offset = other.first_padding_offset;
        self.expected_padding = other.expected_padding;
        self.position = other.position;
        self.closed = other.closed;
    }
}

enum Symbol {
    Data(u8, u8),
    Padding,
    Invalid,
}

fn recognize(byte: u8, alphabet: Alphabet) -> Symbol {
    if byte == b'=' {
        return Symbol::Padding;
    }

    let uppercase = byte.to_ascii_uppercase();
    let table = alphabet_bytes(alphabet);
    match table.iter().position(|&item| item == uppercase) {
        Some(value) => Symbol::Data(value as u8, uppercase),
        None => Symbol::Invalid,
    }
}
