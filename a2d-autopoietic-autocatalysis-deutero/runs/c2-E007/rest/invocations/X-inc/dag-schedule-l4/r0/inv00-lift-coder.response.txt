#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependencies in the specified order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest available task ID chosen each time.
    let mut available = Vec::<usize>::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(task_id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_position = available
            .iter()
            .enumerate()
            .min_by_key(|(_, task_id)| **task_id)
            .map(|(position, _)| position)
            .unwrap();
        let task_id = available.swap_remove(smallest_position);
        order.push(task_id);

        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut removed = vec![false; n];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u64; capacity.len()];
    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));
    let mut current_time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == current_time)
            .collect();
        finishing.sort_by_key(|entry| entry.task);

        if !finishing.is_empty() {
            running.retain(|entry| entry.finish != current_time);
            for entry in finishing {
                let task_id = entry.task;
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(current_time);

                for resource in 0..capacity.len() {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }

                append_event(
                    &mut trace,
                    current_time,
                    "end",
                    task_id,
                    entry.worker,
                    &used,
                );
            }
        }

        // Repeatedly admit one task, reevaluating after every start/completion.
        let mut admitted = false;
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > current_time {
                    continue;
                }
                if !tasks[task_id]
                    .deps
                    .iter()
                    .all(|&predecessor| completed[predecessor])
                {
                    continue;
                }
                if (0..capacity.len()).any(|resource| {
                    u64::from(tasks[task_id].needs[resource])
                        > u64::from(capacity[resource]) - used[resource]
                }) {
                    continue;
                }
                if best.map_or(true, |current| {
                    tail[task_id] > tail[current]
                        || (tail[task_id] == tail[current]
                            && (tasks[task_id].priority > tasks[current].priority
                                || (tasks[task_id].priority == tasks[current].priority
                                    && task_id < current)))
                }) {
                    best = Some(task_id);
                }
            }

            let task_id = match best {
                Some(task_id) => task_id,
                None => break,
            };

            let mut worker = None;
            for offset in 0..worker_count {
                let candidate = (task_id % worker_count + offset) % worker_count;
                if worker_available[candidate] <= current_time {
                    worker = Some(candidate);
                    break;
                }
            }
            let worker = match worker {
                Some(worker) => worker,
                None => break,
            };

            let finish = current_time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let turnaround = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = turnaround;
            runs[task_id] = Some(Run {
                start: current_time,
                finish,
                worker,
            });
            for resource in 0..capacity.len() {
                used[resource] += u64::from(tasks[task_id].needs[resource]);
            }
            append_event(&mut trace, current_time, "start", task_id, worker, &used);
            running.push(Running {
                task: task_id,
                worker,
                finish,
            });
            admitted = true;

            if finish == current_time {
                running.retain(|entry| !(entry.task == task_id && entry.finish == current_time));
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(current_time);
                for resource in 0..capacity.len() {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }
                append_event(&mut trace, current_time, "end", task_id, worker, &used);
            }
        }

        if completed_count == n {
            break;
        }

        // If tasks can still start now, the admission loop above has exhausted
        // all worker or task opportunities. Advance to the next relevant event.
        let mut next_time: Option<u64> = None;
        for entry in &running {
            if entry.finish > current_time {
                next_time = Some(next_time.map_or(entry.finish, |value| value.min(entry.finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > current_time {
                next_time = Some(
                    next_time.map_or(tasks[task_id].release, |value| {
                        value.min(tasks[task_id].release)
                    }),
                );
            }
        }
        for &available_at in &worker_available {
            if available_at > current_time {
                next_time = Some(next_time.map_or(available_at, |value| value.min(available_at)));
            }
        }

        current_time = match next_time {
            Some(time) => time,
            None => {
                // With an acyclic graph, unfinished tasks always have a future
                // release, completion, or cooldown event.
                let _ = admitted;
                return Err(Error::Cycle {
                    blocked: (0..n).filter(|&task_id| !completed[task_id]).collect(),
                });
            }
        };
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn append_event(
    trace: &mut String,
    time: u64,
    event: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    trace.push_str(&format!("{}|{}|{}|{}|", time, event, task, worker));
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, count) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            trace.push_str(&count.to_string());
        }
    }
    trace.push('\n');
}
