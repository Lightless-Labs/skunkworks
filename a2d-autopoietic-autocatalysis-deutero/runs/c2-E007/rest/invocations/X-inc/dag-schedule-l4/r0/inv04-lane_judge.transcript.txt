pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in the specified order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shapes and demands task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and perform deterministic Kahn traversal.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(task_id);
        }
    }

    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);
        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut seen = vec![false; n];
            for &task_id in &order {
                seen[task_id] = true;
            }
            seen
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut used = vec![0u64; capacity.len()];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();
    let mut worker_available_at = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // finish, task, worker
    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        n
    ];
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete every task finishing now in ascending task ID.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        finishing.sort_by_key(|&(_, task_id, _)| task_id);

        for (finish, task_id, worker) in finishing {
            if let Some(index) = running
                .iter()
                .position(|&(f, t, w)| f == finish && t == task_id && w == worker)
            {
                running.swap_remove(index);
            }
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            append_trace_event(&mut trace, now, "end", task_id, worker, &used);
            for &child in &successors[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= now
                        && remaining_predecessors[task_id] == 0
                        && tasks[task_id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(r, &need)| {
                                u64::from(need) <= u64::from(capacity[r]) - used[r]
                            })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available_at[worker] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available_at[worker] = available_at;
            runs[task_id] = Run {
                start: now,
                finish,
                worker,
            };
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            append_trace_event(&mut trace, now, "start", task_id, worker, &used);

            if finish == now {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                append_trace_event(&mut trace, now, "end", task_id, worker, &used);
                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now && next_time.map_or(true, |current| candidate < current) {
                next_time = Some(candidate);
            }
        };

            consider(finish);
        }
        for (task_id, task) in tasks.iter().enumerate() {
            if !started[task_id] {
                consider(task.release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        let Some(next) = next_time else {
            // With an acyclic graph and validated resources, progress must be
            // possible whenever unfinished tasks remain.
            break;
        };
        now = next;
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn append_trace_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task_id: usize,
    worker: usize,
    used: &[u64],
) {
    use std::fmt::Write;

    let _ = write!(trace, "{time}|{kind}|{task_id}|{worker}|");
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, count) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            let _ = write!(trace, "{count}");
        }
    }
    trace.push('\n');
}

            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest available task ID.
    let mut available = BTreeSet::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(task_id) = available.pop_first() {
        order.push(task_id);
        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &task_id in &order {
                flags[task_id] = true;
            }
            flags
        };
        let blocked = (0..n).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));

    loop {
        // Complete every task finishing now in ascending task ID order.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|&(finish, _, _)| finish == time)
            .collect();
        finishing.sort_by_key(|&(_, task_id, _)| task_id);

        if !finishing.is_empty() {
            running.retain(|&(finish, _, _)| finish != time);
            for &(finish, task_id, worker) in &finishing {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    time,
                    task_id,
                    worker,
                    format_used(&used)
                ));
            }
        }

        // Repeatedly admit one task, reevaluating eligibility and ranking.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > time {
                    continue;
                }
                if !tasks[task_id]
                    .deps
                    .iter()
                    .all(|&predecessor| completed[predecessor])
                {
                    continue;
                }

                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(resource, &need)| {
                        u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                    });
                if !fits {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tail[task_id] > tail[current]
                            || (tail[task_id] == tail[current]
                                && tasks[task_id].priority > tasks[current].priority)
                            || (tail[task_id] == tail[current]
                                && tasks[task_id].priority == tasks[current].priority
                                && task_id < current)
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= time);

            let Some(worker) = worker else {
                break;
            };

            let finish = time.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                time,
                task_id,
                worker,
                format_used(&used)
            ));

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    time,
                    task_id,
                    worker,
                    format_used(&used)
                ));
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change the
        // availability of a task or worker.
        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > time {
                consider(tasks[task_id].release);
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > time {
                consider(worker_available[worker]);
            }
        }

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    trace.push_str(&format!("makespan={}\n", makespan));

    let runs = runs
        .into_iter()
        .map(|run| run.expect("all tasks must have been started"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn format_used(used: &[u64]) -> String {
    if used.is_empty() {
        return "-".to_string();
    }

    let mut result = String::new();
    for (index, value) in used.iter().enumerate() {
        if index > 0 {
            result.push(',');
        }
        result.push_str(&value.to_string());
    }
    result
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependencies in the required task and entry order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Validate resource shapes and demands task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest available task ID.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        topo.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if topo.len() != n {
        let mut blocked = Vec::new();
        let removed = {
            let mut is_removed = vec![false; n];
            for &task_id in &topo {
                is_removed[task_id] = true;
            }
            is_removed
        };
        for (task_id, &was_removed) in removed.iter().enumerate() {
            if !was_removed {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in topo.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut used_resources = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];

    // Running entries are (finish time, task ID, worker ID).
    let mut running: Vec<(u64, usize, usize)> = Vec::new();
    let mut trace = String::new();
    use std::fmt::Write as _;
    let _ = writeln!(
        trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    );

    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks finishing now in ascending task ID order.
        let mut ending: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        ending.sort_by_key(|&(_, task_id, _)| task_id);

        for (_, task_id, worker_id) in ending {
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used_resources[resource] -= u64::from(need);
            }
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(now);

            let _ = write!(trace, "{}|end|{}|{}|", now, task_id, worker_id);
            write_resources(&mut trace, &used_resources);
            trace.push('\n');

            for &child in &children[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }
        running.retain(|(finish, _, _)| *finish != now);

        // Start one task at a time and reevaluate eligibility after each start.
        loop {
            let mut best: Option<usize> = None;

            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > now
                    || remaining_predecessors[task_id] != 0
                {
                    continue;
                }

                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(resource, &need)| {
                        used_resources[resource] + u64::from(need)
                            <= u64::from(capacity[resource])
                    });
                if !fits {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[task_id] > tails[current]
                            || (tails[task_id] == tails[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            // Select a worker only after selecting the task.
            let mut selected_worker = None;
            let first_worker = task_id % worker_count;
            for offset in 0..worker_count {
                let worker_id = (first_worker + offset) % worker_count;
                if worker_available[worker_id] <= now {
                    selected_worker = Some(worker_id);
                    break;
                }
            }
            let Some(worker_id) = selected_worker else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let turnaround = finish
                .checked_add(cooldowns[worker_id])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker_id] = turnaround;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used_resources[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });

            let _ = write!(trace, "{}|start|{}|{}|", now, task_id, worker_id);
            write_resources(&mut trace, &used_resources);
            trace.push('\n');

            if finish == now {
                // A zero-duration task completes immediately before another admission.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used_resources[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);

                let _ = write!(trace, "{}|end|{}|{}|", now, task_id, worker_id);
                write_resources(&mut trace, &used_resources);
                trace.push('\n');

                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change the state.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |current| current.min(time)));
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                consider(tasks[task_id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        now = next_time.expect("unfinished tasks must have a future event");
    }

    let mut final_trace = trace;
    let _ = writeln!(final_trace, "makespan={}", makespan);

    Ok(Schedule {
        runs: runs
            .into_iter()
            .map(|run| run.expect("every task must start exactly once"))
            .collect(),
        makespan,
        trace: final_trace,
    })
}

fn write_resources(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
        return;
    }

    for (index, value) in used.iter().enumerate() {
        if index > 0 {
            trace.push(',');
        }
        use std::fmt::Write as _;
        let _ = write!(trace, "{}", value);
    }
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependencies in the specified order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest available task ID chosen each time.
    let mut available = Vec::<usize>::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(task_id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_position = available
            .iter()
            .enumerate()
            .min_by_key(|(_, task_id)| **task_id)
            .map(|(position, _)| position)
            .unwrap();
        let task_id = available.swap_remove(smallest_position);
        order.push(task_id);

        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut removed = vec![false; n];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u64; capacity.len()];
    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));
    let mut current_time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == current_time)
            .collect();
        finishing.sort_by_key(|entry| entry.task);

        if !finishing.is_empty() {
            running.retain(|entry| entry.finish != current_time);
            for entry in finishing {
                let task_id = entry.task;
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(current_time);

                for resource in 0..capacity.len() {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }

                append_event(
                    &mut trace,
                    current_time,
                    "end",
                    task_id,
                    entry.worker,
                    &used,
                );
            }
        }

        // Repeatedly admit one task, reevaluating after every start/completion.
        let mut admitted = false;
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > current_time {
                    continue;
                }
                if !tasks[task_id]
                    .deps
                    .iter()
                    .all(|&predecessor| completed[predecessor])
                {
                    continue;
                }
                if (0..capacity.len()).any(|resource| {
                    u64::from(tasks[task_id].needs[resource])
                        > u64::from(capacity[resource]) - used[resource]
                }) {
                    continue;
                }
                if best.map_or(true, |current| {
                    tail[task_id] > tail[current]
                        || (tail[task_id] == tail[current]
                            && (tasks[task_id].priority > tasks[current].priority
                                || (tasks[task_id].priority == tasks[current].priority
                                    && task_id < current)))
                }) {
                    best = Some(task_id);
                }
            }

            let task_id = match best {
                Some(task_id) => task_id,
                None => break,
            };

            let mut worker = None;
            for offset in 0..worker_count {
                let candidate = (task_id % worker_count + offset) % worker_count;
                if worker_available[candidate] <= current_time {
                    worker = Some(candidate);
                    break;
                }
            }
            let worker = match worker {
                Some(worker) => worker,
                None => break,
            };

            let finish = current_time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let turnaround = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = turnaround;
            runs[task_id] = Some(Run {
                start: current_time,
                finish,
                worker,
            });
            for resource in 0..capacity.len() {
                used[resource] += u64::from(tasks[task_id].needs[resource]);
            }
            append_event(&mut trace, current_time, "start", task_id, worker, &used);
            running.push(Running {
                task: task_id,
                worker,
                finish,
            });
            admitted = true;

            if finish == current_time {
                running.retain(|entry| !(entry.task == task_id && entry.finish == current_time));
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(current_time);
                for resource in 0..capacity.len() {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }
                append_event(&mut trace, current_time, "end", task_id, worker, &used);
            }
        }

        if completed_count == n {
            break;
        }

        // If tasks can still start now, the admission loop above has exhausted
        // all worker or task opportunities. Advance to the next relevant event.
        let mut next_time: Option<u64> = None;
        for entry in &running {
            if entry.finish > current_time {
                next_time = Some(next_time.map_or(entry.finish, |value| value.min(entry.finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > current_time {
                next_time = Some(
                    next_time.map_or(tasks[task_id].release, |value| {
                        value.min(tasks[task_id].release)
                    }),
                );
            }
        }
        for &available_at in &worker_available {
            if available_at > current_time {
                next_time = Some(next_time.map_or(available_at, |value| value.min(available_at)));
            }
        }

        current_time = match next_time {
            Some(time) => time,
            None => {
                // With an acyclic graph, unfinished tasks always have a future
                // release, completion, or cooldown event.
                let _ = admitted;
                return Err(Error::Cycle {
                    blocked: (0..n).filter(|&task_id| !completed[task_id]).collect(),
                });
            }
        };
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn append_event(
    trace: &mut String,
    time: u64,
    event: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    trace.push_str(&format!("{}|{}|{}|{}|", time, event, task, worker));
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, count) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            trace.push_str(&count.to_string());
        }
    }
    trace.push('\n');
}
