#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in the specified order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shapes and demands task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and perform deterministic Kahn traversal.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(task_id);
        }
    }

    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);
        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut seen = vec![false; n];
            for &task_id in &order {
                seen[task_id] = true;
            }
            seen
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut used = vec![0u64; capacity.len()];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();
    let mut worker_available_at = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // finish, task, worker
    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        n
    ];
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete every task finishing now in ascending task ID.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        finishing.sort_by_key(|&(_, task_id, _)| task_id);

        for (finish, task_id, worker) in finishing {
            if let Some(index) = running
                .iter()
                .position(|&(f, t, w)| f == finish && t == task_id && w == worker)
            {
                running.swap_remove(index);
            }
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            append_trace_event(&mut trace, now, "end", task_id, worker, &used);
            for &child in &successors[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= now
                        && remaining_predecessors[task_id] == 0
                        && tasks[task_id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(r, &need)| {
                                u64::from(need) <= u64::from(capacity[r]) - used[r]
                            })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available_at[worker] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available_at[worker] = available_at;
            runs[task_id] = Run {
                start: now,
                finish,
                worker,
            };
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            append_trace_event(&mut trace, now, "start", task_id, worker, &used);

            if finish == now {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                append_trace_event(&mut trace, now, "end", task_id, worker, &used);
                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now && next_time.map_or(true, |current| candidate < current) {
                next_time = Some(candidate);
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for (task_id, task) in tasks.iter().enumerate() {
            if !started[task_id] {
                consider(task.release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        let Some(next) = next_time else {
            // With an acyclic graph and validated resources, progress must be
            // possible whenever unfinished tasks remain.
            break;
        };
        now = next;
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn append_trace_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task_id: usize,
    worker: usize,
    used: &[u64],
) {
    use std::fmt::Write;

    let _ = write!(trace, "{time}|{kind}|{task_id}|{worker}|");
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, count) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            let _ = write!(trace, "{count}");
        }
    }
    trace.push('\n');
}
