```rust
use std::array;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<u8>; 5],
    day_of_month_star: bool,
    day_of_week_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if is_step {
        let (_, max) = domain(field);
        if value == 0 || value > max - domain(field).0 + 1 {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }

    Ok(value)
}

fn parse_item(
    item: &[u8],
    field: Field,
    offset: usize,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (
            &item[..slash],
            Some(&item[slash + 1..]),
            offset + slash + 1,
        )
    } else {
        (item, None, 0)
    };

    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let is_star = base == b"*";
    let range = hyphens.len() == 1;
    let single_number = !is_star && hyphens.is_empty();

    let structurally_valid = if is_star {
        true
    } else if range {
        hyphens[0] > 0
            && hyphens[0] + 1 < base.len()
            && base[..hyphens[0]].iter().all(u8::is_ascii_digit)
            && base[hyphens[0] + 1..].iter().all(u8::is_ascii_digit)
    } else if single_number {
        base.iter().all(u8::is_ascii_digit)
    } else {
        false
    };

    if !structurally_valid || base.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    if let Some(step) = step_bytes {
        if step.is_empty() || !step.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
    }

    let (min, max) = domain(field);
    let (start, end) = if is_star {
        (min, max)
    } else if range {
        let hyphen = hyphens[0];
        let start = parse_number(&base[..hyphen], field, offset, false)?;
        let end = parse_number(&base[hyphen + 1..], field, offset + hyphen + 1, false)?;
        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: offset + hyphen + 1,
            });
        }
        (start, end)
    } else {
        let start = parse_number(base, field, offset, false)?;
        (start, if step_bytes.is_some() { max } else { start })
    };

    let step = if let Some(bytes) = step_bytes {
        parse_number(bytes, field, step_offset, true)?
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = [false; 60];
    let mut start = 0;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(&bytes[start..i], field, start, &mut selected)?;
            start = i + 1;
        }
    }
    parse_item(&bytes[start..], field, start, &mut selected)?;

    let (min, max) = domain(field);
    Ok((min..=max)
        .filter(|&value| selected[value as usize])
        .map(|value| value as u8)
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut allowed: [Vec<u8>; 5] = array::from_fn(|_| Vec::new());
    for i in 0..5 {
        allowed[i] = parse_field(pieces[i], FIELDS[i])?;
    }

    Ok(Cron {
        allowed,
        day_of_month_star: pieces[2].as_bytes() == b"*",
        day_of_week_star: pieces[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at);
        Ok(self.allowed[0].contains(&at.minute)
            && self.allowed[1].contains(&at.hour)
            && self.allowed[2].contains(&at.day)
            && self.allowed[3].contains(&at.month)
            && self.allowed[4].contains(&weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            current = increment_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if is_leap(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += at.day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
```
