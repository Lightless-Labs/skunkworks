#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn bad(field: Field, offset: usize, kind: fn(Field, usize) -> ParseError) -> ParseError {
    kind(field, offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax { field, offset: start });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    let (min, max) = domain(field);
    if step {
        let width = max - min + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else if value < min || value > max {
        return Err(ParseError::OutOfRange { field, offset: start });
    }

    Ok(value)
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_item(
    bytes: &[u8],
    base_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: base_offset,
        });
    }

    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, base_offset));
    }

    let (base_end, step) = if let Some(&slash) = slash_positions.first() {
        (slash, Some(slash + 1))
    } else {
        (bytes.len(), None)
    };
    let base = &bytes[..base_end];

    let (kind, start_num, end_num) = if base == b"*" {
        (0u8, None, None)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1 {
            return Err(invalid_syntax(field, base_offset));
        }

        if let Some(&hyphen) = hyphens.first() {
            if hyphen == 0 || hyphen + 1 == base.len() {
                return Err(invalid_syntax(field, base_offset));
            }
            (
                2u8,
                Some((0usize, hyphen)),
                Some((hyphen + 1, base.len())),
            )
        } else {
            if base.is_empty() || !base.iter().all(u8::is_ascii_digit) {
                return Err(invalid_syntax(field, base_offset));
            }
            (1u8, Some((0usize, base.len())), None)
        }
    };

    if kind == 0 && base != b"*" {
        return Err(invalid_syntax(field, base_offset));
    }
    if base.is_empty() {
        return Err(invalid_syntax(field, base_offset));
    }

    let step_value = if let Some(step_start) = step {
        if step_start == bytes.len() {
            return Err(invalid_syntax(field, base_offset));
        }
        if !bytes[step_start..].iter().all(u8::is_ascii_digit) {
            return Err(invalid_syntax(field, base_offset));
        }
        Some(parse_number(
            bytes,
            step_start,
            bytes.len(),
            field,
            true,
        )?)
    } else {
        None
    };

    let (min, max) = domain(field);
    let (start, end) = match kind {
        0 => (min, max),
        1 => {
            let (s, e) = start_num.unwrap();
            let value = parse_number(bytes, s, e, field, false)?;
            (value, if step_value.is_some() { max } else { value })
        }
        2 => {
            let (s, e) = start_num.unwrap();
            let start_value = parse_number(bytes, s, e, field, false)?;
            let (es, ee) = end_num.unwrap();
            let end_value = parse_number(bytes, es, ee, field, false)?;
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: base_offset + es,
                });
            }
            (start_value, end_value)
        }
        _ => unreachable!(),
    };

    let stride = step_value.unwrap_or(1);
    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(stride) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut offset = 0usize;
    for item in text.split(',') {
        parse_item(item.as_bytes(), offset, field, &mut selected)?;
        offset += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = Vec::with_capacity(5);
    for (text, field) in parts.iter().zip(FIELDS) {
        selected.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = domain(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][(at.day - 1) as usize]
            && self.selected[3][(at.month - 1) as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            current = match advance_minute(current) {
                Some(next) => next,
                None => return Ok(None),
            };
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn advance_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        return Some(CivilDateTime {
            minute: at.minute + 1,
            ..at
        });
    }
    if at.hour < 23 {
        return Some(CivilDateTime {
            hour: at.hour + 1,
            minute: 0,
            ..at
        });
    }
    if at.day < days_in_month(at.year, at.month) {
        return Some(CivilDateTime {
            day: at.day + 1,
            hour: 0,
            minute: 0,
            ..at
        });
    }
    if at.month < 12 {
        return Some(CivilDateTime {
            month: at.month + 1,
            day: 1,
            hour: 0,
            minute: 0,
            ..at
        });
    }
    if at.year < 2399 {
        return Some(CivilDateTime {
            year: at.year + 1,
            month: 1,
            day: 1,
            hour: 0,
            minute: 0,
            ..at
        });
    }
    None
}
