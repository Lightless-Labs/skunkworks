```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    job: Job,
    dependencies: Vec<usize>,
}

pub struct Runner {
    jobs: Vec<JobData>,
    states: Vec<State>,
    attempts: Vec<u8>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    has_cycle: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job.id.bytes().all(|b| {
                    b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
                })
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut data = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.into_iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dep) {
                    Some(&index) => dependencies.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            data.push(JobData { job, dependencies });
        }

        fn has_cycle(
            node: usize,
            jobs: &[JobData],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return true;
            }
            if marks[node] == 2 {
                return false;
            }
            marks[node] = 1;
            for &dep in &jobs[node].dependencies {
                if has_cycle(dep, jobs, marks) {
                    return true;
                }
            }
            marks[node] = 2;
            false
        }

        let mut marks = vec![0; data.len()];
        for i in 0..data.len() {
            if has_cycle(i, &data, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let n = data.len();
        Ok(Self {
            jobs: data,
            states: vec![State::Waiting { ready_at: 0 }; n],
            attempts: vec![0; n],
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.has_cycle
            && self.states.iter().any(|state| {
                matches!(state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        self.states = vec![State::Waiting { ready_at: now }; self.jobs.len()];
        self.attempts.fill(0);
        self.last_time = now;
        self.has_cycle = true;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for i in 0..self.jobs.len() {
            let ready_at = match self.states[i] {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !self.jobs[i]
                .dependencies
                .iter()
                .all(|&dep| matches!(self.states[dep], State::Succeeded))
            {
                continue;
            }

            let base = self.jobs[i].job.priority;
            let effective = if self.aging_ms == 0 {
                base
            } else {
                let increase = (now - ready_at) / self.aging_ms;
                ((base as u64 + increase).min(9)) as u8
            };

            if best.map_or(true, |(_, priority)| effective > priority) {
                best = Some((i, effective));
            }
        }

        let result = if let Some((i, _)) = best {
            self.attempts[i] += 1;
            let attempt = self.attempts[i];
            self.states[i] = State::Running { attempt };
            Some(Ticket {
                id: self.jobs[i].job.id.clone(),
                attempt,
            })
        } else {
            None
        };
        self.last_time = now;
        Ok(result)
    }

    pub fn complete(
        &mut self,
        now: u64,
        id: &str,
        success: bool,
    ) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let i = self
            .jobs
            .iter()
            .position(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;

        if !matches!(self.states[i], State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.states[i] = State::Succeeded;
        } else if self.attempts[i] == self.jobs[i].job.max_attempts {
            self.states[i] = State::Failed;
            self.propagate_blocked();
        } else {
            let exponent = self.attempts[i] - 1;
            let mut delay = self.retry_base;
            for _ in 0..exponent {
                delay = delay.saturating_mul(2).min(self.retry_cap);
            }
            delay = delay.min(self.retry_cap);
            self.states[i] = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.last_time = now;
        Ok(())
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if matches!(self.states[i], State::Waiting { .. })
                    && self.jobs[i].dependencies.iter().any(|&dep| {
                        matches!(self.states[dep], State::Failed | State::Blocked)
                    })
                {
                    self.states[i] = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let i = self
            .jobs
            .iter()
            .position(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;
        Ok(self.states[i].clone())
    }
}
```
