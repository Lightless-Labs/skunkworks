#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut running: Option<(usize, u64)> = None;
    let mut previous_finish: Option<u64> = None;
    let mut started_once = false;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        let next_arrival_time = arrivals
            .get(next_arrival)
            .map(|&index| customers[index].arrival);
        let next_finish_time = running.map(|(_, finish)| finish);

        let tick = match (next_arrival_time, next_finish_time) {
            (Some(arrival), Some(finish)) => arrival.min(finish),
            (Some(arrival), None) => arrival,
            (None, Some(finish)) => finish,
            (None, None) => break,
        };
        now = tick;

        if let Some((index, finish)) = running {
            if finish == tick {
                let customer = &customers[index];
                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                running = None;
                previous_finish = Some(tick);
                finished_count += 1;
            }
        }

        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == tick
        {
            let index = arrivals[next_arrival];
            let customer = &customers[index];
            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(index);
            next_arrival += 1;
        }

        while running.is_none() {
            let Some(index) = waiting.pop_front() else {
                break;
            };
            let customer = &customers[index];
            let wake = if !started_once {
                true
            } else {
                previous_finish.is_some_and(|finish| tick > finish)
            };

            let service_finish = tick.checked_add(customer.service).ok_or(
                Error::TimeOverflow {
                    customer: customer.id,
                    time: tick,
                },
            )?;
            let finish = service_finish
                .checked_add(if wake { wake_cost } else { 0 })
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time: tick,
                })?;

            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Start { finish, wake },
            });
            started_once = true;

            if finish == tick {
                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                previous_finish = Some(tick);
                finished_count += 1;
            } else {
                running = Some((index, finish));
            }
        }
    }

    Ok(events)
}
