#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(std::mem::take(&mut word));
                    active = false;
                }
            }
            '\'' => {
                active = true;
                let opening = pos;
                let mut closed = false;
                while let Some((_, quoted)) = chars.next() {
                    if quoted == '\'' {
                        closed = true;
                        break;
                    }
                    word.push(quoted);
                }
                if !closed {
                    return Err(SplitError::UnterminatedSingle { pos: opening });
                }
            }
            '"' => {
                active = true;
                let opening = pos;
                let mut closed = false;

                while let Some((escape_pos, quoted)) = chars.next() {
                    match quoted {
                        '"' => {
                            closed = true;
                            break;
                        }
                        '\\' => {
                            let Some((_, next)) = chars.next() else {
                                return Err(SplitError::TrailingEscape { pos: escape_pos });
                            };
                            match next {
                                '"' | '\\' | '$' => word.push(next),
                                '\n' => {}
                                other => {
                                    word.push('\\');
                                    word.push(other);
                                }
                            }
                        }
                        other => word.push(other),
                    }
                }

                if !closed {
                    return Err(SplitError::UnterminatedDouble { pos: opening });
                }
            }
            '\\' => {
                if let Some(&(_, next)) = chars.peek() {
                    if next == '\n' {
                        chars.next();
                    } else {
                        let (_, escaped) = chars.next().unwrap();
                        word.push(escaped);
                        active = true;
                    }
                } else {
                    return Err(SplitError::TrailingEscape { pos });
                }
            }
            other => {
                word.push(other);
                active = true;
            }
        }
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
