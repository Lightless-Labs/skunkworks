#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: std::collections::BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }
        Ok(Self::canonical(&parts))
    }

    fn parent(path: &str) -> String {
        let parts = Self::components(path);
        if parts.len() <= 1 {
            "/".to_string()
        } else {
            Self::canonical(&parts[..parts.len() - 1])
        }
    }

    fn basename(path: &str) -> String {
        Self::components(path).last().cloned().unwrap_or_default()
    }

    fn join(base: &str, suffix: &[String]) -> String {
        let mut parts = Self::components(base);
        parts.extend_from_slice(suffix);
        Self::canonical(&parts)
    }

    fn has_prefix(path: &str, prefix: &str) -> bool {
        if prefix == "/" {
            return path.starts_with('/');
        }
        path == prefix
            || path
                .strip_prefix(prefix)
                .map_or(false, |s| s.starts_with('/'))
    }

    fn direct_children<'a>(
        &'a self,
        path: &'a str,
    ) -> impl Iterator<Item = (&'a String, &'a Entry)> + 'a {
        self.entries.iter().filter(move |(p, _)| {
            *p != path && Self::parent(p) == path
        })
    }

    fn resolve(&self, input: &str) -> Result<String, Error> {
        let mut path = input.to_string();
        let mut visited = std::collections::HashSet::<String>::new();
        let mut expansions = 0usize;

        loop {
            let parts = Self::components(&path);
            let mut current = Vec::<String>::new();
            let mut restarted = false;

            for (i, component) in parts.iter().enumerate() {
                current.push(component.clone());
                let prefix = Self::canonical(&current);
                let entry = match self.entries.get(&prefix) {
                    Some(e) => e,
                    None => return Err(Error::Missing(prefix)),
                };

                if i + 1 < parts.len() && entry.kind() != Kind::Dir {
                    return Err(Error::NotDir(prefix));
                }

                if let Entry::Link(target) = entry {
                    if !visited.insert(prefix.clone()) {
                        return Err(Error::Loop(prefix));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(prefix));
                    }
                    expansions += 1;

                    let base = if target.starts_with('/') {
                        "/".to_string()
                    } else {
                        Self::parent(&prefix)
                    };
                    let mut expanded =
                        Self::normalize(target, &base).map_err(|_| Error::InvalidPath(0))?;
                    let suffix = &parts[i + 1..];
                    expanded = Self::join(&expanded, suffix);
                    path = expanded;
                    restarted = true;
                    break;
                }
            }

            if restarted {
                continue;
            }
            return Ok(path);
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let name = Self::basename(path);
        let resolved_parent = self.resolve(&parent)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        let full = if resolved_parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", resolved_parent, name)
        };
        Ok((resolved_parent, full))
    }

    fn check_directory_parent(&self, parent: &str) -> Result<(), Error> {
        match self.entries.get(parent) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(parent.to_string())),
            None => Err(Error::Missing(parent.to_string())),
        }
    }

    fn subtree_size(&self, path: &str) -> u64 {
        match self.entries.get(path) {
            Some(Entry::File(bytes)) => bytes.len() as u64,
            Some(Entry::Link(target)) => target.len() as u64,
            Some(Entry::Dir) => self
                .direct_children(path)
                .map(|(child, _)| self.subtree_size(child))
                .sum(),
            None => 0,
        }
    }

    fn contains_cwd(&self, path: &str) -> bool {
        Self::has_prefix(&self.cwd, path)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = Self::normalize(path, &self.cwd)?;
        let (parent, final_path) = self.resolve_parent(&normalized)?;
        self.check_directory_parent(&parent)?;
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = Self::normalize(path, &self.cwd)?;
        let resolved = match self.resolve(&normalized) {
            Ok(p) => p,
            Err(Error::Missing(missing)) if missing == normalized => {
                let (parent, final_path) = self.resolve_parent(&normalized)?;
                self.check_directory_parent(&parent)?;
                self.entries
                    .insert(final_path, Entry::File(data.to_vec()));
                return Ok(());
            }
            Err(e) => return Err(e),
        };

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, final_path) = self.resolve_parent(&resolved)?;
                self.check_directory_parent(&parent)?;
                self.entries
                    .insert(final_path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = Self::normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = Self::normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut rows: Vec<(String, Kind)> = self
                    .direct_children(&resolved)
                    .map(|(p, e)| (Self::basename(p), e.kind()))
                    .collect();
                rows.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(rows)
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = Self::normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(
        &mut self,
        from: &str,
        to: &str,
        overwrite: bool,
    ) -> Result<(), Error> {
        let src = Self::normalize(from, &self.cwd)?;
        let dst = Self::normalize(to, &self.cwd)?;

        if src == "/" || dst == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_final) = self.resolve_parent(&src)?;
        self.check_directory_parent(&src_parent)?;
        if !self.entries.contains_key(&src_final) {
            return Err(Error::Missing(src_final));
        }

        if matches!(self.entries.get(&src_final), Some(Entry::Dir))
            && self.contains_cwd(&src_final)
        {
            return Err(Error::Busy(src_final));
        }

        let (dst_parent, dst_final) = self.resolve_parent(&dst)?;
        self.check_directory_parent(&dst_parent)?;

        if matches!(self.entries.get(&dst_final), Some(Entry::Dir))
            && self.contains_cwd(&dst_final)
        {
            return Err(Error::Busy(dst_final));
        }

        if src_final == dst_final {
            return Ok(());
        }

        if matches!(self.entries.get(&src_final), Some(Entry::Dir))
            && Self::has_prefix(&dst_final, &src_final)
        {
            return Err(Error::Cycle(dst_final));
        }

        let source = self.entries.get(&src_final).cloned().unwrap();
        let destination = self.entries.get(&dst_final).cloned();

        if destination.is_some() && !overwrite {
            return Err(Error::Exists(dst_final));
        }

        if let Some(dest) = &destination {
            match (&source, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if self.direct_children(&dst_final).next().is_some() {
                        return Err(Error::NotEmpty(dst_final));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst_final)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst_final)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| Self::has_prefix(p, &src_final))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if destination.is_some() {
            let old: Vec<String> = self
                .entries
                .keys()
                .filter(|p| Self::has_prefix(p, &dst_final))
                .cloned()
                .collect();
            for p in old {
                self.entries.remove(&p);
            }
        }
        for (p, _) in &moving {
            self.entries.remove(p);
        }

        for (old, entry) in moving {
            let suffix = if old == src_final {
                Vec::new()
            } else {
                Self::components(&old[src_final.len() + 1..])
            };
            let new = Self::join(&dst_final, &suffix);
            self.entries.insert(new, entry);
        }
        Ok(())
    }

        let normalized = Self::normalize(path, &self.cwd)?;
        if normalized == "/" {
            return Err(Error::Root);
        }

        self.check_directory_parent(&parent)?;

            .entries
            .get(&final_path)
            .cloned()

            return Err(Error::Busy(final_path));
        }

            && self.direct_children(&final_path).next().is_some()
            && !recursive
        {
            return Err(Error::NotEmpty(final_path));
        }

        let remove: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::has_prefix(p, &final_path))
            .cloned()
            .collect();
        for p in remove {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized = Self::normalize(link, &self.cwd)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (parent, final_path) = self.resolve_parent(&normalized)?;
        self.check_directory_parent(&parent)?;
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries
            .insert(final_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = Self::normalize(path, &self.cwd)?;
        if normalized == "/" {
            return Err(Error::InvalidTarget(0));
        }

        let (parent, final_path) = self.resolve_parent(&normalized)?;
        self.check_directory_parent(&parent)?;
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = Self::normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut rows: Vec<(String, u64)> = self
            .entries
            .keys()
            .filter(|p| Self::has_prefix(p, &resolved))
            .map(|p| (p.clone(), self.subtree_size(p)))
            .collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

    Dir,
    File,
    Link,

pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn components(path: &str) -> Result<Vec<String>, Error> {
    if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(path
        .split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .map(str::to_string)
        .collect())
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
    let input = components(path)?;
    let mut out = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd).unwrap_or_default()
    };
    for part in input {
        if part == ".." {
            out.pop();
        } else {
            out.push(part);
        }
    }
    Ok(canonical(&out))
}

fn split_path(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_string(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_string(), path[1..].to_string()),
        Some(i) => (path[..i].to_string(), path[i + 1..].to_string()),
        None => ("/".to_string(), path.to_string()),
    }
}

fn join_parts(base: &str, tail: &[String]) -> String {
    let mut parts = components(base).unwrap_or_default();
    parts.extend_from_slice(tail);
    canonical(&parts)
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn resolve(&self, path: &str, follow_final: bool) -> Result<String, Error> {
        let raw = components(path)?;
        let mut pending = if path.starts_with('/') {
            raw
        } else {
            let mut base = components(&self.cwd)?;
            base.extend(raw);
            base
        };
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(canonical(&resolved));
            }
            let part = pending.remove(0);
            if part == ".." {
                resolved.pop();
                continue;
            }
            resolved.push(part);
            let current = canonical(&resolved);
            let final_component = pending.is_empty();

            if let Some(entry) = self.entries.get(&current) {
                if let Entry::Link(target) = entry {
                    if follow_final || !final_component {
                        if visited.contains(&current) {
                            return Err(Error::Loop(current));
                        }
                        if expansions >= 8 {
                            return Err(Error::Depth(current));
                        }
                        visited.insert(current.clone());
                        expansions += 1;

                        let target_parts = components(target)
                            .map_err(|e| match e {
                                Error::InvalidPath(i) => Error::InvalidTarget(i),
                                _ => Error::InvalidTarget(0),
                            })?;
                        let mut expanded = if target.starts_with('/') {
                            Vec::new()
                        } else {
                            resolved[..resolved.len() - 1].to_vec()
                        };
                        for p in target_parts {
                            if p == ".." {
                                expanded.pop();
                            } else {
                                expanded.push(p);
                            }
                        }
                        expanded.extend(pending);
                        pending = expanded;
                        resolved.clear();
                        continue;
                    }
                }
            } else {
                return Err(Error::Missing(current));
            }

            if !pending.is_empty() {
                match self.entries.get(&current) {
                    Some(Entry::Dir) => {}
                    Some(_) => return Err(Error::NotDir(current)),
                    None => return Err(Error::Missing(current)),
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = split_path(path);
        let resolved_parent = self.resolve(&parent, true)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };
        self.entries
            .keys()
            .filter(|p| {
                if !p.starts_with(&prefix) {
                    return false;
                }
                let rest = &p[prefix.len()..];
                !rest.is_empty() && !rest.contains('/')
            })
            .cloned()
            .collect()
    }

    fn is_within(path: &str, dir: &str) -> bool {
        path == dir || (dir == "/" && path.starts_with('/')) || path.starts_with(&format!("{}/", dir))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        let (parent, name) = split_path(&normalized);
        let resolved_parent = self.resolve(&parent, true)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        let target = if name.is_empty() {
            "/".to_string()
        } else if resolved_parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", resolved_parent, name)
        };
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        let target = match self.resolve(&normalized, true) {
            Ok(path) => path,
            Err(Error::Missing(missing)) if missing == normalized => {
                let (parent, name) = split_path(&normalized);
                let resolved_parent = self.resolve(&parent, true)?;
                match self.entries.get(&resolved_parent) {
                    Some(Entry::Dir) => {}
                    Some(_) => return Err(Error::NotDir(resolved_parent)),
                    None => return Err(Error::Missing(resolved_parent)),
                }
                if name.is_empty() {
                    return Err(Error::IsDir("/".to_string()));
                }
                if resolved_parent == "/" {
                    format!("/{}", name)
                } else {
                    format!("{}/{}", resolved_parent, name)
                }
            }
            Err(e) => return Err(e),
        };
        match self.entries.get(&target) {
            Some(Entry::Dir) => return Err(Error::IsDir(target)),
            Some(Entry::File(_)) => {}
            Some(Entry::Link(_)) => return Err(Error::Missing(target)),
            None => {
                let (parent, _) = split_path(&target);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {}
                    Some(_) => return Err(Error::NotDir(parent)),
                    None => return Err(Error::Missing(parent)),
                }
            }
        }
        self.entries.insert(target, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => Err(Error::Missing(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut result = Vec::new();
        for child in self.children(&resolved) {
            let basename = child.rsplit('/').next().unwrap_or("").to_string();
            if let Some(entry) = self.entries.get(&child) {
                result.push((basename, entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) | None => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from_norm = normalize(from, &self.cwd)?;
        let to_norm = normalize(to, &self.cwd)?;
        if from_norm == "/" || to_norm == "/" {
            return Err(Error::Root);
        }

        let (src_parent_raw, src_name) = split_path(&from_norm);
        let src_parent = self.resolve(&src_parent_raw, true)?;
        match self.entries.get(&src_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(src_parent)),
            None => return Err(Error::Missing(src_parent)),
        }
        let src = if src_parent == "/" {
            format!("/{}", src_name)
        } else {
            format!("{}/{}", src_parent, src_name)
        };
        let source_entry = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;
        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent_raw, dst_name) = split_path(&to_norm);
        let dst_parent = self.resolve(&dst_parent_raw, true)?;
        match self.entries.get(&dst_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(dst_parent)),
            None => return Err(Error::Missing(dst_parent)),
        }
        let dst = if dst_parent == "/" {
            format!("/{}", dst_name)
        } else {
            format!("{}/{}", dst_parent, dst_name)
        };

        if matches!(self.entries.get(&dst), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }
        if src == dst {
            return Ok(());
        }
        if matches!(source_entry, Entry::Dir) && Self::is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let dest_entry = self.entries.get(&dst).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }
        if let Some(dest) = &dest_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&dst).is_empty() {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| Self::is_within(p, &src))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if dest_entry.is_some() {
            let remove: Vec<String> = self
                .entries
                .keys()
                .filter(|p| Self::is_within(p, &dst))
                .cloned()
                .collect();
            for p in remove {
                self.entries.remove(&p);
            }
        }

        for (old, _) in &moving {
            self.entries.remove(old);
        }
        for (old, entry) in moving {
            let suffix = &old[src.len()..];
            self.entries.insert(format!("{}{}", dst, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent_raw, name) = split_path(&normalized);
        let parent = self.resolve(&parent_raw, true)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        let target = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }
        let remove: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &target))
            .cloned()
            .collect();
        for p in remove {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized_link = normalize(link, &self.cwd)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent_raw, name) = split_path(&normalized_link);
        let parent = self.resolve(&parent_raw, true)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        let final_path = if name.is_empty() {
            "/".to_string()
        } else if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries
            .insert(final_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true)?;
        let root_entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &resolved))
            .cloned()
            .collect();

        fn size(path: &str, entries: &BTreeMap<String, Entry>, cache: &mut BTreeMap<String, u64>) -> u64 {
            if let Some(value) = cache.get(path) {
                return *value;
            }
            let value = match entries.get(path) {
                Some(Entry::File(bytes)) => bytes.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                Some(Entry::Dir) => {
                    let prefix = if path == "/" {
                        "/".to_string()
                    } else {
                        format!("{}/", path)
                    };
                    entries
                        .iter()
                        .filter(|(child, _)| {
                            child.starts_with(&prefix)
                                && !child[prefix.len()..].contains('/')
                                && !child[prefix.len()..].is_empty()
                        })
                        .map(|(child, _)| size(child, entries, cache))
                        .sum()
                }
                None => 0,
            };
            cache.insert(path.to_string(), value);
            value
        }

        let mut cache = BTreeMap::new();
        let mut result: Vec<(String, u64)> = paths
            .into_iter()
            .map(|p| {
                let n = size(&p, &self.entries, &mut cache);
                (p, n)
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        if !matches!(root_entry, Entry::Dir | Entry::File(_) | Entry::Link(_)) {
            return Err(Error::Missing(resolved));
        }
        Ok(result)
    }
}

enum Entry {

    Link(String),

pub struct Fs {

}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn path_parts(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(str::to_string).collect()
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
    validate_path(path)?;
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        path_parts(cwd)
    };
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_string()),
        }
    }
    Ok(canonical(&parts))
}

fn parent_and_name(path: &str) -> (String, String) {
    let mut parts = path_parts(path);
    let name = parts.pop().unwrap_or_default();
    (canonical(&parts), name)
}

fn join_child(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{}", name)
    } else {
        format!("{}/{}", parent, name)
    }
}

fn is_within(path: &str, directory: &str) -> bool {
    path == directory
        || (directory == "/" && path.starts_with('/'))
        || (path.starts_with(directory)
            && path.as_bytes().get(directory.len()) == Some(&b'/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn resolve(
        &self,
        initial: &str,
        follow_final: bool,
        allow_missing_final: bool,
    ) -> Result<String, Error> {
        let mut pending = path_parts(initial);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(canonical(&resolved));
            }

            let name = pending.remove(0);
            let prefix_parts = {
                let mut v = resolved.clone();
                v.push(name.clone());
                v
            };
            let prefix = canonical(&prefix_parts);
            let final_component = pending.is_empty();

            match self.entries.get(&prefix) {
                None if final_component && allow_missing_final => return Ok(prefix),
                None => return Err(Error::Missing(prefix)),
                Some(entry) => {
                    let should_follow = matches!(entry, Entry::Link(_))
                        && (!final_component || follow_final);
                    if should_follow {
                        if visited.contains(&prefix) {
                            return Err(Error::Loop(prefix));
                        }
                        if expansions >= 8 {
                            return Err(Error::Depth(prefix));
                        }
                        visited.insert(prefix.clone());
                        expansions += 1;

                        let target = match entry {
                            Entry::Link(target) => target,
                            _ => unreachable!(),
                        };
                        let target_path = if target.starts_with('/') {
                            normalize(target, "/")?
                        } else {
                            let link_parent = canonical(&resolved);
                            normalize(target, &link_parent)?
                        };

                        let mut combined = path_parts(&target_path);
                        combined.extend(pending.into_iter());
                        let combined_path = canonical(&combined);
                        pending = path_parts(&combined_path);
                        resolved.clear();
                        continue;
                    }

                    if !final_component && !matches!(entry, Entry::Dir) {
                        return Err(Error::NotDir(prefix));
                    }
                    resolved.push(name);
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = parent_and_name(path);
        let physical_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&physical_parent) {
            Some(Entry::Dir) => Ok((physical_parent, name)),
            Some(_) => Err(Error::NotDir(physical_parent)),
            None => Err(Error::Missing(physical_parent)),
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if path == "/" {
                    candidate
                        .strip_prefix('/')
                        .is_some_and(|tail| !tail.is_empty() && !tail.contains('/'))
                } else {
                    candidate
                        .strip_prefix(path)
                        .is_some_and(|tail| tail.starts_with('/') && !tail[1..].is_empty() && !tail[1..].contains('/'))
                }
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        let (parent, name) = self.resolve_parent(&normalized)?;
        let destination = join_child(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        let destination = self.resolve(&normalized, true, true)?;
        match self.entries.get(&destination) {
            Some(Entry::Dir) => return Err(Error::IsDir(destination)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(destination, Entry::File(data.to_vec()));
                return Ok(());
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {}
        }

        let (parent, _) = parent_and_name(&destination);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        self.entries
            .insert(destination, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }

        let mut result = Vec::new();
        for child in self.children(&resolved) {
            let name = child.rsplit('/').next().unwrap_or("").to_string();
            let kind = match self.entries.get(&child) {
                Some(Entry::Dir) => Kind::Dir,
                Some(Entry::File(_)) => Kind::File,
                Some(Entry::Link(_)) => Kind::Link,
                None => continue,
            };
            result.push((name, kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source_input = normalize(from, &self.cwd)?;
        let destination_input = normalize(to, &self.cwd)?;

        if source_input == "/" || destination_input == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source_input)?;
        let source = join_child(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination_input)?;
        let destination = join_child(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest_entry) = destination_entry {
            let source_is_dir = matches!(source_entry, Entry::Dir);
            let dest_is_dir = matches!(dest_entry, Entry::Dir);
            match (source_is_dir, dest_is_dir) {
                (true, false) => return Err(Error::NotDir(destination)),
                (false, true) => return Err(Error::IsDir(destination)),
                (true, true) => {
                    if !self.children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(key, _)| is_within(key, &source))
            .map(|(key, value)| (key.clone(), value.clone()))
            .collect();

        for (key, _) in &moved {
            self.entries.remove(key);
        }
        if destination_entry.is_some() {
            self.entries.remove(&destination);
        }
        for (old_path, entry) in moved {
            let suffix = &old_path[source.len()..];
            let new_path = format!("{}{}", destination, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = normalize(path, &self.cwd)?;
        if normalized == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = join_child(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        let removals: Vec<String> = self
            .entries
            .keys()
            .filter(|key| is_within(key, &target))
            .cloned()
            .collect();
        for key in removals {
            self.entries.remove(&key);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized_link = normalize(link, &self.cwd)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (parent, name) = self.resolve_parent(&normalized_link)?;
        let destination = join_child(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let (parent, name) = parent_and_name(&normalized);
        let physical_parent = self.resolve(&parent, true, false)?;
        let final_path = join_child(&physical_parent, &name);

        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&normalized, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| is_within(candidate, &resolved))
            .cloned()
            .collect();

        let mut sizes = BTreeMap::new();
        for entry_path in &paths {
            let size = match self.entries.get(entry_path) {
                Some(Entry::File(data)) => data.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                Some(Entry::Dir) => 0,
                None => 0,
            };
            sizes.insert(entry_path.clone(), size);
        }

        let mut directory_paths = paths.clone();
        directory_paths.sort_by_key(|p| std::cmp::Reverse(path_parts(p).len()));
        for directory in directory_paths {
            if matches!(self.entries.get(&directory), Some(Entry::Dir)) {
                let total = self
                    .entries
                    .keys()
                    .filter(|candidate| candidate.as_str() != directory
                        && is_within(candidate, &directory))
                    .filter(|candidate| {
                        let (parent, _) = parent_and_name(candidate);
                        parent == directory
                    })
                    .map(|child| *sizes.get(child).unwrap_or(&0))
                    .sum::<u64>();
                sizes.insert(directory, total);
            }
        }

        Ok(paths
            .into_iter()
            .map(|entry_path| {
                let size = *sizes.get(&entry_path).unwrap_or(&0);
                (entry_path, size)
            })
            .collect())
    }
}

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) | None => {

                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())

            Some(Entry::Link(_)) => unreachable!(),
        }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_existing(&canonical, true)?;
        match resolved.entry {
            Entry::File(bytes) => Ok(bytes),
            Entry::Dir => Err(Error::IsDir(resolved.path)),
            Entry::Link(_) => unreachable!(),
        }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {

        let resolved = self.resolve_existing(&canonical, true)?;
        if !matches!(resolved.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved.path));
        }

        let mut children = self
            .entries
            .iter()
            .filter_map(|(child_path, entry)| {
                if child_path == "/" {
                    return None;
                }
                let (parent, name) = split_parent(child_path);
                (parent == resolved.path).then(|| {
                    let kind = match entry {
                        Entry::Dir => Kind::Dir,
                        Entry::File(_) => Kind::File,
                        Entry::Link(_) => Kind::Link,
                    };
                    (name, kind)
                })
            })
            .collect::<Vec<_>>();
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_existing(&canonical, true)?;
        if !matches!(resolved.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved.path));
        }
        self.cwd = resolved.path;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source_path = normalize(from, &self.cwd)?;
        let destination_path = normalize(to, &self.cwd)?;

        if source_path == "/" || destination_path == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = split_parent(&source_path);
        let physical_source_parent = self.resolve_existing(&source_parent, true)?.path;
        let source = join(&physical_source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (destination_parent, destination_name) = split_parent(&destination_path);
        let physical_destination_parent =
            self.resolve_existing(&destination_parent, true)?.path;
        let destination = join(&physical_destination_parent, &destination_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(existing) = &destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination) {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let moving = self
            .entries
            .iter()
            .filter(|(path, _)| contains_path(&source, path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect::<Vec<_>>();

        if destination_entry.is_some() {
            self.remove_subtree(&destination);
        }
        self.remove_subtree(&source);
        for (old_path, entry) in moving {
            let suffix = old_path.strip_prefix(&source).unwrap_or("");
            let new_path = if suffix.is_empty() {
                destination.clone()
            } else if destination == "/" {
                format!("{destination}{suffix}")
            } else {
                format!("{destination}{suffix}")
            };
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = split_parent(&canonical);
        let physical_parent = self.resolve_existing(&parent, true)?.path;
        let target = join(&physical_parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && contains_path(&target, &self.cwd) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&target) && !recursive {
            return Err(Error::NotEmpty(target));
        }
        self.remove_subtree(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = normalize(link, &self.cwd)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(index) = target.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidTarget(index));
        }

        let (parent, name) = split_parent(&canonical_link);
        let physical_parent = self.resolve_existing(&parent, true)?.path;
        let resolved_link = join(&physical_parent, &name);
        if self.entries.contains_key(&resolved_link) {
            return Err(Error::Exists(resolved_link));
        }
        self.entries
            .insert(resolved_link, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_existing(&canonical, false)?;
        match resolved.entry {
            Entry::Link(target) => Ok(target),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_existing(&canonical, true)?;
        let mut result = Vec::new();

        for (entry_path, entry) in self
            .entries
            .iter()
            .filter(|(entry_path, _)| contains_path(&resolved.path, entry_path))
        {
            let size = match entry {
                Entry::Dir => self
                    .entries
                    .iter()
                    .filter(|(child_path, _)| {
                        *child_path != entry_path && contains_path(entry_path, child_path)
                    })
                    .filter_map(|(_, child)| match child {
                        Entry::File(bytes) => Some(bytes.len() as u64),
                        Entry::Link(target) => Some(target.len() as u64),
                        Entry::Dir => None,
                    })
                    .sum(),
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            result.push((entry_path.clone(), size));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    fn resolve_existing(&self, path: &str, follow_final: bool) -> Result<Lookup, Error> {
        let mut pending = components(path);
        let mut current = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        if pending.is_empty() {
            return Ok(Lookup {
                path: "/".to_owned(),
                entry: Entry::Dir,
            });
        }

        while let Some(component) = pending.first().cloned() {
            pending.remove(0);
            current.push(component);
            let physical = absolute(&current);
            let entry = self
                .entries
                .get(&physical)
                .cloned()
                .ok_or_else(|| Error::Missing(physical.clone()))?;
            let final_component = pending.is_empty();

            if !final_component && !matches!(entry, Entry::Dir | Entry::Link(_)) {
                return Err(Error::NotDir(physical));
            }

            if let Entry::Link(target) = &entry {
                if follow_final || !final_component {
                    if visited.contains(&physical) {
                        return Err(Error::Loop(physical));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(physical));
                    }
                    visited.insert(physical.clone());
                    expansions += 1;

                    let target_path = if target.starts_with('/') {
                        normalize(target, "/")?
                    } else {
                        let parent = parent_of(&physical);
                        normalize(target, &parent)?
                    };
                    let mut combined = components(&target_path);
                    combined.extend(pending);
                    let expanded = absolute(&combined);

                    current.clear();
                    pending = components(&expanded);
                    if pending.is_empty() {
                        return Ok(Lookup {
                            path: "/".to_owned(),
                            entry: Entry::Dir,
                        });
                    }
                    continue;
                }
            }

            if final_component {
                return Ok(Lookup {
                    path: physical,
                    entry,
                });
            }
        }

        let physical = absolute(&current);
        let entry = self
            .entries
            .get(&physical)
            .cloned()
            .ok_or_else(|| Error::Missing(physical.clone()))?;
        Ok(Lookup {
            path: physical,
            entry,
        })
    }

    fn resolve_for_write(&self, path: &str) -> Result<String, Error> {
        match self.resolve_existing(path, true) {
            Ok(resolved) => Ok(resolved.path),
            Err(Error::Missing(missing)) => {
                let (parent, name) = split_parent(&missing);
                let resolved_parent = self.resolve_existing(&parent, true)?;
                if !matches!(resolved_parent.entry, Entry::Dir) {
                    return Err(Error::NotDir(resolved_parent.path));
                }
                Ok(join(&resolved_parent.path, &name))
            }
            Err(error) => Err(error),
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|candidate| {
            candidate != path && parent_of(candidate) == path
        })
    }

    fn remove_subtree(&mut self, path: &str) {
        let doomed = self
            .entries
            .keys()
            .filter(|candidate| contains_path(path, candidate))
            .cloned()
            .collect::<Vec<_>>();
        for candidate in doomed {
            self.entries.remove(&candidate);
        }
    }
}

fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
    if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(Error::InvalidPath(index));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }

    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd)
    };

    for part in path.split('/') {
        match part {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_owned()),
        }
    }
    Ok(absolute(&parts))
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(ToOwned::to_owned)
        .collect()
}

fn absolute(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn split_parent(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_owned(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_owned(), path[1..].to_owned()),
        Some(index) => (path[..index].to_owned(), path[index + 1..].to_owned()),
        None => ("/".to_owned(), path.to_owned()),
    }
}

fn parent_of(path: &str) -> String {
    split_parent(path).0
}

fn join(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else if name.is_empty() {
        parent.to_owned()
    } else {
        format!("{parent}/{name}")
    }
}

fn contains_path(parent: &str, candidate: &str) -> bool {
    if parent == "/" {
        return candidate.starts_with('/');
    }
    candidate == parent
        || candidate
            .strip_prefix(parent)
            .is_some_and(|suffix| suffix.starts_with('/'))
}
