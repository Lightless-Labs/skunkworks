#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_close = if bytes[0] == b'"' {
        let mut i = 1;
        let mut close = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match close {
            Some(index) => Some(index),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(close) = quoted_close {
            if i <= close {
                i = close + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let mut local = String::new();
    let quoted;

    if let Some(close) = quoted_close {
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut origins = Vec::new();
        let mut pos = 1;
        while pos < close {
            let byte = bytes[pos];
            if byte == b'\\' {
                let Some(&escaped) = bytes.get(pos + 1).filter(|_| pos + 1 < close) else {
                    return Err(EmailError::BadEscape { at: pos });
                };
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: pos });
                }
                local.push(escaped as char);
                origins.push(pos);
                pos += 2;
            } else {
                if !(0x20..=0x7e).contains(&byte) {
                    return Err(EmailError::QuotedChar { at: pos });
                }
                local.push(byte as char);
                origins.push(pos);
                pos += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        quoted = true;
    } else {
        let local_bytes = &bytes[..separator];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (index, &byte) in local_bytes.iter().enumerate() {
            if byte == b'.' {
                if index == 0
                    || index + 1 == local_bytes.len()
                    || local_bytes[index - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: index });
                }
            } else if !(byte.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&byte))
            {
                return Err(EmailError::LocalChar { at: index });
            }
            local.push(byte as char);
        }
        quoted = false;
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_bytes = &raw_domain[..domain_len];
    let mut label_count = 0usize;
    let mut label_start = 0usize;

    for split in 0..=domain_len {
        if split == domain_len || domain_bytes[split] == b'.' {
            let label = &domain_bytes[label_start..split];
            let absolute_start = domain_start + label_start;

            if label.is_empty() {
                let at = if split == domain_len && has_root_dot {
                    input.len() - 1
                } else {
                    absolute_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }
            for (offset, &byte) in label.iter().enumerate() {
                if !(byte.is_ascii_alphanumeric() || byte == b'-') {
                    return Err(EmailError::DomainChar {
                        at: absolute_start + offset,
                    });
                }
            }

            label_count += 1;
            label_start = split + 1;
        }
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_label_start = domain_bytes
        .iter()
        .rposition(|&byte| byte == b'.')
        .map_or(0, |index| index + 1);
    let last_label = &domain_bytes[last_label_start..];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_label_start,
        });
    }

    let domain = domain_bytes
        .iter()
        .map(|&byte| byte.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
