#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1i128 << 120) - 1;

fn checksum(payload: &[u8]) -> u8 {
    payload
        .iter()
        .enumerate()
        .fold(0u32, |sum, (i, &byte)| {
            let code = match byte {
                b'T' => 2,
                b'0' => 0,
                b'1' => 1,
                _ => 0,
            };
            (sum + ((i as u32 + 1) * code)) % 7
        }) as u8
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    checksum_enabled: bool,
) -> Result<String, SignedError> {
    if !(-LIMIT..=LIMIT).contains(&value) {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut reversed = Vec::new();

    if n == 0 {
        reversed.push(b'0');
    } else {
        match system {
            SignedSystem::BalancedTernary => {
                while n != 0 {
                    let mut rem = n % 3;
                    n /= 3;
                    if rem == 2 {
                        rem = -1;
                        n += 1;
                    } else if rem == -2 {
                        rem = 1;
                        n -= 1;
                    }
                    reversed.push(match rem {
                        -1 => b'T',
                        0 => b'0',
                        1 => b'1',
                        _ => unreachable!(),
                    });
                }
            }
            SignedSystem::Negabinary => {
                while n != 0 {
                    let rem = n.rem_euclid(2);
                    n = (n - rem) / -2;
                    reversed.push(if rem == 0 { b'0' } else { b'1' });
                }
            }
        }
    }

    reversed.reverse();
    let mut output = String::from_utf8(reversed).expect("ASCII digits");
    if checksum_enabled {
        let check = checksum(output.as_bytes());
        output.push(':');
        output.push(char::from(b'0' + check));
    }
    Ok(output)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(SignedError::Empty);
    }

    let colon = bytes.iter().position(|&b| b == b':');
    let payload_end = colon.unwrap_or(bytes.len());
    if payload_end == 0 {
        return Err(SignedError::Empty);
    }

    for (index, &byte) in bytes[..payload_end].iter().enumerate() {
        let valid = match system {
            SignedSystem::BalancedTernary => matches!(byte, b'T' | b'0' | b'1'),
            SignedSystem::Negabinary => matches!(byte, b'0' | b'1'),
        };
        if !valid {
            return Err(SignedError::InvalidByte { index });
        }
    }

    let supplied = if let Some(colon_index) = colon {
        let suffix = &bytes[colon_index + 1..];
        if suffix.len() != 1 || !matches!(suffix[0], b'0'..=b'6') {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
        Some((colon_index, suffix[0] - b'0'))
    } else {
        None
    };

    let payload = &bytes[..payload_end];
    if payload.len() > 1 && payload[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let (base, min_digit, max_digit) = match system {
        SignedSystem::BalancedTernary => (3i128, -1i128, 1i128),
        SignedSystem::Negabinary => (-2i128, 0i128, 1i128),
    };

    let mut value = 0i128;
    for (index, &byte) in payload.iter().enumerate() {
        let digit = match byte {
            b'T' => -1,
            b'0' => 0,
            b'1' => 1,
            _ => unreachable!(),
        };
        debug_assert!((min_digit..=max_digit).contains(&digit));

        value = value
            .checked_mul(base)
            .and_then(|v| v.checked_add(digit))
            .ok_or(SignedError::Range { index })?;
        if !(-LIMIT..=LIMIT).contains(&value) {
            return Err(SignedError::Range { index });
        }
    }

    if let Some((_, found)) = supplied {
        let expected = checksum(payload);
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(value)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    checksum_enabled: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, checksum_enabled)
}
