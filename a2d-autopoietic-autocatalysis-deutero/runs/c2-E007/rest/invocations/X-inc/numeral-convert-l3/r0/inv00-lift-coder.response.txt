#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1_i128 << 120) - 1;

fn base(system: &SignedSystem) -> i128 {
    match system {
        SignedSystem::BalancedTernary => 3,
        SignedSystem::Negabinary => -2,
    }
}

fn digit_value(byte: u8, system: &SignedSystem) -> Option<i128> {
    match system {
        SignedSystem::BalancedTernary => match byte {
            b'T' => Some(-1),
            b'0' => Some(0),
            b'1' => Some(1),
            _ => None,
        },
        SignedSystem::Negabinary => match byte {
            b'0' => Some(0),
            b'1' => Some(1),
            _ => None,
        },
    }
}

fn digit_char(value: i128, system: &SignedSystem) -> char {
    match system {
        SignedSystem::BalancedTernary => match value {
            -1 => 'T',
            0 => '0',
            1 => '1',
            _ => unreachable!(),
        },
        SignedSystem::Negabinary => match value {
            0 => '0',
            1 => '1',
            _ => unreachable!(),
        },
    }
}

fn checksum(payload: &str) -> u8 {
    payload
        .bytes()
        .enumerate()
        .fold(0_u32, |sum, (index, byte)| {
            let code = match byte {
                b'T' => 2,
                b'0' => 0,
                b'1' => 1,
                _ => 0,
            };
            (sum + ((index as u32 + 1) * code)) % 7
        }) as u8
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    checksum_enabled: bool,
) -> Result<String, SignedError> {
    if !(-LIMIT..=LIMIT).contains(&value) {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut reversed = Vec::new();

    if n == 0 {
        reversed.push('0');
    } else {
        while n != 0 {
            let digit = match system {
                SignedSystem::BalancedTernary => {
                    let rem = n.rem_euclid(3);
                    let d = if rem == 2 { -1 } else { rem };
                    n = (n - d) / 3;
                    d
                }
                SignedSystem::Negabinary => {
                    let d = n.rem_euclid(2);
                    n = (n - d) / -2;
                    d
                }
            };
            reversed.push(digit_char(digit, &system));
        }
    }

    reversed.reverse();
    let mut output: String = reversed.into_iter().collect();

    if checksum_enabled {
        let check = checksum(&output);
        output.push(':');
        output.push(char::from(b'0' + check));
    }

    Ok(output)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(SignedError::Empty);
    }

    let colon = bytes.iter().position(|&byte| byte == b':');
    let payload_end = colon.unwrap_or(bytes.len());

    if payload_end == 0 {
        return Err(SignedError::Empty);
    }

    for (index, &byte) in bytes[..payload_end].iter().enumerate() {
        if digit_value(byte, &system).is_none() {
            return Err(SignedError::InvalidByte { index });
        }
    }

    if let Some(colon_index) = colon {
        let suffix = &bytes[colon_index + 1..];
        if suffix.len() != 1 || !(b'0'..=b'6').contains(&suffix[0]) {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
    }

    let payload = &input[..payload_end];
    if payload_end > 1 && bytes[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let b = base(&system);
    let mut accumulator = 0_i128;

    for (index, &byte) in bytes[..payload_end].iter().enumerate() {
        let digit = digit_value(byte, &system).expect("payload was validated");
        accumulator = accumulator
            .checked_mul(b)
            .and_then(|value| value.checked_add(digit))
            .ok_or(SignedError::Range { index })?;

        if !(-LIMIT..=LIMIT).contains(&accumulator) {
            return Err(SignedError::Range { index });
        }
    }

    if let Some(colon_index) = colon {
        let expected = checksum(payload);
        let found = bytes[colon_index + 1] - b'0';
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(accumulator)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    checksum_enabled: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, checksum_enabled)
}
