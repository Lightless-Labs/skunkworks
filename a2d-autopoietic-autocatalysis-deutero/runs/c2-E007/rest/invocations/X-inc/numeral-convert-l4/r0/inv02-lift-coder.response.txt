#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];
const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];
const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
    (1, ""),
];

fn append_under_thousand(n: u128, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_under_hundred(rem, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

fn append_under_hundred(n: u128, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut groups: Vec<(u128, &str)> = Vec::new();
    let mut remaining = magnitude;
    for &(scale, name) in &SCALES {
        let group = remaining / scale;
        remaining %= scale;
        if group != 0 {
            groups.push((group, name));
        }
    }

    for (i, &(group, scale)) in groups.iter().enumerate() {
        if i > 0 {
            let last_is_units = i + 1 == groups.len();
            if last_is_units && group < 100 {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        append_under_thousand(group, &mut result);
        if !scale.is_empty() {
            result.push(' ');
            result.push_str(scale);
        }
    }

    Ok(result)
}

fn recognized_word(word: &str, first: bool) -> bool {
    if word == "minus" {
        return first;
    }
    SMALL.contains(&word)
        || TENS.contains(&word)
        || matches!(
            word,
            "hundred"
                | "thousand"
                | "million"
                | "billion"
                | "trillion"
                | "quadrillion"
                | "quintillion"
                | "and"
        )
}

fn word_value(word: &str) -> Option<u128> {
    if let Some(i) = SMALL.iter().position(|&w| w == word) {
        return Some(i as u128);
    }
    if let Some(i) = TENS.iter().position(|&w| w == word) {
        return Some((i as u128 + 2) * 10);
    }
    None
}

fn scale_value(word: &str) -> Option<u128> {
    match word {
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let tokens: Vec<(usize, &str)> = input
        .split(|c| c == ' ' || c == '-')
        .scan(0usize, |offset, word| {
            let start = *offset;
            *offset += word.len() + 1;
            Some((start, word))
        })
        .collect();

    for (i, &(index, word)) in tokens.iter().enumerate() {
        if !recognized_word(word, i == 0) {
            return Err(WordsError::InvalidWord { index });
        }
    }

    let mut negative = false;
    let start = if tokens.first().is_some_and(|(_, w)| *w == "minus") {
        negative = true;
        1
    } else {
        0
    };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(_, word) in &tokens[start..] {
        if let Some(value) = word_value(word) {
            group += value;
        } else if word == "hundred" {
            group *= 100;
        } else if let Some(scale) = scale_value(word) {
            total += group * scale;
            group = 0;
        }

        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let index = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index });
    }

    Ok(value)
}
