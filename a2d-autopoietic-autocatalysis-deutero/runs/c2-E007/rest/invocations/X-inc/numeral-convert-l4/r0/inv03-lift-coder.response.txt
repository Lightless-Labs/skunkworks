#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
    (1, ""),
];

fn encode_under_thousand(n: u128) -> String {
    let mut out = String::new();
    let mut rem = n;

    if rem >= 100 {
        out.push_str(SMALL[(rem / 100) as usize]);
        out.push_str(" hundred");
        rem %= 100;
        if rem != 0 {
            out.push_str(" and ");
        }
    }

    if rem >= 20 {
        out.push_str(TENS[(rem / 10 - 2) as usize]);
        rem %= 10;
        if rem != 0 {
            out.push('-');
            out.push_str(SMALL[rem as usize]);
        }
    } else if rem != 0 {
        out.push_str(SMALL[rem as usize]);
    }

    out
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    if magnitude == LIMIT {
        out.push_str("one quintillion");
        return Ok(out);
    }

    let mut remaining = magnitude;
    let mut groups: Vec<(u128, &'static str)> = Vec::new();

    for &(scale, name) in &SCALES {
        if scale == 1 {
            let group = remaining % 1000;
            if group != 0 {
                groups.push((group, ""));
            }
        } else {
            let group = remaining / scale;
            if group != 0 {
                groups.push((group, name));
                remaining %= scale;
            }
        }
    }

    for (i, (group, scale)) in groups.iter().enumerate() {
        if i > 0 {
            out.push(' ');
        }
        out.push_str(&encode_under_thousand(*group));
        if !scale.is_empty() {
            out.push(' ');
            out.push_str(scale);
        }
    }

    if groups.len() > 1 {
        let last = groups.last().unwrap().0;
        if last < 100 {
            let units_len = encode_under_thousand(last).len();
            let units_start = out.len() - units_len;
            out.insert_str(units_start, "and ");
        }
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<(u128, WordKind)> {
    if word == "zero" {
        return Some((0, WordKind::Number));
    }
    for (i, &w) in SMALL.iter().enumerate().skip(1) {
        if word == w {
            return Some((i as u128, WordKind::Number));
        }
    }
    for (i, &w) in TENS.iter().enumerate() {
        if word == w {
            return Some(((i as u128 + 2) * 10, WordKind::Number));
        }
    }

    match word {
        "hundred" => Some((100, WordKind::Hundred)),
        "thousand" => Some((1_000, WordKind::Scale)),
        "million" => Some((1_000_000, WordKind::Scale)),
        "billion" => Some((1_000_000_000, WordKind::Scale)),
        "trillion" => Some((1_000_000_000_000, WordKind::Scale)),
        "quadrillion" => Some((1_000_000_000_000_000, WordKind::Scale)),
        "quintillion" => Some((1_000_000_000_000_000_000, WordKind::Scale)),
        "and" => Some((0, WordKind::And)),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_lowercase() && b != b' ' && b != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            if i == 0 || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            tokens.push((&input[start..i], start));
            start = i + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (index, (token, position)) in tokens.iter().enumerate() {
        if *token == "minus" {
            if index != 0 {
                return Err(WordsError::InvalidWord { index: *position });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let negative = tokens.first().map(|(t, _)| *t == "minus").unwrap_or(false);
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (token, _) in arithmetic_tokens {
        match word_value(token).unwrap() {
            (value, WordKind::Number) => group += value,
            (_, WordKind::Hundred) => group *= 100,
            (scale, WordKind::Scale) => {
                total += group * scale;
                group = 0;
            }
            (_, WordKind::And) => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let mismatch = canonical
            .bytes()
            .zip(bytes.iter().copied())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(value)
}
