#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        let in_quotes = matches!(state, State::Quoted { .. });
        if !in_quotes && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        match state {
            State::Start => match b {
                b'"' => {
                    state = State::Quoted { opening: pos };
                    pos += 1;
                }
                b',' => {
                    record.push(std::mem::take(&mut field));
                    pos += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    pos += 1;
                }
                b'\r' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    pos += 2;
                }
                b => {
                    let start = pos;
                    while pos < bytes.len()
                        && bytes[pos] != b','
                        && bytes[pos] != b'\n'
                        && bytes[pos] != b'\r'
                        && bytes[pos] != b'"'
                        && bytes[pos] != 0
                    {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                    state = State::Unquoted;
                }
            },
            State::Unquoted => match b {
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                }
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    pos += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::Start;
                    pos += 1;
                }
                b'\r' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::Start;
                    pos += 2;
                }
                _ => {
                    let start = pos;
                    while pos < bytes.len()
                        && bytes[pos] != b','
                        && bytes[pos] != b'\n'
                        && bytes[pos] != b'\r'
                        && bytes[pos] != b'"'
                        && bytes[pos] != 0
                    {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                }
            },
            State::Quoted { opening } => {
                if b == b'"' {
                    if bytes.get(pos + 1) == Some(&b'"') {
                        field.push('"');
                        pos += 2;
                    } else {
                        state = State::AfterQuote;
                        pos += 1;
                    }
                } else {
                    let start = pos;
                    while pos < bytes.len() && bytes[pos] != b'"' && bytes[pos] != 0 {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                    if pos == bytes.len() {
                        return Err(CsvError::UnclosedQuote {
                            record: record_index,
                            byte: opening,
                        });
                    }
                }
            }
            State::AfterQuote => match b {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    pos += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::Start;
                    pos += 1;
                }
                b'\r' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::Start;
                    pos += 2;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            },
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if !matches!(state, State::Start) || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
