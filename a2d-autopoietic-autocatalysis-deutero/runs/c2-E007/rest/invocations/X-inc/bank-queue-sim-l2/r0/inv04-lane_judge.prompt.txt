#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    occupied_until: Option<u64>,
    customer: Option<u32>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<Customer> = customers.to_vec();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied_until: None,
            customer: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;

    loop {
        // Finish events first, ordered by ascending server ID.
        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (state.occupied_until == Some(now)).then_some(i)
            })
            .collect();
        finishing.sort_by_key(|&i| states[i].server.id);
        for i in finishing {
            let customer = states[i].customer.take().unwrap();
            events.push(Event::Finish {
                time: now,
                customer,
                server: states[i].server.id,
            });
            states[i].occupied_until = None;
            states[i].idle_since = now;
        }

        // Open events next, ordered by descending server ID.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.server.opens == now).then_some(i)
            })
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[i].server.id,
            });
        }

        // Arrivals at this tick, already ordered by descending customer ID.
        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == now
        {
            let customer = &ordered_customers[next_customer];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(next_customer);
            next_customer += 1;
        }

        // Dispatch in FIFO order, recalculating the best server each time.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &ordered_customers[customer_index];
            let server = &states[server_index].server;

            let finish = customer
                .service
                .checked_add(now)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].occupied_until = Some(finish);
                states[server_index].customer = Some(customer.id);
            }
        }

        let all_done = next_customer == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|s| s.occupied_until.is_none());
        if all_done && states.iter().all(|s| s.opened) {
            break;
        }

        let next_arrival = ordered_customers
            .get(next_customer)
            .map(|c| c.arrival);
        let next_open = states
            .iter()
            .filter(|s| !s.opened)
            .map(|s| s.server.opens)
            .min();
        let next_finish = states.iter().filter_map(|s| s.occupied_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&t| t > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => {
                // All remaining events, if any, are at the current tick.
                // This can occur at u64::MAX.
                let has_current_event = ordered_customers
                    .get(next_customer)
                    .is_some_and(|c| c.arrival == now)
                    || states.iter().any(|s| {
                        (!s.opened && s.server.opens == now)
                            || s.occupied_until == Some(now)
                    });
                if has_current_event {
                    continue;
                }
                break;
            }
        }
    }

    Ok(events)
}

    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    occupied: Option<(u64, u32)>, // (finish time, customer id)
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_arrival = 0usize;
    let mut time = 0u64;

    loop {
        let mut finishes: Vec<(usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| match state.occupied {
                Some((finish, customer)) if finish == time => Some((i, customer)),
                _ => None,
            })
            .collect();
        finishes.sort_by_key(|(i, _)| states[*i].server.id);
        for (index, customer) in finishes {
            let server = states[index].server.id;
            events.push(Event::Finish {
                time,
                customer,
                server,
            });
            states[index].occupied = None;
            states[index].idle_since = time;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.server.opens == time).then_some(i)
            })
            .collect();
        opening.sort_by_key(|i| std::cmp::Reverse(states[*i].server.id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time,
                server: states[index].server.id,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = &arrivals[next_arrival];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(next_arrival);
            next_arrival += 1;
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &arrivals[customer_index];
            let server = &states[server_index].server;
            let service_finish = time.checked_add(customer.service).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server.id,
                time,
            })?;
            let finish = service_finish.checked_add(server.overhead).ok_or(
                Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                },
            )?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
            } else {
                states[server_index].occupied = Some((finish, customer.id));
            }
        }

        if next_arrival == arrivals.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.opened)
            && states.iter().all(|state| state.occupied.is_none())
        {
            break;
        }

        let next_arrival_time = arrivals.get(next_arrival).map(|customer| customer.arrival);
        let next_open_time = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_finish_time = states
            .iter()
            .filter_map(|state| state.occupied.map(|(finish, _)| finish))
            .min();

        let next_time = [next_arrival_time, next_open_time, next_finish_time]
            .into_iter()
            .flatten()
            .filter(|candidate| *candidate > time)
            .min();

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(events)
}
