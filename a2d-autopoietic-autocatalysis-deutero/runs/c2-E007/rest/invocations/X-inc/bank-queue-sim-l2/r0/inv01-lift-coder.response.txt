#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    occupied_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied_until: None,
        })
        .collect();

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut arrival_index = 0usize;
    let mut waiting: Vec<Customer> = Vec::new();
    let mut events = Vec::new();
    let mut time = 0u64;

    loop {
        let mut finished: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                state
                    .occupied_until
                    .filter(|&finish| finish == time)
                    .map(|_| {
                        // The customer ID is tracked separately below through the active job.
                        (state.server.id, index, 0)
                    })
            })
            .collect();
        finished.sort_by_key(|&(server_id, _, _)| server_id);

        // Occupied jobs are represented by their scheduled finish and customer ID.
        for (_, index, _) in finished {
            if let Some((customer_id, finish)) = active_jobs[index].take() {
                debug_assert_eq!(finish, time);
                events.push(Event::Finish {
                    time,
                    customer: customer_id,
                    server: states[index].server.id,
                });
                states[index].occupied_until = None;
                states[index].idle_since = time;
            }
        }

        let mut opening_indices: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == time).then_some(index)
            })
            .collect();
        opening_indices.sort_by_key(|&index| std::cmp::Reverse(states[index].server.id));
        for index in opening_indices {
            states[index].opened = true;
            states[index].idle_since = time;
            events.push(Event::Open {
                time,
                server: states[index].server.id,
            });
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            let customer = arrivals[arrival_index].clone();
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer);
            arrival_index += 1;
        }

        loop {
            if waiting.is_empty() {
                break;
            }
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting.remove(0);
            let server = &states[server_index].server;
            let finish = customer
                .service
                .checked_add(server.overhead)
                .and_then(|duration| time.checked_add(duration))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
            } else {
                states[server_index].occupied_until = Some(finish);
                active_jobs[server_index] = Some((customer.id, finish));
            }
        }

        let all_done = arrival_index == arrivals.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.occupied_until.is_none())
            && states.iter().all(|state| state.opened);
        if all_done {
            break;
        }

        let next_arrival = arrivals.get(arrival_index).map(|customer| customer.arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.occupied_until).min();

        time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&next| next > time)
            .min()
            .expect("simulation has a future event");
    }

    Ok(events)
}
