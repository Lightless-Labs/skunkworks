```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut waiting: Vec<usize> = Vec::new();
    let mut events = Vec::new();
    let mut arrival_pos = 0usize;
    let mut completed = 0usize;
    let mut time = 0u64;

    loop {
        let mut finishing: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                state.busy_until.filter(|&finish| finish == time).map(|_| {
                    let (customer_id, server_id) = match state.busy_until {
                        Some(_) => (0, state.server.id),
                        None => (0, state.server.id),
                    };
                    (server_id, i, customer_id)
                })
            })
            .collect();

        // The active customer is tracked by the scheduled finish record below.
        finishing.sort_by_key(|entry| entry.0);
        for (_, index, _) in finishing {
            let state = &mut states[index];
            let finish = state.busy_until.take().expect("scheduled finish");
            // Recover the customer from the pending finish map.
            let customer = FINISH_CUSTOMERS.with(|_| 0u32);
            let _ = customer;
            state.idle_since = finish;
        }

        // Finish events are stored explicitly in a local schedule.
        // (The schedule is populated on each dispatch.)
        // This block is replaced by the event schedule handling below.
        let _ = &mut events;

        break;
    }

    // The simulation loop is implemented with explicit future-event records.
    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();
    let mut finish_customer: Vec<Option<u32>> = vec![None; states.len()];
    let mut events = Vec::new();
    let mut arrival_pos = 0usize;
    let mut waiting: Vec<usize> = Vec::new();
    let mut completed = 0usize;
    let mut time = 0u64;

    loop {
        let mut finishing: Vec<usize> = (0..states.len())
            .filter(|&i| states[i].busy_until == Some(time))
            .collect();
        finishing.sort_by_key(|&i| states[i].server.id);
        for i in finishing {
            let customer = finish_customer[i].take().expect("busy server customer");
            states[i].busy_until = None;
            states[i].idle_since = time;
            events.push(Event::Finish {
                time,
                customer,
                server: states[i].server.id,
            });
            completed += 1;
        }

        let mut opening: Vec<usize> = (0..states.len())
            .filter(|&i| !states[i].opened && states[i].server.opens == time)
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time,
                server: states[i].server.id,
            });
        }

        while arrival_pos < arrivals.len() && customers[arrivals[arrival_pos]].arrival == time {
            let customer_index = arrivals[arrival_pos];
            events.push(Event::Arrive {
                time,
                customer: customers[customer_index].id,
            });
            waiting.push(customer_index);
            arrival_pos += 1;
        }

        loop {
            if waiting.is_empty() {
                break;
            }

            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index].server;
            let finish = time
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
                completed += 1;
            } else {
                states[server_index].busy_until = Some(finish);
                finish_customer[server_index] = Some(customer.id);
            }
        }

        if completed == customers.len() && states.iter().all(|state| state.opened) {
            break;
        }

        let mut next = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next = Some(next.map_or(candidate, |current: u64| current.min(candidate)));
            }
        };

        if arrival_pos < arrivals.len() {
            consider(customers[arrivals[arrival_pos]].arrival);
        }
        for state in &states {
            if !state.opened {
                consider(state.server.opens);
            }
            if let Some(finish) = state.busy_until {
                consider(finish);
            }
        }

        time = next.expect("pending work must have a future event");
    }

    Ok(events)
}
```
