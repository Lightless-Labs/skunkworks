struct Run {
    start: usize,
    end: usize,

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::<u8>::with_capacity(bytes.len());
    let mut positions = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<Run>::new();

    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut active_run: Option<usize> = None;
    let mut first_retained = true;
    let mut quote_protected = false;

    while i < bytes.len() {
        if quote_protected {
            let b = bytes[i];
            clean.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' {
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                quote_protected = false;
            }
            continue;
        }

        if depth > 0 {
            let b = bytes[i];
            match b {
                b'\\' => {
                    if i + 1 >= bytes.len()
                        || !(0x20..=0x7e).contains(&bytes[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                }
                b'\r' | b'\n' => {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                }
                b'(' => {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                }
                b')' => {
                    depth -= 1;
                    i += 1;
                }
                b'\t' | b' '..=b'~' => i += 1,
                _ => return Err(EmailError::CommentChar { at: i }),
            }
            continue;
        }

        let b = bytes[i];

        if b == b'"' && first_retained {
            if let Some(start) = active_run.take() {
                runs.push(Run { start, end: i });
            }
            clean.push(b);
            positions.push(i);
            i += 1;
            first_retained = false;
            quote_protected = true;
            continue;
        }

        if is_hws(b) || b == b'(' || b == b'\r' || b == b'\n' {
            if active_run.is_none() {
                active_run = Some(i);
            }
            match b {
                b' ' | b'\t' => i += 1,
                b'(' => {
                    depth = 1;
                    outer_open = i;
                    i += 1;
                }
                b'\r' | b'\n' => {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                }
                _ => unreachable!(),
            }
            continue;

            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        positions.push(i);
        first_retained = false;
        i += 1;

            end: bytes.len(),

                j += 2;
            } else if clean[j] == b'"' {
                close = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        close
    } else {
        None
    };

            && quoted_prefix_end.is_some_and(|end| idx <= end);
        if b == b'@' && !protected {
            separators.push(idx);
        }

        let left_adjacent_at = separator_clean
            .is_some_and(|s| s + 1 == clean.len() || (s + 1 < clean.len() && positions[s + 1] >= run.end));
        let right_adjacent_at = separator_clean
            .is_some_and(|s| s < clean.len() && positions[s] < run.start);
        let at_boundary = run.start == 0 || run.end == bytes.len();
        let left_edge = run.start == 0;
        let right_edge = run.end == bytes.len();

            let left = s.checked_sub(1).is_some_and(|p| {

                && positions[s + 1] >= run.end
                && s + 1 == positions.iter().position(|&p| p >= run.end).unwrap_or(usize::MAX);

        if !left_edge && !right_edge && !adjacent {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

        return Err(EmailError::EmptyInput);
    }

        let mut close = None;

            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                close = Some(j);
                break;
            } else {
                j += 1;
            }

                at: positions[0],

                j += 2;

                break;
            } else {
                j += 1;

            return Err(EmailError::TrailingLocal {
                at: local_pos[close + 1],
            });
        }

        let mut origins = Vec::new();

            let b = local_bytes[k];

                if k + 1 >= close

                {
                    return Err(EmailError::BadEscape {
                        at: local_pos[k],
                    });

                origins.push(local_pos[k]);

                    return Err(EmailError::QuotedChar {
                        at: local_pos[k],
                    });
                }

                origins.push(local_pos[k]);
                k += 1;
            }

                at: local_pos[close],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).unwrap();

    let domain_raw = &clean[domain_start..];
    let domain_positions = &positions[domain_start..];
    let mut domain_len = domain_raw.len();

        domain_len -= 1;
        Some(domain_positions[domain_len])

        None
    };

        return Err(EmailError::EmptyDomain {
            at: if domain_start < positions.len() {
                positions[domain_start]
            } else {
                input.len()
            },
        });
    }

        return Err(EmailError::DomainTooLong {
            at: domain_positions[253],
        });

            label_start = idx + 1;

            } else {
                input.len()
            };

        }

            return Err(EmailError::LabelTooLong {
                at: domain_positions[start + 63],
            });
        }
        if domain_raw[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_positions[start],
            });
        }
        if domain_raw[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_positions[end - 1],
            });
        }
        for idx in start..end {
            let b = domain_raw[idx];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar {
                    at: domain_positions[idx],
                });
            }
        }
    }

        return Err(EmailError::DomainNeedsDot {
            at: if domain_start < positions.len() {
                positions[domain_start]
            } else {
                input.len()
            },
        });

        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: domain_positions[last_start],
        });
    }

        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

        local,
        domain,
        quoted,
    })

    first_original: usize,
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::with_capacity(bytes.len());
    let mut map = Vec::with_capacity(bytes.len());
    let mut runs: Vec<CfwsRun> = Vec::new();
    let mut i = 0usize;
    let mut protected = false;

    while i < bytes.len() {
        if protected {
            let b = bytes[i];
            clean.push(b);
            map.push(i);
            i += 1;
            if b == b'\\' && i < bytes.len() {
                clean.push(bytes[i]);
                map.push(i);
                i += 1;
            } else if b == b'"' {
                protected = false;
            }
            continue;
        }

        let b = bytes[i];
        if b == b'"' && clean.is_empty() {
            clean.push(b);
            map.push(i);
            i += 1;
            protected = true;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let run_start = clean.len();
            let first_original = i;

            loop {
                if i >= bytes.len() {
                    break;
                }
                match bytes[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        let fold_at = i;
                        if bytes[i] != b'\r'
                            || i + 2 >= bytes.len()
                            || bytes[i + 1] != b'\n'
                            || (bytes[i + 2] != b' ' && bytes[i + 2] != b'\t')
                        {
                            return Err(EmailError::BadFold { at: fold_at });
                        }
                        i += 2;
                        while i < bytes.len() && (bytes[i] == b' ' || bytes[i] == b'\t') {
                            i += 1;
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut openings = [0usize; 3];
                        openings[0] = i;
                        let mut depth = 1usize;
                        i += 1;

                        while i < bytes.len() {
                            let c = bytes[i];
                            if c == b'\\' {
                                let slash = i;
                                if i + 1 >= bytes.len()
                                    || !(0x20..=0x7e).contains(&bytes[i + 1])
                                {
                                    return Err(EmailError::CommentEscape { at: slash });
                                }
                                i += 2;
                            } else if c == b'(' {
                                if depth == 3 {
                                    return Err(EmailError::CommentDepth { at: i });
                                }
                                openings[depth] = i;
                                depth += 1;
                                i += 1;
                            } else if c == b')' {
                                depth -= 1;
                                i += 1;
                                if depth == 0 {
                                    break;
                                }
                            } else if c == b'\r' || c == b'\n' {
                                let fold_at = i;
                                if c != b'\r'
                                    || i + 2 >= bytes.len()
                                    || bytes[i + 1] != b'\n'
                                    || (bytes[i + 2] != b' ' && bytes[i + 2] != b'\t')
                                {
                                    return Err(EmailError::BadFold { at: fold_at });
                                }
                                i += 2;
                                while i < bytes.len()
                                    && (bytes[i] == b' ' || bytes[i] == b'\t')
                                {
                                    i += 1;
                                }
                            } else if c == b'\t' || (0x20..=0x7e).contains(&c) {
                                i += 1;
                            } else {
                                return Err(EmailError::CommentChar { at: i });
                            }
                        }

                        if depth != 0 {
                            return Err(EmailError::UnclosedComment { at: outer });
                        }
                    }
                    _ => break,
                }
            }

            runs.push(CfwsRun {
                start: run_start,
                end: clean.len(),
                first_original,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    for run in &runs {
        let legal = run.start == 0
            || run.end == clean.len()
            || (run.start > 0 && clean[run.start - 1] == b'@')
            || (run.end < clean.len() && clean[run.end] == b'@');
        if !legal {
            return Err(EmailError::CfwsBoundary {
                at: run.first_original,
            });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1usize;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: map[0] });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut separators = Vec::new();
    for (j, &b) in clean.iter().enumerate() {
        if b == b'@' && (quote_end.is_none() || j > protected_end) {
            separators.push(j);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: map[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: map[sep] });
    }

    let local_bytes = &clean[..sep];
    let mut local = String::new();
    let quoted;

    if let Some(end_quote) = quote_end {
        if end_quote + 1 != sep {
            let at = map[end_quote + 1];
            return Err(EmailError::TrailingLocal { at });
        }

        quoted = true;
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < end_quote {
            let b = clean[j];
            if b == b'\\' {
                if j + 1 >= end_quote {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                let escaped = clean[j + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                local.push(escaped as char);
                origins.push(map[j]);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: map[j] });
                }
                local.push(b as char);
                origins.push(map[j]);
                j += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: map[end_quote],
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: map[64] });
        }

        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: map[j] });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: map[j] });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
    }

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let had_root_dot = domain_end > domain_start && clean[domain_end - 1] == b'.';
    if had_root_dot {
        domain_end -= 1;
    }

    if domain_start == domain_end {
        let at = if had_root_dot {
            map[domain_end]
        } else {
            input.len()
        };
        return Err(EmailError::EmptyDomain { at });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: map[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean[j] == b'.' {
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for (start, end) in &labels {
        if start == end {
            let at = if *start < clean.len() {
                map[*start]
            } else if had_root_dot {
                map[domain_end]
            } else {
                input.len()
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: map[start + 63],
            });
        }
        if clean[*start] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[*start] });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[end - 1] });
        }
        for j in *start..*end {
            let b = clean[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: map[j] });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: map[domain_start],
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end]
        .iter()
        .all(|b| b.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: map[last_start],
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

    CommentChar { at: usize },

#[derive(Clone, Copy)]
struct Byte {
    value: u8,
    pos: usize,
}

struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn push_run(runs: &mut Vec<Run>, start: usize, end: usize) {
    if let Some(last) = runs.last_mut() {
        if last.end == start {
            last.end = end;
            return;
        }
    }
    runs.push(Run { start, end });
}

fn cleanup(input: &[u8]) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let mut retained = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut in_quote = false;
    let mut quote_escape = false;

    while i < input.len() {
        let b = input[i];

        if in_quote {
            retained.push(Byte { value: b, pos: i });
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                in_quote = false;
            }
            i += 1;
            continue;
        }

        if b == b'"' && retained.is_empty() {
            in_quote = true;
            retained.push(Byte { value: b, pos: i });
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'\r' || b == b'\n' {
            let start = i;
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else {
                i += 1;
                while i < input.len() && is_hws(input[i]) {
                    i += 1;
                }
            }
            push_run(&mut runs, start, i);
            continue;
        }

        if b == b'(' {
            let start = i;
            let outer = i;
            let mut depth = 1usize;
            i += 1;
            while i < input.len() {
                let c = input[i];

                if c == b'\\' {
                    let slash = i;
                    i += 1;
                    if i >= input.len() || !printable(input[i]) {
                        return Err(EmailError::CommentEscape { at: slash });
                    }
                    i += 1;
                    continue;
                }

                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if c == b'(' {
                    depth += 1;
                    if depth > 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    i += 1;
                    continue;
                }

                if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 {
                        break;
                    }
                    continue;
                }

                if printable(c) || c == b'\t' {
                    i += 1;
                    continue;
                }

                return Err(EmailError::CommentChar { at: i });
            }

            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            push_run(&mut runs, start, i);
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        retained.push(Byte { value: b, pos: i });
        i += 1;
    }

    Ok((retained, runs))
}

fn at_of(byte: Option<&Byte>, input_len: usize) -> usize {
    byte.map_or(input_len, |b| b.pos)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (cleaned, runs) = cleanup(raw)?;

    for run in &runs {
        let left_boundary = run.start == 0
            || cleaned.first().is_some_and(|b| b.pos >= run.end);
        let right_boundary = run.end == raw.len()
            || cleaned.last().is_some_and(|b| b.pos < run.start);

        let adjacent_left = cleaned
            .iter()
            .rev()
            .find(|b| b.pos < run.start)
            .is_some_and(|b| b.value == b'@');
        let adjacent_right = cleaned
            .iter()
            .find(|b| b.pos >= run.end)
            .is_some_and(|b| b.value == b'@');

        if !(left_boundary || right_boundary || adjacent_left || adjacent_right) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if cleaned.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if cleaned[0].value == b'"' {
        let mut i = 1;
        while i < cleaned.len() {
            if cleaned[i].value == b'\\' {
                i += 2;
            } else if cleaned[i].value == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote { at: cleaned[0].pos });
        }
    }

    let mut separators = Vec::new();
    let protected_end = quote_close;
    for (i, byte) in cleaned.iter().enumerate() {
        if byte.value == b'@' && !protected_end.is_some_and(|end| i <= end) {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: cleaned[separators[1]].pos,
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: cleaned[sep].pos,
        });
    }

    let mut local = String::new();
    let quoted = cleaned[0].value == b'"';

    if !quoted {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: cleaned[64].pos,
            });
        }
        for i in 0..sep {
            let b = cleaned[i].value;
            if b == b'.' {
                if i == 0 || i + 1 == sep || cleaned[i - 1].value == b'.' {
                    return Err(EmailError::LocalDot { at: cleaned[i].pos });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar { at: cleaned[i].pos });
            }
            local.push(b as char);
        }
    } else {
        let close = quote_close.expect("quoted prefix must have a closing quote");
        if close + 1 != sep {
            let at = cleaned.get(close + 1).map_or(input.len(), |b| b.pos);
            return Err(EmailError::TrailingLocal { at });
        }

        let mut decoded = Vec::new();
        let mut i = 1;
        while i < close {
            let b = cleaned[i].value;
            if b == b'\\' {
                let Some(next) = cleaned.get(i + 1).filter(|_| i + 1 < close) else {
                    return Err(EmailError::BadEscape { at: cleaned[i].pos });
                };
                if !matches!(next.value, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: cleaned[i].pos });
                }
                decoded.push((next.value, cleaned[i].pos));
                i += 2;
            } else {
                if !printable(b) {
                    return Err(EmailError::QuotedChar { at: cleaned[i].pos });
                }
                decoded.push((b, cleaned[i].pos));
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: cleaned[close].pos,
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded[64].1,
            });
        }
        local.extend(decoded.into_iter().map(|(b, _)| b as char));
    }

    let domain_start = sep + 1;
    let raw_domain = &cleaned[domain_start..];
    let mut domain_end = cleaned.len();
    if domain_end > domain_start && cleaned[domain_end - 1].value == b'.' {
        domain_end -= 1;
    }

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: at_of(cleaned.get(domain_start), input.len()),
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: cleaned[domain_start + 253].pos,
        });
    }

    let root_pos = if domain_end < cleaned.len() {
        cleaned[domain_end].pos
    } else {
        input.len()
    };

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for i in domain_start..domain_end {
        if cleaned[i].value == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }
    labels.push((label_start, domain_end));

    for (start, end) in &labels {
        if start == end {
            let at = if *start == domain_end && domain_end < cleaned.len() {
                root_pos
            } else {
                at_of(cleaned.get(*start), input.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: cleaned[start + 63].pos,
            });
        }

        if cleaned[*start].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: cleaned[*start].pos,
            });
        }
        if cleaned[end - 1].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: cleaned[end - 1].pos,
            });
        }

        for i in *start..*end {
            let b = cleaned[i].value;
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: cleaned[i].pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: at_of(cleaned.get(domain_start), input.len()),
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if cleaned[last_start..last_end]
        .iter()
        .all(|b| b.value.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: cleaned[last_start].pos,
        });
    }

    let domain: String = cleaned[domain_start..domain_end]
        .iter()
        .map(|b| (b.value as char).to_ascii_lowercase())
        .collect();

    let _ = raw_domain;
    Ok(Address {
        local,
        domain,
        quoted,
    })
}
