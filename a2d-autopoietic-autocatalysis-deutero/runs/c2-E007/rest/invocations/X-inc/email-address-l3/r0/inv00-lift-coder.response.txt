#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Debug)]
struct CfwsRun {
    start: usize,
    end: usize,
    first_original: usize,
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::with_capacity(bytes.len());
    let mut map = Vec::with_capacity(bytes.len());
    let mut runs: Vec<CfwsRun> = Vec::new();
    let mut i = 0usize;
    let mut protected = false;

    while i < bytes.len() {
        if protected {
            let b = bytes[i];
            clean.push(b);
            map.push(i);
            i += 1;
            if b == b'\\' && i < bytes.len() {
                clean.push(bytes[i]);
                map.push(i);
                i += 1;
            } else if b == b'"' {
                protected = false;
            }
            continue;
        }

        let b = bytes[i];
        if b == b'"' && clean.is_empty() {
            clean.push(b);
            map.push(i);
            i += 1;
            protected = true;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let run_start = clean.len();
            let first_original = i;

            loop {
                if i >= bytes.len() {
                    break;
                }
                match bytes[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        let fold_at = i;
                        if bytes[i] != b'\r'
                            || i + 2 >= bytes.len()
                            || bytes[i + 1] != b'\n'
                            || (bytes[i + 2] != b' ' && bytes[i + 2] != b'\t')
                        {
                            return Err(EmailError::BadFold { at: fold_at });
                        }
                        i += 2;
                        while i < bytes.len() && (bytes[i] == b' ' || bytes[i] == b'\t') {
                            i += 1;
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut openings = [0usize; 3];
                        openings[0] = i;
                        let mut depth = 1usize;
                        i += 1;

                        while i < bytes.len() {
                            let c = bytes[i];
                            if c == b'\\' {
                                let slash = i;
                                if i + 1 >= bytes.len()
                                    || !(0x20..=0x7e).contains(&bytes[i + 1])
                                {
                                    return Err(EmailError::CommentEscape { at: slash });
                                }
                                i += 2;
                            } else if c == b'(' {
                                if depth == 3 {
                                    return Err(EmailError::CommentDepth { at: i });
                                }
                                openings[depth] = i;
                                depth += 1;
                                i += 1;
                            } else if c == b')' {
                                depth -= 1;
                                i += 1;
                                if depth == 0 {
                                    break;
                                }
                            } else if c == b'\r' || c == b'\n' {
                                let fold_at = i;
                                if c != b'\r'
                                    || i + 2 >= bytes.len()
                                    || bytes[i + 1] != b'\n'
                                    || (bytes[i + 2] != b' ' && bytes[i + 2] != b'\t')
                                {
                                    return Err(EmailError::BadFold { at: fold_at });
                                }
                                i += 2;
                                while i < bytes.len()
                                    && (bytes[i] == b' ' || bytes[i] == b'\t')
                                {
                                    i += 1;
                                }
                            } else if c == b'\t' || (0x20..=0x7e).contains(&c) {
                                i += 1;
                            } else {
                                return Err(EmailError::CommentChar { at: i });
                            }
                        }

                        if depth != 0 {
                            return Err(EmailError::UnclosedComment { at: outer });
                        }
                    }
                    _ => break,
                }
            }

            runs.push(CfwsRun {
                start: run_start,
                end: clean.len(),
                first_original,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    for run in &runs {
        let legal = run.start == 0
            || run.end == clean.len()
            || (run.start > 0 && clean[run.start - 1] == b'@')
            || (run.end < clean.len() && clean[run.end] == b'@');
        if !legal {
            return Err(EmailError::CfwsBoundary {
                at: run.first_original,
            });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1usize;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: map[0] });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut separators = Vec::new();
    for (j, &b) in clean.iter().enumerate() {
        if b == b'@' && (quote_end.is_none() || j > protected_end) {
            separators.push(j);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: map[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: map[sep] });
    }

    let local_bytes = &clean[..sep];
    let mut local = String::new();
    let quoted;

    if let Some(end_quote) = quote_end {
        if end_quote + 1 != sep {
            let at = map[end_quote + 1];
            return Err(EmailError::TrailingLocal { at });
        }

        quoted = true;
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < end_quote {
            let b = clean[j];
            if b == b'\\' {
                if j + 1 >= end_quote {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                let escaped = clean[j + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                local.push(escaped as char);
                origins.push(map[j]);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: map[j] });
                }
                local.push(b as char);
                origins.push(map[j]);
                j += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: map[end_quote],
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: map[64] });
        }

        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: map[j] });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: map[j] });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
    }

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let had_root_dot = domain_end > domain_start && clean[domain_end - 1] == b'.';
    if had_root_dot {
        domain_end -= 1;
    }

    if domain_start == domain_end {
        let at = if had_root_dot {
            map[domain_end]
        } else {
            input.len()
        };
        return Err(EmailError::EmptyDomain { at });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: map[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean[j] == b'.' {
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for (start, end) in &labels {
        if start == end {
            let at = if *start < clean.len() {
                map[*start]
            } else if had_root_dot {
                map[domain_end]
            } else {
                input.len()
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: map[start + 63],
            });
        }
        if clean[*start] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[*start] });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[end - 1] });
        }
        for j in *start..*end {
            let b = clean[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: map[j] });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: map[domain_start],
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end]
        .iter()
        .all(|b| b.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: map[last_start],
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
