#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
enum InternalState {
    Waiting {
        ready_at: u64,
        attempt: u8,
        remaining: Option<u64>,
    },
    Running {
        attempt: u8,
        finish_at: u64,
        started_at: u64,
    },
    Succeeded,
    Failed,
    Blocked,
}

pub struct Pool {
    jobs: Vec<Job>,
    states: Vec<InternalState>,
    dependencies: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    first_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|tag| !valid_token(tag, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if ids.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            ids.insert(job.id.clone(), i);
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dep) {
                    Some(&idx) => dependencies[i].push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        let mut marks = vec![0u8; jobs.len()];
        fn visit(node: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &d in &deps[node] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|job| InternalState::Waiting {
                ready_at: job.release_at,
                attempt: 0,
                remaining: None,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            dependencies,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let idx = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;
        Ok(match self.states[idx] {
            InternalState::Waiting {
                ready_at, attempt, ..
            } => State::Waiting { ready_at, attempt },
            InternalState::Running {
                attempt, finish_at, ..
            } => State::Running { attempt, finish_at },
            InternalState::Succeeded => State::Succeeded,
            InternalState::Failed => State::Failed,
            InternalState::Blocked => State::Blocked,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if !self.first_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut now = if self.first_run { 0 } else { self.horizon };
        let include_now = self.first_run;

        loop {
            if !include_now && now == self.horizon {
                // Previous horizon was fully processed; move only to a future event.
            } else {
                self.process_timestamp(now, &mut lines);
            }

            let next = self.next_event(now);
            match next {
                Some(t) if t <= until => {
                    now = t;
                    self.process_timestamp(now, &mut lines);
                    // Avoid processing this timestamp twice on loop entry.
                    let next_after = self.next_event(now);
                    if let Some(t2) = next_after.filter(|&t2| t2 <= until) {
                        now = t2;
                        self.process_timestamp(now, &mut lines);
                        loop {
                            let n = self.next_event(now);
                            if let Some(nt) = n.filter(|&nt| nt <= until) {
                                now = nt;
                                self.process_timestamp(now, &mut lines);
                            } else {
                                break;
                            }
                        }
                    }
                    break;
                }
                _ => break,
            }
        }

        self.horizon = until;
        self.first_run = false;
        Ok(lines)
    }

    fn next_event(&self, now: u64) -> Option<u64> {
        self.states
            .iter()
            .filter_map(|state| match state {
                InternalState::Running { finish_at, .. } if *finish_at > now => Some(*finish_at),
                InternalState::Waiting { ready_at, .. } if *ready_at > now => Some(*ready_at),
                _ => None,
            })
            .min()
    }

    fn process_timestamp(&mut self, now: u64, lines: &mut Vec<String>) {
        let mut completed: Vec<usize> = self
            .states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| match state {
                InternalState::Running { finish_at, .. } if *finish_at == now => Some(i),
                _ => None,
            })
            .collect();
        completed.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));

        for i in completed {
            let attempt = match self.states[i] {
                InternalState::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let success = attempt > self.jobs[i].failures;
            if success {
                self.states[i] = InternalState::Succeeded;
                lines.push(format!("{now} finish {} {attempt} ok", self.jobs[i].id));
            } else {
                lines.push(format!("{now} finish {} {attempt} fail", self.jobs[i].id));
                if attempt >= self.jobs[i].max_attempts {
                    self.states[i] = InternalState::Failed;
                } else {
                    let shift = (attempt - 1) as u32;
                    let delay = self
                        .retry_base
                        .checked_shl(shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready_at = now.saturating_add(delay);
                    self.states[i] = InternalState::Waiting {
                        ready_at,
                        attempt,
                        remaining: None,
                    };
                    lines.push(format!("{now} retry {} {attempt} {ready_at}", self.jobs[i].id));
                }
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if matches!(self.states[i], InternalState::Waiting { .. })
                    && self.dependencies[i].iter().any(|&d| {
                        matches!(
                            self.states[d],
                            InternalState::Failed | InternalState::Blocked
                        )
                    })
                {
                    self.states[i] = InternalState::Blocked;
                    newly_blocked.push(i);
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));
        for i in newly_blocked {
            let dep = self.dependencies[i]
                .iter()
                .filter(|&&d| matches!(self.states[d], InternalState::Failed | InternalState::Blocked))
                .min_by_key(|&&d| self.jobs[d].id.as_str())
                .copied();
            if let Some(d) = dep {
                lines.push(format!("{now} skip {} {}", self.jobs[i].id, self.jobs[d].id));
            }
        }

        self.schedule(now, lines);
    }

    fn schedule(&mut self, now: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self
                .states
                .iter()
                .enumerate()
                .filter_map(|(i, state)| match state {
                    InternalState::Waiting { ready_at, .. }
                        if *ready_at <= now
                            && self.dependencies[i]
                                .iter()
                                .all(|&d| matches!(self.states[d], InternalState::Succeeded)) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                self.effective_priority(b, now)
                    .cmp(&self.effective_priority(a, now))
                    .then(a.cmp(&b))
            });

            let mut chosen = None;
            for candidate in candidates {
                if let Some((victim, remaining)) = self.find_victim(candidate, now) {
                    chosen = Some((candidate, Some((victim, remaining))));
                    break;
                }
                if self.can_start(candidate) {
                    chosen = Some((candidate, None));
                    break;
                }
            }

            let Some((candidate, victim)) = chosen else {
                break;
            };

            if let Some((victim, remaining)) = victim {
                lines.push(format!(
                    "{now} preempt {} by {}",
                    self.jobs[victim].id, self.jobs[candidate].id
                ));
                let attempt = match self.states[victim] {
                    InternalState::Running { attempt, .. } => attempt,
                    _ => 0,
                };
                self.states[victim] = InternalState::Waiting {
                    ready_at: now,
                    attempt,
                    remaining: Some(remaining),
                };
            }

            let (attempt, remaining) = match self.states[candidate].clone() {
                InternalState::Waiting {
                    attempt,
                    remaining: Some(rem),
                    ..
                } => (attempt, rem),
                InternalState::Waiting { attempt, .. } => {
                    (attempt.saturating_add(1), self.jobs[candidate].duration)
                }
                _ => continue,
            };
            let finish_at = now.saturating_add(remaining);
            self.states[candidate] = InternalState::Running {
                attempt,
                finish_at,
                started_at: now,
            };
            lines.push(format!(
                "{now} start {} {attempt} {remaining}",
                self.jobs[candidate].id
            ));
        }
    }

    fn effective_priority(&self, i: usize, now: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[i].priority;
        }
        let ready_at = match self.states[i] {
            InternalState::Waiting { ready_at, .. } => ready_at,
            _ => now,
        };
        let age = (now - ready_at) / self.aging_ms;
        (self.jobs[i].priority as u64 + age).min(9) as u8
    }

    fn can_start(&self, candidate: usize) -> bool {
        let running: Vec<usize> = self
            .states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| matches!(state, InternalState::Running { .. }).then_some(i))
            .collect();
        if running.len() >= self.capacity {
            return false;
        }
        self.tag_free(candidate, &running)
    }

    fn tag_free(&self, candidate: usize, running: &[usize]) -> bool {
        self.jobs[candidate].tag.as_ref().is_none_or(|tag| {
            running
                .iter()
                .all(|&i| self.jobs[i].tag.as_ref() != Some(tag))
        })
    }

    fn find_victim(&self, candidate: usize, now: u64) -> Option<(usize, u64)> {
        let running: Vec<usize> = self
            .states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| matches!(state, InternalState::Running { .. }).then_some(i))
            .collect();
        let candidate_tag = self.jobs[candidate].tag.as_ref();
        let tag_occupied = candidate_tag.is_some_and(|tag| {
            running
                .iter()
                .any(|&i| self.jobs[i].tag.as_ref() == Some(tag))
        });
        let slot_needed = running.len() >= self.capacity;
        if !slot_needed && !tag_occupied {
            return None;
        }

        let mut victims: Vec<(usize, u64, u64)> = running
            .iter()
            .filter_map(|&i| {
                let InternalState::Running {
                    finish_at,
                    started_at,
                    ..
                } = self.states[i]
                else {
                    return None;
                };
                if self.jobs[candidate].priority < self.jobs[i].priority.saturating_add(2) {
                    return None;
                }
                let removes_slot = slot_needed;
                let removes_tag = tag_occupied
                    && candidate_tag.is_some_and(|tag| self.jobs[i].tag.as_ref() == Some(tag));
                if (slot_needed && !removes_slot) || (tag_occupied && !removes_tag) {
                    return None;
                }
                Some((i, finish_at - now, started_at))
            })
            .collect();

        victims.sort_by(|a, b| {
            self.jobs[a.0]
                .priority
                .cmp(&self.jobs[b.0].priority)
                .then(b.2.cmp(&a.2))
                .then_with(|| self.jobs[b.0].id.cmp(&self.jobs[a.0].id))
        });
        victims.first().map(|&(i, remaining, _)| (i, remaining))
    }
}

fn valid_token(value: &str, min_len: usize, max_len: usize) -> bool {
    let bytes = value.as_bytes();
    (min_len..=max_len).contains(&bytes.len())
        && bytes.iter().all(|b| {
            b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-'
        })
}
