#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
enum InternalState {
    Waiting {
        ready_at: u64,
        attempt: u8,
        remaining: Option<u64>,
    },
    Running {
        attempt: u8,
        finish_at: u64,
        started_at: u64,
    },
    Succeeded,
    Failed,
    Blocked,
}

pub struct Pool {
    jobs: Vec<Job>,
    states: Vec<InternalState>,
    dependencies: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_tag(tag) {
                    return Err(Error::BadTag(i));
                }
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&dep_index) = ids.get(dep) else {
                    return Err(Error::MissingDependency(i, j));
                };
                dependencies[i].push(dep_index);
            }
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|job| InternalState::Waiting {
                ready_at: job.release_at,
                attempt: 0,
                remaining: None,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            dependencies,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let Some(index) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        Ok(match self.states[index] {
            InternalState::Waiting {
                ready_at, attempt, ..
            } => State::Waiting { ready_at, attempt },
            InternalState::Running {
                attempt, finish_at, ..
            } => State::Running { attempt, finish_at },
            InternalState::Succeeded => State::Succeeded,
            InternalState::Failed => State::Failed,
            InternalState::Blocked => State::Blocked,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.has_run {
            self.horizon
        } else {
            0
        };
        let mut process_current = !self.has_run;

        loop {
            if process_current {
                let mut completed: Vec<usize> = (0..self.jobs.len())
                    .filter(|&i| {
                        matches!(
                            self.states[i],
                            InternalState::Running { finish_at, .. } if finish_at == time
                        )
                    })
                    .collect();
                completed.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));

                for i in completed {
                    let attempt = match self.states[i] {
                        InternalState::Running { attempt, .. } => attempt,
                        _ => unreachable!(),
                    };
                    let ok = attempt > self.jobs[i].failures;
                    lines.push(format!(
                        "{} finish {} {} {}",
                        time,
                        self.jobs[i].id,
                        attempt,
                        if ok { "ok" } else { "fail" }
                    ));
                    if ok {
                        self.states[i] = InternalState::Succeeded;
                    } else if attempt >= self.jobs[i].max_attempts {
                        self.states[i] = InternalState::Failed;
                    } else {
                        let shift = u32::from(attempt.saturating_sub(1));
                        let delay = self
                            .retry_base
                            .checked_shl(shift)
                            .unwrap_or(u64::MAX)
                            .min(self.retry_cap);
                        let ready_at = time.saturating_add(delay);
                        lines.push(format!(
                            "{} retry {} {} {}",
                            time, self.jobs[i].id, attempt, ready_at
                        ));
                        self.states[i] = InternalState::Waiting {
                            ready_at,
                            attempt,
                            remaining: None,
                        };
                    }
                }

                loop {
                    let mut newly_blocked = Vec::new();
                    for i in 0..self.jobs.len() {
                        if !matches!(self.states[i], InternalState::Waiting { .. }) {
                            continue;
                        }
                        if self.dependencies[i].iter().any(|&d| {
                            matches!(
                                self.states[d],
                                InternalState::Failed | InternalState::Blocked
                            )
                        }) {
                            self.states[i] = InternalState::Blocked;
                            newly_blocked.push(i);
                        }
                    }
                    if newly_blocked.is_empty() {
                        break;
                    }
                    newly_blocked.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));
                    for i in newly_blocked {
                        let failed_dep = self.dependencies[i]
                            .iter()
                            .filter(|&&d| {
                                matches!(
                                    self.states[d],
                                    InternalState::Failed | InternalState::Blocked
                                )
                            })
                            .min_by_key(|&&d| self.jobs[d].id.as_str())
                            .copied();
                        if let Some(dep) = failed_dep {
                            lines.push(format!(
                                "{} skip {} {}",
                                time, self.jobs[i].id, self.jobs[dep].id
                            ));
                        }
                    }
                }

                loop {
                    let mut candidates: Vec<usize> = (0..self.jobs.len())
                        .filter(|&i| {
                            matches!(
                                self.states[i],
                                InternalState::Waiting { ready_at, .. } if ready_at <= time
                            ) && self.dependencies[i].iter().all(|&d| {
                                matches!(self.states[d], InternalState::Succeeded)
                            })
                        })
                        .collect();

                    candidates.sort_by(|&a, &b| {
                        let pa = effective_priority(
                            self.jobs[a].priority,
                            waiting_since(&self.states[a]),
                            time,
                            self.aging_ms,
                        );
                        let pb = effective_priority(
                            self.jobs[b].priority,
                            waiting_since(&self.states[b]),
                            time,
                            self.aging_ms,
                        );
                        pb.cmp(&pa).then(a.cmp(&b))
                    });

                    let running: Vec<usize> = (0..self.jobs.len())
                        .filter(|&i| matches!(self.states[i], InternalState::Running { .. }))
                        .collect();

                    let mut selected = None;
                    for candidate in candidates {
                        let slot_free = running.len() < self.capacity;
                        let tag_free = self.jobs[candidate].tag.as_ref().is_none_or(|tag| {
                            !running.iter().any(|&r| self.jobs[r].tag.as_ref() == Some(tag))
                        });

                        if slot_free && tag_free {
                            selected = Some((candidate, None));
                            break;
                        }

                        let mut victims: Vec<usize> = running
                            .iter()
                            .copied()
                            .filter(|&victim| {
                                self.jobs[candidate].priority
                                    >= self.jobs[victim].priority.saturating_add(2)
                                    && (running.len() > self.capacity || slot_free)
                                    && self.jobs[candidate].tag.as_ref().is_some_and(|tag| {
                                        self.jobs[victim].tag.as_ref() == Some(tag)
                                    })
                            })
                            .collect();

                        victims.sort_by(|&a, &b| {
                            self.jobs[a]
                                .priority
                                .cmp(&self.jobs[b].priority)
                                .then_with(|| {
                                    let sa = match self.states[a] {
                                        InternalState::Running { started_at, .. } => started_at,
                                        _ => 0,
                                    };
                                    let sb = match self.states[b] {
                                        InternalState::Running { started_at, .. } => started_at,
                                        _ => 0,
                                    };
                                    sb.cmp(&sa)
                                })
                                .then_with(|| self.jobs[b].id.cmp(&self.jobs[a].id))
                        });

                        if let Some(victim) = victims.first().copied() {
                            let remaining = match self.states[victim] {
                                InternalState::Running { finish_at, .. } => finish_at - time,
                                _ => 0,
                            };
                            let attempt = match self.states[victim] {
                                InternalState::Running { attempt, .. } => attempt,
                                _ => 0,
                            };
                            selected = Some((candidate, Some((victim, attempt, remaining))));
                            break;
                        }
                    }

                    let Some((candidate, preempt)) = selected else {
                        break;
                    };

                    if let Some((victim, attempt, remaining)) = preempt {
                        lines.push(format!(
                            "{} preempt {} by {}",
                            time, self.jobs[victim].id, self.jobs[candidate].id
                        ));
                        self.states[victim] = InternalState::Waiting {
                            ready_at: time,
                            attempt,
                            remaining: Some(remaining),
                        };
                    }

                    let (attempt, duration) = match self.states[candidate] {
                        InternalState::Waiting {
                            attempt,
                            remaining,
                            ..
                        } => match remaining {
                            Some(rem) => (attempt, rem),
                            None => (attempt.saturating_add(1), self.jobs[candidate].duration),
                        },
                        _ => unreachable!(),
                    };
                    let finish_at = time.saturating_add(duration);
                    self.states[candidate] = InternalState::Running {
                        attempt,
                        finish_at,
                        started_at: time,
                    };
                    lines.push(format!(
                        "{} start {} {} {}",
                        time, self.jobs[candidate].id, attempt, duration
                    ));
                }
            }

            if time == until {
                break;
            }

            let mut next = None;
            for state in &self.states {
                let event = match state {
                    InternalState::Waiting { ready_at, .. } if *ready_at > time => Some(*ready_at),
                    InternalState::Running { finish_at, .. } if *finish_at > time => Some(*finish_at),
                    _ => None,
                };
                if let Some(event) = event {
                    next = Some(next.map_or(event, |n: u64| n.min(event)));
                }
            }
            let Some(next_time) = next.filter(|&t| t <= until) else {
                break;
            };
            time = next_time;
            process_current = true;
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }
}

fn valid_id(s: &str) -> bool {
    (1..=16).contains(&s.len())
        && s.bytes()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn valid_tag(s: &str) -> bool {
    (1..=8).contains(&s.len())
        && s.bytes()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn waiting_since(state: &InternalState) -> u64 {
    match state {
        InternalState::Waiting { ready_at, .. } => *ready_at,
        _ => 0,
    }
}

fn effective_priority(base: u8, since: u64, time: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        base
    } else {
        let aged = ((time - since) / aging_ms).min(9) as u8;
        base.saturating_add(aged).min(9)
    }
}
