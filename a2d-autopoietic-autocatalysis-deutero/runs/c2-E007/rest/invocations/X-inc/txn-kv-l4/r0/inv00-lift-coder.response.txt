use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
enum Data {
    Value { value: String, deadline: Option<u64> },
    Tombstone,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    data: Data,
}

#[derive(Clone)]
struct Write {
    data: Data,
}

struct Transaction {
    revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    versions: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, (String, Option<u64>)>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            versions: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn require_tx(&self, id: u64) -> Result<(), Error> {
        if self.transactions.contains_key(&id) {
            Ok(())
        } else {
            Err(Error::UnknownTransaction(id))
        }
    }

    fn validate_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            self.require_tx(id)?;
        }
        Ok(())
    }

    fn selected_data(&self, key: &str, tx: Option<u64>) -> Option<Data> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(write) = transaction.writes.get(key) {
                return Some(write.data.clone());
            }
            self.versions.get(key)?.iter()
                .rev()
                .find(|v| v.revision <= transaction.revision)
                .map(|v| v.data.clone())
        } else {
            self.versions.get(key)?.last().map(|v| v.data.clone())
        }
    }

    fn live_value(&self, key: &str, tx: Option<u64>) -> Option<String> {
        match self.selected_data(key, tx)? {
            Data::Value { value, deadline } if deadline.map_or(true, |d| d > self.clock) => {
                Some(value)
            }
            _ => None,
        }
    }

    fn visible(&self, tx: Option<u64>) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.versions.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(t) = self.transactions.get(&id) {
                keys.extend(t.writes.keys().cloned());
            }
        }
        keys.into_iter()
            .filter_map(|key| self.live_value(&key, tx).map(|value| (key, value)))
            .collect()
    }

    fn append(&mut self, key: String, data: Data, revision: u64) {
        self.versions.entry(key).or_default().push(Version { revision, data });
    }

    fn write(&mut self, tx: Option<u64>, key: String, data: Data) {
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key, Write { data });
        } else {
            self.revision += 1;
            self.append(key, data, self.revision);
        }
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(id, Transaction {
            revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        self.require_tx(id)?;
        let transaction = self.transactions.get(&id).unwrap();
        if !transaction.writes.is_empty() {
            for key in transaction.writes.keys() {
                if self.versions.get(key)
                    .and_then(|v| v.last())
                    .is_some_and(|v| v.revision > transaction.revision)
                {
                    return Err(Error::Conflict { tx: id, key: key.clone() });
                }
            }
            let writes = transaction.writes.iter()
                .map(|(k, w)| (k.clone(), w.data.clone()))
                .collect::<Vec<_>>();
            self.revision += 1;
            let revision = self.revision;
            for (key, data) in writes {
                self.append(key, data, revision);
            }
        }
        self.transactions.remove(&id);
        Ok(self.revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        self.require_tx(id)?;
        self.transactions.remove(&id);
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let data = match ttl {
            Some(0) => Data::Tombstone,
            Some(duration) => Data::Value {
                value: value.to_string(),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            },
            None => Data::Value { value: value.to_string(), deadline: None },
        };
        self.write(tx, key.to_string(), data);
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let existed = self.live_value(key, tx).is_some();
        self.write(tx, key.to_string(), Data::Tombstone);
        Ok(existed)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.live_value(key, tx))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Ok(self.visible(tx).values().filter(|v| v.as_str() == value).count())
    }

    fn range_matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::range_matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }
        let visible = self.visible(tx);
        let mut matching = visible.into_iter().filter(|(key, _)| {
            Self::range_matches(key, start, end, prefix)
                && cursor.map_or(true, |c| key.as_str() > c.last.as_str())
        });
        if cursor.is_none() {
            for _ in 0..offset {
                if matching.next().is_none() {
                    return Ok(Page { items: Vec::new(), next: None });
                }
            }
        }
        if limit == 0 {
            return Ok(Page { items: Vec::new(), next: None });
        }
        let mut items = Vec::new();
        for _ in 0..limit {
            match matching.next() {
                Some(item) => items.push(item),
                None => break,
            }
        }
        let next = if items.is_empty() {
            None
        } else if matching.next().is_some() {
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: items.last().unwrap().0.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        let mut result = Aggregate { count: 0, sum: 0 };
        for (key, value) in self.visible(tx) {
            if !Self::range_matches(&key, start, end, prefix) {
                continue;
            }
            let parsed = Self::parse_number(&key, &value)?;
            result.sum = result.sum.checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            result.count += 1;
        }
        Ok(result)
    }

    fn parse_number(key: &str, value: &str) -> Result<i64, Error> {
        let bytes = value.as_bytes();
        let valid = if bytes == b"0" {
            true
        } else if bytes.first() == Some(&b'-') {
            bytes.len() > 1
                && bytes[1] >= b'1'
                && bytes[1] <= b'9'
                && bytes[2..].iter().all(u8::is_ascii_digit)
        } else {
            !bytes.is_empty()
                && bytes[0] >= b'1'
                && bytes[0] <= b'9'
                && bytes[1..].iter().all(u8::is_ascii_digit)
        };
        if !valid {
            return Err(Error::InvalidNumber { key: key.to_string(), value: value.to_string() });
        }
        value.parse::<i64>().map_err(|_| Error::InvalidNumber {
            key: key.to_string(),
            value: value.to_string(),
        })
    }

    fn valid_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let data = self.visible(None).into_iter().filter_map(|(key, _)| {
            match self.selected_data(&key, None).unwrap() {
                Data::Value { value, deadline } => Some((key, (value, deadline))),
                Data::Tombstone => None,
            }
        }).collect();
        self.snapshots.insert(name.to_string(), data);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        Self::valid_name(name)?;
        let saved = self.snapshots.get(name)
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }
        let saved_live: BTreeMap<String, (String, Option<u64>)> = saved.iter()
            .filter(|(_, (_, deadline))| deadline.map_or(true, |d| d > now))
            .map(|(k, v)| (k.clone(), v.clone()))
            .collect();
        let current_live = self.visible(None);
        let keys: BTreeSet<String> = current_live.keys()
            .chain(saved_live.keys())
            .cloned()
            .collect();
        if keys.is_empty() {
            return Ok(self.revision);
        }
        self.revision += 1;
        let revision = self.revision;
        for key in keys {
            let data = match saved_live.get(&key) {
                Some((value, deadline)) => Data::Value {
                    value: value.clone(),
                    deadline: *deadline,
                },
                None => Data::Tombstone,
            };
            self.append(key, data, revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_name(name)?;
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.versions.values_mut() {
            let mut keep = BTreeSet::new();
            if let Some(i) = versions.len().checked_sub(1) {
                keep.insert(i);
            }
            for transaction in self.transactions.values() {
                if let Some(i) = versions.iter().rposition(|v| v.revision <= transaction.revision) {
                    keep.insert(i);
                }
            }
            let mut i = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&i);
                i += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let live_keys = self.visible(None).len();
        let versions = self.versions.values().map(Vec::len).sum();
        let tombstones = self.versions.values()
            .flat_map(|v| v.iter())
            .filter(|v| matches!(v.data, Data::Tombstone))
            .count();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.versions.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        })
    }
}
