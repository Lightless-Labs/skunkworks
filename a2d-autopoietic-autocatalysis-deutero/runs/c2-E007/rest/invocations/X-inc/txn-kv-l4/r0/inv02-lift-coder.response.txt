```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum StoredValue {
    Value {
        value: String,
        deadline: Option<u64>,
    },
    Tombstone,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Version {
    revision: u64,
    data: StoredValue,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Write {
    data: StoredValue,
}

#[derive(Debug, Clone)]
struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

#[derive(Debug, Clone)]
struct Snapshot {
    entries: BTreeMap<String, (String, Option<u64>)>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    versions: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, Snapshot>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            versions: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn validate_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn validate_key(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    fn live(data: &StoredValue, now: u64) -> Option<&str> {
        match data {
            StoredValue::Value { value, deadline }
                if deadline.is_none_or(|deadline| deadline > now) =>
            {
                Some(value)
            }
            _ => None,
        }
    }

    fn selected_version(&self, key: &str, tx: Option<u64>) -> Option<&Version> {
        let versions = self.versions.get(key)?;
        match tx {
            None => versions.last(),
            Some(id) => {
                let begin_revision = self.transactions.get(&id)?.begin_revision;
                versions
                    .iter()
                    .rev()
                    .find(|version| version.revision <= begin_revision)
            }
        }
    }

    fn visible_data(&self, key: &str, tx: Option<u64>) -> Option<&StoredValue> {
        if let Some(id) = tx {
            if let Some(write) = self
                .transactions
                .get(&id)
                .and_then(|transaction| transaction.writes.get(key))
            {
                return Some(&write.data);
            }
        }
        self.selected_version(key, tx).map(|version| &version.data)
    }

    fn visible_live(&self, key: &str, tx: Option<u64>, now: u64) -> Option<&str> {
        self.visible_data(key, tx)
            .and_then(|data| Self::live(data, now))
    }

    fn append_version(&mut self, key: String, data: StoredValue, revision: u64) {
        self.versions
            .entry(key)
            .or_default()
            .push(Version { revision, data });
    }

    fn matching_keys(
        &self,
        tx: Option<u64>,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Vec<(String, String)> {
        let mut keys: BTreeSet<String> = self.versions.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                keys.extend(transaction.writes.keys().cloned());
            }
        }

        keys.into_iter()
            .filter(|key| {
                key.as_str() >= start
                    && end.is_none_or(|end| key.as_str() < end)
                    && prefix.is_none_or(|prefix| key.starts_with(prefix))
            })
            .filter_map(|key| {
                self.visible_live(&key, tx, now)
                    .map(|value| (key, value.to_owned()))
            })
            .collect()
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;

        if !transaction.writes.is_empty() {
            let begin_revision = transaction.begin_revision;
            for key in transaction.writes.keys() {
                if self
                    .versions
                    .get(key)
                    .and_then(|versions| versions.last())
                    .is_some_and(|version| version.revision > begin_revision)
                {
                    return Err(Error::Conflict {
                        tx: id,
                        key: key.clone(),
                    });
                }
            }

            let writes = transaction.writes.clone();
            let revision = self.revision + 1;
            for (key, write) in writes {
                self.append_version(key, write.data, revision);
            }
            self.revision = revision;
        }

        self.transactions.remove(&id);
        Ok(self.revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if !self.transactions.contains_key(&id) {
            return Err(Error::UnknownTransaction(id));
        }
        self.transactions.remove(&id);
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_key(key)?;

        if let Some(id) = tx {
            if self.transactions.get(&id).is_some_and(|t| t.read_only) {
                return Err(Error::ReadOnly(id));
            }
        }

        let data = match ttl {
            Some(0) => StoredValue::Tombstone,
            Some(duration) => StoredValue::Value {
                value: value.to_owned(),
                deadline: Some(
                    now.checked_add(duration)
                        .ok_or(Error::DeadlineOverflow)?,
                ),
            },
            None => StoredValue::Value {
                value: value.to_owned(),
                deadline: None,
            },
        };

        match tx {
            Some(id) => {
                self.transactions
                    .get_mut(&id)
                    .expect("transaction validated")
                    .writes
                    .insert(key.to_owned(), Write { data });
            }
            None => {
                self.revision += 1;
                self.append_version(key.to_owned(), data, self.revision);
            }
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_key(key)?;

        if let Some(id) = tx {
            if self.transactions.get(&id).is_some_and(|t| t.read_only) {
                return Err(Error::ReadOnly(id));
            }
        }

        let existed = self.visible_live(key, tx, now).is_some();
        match tx {
            Some(id) => {
                self.transactions
                    .get_mut(&id)
                    .expect("transaction validated")
                    .writes
                    .insert(
                        key.to_owned(),
                        Write {
                            data: StoredValue::Tombstone,
                        },
                    );
            }
            None => {
                self.revision += 1;
                self.append_version(key.to_owned(), StoredValue::Tombstone, self.revision);
            }
        }
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_key(key)?;
        Ok(self
            .visible_live(key, tx, now)
            .map(str::to_owned))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Ok(self
            .matching_keys(tx, now, "", None, None)
            .iter()
            .filter(|(_, current)| current == value)
            .count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.is_some_and(|end| end <= start) {
            return Err(Error::InvalidRange);
        }

        if let Some(cursor) = cursor {
            if cursor.start != start
                || cursor.end.as_deref() != end
                || cursor.prefix.as_deref() != prefix
                || cursor.offset != offset
                || cursor.last.is_empty()
                || cursor.last < start
                || end.is_some_and(|end| cursor.last.as_str() >= end)
                || prefix.is_some_and(|prefix| !cursor.last.starts_with(prefix))
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let all = self.matching_keys(tx, now, start, end, prefix);
        let candidates: Vec<_> = match cursor {
            Some(cursor) => all
                .into_iter()
                .filter(|(key, _)| key > &cursor.last)
                .collect(),
            None => all.into_iter().skip(offset).collect(),
        };

        let has_more = candidates.len() > limit;
        let items: Vec<_> = candidates.into_iter().take(limit).collect();
        let next = if has_more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };

        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.is_some_and(|end| end <= start) {
            return Err(Error::InvalidRange);
        }

        let entries = self.matching_keys(tx, now, start, end, prefix);
        let mut count = 0usize;
        let mut sum = 0i64;

        for (key, value) in entries {
            let parsed = Self::parse_number(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }

        Ok(Aggregate { count, sum })
    }

    fn parse_number(value: &str) -> Option<i64> {
        if value == "0" {
            return Some(0);
        }

        let digits = if let Some(rest) = value.strip_prefix('-') {
            if rest.is_empty() || rest.starts_with('0') {
                return None;
            }
            rest
        } else {
            if value.is_empty() || value.starts_with('0') {
                return None;
            }
            value
        };

        if !digits.bytes().all(|byte| byte.is_ascii_digit()) {
            return None;
        }
        value.parse::<i64>().ok()
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;

        let exists = self.snapshots.contains_key(name);
        if !exists && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }

        let entries = self
            .versions
            .iter()
            .filter_map(|(key, versions)| {
                let data = &versions.last()?.data;
                match data {
                    StoredValue::Value { value, deadline }
                        if deadline.is_none_or(|deadline| deadline > now) =>
                    {
                        Some((key.clone(), (value.clone(), *deadline)))
                    }
                    _ => None,
                }
            })
            .collect();

        self.snapshots
            .insert(name.to_owned(), Snapshot { entries });
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let snapshot = self
            .snapshots
            .get(name)
            .ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let saved_live: BTreeMap<String, (String, Option<u64>)> = snapshot
            .entries
            .iter()
            .filter(|(_, (_, deadline))| deadline.is_none_or(|deadline| *deadline > now))
            .map(|(key, (value, deadline))| {
                (key.clone(), (value.clone(), *deadline))
            })
            .collect();

        let current_live: BTreeSet<String> = self
            .versions
            .iter()
            .filter_map(|(key, versions)| {
                Self::live(&versions.last()?.data, now).map(|_| key.clone())
            })
            .collect();

        let union: BTreeSet<String> = current_live
            .into_iter()
            .chain(saved_live.keys().cloned())
            .collect();

        if union.is_empty() {
            return Ok(self.revision);
        }

        let revision = self.revision + 1;
        for key in union {
            let data = match saved_live.get(&key) {
                Some((value, deadline)) => StoredValue::Value {
                    value: value.clone(),
                    deadline: *deadline,
                },
                None => StoredValue::Tombstone,
            };
            self.append_version(key, data, revision);
        }
        self.revision = revision;
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if !self.snapshots.contains_key(name) {
            return Err(Error::SnapshotMissing(name.to_owned()));
        }
        self.snapshots.remove(name);
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let begin_revisions: Vec<u64> = self
            .transactions
            .values()
            .map(|transaction| transaction.begin_revision)
            .collect();

        let mut removed = 0usize;
        for versions in self.versions.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for begin_revision in &begin_revisions {
                if let Some(index) = versions
                    .iter()
                    .rposition(|version| version.revision <= *begin_revision)
                {
                    keep.insert(index);
                }
            }

            let mut index = 0usize;
            versions.retain(|_| {
                let should_keep = keep.contains(&index);
                index += 1;
                if !should_keep {
                    removed += 1;
                }
                should_keep
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;

        let mut versions = 0usize;
        let mut tombstones = 0usize;
        let mut live_keys = 0usize;

        for key_versions in self.versions.values() {
            versions += key_versions.len();
            tombstones += key_versions
                .iter()
                .filter(|version| matches!(version.data, StoredValue::Tombstone))
                .count();
            if key_versions
                .last()
                .is_some_and(|version| Self::live(&version.data, now).is_some())
            {
                live_keys += 1;
            }
        }

        let read_only_transactions = self
            .transactions
            .values()
            .filter(|transaction| transaction.read_only)
            .count();
        let staged_writes = self
            .transactions
            .values()
            .map(|transaction| transaction.writes.len())
            .sum();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.versions.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}

impl Default for Store {
    fn default() -> Self {
        Self::new()
    }
}
```
