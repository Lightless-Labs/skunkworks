use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn render_absolute(parts: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if intent && !parts.is_empty() {
        result.push('/');
    }
    result
}

fn render_relative(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return ".".to_owned();
    }
    let mut result = parts.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce_plain(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    let mut stack: Vec<String> = Vec::new();
    let mut offset = 0usize;

    for component in path.split('/') {
        let component_offset = offset;
        offset += component.len() + 1;
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|part| part != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at: component_offset });
            } else {
                stack.push("..".to_owned());
            }
        } else {
            stack.push(component.to_owned());
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let parts = reduce_plain(path, absolute)?;
    if absolute {
        Ok(render_absolute(&parts, intent))
    } else {
        Ok(render_relative(&parts, intent))
    }
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized)),
        Err(_) => Err(PathError::InvalidCwd),
    }
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            match normalize(path) {
                Ok(normalized) => Ok(Some(components(&normalized))),
                Err(_) => Err(PathError::InvalidHome),
            }
        }
    }
}

fn expand_tilde<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(Option<Vec<String>>, &'a str, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((None, path, 0));
    }
    if path == "~" || path.starts_with("~/") {
        let home = home.ok_or(PathError::MissingHome)?;
        let remainder = if path == "~" { "" } else { &path[2..] };
        let consumed = path.len() - remainder.len();
        return Ok((Some(home.to_vec()), remainder, consumed));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (tilde_start, request, request_offset) =
        expand_tilde(path, home_parts.as_deref())?;

    let (mut stack, request, request_offset) = if let Some(parts) = tilde_start {
        (parts, request, request_offset)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_parts, path, 0)
    };

    let mut offset = request_offset;
    for component in request.split('/') {
        let at = offset;
        offset += component.len() + 1;
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            stack.push(component.to_owned());
        }
    }
    Ok(render_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut indexes = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path)
            && link.path != "/";
        let valid_target = validate_raw(&link.target).is_ok();
        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = indexes.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indexes.insert(link.path.clone(), index);
    }
    Ok(indexes)
}

struct WorkComponent {
    value: String,
    at: usize,
}

fn target_components(target: &str) -> Vec<WorkComponent> {
    let mut result = Vec::new();
    let mut offset = 0usize;
    for component in target.split('/') {
        if !component.is_empty() {
            result.push(WorkComponent {
                value: component.to_owned(),
                at: offset,
            });
        }
        offset += component.len() + 1;
    }
    result
}

fn requested_components(path: &str, offset: usize) -> Vec<WorkComponent> {
    let mut result = Vec::new();
    let mut local = 0usize;
    for component in path.split('/') {
        if !component.is_empty() {
            result.push(WorkComponent {
                value: component.to_owned(),
                at: offset + local,
            });
        }
        local += component.len() + 1;
    }
    result
}

fn lookup_absolute(stack: &[String]) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    result
}

fn process_stream(
    initial: &mut Vec<String>,
    stream: Vec<WorkComponent>,
    links: &[Link],
    link_map: &HashMap<String, usize>,
    max_expansions: usize,
    expansion_count: &mut usize,
    active: &mut Vec<usize>,
) -> Result<(), PathError> {
    let mut pending = stream;
    while let Some(component) = pending.first().cloned() {
        pending.remove(0);
        match component.value.as_str() {
            "." => continue,
            ".." => {
                if initial.pop().is_none() {
                    return Err(PathError::Escape { at: component.at });
                }
            }
            _ => {
                initial.push(component.value);
                let key = lookup_absolute(initial);
                if let Some(&index) = link_map.get(&key) {
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if *expansion_count >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    *expansion_count += 1;

                    initial.pop();
                    active.push(index);
                    let target = &links[index].target;
                    let mut target_stack = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        initial.clone()
                    };
                    process_stream(
                        &mut target_stack,
                        target_components(target),
                        links,
                        link_map,
                        max_expansions,
                        expansion_count,
                        active,
                    )?;
                    *initial = target_stack;
                    active.pop();
                }
            }
        }
    }
    Ok(())
}

        max_expansions,

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {

    let home_parts = validate_home(home)?;
    let link_map = validate_links(links)?;
    resolve_links_validated(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        &link_map,
        max_expansions,
    )
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(left, right)| left == right)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

fn is_directory_result(path: &str) -> bool {
    path.ends_with('/')
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let base_resolved = resolve_links(base, cwd, home, links, max_expansions)?;
    let target_resolved = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = components(&base_resolved);
    let target_parts = components(&target_resolved);
    let target_directory = is_directory_result(&target_resolved);

    let mut candidates = Vec::new();
    candidates.push(route(&base_parts, &target_parts));

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let matched = components(&probe);
        if matched.len() > target_parts.len()
            || !matched
                .iter()
                .zip(target_parts.iter())
                .all(|(left, right)| left == right)
        {
            continue;
        }

        let mut candidate = route(&base_parts, &components(&link.path));
        let suffix = &target_parts[matched.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &base_resolved,
            home,
            links,
            max_expansions,
        ) {
            if components(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|left, right| {
        left.len()
            .cmp(&right.len())
            .then_with(|| left.as_bytes().cmp(right.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Empty)
}

    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/').find(|part| !part.is_empty()) == Some(".")
        || path.rsplit('/').find(|part| !part.is_empty()) == Some("..")
}

fn split_components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn reduce_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    offset: usize,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|part| part != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at: offset });
            } else {
                stack.push("..".to_string());
            }
        }
        _ => stack.push(component.to_string()),
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = stack.join("/");
    if absolute {
        result.insert(0, '/');
    }
    if result.is_empty() {
        result = if absolute { "/".into() } else { ".".into() };
    }
    if intent && result != "/" && result != "." && !result.ends_with('/') {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    let mut search = 0usize;
    for part in path.split('/') {
        if part.is_empty() {
            search += 1;
            continue;
        }
        let at = search;
        reduce_component(&mut stack, part, absolute, at)?;
        search += part.len() + 1;
    }
    Ok(render(&stack, absolute, intent))
}

fn validate_context(value: &str) -> Result<Vec<String>, PathError> {
    if !value.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    match normalize(value) {
        Ok(normalized) => Ok(split_components(&normalized)
            .into_iter()
            .map(str::to_string)
            .collect()),
        Err(_) => Err(PathError::InvalidCwd),
    }
}

fn validate_home(value: &str) -> Result<Vec<String>, PathError> {
    if !value.starts_with('/') {
        return Err(PathError::InvalidHome);
    }
    match normalize(value) {
        Ok(normalized) => Ok(split_components(&normalized)
            .into_iter()
            .map(str::to_string)
            .collect()),
        Err(_) => Err(PathError::InvalidHome),
    }
}

fn requested_parts<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<(bool, Vec<(String, usize)>), PathError> {
    validate_raw(path)?;
    let bytes = path.as_bytes();
    if bytes[0] == b'~' {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let h = home.ok_or(PathError::MissingHome)?;
            let start = if bytes.len() > 1 { 2 } else { 1 };
            let mut parts = Vec::new();
            let mut pos = start;
            for part in path[start..].split('/') {
                if !part.is_empty() {
                    parts.push((part.to_string(), pos));
                }
                pos += part.len() + 1;
            }
            return Ok((false, parts));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let absolute = bytes[0] == b'/';
    let mut parts = Vec::new();
    let mut pos = 0usize;
    for part in path.split('/') {
        if !part.is_empty() {
            parts.push((part.to_string(), pos));
        }
        pos += part.len() + 1;
    }
    Ok((absolute, parts))
}

fn render_absolute(stack: &[String], intent: bool) -> String {
    render(stack, true, intent)
}

fn resolve_core(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
) -> Result<String, PathError> {
    let intent = directory_intent(path);
    let (absolute, parts) = requested_parts(path, home)?;
    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        home.unwrap_or(&[]).to_vec()
    } else {
        cwd.to_vec()
    };
    for (part, at) in parts {
        reduce_component(&mut stack, &part, true, at)?;
    }
    Ok(render_absolute(&stack, intent))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd = validate_context(cwd)?;
    let home = match home {
        Some(value) => Some(validate_home(value)?),
        None => None,
    };
    resolve_core(path, &cwd, home.as_deref())
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut table = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();
        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        table.insert(link.path.clone(), (index, link.target.clone()));
    }
    Ok(table)
}

fn absolute_key(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    }
}

#[derive(Clone)]
struct WorkItem {
    component: String,
    offset: usize,
    source: String,
}

fn traversal_with_links(
    initial: Vec<String>,
    queue: Vec<WorkItem>,
    table: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    let mut pending: Vec<WorkItem> = queue.into_iter().rev().collect();
    let mut active: Vec<usize> = Vec::new();
    let mut expansion_count = 0usize;

    while let Some(item) = pending.pop() {
        match item.component.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: item.offset });
                }
            }
            _ => {
                stack.push(item.component.clone());
                if let Some((index, target)) = table.get(&absolute_key(&stack)) {
                    if active.contains(index) {
                        return Err(PathError::Loop { index: *index });
                    }
                    if expansion_count >= max_expansions {
                        return Err(PathError::Limit { index: *index });
                    }
                    expansion_count += 1;
                    stack.pop();
                    active.push(*index);

                    let mut target_parts = Vec::new();
                    for (component, offset) in target_components(target) {
                        target_parts.push(WorkItem {
                            component,
                            offset,
                            source: target.clone(),
                        });
                    }
                    pending.push(WorkItem {
                        component: String::new(),
                        offset: 0,
                        source: format!("\0finish:{index}"),
                    });
                    for target_part in target_parts.into_iter().rev() {
                        pending.push(target_part);
                    }
                }
            }
        }

        while pending.last().is_some_and(|next| next.source.starts_with("\0finish:")) {
            let finish = pending.pop().unwrap();
            let index = finish.source["\0finish:".len()..]
                .parse::<usize>()
                .unwrap();
            active.retain(|value| *value != index);
        }
    }

    Ok(stack)
}

fn target_components(target: &str) -> Vec<(String, usize)> {
    let mut out = Vec::new();
    let mut pos = 0usize;
    for part in target.split('/') {
        if !part.is_empty() {
            out.push((part.to_string(), pos));
        }
        pos += part.len() + 1;
    }
    out
}

fn make_queue(parts: Vec<(String, usize)>, source: &str) -> Vec<WorkItem> {
    parts
        .into_iter()
        .map(|(component, offset)| WorkItem {
            component,
            offset,
            source: source.to_string(),
        })
        .collect()
}

fn resolve_links_with_context(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
    table: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<String, PathError> {
    let intent = directory_intent(path);
    let (absolute, parts) = requested_parts(path, home)?;
    let initial = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        home.unwrap_or(&[]).to_vec()
    } else {
        cwd.to_vec()
    };
    let result = traversal_with_links(
        initial,
        make_queue(parts, path),
        table,
        max_expansions,
    )?;
    Ok(render_absolute(&result, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd = validate_context(cwd)?;
    let home = match home {
        Some(value) => Some(validate_home(value)?),
        None => None,
    };
    let table = validate_links(links)?;
    resolve_links_with_context(path, &cwd, home.as_deref(), &table, max_expansions)
}

fn components_of_absolute(path: &str) -> Vec<String> {
    split_components(path)
        .into_iter()
        .map(str::to_string)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = Vec::new();
    for _ in common..base.len() {
        parts.push("..".to_string());
    }
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

fn prefix_matches(prefix: &[String], full: &[String]) -> bool {
    prefix.len() <= full.len() && prefix.iter().zip(full.iter()).all(|(a, b)| a == b)
}

fn protect_tilde(candidate: String) -> String {
    if candidate.starts_with('~') {
        format!("./{candidate}")
    } else {
        candidate
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = components_of_absolute(&resolved_base);
    let target_components = components_of_absolute(&resolved_target);
    let target_dir = resolved_target.ends_with('/');
    let table = validate_links(links)?;

    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = resolve_links(
            &link.path,
            "/",
            None,
            links,
            max_expansions,
        );
        let Ok(probe) = probe else {
            continue;
        };
        let alias_components = components_of_absolute(&probe);
        if prefix_matches(&alias_components, &target_components) {
            let suffix = &target_components[alias_components.len()..];
            let mut candidate = route(
                &base_components,
                &components_of_absolute(&link.path),
            );
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();
    for candidate in candidates {
        let mut candidate = protect_tilde(candidate);
        if target_dir && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }
        let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) else {
            continue;
        };
        if components_of_absolute(&resolved) == target_components {
            valid.push(candidate);
        }
    }

    let _ = table;
    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Empty)
}
