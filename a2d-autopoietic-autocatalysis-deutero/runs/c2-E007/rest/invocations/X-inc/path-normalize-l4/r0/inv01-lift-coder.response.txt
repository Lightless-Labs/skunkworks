use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|last| last == "." || last == "..")
}

fn spell(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if absolute && parts.is_empty() {
        return "/".to_owned();
    }
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce(
    stack: &mut Vec<String>,
    path: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let mut offset = 0usize;
    for part in path.split('/') {
        let start = offset;
        offset += part.len() + 1;
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: offset_base + start,
                });
            } else {
                stack.push("..".to_owned());
            }
        } else {
            stack.push(part.to_owned());
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute)?;
    Ok(spell(&stack, absolute, intent))
}

fn validated_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized))
}

fn resolve_internal(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);
    let mut request = path;
    let mut offset_base = 0usize;
    let mut stack;

    if request.starts_with('~') {
        let rest = &request[1..];
        if !rest.is_empty() && !rest.starts_with('/') {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        stack = home_parts.ok_or(PathError::MissingHome)?;
        request = rest.strip_prefix('/').unwrap_or(rest);
        offset_base = 1 + usize::from(rest.starts_with('/'));
    } else if request.starts_with('/') {
        stack = Vec::new();
    } else {
        stack = cwd_parts;
    }

    reduce(&mut stack, request, offset_base, true)?;
    Ok(spell(&stack, true, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_internal(path, cwd, home)
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut map = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();
        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = map.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        map.insert(link.path.clone(), index);
    }
    Ok(map)
}

#[derive(Clone)]
struct WorkPart {
    value: String,
    at: usize,
}

fn make_work(path: &str, base: usize) -> Vec<WorkPart> {
    let mut result = Vec::new();
    let mut offset = 0usize;
    for part in path.split('/') {
        if !part.is_empty() {
            result.push(WorkPart {
                value: part.to_owned(),
                at: base + offset,
            });
        }
        offset += part.len() + 1;
    }
    result
}

fn stack_key(stack: &[String]) -> String {
    format!("/{}", stack.join("/"))
}

fn resolve_links_internal(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    let link_map = validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let mut request = path;
    let mut request_offset = 0usize;
    let mut stack;

    if request.starts_with('~') {
        let rest = &request[1..];
        if !rest.is_empty() && !rest.starts_with('/') {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        stack = home_parts.ok_or(PathError::MissingHome)?;
        let had_slash = rest.starts_with('/');
        request = rest.strip_prefix('/').unwrap_or(rest);
        request_offset = 1 + usize::from(had_slash);
    } else if request.starts_with('/') {
        stack = Vec::new();
    } else {
        stack = cwd_parts;
    }

    let mut queue = make_work(request, request_offset);
    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    while let Some(part) = queue.first().cloned() {
        queue.remove(0);
        if part.value == "." {
            continue;
        }
        if part.value == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at: part.at });
            }
            continue;
        }

        stack.push(part.value);
        let key = stack_key(&stack);
        if let Some(&index) = link_map.get(&key) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);

            let link = &links[index];
            stack.pop();
            if link.target.starts_with('/') {
                stack.clear();
            }
            let target_parts = make_work(&link.target, 0);
            let remaining = std::mem::take(&mut queue);
            queue = target_parts;
            queue.extend(remaining);

            // A completion marker keeps this link active through nested target expansion.
            queue.insert(
                target_parts.len(),
                WorkPart {
                    value: format!("\0LINK_END:{index}"),
                    at: usize::MAX,
                },
            );
        }

        if let Some(marker) = queue.first() {
            if marker.at == usize::MAX && marker.value.starts_with("\0LINK_END:") {
                let marker = queue.remove(0);
                let index = marker.value["\0LINK_END:".len()..]
                    .parse::<usize>()
                    .unwrap();
                if active.last() == Some(&index) {
                    active.pop();
                }
            }
        }
    }

    Ok(spell(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_internal(path, cwd, home, links, max_expansions)
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = components(&resolved_base);
    let target_parts = components(&resolved_target);
    let target_dir = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) else {
            continue;
        };
        let alias = components(&probe);
        if alias.len() > target_parts.len()
            || alias
                .iter()
                .zip(target_parts.iter())
                .any(|(a, b)| a != b)
        {
            continue;
        }

        let key_parts = components(&link.path);
        let mut candidate = route(&base_parts, &key_parts);
        let suffix = &target_parts[alias.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_dir && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }
        if let Ok(resolved) =
            resolve_links(&candidate, &resolved_base, home, links, max_expansions)
        {
            if components(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    valid
        .into_iter()
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())))
        .ok_or(PathError::Empty)
}
