use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone)]
struct Component {
    value: String,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn raw_components(path: &str, base: usize) -> Vec<Component> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push(Component {
            value: path[start..i].to_string(),
            at: base + start,
        });
    }
    result
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    raw_components(path, 0)
        .last()
        .is_some_and(|c| c.value == "." || c.value == "..")
}

fn render_absolute(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return "/".to_string();
    }
    let mut result = format!("/{}", stack.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn render_normalized(stack: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return render_absolute(stack, intent);
    }
    if stack.is_empty() {
        return ".".to_string();
    }
    let mut result = stack.join("/");
    if intent {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    for component in raw_components(path, 0) {
        match component.value.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|s| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component.at });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(component.value),
        }
    }
    Ok(render_normalized(&stack, absolute, intent))
}

fn canonical_components(path: &str) -> Vec<String> {
    raw_components(path, 0)
        .into_iter()
        .map(|c| c.value)
        .collect()
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(canonical_components(&normalized))
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let normalized = normalize(path).map_err(|_| PathError::InvalidHome)?;
            Ok(Some(canonical_components(&normalized)))
        }
    }
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a Vec<String>>,
) -> Result<(bool, Vec<Component>, bool), PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let mut request = path;
    let mut offset = 0usize;
    let mut expanded = false;

    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            expanded = true;
            if bytes.len() > 1 {
                request = &path[2..];
                offset = 2;
            } else {
                request = "";
                offset = 1;
            }
            return Ok((expanded, raw_components(request, offset), intent).map_with_home(home));
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    }

    let absolute = path.starts_with('/');
    Ok((absolute, raw_components(request, offset), intent))
}

trait HomeTuple {
    fn map_with_home(self, home: &[String]) -> (bool, Vec<Component>, bool, Vec<String>);
}

impl HomeTuple for (bool, Vec<Component>, bool) {
    fn map_with_home(self, home: &[String]) -> (bool, Vec<Component>, bool, Vec<String>) {
        (self.0, self.1, self.2, home.to_vec())
    }
}

fn request_parts(
    path: &str,
    home: Option<&Vec<String>>,
) -> Result<(bool, Vec<Component>, bool, Option<Vec<String>>), PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);

    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let remainder = if bytes.len() > 1 { &path[2..] } else { "" };
            let offset = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((
                false,
                raw_components(remainder, offset),
                intent,
                Some(home.clone()),
            ));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    let absolute = path.starts_with('/');
    Ok((
        absolute,
        raw_components(path, 0),
        intent,
        None,
    ))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd)?;
    let home_components = validate_home(home)?;
    let (absolute, components, intent, tilde_home) =
        request_parts(path, home_components.as_ref())?;

    let mut stack = if absolute {
        Vec::new()
    } else if let Some(home) = tilde_home {
        home
    } else {
        cwd_components
    };

    for component in components {
        match component.value.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: component.at });
                }
            }
            _ => stack.push(component.value),
        }
    }
    Ok(render_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut table = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        table.insert(link.path.clone(), (index, link.target.clone()));
    }
    Ok(table)
}

fn process_component(
    value: &str,
    at: usize,
    stack: &mut Vec<String>,
) -> Result<(), PathError> {
    match value {
        "." => {}
        ".." => {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        }
        _ => stack.push(value.to_string()),
    }
    Ok(())
}

fn traverse_links(
    components: Vec<Component>,
    mut stack: Vec<String>,
    table: &HashMap<String, (usize, String)>,
    active: &mut Vec<usize>,
    count: &mut usize,
    limit: usize,
) -> Result<Vec<String>, PathError> {
    let mut pending = components;
    while let Some(component) = pending.first().cloned() {
        pending.remove(0);
        if component.value == "." || component.value == ".." {
            process_component(&component.value, component.at, &mut stack)?;
            continue;
        }

        stack.push(component.value);
        let key = format!("/{}", stack.join("/"));
        if let Some((index, target)) = table.get(&key) {
            if active.contains(index) {
                return Err(PathError::Loop { index: *index });
            }
            if *count >= limit {
                return Err(PathError::Limit { index: *index });
            }
            *count += 1;

            // The matched link name is removed before a relative target is traversed.
            let mut target_stack = stack.clone();
            target_stack.pop();
            let absolute = target.starts_with('/');
            if absolute {
                target_stack.clear();
            }
            let target_components = raw_components(target, 0);
            active.push(*index);
            stack = traverse_links(
                target_components,
                target_stack,
                table,
                active,
                count,
                limit,
            )?;
            active.pop();
        }
    }
    Ok(stack)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd)?;
    let home_components = validate_home(home)?;
    let table = validate_links(links)?;
    let (absolute, components, intent, tilde_home) =
        request_parts(path, home_components.as_ref())?;

    let start = if absolute {
        Vec::new()
    } else if let Some(home) = tilde_home {
        home
    } else {
        cwd_components
    };
    let mut count = 0;
    let stack = traverse_links(
        components,
        start,
        &table,
        &mut Vec::new(),
        &mut count,
        max_expansions,
    )?;
    Ok(render_absolute(&stack, intent))
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_string(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

fn is_directory_result(path: &str) -> bool {
    path.ends_with('/')
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;

    let base_components = canonical_components(&resolved_base);
    let target_components = canonical_components(&resolved_target);
    let target_directory = is_directory_result(&resolved_target);
    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = match resolve_links(&link.path, "/", None, links, max_expansions) {
            Ok(value) => value,
            Err(_) => continue,
        };
        let alias_components = canonical_components(&probe);
        if alias_components.len() <= target_components.len()
            && alias_components
                .iter()
                .zip(target_components.iter())
                .all(|(a, b)| a == b)
        {
            let mut candidate = route(&base_components, &canonical_components(&link.path));
            let suffix = &target_components[alias_components.len()..];
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if canonical_components(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
