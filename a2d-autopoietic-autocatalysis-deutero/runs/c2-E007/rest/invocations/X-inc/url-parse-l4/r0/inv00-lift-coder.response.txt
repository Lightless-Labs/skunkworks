    InvalidByte { at: usize },
    InvalidScheme { at: usize },

    InvalidHost { at: usize },
    InvalidPort { at: usize },

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(err) => {
            let index = err.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(index).copied().unwrap_or(input.len()),
            })
        }
    }
}

fn component_safe(component: Component, byte: u8) -> bool {
    let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
    if unreserved {
        return true;
    }
    match component {
        Component::UserInfo => byte == b':',
        Component::Path => byte == b'/',
        Component::Query => matches!(byte, b'/' | b'?'),
        Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if component_safe(component, byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn map_codec_parse(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(
    component: Component,
    value: &str,
    base: usize,
) -> Result<(), ParseError> {
    decode_component(component, value)
        .map(|_| ())
        .map_err(|error| map_codec_parse(error, base))
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (index, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + index,
            });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
        {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }
    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let ats: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + ats[1],
        });
    }
    if ats.len() == 1 && ats[0] == 0 {
        return Err(ParseError::InvalidUserInfo { at: authority_start });
    }

    let (userinfo, host_start_offset) = if let Some(&at) = ats.first() {
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };
    let host_start = authority_start + host_start_offset;
    let host_port = &authority[host_start_offset..];

    let mut ipv6 = false;
    let (host, port, port_offset) = if host_port.starts_with('[') {
        let closing = match host_port.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(relative) => relative + 1,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &host_port[1..closing];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix = &host_port[closing + 1..];
        ipv6 = true;
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                host_start + closing + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + closing + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port, None)
    } else {
        let host_colon = host_port.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, parsed_port, parsed_port_offset) = match host_colon {
            Some(offset) => (
                &host_port[..offset],
                Some(parse_port(
                    &host_port[offset + 1..],
                    host_start + offset + 1,
                )?),
                Some(offset),
            ),
            None => (host_port, None, None),
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, byte) in raw_host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        (raw_host.to_ascii_lowercase(), parsed_port, parsed_port_offset)
    };
    let _ = port_offset;

    let tail = &input[authority_end..];
    let fragment_relative = tail.find('#');
    let before_fragment = fragment_relative.map(|i| &tail[..i]).unwrap_or(tail);
    let fragment = fragment_relative.map(|i| tail[i + 1..].to_owned());
    let query_relative = before_fragment.find('?');
    let path_raw = query_relative
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_relative.map(|i| before_fragment[i + 1..].to_owned());
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(info) = userinfo.as_deref() {
        validate_component(Component::UserInfo, info, authority_start)?;
    }
    if !path_raw.is_empty() {
        validate_component(Component::Path, path_raw, authority_end)?;
    }
    if let Some(query_value) = query.as_deref() {
        let query_base = authority_end + query_relative.unwrap_or(0) + 1;
        validate_component(Component::Query, query_value, query_base)?;
    }
    if let Some(fragment_value) = fragment.as_deref() {
        let fragment_base = authority_end + fragment_relative.unwrap_or(0) + 1;
        validate_component(Component::Fragment, fragment_value, fragment_base)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded[1..].split('/').collect();
    let final_navigation = decoded_segments
        .last()
        .map(|segment| *segment == "." || *segment == "..")
        .unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for (raw, decoded_segment) in raw_segments.iter().zip(decoded_segments.iter()) {
        if *decoded_segment == "." {
            continue;
        } else if *decoded_segment == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }
    if final_navigation {
        stack.push("");
    }

    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    Ok(result)
}

fn scan_reference_bytes(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn split_reference(input: &str) -> (&str, Option<&str>, Option<&str>) {
    let hash = input.find('#');
    let before_fragment = hash.map(|i| &input[..i]).unwrap_or(input);
    let fragment = hash.map(|i| &input[i + 1..]);
    let question = before_fragment.find('?');
    let path = question
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = question.map(|i| &before_fragment[i + 1..]);
    (path, query, fragment)
}

fn url_authority(url: &Url) -> String {
    let mut authority = String::new();
    if let Some(userinfo) = &url.userinfo {
        authority.push_str(userinfo);
        authority.push('@');
    }
    if url.ipv6 {
        authority.push('[');
        authority.push_str(&url.host);
        authority.push(']');
    } else {
        authority.push_str(&url.host);
    }
    if let Some(port) = url.port {
        authority.push(':');
        authority.push_str(&port.to_string());
    }
    authority
}

fn build_absolute(scheme: &str, authority: &str, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut result = String::new();
    result.push_str(scheme);
    result.push_str("://");
    result.push_str(authority);
    result.push_str(path);
    if let Some(query) = query {
        result.push('?');
        result.push_str(query);
    }
    if let Some(fragment) = fragment {
        result.push('#');
        result.push_str(fragment);
    }
    result
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_reference_bytes(reference).map_err(ResolveError::Reference)?;

    let first_delimiter = reference
        .char_indices()
        .find(|(_, c)| matches!(c, '/' | '?' | '#'))
        .map(|(i, _)| i)
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);
    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_path.clone(),
            reference_query.or(base_url.query.as_deref()),
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_path.rfind('/').unwrap_or(0);
            let mut value = base_path[..=slash].to_owned();
            value.push_str(reference_path);
            value
        };
        (merged, reference_query)
    };

    let authority = url_authority(&base_url);
    let constructed = build_absolute(
        &base_url.scheme,
        &authority,
        &selected_path,
        selected_query,
        reference_fragment,
    );
    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if byte == b' ' {
            output.push('+');
        } else if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut cursor = 0;

    for field in input.split('&') {
        let field_start = cursor;
        cursor += field.len() + 1;
        if field.is_empty() {
            continue;
        }

        let equal = field.find('=');
        let raw_key = equal.map(|i| &field[..i]).unwrap_or(field);
        let raw_value = equal.map(|i| &field[i + 1..]).unwrap_or("");
        let key_input = raw_key.replace('+', " ");
        let value_input = raw_value.replace('+', " ");

        let key = decode_component(Component::Query, &key_input).map_err(|error| match error {
            CodecError::BadEscape { at } => QueryError::BadEscape { at: field_start + at },
            CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: field_start + at },
        })?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value = decode_component(Component::Query, &value_input).map_err(|error| match error {
            CodecError::BadEscape { at } => {
                QueryError::BadEscape { at: field_start + raw_key.len() + equal.map(|_| 1).unwrap_or(0) + at }
            }
            CodecError::InvalidUtf8 { at } => {
                QueryError::InvalidUtf8 { at: field_start + raw_key.len() + equal.map(|_| 1).unwrap_or(0) + at }
            }
        })?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| *existing == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, collected)) = merged.iter_mut().find(|(existing, _)| existing == key) {
            collected.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = String::new();
    output.push_str(&url.scheme);
    output.push_str("://");

    if let Some(userinfo) = url.userinfo.as_deref() {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .map_err(|error| CanonicalError::Parse(map_codec_parse(error, 0)))?;
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }

    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host).expect("validated IPv6 address");
        output.push('[');
        output.push_str(&address.to_string());
        output.push(']');
    } else {
        output.push_str(&url.host);
    }

    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            output.push(':');
            output.push_str(&port.to_string());
        }
    }

    let segments: Vec<String> = path
        .split('/')
        .map(|segment| {
            decode_component(Component::Path, segment)
                .map(|decoded| encode_unreserved(&decoded))
                .unwrap_or_default()
        })
        .collect();
    output.push_str(&segments.join("/"));

    if let Some(map) = query_map {
        output.push('?');
        output.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = url.fragment.as_deref() {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|error| CanonicalError::Parse(map_codec_parse(error, 0)))?;
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}

fn encode_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}
