#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/');

    let mut components: Vec<(&str, usize)> = Vec::new();
    let mut last_nonempty: Option<&str> = None;

    for (start, component) in path
        .split('/')
        .scan(0usize, |offset, component| {
            let start = *offset;
            *offset += component.len() + 1;
            Some((start, component))
        })
    {
        if component.is_empty() {
            continue;
        }

        last_nonempty = Some(component);

        match component {
            "." => {}
            ".." => {
                if let Some(index) = components.iter().rposition(|(name, _)| *name != "..") {
                    components.remove(index);
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push(("..", start));
                }
            }
            _ => components.push((component, start)),
        }
    }

    let directory_intent =
        directory_intent || matches!(last_nonempty, Some(".") | Some(".."));

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    for (index, (component, _)) in components.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
