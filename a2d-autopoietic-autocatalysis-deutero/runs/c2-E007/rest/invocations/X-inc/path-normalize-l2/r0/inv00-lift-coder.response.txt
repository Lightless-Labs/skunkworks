#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn has_directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce(
    components: &mut Vec<String>,
    path: &str,
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let bytes = path.as_bytes();
    let mut pos = offset;

    while pos < bytes.len() {
        while pos < bytes.len() && bytes[pos] == b'/' {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let start = pos;
        while pos < bytes.len() && bytes[pos] != b'/' {
            pos += 1;
        }
        let component = &path[start..pos];

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|c| c != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push("..".to_string());
                }
            }
            _ => components.push(component.to_string()),
        }
    }
    Ok(())
}

fn assemble(components: Vec<String>, absolute: bool, directory_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", components.join("/"))
    } else if components.is_empty() {
        ".".to_string()
    } else {
        components.join("/")
    };

    if absolute && components.is_empty() {
        result = "/".to_string();
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory_intent = has_directory_intent(path);
    let mut components = Vec::new();

    reduce(&mut components, path, 0, absolute)?;
    Ok(assemble(components, absolute, directory_intent))
}

fn validated_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_string)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd)?;

    let home_components = match home {
        Some(home_path) => {
            if !home_path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(validated_context(home_path).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate_raw(path)?;

    let (absolute, mut components, offset) = if path.starts_with('/') {
        (true, Vec::new(), 0)
    } else if path == "~" || path.starts_with("~/") {
        let mut home_components = home_components.ok_or(PathError::MissingHome)?;
        let offset = if path.starts_with("~/") { 2 } else { 1 };
        (true, std::mem::take(&mut home_components), offset)
    } else if path.starts_with('~') {
        return Err(PathError::UnsupportedTilde { at: 1 });
    } else {
        (true, cwd_components, 0)
    };

    let directory_intent = has_directory_intent(path);
    reduce(&mut components, path, offset, absolute)?;
    Ok(assemble(components, true, directory_intent))
}
