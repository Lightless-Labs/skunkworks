#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((start, &path[start..i]));
    }
    result
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, component)| *component == "." || *component == "..")
}

fn append_result(
    parts: &[String],
    absolute: bool,
    wants_directory: bool,
) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };

    if result.is_empty() && absolute {
        result.push('/');
    }

    if wants_directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce(
    parts: &mut Vec<String>,
    path: &str,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, component) in components(path) {
        match component {
            "." => {}
            ".." => {
                if parts.last().is_some_and(|part| part != "..") {
                    parts.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    parts.push("..".to_string());
                }
            }
            name => parts.push(name.to_string()),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let wants_directory = directory_intent(path);
    let mut parts = Vec::new();
    reduce(&mut parts, path, absolute)?;

    Ok(append_result(&parts, absolute, wants_directory))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let normalized_cwd = if cwd.starts_with('/') {
        normalize(cwd).map_err(|_| PathError::InvalidCwd)?
    } else {
        return Err(PathError::InvalidCwd);
    };

    let normalized_home = match home {
        Some(value) => {
            if !value.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(normalize(value).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate_raw(path)?;

    let tilde_expansion = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            Some(bytes.len() > 1)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        None
    };

    let wants_directory = directory_intent(path);
    let mut parts: Vec<String>;
    let remainder: &str;

    if let Some(has_slash) = tilde_expansion {
        let home_path = normalized_home.ok_or(PathError::MissingHome)?;
        parts = components(&home_path)
            .into_iter()
            .map(|(_, component)| component.to_string())
            .collect();
        remainder = if has_slash { &path[2..] } else { "" };
    } else if path.starts_with('/') {
        parts = Vec::new();
        remainder = path;
    } else {
        parts = components(&normalized_cwd)
            .into_iter()
            .map(|(_, component)| component.to_string())
            .collect();
        remainder = path;
    }

    reduce(&mut parts, remainder, true)?;
    Ok(append_result(&parts, true, wants_directory))
}
