pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn has_directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce(
    components: &mut Vec<String>,
    path: &str,
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let bytes = path.as_bytes();
    let mut pos = offset;

    while pos < bytes.len() {
        while pos < bytes.len() && bytes[pos] == b'/' {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let start = pos;
        while pos < bytes.len() && bytes[pos] != b'/' {
            pos += 1;
        }
        let component = &path[start..pos];

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|c| c != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push("..".to_string());
                }
            }
            _ => components.push(component.to_string()),
        }
    }
    Ok(())
}

fn assemble(components: Vec<String>, absolute: bool, directory_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", components.join("/"))
    } else if components.is_empty() {
        ".".to_string()
    } else {
        components.join("/")
    };

    if absolute && components.is_empty() {
        result = "/".to_string();
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory_intent = has_directory_intent(path);
    let mut components = Vec::new();

    reduce(&mut components, path, 0, absolute)?;
    Ok(assemble(components, absolute, directory_intent))
}

fn validated_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_string)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd)?;

    let home_components = match home {
        Some(home_path) => {
            if !home_path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(validated_context(home_path).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    let (absolute, mut components, offset) = if path.starts_with('/') {
        (true, Vec::new(), 0)

    } else {
        (true, cwd_components, 0)
    };

    let directory_intent = has_directory_intent(path);
    reduce(&mut components, path, offset, absolute)?;
    Ok(assemble(components, true, directory_intent))
}

    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let mut start = None;

    for (i, b) in path.bytes().enumerate() {
        if b == b'/' {
            if let Some(s) = start.take() {
                result.push((s, &path[s..i]));
            }
        } else if start.is_none() {
            start = Some(i);
        }
    }

    if let Some(s) = start {
        result.push((s, &path[s..]));
    }
    result
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    components(path)
        .last()
        .is_some_and(|(_, c)| *c == "." || *c == "..")
}

fn reduce(
    path: &str,
    absolute: bool,
    initial: Vec<String>,
    skip_initial_bytes: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;

    for (at, component) in components(path) {
        if at < skip_initial_bytes {
            continue;
        }
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_string());
                }
            }
            ordinary => stack.push(ordinary.to_string()),
        }
    }

    Ok(stack)
}

fn join_result(absolute: bool, parts: Vec<String>, directory: bool) -> String {
    let joined = parts.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };

    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory = directory_intent(path);
    let parts = reduce(path, absolute, Vec::new(), 0)?;
    Ok(join_result(absolute, parts, directory))
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    let trimmed = normalized.trim_end_matches('/');
    if trimmed.is_empty() {
        Ok(Vec::new())
    } else {
        Ok(trimmed
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect())
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let expanded_tilde = if tilde {
        match bytes.get(1) {
            None => true,
            Some(b'/') => true,
            Some(_) => return Err(PathError::UnsupportedTilde { at: 1 }),
        }
    } else {
        false
    };

    let (absolute, initial, skip) = if expanded_tilde {
        let initial = home_parts.ok_or(PathError::MissingHome)?;
        (true, initial, if bytes.len() > 1 { 2 } else { 1 })
    } else if bytes.first() == Some(&b'/') {
        (true, Vec::new(), 0)
    } else {
        (true, cwd_parts, 0)
    };

    let directory = directory_intent(path);
    let parts = reduce(path, absolute, initial, skip)?;
    Ok(join_result(true, parts, directory))
}

fn reduce_components<'a, I>(
    components: I,
    stack: &mut Vec<&'a str>,
    absolute: bool,
) -> Result<(), PathError>
where
    I: IntoIterator<Item = (&'a str, usize)>,
{
    for (component, at) in components {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if stack.last().is_some_and(|last| *last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..");
            }
        } else {
            stack.push(component);
        }
    }
    Ok(())
}

fn assemble(stack: &[&str], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        return result;
    }
    if result == "." {
        return result;
    }
    if directory {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let directory = directory_intent(path);
    let mut stack = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        if !component.is_empty() {
            reduce_components(
                std::iter::once((component, offset)),
                &mut stack,
                absolute,
            )?;
        }
        offset += component.len() + 1;
    }

    Ok(assemble(&stack, absolute, directory))
}

fn normalized_absolute_components(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::Escape { at: 0 });
    }
    validate_raw(path).map_err(|_| PathError::Escape { at: 0 })?;

    let mut stack: Vec<String> = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        if !component.is_empty() && component != "." {

                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: offset });
                }
            } else {
                stack.push(component.to_owned());
            }
        }
        offset += component.len() + 1;
    }

    Ok(stack)
}

    let cwd_components =

        }
        offset += component.len() + 1;
    }

    let refs: Vec<&str> = stack.iter().map(String::as_str).collect();

    Empty,
    Nul { at: usize },
    Backslash { at: usize },

    InvalidHome,

}

    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });

        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((start, &path[start..i]));
    }
    result
}

}

    };

    let wants_directory = directory_intent(path);
    let mut parts: Vec<String>;
    let remainder: &str;

    if let Some(has_slash) = tilde_expansion {
        let home_path = normalized_home.ok_or(PathError::MissingHome)?;
        parts = components(&home_path)
            .into_iter()
            .map(|(_, component)| component.to_string())
            .collect();
        remainder = if has_slash { &path[2..] } else { "" };
    } else if path.starts_with('/') {
        parts = Vec::new();
        remainder = path;
    } else {
        parts = components(&normalized_cwd)
            .into_iter()
            .map(|(_, component)| component.to_string())
            .collect();
        remainder = path;
    }

    reduce(&mut parts, remainder, true)?;
    Ok(append_result(&parts, true, wants_directory))
}
