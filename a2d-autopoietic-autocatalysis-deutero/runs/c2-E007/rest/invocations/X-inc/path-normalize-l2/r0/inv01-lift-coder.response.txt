#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a, I>(
    components: I,
    stack: &mut Vec<&'a str>,
    absolute: bool,
) -> Result<(), PathError>
where
    I: IntoIterator<Item = (&'a str, usize)>,
{
    for (component, at) in components {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if stack.last().is_some_and(|last| *last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..");
            }
        } else {
            stack.push(component);
        }
    }
    Ok(())
}

fn assemble(stack: &[&str], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        return result;
    }
    if result == "." {
        return result;
    }
    if directory {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let directory = directory_intent(path);
    let mut stack = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        if !component.is_empty() {
            reduce_components(
                std::iter::once((component, offset)),
                &mut stack,
                absolute,
            )?;
        }
        offset += component.len() + 1;
    }

    Ok(assemble(&stack, absolute, directory))
}

fn normalized_absolute_components(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::Escape { at: 0 });
    }
    validate_raw(path).map_err(|_| PathError::Escape { at: 0 })?;

    let mut stack: Vec<String> = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        if !component.is_empty() && component != "." {
            if component == ".." {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: offset });
                }
            } else {
                stack.push(component.to_owned());
            }
        }
        offset += component.len() + 1;
    }

    Ok(stack)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components =
        normalized_absolute_components(cwd).map_err(|_| PathError::InvalidCwd)?;

    let home_components = match home {
        Some(home) => Some(
            normalized_absolute_components(home).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes[0] == b'~';
    let expanded = if tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        Some(home_components.as_ref().ok_or(PathError::MissingHome)?)
    } else {
        None
    };

    let absolute = !tilde && bytes[0] == b'/';
    let directory = directory_intent(path);
    let mut stack: Vec<String> = if absolute {
        Vec::new()
    } else if let Some(home) = expanded {
        home.clone()
    } else {
        cwd_components
    };

    let remainder_start = if tilde {
        if bytes.len() > 1 && bytes[1] == b'/' {
            2
        } else {
            bytes.len()
        }
    } else {
        0
    };

    let remainder = &path[remainder_start..];
    let mut offset = remainder_start;
    for component in remainder.split('/') {
        if !component.is_empty() && component != "." {
            if component == ".." {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: offset });
                }
            } else {
                stack.push(component.to_owned());
            }
        }
        offset += component.len() + 1;
    }

    let refs: Vec<&str> = stack.iter().map(String::as_str).collect();
    Ok(assemble(&refs, true, directory))
}
