#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(invalid_syntax(field, start));
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if step {
        let (_, max) = bounds(field);
        let width = max - bounds(field).0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; 256];
    let mut item_start = 0usize;
    let mut exact_star = false;

    for item_end in (0..=bytes.len())
        .filter(|&i| i == bytes.len() || bytes[i] == b',')
    {
        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(invalid_syntax(field, item_start));
        }

        let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
            (slash, Some(slash + 1))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            0
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            1
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen == 0
                || hyphen + 1 == base.len()
                || !base[..hyphen].iter().all(u8::is_ascii_digit)
                || !base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(invalid_syntax(field, item_start));
            }
            2
        } else {
            return Err(invalid_syntax(field, item_start));
        };

        if base_end == 0 {
            return Err(invalid_syntax(field, item_start));
        }

        if step_start.is_some_and(|start| {
            start >= item.len() || !item[start..].iter().all(u8::is_ascii_digit)
        }) {
            return Err(invalid_syntax(field, item_start));
        }

        let (min, max) = bounds(field);
        let (start, end, start_offset, end_offset) = match base_kind {
            0 => (min, max, item_start, item_start),
            1 => {
                let number = parse_number(item, 0, base.len(), field, false)?;
                (number, number, item_start, item_start)
            }
            _ => {
                let hyphen = hyphens[0];
                let first = parse_number(item, 0, hyphen, field, false)?;
                let second = parse_number(item, hyphen + 1, base.len(), field, false)?;
                if first > second {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
                (first, second, item_start, item_start + hyphen + 1)
            }
        };

        let step = if let Some(step_pos) = step_start {
            parse_number(item, step_pos, item.len(), field, true)?
        } else {
            1
        };

        let (selection_start, selection_end) = if step_start.is_some() {
            match base_kind {
                0 => (min, max),
                1 => (start, max),
                _ => (start, end),
            }
        } else {
            (start, end)
        };

        let mut value = selection_start;
        while value <= selection_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_start == 0 && item_end == bytes.len() && item == b"*" {
            exact_star = true;
        }

        let _ = (start_offset, end_offset);
        item_start = item_end + 1;
    }

    let values = selected
        .iter()
        .enumerate()
        .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
        .collect();

    Ok((values, exact_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed = Vec::with_capacity(5);
    for (text, field) in fields.iter().zip(field_kinds) {
        parsed.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: std::array::from_fn(|i| parsed[i].0.clone()),
        day_stars: [parsed[2].1, parsed[4].1],
    })
}
