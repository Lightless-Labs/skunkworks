        }

    }

    Ok(())

            return Err(ParseError::ReversedRange {
                field,
                offset: end_offset,
            });
        }

            return Err(ParseError::BadStep {
                field,
                offset: step_offset,

        Some(value)

        None

        (None, _) => (min, max),
        (Some(s), Some(e)) => (s, e),

                None => break,
            }
        }
    } else {
        for value in range_start..=range_end {
            selected[(value - min) as usize] = true;
        }
    }

    let _ = (start_offset, end_offset);
    Ok(())
}

fn parse_number(text: &str, offset: usize, field: Field) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn check_domain(value: u32, offset: usize, field: Field) -> Result<(), ParseError> {
    if value < field_min(field) as u32 || value > field_max(field) {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(())
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end == start + 1 && bytes[start] == b'0' {
        return Ok(0);
    }
    if bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });

            .checked_mul(10)

                field,
                offset: start,

    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    minimum: u32,
    maximum: u32,
) -> Result<(u32, u32, Option<u32>), ParseError> {
    let slash = bytes[item_start..item_end]
        .iter()
        .position(|&b| b == b'/')
        .map(|p| item_start + p);

    let base_end = slash.unwrap_or(item_end);
    let step_bounds = slash.map(|s| (s + 1, item_end));

    let (start_bounds, end_bounds) = if bytes[item_start] == b'*' {
        if base_end != item_start + 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        (None, None)
    } else {
        let hyphen = bytes[item_start..base_end]
            .iter()
            .position(|&b| b == b'-')
            .map(|p| item_start + p);

        match hyphen {
            Some(h) => {
                if h == item_start || h + 1 == base_end
                    || bytes[h + 1..base_end].contains(&b'-')
                    || bytes[item_start..h].contains(&b'*')
                    || bytes[h + 1..base_end].contains(&b'*')
                {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                (Some((item_start, h)), Some((h + 1, base_end)))
            }
            None => {
                if base_end == item_start
                    || bytes[item_start..base_end].contains(&b'*')
                {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                (Some((item_start, base_end)), None)
            }
        }
    };

    if let Some((s, e)) = step_bounds {
        if s == e {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        if bytes[s..e].contains(&b'/') || bytes[s..e].contains(&b'-') || bytes[s..e].contains(&b'*') {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    }

    let mut start = minimum;
    let mut end = maximum;
    if let Some((s, e)) = start_bounds {
        start = parse_number(bytes, s, e, field)?;
        if start < minimum || start > maximum {
            return Err(ParseError::OutOfRange { field, offset: s });
        }
        end = start;

        if let Some((es, ee)) = end_bounds {
            end = parse_number(bytes, es, ee, field)?;
            if end < minimum || end > maximum {
                return Err(ParseError::OutOfRange { field, offset: es });
            }
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: es,
                });
            }
        }
    }

        let value = parse_number(bytes, s, e, field)?;
        let width = maximum - minimum + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset: s });
        }
        Some(value)
    } else {
        None
    };

    if step.is_some() && start_bounds.is_some() && end_bounds.is_none() {
        end = maximum;
    }

    Ok((start, end, step))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_stars = [false; 2];

    for i in 0..5 {
        let field = fields[i];
        let text = parts[i];
        let bytes = text.as_bytes();
        let (minimum, maximum) = field_domain(field);

        for (offset, &byte) in bytes.iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte == b'*'
                || byte == b'-'
                || byte == b'/'
                || byte == b',')
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if i == 2 {
            day_stars[0] = text == "*";
        } else if i == 4 {
            day_stars[1] = text == "*";
        }

        let mut bitmap = vec![false; (maximum + 1) as usize];
        let mut item_start = 0;
        loop {
            let item_end = bytes[item_start..]
                .iter()
                .position(|&b| b == b',')
                .map(|p| item_start + p)
                .unwrap_or(bytes.len());

            if item_start == item_end {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: item_start,
                });
            }

            let (start, end, step) =
                parse_item(bytes, item_start, item_end, field, minimum, maximum)?;
            let stride = step.unwrap_or(1);
            let mut value = start;
            while value <= end {
                bitmap[value as usize] = true;
                match value.checked_add(stride) {
                    Some(next) => value = next,
                    None => break,
                }
            }

            if item_end == bytes.len() {
                break;
            }
            item_start = item_end + 1;
        }

        selected[i] = bitmap
            .iter()
            .enumerate()
            .filter_map(|(value, &enabled)| enabled.then_some(value as u8))
            .collect();
    }

    Ok(Cron {
        allowed: selected,
        day_stars,
    })
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(invalid_syntax(field, start));
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if step {
        let (_, max) = bounds(field);
        let width = max - bounds(field).0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; 256];
    let mut item_start = 0usize;
    let mut exact_star = false;

    for item_end in (0..=bytes.len())
        .filter(|&i| i == bytes.len() || bytes[i] == b',')
    {
        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(invalid_syntax(field, item_start));
        }

        let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
            (slash, Some(slash + 1))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            0
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            1
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen == 0
                || hyphen + 1 == base.len()
                || !base[..hyphen].iter().all(u8::is_ascii_digit)
                || !base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(invalid_syntax(field, item_start));
            }
            2
        } else {
            return Err(invalid_syntax(field, item_start));
        };

        if base_end == 0 {
            return Err(invalid_syntax(field, item_start));
        }

        if step_start.is_some_and(|start| {
            start >= item.len() || !item[start..].iter().all(u8::is_ascii_digit)
        }) {
            return Err(invalid_syntax(field, item_start));
        }

        let (min, max) = bounds(field);
        let (start, end, start_offset, end_offset) = match base_kind {
            0 => (min, max, item_start, item_start),
            1 => {
                let number = parse_number(item, 0, base.len(), field, false)?;
                (number, number, item_start, item_start)
            }
            _ => {
                let hyphen = hyphens[0];
                let first = parse_number(item, 0, hyphen, field, false)?;
                let second = parse_number(item, hyphen + 1, base.len(), field, false)?;
                if first > second {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
                (first, second, item_start, item_start + hyphen + 1)
            }
        };

        let step = if let Some(step_pos) = step_start {
            parse_number(item, step_pos, item.len(), field, true)?
        } else {
            1
        };

        let (selection_start, selection_end) = if step_start.is_some() {
            match base_kind {
                0 => (min, max),
                1 => (start, max),
                _ => (start, end),
            }
        } else {
            (start, end)
        };

        let mut value = selection_start;
        while value <= selection_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_start == 0 && item_end == bytes.len() && item == b"*" {
            exact_star = true;
        }

        let _ = (start_offset, end_offset);
        item_start = item_end + 1;
    }

    let values = selected
        .iter()
        .enumerate()
        .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
        .collect();

    Ok((values, exact_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed = Vec::with_capacity(5);
    for (text, field) in fields.iter().zip(field_kinds) {
        parsed.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: std::array::from_fn(|i| parsed[i].0.clone()),
        day_stars: [parsed[2].1, parsed[4].1],
    })
}
