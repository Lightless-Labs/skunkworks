#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    let error = |kind| kind;
    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (_, maximum) = bounds(field);
        let width = maximum - bounds(field).0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (minimum, maximum) = bounds(field);
        if value < minimum || value > maximum {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    let _ = error;
    Ok(value)
}

fn parse_field(
    text: &str,
    field: Field,
) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let day_star = matches!(field, Field::DayOfMonth | Field::DayOfWeek)
        && bytes == b"*";

    let mut selected = vec![false; (bounds(field).1 + 1) as usize];
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&byte| byte == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash = bytes[item_start..item_end]
            .iter()
            .position(|&byte| byte == b'/')
            .map(|relative| item_start + relative);

        let (base_end, step) = if let Some(slash_pos) = slash {
            if bytes[slash_pos + 1..item_end].contains(&b'/') {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (slash_pos, Some((slash_pos + 1, item_end)))
        } else {
            (item_end, None)
        };

        let base = &bytes[item_start..base_end];
        let (kind, first_start, first_end, second) = if base == b"*" {
            (0u8, 0usize, 0usize, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &byte)| (byte == b'-').then_some(i))
                .collect();

            match hyphens.as_slice() {
                [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                    (1, item_start, base_end, None)
                }
                [hyphen]
                    if *hyphen > 0
                        && *hyphen + 1 < base.len()
                        && base[..*hyphen].iter().all(u8::is_ascii_digit)
                        && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
                {
                    (
                        2,
                        item_start,
                        item_start + *hyphen,
                        Some((item_start + *hyphen + 1, base_end)),
                    )
                }
                _ => {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
            }
        };

        let (start_value, end_value) = match kind {
            0 => {
                let (minimum, maximum) = bounds(field);
                (minimum, maximum)
            }
            1 => {
                let value = parse_number(bytes, first_start, first_end, field, false)?;
                (value, value)
            }
            _ => {
                let start_value =
                    parse_number(bytes, first_start, first_end, field, false)?;
                let (end_start, end_end) = second.unwrap();
                let end_value = parse_number(bytes, end_start, end_end, field, false)?;
                if start_value > end_value {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_start,
                    });
                }
                (start_value, end_value)
            }
        };

        let step_value = if let Some((step_start, step_end)) = step {
            Some(parse_number(bytes, step_start, step_end, field, true)?)
        } else {
            None
        };

        let increment = step_value.unwrap_or(1);
        let mut value = if kind == 1 && step_value.is_none() {
            start_value
        } else {
            start_value
        };

        while value <= end_value {
            selected[value as usize] = true;
            if step_value.is_none() && kind == 1 {
                break;
            }
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    let values = selected
        .iter()
        .enumerate()
        .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
        .collect();

    Ok((values, day_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_types = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed = Vec::with_capacity(5);
    for (text, field) in fields.into_iter().zip(field_types) {
        parsed.push(parse_field(text, field)?);
    }

    let day_stars = [parsed[2].1, parsed[4].1];
    let selected = std::array::from_fn(|index| parsed[index].0.clone());

    Ok(Cron {
        selected,
        day_stars,
    })
}
