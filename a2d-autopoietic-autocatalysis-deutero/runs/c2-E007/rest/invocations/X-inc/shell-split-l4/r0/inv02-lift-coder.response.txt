use std::fmt::Write;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn append_text(word: &mut Word, text: &str, wildcard_active: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if wildcard_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn append_char(word: &mut Word, c: char, wildcard_active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if matches!(n, '"' | '\\' | '$' | '\n') {
                        if n != '\n' {
                            append_char(&mut word, n, false);
                        }
                        i = after_n;
                    } else {
                        append_char(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            if ch == '\\' {
                                j += ch.len_utf8();
                                if j < input.len() {
                                    let escaped = input[j..].chars().next().unwrap();
                                    j += escaped.len_utf8();
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += ch.len_utf8();
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(split_at) = contents.find(":-") {
                            (&contents[..split_at], Some(&contents[split_at + 2..]))
                        } else {
                            (contents, None)
                        };
                        let mut name_chars = name.chars();
                        let valid_name = name_chars.next().is_some_and(is_name_start)
                            && name_chars.all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match default {
                            Some(default_text) => match env_value(env, name) {
                                Some(v) if !v.is_empty() => v.to_string(),
                                _ => decode_default(default_text),
                            },
                            None => env_value(env, name).unwrap_or("").to_string(),
                        };
                        append_text(&mut word, &value, false);
                        active = true;
                        i = close + 1;
                    } else {
                        let name_start = next_i;
                        if name_start < input.len() {
                            let first = input[name_start..].chars().next().unwrap();
                            if is_name_start(first) {
                                let mut end = name_start + first.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if !is_name_continue(ch) {
                                        break;
                                    }
                                    end += ch.len_utf8();
                                }
                                let value = env_value(env, &input[name_start..end]).unwrap_or("");
                                append_text(&mut word, value, false);
                                active = true;
                                i = end;
                            } else {
                                append_char(&mut word, '$', false);
                                active = true;
                                i = next_i;
                            }
                        } else {
                            append_char(&mut word, '$', false);
                            active = true;
                            i = next_i;
                        }
                    }
                } else {
                    append_char(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '\n' {
                        i = after_n;
                    } else {
                        append_char(&mut word, n, false);
                        active = true;
                        i = after_n;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            if ch == '\\' {
                                j += ch.len_utf8();
                                if j < input.len() {
                                    let escaped = input[j..].chars().next().unwrap();
                                    j += escaped.len_utf8();
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += ch.len_utf8();
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(split_at) = contents.find(":-") {
                            (&contents[..split_at], Some(&contents[split_at + 2..]))
                        } else {
                            (contents, None)
                        };
                        let mut name_chars = name.chars();
                        let valid_name = name_chars.next().is_some_and(is_name_start)
                            && name_chars.all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match default {
                            Some(default_text) => match env_value(env, name) {
                                Some(v) if !v.is_empty() => v.to_string(),
                                _ => decode_default(default_text),
                            },
                            None => env_value(env, name).unwrap_or("").to_string(),
                        };
                        append_text(&mut word, &value, true);
                        active = true;
                        i = close + 1;
                    } else {
                        let name_start = next_i;
                        if name_start < input.len() {
                            let first = input[name_start..].chars().next().unwrap();
                            if is_name_start(first) {
                                let mut end = name_start + first.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if !is_name_continue(ch) {
                                        break;
                                    }
                                    end += ch.len_utf8();
                                }
                                let value = env_value(env, &input[name_start..end]).unwrap_or("");
                                append_text(&mut word, value, true);
                                active = true;
                                i = end;
                            } else {
                                append_char(&mut word, '$', true);
                                active = true;
                                i = next_i;
                            }
                        } else {
                            append_char(&mut word, '$', true);
                            active = true;
                            i = next_i;
                        }
                    }
                } else {
                    append_char(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric()
            || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%')
        {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        if let Some(best) = candidates
            .into_iter()
            .flatten()
            .min_by_key(|candidate| candidate.len())
        {
            encoded.push(best);
        } else {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push('\'');
                    fallback.push('\\');
                    fallback.push('\'');
                    fallback.push('\'');
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            encoded.push(fallback);
        }
    }

    let mut result = String::new();
    for (index, word) in encoded.iter().enumerate() {
        if index > 0 {
            result.push(' ');
        }
        result.push_str(word);
    }
    Ok(result)
}
