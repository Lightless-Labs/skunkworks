#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn lookup_env<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn push_scalar(word: &mut Word, c: char, active_glob: bool) {
    let pos = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(pos);
    }
}

fn push_string(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        push_scalar(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            }
        } else {
            result.push(c);
        }
    }

    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    push_scalar(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            push_scalar(&mut word, next, false);
                        }
                        i = after_next;
                    } else {
                        push_scalar(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '$' {
                    if next_i < input.len() {
                        let following = input[next_i..].chars().next().unwrap();
                        if following == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if is_name_start(following) {
                            let mut end = next_i;
                            while end < input.len() {
                                let name_c = input[end..].chars().next().unwrap();
                                if !is_name_continue(name_c) {
                                    break;
                                }
                                end += name_c.len_utf8();
                            }
                            let name = &input[next_i..end];
                            active = true;
                            if let Some(value) = lookup_env(name, env) {
                                push_string(&mut word, value, false);
                            }
                            i = end;
                        } else if following == '{' {
                            let dollar = i;
                            let content_start = next_i + 1;
                            let mut scan = content_start;
                            let mut closing = None;

                            while scan < input.len() {
                                let scan_c = input[scan..].chars().next().unwrap();
                                if scan_c == '\\' {
                                    scan += scan_c.len_utf8();
                                    if scan < input.len() {
                                        let escaped = input[scan..].chars().next().unwrap();
                                        scan += escaped.len_utf8();
                                    }
                                } else if scan_c == '}' {
                                    closing = Some(scan);
                                    break;
                                } else {
                                    scan += scan_c.len_utf8();
                                }
                            }

                            let close = closing.ok_or(SplitError::UnterminatedExpansion {
                                pos: dollar,
                            })?;
                            let contents = &input[content_start..close];

                            let mut name_end = 0;
                            let mut content_chars = contents.char_indices();
                            match content_chars.next() {
                                Some((_, first)) if is_name_start(first) => {
                                    name_end = first.len_utf8();
                                    for (offset, ch) in content_chars {
                                        if !is_name_continue(ch) {
                                            break;
                                        }
                                        name_end = offset + ch.len_utf8();
                                    }
                                }
                                _ => return Err(SplitError::InvalidExpansion { pos: dollar }),
                            }

                            let name = &contents[..name_end];
                            let suffix = &contents[name_end..];
                            let selected = if suffix.is_empty() {
                                lookup_env(name, env).unwrap_or("")
                            } else if let Some(default) = suffix.strip_prefix(":-") {
                                match lookup_env(name, env) {
                                    Some(value) if !value.is_empty() => value.to_owned(),
                                    _ => decode_default(default),
                                }
                            } else {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            };

                            active = true;
                            push_string(&mut word, &selected, false);
                            i = close + 1;
                        } else {
                            active = true;
                            push_scalar(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        active = true;
                        push_scalar(&mut word, '$', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else {
                    push_scalar(&mut word, c, false);
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        push_scalar(&mut word, next, false);
                        i = after_next;
                    }
                } else if c == '$' {
                    if next_i < input.len() {
                        let following = input[next_i..].chars().next().unwrap();
                        if following == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if is_name_start(following) {
                            let mut end = next_i;
                            while end < input.len() {
                                let name_c = input[end..].chars().next().unwrap();
                                if !is_name_continue(name_c) {
                                    break;
                                }
                                end += name_c.len_utf8();
                            }
                            active = true;
                            if let Some(value) = lookup_env(&input[next_i..end], env) {
                                push_string(&mut word, value, true);
                            }
                            i = end;
                        } else if following == '{' {
                            let dollar = i;
                            let content_start = next_i + 1;
                            let mut scan = content_start;
                            let mut closing = None;

                            while scan < input.len() {
                                let scan_c = input[scan..].chars().next().unwrap();
                                if scan_c == '\\' {
                                    scan += scan_c.len_utf8();
                                    if scan < input.len() {
                                        let escaped = input[scan..].chars().next().unwrap();
                                        scan += escaped.len_utf8();
                                    }
                                } else if scan_c == '}' {
                                    closing = Some(scan);
                                    break;
                                } else {
                                    scan += scan_c.len_utf8();
                                }
                            }

                            let close = closing.ok_or(SplitError::UnterminatedExpansion {
                                pos: dollar,
                            })?;
                            let contents = &input[content_start..close];

                            let mut name_end = 0;
                            let mut content_chars = contents.char_indices();
                            match content_chars.next() {
                                Some((_, first)) if is_name_start(first) => {
                                    name_end = first.len_utf8();
                                    for (offset, ch) in content_chars {
                                        if !is_name_continue(ch) {
                                            break;
                                        }
                                        name_end = offset + ch.len_utf8();
                                    }
                                }
                                _ => return Err(SplitError::InvalidExpansion { pos: dollar }),
                            }

                            let name = &contents[..name_end];
                            let suffix = &contents[name_end..];
                            let selected = if suffix.is_empty() {
                                lookup_env(name, env).unwrap_or("").to_owned()
                            } else if let Some(default) = suffix.strip_prefix(":-") {
                                match lookup_env(name, env) {
                                    Some(value) if !value.is_empty() => value.to_owned(),
                                    _ => decode_default(default),
                                }
                            } else {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            };

                            active = true;
                            push_string(&mut word, &selected, true);
                            i = close + 1;
                        } else {
                            active = true;
                            push_scalar(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        active = true;
                        push_scalar(&mut word, '$', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else {
                    active = true;
                    push_scalar(&mut word, c, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric()
            || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%')
        {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }

    let mut result = String::new();
    result.push('"');
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

fn fallback_candidate(word: &str) -> String {
    let mut result = String::new();
    result.push('\'');
    for c in word.chars() {
        if c == '\'' {
            result.push('\'');
            result.push('\\');
            result.push('\'');
            result.push('\'');
        } else {
            result.push(c);
        }
    }
    result.push('\'');
    result
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates: Vec<(usize, usize, String)> = Vec::new();

        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate.len(), 0, candidate));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate.len(), 1, candidate));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate.len(), 2, candidate));
        }

        let candidate = if let Some((_, _, chosen)) =
            candidates.into_iter().min_by_key(|(len, priority, _)| (*len, *priority))
        {
            chosen
        } else {
            fallback_candidate(word)
        };
        encoded.push(candidate);
    }

    Ok(encoded.join(" "))
}
