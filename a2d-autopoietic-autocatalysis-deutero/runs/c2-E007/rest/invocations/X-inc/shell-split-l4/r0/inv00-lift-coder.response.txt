use std::fmt::Write;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    Bare,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_text(word: &mut Word, text: &str, active_glob: bool) {
    for c in text.chars() {
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn append_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn braced_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let bytes = input.as_bytes();
    let mut close = dollar + 2;
    while close < bytes.len() {
        if bytes[close] == b'\\' {
            close += 1;
            if close < bytes.len() {
                let c = input[close..].chars().next().unwrap();
                close += c.len_utf8();
            }
        } else if bytes[close] == b'}' {
            break;
        } else {
            let c = input[close..].chars().next().unwrap();
            close += c.len_utf8();
        }
    }

    if close >= bytes.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let contents = &input[dollar + 2..close];
    let (name_end, default_start) = if let Some(i) = contents.find(":-") {
        (i, Some(i + 2))
    } else {
        (contents.len(), None)
    };

    let name = &contents[..name_end];
    let mut chars = name.chars();
    let valid_name = chars.next().is_some_and(is_name_start) && chars.all(is_name_continue);
    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    if let Some(start) = default_start {
        if !contents[..name_end].chars().all(is_name_continue)
            || contents[..name_end].is_empty()
        {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        let value = lookup(name, env).filter(|v| !v.is_empty());
        let result = match value {
            Some(value) => value.to_string(),
            None => decode_default(&contents[start..]),
        };
        Ok((close + 1, result))
    } else {
        Ok((close + 1, lookup(name, env).unwrap_or("").to_string()))
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::Bare;
    let mut quote_start = 0usize;
    let mut i = 0usize;
    let bytes = input.as_bytes();

    while i < bytes.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::Bare;
                    i += n;
                } else {
                    append_scalar(&mut word, c, false);
                    i += n;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::Bare;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= bytes.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_scalar(&mut word, next, false);
                        }
                        i += next.len_utf8();
                    } else {
                        append_scalar(&mut word, '\\', false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let (end, value) = braced_expansion(input, i, env)?;
                        append_text(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else if let Some(first) = input[i + n..].chars().next() {
                        if is_name_start(first) {
                            let mut end = i + n;
                            while end < bytes.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(&input[i + n..end], env).unwrap_or("");
                            append_text(&mut word, value, false);
                            active = true;
                            i = end;
                        } else {
                            append_scalar(&mut word, '$', false);
                            active = true;
                            i += n;
                        }
                    } else {
                        append_scalar(&mut word, '$', false);
                        active = true;
                        i += n;
                    }
                } else {
                    append_scalar(&mut word, c, false);
                    i += n;
                }
            }
            Quote::Bare => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = i;
                    active = true;
                    i += n;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = i;
                    active = true;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= bytes.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        append_scalar(&mut word, next, false);
                        active = true;
                    }
                    i += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let (end, value) = braced_expansion(input, i, env)?;
                        append_text(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else if let Some(first) = input[i + n..].chars().next() {
                        if is_name_start(first) {
                            let mut end = i + n;
                            while end < bytes.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(&input[i + n..end], env).unwrap_or("");
                            append_text(&mut word, value, true);
                            active = true;
                            i = end;
                        } else {
                            append_scalar(&mut word, '$', false);
                            active = true;
                            i += n;
                        }
                    } else {
                        append_scalar(&mut word, '$', false);
                        active = true;
                        i += n;
                    }
                } else {
                    append_scalar(&mut word, c, matches!(c, '*' | '?' | '['));
                    active = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::Bare => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut result = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|b| *b == 0) {
            return Err(JoinError::Nul { word: index, pos });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(value) = bare_candidate(word) {
            candidates.push((value, 0usize));
        }
        if let Some(value) = single_candidate(word) {
            candidates.push((value, 1usize));
        }
        if let Some(value) = double_candidate(word) {
            candidates.push((value, 2usize));
        }

        let chosen = if candidates.is_empty() {
            let mut value = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    value.push('\'');
                    value.push('\\');
                    value.push('\'');
                    value.push('\'');
                } else {
                    value.push(c);
                }
            }
            value.push('\'');
            value
        } else {
            candidates.sort_by_key(|(value, priority)| (value.len(), *priority));
            candidates.remove(0).0
        };
        encoded.push(chosen);
    }

    let mut result = String::new();
    for (i, item) in encoded.iter().enumerate() {
        if i > 0 {
            result.push(' ');
        }
        result.push_str(item);
    }
    Ok(result)
}
