#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn symbol_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn is_foreign(byte: u8, alphabet: Alphabet) -> bool {
    match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    }
}

fn is_ignored(byte: u8) -> bool {
    matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

fn encoded_symbol(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + value - 26,
        52..=61 => b'0' + value - 52,
        62 if alphabet == Alphabet::Standard => b'+',
        63 if alphabet == Alphabet::Standard => b'/',
        62 => b'-',
        _ => b'_',
    }
}

fn encode_raw(input: &[u8], options: Options) -> Vec<u8> {
    let mut out = Vec::new();
    let mut i = 0;
    while i + 3 <= input.len() {
        let a = input[i];
        let b = input[i + 1];
        let c = input[i + 2];
        out.push(encoded_symbol(a >> 2, options.alphabet));
        out.push(encoded_symbol(((a & 3) << 4) | (b >> 4), options.alphabet));
        out.push(encoded_symbol(((b & 15) << 2) | (c >> 6), options.alphabet));
        out.push(encoded_symbol(c & 63, options.alphabet));
        i += 3;
    }
    let rem = input.len() - i;
    if rem == 1 {
        let a = input[i];
        out.push(encoded_symbol(a >> 2, options.alphabet));
        out.push(encoded_symbol((a & 3) << 4, options.alphabet));
        if options.padding != Padding::Forbidden {
            out.extend_from_slice(b"==");
        }
    } else if rem == 2 {
        let a = input[i];
        let b = input[i + 1];
        out.push(encoded_symbol(a >> 2, options.alphabet));
        out.push(encoded_symbol(((a & 3) << 4) | (b >> 4), options.alphabet));
        out.push(encoded_symbol((b & 15) << 2, options.alphabet));
        if options.padding != Padding::Forbidden {
            out.push(b'=');
        }
    }
    out
}

fn wrap_symbols(symbols: &[u8], width: usize, ending: LineEnding) -> Vec<u8> {
    let mut out = Vec::new();
    for (i, &byte) in symbols.iter().enumerate() {
        if i > 0 && i % width == 0 {
            match ending {
                LineEnding::Lf => out.push(b'\n'),
                LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
            }
        }
        out.push(byte);
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let symbols = encode_raw(input, options);
    let bytes = match options.whitespace {
        Whitespace::Wrapped { width, ending } => wrap_symbols(&symbols, width, ending),
        _ => symbols,
    };
    Ok(String::from_utf8(bytes).unwrap())
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    symbols_in_row: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            pending: Vec::new(),
            symbols_in_row: 0,
            closed: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.pending.extend_from_slice(input);
        let complete = self.pending.len() / 3 * 3;
        let bytes = self.pending[..complete].to_vec();
        self.pending.drain(..complete);
        Ok(self.emit(&bytes))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let bytes = std::mem::take(&mut self.pending);
        Ok(self.emit(&bytes))
    }

    fn emit(&mut self, input: &[u8]) -> String {
        let symbols = encode_raw(input, self.options);
        let mut out = Vec::new();
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            for byte in symbols {
                if self.symbols_in_row == width {
                    match ending {
                        LineEnding::Lf => out.push(b'\n'),
                        LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
                    }
                    self.symbols_in_row = 0;
                }
                out.push(byte);
                self.symbols_in_row += 1;
            }
        } else {
            out.extend(symbols);
        }
        String::from_utf8(out).unwrap()
    }
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbols: usize,
    group: Vec<(u8, usize)>,
    terminal: bool,
    pending_second_pad: Option<usize>,
    output: Vec<u8>,
    sticky: Option<Error>,
    closed: bool,
    row_symbols: usize,
    wrap_state: u8,
    separator_start: usize,
    expect_symbol_after_separator: bool,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols: 0,
            group: Vec::new(),
            terminal: false,
            pending_second_pad: None,
            output: Vec::new(),
            sticky: None,
            closed: false,
            row_symbols: 0,
            wrap_state: 0,
            separator_start: 0,
            expect_symbol_after_separator: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut produced = Vec::new();
        for &byte in input {
            let current = self.offset;
            self.offset += 1;
            if let Err(error) = self.frame_byte(byte, current) {
                return self.fail(error);
            }
            if self.should_skip(byte) {
                continue;
            }
            if let Some(limit) = self.options.max_symbols {
                if self.symbols >= limit {
                    return self.fail(Error::Overlong { offset: current, limit });
                }
            }
            self.symbols += 1;
            if let Err(error) = self.process_symbol(byte, current, &mut produced) {
                return self.fail(error);
            }
        }
        Ok(produced)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut produced = Vec::new();
        if let Whitespace::Wrapped { ending, .. } = self.options.whitespace {
            if self.wrap_state == 1 {
                let error = Error::BadWrap { offset: self.separator_start };
                return self.fail(error);
            }
            if self.wrap_state == 2 {
                let error = Error::BadWrap { offset: self.separator_start };
                return self.fail(error);
            }
            let _ = ending;
            if self.expect_symbol_after_separator {
                let error = Error::BadWrap { offset: self.separator_start };
                return self.fail(error);
            }
        }
        if let Some(offset) = self.pending_second_pad {
            return self.fail(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            let len = self.group.len();
            if self.options.padding == Padding::Required {
                return self.fail(Error::Truncated {
                    offset: self.group[0].1,
                });
            }
            if len == 1 {
                return self.fail(Error::Truncated {
                    offset: self.group[0].1,
                });
            }
            if self.group.iter().any(|(b, _)| *b == b'=') {
                return self.fail(Error::BadPadding {
                    offset: self.group.iter().find(|(b, _)| *b == b'=').unwrap().1,
                });
            }
            if self.options.strict {
                let (v, offset) = self.group[len - 1];
                let mask = if len == 2 { 15 } else { 3 };
                if v & mask != 0 {
                    return self.fail(Error::NonCanonical { offset });
                }
            }
            self.emit_group(&mut produced);
        }
        self.closed = true;
        Ok(produced)
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn should_skip(&self, byte: u8) -> bool {
        match self.options.whitespace {
            Whitespace::IgnoreAscii => is_ignored(byte),
            _ => false,
        }
    }

    fn frame_byte(&mut self, byte: u8, offset: usize) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => {
                match self.wrap_state {
                    1 => {
                        if ending != LineEnding::CrLf || byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = 2;
                        self.expect_symbol_after_separator = true;
                        return Ok(());
                    }
                    2 => {
                        if byte == b'\r' || byte == b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = 0;
                        self.row_symbols = 0;
                        self.expect_symbol_after_separator = false;
                    }
                    _ => {}
                }
                if self.row_symbols == width {
                    self.separator_start = offset;
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.wrap_state = 2;
                            self.expect_symbol_after_separator = true;
                            return Ok(());
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.wrap_state = 1;
                            return Ok(());
                        }
                        _ => return Err(Error::BadWrap { offset }),
                    }
                }
                if byte == b'\r' || byte == b'\n' || is_ignored(byte) {
                    return Err(Error::BadWrap { offset });
                }
                self.row_symbols += 1;
            }
            _ => {}
        }
        Ok(())
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        produced: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal {
            return Err(Error::BadPadding { offset });
        }
        if let Some(pending) = self.pending_second_pad {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_second_pad = None;
            self.group.push((byte, offset));
            self.terminal = true;
            self.emit_group(produced);
            return Ok(());
        }
        let pos = self.group.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match pos {
                0 | 1 => Err(Error::BadPadding { offset }),
                2 => {
                    self.group.push((byte, offset));
                    self.pending_second_pad = Some(offset);
                    Ok(())
                }
                3 => {
                    self.group.push((byte, offset));
                    self.terminal = true;
                    self.emit_group(produced);
                    Ok(())
                }
                _ => Err(Error::BadPadding { offset }),
            }
        } else {
            let value = match symbol_value(byte, self.options.alphabet) {
                Some(value) => value,
                None if self.options.strict && is_foreign(byte, self.options.alphabet) => {
                    return Err(Error::MixedAlphabet { offset });
                }
                None => return Err(Error::InvalidByte { offset, byte }),
            };
            self.group.push((value, offset));
            if self.group.len() == 4 {
                self.emit_group(produced);
            }
            Ok(())
        }
    }

    fn emit_group(&mut self, produced: &mut Vec<u8>) {
        if self.group.len() < 2 {
            self.group.clear();
            return;
        }
        let values: Vec<u8> = self
            .group
            .iter()
            .map(|(v, _)| if *v == b'=' { 0 } else { *v })
            .collect();
        let mut bytes = Vec::new();
        bytes.push((values[0] << 2) | (values[1] >> 4));
        if self.group.len() >= 3 && self.group[2].0 != b'=' {
            bytes.push((values[1] << 4) | (values[2] >> 2));
        }
        if self.group.len() >= 4 && self.group[3].0 != b'=' {
            bytes.push((values[2] << 6) | values[3]);
        }
        if self.options.strict {
            if self.group.len() == 4 && self.group[2].0 == b'=' && values[1] & 15 != 0 {
                self.sticky = Some(Error::NonCanonical {
                    offset: self.group[1].1,
                });
            } else if self.group.len() == 4
                && self.group[3].0 == b'='
                && self.group[2].0 != b'='
                && values[2] & 3 != 0
            {
                self.sticky = Some(Error::NonCanonical {
                    offset: self.group[2].1,
                });
            }
        }
        produced.extend(bytes);
        self.group.clear();
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
