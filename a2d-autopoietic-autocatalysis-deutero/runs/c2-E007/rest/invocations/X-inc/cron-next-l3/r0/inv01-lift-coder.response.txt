use std::ops::RangeInclusive;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Clone, Copy)]
struct Domain {
    min: u32,
    max: u32,
}

impl Domain {
    fn width(self) -> u32 {
        self.max - self.min + 1
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: fn(Field, usize) -> ParseError) -> ParseError {
    kind(field, offset)
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(invalid_syntax(field, start));
    }

    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if step {
        if !(1..=domain(field).width()).contains(&value) {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let d = domain(field);
        if value < d.min || value > d.max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }
    Ok(value)
}

fn name_value(bytes: &[u8], field: Field) -> Option<u32> {
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => match upper.as_slice() {
            b"JAN" => Some(1),
            b"FEB" => Some(2),
            b"MAR" => Some(3),
            b"APR" => Some(4),
            b"MAY" => Some(5),
            b"JUN" => Some(6),
            b"JUL" => Some(7),
            b"AUG" => Some(8),
            b"SEP" => Some(9),
            b"OCT" => Some(10),
            b"NOV" => Some(11),
            b"DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_slice() {
            b"SUN" => Some(0),
            b"MON" => Some(1),
            b"TUE" => Some(2),
            b"WED" => Some(3),
            b"THU" => Some(4),
            b"FRI" => Some(5),
            b"SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn parse_atom(
    bytes: &[u8],
    start: usize,
    field: Field,
) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        return parse_number(bytes, start, field, false).map(|n| (n, true));
    }
    if let Some(value) = name_value(bytes, field) {
        return Ok((value, false));
    }
    Err(invalid_syntax(field, start))
}

fn item_structure(item: &[u8]) -> Option<(Option<&[u8]>, Option<&[u8]>, Option<&[u8]>)> {
    let mut slash = None;
    for (i, &b) in item.iter().enumerate() {
        if b == b'/' {
            if slash.is_some() {
                return None;
            }
            slash = Some(i);
        }
    }

    let (base, step) = match slash {
        Some(i) => (&item[..i], Some(&item[i + 1..])),
        None => (item, None),
    };
    if base.is_empty() || step.is_some_and(|s| s.is_empty()) {
        return None;
    }

    if base == b"*" {
        return Some((None, None, step));
    }
    if base.contains(&b'*') {
        return None;
    }

    let mut hyphen = None;
    for (i, &b) in base.iter().enumerate() {
        if b == b'-' {
            if hyphen.is_some() {
                return None;
            }
            hyphen = Some(i);
        }
    }

    match hyphen {
        Some(i) if i > 0 && i + 1 < base.len() => {
            Some((Some(&base[..i]), Some(&base[i + 1..]), step))
        }
        Some(_) => None,
        None => Some((Some(base), None, step)),
    }
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let (first, second, step_bytes) =
        item_structure(item).ok_or_else(|| invalid_syntax(field, item_offset))?;

    let is_star = first.is_none();
    let mut start = domain(field).min;
    let mut end = domain(field).max;

    if let Some(first_atom) = first {
        let (value, is_numeric) = parse_atom(first_atom, item_offset, field)?;
        start = value;
        end = value;

        if let Some(second_atom) = second {
            let second_offset = item_offset + first_atom.len() + 1;
            let (end_value, _) = parse_atom(second_atom, second_offset, field)?;
            end = end_value;

            // Numeric base and endpoints are validated before reversed-range checking.
            if is_numeric && start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: second_offset,
                });
            }
            if !is_numeric && start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: second_offset,
                });
            }
        } else if step_bytes.is_some() {
            end = domain(field).max;
        }
    }

    let step = if let Some(bytes) = step_bytes {
        let step_offset = item_offset + item.len() - bytes.len();
        parse_number(bytes, step_offset, field, true)?
    } else {
        1
    };

    if is_star {
        start = domain(field).min;
        end = domain(field).max;
    }

    let mut value = start;
    while value <= end {
        selected[(value - domain(field).min) as usize] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let field_order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let d = domain(field_order[i]);
        vec![false; d.width() as usize]
    });

    for (index, (&field, text)) in field_order.iter().zip(fields.iter()).enumerate() {
        for (offset, &byte) in text.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut item_start = 0;
        for item in text.as_bytes().split(|&b| b == b',') {
            parse_item(item, item_start, field, &mut sets[index])?;
            item_start += item.len() + 1;
        }
    }

    Ok(Cron {
        sets,
        day_of_month_unrestricted: fields[2] == "*",
        day_of_week_unrestricted: fields[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let d = domain(field);
        self.sets[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| {
                yes.then_some((d.min as usize + i) as u8)
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.contains(Field::Minute, at.minute as u32)
            || !self.contains(Field::Hour, at.hour as u32)
            || !self.contains(Field::Month, at.month as u32)
        {
            return false;
        }

        let dom = self.contains(Field::DayOfMonth, at.day as u32);
        let dow = self.contains(Field::DayOfWeek, weekday(at.year, at.month, at.day) as u32);

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn contains(&self, field: Field, value: u32) -> bool {
        let d = domain(field);
        if value < d.min || value > d.max {
            return false;
        }
        self.sets[field_index(field)][(value - d.min) as usize]
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return true;
    }
    false
}
