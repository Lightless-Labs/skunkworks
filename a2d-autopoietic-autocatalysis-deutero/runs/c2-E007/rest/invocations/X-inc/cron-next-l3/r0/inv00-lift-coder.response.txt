#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => match upper.as_slice() {
            b"JAN" => Some(1),
            b"FEB" => Some(2),
            b"MAR" => Some(3),
            b"APR" => Some(4),
            b"MAY" => Some(5),
            b"JUN" => Some(6),
            b"JUL" => Some(7),
            b"AUG" => Some(8),
            b"SEP" => Some(9),
            b"OCT" => Some(10),
            b"NOV" => Some(11),
            b"DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_slice() {
            b"SUN" => Some(0),
            b"MON" => Some(1),
            b"TUE" => Some(2),
            b"WED" => Some(3),
            b"THU" => Some(4),
            b"FRI" => Some(5),
            b"SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn is_atom(field: Field, bytes: &[u8]) -> bool {
    !bytes.is_empty()
        && (bytes.iter().all(u8::is_ascii_digit) || name_value(field, bytes).is_some())
}

fn parse_number(field: Field, bytes: &[u8], offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_endpoint(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        let value = parse_number(field, bytes, offset)?;
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok((value, true))
    } else if let Some(value) = name_value(field, bytes) {
        Ok((value, false))
    } else {
        unreachable!("syntax checked before endpoint parsing")
    }
}

fn parse_item(
    field: Field,
    item: &[u8],
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }
    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), Some(item_offset + slash + 1))
    } else {
        (item, None, None)
    };

    let (start_bytes, end_bytes, end_offset, is_star) = if base == b"*" {
        (None, None, None, true)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();
        if hyphens.len() > 1 {
            return Err(invalid_syntax(field, item_offset));
        }
        if let Some(&hyphen) = hyphens.first() {
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if !is_atom(field, left) || !is_atom(field, right) {
                return Err(invalid_syntax(field, item_offset));
            }
            (
                Some(left),
                Some(right),
                Some(item_offset + hyphen + 1),
                false,
            )
        } else {
            if !is_atom(field, base) {
                return Err(invalid_syntax(field, item_offset));
            }
            (Some(base), None, None, false)
        }
    };

    if let Some(step) = step_bytes {
        if step.is_empty() || !step.iter().all(u8::is_ascii_digit) {
            return Err(invalid_syntax(field, item_offset));
        }
    }

    let (mut start, mut end) = if is_star {
        bounds(field)
    } else {
        let (s, s_numeric) = parse_endpoint(field, start_bytes.unwrap(), item_offset)?;
        let e = if let Some(end_bytes) = end_bytes {
            let (e, e_numeric) =
                parse_endpoint(field, end_bytes, end_offset.unwrap())?;
            if s_numeric && e_numeric && s > e {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset.unwrap(),
                });
            }
            e
        } else {
            s
        };
        (s, e)
    };

    let step = if let Some(step_bytes) = step_bytes {
        let offset = step_offset.unwrap();
        let value = parse_number(field, step_bytes, offset)?;
        let (_, max) = bounds(field);
        let width = max - bounds(field).0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset });
        }
        if end_bytes.is_none() && !is_star {
            end = bounds(field).1;
        }
        value
    } else {
        if !is_star && end_bytes.is_none() {
            end = start;
        }
        1
    };

    while start <= end {
        selected[start as usize] = true;
        match start.checked_add(step) {
            Some(next) if next > start => start = next,
            _ => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (min, max) = bounds(FIELDS[i]);
        vec![false; (max + 1) as usize]
    });

    for (index, (&field, text)) in FIELDS.iter().zip(parts.iter()).enumerate() {
        let bytes = text.as_bytes();
        for (offset, &byte) in bytes.iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut item_start = 0;
        for (i, &byte) in bytes.iter().enumerate() {
            if byte == b',' {
                parse_item(field, &bytes[item_start..i], item_start, &mut selected[index])?;
                item_start = i + 1;
            }
        }
        parse_item(
            field,
            &bytes[item_start..],
            item_start,
            &mut selected[index],
        )?;
    }

    Ok(Cron {
        selected,
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = FIELDS.iter().position(|&f| f == field).unwrap();
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            current = next_minute(current).expect("supported range has a next minute");
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut result = 6u8;
    let mut y = 2000u16;
    let mut m = 1u8;
    let mut d = 1u8;
    while (y, m, d) < (year, month, day) {
        result = (result + 1) % 7;
        d += 1;
        if d > days_in_month(y, m) {
            d = 1;
            m += 1;
            if m > 12 {
                m = 1;
                y += 1;
            }
        }
    }
    result
}

fn next_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    let mut next = at;
    if next.minute < 59 {
        next.minute += 1;
    } else {
        next.minute = 0;
        if next.hour < 23 {
            next.hour += 1;
        } else {
            next.hour = 0;
            if next.day < days_in_month(next.year, next.month) {
                next.day += 1;
            } else {
                next.day = 1;
                if next.month < 12 {
                    next.month += 1;
                } else {
                    next.month = 1;
                    if next.year < 2399 {
                        next.year += 1;
                    } else {
                        return None;
                    }
                }
            }
        }
    }
    Some(next)
}
