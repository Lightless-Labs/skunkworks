```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
            cwd: Vec::new(),
        }
    }

    fn canonical_path(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_owned()),
            }
        }

        Ok(components)
    }

    fn entry_at<'a>(&'a self, components: &[String]) -> Result<&'a Entry, Error> {
        let mut entry = &self.root;
        let mut prefix = Vec::new();

        for (i, component) in components.iter().enumerate() {
            match entry {
                Entry::Dir(children) => {
                    prefix.push(component.clone());
                    entry = children
                        .get(component)
                        .ok_or_else(|| Error::Missing(Self::canonical_path(&prefix)))?;
                }
                Entry::File(_) => return Err(Error::NotDir(Self::canonical_path(&prefix))),
            }

            if i + 1 < components.len() && matches!(entry, Entry::File(_)) {
                return Err(Error::NotDir(Self::canonical_path(&prefix)));
            }
        }

        Ok(entry)
    }

    fn parent_at_mut<'a>(
        &'a mut self,
        components: &[String],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut entry = &mut self.root;
        let mut prefix = Vec::new();

        for component in &components[..components.len() - 1] {
            match entry {
                Entry::Dir(children) => {
                    prefix.push(component.clone());
                    entry = children
                        .get_mut(component)
                        .ok_or_else(|| Error::Missing(Self::canonical_path(&prefix)))?;
                    if matches!(entry, Entry::File(_)) {
                        return Err(Error::NotDir(Self::canonical_path(&prefix)));
                    }
                }
                Entry::File(_) => return Err(Error::NotDir(Self::canonical_path(&prefix))),
            }
        }

        match entry {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_path(
                &components[..components.len() - 1],
            ))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let name = components.last().unwrap().clone();
        let final_path = Self::canonical_path(&components);
        let parent = self.parent_at_mut(&components)?;
        if parent.contains_key(&name) {
            return Err(Error::Exists(final_path));
        }
        parent.insert(name, Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let name = components.last().unwrap().clone();
        let final_path = Self::canonical_path(&components);
        let parent = self.parent_at_mut(&components)?;

        match parent.get_mut(&name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(final_path)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        match self.entry_at(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical_path(&components))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        match self.entry_at(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_path(&components))),
        }
    }

    pub fn cwd(&self) -> String {
        Self::canonical_path(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        match self.entry_at(&components)? {
            Entry::Dir(_) => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(Self::canonical_path(&components))),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
