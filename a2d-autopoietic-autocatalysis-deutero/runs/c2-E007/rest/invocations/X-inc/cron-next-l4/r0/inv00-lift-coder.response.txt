use std::fmt::Write;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum Endpoint {
    Number(u32, usize),
    Name(u32),
}

fn parse_endpoint(s: &str, field: Field, item_start: usize) -> Result<Endpoint, ParseError> {
    if s.is_empty() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }
    if s.bytes().all(|b| b.is_ascii_digit()) {
        let start = item_start;
        if s.len() > 1 && s.as_bytes()[0] == b'0' {
            return Err(ParseError::LeadingZero {
                field,
                offset: start,
            });
        }
        let mut value = 0u32;
        for b in s.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow {
                    field,
                    offset: start,
                })?;
        }
        let (lo, hi) = bounds(field);
        if value < lo || value > hi {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
        return Ok(Endpoint::Number(value, start));
    }
    if let Some(table) = names(field) {
        if s.len() == 3 && s.bytes().all(|b| b.is_ascii_alphabetic()) {
            let upper = s.to_ascii_uppercase();
            if let Some(index) = table.iter().position(|n| *n == upper) {
                let value = if field == Field::Month {
                    index as u32 + 1
                } else {
                    index as u32
                };
                return Ok(Endpoint::Name(value));
            }
        }
    }
    Err(ParseError::InvalidSyntax {
        field,
        offset: item_start,
    })
}

fn parse_item(
    item: &str,
    field: Field,
    offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset });
    }

    let slash_count = item.bytes().filter(|b| *b == b'/').count();
    if slash_count > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    let (base, step_text) = if let Some(i) = item.find('/') {
        (&item[..i], Some(&item[i + 1..]))
    } else {
        (item, None)
    };

    let wildcard = base == "*";
    let hyphen_count = base.bytes().filter(|b| *b == b'-').count();
    if (!wildcard && (base.is_empty() || hyphen_count > 1))
        || (wildcard && hyphen_count != 0)
    {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    let endpoints = if wildcard {
        None
    } else if hyphen_count == 0 {
        Some((parse_endpoint(base, field, offset)?, None))
    } else {
        let i = base.find('-').unwrap();
        let left = &base[..i];
        let right = &base[i + 1..];
        if left.is_empty() || right.is_empty() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        Some((
            parse_endpoint(left, field, offset)?,
            Some(parse_endpoint(right, field, offset + i + 1)?),
        ))
    };

    let mut step = 1u32;
    if let Some(st) = step_text {
        if st.is_empty() || !st.bytes().all(|b| b.is_ascii_digit()) {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        let step_offset = offset + base.len() + 1;
        if st.len() > 1 && st.as_bytes()[0] == b'0' {
            return Err(ParseError::LeadingZero {
                field,
                offset: step_offset,
            });
        }
        let mut value = 0u32;
        for b in st.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow {
                    field,
                    offset: step_offset,
                })?;
        }
        let (_, hi) = bounds(field);
        if value == 0 || value > hi + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        step = value;
    }

    let (min, max) = bounds(field);
    let (start, end) = if wildcard {
        (min, max)
    } else {
        let (first, last) = endpoints.unwrap();
        let (start, end) = match (first, last) {
            (Endpoint::Number(a, _), None) => {
                if step_text.is_none() {
                    (a, a)
                } else {
                    (a, max)
                }
            }
            (Endpoint::Name(a), None) => {
                if step_text.is_none() {
                    (a, a)
                } else {
                    (a, max)
                }
            }
            (a, Some(b)) => {
                let av = match a {
                    Endpoint::Number(v, _) | Endpoint::Name(v) => v,
                };
                let bv = match b {
                    Endpoint::Number(v, _) | Endpoint::Name(v) => v,
                };
                if av > bv {
                    let end_offset = match b {
                        Endpoint::Number(_, o) => o,
                        Endpoint::Name(_) => offset + base.len(),
                    };
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
                (av, bv)
            }
        };
        (start, end)
    };

    let mut n = start;
    while n <= end {
        selected[(n - min) as usize] = true;
        match n.checked_add(step) {
            Some(next) => n = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.starts_with('@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (lo, hi) = bounds(FIELDS[i]);
        vec![false; (hi - lo + 1) as usize]
    });

    for (i, part) in parts.iter().enumerate() {
        let field = FIELDS[i];
        for (offset, b) in part.bytes().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }
        let mut offset = 0usize;
        for item in part.split(',') {
            parse_item(item, field, offset, &mut sets[i])?;
            offset += item.len() + 1;
        }
    }

    Ok(Cron {
        sets,
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = field_index(field);
        let (lo, _) = bounds(field);
        self.sets[i]
            .iter()
            .enumerate()
            .filter_map(|(n, yes)| yes.then_some(lo as u8 + n as u8))
            .collect()
    }

    fn selected(&self, field: Field, value: u8) -> bool {
        let (lo, _) = bounds(field);
        self.sets[field_index(field)][(value as u32 - lo) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected(Field::Minute, at.minute)
            || !self.selected(Field::Hour, at.hour)
            || !self.selected(Field::Month, at.month)
        {
            return false;
        }
        let dom = self.selected(Field::DayOfMonth, at.day);
        let dow = self.selected(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_len(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = (after.year, after.month, after.day);
        loop {
            let (y, m, d) = date;
            let start_hour = if (y, m, d) == (after.year, after.month, after.day) {
                after.hour
            } else {
                0
            };
            for hour in self.allowed(Field::Hour) {
                if hour < start_hour {
                    continue;
                }
                let start_minute = if (y, m, d) == (after.year, after.month, after.day)
                    && hour == after.hour
                {
                    after.minute.saturating_add(1)
                } else {
                    0
                };
                for minute in self.allowed(Field::Minute) {
                    if minute < start_minute {
                        continue;
                    }
                    let at = CivilDateTime {
                        year: y,
                        month: m,
                        day: d,
                        hour,
                        minute,
                    };
                    if at > after && self.matches_valid(at) {
                        return Ok(Some(at));
                    }
                }
            }
            date = match next_date(y, m, d) {
                Some(v) => v,
                None => return Ok(None),
            };
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = (before.year, before.month, before.day);
        loop {
            let (y, m, d) = date;
            let start_hour = if (y, m, d) == (before.year, before.month, before.day) {
                before.hour
            } else {
                23
            };
            for hour in self.allowed(Field::Hour).into_iter().rev() {
                if hour > start_hour {
                    continue;
                }
                let start_minute = if (y, m, d) == (before.year, before.month, before.day)
                    && hour == before.hour
                {
                    before.minute.saturating_sub(1)
                } else {
                    59
                };
                for minute in self.allowed(Field::Minute).into_iter().rev() {
                    if minute > start_minute {
                        continue;
                    }
                    let at = CivilDateTime {
                        year: y,
                        month: m,
                        day: d,
                        hour,
                        minute,
                    };
                    if at < before && self.matches_valid(at) {
                        return Ok(Some(at));
                    }
                }
            }
            date = match previous_date(y, m, d) {
                Some(v) => v,
                None => return Ok(None),
            };
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut cursor = start;
        while cursor < end {
            if self.matches_valid(cursor) {
                if fires.len() == cap {
                    return Ok(FireWindow {
                        fires,
                        truncated: true,
                    });
                }
                fires.push(cursor);
            }
            match self.next_fire(cursor) {
                Ok(Some(next)) if next < end => cursor = next,
                _ => break,
            }
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::new();
        for field in FIELDS {
            let i = field_index(field);
            let (lo, hi) = bounds(field);
            let full = self.sets[i].iter().all(|v| *v);
            let rendered = if field == Field::DayOfMonth && self.unrestricted_dom {
                "*".to_string()
            } else if field == Field::DayOfWeek && self.unrestricted_dow {
                "*".to_string()
            } else if full {
                if matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                    "*/1".to_string()
                } else {
                    "*".to_string()
                }
            } else {
                let mut s = String::new();
                for (idx, yes) in self.sets[i].iter().enumerate() {
                    if *yes {
                        if !s.is_empty() {
                            s.push(',');
                        }
                        write!(&mut s, "{}", lo as usize + idx).unwrap();
                    }
                }
                let _ = hi;
                s
            };
            fields.push(rendered);
        }
        fields.join(" ")
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_len(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_len(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut y = 2000u16;
    let mut m = 1u8;
    let mut d = 1u8;
    let mut w = 6u8;
    while (y, m, d) < (year, month, day) {
        w = (w + 1) % 7;
        if d < month_len(y, m) {
            d += 1;
        } else {
            d = 1;
            if m == 12 {
                m = 1;
                y += 1;
            } else {
                m += 1;
            }
        }
    }
    w
}

fn next_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day < month_len(year, month) {
        Some((year, month, day + 1))
    } else if month < 12 {
        Some((year, month + 1, 1))
    } else if year < 2399 {
        Some((year + 1, 1, 1))
    } else {
        None
    }
}

fn previous_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day > 1 {
        Some((year, month, day - 1))
    } else if month > 1 {
        let m = month - 1;
        Some((year, m, month_len(year, m)))
    } else if year > 2000 {
        Some((year - 1, 12, 31))
    } else {
        None
    }
}
