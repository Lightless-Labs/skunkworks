use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = hex_value(bytes[i + 1]);
            let low = hex_value(bytes[i + 2]);
            match (high, low) {
                (Some(high), Some(low)) => {
                    decoded.push((high << 4) | low);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(error.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };
        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_component(
    component: Component,
    value: &str,
    base: usize,
) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (offset, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }

    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = value
            .saturating_mul(10)
            .saturating_add((byte - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, byte) in bytes.iter().copied().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, byte) in bytes[..colon].iter().copied().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(offset, &byte)| (byte == b'@').then_some(offset))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }
    let (userinfo, host_start_offset) = match at_positions.first().copied() {
        Some(at) if at == 0 => {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        Some(at) => (Some(&authority[..at]), at + 1),
        None => (None, 0),
    };

    let host_start = authority_start + host_start_offset;
    let host_authority = &authority[host_start_offset..];
    let host_authority_bytes = host_authority.as_bytes();

    let (host, ipv6, port, port_start_offset) = if host_authority_bytes.first() == Some(&b'[') {
        let close = host_authority_bytes[1..]
            .iter()
            .position(|&byte| byte == b']')
            .map(|offset| offset + 1);
        let close = match close {
            Some(close) => close,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &host_authority[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }

        let suffix_start = close + 1;
        let suffix = &host_authority[suffix_start..];
        let (port, port_start_offset) = if suffix.is_empty() {
            (None, None)
        } else if suffix.as_bytes()[0] == b':' {
            let port_start = host_start_offset + suffix_start + 1;
            (Some(parse_port(&suffix[1..], host_start + suffix_start + 1)?), Some(port_start))
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        };

        (interior.to_ascii_lowercase(), true, port, port_start_offset)
    } else {
        let host_colon = host_authority_bytes.iter().position(|&byte| byte == b':');
        let (host_text, port) = match host_colon {
            Some(offset) => {
                let port_start = host_start + offset + 1;
                (
                    &host_authority[..offset],
                    Some(parse_port(&host_authority[offset + 1..], port_start)?),
                )
            }
            None => (host_authority, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), false, port, None)
    };

    if let Some(userinfo_text) = userinfo {
        validate_component(
            Component::UserInfo,
            userinfo_text,
            authority_start,
        )?;
    }

    let remainder = &input[authority_end..];
    let fragment_relative = remainder.find('#');
    let before_fragment = fragment_relative.map_or(remainder, |index| &remainder[..index]);
    let fragment = fragment_relative.map(|index| &remainder[index + 1..]);

    let query_relative = before_fragment.find('?');
    let path = query_relative.map_or(before_fragment, |index| &before_fragment[..index]);
    let query = query_relative.map(|index| &before_fragment[index + 1..]);

    let path_value = if path.is_empty() { "/" } else { path };
    if !path.is_empty() {
        validate_component(Component::Path, path, authority_end)?;
    }
    if let Some(query_text) = query {
        let query_start = authority_end + query_relative.unwrap() + 1;
        validate_component(Component::Query, query_text, query_start)?;
    }
    if let Some(fragment_text) = fragment {
        let fragment_start = authority_end + fragment_relative.unwrap() + 1;
        validate_component(Component::Fragment, fragment_text, fragment_start)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo: userinfo.map(str::to_owned),
        host,
        port,
        path: path_value.to_owned(),
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}
