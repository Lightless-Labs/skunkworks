#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { start: usize },
    Double { start: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double { .. } => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let n = input[next_i..].chars().next().unwrap();
                        let after_n = next_i + n.len_utf8();
                        if matches!(n, '"' | '\\' | '$') {
                            word.push(n);
                            i = after_n;
                        } else if n == '\n' {
                            i = after_n;
                        } else {
                            word.push('\\');
                            i = next_i;
                        }
                    }
                    '$' => {
                        let (end, value) = expand(input, i, env)?;
                        if let Some((end, value)) = value {
                            word.push_str(&value);
                            active = true;
                            i = end;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    }
                    _ => {
                        word.push(c);
                        i = next_i;
                    }
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else {
                    match c {
                        '\'' => {
                            quote = Quote::Single { start: i };
                            active = true;
                            i = next_i;
                        }
                        '"' => {
                            quote = Quote::Double { start: i };
                            active = true;
                            i = next_i;
                        }
                        '\\' => {
                            if next_i == input.len() {
                                return Err(SplitError::TrailingEscape { pos: i });
                            }
                            let n = input[next_i..].chars().next().unwrap();
                            let after_n = next_i + n.len_utf8();
                            if n == '\n' {
                                i = after_n;
                            } else {
                                word.push(n);
                                active = true;
                                i = after_n;
                            }
                        }
                        '$' => {
                            let (end, value) = expand(input, i, env)?;
                            if let Some((end, value)) = value {
                                word.push_str(&value);
                                active = true;
                                i = end;
                            } else {
                                word.push('$');
                                active = true;
                                i = next_i;
                            }
                        }
                        _ => {
                            word.push(c);
                            active = true;
                            i = next_i;
                        }
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single { start } => return Err(SplitError::UnterminatedSingle { pos: start }),
        Quote::Double { start } => return Err(SplitError::UnterminatedDouble { pos: start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_char(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

// Returns the next input offset and, when a real expansion was consumed, its value.
fn expand(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, Option<(usize, String)>), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((after_dollar, None));
    }

    let bytes = input.as_bytes();
    if bytes[after_dollar] == b'{' {
        let contents_start = after_dollar + 1;
        let mut p = contents_start;
        let mut close = None;

        while p < input.len() {
            let ch = input[p..].chars().next().unwrap();
            if ch == '\\' {
                p += 1;
                if p < input.len() {
                    let escaped = input[p..].chars().next().unwrap();
                    p += escaped.len_utf8();
                }
            } else if ch == '}' {
                close = Some(p);
                break;
            } else {
                p += ch.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &input[contents_start..close];
        let (name_end, name) = parse_name(content, 0);
        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let default = if name_end == content.len() {
            None
        } else if content[name_end..].starts_with(":-") {
            Some(&content[name_end + 2..])
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        let value = lookup(env, name);
        let selected = match default {
            Some(text) if value.map_or(true, str::is_empty) => decode_default(text),
            _ => value.unwrap_or("").to_string(),
        };
        Ok((close + 1, Some((close + 1, selected))))
    } else if is_name_start(bytes[after_dollar]) {
        let mut end = after_dollar + 1;
        while end < input.len() && is_name_char(bytes[end]) {
            end += 1;
        }
        let name = &input[after_dollar..end];
        Ok((end, Some((end, lookup(env, name).unwrap_or("").to_string()))))
    } else {
        Ok((after_dollar, None))
    }
}

fn parse_name(s: &str, start: usize) -> (usize, &str) {
    let bytes = s.as_bytes();
    if start >= bytes.len() || !is_name_start(bytes[start]) {
        return (start, "");
    }
    let mut end = start + 1;
    while end < bytes.len() && is_name_char(bytes[end]) {
        end += 1;
    }
    (end, &s[start..end])
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let escaped = text[i..].chars().next().unwrap();
                decoded.push(escaped);
                i += escaped.len_utf8();
            }
        } else {
            decoded.push(c);
            i += c.len_utf8();
        }
    }
    decoded
}
