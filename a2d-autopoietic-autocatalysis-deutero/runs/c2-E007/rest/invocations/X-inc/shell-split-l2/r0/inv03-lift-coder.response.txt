#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        let next_i = i + ch.len_utf8();

        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if ch == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if ch == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if ch == '$' {
                    if let Some((value, end)) = expand(input, i, env)? {
                        active = true;
                        word.push_str(&value);
                        i = end;
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    word.push(ch);
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if matches!(ch, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if ch == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if ch == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if ch == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        word.push(next);
                        i = after_next;
                    }
                } else if ch == '$' {
                    if let Some((value, end)) = expand(input, i, env)? {
                        active = true;
                        word.push_str(&value);
                        i = end;
                    } else {
                        active = true;
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    word.push(ch);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<Option<(String, usize)>, SplitError> {
    let after_dollar = dollar_pos + 1;
    let Some(ch) = input[after_dollar..].chars().next() else {
        return Ok(None);
    };

    if ch == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let current = input[cursor..].chars().next().unwrap();
            if current == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    let escaped = input[cursor..].chars().next().unwrap();
                    cursor += escaped.len_utf8();
                }
            } else if current == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += current.len_utf8();
            }
        }

        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
        };

        let contents = &input[content_start..close_pos];
        let (name, default) = if let Some(operator_pos) = contents.find(":-") {
            (&contents[..operator_pos], Some(&contents[operator_pos + 2..]))
        } else {
            (contents, None)
        };

        let mut name_chars = name.chars();
        if !name_chars.next().is_some_and(is_name_start)
            || !name_chars.all(is_name_continue)
        {
            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
        }

        let value = lookup(env, name).unwrap_or("");
        let expanded = if let Some(default_text) = default {
            if value.is_empty() {
                decode_default(default_text)
            } else {
                value.to_string()
            }
        } else {
            value.to_string()
        };

        Ok(Some((expanded, close_pos + 1)))
    } else if is_name_start(ch) {
        let mut end = after_dollar;
        while end < input.len() {
            let current = input[end..].chars().next().unwrap();
            if is_name_continue(current) {
                end += current.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok(Some((lookup(env, name).unwrap_or("").to_string(), end)))
    } else {
        Ok(None)
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let ch = text[i..].chars().next().unwrap();
        if ch == '\\' {
            i += 1;
            if i < text.len() {
                let escaped = text[i..].chars().next().unwrap();
                result.push(escaped);
                i += escaped.len_utf8();
            } else {
                result.push('\\');
            }
        } else {
            result.push(ch);
            i += ch.len_utf8();
        }
    }
    result
}
