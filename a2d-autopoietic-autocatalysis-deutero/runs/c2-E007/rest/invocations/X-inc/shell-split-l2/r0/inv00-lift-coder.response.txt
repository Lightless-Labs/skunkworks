#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { start: usize },
    Double { start: usize },
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, c)) = chars.next() {
        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
            }
            Quote::Double { .. } => {
                match c {
                    '"' => quote = Quote::None,
                    '\\' => {
                        let Some(&(_, next)) = chars.peek() else {
                            return Err(SplitError::TrailingEscape { pos });
                        };
                        if matches!(next, '"' | '\\' | '$' | '\n') {
                            chars.next();
                            if next != '\n' {
                                word.push(next);
                            }
                        } else {
                            word.push('\\');
                        }
                    }
                    '$' => {
                        active = true;
                        expand(input, &mut chars, pos, env, &mut word)?;
                    }
                    _ => word.push(c),
                }
            }
            Quote::None => match c {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    quote = Quote::Single { start: pos };
                }
                '"' => {
                    active = true;
                    quote = Quote::Double { start: pos };
                }
                '\\' => {
                    let Some(&(_, next)) = chars.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    chars.next();
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                }
                '$' => {
                    active = true;
                    expand(input, &mut chars, pos, env, &mut word)?;
                }
                _ => {
                    active = true;
                    word.push(c);
                }
            },
        }
    }

    match quote {
        Quote::Single { start } => return Err(SplitError::UnterminatedSingle { pos: start }),
        Quote::Double { start } => return Err(SplitError::UnterminatedDouble { pos: start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn expand(
    input: &str,
    chars: &mut std::iter::Peekable<std::str::CharIndices<'_>>,
    dollar_pos: usize,
    env: &[(&str, &str)],
    word: &mut String,
) -> Result<(), SplitError> {
    let Some(&(_, next)) = chars.peek() else {
        word.push('$');
        return Ok(());
    };

    if next == '{' {
        chars.next();

        let content_start = chars.peek().map_or(input.len(), |(p, _)| *p);
        let mut close = None;
        let mut scan = input[content_start..].char_indices().peekable();

        while let Some((relative, c)) = scan.next() {
            if c == '\\' {
                scan.next();
            } else if c == '}' {
                close = Some(content_start + relative);
                break;
            }
        }

        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
        };

        let contents = &input[content_start..close_pos];
        let (name, default) = if let Some(separator) = contents.find(":-") {
            (&contents[..separator], Some(&contents[separator + 2..]))
        } else {
            (contents, None)
        };

        let mut name_chars = name.chars();
        let valid_name = name_chars.next().is_some_and(is_name_start)
            && name_chars.all(is_name_continue);

        if !valid_name {
            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
        }

        let value = lookup(name, env).unwrap_or("");
        if let Some(default_text) = default {
            if value.is_empty() {
                word.push_str(&decode_default(default_text));
            } else {
                word.push_str(value);
            }
        } else {
            word.push_str(value);
        }

        while chars.peek().is_some_and(|(p, _)| *p <= close_pos) {
            chars.next();
        }
        if chars.peek().is_some_and(|(p, c)| *p == close_pos && *c == '}') {
            chars.next();
        }

        return Ok(());
    }

    if is_name_start(next) {
        let mut name = String::new();
        while let Some(&(_, c)) = chars.peek() {
            if is_name_continue(c) {
                name.push(c);
                chars.next();
            } else {
                break;
            }
        }
        if let Some(value) = lookup(&name, env) {
            word.push_str(value);
        }
        return Ok(());
    }

    word.push('$');
    Ok(())
}
