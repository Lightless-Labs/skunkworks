```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                if i == start {
                    return Err(Error::InvalidPath(i));
                }
                let component = &path[start..i];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = i + 1;
            }
        }

        Ok(components)
    }

    fn prefix(components: &[&str], count: usize) -> String {
        if count == 0 {
            "/".to_string()
        } else {
            format!("/{}", components[..count].join("/"))
        }
    }

    fn find<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;

        for (index, component) in components.iter().enumerate() {
            let dir = match current {
                Entry::Dir(dir) => dir,
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::prefix(components, index)));
                }
            };
            current = dir
                .get(*component)
                .ok_or_else(|| Error::Missing(Self::prefix(components, index + 1)))?;
        }

        Ok(current)
    }

    fn parent_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut current = &mut self.root;

        for (index, component) in components[..components.len() - 1].iter().enumerate() {
            let dir = match current {
                Entry::Dir(dir) => dir,
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::prefix(components, index)));
                }
            };
            current = dir
                .get_mut(*component)
                .ok_or_else(|| Error::Missing(Self::prefix(components, index + 1)))?;
        }

        match current {
            Entry::Dir(dir) => Ok(dir),
            Entry::File(_) => Err(Error::NotDir(Self::prefix(
                components,
                components.len() - 1,
            ))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let canonical = Self::prefix(&components, components.len());
        let name = components[components.len() - 1].to_string();
        let parent = self.parent_mut(&components)?;
        if parent.contains_key(&name) {
            return Err(Error::Exists(canonical));
        }

        parent.insert(name, Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let canonical = Self::prefix(&components, components.len());
        let name = components[components.len() - 1].to_string();
        let parent = self.parent_mut(&components)?;

        match parent.get_mut(&name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        match self.find(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::prefix(&components, components.len()))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let canonical = Self::prefix(&components, components.len());

        match self.find(&components)? {
            Entry::File(_) => Err(Error::NotDir(canonical)),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
