```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut result = Vec::new();
        let bytes = path.as_bytes();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                let component = &path[start..i];
                if component.is_empty() {
                    return Err(Error::InvalidPath(i.saturating_sub(1)));
                }
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                result.push(component);
                start = i + 1;
            }
        }

        Ok(result)
    }

    fn canonical_prefix(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn get_entry<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut prefix = Vec::new();

        for (index, component) in components.iter().enumerate() {
            if !matches!(current, Entry::Dir(_)) {
                return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
            }

            prefix.push(*component);
            let path = Self::canonical_prefix(&prefix);
            current = match current {
                Entry::Dir(children) => children
                    .get(*component)
                    .ok_or_else(|| Error::Missing(path))?,
                Entry::File(_) => unreachable!(),
            };

            if index + 1 < components.len() && !matches!(current, Entry::Dir(_)) {
                return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
            }
        }

        Ok(current)
    }

    fn get_parent_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut current = &mut self.root;
        let mut prefix = Vec::new();

        for component in components {
            prefix.push(*component);
            let path = Self::canonical_prefix(&prefix);
            current = match current {
                Entry::Dir(children) => children
                    .get_mut(*component)
                    .ok_or_else(|| Error::Missing(path.clone()))?,
                Entry::File(_) => return Err(Error::NotDir(Self::canonical_prefix(&prefix[..prefix.len() - 1]))),
            };

            if !matches!(current, Entry::Dir(_)) {
                return Err(Error::NotDir(path));
            }
        }

        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(&prefix))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let (name, parent) = components.split_last().expect("nonempty path");
        let children = self.get_parent_mut(parent)?;
        if children.contains_key(*name) {
            return Err(Error::Exists(Self::canonical_prefix(&components)));
        }

        children.insert((*name).to_owned(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let (name, parent) = components.split_last().expect("nonempty path");
        let children = self.get_parent_mut(parent)?;
        match children.get_mut(*name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(Self::canonical_prefix(&components))),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                children.insert((*name).to_owned(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::components(path)?;
        match self.get_entry(&components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical_prefix(&components))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::components(path)?;
        match self.get_entry(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(&components))),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
