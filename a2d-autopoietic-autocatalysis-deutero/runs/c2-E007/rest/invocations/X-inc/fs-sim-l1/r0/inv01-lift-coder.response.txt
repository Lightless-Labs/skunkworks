```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn parse_path(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(offset) = path.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidPath(offset));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for index in 1..=bytes.len() {
            if index == bytes.len() || bytes[index] == b'/' {
                if index == start {
                    return Err(Error::InvalidPath(index.saturating_sub(1)));
                }
                let component = &path[start..index];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = index + 1;
            }
        }

        Ok(components)
    }

    fn canonical_prefix(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn lookup<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        for (index, component) in components.iter().enumerate() {
            let prefix = Self::canonical_prefix(&components[..=index]);
            match current {
                Entry::Dir(children) => {
                    current = children
                        .get(*component)
                        .ok_or(Error::Missing(prefix))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(
                        &components[..index],
                    )));
                }
            }
        }
        Ok(current)
    }

    fn lookup_mut<'a>(&'a mut self, components: &[&str]) -> Result<&'a mut Entry, Error> {
        let mut current = &mut self.root;
        for (index, component) in components.iter().enumerate() {
            let prefix = Self::canonical_prefix(&components[..=index]);
            match current {
                Entry::Dir(children) => {
                    current = children
                        .get_mut(*component)
                        .ok_or(Error::Missing(prefix))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(
                        &components[..index],
                    )));
                }
            }
        }
        Ok(current)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::parse_path(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let parent_path = Self::canonical_prefix(parent_components);
        let child_name = components[components.len() - 1];
        let full_path = Self::canonical_prefix(&components);

        match self.lookup_mut(parent_components)? {
            Entry::Dir(children) => {
                if children.contains_key(child_name) {
                    Err(Error::Exists(full_path))
                } else {
                    children.insert(
                        child_name.to_owned(),
                        Entry::Dir(BTreeMap::new()),
                    );
                    Ok(())
                }
            }
            Entry::File(_) => Err(Error::NotDir(parent_path)),
        }
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::parse_path(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let parent_path = Self::canonical_prefix(parent_components);
        let child_name = components[components.len() - 1];
        let full_path = Self::canonical_prefix(&components);

        match self.lookup_mut(parent_components)? {
            Entry::Dir(children) => match children.get_mut(child_name) {
                Some(Entry::Dir(_)) => Err(Error::IsDir(full_path)),
                Some(Entry::File(contents)) => {
                    contents.clear();
                    contents.extend_from_slice(data);
                    Ok(())
                }
                None => {
                    children.insert(child_name.to_owned(), Entry::File(data.to_vec()));
                    Ok(())
                }
            },
            Entry::File(_) => Err(Error::NotDir(parent_path)),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::parse_path(path)?;
        let full_path = Self::canonical_prefix(&components);
        match self.lookup(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(full_path)),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::parse_path(path)?;
        let full_path = Self::canonical_prefix(&components);
        match self.lookup(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(full_path)),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
