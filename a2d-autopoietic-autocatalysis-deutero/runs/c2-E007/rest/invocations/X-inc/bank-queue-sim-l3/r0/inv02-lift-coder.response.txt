#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    remaining: u64,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn best_waiter(waiting: &[Job]) -> Option<usize> {
    waiting.iter().enumerate().min_by(|(_, a), (_, b)| {
        a.customer
            .priority
            .cmp(&b.customer.priority)
            .then_with(|| match (a.pause_time, b.pause_time) {
                (Some(at), Some(bt)) => at.cmp(&bt),
                (Some(_), None) => std::cmp::Ordering::Less,
                (None, Some(_)) => std::cmp::Ordering::Greater,
                (None, None) => a
                    .customer
                    .arrival
                    .cmp(&b.customer.arrival)
                    .then_with(|| b.customer.id.cmp(&a.customer.id)),
            })
    }).map(|(i, _)| i)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|prior| prior.id == customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        if let Some(active) = running.as_ref() {
            if active.finish < time {
                unreachable!();
            }
        }

        // Process a scheduled completion before arrivals and decisions at this tick.
        if running.as_ref().is_some_and(|r| r.finish == time) {
            let active = running.take().unwrap();
            trace.push(Event {
                time,
                customer: active.job.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        // Enqueue the complete arrival cohort before considering preemption.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        let protected = running
            .as_ref()
            .is_some_and(|r| time < r.work_at);

        if !protected {
            if let (Some(active), Some(waiter_index)) =
                (running.as_ref(), best_waiter(&waiting))
            {
                if waiting[waiter_index].customer.priority < active.job.customer.priority {
                    let waiter_id = waiting[waiter_index].customer.id;
                    let mut active = running.take().unwrap();
                    let remaining = active.finish - time;
                    trace.push(Event {
                        time,
                        customer: active.job.customer.id,
                        kind: Kind::Preempt {
                            remaining,
                            by: waiter_id,
                        },
                    });
                    active.job.remaining = remaining;
                    active.job.pause_time = Some(time);
                    waiting.push(active.job);
                }
            }
        }

        // Fill an idle server, repeating for inline zero-duration completions.
        while running.is_none() {
            let Some(index) = best_waiter(&waiting) else {
                break;
            };
            let job = waiting.remove(index);
            let id = job.customer.id;
            let dispatch_cost = if last_dispatched.is_some_and(|last| last != id) {
                switch_cost
            } else {
                0
            };
            let work_at = time.checked_add(dispatch_cost).ok_or(Error::TimeOverflow {
                customer: id,
                time,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time,
            })?;
            last_dispatched = Some(id);

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            let mut job = job;
            job.started = true;
            job.pause_time = None;
            trace.push(Event {
                time,
                customer: id,
                kind,
            });

            if finish == time {
                trace.push(Event {
                    time,
                    customer: id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            } else {
                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = &running {
            for boundary in [Some(active.work_at), Some(active.finish)]
                .into_iter()
                .flatten()
                .filter(|&t| t > time)
            {
                next_time = Some(next_time.map_or(boundary, |n| n.min(boundary)));
            }
        }

        if let Some(next) = next_time {
            time = next;
        } else {
            break;
        }
    }

    Ok(trace)
}
