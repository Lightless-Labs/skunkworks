#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    paused_at: Option<u64>,
}

#[derive(Clone)]
struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(at_a), Some(at_b)) => at_a
                .cmp(&at_b)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter_index(waiting: &[Waiting]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
        .map(|(index, _)| index)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut time = arrivals[0].arrival;

    loop {
        if let Some(active) = running.as_ref() {
            if active.finish == time {
                let finished = running.take().unwrap();
                events.push(Event {
                    time,
                    customer: finished.customer.id,
                    kind: Kind::Finish,
                });
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                paused_at: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                if let Some(index) = best_waiter_index(&waiting) {
                    if waiting[index].customer.priority < active.customer.priority {
                        let by = waiting[index].customer.id;
                        let active = running.take().unwrap();
                        let remaining = active.finish - time;
                        events.push(Event {
                            time,
                            customer: active.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            customer: active.customer,
                            remaining,
                            started: true,
                            paused_at: Some(time),
                        });
                    }
                }
            }
        }

        while running.is_none() {
            let Some(index) = best_waiter_index(&waiting) else {
                break;
            };
            let chosen = waiting.remove(index);
            let cost = if last_dispatched == Some(chosen.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = time
                .checked_add(cost)
                .ok_or(Error::TimeOverflow {
                    customer: chosen.customer.id,
                    time,
                })?;
            let finish = work_at
                .checked_add(chosen.remaining)
                .ok_or(Error::TimeOverflow {
                    customer: chosen.customer.id,
                    time,
                })?;

            let kind = if chosen.started {
                Kind::Resume {
                    remaining: chosen.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: chosen.remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time,
                customer: chosen.customer.id,
                kind,
            });
            last_dispatched = Some(chosen.customer.id);
            running = Some(Running {
                customer: chosen.customer,
                remaining: chosen.remaining,
                work_at,
                finish,
            });

            if finish == time {
                let finished = running.take().unwrap();
                events.push(Event {
                    time,
                    customer: finished.customer.id,
                    kind: Kind::Finish,
                });
            }
        }

        if next_arrival == arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            if active.finish > time {
                next_time = Some(next_time.map_or(active.finish, |t| t.min(active.finish)));
            }
            if active.work_at > time {
                next_time = Some(next_time.map_or(active.work_at, |t| t.min(active.work_at)));
            }
        }
        match next_time {
            Some(t) => time = t,
            None => break,
        }
    }

    Ok(events)
}
