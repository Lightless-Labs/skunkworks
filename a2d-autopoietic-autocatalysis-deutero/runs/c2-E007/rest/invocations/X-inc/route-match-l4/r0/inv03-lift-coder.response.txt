#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_' | b'-'))
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if unreserved(b) {
            decoded.push(b);
            i += 1;
        } else if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(h) = hex_upper(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(l) = hex_upper(bytes[i + 2]) else {
                return Err(base + i);
            };
            let v = h * 16 + l;
            if !(0x20..=0x7e).contains(&v) || v == b'/' || unreserved(v) {
                return Err(base + i);
            }
            decoded.push(v);
            i += 3;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_path(path: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    if !path.starts_with('/') {
        return Err(invalid(0));
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = path.as_bytes();
    let mut raw_segments: Vec<(usize, &str)> = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if i == start {
                    return Err(invalid(start));
                }
                break;
            }
            if i == start {
                return Err(invalid(start));
            }
            raw_segments.push((start, &path[start..i]));
            start = i + 1;
        }
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    if start < end {
        raw_segments.push((start, &path[start..end]));
    } else if !trailing {
        return Err(invalid(start));
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = Vec::<String>::new();
    for (idx, (at, raw)) in raw_segments.iter().enumerate() {
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(invalid(*at));
            }
            if names.iter().any(|n| n == name) {
                return Err(invalid(*at));
            }
            if marker == b'*' && idx + 1 != raw_segments.len() {
                return Err(invalid(*at));
            }
            names.push(name.to_string());
            segments.push(if marker == b':' {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            });
        } else {
            match decode_segment(raw, *at) {
                Ok(value) => segments.push(Segment::Static(value)),
                Err(at) => return Err(invalid(at)),
            }
        }
    }
    Ok((segments, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method != method
                || (self.policy == SlashPolicy::Strict && route.trailing != trailing)
                || route.segments.len() != segments.len()
            {
                continue;
            }
            let same_shape = route
                .segments
                .iter()
                .zip(&segments)
                .all(|(a, b)| match (a, b) {
                    (Segment::Static(x), Segment::Static(y)) => x == y,
                    (Segment::Param(_), Segment::Param(_)) => true,
                    (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                    _ => false,
                });
            if same_shape {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head, false)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>, Vec<u8>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            let wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
            if (!wildcard && route.segments.len() != request_segments.len())
                || (wildcard && route.segments.len() - 1 > request_segments.len())
            {
                continue;
            }

            let mut params = Vec::new();
            let mut rank = Vec::new();
            let mut matched = true;
            let fixed_len = if wildcard {
                route.segments.len() - 1
            } else {
                route.segments.len()
            };

            for i in 0..fixed_len {
                match (&route.segments[i], &request_segments[i]) {
                    (Segment::Static(expected), Segment::Static(actual)) if expected == actual => {
                        rank.push(2);
                    }
                    (Segment::Static(_), _) => {
                        matched = false;
                        break;
                    }
                    (Segment::Param(name), Segment::Static(value)) => {
                        rank.push(1);
                        params.push((name.clone(), value.clone()));
                    }
                    (Segment::Wildcard(_), _) => unreachable!(),
                    (_, _) => {
                        matched = false;
                        break;
                    }
                }
            }

            if !matched {
                continue;
            }

            if wildcard {
                if let Segment::Wildcard(name) = &route.segments[fixed_len] {
                    let tail = request_segments[fixed_len..]
                        .iter()
                        .map(|segment| match segment {
                            Segment::Static(value) => value.as_str(),
                            _ => unreachable!(),
                        })
                        .collect::<Vec<_>>()
                        .join("/");
                    params.push((name.clone(), tail));
                    rank.push(0);
                }
            } else {
                rank.push(3);
            }
            candidates.push((route, params, rank));
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _, _)| route.method == eligible_method)
            .collect();
        if eligible.is_empty() {
            let mut allowed = candidates
                .iter()
                .map(|(route, _, _)| route.method.clone())
                .collect::<Vec<_>>();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let winner = eligible
            .into_iter()
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.0.id.cmp(&a.0.id)))
            .expect("eligible routes are nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && winner.0.trailing != request_trailing
        {
            if winner.0.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.0.id,
            params: winner.1.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = Vec::<(&str, &str)>::new();
        for &(key, value) in params {
            if supplied.iter().any(|(existing, _)| *existing == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        let capture_names = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect::<Vec<_>>();

        for key in &capture_names {
            if !supplied.iter().any(|(given, _)| given == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extras = supplied
            .iter()
            .filter(|(key, _)| !capture_names.iter().any(|capture| capture == key))
            .map(|(key, _)| *key)
            .collect::<Vec<_>>();
        extras.sort();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for segment in &route.segments {
            let name = match segment {
                Segment::Param(name) | Segment::Wildcard(name) => name,
                Segment::Static(_) => continue,
            };
            let value = supplied
                .iter()
                .find(|(key, _)| *key == name)
                .map(|(_, value)| *value)
                .expect("required parameter checked");
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid_value = match segment {
                Segment::Param(_) => !value.is_empty() && !value.contains('/'),
                Segment::Wildcard(_) => {
                    value.is_empty()
                        || value
                            .split('/')
                            .all(|component| !component.is_empty())
                }
                Segment::Static(_) => true,
            };
            if !valid_ascii || !valid_value {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut output = String::new();
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    output.push('/');
                    output.push_str(&encode_segment(value.as_bytes()));
                }
                Segment::Param(name) => {
                    let value = supplied
                        .iter()
                        .find(|(key, _)| *key == name)
                        .map(|(_, value)| *value)
                        .expect("required parameter checked");
                    output.push('/');
                    output.push_str(&encode_segment(value.as_bytes()));
                }
                Segment::Wildcard(name) => {
                    let value = supplied
                        .iter()
                        .find(|(key, _)| *key == name)
                        .map(|(_, value)| *value)
                        .expect("required parameter checked");
                    if !value.is_empty() {
                        for component in value.split('/') {
                            output.push('/');
                            output.push_str(&encode_segment(component.as_bytes()));
                        }
                    }
                }
            }
        }

        if output.is_empty() {
            output.push('/');
        } else if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn encode_segment(bytes: &[u8]) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut output = String::new();
    for &b in bytes {
        if unreserved(b) {
            output.push(b as char);
        } else {
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    output
}
