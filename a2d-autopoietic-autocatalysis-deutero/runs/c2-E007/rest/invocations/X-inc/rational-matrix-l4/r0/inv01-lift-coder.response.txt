use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

fn rational_from_signed_mag(negative: bool, mag: u128, den: u128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if mag == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(mag, den);
    let m = mag / g;
    let d = den / g;
    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    if negative {
        if m > (1u128 << 63) {
            return Err(RationalError::Overflow);
        }
        let n = if m == (1u128 << 63) {
            i64::MIN
        } else {
            -(m as i64)
        };
        Ok(Rational { num: n, den: d as i64 })
    } else {
        if m > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(Rational { num: m as i64, den: d as i64 })
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Rational { num: 0, den: 1 });
        }
        let neg = (num < 0) ^ (den < 0);
        rational_from_signed_mag(neg, num.unsigned_abs() as u128, den.unsigned_abs() as u128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let left = (self.num as i128).checked_mul(b).ok_or(RationalError::Overflow)?;
        let right = (rhs.num as i128).checked_mul(a).ok_or(RationalError::Overflow)?;
        let n = left.checked_add(right).ok_or(RationalError::Overflow)?;
        let d = a.checked_mul(rhs.den as i128).ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let left = (self.num as i128).checked_mul(b).ok_or(RationalError::Overflow)?;
        let right = (rhs.num as i128).checked_mul(a).ok_or(RationalError::Overflow)?;
        let n = left.checked_sub(right).ok_or(RationalError::Overflow)?;
        let d = a.checked_mul(rhs.den as i128).ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128);
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128);
        let a = (self.num as i128) / g1 as i128;
        let b = (rhs.num as i128) / g2 as i128;
        let c = (self.den as i128) / g2 as i128;
        let d = (rhs.den as i128) / g1 as i128;
        let n = a.checked_mul(b).ok_or(RationalError::Overflow)?;
        let den = c.checked_mul(d).ok_or(RationalError::Overflow)?;
        rational_from_i128(n, den)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128);
        let g2 = gcd_u128(self.den as u128, rhs.den as u128);
        let a = self.num as i128 / g1 as i128;
        let b = rhs.den as i128 / g2 as i128;
        let c = self.den as i128 / g2 as i128;
        let d = rhs.num as i128 / g1 as i128;
        let n = a.checked_mul(b).ok_or(RationalError::Overflow)?;
        let den = c.checked_mul(d).ok_or(RationalError::Overflow)?;
        rational_from_i128(n, den)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn err(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = 0usize;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(err(i, if i == bytes.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }

        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let s = i;
            i += 1;
            Some(s)
        } else {
            None
        };

        let den_start = if slash.is_some() { i } else { bytes.len() };
        if slash.is_some() {
            let start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        if i < bytes.len() {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let num_digits = &bytes[num_start..slash.unwrap_or(bytes.len())];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == b"0" {
            return Err(err(0, ParseKind::NonCanonical));
        }

        let den_digits = if slash.is_some() { &bytes[den_start..] } else { &[] };
        if !den_digits.is_empty() && den_digits.len() >= 2 && den_digits[0] == b'0' {
            return Err(err(den_start + 1, ParseKind::NonCanonical));
        }

        fn parse_magnitude(digits: &[u8], max: u128) -> Option<u128> {
            let mut value = 0u128;
            for &b in digits {
                value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
                if value > max {
                    return None;
                }
            }
            Some(value)
        }

        let max_num = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let nmag = parse_magnitude(num_digits, max_num)
            .ok_or_else(|| err(0, ParseKind::OutOfRange))?;
        let dmag = if slash.is_some() {
            parse_magnitude(den_digits, i64::MAX as u128)
                .ok_or_else(|| err(den_start, ParseKind::OutOfRange))?
        } else {
            1
        };

        if slash.is_some() && dmag == 0 {
            return Err(err(den_start, ParseKind::ZeroDenominator));
        }
        if slash.is_some() && dmag == 1 {
            return Err(err(slash.unwrap(), ParseKind::NonCanonical));
        }
        if slash.is_some() && nmag == 0 {
            return Err(err(slash.unwrap(), ParseKind::NonCanonical));
        }
        if slash.is_some() && gcd_u128(nmag, dmag) > 1 {
            return Err(err(slash.unwrap(), ParseKind::NonCanonical));
        }

        rational_from_signed_mag(negative, nmag, dmag).map_err(|_| err(0, ParseKind::OutOfRange))
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

fn checked_area(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = checked_area(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Matrix { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let count = checked_area(self.rows, rhs.cols)?;
        let mut out = Vec::with_capacity(count);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let term = self.data[r * self.cols + k]
                        .mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Matrix { rows: self.rows, cols: rhs.cols, data: out })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut work = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == work.rows {
                break;
            }
            let mut best: Option<usize> = None;
            for row in p..work.rows {
                let value = work.data[row * work.cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), row);
                if best.map_or(true, |b| {
                    let v = work.data[b * work.cols + col];
                    key < (v.den, v.num.unsigned_abs(), b)
                }) {
                    best = Some(row);
                }
            }
            let Some(q) = best else { continue };
            if q != p {
                for c in 0..work.cols {
                    work.data.swap(p * work.cols + c, q * work.cols + c);
                }
                swaps.push((p, q));
            }
            let pivot = work.data[p * work.cols + col];
            let start = p * work.cols;
            for c in 0..work.cols {
                work.data[start + c] = work.data[start + c].div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            }
            for row in 0..work.rows {
                if row == p {
                    continue;
                }
                let factor = work.data[row * work.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..work.cols {
                    let product = factor.mul(&work.data[start + c])
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    let idx = row * work.cols + c;
                    work.data[idx] = work.data[idx].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix: work, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let size = checked_area(n, width)?;
        let mut data = Vec::with_capacity(size);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let aug = Matrix { rows: n, cols: width, data };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut inverse = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                inverse.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix { rows: n, cols: n, data: inverse })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let size = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(size);
        for r in 0..self.rows {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                data.push(rhs.data[r * k + c]);
            }
        }
        let aug = Matrix { rows: self.rows, cols: width, data };
        let red = aug.eliminate(n).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..n).all(|c| red.matrix.data[r * width + c].num == 0);
            if coeff_zero {
                for c in 0..k {
                    if red.matrix.data[r * width + n + c].num != 0 {
                        let witness = red.matrix.data[r * width..(r + 1) * width].to_vec();
                        return Ok(Solution::None { row: r, rhs_col: c, witness });
                    }
                }
            }
        }

        let rank = red.pivots.len();
        let mut is_pivot = vec![false; n];
        for &c in &red.pivots {
            is_pivot[c] = true;
        }
        let free: Vec<usize> = (0..n).filter(|&c| !is_pivot[c]).collect();

        let particular_size = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_size];
        for (j, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] = red.matrix.data[j * width + n + c];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular };

        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: red.pivots });
        }

        let basis_size = n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_size];
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * free.len() + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let coeff = red.matrix.data[i * width + free_col];
                basis[pivot_col * free.len() + j] = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic { row: pivot_col, col: j, step: rank, source })
                })?;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: basis.len() / n.max(1), data: basis },
            pivots: red.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}
