use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn signed_from_magnitude(magnitude: u128, negative: bool) -> Result<i64, RationalError> {
    if negative {
        if magnitude == (i64::MAX as u128) + 1 {
            Ok(i64::MIN)
        } else if magnitude <= i64::MAX as u128 {
            Ok(-(magnitude as i64))
        } else {
            Err(RationalError::Overflow)
        }
    } else if magnitude <= i64::MAX as u128 {
        Ok(magnitude as i64)
    } else {
        Err(RationalError::Overflow)
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let negative = (num < 0) ^ (den < 0);
        let nmag = (num as i128).unsigned_abs();
        let dmag = (den as i128).unsigned_abs();
        let g = gcd_u128(nmag, dmag);
        let n = nmag / g;
        let d = dmag / g;
        let num = signed_from_magnitude(n, negative)?;
        if d > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = (self.num as i128).unsigned_abs();
        let b = (rhs.num as i128).unsigned_abs();
        let c = self.den as u128;
        let d = rhs.den as u128;
        let g1 = gcd_u128(a, d);
        let g2 = gcd_u128(b, c);
        let an = a / g1;
        let dn = d / g1;
        let bn = b / g2;
        let cn = c / g2;
        let mag = an.checked_mul(bn).ok_or(RationalError::Overflow)?;
        let den = cn.checked_mul(dn).ok_or(RationalError::Overflow)?;
        let negative = (self.num < 0) ^ (rhs.num < 0);
        let num = signed_from_magnitude(mag, negative)?;
        if den > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Rational::new(num, den as i64)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let a = (self.num as i128).unsigned_abs();
        let b = (rhs.num as i128).unsigned_abs();
        let c = self.den as u128;
        let d = rhs.den as u128;
        let g1 = gcd_u128(a, b);
        let g2 = gcd_u128(d, c);
        let an = a / g1;
        let bn = b / g1;
        let dn = d / g2;
        let cn = c / g2;
        let mag = an.checked_mul(dn).ok_or(RationalError::Overflow)?;
        let den = cn.checked_mul(bn).ok_or(RationalError::Overflow)?;
        let negative = (self.num < 0) ^ (rhs.num < 0);
        let num = signed_from_magnitude(mag, negative)?;
        if den > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Rational::new(num, den as i64)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(fail(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
            if i == bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(fail(i, ParseKind::InvalidByte));
        }

        let mut slash = None;
        let mut den_start = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            den_start = Some(i);
            if i == bytes.len() || !bytes[i].is_ascii_digit() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
        }
        if i < bytes.len() {
            return Err(fail(i, ParseKind::InvalidByte));
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_digits_start = if bytes[0] == b'-' { 1 } else { 0 };
        if num_end - num_digits_start >= 2 && bytes[num_digits_start] == b'0' {
            return Err(fail(num_digits_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_end == 2 && bytes[1] == b'0' {
            return Err(fail(0, ParseKind::NonCanonical));
        }

        if let Some(ds) = den_start {
            if bytes.len() - ds >= 2 && bytes[ds] == b'0' {
                return Err(fail(ds + 1, ParseKind::NonCanonical));
            }
        }

        let num_mag = parse_magnitude(&bytes[num_digits_start..num_end]);
        let neg = bytes[0] == b'-';
        let num_limit = if neg {
            (i64::MAX as u128) + 1
        } else {
            i64::MAX as u128
        };
        let num_mag = num_mag.ok_or_else(|| fail(0, ParseKind::OutOfRange))?;
        if num_mag > num_limit {
            return Err(fail(0, ParseKind::OutOfRange));
        }

        let num = signed_from_magnitude(num_mag, neg)
            .map_err(|_| fail(0, ParseKind::OutOfRange))?;

        let den_value = if let Some(ds) = den_start {
            let mag = parse_magnitude(&bytes[ds..])
                .ok_or_else(|| fail(ds, ParseKind::OutOfRange))?;
            if mag > i64::MAX as u128 {
                return Err(fail(ds, ParseKind::OutOfRange));
            }
            if mag == 0 {
                return Err(fail(ds, ParseKind::ZeroDenominator));
            }
            Some(mag as i64)
        } else {
            None
        };

        if let (Some(slash_at), Some(den)) = (slash, den_value) {
            if den == 1 {
                return Err(fail(slash_at, ParseKind::NonCanonical));
            }
            if num == 0 {
                return Err(fail(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128((num as i128).unsigned_abs(), den as u128) > 1 {
                return Err(fail(slash_at, ParseKind::NonCanonical));
            }
            Ok(Self { num, den })
        } else {
            Ok(Self { num, den: 1 })
        }
    }
}

fn parse_magnitude(bytes: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
    }
    Some(value)
}

fn rational_from_i128(n: i128, d: i128) -> Result<Rational, RationalError> {
    if d <= 0 {
        return Err(RationalError::Overflow);
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let negative = n < 0;
    let mag = n.unsigned_abs();
    let den = d as u128;
    let g = gcd_u128(mag, den);
    let num_mag = mag / g;
    let den = den / g;
    let num = signed_from_magnitude(num_mag, negative)?;
    if den > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num,
        den: den as i64,
    })
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|v| v.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols, self.cols)
    }

    fn eliminate(
        &self,
        pivot_cols: usize,
        total_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        let candidate_cols = pivot_cols.min(matrix.cols);

        for col in 0..candidate_cols {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (
                    value.den,
                    (value.num as i128).unsigned_abs(),
                    row,
                );
                if chosen.map_or(true, |old| {
                    let v = matrix.data[old * matrix.cols + col];
                    key < (v.den, (v.num as i128).unsigned_abs(), old)
                }) {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else {
                continue;
            };
            if q != p {
                for c in 0..total_cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..total_cols {
                let old = matrix.data[p * matrix.cols + c];
                matrix.data[p * matrix.cols + c] =
                    old.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..total_cols {
                    let pivot_entry = matrix.data[p * matrix.cols + c];
                    let product = factor.mul(&pivot_entry).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                    let current = matrix.data[row * matrix.cols + c];
                    matrix.data[row * matrix.cols + c] = current.sub(&product).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                }
            }

            pivots.push(col);
            p += 1;
        }

        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduced = augmented.eliminate(n, width)?;
        if reduced.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduced.pivots.len(),
            });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                out.push(reduced.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: out,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..m {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                data.push(rhs.data[r * k + c]);
            }
        }
        let augmented = Matrix {
            rows: m,
            cols: width,
            data,
        };
        let reduced = augmented
            .eliminate(n, width)
            .map_err(SolveError::Matrix)?;

        for row in 0..m {
            let coeff_zero = (0..n).all(|c| reduced.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                for rhs_col in 0..k {
                    if reduced.matrix.data[row * width + n + rhs_col].num != 0 {
                        return Ok(Solution::None {
                            row,
                            rhs_col,
                            witness: reduced.matrix.data[row * width..(row + 1) * width].to_vec(),
                        });
                    }
                }
            }
        }

        let rank = reduced.pivots.len();
        let mut free = Vec::new();
        let mut pivot_present = vec![false; n];
        for &p in &reduced.pivots {
            pivot_present[p] = true;
        }
        for c in 0..n {
            if !pivot_present[c] {
                free.push(c);
            }
        }

        let particular_len = n.checked_mul(k).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pivot_col) in reduced.pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular_data[pivot_col * k + rhs_col] =
                    reduced.matrix.data[j * width + n + rhs_col];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: reduced.pivots,
            });
        }

        let basis_len = n.checked_mul(free.len()).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in reduced.pivots.iter().enumerate() {
                let coeff = reduced.matrix.data[i * width + free_col];
                basis_data[pivot_col * free.len() + j] =
                    coeff.neg().map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: j,
                            step: rank,
                            source,
                        })
                    })?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: if n == 0 { 0 } else { basis_len / n },
                data: basis_data,
            },
            pivots: reduced.pivots,
        })
    }
}
