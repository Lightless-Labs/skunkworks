#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|i| i as u8)
}

fn encode_group(bytes: &[u8]) -> String {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    let mut out = String::with_capacity(4);
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    for chunk in input.chunks(3) {
        out.push_str(&encode_group(chunk));
    }
    out
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.tail.len() + input.len());
        data.extend_from_slice(&self.tail);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut out = String::with_capacity(complete_len / 3 * 4);
        for chunk in data[..complete_len].chunks(3) {
            out.push_str(&encode_group(chunk));
        }

        self.tail.clear();
        self.tail.extend_from_slice(&data[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        if self.tail.is_empty() {
            Ok(String::new())
        } else {
            let out = encode_group(&self.tail);
            self.tail.clear();
            Ok(out)
        }
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    value: Option<u8>,
    offset: usize,
}

pub struct Decoder {
    group: Vec<Symbol>,
    pending_padding: Option<usize>,
    terminal_padding: bool,
    offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            pending_padding: None,
            terminal_padding: false,
            offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    fn emit_group(group: &[Symbol], out: &mut Vec<u8>) {
        let a = group[0].value.unwrap();
        let b = group[1].value.unwrap();
        let c = group[2].value;
        let d = group[3].value;

        out.push((a << 2) | (b >> 4));
        if let Some(c) = c {
            out.push((b << 4) | (c >> 2));
            if let Some(d) = d {
                out.push((c << 6) | d);
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding {
                    offset: current_offset,
                });
            }

            if let Some(padding_offset) = self.pending_padding {
                if byte != b'=' {
                    return self.fail(Error::BadPadding {
                        offset: current_offset,
                    });
                }

                self.group.push(Symbol {
                    byte,
                    value: None,
                    offset: current_offset,
                });
                self.pending_padding = None;
                Self::emit_group(&self.group, &mut out);
                self.group.clear();
                self.terminal_padding = true;
                let _ = padding_offset;
                continue;
            }

            let position = self.group.len();
            if byte == b'=' {
                match position {
                    0 | 1 => {
                        return self.fail(Error::BadPadding {
                            offset: current_offset,
                        });
                    }
                    2 => {
                        self.group.push(Symbol {
                            byte,
                            value: None,
                            offset: current_offset,
                        });
                        self.pending_padding = Some(current_offset);
                    }
                    3 => {
                        self.group.push(Symbol {
                            byte,
                            value: None,
                            offset: current_offset,
                        });
                        Self::emit_group(&self.group, &mut out);
                        self.group.clear();
                        self.terminal_padding = true;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let value = match digit(byte) {
                Some(value) => value,
                None => {
                    return self.fail(Error::InvalidByte {
                        offset: current_offset,
                        byte,
                    });
                }
            };

            self.group.push(Symbol {
                byte,
                value: Some(value),
                offset: current_offset,
            });

            if self.group.len() == 4 {
                Self::emit_group(&self.group, &mut out);
                self.group.clear();
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            return self.fail(Error::Truncated {
                offset: self.group[0].offset,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
