#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_literal(word: &mut Word, c: char, wildcard_active: bool) {
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_string(word: &mut Word, value: &str, wildcard_active: bool) {
    for c in value.chars() {
        append_literal(word, c, wildcard_active);
    }
}

fn find_braced_end(input: &str, start: usize) -> Option<usize> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next()?;
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next()?;
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn parse_braced<'a>(
    input: &'a str,
    dollar: usize,
    content_start: usize,
    env: &[(&str, &str)],
) -> Result<(&'a str, usize, Option<&'a str>), SplitError> {
    let close = find_braced_end(input, content_start)
        .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let name_len = contents
        .char_indices()
        .take_while(|(_, c)| is_name_char(*c))
        .map(|(i, c)| i + c.len_utf8())
        .last()
        .unwrap_or(0);

    let name = &contents[..name_len];
    if name.is_empty() || !name.chars().next().map(is_name_start).unwrap_or(false) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let default = if name_len == contents.len() {
        None
    } else if contents[name_len..].starts_with(":-") {
        Some(&contents[name_len + 2..])
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    let value = env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value);
    let end = close + 1;
    Ok((name, end, value.filter(|v| !v.is_empty()).or_else(|| {
        if default.is_some() {
            value
        } else {
            value
        }
    })))
}

fn split_impl(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.text.push(next);
                        }
                        i = after_next;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let content_start = next_i + 1;
                        let close = find_braced_end(input, content_start)
                            .ok_or(SplitError::UnterminatedExpansion { pos: i })?;
                        let contents = &input[content_start..close];
                        let name_len = contents
                            .char_indices()
                            .take_while(|(_, ch)| is_name_char(*ch))
                            .map(|(idx, ch)| idx + ch.len_utf8())
                            .last()
                            .unwrap_or(0);
                        let name = &contents[..name_len];
                        if name.is_empty()
                            || !name.chars().next().map(is_name_start).unwrap_or(false)
                            || (name_len != contents.len()
                                && !contents[name_len..].starts_with(":-"))
                        {
                            return Err(SplitError::InvalidExpansion { pos: i });
                        }
                        let default = if name_len == contents.len() {
                            None
                        } else {
                            Some(&contents[name_len + 2..])
                        };
                        let found = env.iter().find(|(key, _)| *key == name).map(|(_, v)| *v);
                        let selected = match (found, default) {
                            (Some(v), _) if !v.is_empty() => v.to_string(),
                            (_, Some(d)) => decode_default(d),
                            (Some(v), None) => v.to_string(),
                            (None, None) => String::new(),
                        };
                        append_string(&mut word, &selected, false);
                        active = true;
                        i = close + 1;
                    } else {
                        let name_start = next_i;
                        let first = input[name_start..].chars().next();
                        if first.map(is_name_start).unwrap_or(false) {
                            let mut end = name_start;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_char(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[name_start..end];
                            if let Some((_, value)) = env.iter().find(|(key, _)| *key == name) {
                                word.text.push_str(value);
                            }
                            active = true;
                            i = end;
                        } else {
                            word.text.push('$');
                            active = true;
                            i = next_i;
                        }
                    }
                } else {
                    word.text.push(c);
                    i = next_i;
                }
                let _ = open;
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    if next == '\n' {
                        i = next_i + next.len_utf8();
                    } else {
                        append_literal(&mut word, next, false);
                        active = true;
                        i = next_i + next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let content_start = next_i + 1;
                        let close = find_braced_end(input, content_start)
                            .ok_or(SplitError::UnterminatedExpansion { pos: i })?;
                        let contents = &input[content_start..close];
                        let name_len = contents
                            .char_indices()
                            .take_while(|(_, ch)| is_name_char(*ch))
                            .map(|(idx, ch)| idx + ch.len_utf8())
                            .last()
                            .unwrap_or(0);
                        let name = &contents[..name_len];
                        if name.is_empty()
                            || !name.chars().next().map(is_name_start).unwrap_or(false)
                            || (name_len != contents.len()
                                && !contents[name_len..].starts_with(":-"))
                        {
                            return Err(SplitError::InvalidExpansion { pos: i });
                        }
                        let default = if name_len == contents.len() {
                            None
                        } else {
                            Some(&contents[name_len + 2..])
                        };
                        let found = env.iter().find(|(key, _)| *key == name).map(|(_, v)| *v);
                        let selected = match (found, default) {
                            (Some(v), _) if !v.is_empty() => v.to_string(),
                            (_, Some(d)) => decode_default(d),
                            (Some(v), None) => v.to_string(),
                            (None, None) => String::new(),
                        };
                        append_string(&mut word, &selected, true);
                        active = true;
                        i = close + 1;
                    } else {
                        let name_start = next_i;
                        let first = input[name_start..].chars().next();
                        if first.map(is_name_start).unwrap_or(false) {
                            let mut end = name_start;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_char(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[name_start..end];
                            if let Some((_, value)) = env.iter().find(|(key, _)| *key == name) {
                                append_string(&mut word, value, true);
                            }
                            active = true;
                            i = end;
                        } else {
                            word.text.push('$');
                            active = true;
                            i = next_i;
                        }
                    }
                } else {
                    append_literal(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    split_impl(input, env)
}
