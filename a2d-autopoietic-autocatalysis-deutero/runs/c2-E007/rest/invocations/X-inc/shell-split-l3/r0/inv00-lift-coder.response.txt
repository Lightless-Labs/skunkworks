#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_text(word: &mut Word, s: &str, active_glob: bool) {
    for c in s.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_text(&mut current, &input[i..next_i], false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if matches!(n, '"' | '\\' | '$') {
                        append_text(&mut current, &input[next_i..after_n], false);
                        i = after_n;
                    } else if n == '\n' {
                        i = after_n;
                    } else {
                        append_text(&mut current, "\\", false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let escaped = input[after..].chars().next().unwrap();
                                    j = after + escaped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(k) = contents.find(":-") {
                            (&contents[..k], Some(&contents[k + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_char)
                            || (default.is_none() && contents.contains(':'))
                            || (default.is_none() && contents.contains('-'))
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match (env_value(env, name), default) {
                            (Some(v), Some(d)) if !v.is_empty() => v.to_string(),
                            (_, Some(d)) => decode_default(d),
                            (Some(v), None) => v.to_string(),
                            (None, None) => String::new(),
                        };
                        append_text(&mut current, &value, false);
                        active = true;
                        i = close + 1;
                    } else {
                        let after_dollar = next_i;
                        if after_dollar < input.len() {
                            let n = input[after_dollar..].chars().next().unwrap();
                            if is_name_start(n) {
                                let mut end = after_dollar + n.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if is_name_char(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[after_dollar..end];
                                if let Some(value) = env_value(env, name) {
                                    append_text(&mut current, value, false);
                                }
                                active = true;
                                i = end;
                            } else {
                                append_text(&mut current, "$", false);
                                active = true;
                                i = next_i;
                            }
                        } else {
                            append_text(&mut current, "$", false);
                            active = true;
                            i = next_i;
                        }
                    }
                } else {
                    append_text(&mut current, &input[i..next_i], false);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' || c == '"' {
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '\n' {
                        i = after_n;
                    } else {
                        append_text(&mut current, &input[next_i..after_n], false);
                        active = true;
                        i = after_n;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let escaped = input[after..].chars().next().unwrap();
                                    j = after + escaped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(k) = contents.find(":-") {
                            (&contents[..k], Some(&contents[k + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_char)
                            || (default.is_none() && contents.contains(':'))
                            || (default.is_none() && contents.contains('-'))
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match (env_value(env, name), default) {
                            (Some(v), Some(_)) if !v.is_empty() => v.to_string(),
                            (_, Some(d)) => decode_default(d),
                            (Some(v), None) => v.to_string(),
                            (None, None) => String::new(),
                        };
                        append_text(&mut current, &value, true);
                        active = true;
                        i = close + 1;
                    } else {
                        let after_dollar = next_i;
                        if after_dollar < input.len() {
                            let n = input[after_dollar..].chars().next().unwrap();
                            if is_name_start(n) {
                                let mut end = after_dollar + n.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if is_name_char(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[after_dollar..end];
                                if let Some(value) = env_value(env, name) {
                                    append_text(&mut current, value, true);
                                }
                                active = true;
                                i = end;
                            } else {
                                append_text(&mut current, "$", false);
                                active = true;
                                i = next_i;
                            }
                        } else {
                            append_text(&mut current, "$", false);
                            active = true;
                            i = next_i;
                        }
                    }
                } else {
                    append_text(&mut current, &input[i..next_i], matches!(c, '*' | '?' | '['));
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}
