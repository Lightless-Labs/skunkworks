#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_char(word: &mut Word, c: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_str(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_char(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let mut i = content_start;
    let mut closing = None;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                i += input[i..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            closing = Some(i);
            break;
        } else {
            i += c.len_utf8();
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[content_start..close];

    let mut name_end = 0;
    for c in content.chars() {
        if name_end == 0 {
            if !is_name_start(c) {
                return Err(SplitError::InvalidExpansion { pos: dollar });
            }
        } else if !is_name_continue(c) {
            break;
        }
        name_end += c.len_utf8();
    }

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &content[..name_end];
    let remainder = &content[name_end..];

    let value = if remainder.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current: Option<Word> = None;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(current.as_mut().unwrap(), c, false);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        append_char(current.as_mut().unwrap(), next, false);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        append_char(current.as_mut().unwrap(), '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() && input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let (end, value) = expand_braced(input, i, env)?;
                        append_str(current.as_mut().unwrap(), &value, false);
                        i = end;
                    } else if next_i < input.len()
                        && is_name_start(input[next_i..].chars().next().unwrap())
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_i..end];
                        let value = lookup(env, name).unwrap_or("");
                        append_str(current.as_mut().unwrap(), value, false);
                        i = end;
                    } else {
                        append_char(current.as_mut().unwrap(), '$', false);
                        i = next_i;
                    }
                } else {
                    append_char(current.as_mut().unwrap(), c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if let Some(word) = current.take() {
                        words.push(word);
                    }
                    i = next_i;
                } else if c == '\'' {
                    current.get_or_insert_with(|| Word {
                        text: String::new(),
                        glob: Vec::new(),
                    });
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    current.get_or_insert_with(|| Word {
                        text: String::new(),
                        glob: Vec::new(),
                    });
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        let word = current.get_or_insert_with(|| Word {
                            text: String::new(),
                            glob: Vec::new(),
                        });
                        append_char(word, next, false);
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() && input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let word = current.get_or_insert_with(|| Word {
                        text: String::new(),
                        glob: Vec::new(),
                    });
                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let (end, value) = expand_braced(input, i, env)?;
                        append_str(word, &value, true);
                        i = end;
                    } else if next_i < input.len()
                        && is_name_start(input[next_i..].chars().next().unwrap())
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_i..end];
                        let value = lookup(env, name).unwrap_or("");
                        append_str(word, value, true);
                        i = end;
                    } else {
                        append_char(word, '$', false);
                        i = next_i;
                    }
                } else {
                    let word = current.get_or_insert_with(|| Word {
                        text: String::new(),
                        glob: Vec::new(),
                    });
                    append_char(word, c, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if let Some(word) = current {
        words.push(word);
    }

    Ok(words)
}
