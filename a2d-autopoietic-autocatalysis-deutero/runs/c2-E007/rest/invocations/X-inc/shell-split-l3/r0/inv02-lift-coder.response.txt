#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_atomic(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn find_braced_end(input: &str, start: usize) -> Option<usize> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next()?;
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next()?;
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                result.push(next);
                i += next.len_utf8();
            }
        } else {
            result.push(c);
            i += c.len_utf8();
        }
    }
    result
}

fn first_env<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let len = c.len_utf8();

        match quote {
            Quote::Single(open) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    current.text.push(c);
                }
                i += len;
                let _ = open;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i += len;
                } else if c == '\\' {
                    let slash_pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    match next {
                        '"' | '\\' | '$' => {
                            current.text.push(next);
                            i += next.len_utf8();
                        }
                        '\n' => {
                            i += 1;
                        }
                        _ => {
                            current.text.push('\\');
                        }
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let dollar_pos = i;
                    let next_pos = i + 1;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: dollar_pos });
                    }
                    if input[next_pos..].starts_with('{') {
                        let content_start = next_pos + 1;
                        let Some(close) = find_braced_end(input, content_start) else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
                        };
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(sep) = contents.find(":-") {
                            (&contents[..sep], Some(&contents[sep + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty()
                            || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
                        }
                        let selected = match default {
                            Some(text) => match first_env(name, env) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(text),
                            },
                            None => first_env(name, env).unwrap_or("").to_string(),
                        };
                        append_atomic(&mut current, &selected, false);
                        active = true;
                        i = close + 1;
                    } else if next_pos < input.len() {
                        let first = input[next_pos..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_pos + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_char(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            append_atomic(
                                &mut current,
                                first_env(&input[next_pos..end], env).unwrap_or(""),
                                false,
                            );
                            active = true;
                            i = end;
                        } else {
                            current.text.push('$');
                            i += 1;
                        }
                    } else {
                        current.text.push('$');
                        i += 1;
                    }
                } else {
                    current.text.push(c);
                    i += len;
                }
                let _ = open;
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += len;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i += len;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i += len;
                } else if c == '\\' {
                    let slash_pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        current.text.push(next);
                        active = true;
                    }
                    i += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let dollar_pos = i;
                    let next_pos = i + 1;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: dollar_pos });
                    }
                    if input[next_pos..].starts_with('{') {
                        let content_start = next_pos + 1;
                        let Some(close) = find_braced_end(input, content_start) else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
                        };
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(sep) = contents.find(":-") {
                            (&contents[..sep], Some(&contents[sep + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty()
                            || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
                        }
                        let selected = match default {
                            Some(text) => match first_env(name, env) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(text),
                            },
                            None => first_env(name, env).unwrap_or("").to_string(),
                        };
                        append_atomic(&mut current, &selected, true);
                        active = true;
                        i = close + 1;
                    } else if next_pos < input.len() {
                        let first = input[next_pos..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_pos + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_char(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            append_atomic(
                                &mut current,
                                first_env(&input[next_pos..end], env).unwrap_or(""),
                                true,
                            );
                            active = true;
                            i = end;
                        } else {
                            current.text.push('$');
                            active = true;
                            i += 1;
                        }
                    } else {
                        current.text.push('$');
                        active = true;
                        i += 1;
                    }
                } else {
                    append_literal(&mut current, c, true);
                    active = true;
                    i += len;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }

    Ok(words)
}
