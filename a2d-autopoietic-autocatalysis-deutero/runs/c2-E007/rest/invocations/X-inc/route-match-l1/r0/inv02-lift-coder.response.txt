#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { id, segments });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let decoded = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>, Vec<u8>)> = None;

        for route in &self.routes {
            if route.segments.len() != decoded.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut rank = Vec::with_capacity(route.segments.len());
            let mut matches = true;

            for (pattern_segment, path_segment) in route.segments.iter().zip(&decoded) {
                match pattern_segment {
                    Segment::Static(value) => {
                        rank.push(2);
                        if value != path_segment {
                            matches = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        rank.push(1);
                        params.push((name.clone(), path_segment.clone()));
                    }
                }
            }

            if !matches {
                continue;
            }

            let replace = best
                .as_ref()
                .map(|(_, _, best_rank)| rank > *best_rank)
                .unwrap_or(true);

            if replace {
                best = Some((route, params, rank));
            }
        }

        match best {
            Some((route, params, _)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    for end in 1..=bytes.len() {
        if end == bytes.len() || bytes[end] == b'/' {
            if end == start {
                return Err(Error::InvalidPattern { at: start });
            }

            let raw = &pattern[start..end];
            let first = bytes[start];

            if first == b':' {
                let name = &raw[1..];
                if !valid_param_name(name) || !names.insert(name.to_owned()) {
                    return Err(Error::InvalidPattern { at: start });
                }
                segments.push(Segment::Param(name.to_owned()));
            } else if first == b'*' {
                return Err(Error::InvalidPattern { at: start });
            } else {
                segments.push(Segment::Static(decode_segment(bytes, start, end, true)?));
            }

            if end == bytes.len() {
                break;
            }
            start = end + 1;
        }
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;

    for end in 1..=bytes.len() {
        if end == bytes.len() || bytes[end] == b'/' {
            if end == start {
                return Err(Error::InvalidPath { at: start });
            }
            segments.push(decode_segment(bytes, start, end, false)?);
            if end == bytes.len() {
                break;
            }
            start = end + 1;
        }
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn decode_segment(
    bytes: &[u8],
    start: usize,
    end: usize,
    pattern: bool,
) -> Result<String, Error> {
    let mut decoded = Vec::new();
    let mut i = start;

    while i < end {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= end
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(invalid(pattern, i));
            }

            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(pattern, i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid(pattern, i));
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn valid_param_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    if byte.is_ascii_digit() {
        byte - b'0'
    } else {
        byte - b'A' + 10
    }
}

fn invalid(pattern: bool, at: usize) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}
