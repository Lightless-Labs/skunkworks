#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|cell| cell == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|cell| cell == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Quoted { opening } => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
                let _ = opening;
                record_started = true;
            }
            State::FieldStart | State::Unquoted | State::AfterQuote => {
                if b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
                    return Err(CsvError::BareCr {
                        record: record_index,
                        byte: i,
                    });
                }

                match state {
                    State::FieldStart => {
                        if b == quote {
                            state = State::Quoted { opening: i };
                            record_started = true;
                            i += 1;
                        } else if b == delimiter {
                            row.push(std::mem::take(&mut field));
                            record_started = true;
                            i += 1;
                        } else if b == b'\n' || b == b'\r' {
                            row.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut row));
                            record_index += 1;
                            record_started = false;
                            state = State::FieldStart;
                            i += if b == b'\r' { 2 } else { 1 };
                        } else {
                            if b == b'"' {
                                return Err(CsvError::UnexpectedQuote {
                                    record: record_index,
                                    byte: i,
                                });
                            }
                            let ch = input[i..].chars().next().unwrap();
                            field.push(ch);
                            state = State::Unquoted;
                            record_started = true;
                            i += ch.len_utf8();
                        }
                    }
                    State::Unquoted => {
                        if b == quote {
                            return Err(CsvError::UnexpectedQuote {
                                record: record_index,
                                byte: i,
                            });
                        } else if b == delimiter {
                            row.push(std::mem::take(&mut field));
                            state = State::FieldStart;
                            record_started = true;
                            i += 1;
                        } else if b == b'\n' || b == b'\r' {
                            row.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut row));
                            record_index += 1;
                            record_started = false;
                            state = State::FieldStart;
                            i += if b == b'\r' { 2 } else { 1 };
                        } else {
                            let ch = input[i..].chars().next().unwrap();
                            field.push(ch);
                            record_started = true;
                            i += ch.len_utf8();
                        }
                    }
                    State::AfterQuote => {
                        if b == delimiter {
                            row.push(std::mem::take(&mut field));
                            state = State::FieldStart;
                            record_started = true;
                            i += 1;
                        } else if b == b'\n' || b == b'\r' {
                            row.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut row));
                            record_index += 1;
                            record_started = false;
                            state = State::FieldStart;
                            i += if b == b'\r' { 2 } else { 1 };
                        } else {
                            return Err(CsvError::TrailingByte {
                                record: record_index,
                                byte: i,
                            });
                        }
                    }
                    State::Quoted { .. } => unreachable!(),
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started || !row.is_empty() || !field.is_empty() {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let mut records = parse_with_dialect(input, config.delimiter, config.quote)?;

    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(names) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..names.len() {
                if let Some(first) = names[..second].iter().position(|name| name == &names[second]) {
                    return Err(CsvError::DuplicateHeader {
                        name: names[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }
        for (index, row) in records.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> Option<i128> {
    let b = s.as_bytes();
    let (negative, rest) = if b.first() == Some(&b'-') {
        (true, &b[1..])
    } else {
        (false, b)
    };
    let dot = rest.iter().position(|&c| c == b'.')?;
    let whole = &rest[..dot];
    let frac = &rest[dot + 1..];
    if whole.is_empty()
        || frac.len() != 2
        || !frac.iter().all(u8::is_ascii_digit)
        || !(whole == b"0"
            || (whole[0].is_ascii_digit()
                && whole[0] != b'0'
                && whole.iter().all(u8::is_ascii_digit)))
    {
        return None;
    }
    if negative && whole == b"0" && frac == b"00" {
        return None;
    }
    let mut magnitude: i128 = 0;
    for &c in whole.iter().chain(frac.iter()) {
        magnitude = magnitude.checked_mul(10)?.checked_add((c - b'0') as i128)?;
    }
    Some(if negative { -magnitude } else { magnitude })
}

fn convert_cell(
    cell: &str,
    kind: ColumnType,
    nullable: bool,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => {
            if !integer_syntax(cell) {
                return Err(CellErrorKind::InvalidInteger);
            }
            match cell.parse::<i64>() {
                Ok(value) => Ok(Value::Integer(value)),
                Err(_) => Err(CellErrorKind::IntegerOverflow),
            }
        }
        ColumnType::Decimal => {
            let scaled = decimal_syntax(cell).ok_or(CellErrorKind::InvalidDecimal)?;
            if scaled < i64::MIN as i128 || scaled > i64::MAX as i128 {
                return Err(CellErrorKind::DecimalOverflow);
            }
            Ok(Value::Decimal(scaled as i64))
        }
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut had_error = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if had_error && policy != ErrorPolicy::Collect {
                output.push(None);
                continue;
            }

            let (resolved, unknown) = match &spec.column {
                ColumnRef::Index(index) => (Some(*index), false),
                ColumnRef::Name(name) => match table.column_index(name) {
                    Some(index) => (Some(index), false),
                    None => (None, true),
                },
            };

            let error = if unknown {
                Some((None, CellErrorKind::UnknownColumn))
            } else {
                let index = resolved.unwrap();
                match row.get(index) {
                    None => Some((Some(index), CellErrorKind::MissingCell)),
                    Some(cell) => match convert_cell(cell, spec.kind, spec.nullable) {
                        Ok(value) => {
                            output.push(Some(value));
                            continue;
                        }
                        Err(kind) => Some((Some(index), kind)),
                    },
                }
            };

            let (column, kind) = error.unwrap();
            output.push(None);
            result.errors.push(CellError {
                row: row_index,
                spec: spec_index,
                column,
                kind,
            });
            had_error = true;
            if policy == ErrorPolicy::Stop {
                for _ in spec_index + 1..columns.len() {
                    output.push(None);
                }
                result.rows.push(output);
                return result;
            }
            if policy == ErrorPolicy::FirstPerRow {
                for _ in spec_index + 1..columns.len() {
                    output.push(None);
                }
                break;
            }
        }

        result.rows.push(output);
    }

    result
}
