#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum ParseState {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_config(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = ParseState::FieldStart;
    let mut quote_open = 0usize;
    let mut record_index = 0usize;
    let mut i = start;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != ParseState::Quoted && byte == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }

            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            state = ParseState::FieldStart;
            i += 2;
            continue;
        }

        match state {
            ParseState::FieldStart => {
                if byte == quote {
                    quote_open = i;
                    state = ParseState::Quoted;
                    i += 1;
                } else if byte == delimiter {
                    record.push(String::new());
                    i += 1;
                } else if byte == b'\n' {
                    record.push(String::new());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    i += 1;
                } else {
                    field.push(byte as char);
                    state = ParseState::Unquoted;
                    i += 1;
                }
            }
            ParseState::Unquoted => {
                if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = ParseState::FieldStart;
                    i += 1;
                } else if byte == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = ParseState::FieldStart;
                    i += 1;
                } else {
                    field.push(byte as char);
                    i += 1;
                }
            }
            ParseState::Quoted => {
                if byte == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = ParseState::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(byte as char);
                    i += 1;
                }
            }
            ParseState::AfterQuote => {
                if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = ParseState::FieldStart;
                    i += 1;
                } else if byte == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = ParseState::FieldStart;
                    i += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if state == ParseState::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if !record.is_empty() || !field.is_empty() || state != ParseState::FieldStart {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_config(input, b',', b'"')
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let mut parsed = parse_with_config(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers && !parsed.is_empty() {
        let header = parsed.remove(0);

        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..header.len() {
                if let Some(first) = header[..second]
                    .iter()
                    .position(|value| value == &header[second])
                {
                    return Err(CsvError::DuplicateHeader {
                        name: header[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }

        for (row_index, row) in parsed.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }

        (Some(header), parsed)
    } else {
        (None, parsed)
    };

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = build_table(&self.input, self.config);
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    MissingCell,
    Empty,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' {
        &bytes[1..]
    } else {
        bytes
    };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn parse_integer(value: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(value) {
        return Err(CellErrorKind::InvalidInteger);
    }
    value
        .parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn decimal_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    let body = if bytes.first() == Some(&b'-') {
        &bytes[1..]
    } else {
        bytes
    };
    let Some(dot) = body.iter().position(|&byte| byte == b'.') else {
        return false;
    };
    if body[dot + 1..].len() != 2
        || !body[dot + 1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    let integral = &body[..dot];
    if integral.is_empty() {
        return false;
    }
    if integral == b"0" {
        return bytes.first() != Some(&b'-') || body[dot + 1..] != *b"00";
    }
    integral[0].is_ascii_digit()
        && integral[0] != b'0'
        && integral.iter().all(u8::is_ascii_digit)
}

fn parse_decimal(value: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(value) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = value.starts_with('-');
    let body = if negative { &value[1..] } else { value };
    let dot = body.find('.').unwrap();
    let integral = body[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let fractional = body[dot + 1..]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let magnitude = integral
        .checked_mul(100)
        .and_then(|n| n.checked_add(fractional))
        .ok_or(CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn convert_cell(value: &str, kind: ColumnType) -> Result<Value, CellErrorKind> {
    match kind {
        ColumnType::Text => Ok(Value::Text(value.to_string())),
        ColumnType::Integer => parse_integer(value).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(value).map(Value::Decimal),
        ColumnType::Boolean => match value {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output_row = Vec::with_capacity(columns.len());
        let mut stopped = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if stopped {
                output_row.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let result = if resolved.is_none() {
                Err((None, CellErrorKind::UnknownColumn))
            } else {
                let column = resolved.unwrap();
                match row.get(column) {
                    None => Err((Some(column), CellErrorKind::MissingCell)),
                    Some(cell) if cell.is_empty() && spec.nullable => Ok(Value::Null),
                    Some(cell) if cell.is_empty() && spec.kind != ColumnType::Text => {
                        Err((Some(column), CellErrorKind::Empty))
                    }
                    Some(cell) => convert_cell(cell, spec.kind)
                        .map_err(|kind| (Some(column), kind)),
                }
            };

            match result {
                Ok(value) => output_row.push(Some(value)),
                Err((column, kind)) => {
                    output.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    output_row.push(None);
                    if policy != ErrorPolicy::Collect {
                        stopped = true;
                    }
                }
            }
        }

        output.rows.push(output_row);
        if policy == ErrorPolicy::Stop && !output.errors.is_empty() {
            break;
        }
    }

    output
}
