#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Field {
    selected: Vec<bool>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minute: Field,
    hour: Field,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        for (i, b) in text.bytes().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = text
            .bytes()
            .enumerate()
            .filter_map(|(i, b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == text.len() {
            return Err(CronError::Layout(split));
        }

        let minute = parse_field(&text[..split], 0, 59)?;
        let hour = parse_field(&text[split + 1..], split + 1, 23)?;
        Ok(Self { minute, hour })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minute.selected[(minute % 60) as usize]
            && self.hour.selected[((minute / 60) % 24) as usize]
    }
}

fn syntax_non_digit(fragment: &str, start: usize) -> Option<CronError> {
    if fragment.is_empty() {
        return Some(CronError::Syntax(start));
    }
    for (i, b) in fragment.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Some(CronError::Syntax(start + i));
        }
    }
    None
}

fn parse_number(fragment: &str, start: usize, max: usize) -> Result<usize, CronError> {
    if let Some(err) = syntax_non_digit(fragment, start) {
        return Err(err);
    }
    if fragment.len() > 1 && fragment.starts_with('0') {
        return Err(CronError::Number(start));
    }
    let mut value = 0usize;
    for b in fragment.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
    }
    if value > max {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

fn parse_field(text: &str, offset: usize, max: usize) -> Result<Field, CronError> {
    let mut selected = vec![false; max + 1];
    let mut item_start = 0;

    loop {
        let end = text[item_start..]
            .find(',')
            .map(|n| item_start + n)
            .unwrap_or(text.len());
        if end == item_start {
            return Err(CronError::Syntax(offset + item_start));
        }

        parse_item(
            &text[item_start..end],
            offset + item_start,
            max,
            &mut selected,
        )?;

        if end == text.len() {
            break;
        }
        item_start = end + 1;
        if item_start == text.len() {
            return Err(CronError::Syntax(offset + item_start));
        }
    }

    Ok(Field { selected })
}

fn parse_item(
    item: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let slash_positions: Vec<usize> = item
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(CronError::Syntax(offset + slash_positions[1]));
    }

    let slash = slash_positions.first().copied();
    let (base, step_fragment, step_start) = match slash {
        Some(i) => (&item[..i], Some(&item[i + 1..]), offset + i + 1),
        None => (item, None, 0),
    };

    let is_star = base == "*";
    let dash = base.find('-');

    if slash.is_some() && !is_star && dash.is_none() {
        return Err(CronError::Syntax(offset + slash.unwrap()));
    }

    let (start, end, dash_offset) = if is_star {
        (0, max, None)
    } else if let Some(dash_index) = dash {
        let left = parse_number(&base[..dash_index], offset, max)?;
        let right = parse_number(&base[dash_index + 1..], offset + dash_index + 1, max)?;
        if left > right {
            return Err(CronError::Descending(offset + dash_index));
        }
        (left, right, Some(offset + dash_index))
    } else {
        let n = parse_number(base, offset, max)?;
        (n, n, None)
    };

    let step = if let Some(fragment) = step_fragment {
        parse_number(fragment, step_start, max)?
    } else {
        1
    };

    if step == 0 || step > max + 1 {
        return Err(CronError::Step(step_start));
    }

    let _ = dash_offset;
    let mut value = start;
    while value <= end {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|reserved| reserved == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let owned_id = id.to_string();
        let order = self.jobs.len();
        self.reserved.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll;
        let first_minute = match lower {
            Some(last) => last / 60_000 + 1,
            None => 0,
        };
        let last_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if first_minute <= last_minute {
            for job in &self.jobs {
                match job.policy {
                    Policy::All => {
                        for minute in first_minute..=last_minute {
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                    Policy::Once => {
                        if let Some(minute) = (first_minute..=last_minute)
                            .rev()
                            .find(|&minute| job.cron.matches(minute))
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0 {
                            let minute = now / 60_000;
                            if minute >= first_minute
                                && minute <= last_minute
                                && job.cron.matches(minute)
                            {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: now,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
