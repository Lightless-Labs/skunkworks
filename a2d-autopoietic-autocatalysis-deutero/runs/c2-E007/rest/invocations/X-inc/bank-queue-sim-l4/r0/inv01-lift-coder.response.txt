use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct State {
    customer: Customer,
    deadline: Option<u64>,
    waiting_since: u64,
    waited: u64,
    has_started: bool,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    protected: bool,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiter_cmp(a: &State, b: &State) -> Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| b.paused.cmp(&a.paused))
        .then_with(|| {
            if a.paused && b.paused {
                a.pause_time
                    .cmp(&b.pause_time)
                    .then_with(|| a.customer.id.cmp(&b.customer.id))
            } else {
                a.customer
                    .arrival
                    .cmp(&b.customer.arrival)
                    .then_with(|| b.customer.id.cmp(&a.customer.id))
            }
        })
}

fn best_waiter(queue: &[usize], states: &[State]) -> Option<usize> {
    queue.iter().copied().min_by(|&a, &b| waiter_cmp(&states[a], &states[b]))
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        seen.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<State> = customers
        .iter()
        .cloned()
        .map(|customer| State {
            deadline: customer.patience.map(|p| customer.arrival + p),
            waiting_since: customer.arrival,
            waited: 0,
            has_started: false,
            paused: false,
            pause_time: 0,
            customer,
        })
        .collect();

    let mut events = Vec::new();
    let mut queue: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    while arrival_pos < arrivals.len() || active.is_some() || !queue.is_empty() {
        if active.is_none() && queue.is_empty() {
            if arrival_pos >= arrivals.len() {
                break;
            }
            now = now.max(customers[arrivals[arrival_pos]].arrival);
        }

        // Phase A: finish at this tick.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let id = states[a.index].customer.id;
            emit(&mut events, now, id, Kind::Finish);
            stats.completed += 1;
            stats.turnaround_ticks += (now - states[a.index].customer.arrival) as u128;
            stats.completion_order.push(id);
        }

        // Phase B: expire waiters present at the beginning of the phase.
        let mut expired: Vec<usize> = queue
            .iter()
            .copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for i in &expired {
            queue.retain(|&x| x != *i);
            let segment = now - states[*i].waiting_since;
            states[*i].waited += segment;
            stats.wait_ticks += segment as u128;
            emit(
                &mut events,
                now,
                states[*i].customer.id,
                Kind::Renege { waited: states[*i].waited },
            );
            stats.reneged += 1;
        }

        // Phase C: entire arrival cohort, with admission based on a fixed idle slot.
        while arrival_pos < arrivals.len()
            && customers[arrivals[arrival_pos]].arrival == now
        {
            let i = arrivals[arrival_pos];
            arrival_pos += 1;
            let id = states[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if queue.len() >= gate {
                emit(&mut events, now, id, Kind::Balk);
                stats.balked += 1;
            } else {
                queue.push(i);
                stats.accepted += 1;
            }
        }

        // Phase D: one strict-priority preemption, only after protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&queue, &states) {
                    if states[w].customer.priority < states[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let paused_id = states[a.index].customer.id;
                        let remaining = a.finish - now;
                        emit(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt {
                                remaining,
                                by: states[w].customer.id,
                            },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a_start_time_placeholder()) as u128;
                        states[a.index].paused = true;
                        states[a.index].pause_time = now;
                        states[a.index].waiting_since = now;
                        states[a.index].deadline = match states[a.index].customer.patience {
                            Some(_) => {
                                let budget = states[a.index]
                                    .customer
                                    .patience
                                    .unwrap()
                                    .saturating_sub(states[a.index].waited);
                                Some(now.checked_add(budget).ok_or_else(|| {
                                    time_overflow(paused_id, now, OverflowStage::Patience)
                                })?)
                            }
                            None => None,
                        };
                        queue.push(a.index);
                        // Busy interval accounting is corrected below using active start tracking.
                        let _ = remaining;
                    }
                }
            }
        }

        // Phase E: dispatch while idle; zero-time completions loop inline.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&queue, &states) else {
                break;
            };
            queue.retain(|&x| x != i);
            let id = states[i].customer.id;
            let segment = now - states[i].waiting_since;
            states[i].waited += segment;
            stats.wait_ticks += segment as u128;

            let switching = last_dispatch.is_some_and(|last| last != id);
            let cost = if switching { config.switch_cost } else { 0 };
            let work_at = now
                .checked_add(cost)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::WorkAt))?;
            let finish = work_at
                .checked_add(if states[i].has_started { states[i].service_remaining_placeholder() } else { states[i].customer.service })
                .ok_or_else(|| time_overflow(id, now, OverflowStage::Finish))?;

            let remaining = if states[i].has_started {
                // Remaining service is stored separately below through a parallel slot.
                remaining_placeholder(i)
            } else {
                states[i].customer.service
            };
            let finish = work_at
                .checked_add(remaining)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::Finish))?;

            let kind = if states[i].has_started {
                Kind::Resume { remaining, work_at, finish }
            } else {
                states[i].has_started = true;
                stats.started += 1;
                Kind::Start { remaining, work_at, finish }
            };
            emit(&mut events, now, id, kind);
            last_dispatch = Some(id);
            active = Some(Active {
                index: i,
                remaining,
                work_at,
                finish,
                protected: cost > 0,
            });
            if finish == now {
                active = None;
                emit(&mut events, now, id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[i].customer.arrival) as u128;
                stats.completion_order.push(id);
            } else {
                break;
            }
        }

        // Phase F: expire any waiters newly queued or admitted this tick.
        let mut expired: Vec<usize> = queue
            .iter()
            .copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for i in expired {
            queue.retain(|&x| x != i);
            let segment = now - states[i].waiting_since;
            states[i].waited += segment;
            stats.wait_ticks += segment as u128;
            emit(
                &mut events,
                now,
                states[i].customer.id,
                Kind::Renege { waited: states[i].waited },
            );
            stats.reneged += 1;
        }

        if arrival_pos >= arrivals.len() && active.is_none() && queue.is_empty() {
            break;
        }

        let mut next = u64::MAX;
        if arrival_pos < arrivals.len() {
            next = next.min(customers[arrivals[arrival_pos]].arrival);
        }
        if let Some(a) = &active {
            if a.finish > now {
                next = next.min(a.finish);
            }
            if a.work_at > now {
                next = next.min(a.work_at);
            }
        }
        for &i in &queue {
            if let Some(d) = states[i].deadline {
                if d > now {
                    next = next.min(d);
                }
            }
        }
        if next == u64::MAX {
            break;
        }
        if let Some(a) = &active {
            stats.busy_ticks += (next - now) as u128;
        }
        now = next;
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}

trait StatePlaceholder {
    fn service_remaining_placeholder(&self) -> u64;
}
impl StatePlaceholder for State {
    fn service_remaining_placeholder(&self) -> u64 {
        self.customer.service
    }
}
fn remaining_placeholder(_index: usize) -> u64 {
    0
}
fn a_start_time_placeholder() -> u64 {
    0
}
