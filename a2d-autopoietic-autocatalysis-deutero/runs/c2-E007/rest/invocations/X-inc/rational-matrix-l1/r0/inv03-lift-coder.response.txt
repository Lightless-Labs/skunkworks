```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let num = -(self.num as i128);
        if num < i64::MIN as i128 || num > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: num as i64, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        use ParseKind::*;

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }

        let numerator_start = 0;
        let numerator_digits_start = usize::from(bytes[0] == b'-');
        let mut numerator_end = i;

        if i == bytes.len() || !bytes[i].is_ascii_digit() {
            return Err(parse_error(i, InvalidByte));
        }
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        numerator_end = i;

        let mut slash_at = None;
        let mut denominator_start = None;
        let mut denominator_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, InvalidByte));
            }
            slash_at = Some(i);
            i += 1;
            denominator_start = Some(i);

            if i == bytes.len() || !bytes[i].is_ascii_digit() {
                return Err(parse_error(i, InvalidByte));
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            denominator_end = Some(i);

            if i < bytes.len() {
                return Err(parse_error(i, InvalidByte));
            }
        }

        let num_token = &bytes[numerator_start..numerator_end];
        let negative = num_token[0] == b'-';
        let num_digits = &bytes[numerator_digits_start..numerator_end];

        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_error(numerator_digits_start + 1, NonCanonical));
        }
        if negative && num_digits == b"0" {
            return Err(parse_error(0, NonCanonical));
        }

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, NonCanonical));
            }
        }

        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_mag = parse_magnitude(num_digits);
        if num_mag > num_limit {
            return Err(parse_error(0, OutOfRange));
        }

        let denominator_value = if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            let magnitude = parse_magnitude(&bytes[start..end]);
            if magnitude > i64::MAX as u128 {
                return Err(parse_error(start, OutOfRange));
            }
            Some((start, magnitude as i64))
        } else {
            None
        };

        if let Some((start, 0)) = denominator_value {
            return Err(parse_error(start, ZeroDenominator));
        }

        if let (Some(slash), Some((_, 1))) = (slash_at, denominator_value) {
            return Err(parse_error(slash, NonCanonical));
        }
        if let (Some(slash), Some(_)) = (slash_at, denominator_value) {
            if num_mag == 0 {
                return Err(parse_error(slash, NonCanonical));
            }
        }

        if let (Some(slash), Some((_, den))) = (slash_at, denominator_value) {
            if gcd_u128(num_mag, den as u128) > 1 {
                return Err(parse_error(slash, NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        match denominator_value {
            Some((_, den)) => Ok(Self { num: signed_num, den }),
            None => Ok(Self { num: signed_num, den: 1 }),
        }
    }

    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = -num;
            den = -den;
        }

        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let gcd = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= gcd;
        den /= gcd;

        if num < i64::MIN as i128 || num > i64::MAX as i128 || den < 1 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }

        Ok(Self { num: num as i64, den: den as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(digits: &[u8]) -> u128 {
    let mut value = 0u128;
    for &digit in digits {
        value = value.saturating_mul(10).saturating_add((digit - b'0') as u128);
    }
    value
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}
```
