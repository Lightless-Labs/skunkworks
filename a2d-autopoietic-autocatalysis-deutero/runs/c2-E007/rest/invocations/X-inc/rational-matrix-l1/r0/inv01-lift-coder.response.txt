```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (num, den) = if den < 0 {
        (num.checked_neg().ok_or(RationalError::Overflow)?, -den)
    } else {
        (num, den)
    };

    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    let num = num / g;
    let den = den / g;

    if num < i64::MIN as i128
        || num > i64::MAX as i128
        || den < 1
        || den > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: num as i64,
        den: den as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left = a * (d / g);
        let right = c * (b / g);
        from_parts(left + right, (b / g) * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left = a * (d / g);
        let right = c * (b / g);
        from_parts(left - right, (b / g) * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(c.unsigned_abs(), b as u128) as i128;
        from_parts((a / g1) * (c / g2), (b / g2) * (d / g1))
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g1 = gcd_u128(a.unsigned_abs(), c.unsigned_abs()) as i128;
        let g2 = gcd_u128(d as u128, b as u128) as i128;
        from_parts((a / g1) * (d / g2), (b / g2) * (c / g1))
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        from_parts(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| Err(RationalError::Parse { at, kind });

        if bytes.is_empty() {
            return fail(0, ParseKind::Empty);
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return fail(i, ParseKind::InvalidByte);
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;

        if i < bytes.len() && bytes[i] == b'/' {
            i += 1;
            let start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return fail(i, ParseKind::InvalidByte);
            }
            den_start = Some(start);
            den_end = Some(i);
        }

        if i < bytes.len() {
            return fail(i, ParseKind::InvalidByte);
        }

        let num_digits = num_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return fail(num_start + 1, ParseKind::NonCanonical);
        }
        if negative && num_digits == 1 && bytes[num_start] == b'0' {
            return fail(0, ParseKind::NonCanonical);
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return fail(start + 1, ParseKind::NonCanonical);
            }
        }

        let mut num_mag: u128 = 0;
        for &b in &bytes[num_start..num_end] {
            num_mag = num_mag * 10 + (b - b'0') as u128;
            let limit = if negative {
                1u128 << 63
            } else {
                i64::MAX as u128
            };
            if num_mag > limit {
                return fail(0, ParseKind::OutOfRange);
            }
        }

        let mut den_mag: u128 = 1;
        if let (Some(start), Some(end)) = (den_start, den_end) {
            den_mag = 0;
            for &b in &bytes[start..end] {
                den_mag = den_mag * 10 + (b - b'0') as u128;
                if den_mag > i64::MAX as u128 {
                    return fail(start, ParseKind::OutOfRange);
                }
            }
            if den_mag == 0 {
                return fail(start, ParseKind::ZeroDenominator);
            }
            if den_mag == 1 {
                return fail(start - 1, ParseKind::NonCanonical);
            }
            if num_mag == 0 {
                return fail(start - 1, ParseKind::NonCanonical);
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return fail(start - 1, ParseKind::NonCanonical);
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        Ok(Rational {
            num: signed_num,
            den: den_mag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
