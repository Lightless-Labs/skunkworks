#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if (2..=36).contains(&value) {
        Ok(())
    } else {
        Err(BaseError::BadBase { which, value })
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input[..hash] != expected {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u32>, multiplier: u32, addend: u32) {
    let mut carry = addend;
    for limb in value.iter_mut() {
        let product = *limb * multiplier + carry;
        *limb = product % 1_000_000_000;
        carry = product / 1_000_000_000;
    }
    if carry != 0 {
        value.push(carry);
    }
}

fn from_digits(digits: &[u8], radix: u8) -> Vec<u32> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_add(&mut value, radix as u32, digit as u32);
    }
    value
}

fn divide(value: &mut Vec<u32>, divisor: u32) -> u32 {
    let mut remainder = 0u64;
    for limb in value.iter_mut().rev() {
        let current = remainder * 1_000_000_000 + *limb as u64;
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    remainder as u32
}

fn format_value(mut value: Vec<u32>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if value.iter().all(|&limb| limb == 0) {
        digits.push(0u8);
    } else {
        while value.iter().any(|&limb| limb != 0) {
            digits.push(divide(&mut value, radix as u32) as u8);
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let mut padded = vec![0; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::new();
    for (index, digit) in digits.iter().copied().enumerate() {
        if style.group != 0
            && index != 0
            && (digits.len() - index) % style.group == 0
        {
            output.push('_');
        }
        let byte = if digit < 10 {
            b'0' + digit
        } else if style.upper {
            b'A' + digit - 10
        } else {
            b'a' + digit - 10
        };
        output.push(byte as char);
    }
    output
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse(input, from, 0)?;
    Ok(format_value(from_digits(&digits, from), to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse(left, base, 0)?;
    let right_digits = parse(right, base, 1)?;
    let mut sum = from_digits(&left_digits, base);
    let right_value = from_digits(&right_digits, base);

    if sum.len() < right_value.len() {
        sum.resize(right_value.len(), 0);
    }
    let mut carry = 0u64;
    for index in 0..sum.len() {
        let rhs = right_value.get(index).copied().unwrap_or(0) as u64;
        let total = sum[index] as u64 + rhs + carry;
        sum[index] = (total % 1_000_000_000) as u32;
        carry = total / 1_000_000_000;
    }
    if carry != 0 {
        sum.push(carry as u32);
    }

    Ok(format_value(sum, base, style))
}
