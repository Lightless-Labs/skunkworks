#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let (payload, offset) = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        (&bytes[hash + 1..], hash + 1)
    } else {
        (bytes, 0)
    };

    if payload.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut digits = Vec::new();

    // Scan all non-separator bytes before checking separator placement.
    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: offset + i,
                });
            }
        }
    }

    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = i > 0 && payload[i - 1] != b'_';
            let has_digit_after = i + 1 < payload.len() && payload[i + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: offset + i,
                });
            }
        }
    }

    if digits.is_empty() {
        return Err(BaseError::Separator {
            operand,
            index: offset,
        });
    }
    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: offset,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in value.iter_mut() {
        let n = (*limb as u16) * (radix as u16) + carry;
        *limb = (n & 0xff) as u8;
        carry = n >> 8;
    }
    while carry != 0 {
        value.push((carry & 0xff) as u8);
        carry >>= 8;
    }
}

fn add_values(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for i in 0..left.len().max(right.len()) {
        let a = left.get(i).copied().unwrap_or(0) as u16;
        let b = right.get(i).copied().unwrap_or(0) as u16;
        let n = a + b + carry;
        result.push((n & 0xff) as u8);
        carry = n >> 8;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn divide_value(value: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for byte in value.iter_mut().rev() {
        let n = (remainder << 8) | (*byte as u16);
        *byte = (n / divisor as u16) as u8;
        remainder = n % divisor as u16;
    }
    while value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn format_value(mut value: Vec<u8>, radix: u8, style: BaseStyle) -> String {
    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };

    let mut digits = Vec::new();
    if value.is_empty() {
        digits.push(b'0');
    } else {
        while !value.is_empty() {
            digits.push(alphabet[divide_value(&mut value, radix) as usize]);
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let mut padded = vec![b'0'; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    if style.group == 0 {
        return String::from_utf8(digits).expect("ASCII output");
    }

    let extra = (digits.len() - 1) / style.group;
    let mut output = Vec::with_capacity(digits.len() + extra);
    for (i, byte) in digits.iter().copied().enumerate() {
        if i > 0 && (digits.len() - i) % style.group == 0 {
            output.push(b'_');
        }
        output.push(byte);
    }
    String::from_utf8(output).expect("ASCII output")
}

fn parse_to_value(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = Vec::new();
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    value
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    let value = parse_to_value(&digits, from);
    Ok(format_value(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = parse_to_value(&left_digits, base);
    let right_value = parse_to_value(&right_digits, base);
    Ok(format_value(add_values(&left_value, &right_value), base, style))
}
