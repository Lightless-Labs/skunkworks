#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize },
    InvalidPadding { offset: usize },
    InvalidLength { offset: usize },
    TrailingData { offset: usize },
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StreamError {
    Finished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_triple(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

fn encode_tail(bytes: &[u8]) -> String {
    let mut out = String::new();
    match bytes.len() {
        0 => {}
        1 => {
            let a = bytes[0];
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            out.push_str("==");
        }
        2 => {
            let a = bytes[0];
            let b = bytes[1];
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            out.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            out.push('=');
        }
        _ => unreachable!(),
    }
    out
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        let encoded = encode_triple(chunk[0], chunk[1], chunk[2]);
        out.push_str(std::str::from_utf8(&encoded).expect("base64 alphabet is ASCII"));
    }
    out.push_str(&encode_tail(chunks.remainder()));
    out
}

fn sextet(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn is_ignored_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\n' | 0x0b | 0x0c | b'\r')
}

#[derive(Clone, Copy)]
enum QuartetByte {
    Value(u8),
    Padding,
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut out = String::with_capacity((self.tail_len + input.len()) / 3 * 4);
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                let encoded = encode_triple(self.tail[0], self.tail[1], input[index - 1]);
                out.push_str(std::str::from_utf8(&encoded).expect("base64 alphabet is ASCII"));
                self.tail_len = 0;
            } else {
                return Ok(out);
            }
        }

        while index + 3 <= input.len() {
            let encoded = encode_triple(input[index], input[index + 1], input[index + 2]);
            out.push_str(std::str::from_utf8(&encoded).expect("base64 alphabet is ASCII"));
            index += 3;
        }

        let remaining = &input[index..];
        self.tail_len = remaining.len();
        self.tail[..self.tail_len].copy_from_slice(remaining);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }
        self.finished = true;
        Ok(encode_tail(&self.tail[..self.tail_len]))
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    ignore_whitespace: bool,
    quartet: [QuartetByte; 4],
    quartet_len: usize,
    offset: usize,
    padded: bool,
    finished: bool,
    error: Option<DecodeError>,
}

impl Decoder {
    pub fn new(ignore_whitespace: bool) -> Self {
        Self {
            ignore_whitespace,
            quartet: [QuartetByte::Padding; 4],
            quartet_len: 0,
            offset: 0,
            padded: false,
            finished: false,
            error: None,
        }
    }

    fn push_inner(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        let mut out = Vec::new();

        for byte in input.bytes() {
            let current_offset = self.offset;
            self.offset += 1;

            if self.ignore_whitespace && is_ignored_whitespace(byte) {
                continue;
            }

            if self.padded {
                return Err(DecodeError::TrailingData {
                    offset: current_offset,
                });
            }

            let position = self.quartet_len;

            if byte == b'=' {
                if position <= 1 {
                    return Err(DecodeError::InvalidPadding {
                        offset: current_offset,
                    });
                }
                if position == 3 {
                    match self.quartet[2] {
                        QuartetByte::Padding => {
                            self.quartet[3] = QuartetByte::Padding;
                        }
                        QuartetByte::Value(_) => {
                            self.quartet[3] = QuartetByte::Padding;
                        }
                    }
                } else {
                    self.quartet[2] = QuartetByte::Padding;
                }
            } else if position == 3 && matches!(self.quartet[2], QuartetByte::Padding) {
                return Err(DecodeError::InvalidPadding {
                    offset: current_offset,
                });
            } else if let Some(value) = sextet(byte) {
                self.quartet[position] = QuartetByte::Value(value);
            } else {
                return Err(DecodeError::InvalidByte {
                    offset: current_offset,
                });
            }

            self.quartet_len += 1;

            if self.quartet_len == 4 {
                let a = match self.quartet[0] {
                    QuartetByte::Value(v) => v,
                    QuartetByte::Padding => {
                        return Err(DecodeError::InvalidPadding {
                            offset: current_offset,
                        })
                    }
                };
                let b = match self.quartet[1] {
                    QuartetByte::Value(v) => v,
                    QuartetByte::Padding => {
                        return Err(DecodeError::InvalidPadding {
                            offset: current_offset,
                        })
                    }
                };

                match (self.quartet[2], self.quartet[3]) {
                    (QuartetByte::Value(c), QuartetByte::Value(d)) => {
                        out.push((a << 2) | (b >> 4));
                        out.push((b << 4) | (c >> 2));
                        out.push((c << 6) | d);
                    }
                    (QuartetByte::Value(c), QuartetByte::Padding) => {
                        out.push((a << 2) | (b >> 4));
                        out.push((b << 4) | (c >> 2));
                        self.padded = true;
                    }
                    (QuartetByte::Padding, QuartetByte::Padding) => {
                        out.push((a << 2) | (b >> 4));
                        self.padded = true;
                    }
                    (QuartetByte::Padding, QuartetByte::Value(_)) => {
                        return Err(DecodeError::InvalidPadding {
                            offset: current_offset,
                        })
                    }
                }
                self.quartet_len = 0;
            }
        }

        Ok(out)
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        match self.push_inner(input) {
            Ok(output) => Ok(output),
            Err(error) => {
                self.error = Some(error);
                Err(error)
            }
        }
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }
        if self.quartet_len != 0 {
            let error = DecodeError::InvalidLength {
                offset: self.offset,
            };
            self.error = Some(error);
            return Err(error);
        }
        self.finished = true;
        Ok(Vec::new())
    }
}

pub fn decode(input: &str, ignore_whitespace: bool) -> Result<Vec<u8>, DecodeError> {
    let mut decoder = Decoder::new(ignore_whitespace);
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
