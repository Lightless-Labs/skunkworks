use std::collections::HashMap;

#[derive(Debug, PartialEq, Eq)]
pub enum Error {
    NoTransaction,
    UnknownSnapshot(u64),
    ClockWentBackwards { last: u64, now: u64 },
    DeadlineOverflow,
}

#[derive(Clone)]
struct Entry {
    value: String,
    deadline: Option<u64>,
}

impl Entry {
    fn is_live_at(&self, now: u64) -> bool {
        self.deadline.is_none_or(|deadline| deadline > now)
    }
}

pub struct Store {
    entries: HashMap<String, Entry>,
    scopes: Vec<HashMap<String, Entry>>,
    snapshots: HashMap<u64, HashMap<String, Entry>>,
    last_time: u64,
}

impl Store {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            scopes: Vec::new(),
            snapshots: HashMap::new(),
            last_time: 0,
        }
    }

    fn advance(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::ClockWentBackwards {
                last: self.last_time,
                now,
            });
        }

        self.last_time = now;
        self.entries
            .retain(|_, entry| entry.is_live_at(now));
        Ok(())
    }

    pub fn set(
        &mut self,
        key: &str,
        value: &str,
        ttl: Option<u64>,
        now: u64,
    ) -> Result<(), Error> {
        self.advance(now)?;

        let deadline = match ttl {
            Some(ttl) => Some(now.checked_add(ttl).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };

        if deadline.is_some_and(|deadline| deadline <= now) {
            self.entries.remove(key);
        } else {
            self.entries.insert(
                key.to_owned(),
                Entry {
                    value: value.to_owned(),
                    deadline,
                },
            );
        }

        Ok(())
    }

    pub fn get(&mut self, key: &str, now: u64) -> Result<Option<String>, Error> {
        self.advance(now)?;
        Ok(self.entries.get(key).map(|entry| entry.value.clone()))
    }

    pub fn delete(&mut self, key: &str, now: u64) -> Result<bool, Error> {
        self.advance(now)?;
        Ok(self.entries.remove(key).is_some())
    }

    pub fn count(&mut self, value: &str, now: u64) -> Result<usize, Error> {
        self.advance(now)?;
        Ok(self
            .entries
            .values()
            .filter(|entry| entry.value == value)
            .count())
    }

    pub fn begin(&mut self, now: u64) -> Result<(), Error> {
        self.advance(now)?;
        self.scopes.push(self.entries.clone());
        Ok(())
    }

    pub fn commit(&mut self, now: u64) -> Result<(), Error> {
        self.advance(now)?;
        if self.scopes.is_empty() {
            return Err(Error::NoTransaction);
        }
        self.scopes.clear();
        Ok(())
    }

    pub fn rollback(&mut self, now: u64) -> Result<(), Error> {
        self.advance(now)?;
        let Some(saved) = self.scopes.pop() else {
            return Err(Error::NoTransaction);
        };
        self.entries = saved
            .into_iter()
            .filter(|(_, entry)| entry.is_live_at(now))
            .collect();
        Ok(())
    }

    pub fn snapshot(&mut self, id: u64, now: u64) -> Result<(), Error> {
        self.advance(now)?;
        self.snapshots.insert(id, self.entries.clone());
        Ok(())
    }

    pub fn restore(&mut self, id: u64, now: u64) -> Result<(), Error> {
        self.advance(now)?;
        let Some(saved) = self.snapshots.get(&id) else {
            return Err(Error::UnknownSnapshot(id));
        };
        self.entries = saved
            .iter()
            .filter(|(_, entry)| entry.is_live_at(now))
            .map(|(key, entry)| (key.clone(), entry.clone()))
            .collect();
        Ok(())
    }
}
