```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum BucketError {
    ZeroInterval,
    TimeWentBackwards,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct TokenBucket {
    capacity: u64,
    refill: u64,
    interval_ms: u64,
    tokens: u64,
    remainder: u64,
    last_ms: u64,
}

impl TokenBucket {
    pub fn new(
        capacity: u64,
        refill: u64,
        interval_ms: u64,
        start_ms: u64,
    ) -> Result<Self, BucketError> {
        if interval_ms == 0 {
            return Err(BucketError::ZeroInterval);
        }

        Ok(Self {
            capacity,
            refill,
            interval_ms,
            tokens: capacity,
            remainder: 0,
            last_ms: start_ms,
        })
    }

    pub fn available(&mut self, now: u64) -> Result<u64, BucketError> {
        self.advance(now)?;
        Ok(self.tokens)
    }

    pub fn try_acquire(&mut self, n: u64, now: u64) -> Result<bool, BucketError> {
        self.advance(now)?;
        if n <= self.tokens {
            self.tokens -= n;
            Ok(true)
        } else {
            Ok(false)
        }
    }

    fn advance(&mut self, now: u64) -> Result<(), BucketError> {
        if now < self.last_ms {
            return Err(BucketError::TimeWentBackwards);
        }

        let elapsed = now - self.last_ms;

        // Compute elapsed * refill + remainder exactly as a 128-bit integer.
        let numerator = (elapsed as u128) * (self.refill as u128) + (self.remainder as u128);
        let interval = self.interval_ms as u128;
        let whole = numerator / interval;
        let remainder = (numerator % interval) as u64;

        let total = (self.tokens as u128) + whole;
        if total >= self.capacity as u128 {
            self.tokens = self.capacity;
            self.remainder = 0;
        } else {
            self.tokens = total as u64;
            self.remainder = remainder;
        }

        self.last_ms = now;
        Ok(())
    }
}
```
