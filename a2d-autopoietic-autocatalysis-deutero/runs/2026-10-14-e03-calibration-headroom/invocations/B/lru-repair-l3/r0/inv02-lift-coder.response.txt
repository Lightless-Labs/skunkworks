```rust
use std::collections::VecDeque;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum PutError {
    TooHeavy { weight: usize, capacity: usize },
}

struct Entry {
    key: i32,
    value: i32,
    weight: usize,
}

pub struct LruCache {
    capacity: usize,
    total_weight: usize,
    entries: VecDeque<Entry>,
}

impl LruCache {
    pub fn new(capacity: usize) -> Self {
        Self {
            capacity,
            total_weight: 0,
            entries: VecDeque::new(),
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn total_weight(&self) -> usize {
        self.total_weight
    }

    pub fn keys_lru(&self) -> Vec<i32> {
        self.entries.iter().map(|entry| entry.key).collect()
    }

    pub fn get(&mut self, key: i32) -> Option<i32> {
        let pos = self.entries.iter().position(|entry| entry.key == key)?;
        let entry = self.entries.remove(pos)?;
        let value = entry.value;
        self.entries.push_back(entry);
        Some(value)
    }

    pub fn peek(&mut self, key: i32) -> Option<i32> {
        self.entries
            .iter()
            .find(|entry| entry.key == key)
            .map(|entry| entry.value)
    }

    pub fn remove(&mut self, key: i32) -> Option<i32> {
        let pos = self.entries.iter().position(|entry| entry.key == key)?;
        let entry = self.entries.remove(pos)?;
        self.total_weight -= entry.weight;
        Some(entry.value)
    }

    pub fn put(
        &mut self,
        key: i32,
        value: i32,
        weight: usize,
    ) -> Result<Option<i32>, PutError> {
        if weight > self.capacity {
            return Err(PutError::TooHeavy {
                weight,
                capacity: self.capacity,
            });
        }

        let old = self.remove(key);

        while self.total_weight > self.capacity - weight {
            if let Some(victim) = self.entries.pop_front() {
                self.total_weight -= victim.weight;
            } else {
                break;
            }
        }

        self.entries.push_back(Entry { key, value, weight });
        self.total_weight += weight;
        Ok(old)
    }
}
```
