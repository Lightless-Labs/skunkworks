#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte,
    InvalidPadding,
    InvalidLength,
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StreamError {
    Finished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_triple(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

fn encode_tail(input: &[u8], output: &mut String) {
    match input.len() {
        0 => {}
        1 => {
            output.push(ALPHABET[(input[0] >> 2) as usize] as char);
            output.push(ALPHABET[((input[0] & 0x03) << 4) as usize] as char);
            output.push_str("==");
        }
        2 => {
            output.push(ALPHABET[(input[0] >> 2) as usize] as char);
            output.push(ALPHABET[(((input[0] & 0x03) << 4) | (input[1] >> 4)) as usize] as char);
            output.push(ALPHABET[((input[1] & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        _ => unreachable!(),
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        let encoded = encode_triple(chunk[0], chunk[1], chunk[2]);
        output.push_str(std::str::from_utf8(&encoded).expect("base64 is ASCII"));
    }
    encode_tail(chunks.remainder(), &mut output);
    output
}

fn decode_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Symbol {
    Value(u8),
    Padding,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum QuartetState {
    Open,
    NeedSecondPadding,
    Padded,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct DecoderCore {
    quartet: [u8; 4],
    len: usize,
    state: QuartetState,
    offset: usize,
}

impl DecoderCore {
    fn new() -> Self {
        Self {
            quartet: [0; 4],
            len: 0,
            state: QuartetState::Open,
            offset: 0,
        }
    }

    fn consume(&mut self, byte: u8) -> Result<Option<[u8; 3]>, DecodeError> {
        self.offset += 1;

        if self.state == QuartetState::Padded {
            return Err(DecodeError::InvalidPadding);
        }

        if self.state == QuartetState::NeedSecondPadding {
            if byte != b'=' {
                return Err(DecodeError::InvalidPadding);
            }
            self.quartet[3] = 0;
            self.len = 0;
            self.state = QuartetState::Padded;
            return Ok(Some([
                (self.quartet[0] << 2) | (self.quartet[1] >> 4),
                0,
                0,
            ]));
        }

        let position = self.len;
        if byte == b'=' {
            match position {
                0 | 1 => return Err(DecodeError::InvalidPadding),
                2 => {
                    self.state = QuartetState::NeedSecondPadding;
                    self.quartet[2] = 0;
                    self.len = 3;
                    return Ok(None);
                }
                3 => {
                    self.quartet[3] = 0;
                    let result = [
                        (self.quartet[0] << 2) | (self.quartet[1] >> 4),
                        (self.quartet[1] << 4) | (self.quartet[2] >> 2),
                        0,
                    ];
                    self.len = 0;
                    self.state = QuartetState::Padded;
                    return Ok(Some(result));
                }
                _ => unreachable!(),
            }
        }

        let value = decode_value(byte).ok_or(DecodeError::InvalidByte)?;
        self.quartet[position] = value;
        self.len += 1;

        if self.len == 4 {
            let result = [
                (self.quartet[0] << 2) | (self.quartet[1] >> 4),
                (self.quartet[1] << 4) | (self.quartet[2] >> 2),
                (self.quartet[2] << 6) | self.quartet[3],
            ];
            self.len = 0;
            Ok(Some(result))
        } else {
            Ok(None)
        }
    }

    fn finish(&self) -> Result<(), DecodeError> {
        match self.state {
            QuartetState::NeedSecondPadding => Err(DecodeError::InvalidLength),
            QuartetState::Padded => Ok(()),
            QuartetState::Open if self.len == 0 => Ok(()),
            QuartetState::Open => Err(DecodeError::InvalidLength),
        }
    }
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let mut core = DecoderCore::new();
    let mut output = Vec::with_capacity(input.len() / 4 * 3);

    for &byte in input.as_bytes() {
        if let Some(decoded) = core.consume(byte)? {
            match core.state {
                QuartetState::Padded => {
                    let count = if input.as_bytes().iter().rev().find(|&&b| b != b'=').is_some() {
                        if core.quartet[2] == 0 { 1 } else { 2 }
                    } else {
                        3
                    };
                    output.extend_from_slice(&decoded[..count]);
                }
                _ => output.extend_from_slice(&decoded),
            }
        }
    }

    core.finish()?;
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut output = String::with_capacity((self.tail_len + input.len()) / 3 * 4);
        let mut index = 0;

        if self.tail_len != 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;

                if self.tail_len == 3 {
                    let encoded = encode_triple(self.tail[0], self.tail[1], input[index - 1]);
                    output.push_str(std::str::from_utf8(&encoded).expect("base64 is ASCII"));
                    self.tail_len = 0;
                }
            }
        }

        while index + 3 <= input.len() {
            let encoded = encode_triple(input[index], input[index + 1], input[index + 2]);
            output.push_str(std::str::from_utf8(&encoded).expect("base64 is ASCII"));
            index += 3;
        }

        let remaining = &input[index..];
        self.tail_len = remaining.len();
        self.tail[..self.tail_len].copy_from_slice(remaining);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }
        self.finished = true;
        let mut output = String::new();
        encode_tail(&self.tail[..self.tail_len], &mut output);
        self.tail_len = 0;
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Decoder {
    core: DecoderCore,
    error: Option<DecodeError>,
    finished: bool,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            core: DecoderCore::new(),
            error: None,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        let mut output = Vec::with_capacity(input.len() / 4 * 3);
        for &byte in input.as_bytes() {
            match self.core.consume(byte) {
                Ok(Some(decoded)) => match self.core.state {
                    QuartetState::Padded => {
                        let count = if self.core.quartet[2] == 0 { 1 } else { 2 };
                        output.extend_from_slice(&decoded[..count]);
                    }
                    _ => output.extend_from_slice(&decoded),
                },
                Ok(None) => {}
                Err(error) => {
                    self.error = Some(error);
                    return Err(error);
                }
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        match self.core.finish() {
            Ok(()) => {
                self.finished = true;
                Ok(Vec::new())
            }
            Err(error) => {
                self.error = Some(error);
                Err(error)
            }
        }
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
