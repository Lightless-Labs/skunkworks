#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte,
    InvalidPadding,
    InvalidLength,
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StreamError {
    Finished,
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

pub struct Decoder {
    quartet: [u8; 4],
    quartet_len: usize,
    padded: bool,
    finished: bool,
    error: Option<DecodeError>,
    offset: usize,
}

fn alphabet_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_triple(a: u8, b: u8, c: u8, output: &mut String) {
    const ALPHABET: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

fn decode_quartet(quartet: &[u8; 4]) -> Vec<u8> {
    let a = alphabet_value(quartet[0]).unwrap();
    let b = alphabet_value(quartet[1]).unwrap();
    let mut output = vec![(a << 2) | (b >> 4)];

    if quartet[2] != b'=' {
        let c = alphabet_value(quartet[2]).unwrap();
        output.push((b << 4) | (c >> 2));

        if quartet[3] != b'=' {
            let d = alphabet_value(quartet[3]).unwrap();
            output.push((c << 6) | d);
        }
    }

    output
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::new();
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        encode_triple(chunk[0], chunk[1], chunk[2], &mut output);
    }

    match chunks.remainder() {
        [] => {}
        [a] => {
            const ALPHABET: &[u8; 64] =
                b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push_str("==");
        }
        [a, b] => {
            const ALPHABET: &[u8; 64] =
                b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        _ => unreachable!(),
    }

    output
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                encode_triple(self.tail[0], self.tail[1], input[index - 1], &mut output);
                self.tail_len = 0;
            }
        }

        while index + 3 <= input.len() {
            encode_triple(
                input[index],
                input[index + 1],
                input[index + 2],
                &mut output,
            );
            index += 3;
        }

        let remainder = &input[index..];
        self.tail_len = remainder.len();
        for (slot, byte) in self.tail.iter_mut().zip(remainder.iter().copied()) {
            *slot = byte;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            0 => {}
            1 => {
                const ALPHABET: &[u8; 64] =
                    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                const ALPHABET: &[u8; 64] =
                    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => unreachable!(),
        }
        self.tail_len = 0;
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            quartet: [0; 4],
            quartet_len: 0,
            padded: false,
            finished: false,
            error: None,
            offset: 0,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        let mut output = Vec::new();

        for byte in input.bytes() {
            let position = self.quartet_len;

            let failure = if self.padded {
                Some(DecodeError::InvalidPadding)
            } else if position == 3
                && self.quartet[2] == b'='
                && byte != b'='
            {
                Some(DecodeError::InvalidPadding)
            } else if byte == b'=' {
                if position < 2 {
                    Some(DecodeError::InvalidPadding)
                } else {
                    self.quartet[position] = byte;
                    self.quartet_len += 1;
                    if self.quartet_len == 4 {
                        output.extend(decode_quartet(&self.quartet));
                        self.quartet_len = 0;
                        self.padded = true;
                    }
                    None
                }
            } else if let Some(value) = alphabet_value(byte) {
                self.quartet[position] = byte;
                self.quartet_len += 1;
                if self.quartet_len == 4 {
                    output.extend(decode_quartet(&self.quartet));
                    self.quartet_len = 0;
                }
                let _ = value;
                None
            } else {
                Some(DecodeError::InvalidByte)
            };

            if let Some(error) = failure {
                self.error = Some(error);
                return Err(error);
            }

            self.offset += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }
        if self.quartet_len != 0 {
            self.error = Some(DecodeError::InvalidLength);
            return Err(DecodeError::InvalidLength);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
