#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte,
    InvalidPadding,
    InvalidLength,
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StreamError {
    Finished,
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

pub struct Decoder {
    quartet: [u8; 4],
    quartet_len: usize,
    padded: bool,
    finished: bool,
    error: Option<DecodeError>,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_triple(a: u8, b: u8, c: u8, out: &mut String) {
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    out.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        encode_triple(chunk[0], chunk[1], chunk[2], &mut out);
    }

    match chunks.remainder() {
        [] => {}
        [a] => {
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            out.push_str("==");
        }
        [a, b] => {
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            out.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            out.push('=');
        }
        _ => unreachable!(),
    }

    out
}

fn decode_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn decode_quartet(q: &[u8; 4]) -> Vec<u8> {
    let a = decode_value(q[0]).unwrap();
    let b = decode_value(q[1]).unwrap();
    let mut out = Vec::with_capacity(3);

    out.push((a << 2) | (b >> 4));

    match (q[2], q[3]) {
        (b'=', b'=') => {}
        (c, b'=') => {
            let c = decode_value(c).unwrap();
            out.push((b << 4) | (c >> 2));
        }
        (c, d) => {
            let c = decode_value(c).unwrap();
            let d = decode_value(d).unwrap();
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);
        }
    }

    out
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let mut decoder = Decoder::new();
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut out = String::new();
        let mut pos = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && pos < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[pos];
                }
                self.tail_len += 1;
                pos += 1;
            }

            if self.tail_len == 3 {
                encode_triple(self.tail[0], self.tail[1], input[pos - 1], &mut out);
                self.tail_len = 0;
            } else {
                return Ok(out);
            }
        }

        while pos + 3 <= input.len() {
            encode_triple(input[pos], input[pos + 1], input[pos + 2], &mut out);
            pos += 3;
        }

        let remainder = &input[pos..];
        self.tail_len = remainder.len();
        self.tail[..self.tail_len].copy_from_slice(remainder);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }
        self.finished = true;

        let mut out = String::new();
        match self.tail_len {
            0 => {}
            1 => {
                let a = self.tail[0];
                out.push(ALPHABET[(a >> 2) as usize] as char);
                out.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                out.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                out.push(ALPHABET[(a >> 2) as usize] as char);
                out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                out.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                out.push('=');
            }
            _ => unreachable!(),
        }
        self.tail_len = 0;
        Ok(out)
    }
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            quartet: [0; 4],
            quartet_len: 0,
            padded: false,
            finished: false,
            error: None,
        }
    }

    fn fail(&mut self, error: DecodeError) -> DecodeError {
        self.error = Some(error);
        error
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        let mut out = Vec::new();

        for &byte in input.as_bytes() {
            if self.padded {
                return Err(self.fail(DecodeError::InvalidPadding));
            }

            let pos = self.quartet_len;
            if byte == b'=' {
                if pos < 2 {
                    return Err(self.fail(DecodeError::InvalidPadding));
                }
                if pos == 2 {
                    self.quartet[pos] = byte;
                    self.quartet_len += 1;
                    continue;
                }
                if pos == 3 {
                    if self.quartet[2] != b'=' {
                        self.quartet[3] = byte;
                        out.extend(decode_quartet(&self.quartet));
                        self.quartet_len = 0;
                        self.padded = true;
                        continue;
                    }
                    self.quartet[3] = byte;
                    out.extend(decode_quartet(&self.quartet));
                    self.quartet_len = 0;
                    self.padded = true;
                    continue;
                }
            }

            if pos == 3 && self.quartet[2] == b'=' {
                return Err(self.fail(DecodeError::InvalidPadding));
            }

            if let Some(_) = decode_value(byte) {
                self.quartet[pos] = byte;
                self.quartet_len += 1;
                if self.quartet_len == 4 {
                    out.extend(decode_quartet(&self.quartet));
                    self.quartet_len = 0;
                }
            } else {
                return Err(self.fail(DecodeError::InvalidByte));
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }
        if self.quartet_len != 0 {
            return Err(self.fail(DecodeError::InvalidLength));
        }
        self.finished = true;
        Ok(Vec::new())
    }
}
