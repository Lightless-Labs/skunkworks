#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte,
    InvalidPadding,
    InvalidLength,
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StreamError {
    Finished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_triplet(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        output.extend(encode_triplet(chunk[0], chunk[1], chunk[2]).map(char::from));
    }

    match chunks.remainder() {
        [] => {}
        [a] => {
            output.push(char::from(ALPHABET[(a >> 2) as usize]));
            output.push(char::from(ALPHABET[((a & 0x03) << 4) as usize]));
            output.push_str("==");
        }
        [a, b] => {
            output.push(char::from(ALPHABET[(a >> 2) as usize]));
            output.push(char::from(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize]));
            output.push(char::from(ALPHABET[((b & 0x0f) << 2) as usize]));
            output.push('=');
        }
        _ => unreachable!(),
    }

    output
}

fn sextet(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn decode_quartet(q: &[u8; 4]) -> Vec<u8> {
    let a = sextet(q[0]).unwrap();
    let b = sextet(q[1]).unwrap();
    let mut output = vec![(a << 2) | (b >> 4)];

    if q[2] != b'=' {
        let c = sextet(q[2]).unwrap();
        output.push((b << 4) | (c >> 2));

        if q[3] != b'=' {
            let d = sextet(q[3]).unwrap();
            output.push((c << 6) | d);
        }
    }

    output
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum PaddingState {
    None,
    NeedSecondEquals,
}

fn process_decoder_byte(
    quartet: &mut [u8; 4],
    len: &mut usize,
    padded: &mut bool,
    padding: &mut PaddingState,
    byte: u8,
    output: &mut Vec<u8>,
) -> Result<(), DecodeError> {
    if *padded {
        return Err(DecodeError::InvalidPadding);
    }

    if *padding == PaddingState::NeedSecondEquals {
        if byte != b'=' {
            return Err(DecodeError::InvalidPadding);
        }
        quartet[3] = byte;
        *len = 4;
        *padding = PaddingState::None;
        *padded = true;
        output.extend(decode_quartet(quartet));
        *len = 0;
        return Ok(());
    }

    let position = *len;
    if byte == b'=' {
        match position {
            0 | 1 => return Err(DecodeError::InvalidPadding),
            2 => {
                quartet[2] = byte;
                *len = 3;
                *padding = PaddingState::NeedSecondEquals;
                return Ok(());
            }
            3 => {
                if sextet(quartet[2]).is_none() {
                    return Err(DecodeError::InvalidByte);
                }
                quartet[3] = byte;
                *len = 4;
                *padded = true;
                output.extend(decode_quartet(quartet));
                *len = 0;
                return Ok(());
            }
            _ => return Err(DecodeError::InvalidPadding),
        }
    }

    let value = sextet(byte).ok_or(DecodeError::InvalidByte)?;
    quartet[position] = byte;
    *len += 1;

    if *len == 4 {
        let _ = value;
        output.extend(decode_quartet(quartet));
        *len = 0;
    }

    Ok(())
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let mut quartet = [0; 4];
    let mut len = 0;
    let mut padded = false;
    let mut padding = PaddingState::None;
    let mut output = Vec::new();

    for &byte in input.as_bytes() {
        process_decoder_byte(
            &mut quartet,
            &mut len,
            &mut padded,
            &mut padding,
            byte,
            &mut output,
        )?;
    }

    if len != 0 {
        return Err(DecodeError::InvalidLength);
    }

    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                output.extend(
                    encode_triplet(self.tail[0], self.tail[1], input[index - 1])
                        .map(char::from),
                );
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            output.extend(
                encode_triplet(input[index], input[index + 1], input[index + 2])
                    .map(char::from),
            );
            index += 3;
        }

        let remainder = &input[index..];
        self.tail_len = remainder.len();
        self.tail[..self.tail_len].copy_from_slice(remainder);

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        self.finished = true;
        let output = match self.tail_len {
            0 => String::new(),
            1 => encode(&self.tail[..1]),
            2 => encode(&self.tail[..2]),
            _ => unreachable!(),
        };
        self.tail_len = 0;
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    quartet: [u8; 4],
    len: usize,
    padded: bool,
    padding: PaddingState,
    finished: bool,
    error: Option<DecodeError>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            quartet: [0; 4],
            len: 0,
            padded: false,
            padding: PaddingState::None,
            finished: false,
            error: None,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        let mut output = Vec::new();
        for &byte in input.as_bytes() {
            match process_decoder_byte(
                &mut self.quartet,
                &mut self.len,
                &mut self.padded,
                &mut self.padding,
                byte,
                &mut output,
            ) {
                Ok(()) => {}
                Err(error) => {
                    self.error = Some(error);
                    return Err(error);
                }
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        if self.len != 0 {
            self.error = Some(DecodeError::InvalidLength);
            return Err(DecodeError::InvalidLength);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
