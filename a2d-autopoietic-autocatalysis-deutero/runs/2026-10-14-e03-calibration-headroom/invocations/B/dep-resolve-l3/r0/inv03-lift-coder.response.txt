```rust
use std::cmp::Reverse;
use std::collections::{BTreeMap, BTreeSet, BinaryHeap};

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Feature {
    pub name: String,
    pub dependencies: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub dependencies: Vec<String>,
    pub optional_dependencies: Vec<String>,
    pub features: Vec<Feature>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Request {
    pub package: String,
    pub features: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    MissingPackage(String),
    UnknownFeature { package: String, feature: String },
    Cycle(Vec<String>),
}

pub fn resolve(
    packages: &[Package],
    requests: &[Request],
) -> Result<Vec<String>, ResolveError> {
    let catalog: BTreeMap<&str, &Package> = packages
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut requested_features: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    for request in requests {
        requested_features
            .entry(request.package.clone())
            .or_default()
            .extend(request.features.iter().cloned());
    }

    if requested_features.is_empty() {
        return Ok(Vec::new());
    }

    let mut relevant = BTreeSet::new();
    let mut missing = BTreeSet::new();
    let mut unknown_features = BTreeSet::new();
    let mut pending: BTreeSet<String> = requested_features.keys().cloned().collect();

    while let Some(name) = pending.pop_first() {
        if !relevant.insert(name.clone()) {
            continue;
        }

        let Some(package) = catalog.get(name.as_str()).copied() else {
            missing.insert(name);
            continue;
        };

        let mut enabled_optional = BTreeSet::new();
        if let Some(features) = requested_features.get(&name) {
            let feature_map: BTreeMap<&str, &Feature> = package
                .features
                .iter()
                .map(|feature| (feature.name.as_str(), feature))
                .collect();

            for feature_name in features {
                if let Some(feature) = feature_map.get(feature_name.as_str()) {
                    enabled_optional.extend(feature.dependencies.iter().cloned());
                } else {
                    unknown_features.insert((name.clone(), feature_name.clone()));
                }
            }
        }

        for dependency in &package.dependencies {
            if !relevant.contains(dependency) {
                pending.insert(dependency.clone());
            }
        }
        for dependency in &package.optional_dependencies {
            if enabled_optional.contains(dependency) && !relevant.contains(dependency) {
                pending.insert(dependency.clone());
            }
        }
    }

    if let Some(name) = missing.into_iter().next() {
        return Err(ResolveError::MissingPackage(name));
    }
    if let Some((package, feature)) = unknown_features.into_iter().next() {
        return Err(ResolveError::UnknownFeature { package, feature });
    }

    let mut graph: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    for name in &relevant {
        let package = catalog[name.as_str()];
        let enabled_optional: BTreeSet<String> = requested_features
            .get(name)
            .into_iter()
            .flatten()
            .filter_map(|feature_name| {
                package
                    .features
                    .iter()
                    .find(|feature| feature.name == *feature_name)
            })
            .flat_map(|feature| feature.dependencies.iter().cloned())
            .collect();

        let mut targets: BTreeSet<String> = package.dependencies.iter().cloned().collect();
        targets.extend(
            package
                .optional_dependencies
                .iter()
                .filter(|dependency| enabled_optional.contains(*dependency))
                .cloned(),
        );
        graph.insert(name.clone(), targets);
    }

    fn find_cycle(
        node: &str,
        graph: &BTreeMap<String, BTreeSet<String>>,
        visited: &mut BTreeSet<String>,
        stack: &mut Vec<String>,
        on_stack: &mut BTreeMap<String, usize>,
    ) -> Option<Vec<String>> {
        visited.insert(node.to_string());
        on_stack.insert(node.to_string(), stack.len());
        stack.push(node.to_string());

        if let Some(targets) = graph.get(node) {
            for target in targets {
                if let Some(&start) = on_stack.get(target) {
                    let mut cycle = stack[start..].to_vec();
                    let smallest = cycle
                        .iter()
                        .enumerate()
                        .min_by(|(_, left), (_, right)| left.cmp(right))
                        .map(|(index, _)| index)
                        .unwrap_or(0);
                    cycle.rotate_left(smallest);
                    cycle.push(cycle[0].clone());
                    return Some(cycle);
                }
                if !visited.contains(target) {
                    if let Some(cycle) =
                        find_cycle(target, graph, visited, stack, on_stack)
                    {
                        return Some(cycle);
                    }
                }
            }
        }

        stack.pop();
        on_stack.remove(node);
        None
    }

    let mut visited = BTreeSet::new();
    let mut stack = Vec::new();
    let mut on_stack = BTreeMap::new();
    for name in graph.keys() {
        if !visited.contains(name) {
            if let Some(cycle) =
                find_cycle(name, &graph, &mut visited, &mut stack, &mut on_stack)
            {
                return Err(ResolveError::Cycle(cycle));
            }
        }
    }

    let mut remaining_dependencies: BTreeMap<String, usize> =
        graph.keys().map(|name| (name.clone(), 0)).collect();
    let mut dependents: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();

    for (dependent, dependencies) in &graph {
        remaining_dependencies.insert(dependent.clone(), dependencies.len());
        for dependency in dependencies {
            dependents
                .entry(dependency.clone())
                .or_default()
                .insert(dependent.clone());
        }
    }

    let mut ready: BinaryHeap<Reverse<String>> = remaining_dependencies
        .iter()
        .filter(|(_, count)| **count == 0)
        .map(|(name, _)| Reverse(name.clone()))
        .collect();

    let mut order = Vec::with_capacity(relevant.len());
    while let Some(Reverse(name)) = ready.pop() {
        order.push(name.clone());
        if let Some(next_dependents) = dependents.get(&name) {
            for dependent in next_dependents {
                let count = remaining_dependencies
                    .get_mut(dependent)
                    .expect("all graph nodes have a dependency count");
                *count -= 1;
                if *count == 0 {
                    ready.push(Reverse(dependent.clone()));
                }
            }
        }
    }

    Ok(order)
}
```
