pub enum PutError {
    TooHeavy { weight: usize, capacity: usize },
}

    key: i32,
    value: i32,
    weight: usize,
    deadline: Option<u64>,

        let total = &mut self.total_weight;

            if reached_live { return true; }
            let expired = e.deadline.is_some_and(|d| d < now);
            if !expired { reached_live = true; }
            if expired { *total -= e.weight; }
            !expired
        });

        self.purge(now);
        self.entries.len()
