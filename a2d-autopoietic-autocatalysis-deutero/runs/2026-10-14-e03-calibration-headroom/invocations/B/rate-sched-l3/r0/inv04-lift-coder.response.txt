```rust
use std::collections::{BTreeMap, VecDeque};

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum WindowError {
    ZeroWindow,
    ZeroIdleTtl,
    TimeWentBackwards,
}

struct KeyState {
    requests: VecDeque<(u64, u64)>,
    last_activity: u64,
}

pub struct SlidingWindow {
    limit: u64,
    window_ms: u64,
    idle_ttl_ms: u64,
    clock: u64,
    keys: BTreeMap<String, KeyState>,
}

impl SlidingWindow {
    pub fn new(
        limit: u64,
        window_ms: u64,
        idle_ttl_ms: u64,
        start_ms: u64,
    ) -> Result<Self, WindowError> {
        if window_ms == 0 {
            return Err(WindowError::ZeroWindow);
        }
        if idle_ttl_ms == 0 {
            return Err(WindowError::ZeroIdleTtl);
        }

        Ok(Self {
            limit,
            window_ms,
            idle_ttl_ms,
            clock: start_ms,
            keys: BTreeMap::new(),
        })
    }

    fn advance_clock(&mut self, now: u64) -> Result<(), WindowError> {
        if now < self.clock {
            return Err(WindowError::TimeWentBackwards);
        }
        self.clock = now;
        Ok(())
    }

    fn purge_expired(state: &mut KeyState, now: u64, window_ms: u64) {
        while let Some(&(timestamp, _)) = state.requests.front() {
            if now - timestamp >= window_ms {
                state.requests.pop_front();
            } else {
                break;
            }
        }
    }

    fn live_weight(state: &KeyState) -> u64 {
        state
            .requests
            .iter()
            .fold(0u64, |sum, &(_, weight)| sum.saturating_add(weight))
    }

    pub fn try_acquire(
        &mut self,
        key: &str,
        n: u64,
        now: u64,
    ) -> Result<bool, WindowError> {
        self.advance_clock(now)?;

        if let Some(state) = self.keys.get_mut(key) {
            Self::purge_expired(state, now, self.window_ms);
        }

        if n == 0 {
            return Ok(true);
        }

        let state = self.keys.entry(key.to_owned()).or_insert_with(|| KeyState {
            requests: VecDeque::new(),
            last_activity: now,
        });
        state.last_activity = now;

        let used = Self::live_weight(state);
        let remaining = self.limit.saturating_sub(used);
        if n > remaining {
            return Ok(false);
        }

        state.requests.push_back((now, n));
        Ok(true)
    }

    pub fn remaining(&mut self, key: &str, now: u64) -> Result<u64, WindowError> {
        self.advance_clock(now)?;

        let Some(state) = self.keys.get_mut(key) else {
            return Ok(self.limit);
        };

        Self::purge_expired(state, now, self.window_ms);
        let used = Self::live_weight(state);
        Ok(self.limit.saturating_sub(used))
    }

    pub fn evict_idle(&mut self, now: u64) -> Result<Vec<String>, WindowError> {
        self.advance_clock(now)?;

        let idle_ttl_ms = self.idle_ttl_ms;
        let evicted: Vec<String> = self
            .keys
            .iter()
            .filter_map(|(key, state)| {
                (now - state.last_activity >= idle_ttl_ms).then(|| key.clone())
            })
            .collect();

        for key in &evicted {
            self.keys.remove(key);
        }

        Ok(evicted)
    }

    pub fn tracked_keys(&self) -> usize {
        self.keys.len()
    }
}
```
