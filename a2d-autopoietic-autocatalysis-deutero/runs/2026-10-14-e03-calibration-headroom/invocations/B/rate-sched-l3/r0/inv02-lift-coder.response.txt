```rust
use std::collections::{BTreeMap, VecDeque};

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum WindowError {
    ZeroWindow,
    ZeroIdleTtl,
    TimeWentBackwards,
}

struct KeyState {
    requests: VecDeque<(u64, u64)>,
    last_activity: u64,
}

pub struct SlidingWindow {
    limit: u64,
    window_ms: u64,
    idle_ttl_ms: u64,
    clock: u64,
    keys: BTreeMap<String, KeyState>,
}

impl SlidingWindow {
    pub fn new(
        limit: u64,
        window_ms: u64,
        idle_ttl_ms: u64,
        start_ms: u64,
    ) -> Result<Self, WindowError> {
        if window_ms == 0 {
            return Err(WindowError::ZeroWindow);
        }
        if idle_ttl_ms == 0 {
            return Err(WindowError::ZeroIdleTtl);
        }

        Ok(Self {
            limit,
            window_ms,
            idle_ttl_ms,
            clock: start_ms,
            keys: BTreeMap::new(),
        })
    }

    fn advance_clock(&mut self, now: u64) -> Result<(), WindowError> {
        if now < self.clock {
            return Err(WindowError::TimeWentBackwards);
        }
        self.clock = now;
        Ok(())
    }

    fn purge_expired(&mut self, key: &str, now: u64) {
        if let Some(state) = self.keys.get_mut(key) {
            while let Some(&(timestamp, _)) = state.requests.front() {
                if now - timestamp >= self.window_ms {
                    state.requests.pop_front();
                } else {
                    break;
                }
            }
        }
    }

    fn live_weight(&self, key: &str) -> u64 {
        self.keys
            .get(key)
            .map(|state| {
                state
                    .requests
                    .iter()
                    .fold(0u64, |sum, &(_, weight)| sum.saturating_add(weight))
            })
            .unwrap_or(0)
    }

    pub fn try_acquire(
        &mut self,
        key: &str,
        n: u64,
        now: u64,
    ) -> Result<bool, WindowError> {
        self.advance_clock(now)?;
        self.purge_expired(key, now);

        if n == 0 {
            return Ok(true);
        }

        if !self.keys.contains_key(key) {
            self.keys.insert(
                key.to_owned(),
                KeyState {
                    requests: VecDeque::new(),
                    last_activity: now,
                },
            );
        }

        let used = self.live_weight(key);
        let available = self.limit.saturating_sub(used);
        let accepted = n <= available;

        if let Some(state) = self.keys.get_mut(key) {
            state.last_activity = now;
            if accepted {
                state.requests.push_back((now, n));
            }
        }

        Ok(accepted)
    }

    pub fn remaining(&mut self, key: &str, now: u64) -> Result<u64, WindowError> {
        self.advance_clock(now)?;
        self.purge_expired(key, now);

        Ok(self.limit.saturating_sub(self.live_weight(key)))
    }

    pub fn evict_idle(&mut self, now: u64) -> Result<Vec<String>, WindowError> {
        self.advance_clock(now)?;

        let idle_ttl_ms = self.idle_ttl_ms;
        let evicted: Vec<String> = self
            .keys
            .iter()
            .filter(|(_, state)| now - state.last_activity >= idle_ttl_ms)
            .map(|(key, _)| key.clone())
            .collect();

        for key in &evicted {
            self.keys.remove(key);
        }

        Ok(evicted)
    }

    pub fn tracked_keys(&self) -> usize {
        self.keys.len()
    }
}
```
