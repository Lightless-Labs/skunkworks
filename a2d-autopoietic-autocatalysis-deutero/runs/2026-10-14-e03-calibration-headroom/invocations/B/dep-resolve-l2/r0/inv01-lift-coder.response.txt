#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub dependencies: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    MissingDependency { package: String, dependency: String },
    Cycle(Vec<String>),
}

pub fn install_order(packages: &[Package]) -> Result<Vec<String>, ResolveError> {
    use std::collections::{BTreeMap, BTreeSet};

    let by_name: BTreeMap<&str, &Package> = packages
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut missing: Option<(&str, &str)> = None;
    for package in packages {
        for dependency in &package.dependencies {
            if !by_name.contains_key(dependency.as_str()) {
                let candidate = (package.name.as_str(), dependency.as_str());
                if missing.is_none_or(|current| candidate < current) {
                    missing = Some(candidate);
                }
            }
        }
    }

    if let Some((package, dependency)) = missing {
        return Err(ResolveError::MissingDependency {
            package: package.to_string(),
            dependency: dependency.to_string(),
        });
    }

    fn visit<'a>(
        name: &'a str,
        by_name: &BTreeMap<&'a str, &'a Package>,
        states: &mut BTreeMap<&'a str, u8>,
        stack: &mut Vec<&'a str>,
    ) -> Option<Vec<String>> {
        states.insert(name, 1);
        stack.push(name);

        let dependencies: BTreeSet<&str> = by_name[name]
            .dependencies
            .iter()
            .map(String::as_str)
            .collect();

        for dependency in dependencies {
            match states.get(dependency).copied().unwrap_or(0) {
                0 => {
                    if let Some(cycle) = visit(dependency, by_name, states, stack) {
                        return Some(cycle);
                    }
                }
                1 => {
                    let start = stack.iter().position(|&node| node == dependency).unwrap();
                    let mut path: Vec<&str> = stack[start..].to_vec();
                    let smallest = path
                        .iter()
                        .enumerate()
                        .min_by_key(|(_, node)| *node)
                        .map(|(index, _)| index)
                        .unwrap();
                    path.rotate_left(smallest);
                    let mut cycle: Vec<String> =
                        path.into_iter().map(str::to_string).collect();
                    cycle.push(cycle[0].clone());
                    return Some(cycle);
                }
                _ => {}
            }
        }

        stack.pop();
        states.insert(name, 2);
        None
    }

    let mut states: BTreeMap<&str, u8> = BTreeMap::new();
    let mut stack = Vec::new();

    for &name in by_name.keys() {
        if states.get(name).copied().unwrap_or(0) == 0 {
            if let Some(cycle) = visit(name, &by_name, &mut states, &mut stack) {
                return Err(ResolveError::Cycle(cycle));
            }
        }
    }

    let mut remaining_dependencies: BTreeMap<&str, BTreeSet<&str>> = BTreeMap::new();
    let mut dependents: BTreeMap<&str, Vec<&str>> = BTreeMap::new();

    for package in packages {
        let dependencies: BTreeSet<&str> = package
            .dependencies
            .iter()
            .map(String::as_str)
            .collect();
        for &dependency in &dependencies {
            dependents.entry(dependency).or_default().push(package.name.as_str());
        }
        remaining_dependencies.insert(package.name.as_str(), dependencies);
    }

    let mut ready: BTreeSet<&str> = remaining_dependencies
        .iter()
        .filter_map(|(&name, dependencies)| dependencies.is_empty().then_some(name))
        .collect();

    let mut order = Vec::with_capacity(packages.len());
    while let Some(&name) = ready.iter().next() {
        ready.remove(name);
        order.push(name.to_string());

        if let Some(children) = dependents.get(name) {
            for &dependent in children {
                let dependencies = remaining_dependencies.get_mut(dependent).unwrap();
                dependencies.remove(name);
                if dependencies.is_empty() {
                    ready.insert(dependent);
                }
            }
        }
    }

    Ok(order)
}
