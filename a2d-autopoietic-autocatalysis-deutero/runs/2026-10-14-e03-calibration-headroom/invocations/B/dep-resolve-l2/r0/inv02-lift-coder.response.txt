```rust
use std::collections::{BTreeSet, HashMap};

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub dependencies: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    MissingDependency { package: String, dependency: String },
    Cycle(Vec<String>),
}

pub fn install_order(packages: &[Package]) -> Result<Vec<String>, ResolveError> {
    let mut by_name: HashMap<&str, usize> = HashMap::new();
    for (index, package) in packages.iter().enumerate() {
        by_name.insert(&package.name, index);
    }

    let mut package_indices: Vec<usize> = (0..packages.len()).collect();
    package_indices.sort_by(|&a, &b| packages[a].name.cmp(&packages[b].name));

    let mut missing: Option<(String, String)> = None;
    for &index in &package_indices {
        let package = &packages[index];
        for dependency in &package.dependencies {
            if !by_name.contains_key(dependency.as_str()) {
                let candidate = (package.name.clone(), dependency.clone());
                if missing.as_ref().is_none_or(|current| candidate < *current) {
                    missing = Some(candidate);
                }
            }
        }
    }
    if let Some((package, dependency)) = missing {
        return Err(ResolveError::MissingDependency {
            package,
            dependency,
        });
    }

    let adjacency: Vec<Vec<usize>> = packages
        .iter()
        .map(|package| {
            let mut targets: Vec<usize> = package
                .dependencies
                .iter()
                .map(|dependency| by_name[dependency.as_str()])
                .collect();
            targets.sort_by(|&a, &b| packages[a].name.cmp(&packages[b].name));
            targets.dedup();
            targets
        })
        .collect();

    fn visit(
        node: usize,
        packages: &[Package],
        adjacency: &[Vec<usize>],
        states: &mut [u8],
        stack: &mut Vec<usize>,
    ) -> Option<Vec<String>> {
        states[node] = 1;
        stack.push(node);

        for &target in &adjacency[node] {
            if states[target] == 0 {
                if let Some(cycle) = visit(target, packages, adjacency, states, stack) {
                    return Some(cycle);
                }
            } else if states[target] == 1 {
                let start = stack.iter().position(|&item| item == target).unwrap();
                let mut cycle: Vec<String> = stack[start..]
                    .iter()
                    .map(|&item| packages[item].name.clone())
                    .collect();

                let smallest = (0..cycle.len())
                    .min_by(|&a, &b| cycle[a].cmp(&cycle[b]))
                    .unwrap();
                cycle.rotate_left(smallest);
                cycle.push(cycle[0].clone());
                return Some(cycle);
            }
        }

        stack.pop();
        states[node] = 2;
        None
    }

    let mut states = vec![0u8; packages.len()];
    let mut stack = Vec::new();
    for &index in &package_indices {
        if states[index] == 0 {
            if let Some(cycle) = visit(index, packages, &adjacency, &mut states, &mut stack) {
                return Err(ResolveError::Cycle(cycle));
            }
        }
    }

    let mut remaining_dependencies: Vec<usize> =
        adjacency.iter().map(Vec::len).collect();
    let mut dependents = vec![Vec::new(); packages.len()];
    for (dependent, targets) in adjacency.iter().enumerate() {
        for &dependency in targets {
            dependents[dependency].push(dependent);
        }
    }

    let mut ready = BTreeSet::new();
    for (index, &count) in remaining_dependencies.iter().enumerate() {
        if count == 0 {
            ready.insert((packages[index].name.clone(), index));
        }
    }

    let mut order = Vec::with_capacity(packages.len());
    while let Some((name, index)) = ready.pop_first() {
        order.push(name);
        for &dependent in &dependents[index] {
            remaining_dependencies[dependent] -= 1;
            if remaining_dependencies[dependent] == 0 {
                ready.insert((packages[dependent].name.clone(), dependent));
            }
        }
    }

    Ok(order)
}
```
