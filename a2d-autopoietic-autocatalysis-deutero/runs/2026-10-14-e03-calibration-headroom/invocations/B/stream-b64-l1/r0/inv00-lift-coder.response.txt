#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte,
    InvalidPadding,
    InvalidLength,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum EncodeState {
    Empty,
    One(u8),
    Two(u8, u8),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum DecodeState {
    Empty,
    One(u8),
    Two(u8, u8),
    Three(u8, u8, u8),
    NeedSecondPad(u8, u8),
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Encoder {
    state: EncodeState,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Decoder {
    state: DecodeState,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_sextet(value: u8) -> u8 {
    ALPHABET[value as usize]
}

fn decode_sextet(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            state: EncodeState::Empty,
        }
    }

    pub fn update(&mut self, input: &[u8]) -> String {
        let mut output = String::with_capacity((input.len() * 4 + 2) / 3);

        for &byte in input {
            match self.state {
                EncodeState::Empty => self.state = EncodeState::One(byte),
                EncodeState::One(first) => {
                    output.push(encode_sextet(first >> 2) as char);
                    output.push(encode_sextet(((first & 0x03) << 4) | (byte >> 4)) as char);
                    self.state = EncodeState::Two(first, byte);
                }
                EncodeState::Two(first, second) => {
                    output.push(encode_sextet(((second & 0x0f) << 2) | (byte >> 6)) as char);
                    output.push(encode_sextet(byte & 0x3f) as char);
                    self.state = EncodeState::Empty;
                    let _ = first;
                }
            }
        }

        output
    }

    pub fn finalize(&mut self) -> String {
        let mut output = String::new();
        match self.state {
            EncodeState::Empty => {}
            EncodeState::One(first) => {
                output.push(encode_sextet(first >> 2) as char);
                output.push(encode_sextet((first & 0x03) << 4) as char);
                output.push_str("==");
            }
            EncodeState::Two(first, second) => {
                output.push(encode_sextet(first >> 2) as char);
                output.push(encode_sextet(((first & 0x03) << 4) | (second >> 4)) as char);
                output.push(encode_sextet((second & 0x0f) << 2) as char);
                output.push('=');
            }
        }
        self.state = EncodeState::Empty;
        output
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            state: DecodeState::Empty,
        }
    }

    pub fn update(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        let mut output = Vec::new();

        for byte in input.bytes() {
            match self.state {
                DecodeState::Finished => return Err(DecodeError::InvalidPadding),
                DecodeState::NeedSecondPad(first, second) => {
                    if byte != b'=' {
                        return Err(DecodeError::InvalidPadding);
                    }
                    output.push((first << 2) | (second >> 4));
                    self.state = DecodeState::Finished;
                }
                DecodeState::Empty => {
                    if byte == b'=' {
                        return Err(DecodeError::InvalidPadding);
                    }
                    let value = decode_sextet(byte).ok_or(DecodeError::InvalidByte)?;
                    self.state = DecodeState::One(value);
                }
                DecodeState::One(first) => {
                    if byte == b'=' {
                        return Err(DecodeError::InvalidPadding);
                    }
                    let value = decode_sextet(byte).ok_or(DecodeError::InvalidByte)?;
                    self.state = DecodeState::Two(first, value);
                }
                DecodeState::Two(first, second) => {
                    if byte == b'=' {
                        self.state = DecodeState::NeedSecondPad(first, second);
                    } else {
                        let value = decode_sextet(byte).ok_or(DecodeError::InvalidByte)?;
                        output.push((first << 2) | (second >> 4));
                        self.state = DecodeState::Three(first, second, value);
                    }
                }
                DecodeState::Three(first, second, third) => {
                    if byte == b'=' {
                        output.push((first << 2) | (second >> 4));
                        output.push((second << 4) | (third >> 2));
                        self.state = DecodeState::Finished;
                    } else {
                        let fourth = decode_sextet(byte).ok_or(DecodeError::InvalidByte)?;
                        output.push((first << 2) | (second >> 4));
                        output.push((second << 4) | (third >> 2));
                        output.push((third << 6) | fourth);
                        self.state = DecodeState::Empty;
                    }
                }
            }
        }

        Ok(output)
    }

    pub fn finalize(&mut self) -> Result<Vec<u8>, DecodeError> {
        match self.state {
            DecodeState::Empty | DecodeState::Finished => {
                self.state = DecodeState::Empty;
                Ok(Vec::new())
            }
            DecodeState::One(_) | DecodeState::Two(_, _) | DecodeState::Three(_, _, _) |
            DecodeState::NeedSecondPad(_, _) => Err(DecodeError::InvalidLength),
        }
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.update(input);
    output.push_str(&encoder.finalize());
    output
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let mut decoder = Decoder::new();
    let mut output = decoder.update(input)?;
    output.extend(decoder.finalize()?);
    Ok(output)
}
