#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte,
    InvalidPadding,
    InvalidLength,
}

pub fn encode(input: &[u8]) -> String {
    const ALPHABET: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);

    for chunk in input.chunks(3) {
        let first = chunk[0];
        let second = chunk.get(1).copied().unwrap_or(0);
        let third = chunk.get(2).copied().unwrap_or(0);

        output.push(ALPHABET[(first >> 2) as usize] as char);
        output.push(ALPHABET[(((first & 0x03) << 4) | (second >> 4)) as usize] as char);

        if chunk.len() >= 2 {
            output.push(ALPHABET[(((second & 0x0f) << 2) | (third >> 6)) as usize] as char);
        } else {
            output.push('=');
        }

        if chunk.len() == 3 {
            output.push(ALPHABET[(third & 0x3f) as usize] as char);
        } else {
            output.push('=');
        }
    }

    output
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let mut output = Vec::with_capacity(input.len() / 4 * 3);
    let mut quartet = [0u8; 4];
    let mut position = 0usize;
    let mut padding_at_position_two = false;
    let mut finished = false;

    for byte in input.bytes() {
        if byte.is_ascii_whitespace() {
            continue;
        }

        if finished {
            return Err(DecodeError::InvalidPadding);
        }

        if padding_at_position_two && byte != b'=' {
            return Err(DecodeError::InvalidPadding);
        }

        if byte == b'=' {
            match position {
                0 | 1 => return Err(DecodeError::InvalidPadding),
                2 => {
                    quartet[position] = 0;
                    padding_at_position_two = true;
                }
                3 => {
                    quartet[position] = 0;
                }
                _ => unreachable!(),
            }
        } else {
            let value = match decode_value(byte) {
                Some(value) => value,
                None => return Err(DecodeError::InvalidByte),
            };
            quartet[position] = value;
        }

        position += 1;

        if position == 4 {
            if padding_at_position_two {
                output.push((quartet[0] << 2) | (quartet[1] >> 4));
                finished = true;
            } else if input_padding_at_position_three(input, &quartet, position) {
                output.push((quartet[0] << 2) | (quartet[1] >> 4));
                output.push((quartet[1] << 4) | (quartet[2] >> 2));
                finished = true;
            } else {
                output.push((quartet[0] << 2) | (quartet[1] >> 4));
                output.push((quartet[1] << 4) | (quartet[2] >> 2));
                output.push((quartet[2] << 6) | quartet[3]);
            }

            position = 0;
            padding_at_position_two = false;
        }
    }

    if position != 0 {
        return Err(DecodeError::InvalidLength);
    }

    Ok(output)
}

fn decode_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn input_padding_at_position_three(input: &str, quartet: &[u8; 4], position: usize) -> bool {
    let mut seen = 0usize;
    for byte in input.bytes() {
        if byte.is_ascii_whitespace() {
            continue;
        }
        seen += 1;
        if seen == position {
            return byte == b'=';
        }
    }
    false
}
