#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte,
    InvalidPadding,
    InvalidLength,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];

        output.push(ALPHABET[(a >> 2) as usize] as char);
        output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
        output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
        output.push(ALPHABET[(c & 0x3f) as usize] as char);
    }

    match chunks.remainder() {
        [a] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push('=');
            output.push('=');
        }
        [a, b] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        _ => {}
    }

    output
}

fn decode_sextet(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let mut output = Vec::with_capacity(input.len() / 4 * 3);
    let mut quartet = [0u8; 4];
    let mut position = 0usize;
    let mut padding_at_position_two = false;
    let mut finished = false;

    for byte in input.bytes() {
        if finished {
            return Err(DecodeError::InvalidPadding);
        }

        if byte == b'=' {
            if position < 2 {
                return Err(DecodeError::InvalidPadding);
            }
            if padding_at_position_two {
                if position != 3 {
                    return Err(DecodeError::InvalidPadding);
                }
                output.push((quartet[0] << 2) | (quartet[1] >> 4));
                position = 0;
                padding_at_position_two = false;
                finished = true;
            } else if position == 2 {
                padding_at_position_two = true;
                position += 1;
            } else {
                output.push((quartet[0] << 2) | (quartet[1] >> 4));
                output.push((quartet[1] << 4) | (quartet[2] >> 2));
                position = 0;
                finished = true;
            }
            continue;
        }

        if padding_at_position_two {
            return Err(DecodeError::InvalidPadding);
        }

        let sextet = decode_sextet(byte).ok_or(DecodeError::InvalidByte)?;
        quartet[position] = sextet;
        position += 1;

        if position == 4 {
            output.push((quartet[0] << 2) | (quartet[1] >> 4));
            output.push((quartet[1] << 4) | (quartet[2] >> 2));
            output.push((quartet[2] << 6) | quartet[3]);
            position = 0;
        }
    }

    if position != 0 {
        return Err(DecodeError::InvalidLength);
    }

    Ok(output)
}
