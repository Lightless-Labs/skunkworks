```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u64,
    pub priority: u32,
    pub ready_at: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Dispatch {
    pub job: Job,
    pub enqueued_at: u64,
    pub started_at: u64,
    pub effective_priority: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SchedulerError {
    ZeroAgingInterval,
    TimeWentBackwards,
    DuplicateId(u64),
}

struct PendingJob {
    job: Job,
    enqueued_at: u64,
    insertion_order: u128,
}

pub struct Scheduler {
    aging_interval_ms: u64,
    last_time: u64,
    next_order: u128,
    pending: Vec<PendingJob>,
    id_to_index: HashMap<u64, usize>,
}

impl Scheduler {
    pub fn new(aging_interval_ms: u64, start_ms: u64) -> Result<Self, SchedulerError> {
        if aging_interval_ms == 0 {
            return Err(SchedulerError::ZeroAgingInterval);
        }

        Ok(Self {
            aging_interval_ms,
            last_time: start_ms,
            next_order: 0,
            pending: Vec::new(),
            id_to_index: HashMap::new(),
        })
    }

    pub fn enqueue(&mut self, job: Job, now: u64) -> Result<(), SchedulerError> {
        self.check_time(now)?;
        if self.id_to_index.contains_key(&job.id) {
            return Err(SchedulerError::DuplicateId(job.id));
        }

        let order = self.next_order;
        self.pending.push(PendingJob {
            job,
            enqueued_at: now,
            insertion_order: order,
        });
        let index = self.pending.len() - 1;
        self.id_to_index.insert(self.pending[index].job.id, index);

        // At most u64::MAX insertions can be represented over the scheduler's
        // lifetime; saturating preserves the FIFO order for all practical and
        // API-representable pending sets.
        self.next_order = self.next_order.saturating_add(1);
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, SchedulerError> {
        self.check_time(now)?;

        let mut best: Option<(usize, u32, u128)> = None;
        for (index, pending) in self.pending.iter().enumerate() {
            if pending.job.ready_at > now {
                continue;
            }

            let age_steps = (now - pending.enqueued_at) / self.aging_interval_ms;
            let effective = (pending.job.priority as u64)
                .saturating_add(age_steps)
                .min(u32::MAX as u64) as u32;

            match best {
                None => best = Some((index, effective, pending.insertion_order)),
                Some((_, best_priority, best_order))
                    if effective > best_priority
                        || (effective == best_priority
                            && pending.insertion_order < best_order) =>
                {
                    best = Some((index, effective, pending.insertion_order));
                }
                _ => {}
            }
        }

        self.last_time = now;
        let Some((index, effective_priority, _)) = best else {
            return Ok(None);
        };

        let selected = self.pending.remove(index);
        self.rebuild_indices();
        Ok(Some(Dispatch {
            job: selected.job,
            enqueued_at: selected.enqueued_at,
            started_at: now,
            effective_priority,
        }))
    }

    pub fn cancel(&mut self, id: u64, now: u64) -> Result<Option<Job>, SchedulerError> {
        self.check_time(now)?;

        self.last_time = now;
        let Some(index) = self.id_to_index.get(&id).copied() else {
            return Ok(None);
        };

        let removed = self.pending.remove(index);
        self.rebuild_indices();
        Ok(Some(removed.job))
    }

    pub fn len(&self) -> usize {
        self.pending.len()
    }

    fn check_time(&self, now: u64) -> Result<(), SchedulerError> {
        if now < self.last_time {
            Err(SchedulerError::TimeWentBackwards)
        } else {
            Ok(())
        }
    }

    fn rebuild_indices(&mut self) {
        self.id_to_index.clear();
        for (index, pending) in self.pending.iter().enumerate() {
            self.id_to_index.insert(pending.job.id, index);
        }
    }
}
```
