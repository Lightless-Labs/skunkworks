```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u64,
    pub priority: u32,
    pub ready_at: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Dispatch {
    pub job: Job,
    pub enqueued_at: u64,
    pub started_at: u64,
    pub effective_priority: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SchedulerError {
    ZeroAgingInterval,
    TimeWentBackwards,
    DuplicateId(u64),
}

struct PendingJob {
    job: Job,
    enqueued_at: u64,
    insertion_order: u128,
}

pub struct Scheduler {
    aging_interval_ms: u64,
    last_time: u64,
    next_order: u128,
    pending: Vec<PendingJob>,
    by_id: HashMap<u64, usize>,
}

impl Scheduler {
    pub fn new(aging_interval_ms: u64, start_ms: u64) -> Result<Self, SchedulerError> {
        if aging_interval_ms == 0 {
            return Err(SchedulerError::ZeroAgingInterval);
        }

        Ok(Self {
            aging_interval_ms,
            last_time: start_ms,
            next_order: 0,
            pending: Vec::new(),
            by_id: HashMap::new(),
        })
    }

    pub fn enqueue(&mut self, job: Job, now: u64) -> Result<(), SchedulerError> {
        self.check_time(now)?;
        if self.by_id.contains_key(&job.id) {
            return Err(SchedulerError::DuplicateId(job.id));
        }

        let index = self.pending.len();
        self.pending.push(PendingJob {
            job,
            enqueued_at: now,
            insertion_order: self.next_order,
        });
        self.by_id.insert(self.pending[index].job.id, index);
        self.next_order = self.next_order.wrapping_add(1);
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, SchedulerError> {
        self.check_time(now)?;

        let mut selected: Option<(usize, u32, u128)> = None;
        for (index, pending) in self.pending.iter().enumerate() {
            if pending.job.ready_at > now {
                continue;
            }

            let age_steps = (now - pending.enqueued_at) / self.aging_interval_ms;
            let effective = (pending.job.priority as u64)
                .saturating_add(age_steps)
                .min(u32::MAX as u64) as u32;

            let should_select = match selected {
                None => true,
                Some((_, best_priority, best_order)) => {
                    effective > best_priority
                        || (effective == best_priority && pending.insertion_order < best_order)
                }
            };
            if should_select {
                selected = Some((index, effective, pending.insertion_order));
            }
        }

        self.last_time = now;
        let Some((index, effective_priority, _)) = selected else {
            return Ok(None);
        };

        let pending = self.pending.remove(index);
        self.rebuild_indices();
        Ok(Some(Dispatch {
            job: pending.job,
            enqueued_at: pending.enqueued_at,
            started_at: now,
            effective_priority,
        }))
    }

    pub fn cancel(&mut self, id: u64, now: u64) -> Result<Option<Job>, SchedulerError> {
        self.check_time(now)?;

        let result = self.by_id.get(&id).copied().map(|index| {
            let pending = self.pending.remove(index);
            self.rebuild_indices();
            pending.job
        });

        self.last_time = now;
        Ok(result)
    }

    pub fn len(&self) -> usize {
        self.pending.len()
    }

    fn check_time(&self, now: u64) -> Result<(), SchedulerError> {
        if now < self.last_time {
            Err(SchedulerError::TimeWentBackwards)
        } else {
            Ok(())
        }
    }

    fn rebuild_indices(&mut self) {
        self.by_id.clear();
        for (index, pending) in self.pending.iter().enumerate() {
            self.by_id.insert(pending.job.id, index);
        }
    }
}
```
