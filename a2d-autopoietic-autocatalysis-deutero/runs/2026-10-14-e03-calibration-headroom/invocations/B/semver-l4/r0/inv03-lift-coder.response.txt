#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub prerelease: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Section {
    Prerelease,
    Build,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    EmptyInput,
    ComponentCount { found: usize },
    EmptyComponent { index: usize },
    InvalidDigit { index: usize, offset: usize },
    LeadingZero { index: usize },
    Overflow { index: usize },
    EmptyIdentifier { section: Section, index: usize },
    InvalidIdentifier { section: Section, index: usize, offset: usize },
    NumericLeadingZero { index: usize },
}

fn parse_core(core: &str) -> Result<(u64, u64, u64), ParseError> {
    if core.is_empty() {
        return Err(ParseError::EmptyInput);
    }
    let parts: Vec<&str> = core.split('.').collect();
    if parts.len() != 3 {
        return Err(ParseError::ComponentCount { found: parts.len() });
    }

    let mut values = [0u64; 3];
    for (index, part) in parts.iter().enumerate() {
        if part.is_empty() {
            return Err(ParseError::EmptyComponent { index });
        }
        if let Some(offset) = part.bytes().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidDigit { index, offset });
        }
        if part.len() > 1 && part.starts_with('0') {
            return Err(ParseError::LeadingZero { index });
        }
        values[index] = part
            .parse::<u64>()
            .map_err(|_| ParseError::Overflow { index })?;
    }
    Ok((values[0], values[1], values[2]))
}

fn parse_identifiers(text: &str, section: Section) -> Result<Vec<String>, ParseError> {
    let mut result = Vec::new();
    for (index, identifier) in text.split('.').enumerate() {
        if identifier.is_empty() {
            return Err(ParseError::EmptyIdentifier { section, index });
        }
        if let Some(offset) = identifier.bytes().position(|b| {
            !(b.is_ascii_alphanumeric() || b == b'-')
        }) {
            return Err(ParseError::InvalidIdentifier { section, index, offset });
        }
        if section == Section::Prerelease
            && identifier.len() > 1
            && identifier.starts_with('0')
            && identifier.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(ParseError::NumericLeadingZero { index });
        }
        result.push(identifier.to_owned());
    }
    Ok(result)
}

pub fn parse_version(input: &str) -> Result<Version, ParseError> {
    if input.is_empty() {
        return Err(ParseError::EmptyInput);
    }

    let (before_build, build_text) = match input.split_once('+') {
        Some((left, right)) => (left, Some(right)),
        None => (input, None),
    };
    let (core_text, prerelease_text) = match before_build.split_once('-') {
        Some((left, right)) => (left, Some(right)),
        None => (before_build, None),
    };

    let (major, minor, patch) = parse_core(core_text)?;
    let prerelease = match prerelease_text {
        Some(text) => parse_identifiers(text, Section::Prerelease)?,
        None => Vec::new(),
    };
    let build = match build_text {
        Some(text) => parse_identifiers(text, Section::Build)?,
        None => Vec::new(),
    };

    Ok(Version {
        major,
        minor,
        patch,
        prerelease,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_strings(a: &str, b: &str) -> std::cmp::Ordering {
    let a_trimmed = a.trim_start_matches('0');
    let b_trimmed = b.trim_start_matches('0');
    let a_norm = if a_trimmed.is_empty() { "0" } else { a_trimmed };
    let b_norm = if b_trimmed.is_empty() { "0" } else { b_trimmed };
    a_norm
        .len()
        .cmp(&b_norm.len())
        .then_with(|| a_norm.cmp(b_norm))
}

pub fn precedence(a: &Version, b: &Version) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    let core_cmp = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core_cmp != Ordering::Equal {
        return core_cmp;
    }

    match (a.prerelease.is_empty(), b.prerelease.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.prerelease.iter().zip(&b.prerelease) {
        let left_numeric = numeric_identifier(left);
        let right_numeric = numeric_identifier(right);
        let cmp = match (left_numeric, right_numeric) {
            (true, true) => compare_numeric_strings(left, right),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };
        if cmp != Ordering::Equal {
            return cmp;
        }
    }
    a.prerelease.len().cmp(&b.prerelease.len())
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Op {
    Lt,
    Le,
    Eq,
    Ge,
    Gt,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Comparator {
    pub op: Op,
    pub version: Version,
    pub allows_prerelease: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Range {
    pub alternatives: Vec<Vec<Comparator>>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RangeError {
    EmptyAlternative { alternative: usize },
    InvalidVersion { alternative: usize, index: usize, error: ParseError },
    InvalidToken { alternative: usize, index: usize },
    BoundOverflow { alternative: usize, index: usize },
}

fn ascii_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\n' | b'\r' | 0x0b | 0x0c)
}

fn split_ascii_whitespace(s: &str) -> Vec<&str> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut start = None;
    for (i, &b) in bytes.iter().enumerate() {
        if ascii_whitespace(b) {
            if let Some(begin) = start.take() {
                tokens.push(&s[begin..i]);
            }
        } else if start.is_none() {
            start = Some(i);
        }
    }
    if let Some(begin) = start {
        tokens.push(&s[begin..])
    }
    tokens
}

fn is_all_ascii_whitespace(s: &str) -> bool {
    s.bytes().all(ascii_whitespace)
}

#[derive(Clone, Copy)]
struct Endpoint {
    version: Version,
    supplied: usize,
}

fn parse_endpoint(text: &str) -> Option<Endpoint> {
    if text.is_empty() || text.starts_with(['^', '~', '>', '<', '=', '*']) {
        return None;
    }
    if text.contains('*') || text.contains('x') || text.contains('X') {
        return None;
    }

    let plus = text.find('+');
    let before_build = plus.map_or(text, |i| &text[..i]);
    let has_suffix = before_build.contains('-') || plus.is_some();
    let core = before_build.split_once('-').map_or(before_build, |(c, _)| c);
    let supplied = core.split('.').count();
    if !(1..=3).contains(&supplied) || (has_suffix && supplied != 3) {
        return None;
    }

    let parts: Vec<&str> = core.split('.').collect();
    let mut padded = parts.join(".");
    for _ in supplied..3 {
        padded.push_str(".0");
    }

    let version = if supplied == 3 {
        parse_version(text).ok()?
    } else {
        let (major, minor, patch) = parse_core(&padded).ok()?;
        Version { major, minor, patch, prerelease: Vec::new(), build: Vec::new() }
    };
    Some(Endpoint { version, supplied })
}

fn upper_bound(
    endpoint: &Endpoint,
    axis: usize,
    alternative: usize,
    index: usize,
) -> Result<Version, RangeError> {
    let mut version = endpoint.version.clone();
    let component = match axis {
        0 => &mut version.major,
        1 => &mut version.minor,
        _ => &mut version.patch,
    };
    *component = component.checked_add(1).ok_or(RangeError::BoundOverflow {
        alternative,
        index,
    })?;
    match axis {
        0 => {
            version.minor = 0;
            version.patch = 0;
        }
        1 => version.patch = 0,
        _ => {}
    }
    version.prerelease = vec!["0".to_owned()];
    version.build.clear();
    Ok(version)
}

fn comparator(op: Op, version: Version) -> Comparator {
    let allows_prerelease = !version.prerelease.is_empty();
    Comparator { op, version, allows_prerelease }
}

fn exclusive_upper(version: Version) -> Comparator {
    Comparator { op: Op::Lt, version, allows_prerelease: false }
}

fn invalid_token(alternative: usize, index: usize) -> RangeError {
    RangeError::InvalidToken { alternative, index }
}

fn expand_token(
    token: &str,
    alternative: usize,
    index: usize,
) -> Result<Vec<Comparator>, RangeError> {
    let explicit = if token.starts_with(">=") {
        Some((Op::Ge, &token[2..]))
    } else if token.starts_with("<=") {
        Some((Op::Le, &token[2..]))
    } else if token.starts_with('>') {
        Some((Op::Gt, &token[1..]))
    } else if token.starts_with('<') {
        Some((Op::Lt, &token[1..]))
    } else if token.starts_with('=') {
        Some((Op::Eq, &token[1..]))
    } else {
        None
    };
    if let Some((op, remainder)) = explicit {
        return match parse_version(remainder) {
            Ok(version) => Ok(vec![comparator(op, version)]),
            Err(error) => Err(RangeError::InvalidVersion { alternative, index, error }),
        };
    }

    if matches!(token, "*" | "x" | "X") {
        return Ok(Vec::new());
    }

    if let Some(rest) = token.strip_prefix('^') {
        let endpoint = parse_endpoint(rest).ok_or_else(|| invalid_token(alternative, index))?;
        let axis = if endpoint.version.major > 0 || endpoint.supplied == 1 {
            0
        } else if endpoint.version.minor > 0 || endpoint.supplied == 2 {
            1
        } else {
            2
        };
        let upper = upper_bound(&endpoint, axis, alternative, index)?;
        return Ok(vec![
            comparator(Op::Ge, endpoint.version),
            exclusive_upper(upper),
        ]);
    }

    if let Some(rest) = token.strip_prefix('~') {
        let endpoint = parse_endpoint(rest).ok_or_else(|| invalid_token(alternative, index))?;
        let axis = if endpoint.supplied == 1 { 0 } else { 1 };
        let upper = upper_bound(&endpoint, axis, alternative, index)?;
        return Ok(vec![
            comparator(Op::Ge, endpoint.version),
            exclusive_upper(upper),
        ]);
    }

    let components: Vec<&str> = token.split('.').collect();
    if let Some(last) = components.last().copied() {
        if matches!(last, "*" | "x" | "X") {
            if components.len() < 2 || components.len() > 3 {
                return Err(invalid_token(alternative, index));
            }
            if components[..components.len() - 1]
                .iter()
                .any(|p| matches!(*p, "*" | "x" | "X"))
            {
                return Err(invalid_token(alternative, index));
            }
            let numeric = components[..components.len() - 1].join(".");
            let endpoint = parse_endpoint(&numeric)
                .filter(|e| e.supplied == components.len() - 1)
                .ok_or_else(|| invalid_token(alternative, index))?;
            let axis = endpoint.supplied - 1;
            let upper = upper_bound(&endpoint, axis, alternative, index)?;
            return Ok(vec![
                comparator(Op::Ge, endpoint.version),
                exclusive_upper(upper),
            ]);
        }
    }

    let endpoint = parse_endpoint(token).ok_or_else(|| invalid_token(alternative, index))?;
    if endpoint.supplied == 3 {
        Ok(vec![comparator(Op::Eq, endpoint.version)])
    } else {
        let upper = upper_bound(&endpoint, endpoint.supplied - 1, alternative, index)?;
        Ok(vec![
            comparator(Op::Ge, endpoint.version),
            exclusive_upper(upper),
        ])
    }
}

fn expand_hyphen(
    left: &str,
    right: &str,
    alternative: usize,
) -> Result<Vec<Comparator>, RangeError> {
    let left_endpoint = parse_endpoint(left).ok_or_else(|| invalid_token(alternative, 0))?;
    let right_endpoint = parse_endpoint(right).ok_or_else(|| invalid_token(alternative, 2))?;

    let mut result = vec![comparator(Op::Ge, left_endpoint.version)];
    if right_endpoint.supplied == 3 {
        result.push(comparator(Op::Le, right_endpoint.version));
    } else {
        let upper = upper_bound(
            &right_endpoint,
            right_endpoint.supplied - 1,
            alternative,
            2,
        )?;
        result.push(exclusive_upper(upper));
    }
    Ok(result)
}

pub fn parse_range(input: &str) -> Result<Range, RangeError> {
    if is_all_ascii_whitespace(input) {
        return Ok(Range { alternatives: vec![Vec::new()] });
    }

    let raw_alternatives: Vec<&str> = input.split("||").collect();
    let mut alternatives = Vec::with_capacity(raw_alternatives.len());
    for (alternative, raw) in raw_alternatives.iter().enumerate() {
        let tokens = split_ascii_whitespace(raw);
        if tokens.is_empty() {
            return Err(RangeError::EmptyAlternative { alternative });
        }
        if tokens.len() == 3 && tokens[1] == "-" {
            alternatives.push(expand_hyphen(tokens[0], tokens[2], alternative)?);
            continue;
        }

        let mut comparators = Vec::new();
        for (index, token) in tokens.iter().enumerate() {
            comparators.extend(expand_token(token, alternative, index)?);
        }
        alternatives.push(comparators);
    }
    Ok(Range { alternatives })
}

fn relation_holds(candidate: &Version, comparator: &Comparator) -> bool {
    use std::cmp::Ordering;
    match precedence(candidate, &comparator.version) {
        Ordering::Less => matches!(comparator.op, Op::Lt | Op::Le),
        Ordering::Equal => matches!(comparator.op, Op::Le | Op::Eq | Op::Ge),
        Ordering::Greater => matches!(comparator.op, Op::Gt | Op::Ge),
    }
}

pub fn satisfies(version: &Version, range: &Range) -> bool {
    range.alternatives.iter().any(|alternative| {
        if !alternative.iter().all(|c| relation_holds(version, c)) {
            return false;
        }
        if version.prerelease.is_empty() {
            return true;
        }
        alternative.iter().any(|c| {
            c.allows_prerelease
                && (c.version.major, c.version.minor, c.version.patch)
                    == (version.major, version.minor, version.patch)
        })
    })
}

pub fn max_satisfying<'a>(versions: &'a [Version], range: &Range) -> Option<&'a Version> {
    let mut best: Option<&Version> = None;
    for version in versions {
        if satisfies(version, range)
            && best.map_or(true, |current| precedence(version, current).is_gt())
        {
            best = Some(version);
        }
    }
    best
}
