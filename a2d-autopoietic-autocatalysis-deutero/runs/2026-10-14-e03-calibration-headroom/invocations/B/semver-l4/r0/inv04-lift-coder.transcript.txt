fn parse_core(parts: &[&str]) -> Result<[u64; 3], ParseError> {
    if parts.len() != 3 {
        return Err(ParseError::ComponentCount { found: parts.len() });
    }

    let mut values = [0u64; 3];
    for (index, part) in parts.iter().enumerate() {
        if part.is_empty() {
            return Err(ParseError::EmptyComponent { index });
        }
        if let Some(offset) = part.bytes().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidDigit { index, offset });
        }
        if part.len() > 1 && part.starts_with('0') {
            return Err(ParseError::LeadingZero { index });
        }
        values[index] = part.parse::<u64>().map_err(|_| ParseError::Overflow { index })?;
    }
    Ok(values)
}

fn parse_identifiers(text: &str, section: Section) -> Result<Vec<String>, ParseError> {

    for (index, identifier) in text.split('.').enumerate() {

pub fn parse_version(input: &str) -> Result<Version, ParseError> {
    if input.is_empty() {
        return Err(ParseError::EmptyInput);

    let (before_build, build_text) = match input.split_once('+') {
        Some((left, right)) => (left, Some(right)),
        None => (input, None),
    };
    let (core_text, prerelease_text) = match before_build.split_once('-') {
        Some((left, right)) => (left, Some(right)),
        None => (before_build, None),

fn tokens(s: &str) -> Vec<&str> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ascii_whitespace(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !ascii_whitespace(bytes[i]) {
            i += 1;
        }
        out.push(&s[start..i]);
    }
    out
}

fn parse_endpoint(s: &str) -> Option<Endpoint> {
    if s.is_empty() || s.starts_with(['^', '~', '<', '>', '=', '+', '-']) {

    } else {

        if ts.len() == 3 && ts[1] == "-" {
            expand_hyphen(ts[0], ts[2], alternative, &mut comparators)?;
        } else {
            for (index, token) in ts.iter().enumerate() {
                expand_token(token, alternative, index, &mut comparators)?;
            }
        }
        alternatives.push(comparators);
    }
    Ok(Range { alternatives })
}

fn relation(candidate: &Version, comparator: &Comparator) -> bool {
    use std::cmp::Ordering;
    let ord = precedence(candidate, &comparator.version);
    match comparator.op {
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Eq => ord == Ordering::Equal,
        Op::Ge => ord != Ordering::Less,
        Op::Gt => ord == Ordering::Greater,
    }
}

pub fn satisfies(version: &Version, range: &Range) -> bool {
    range.alternatives.iter().any(|alternative| {
        if !alternative.iter().all(|c| relation(version, c)) {
            return false;
        }
        if version.prerelease.is_empty() {
            return true;
        }
        alternative.iter().any(|c| {
            c.allows_prerelease
                && (version.major, version.minor, version.patch)
                    == (c.version.major, c.version.minor, c.version.patch)
        })
    })
}

pub fn max_satisfying<'a>(versions: &'a [Version], range: &Range) -> Option<&'a Version> {
    let mut best: Option<&Version> = None;
    for version in versions {
        if satisfies(version, range)
            && best.is_none_or(|current| precedence(version, current).is_gt())
        {
            best = Some(version);
        }
    }
    best
}
