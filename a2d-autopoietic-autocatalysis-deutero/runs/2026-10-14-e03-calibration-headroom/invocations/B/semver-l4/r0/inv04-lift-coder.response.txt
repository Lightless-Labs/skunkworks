#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub prerelease: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Section {
    Prerelease,
    Build,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    EmptyInput,
    ComponentCount { found: usize },
    EmptyComponent { index: usize },
    InvalidDigit { index: usize, offset: usize },
    LeadingZero { index: usize },
    Overflow { index: usize },
    EmptyIdentifier { section: Section, index: usize },
    InvalidIdentifier { section: Section, index: usize, offset: usize },
    NumericLeadingZero { index: usize },
}

fn parse_core(parts: &[&str]) -> Result<[u64; 3], ParseError> {
    if parts.len() != 3 {
        return Err(ParseError::ComponentCount { found: parts.len() });
    }

    let mut values = [0u64; 3];
    for (index, part) in parts.iter().enumerate() {
        if part.is_empty() {
            return Err(ParseError::EmptyComponent { index });
        }
        if let Some(offset) = part.bytes().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidDigit { index, offset });
        }
        if part.len() > 1 && part.starts_with('0') {
            return Err(ParseError::LeadingZero { index });
        }
        values[index] = part.parse::<u64>().map_err(|_| ParseError::Overflow { index })?;
    }
    Ok(values)
}

fn parse_identifiers(text: &str, section: Section) -> Result<Vec<String>, ParseError> {
    let mut result = Vec::new();
    for (index, identifier) in text.split('.').enumerate() {
        if identifier.is_empty() {
            return Err(ParseError::EmptyIdentifier { section, index });
        }
        if let Some(offset) = identifier.bytes().position(|b| {
            !(b.is_ascii_alphanumeric() || b == b'-')
        }) {
            return Err(ParseError::InvalidIdentifier { section, index, offset });
        }
        if section == Section::Prerelease
            && identifier.len() > 1
            && identifier.starts_with('0')
            && identifier.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(ParseError::NumericLeadingZero { index });
        }
        result.push(identifier.to_string());
    }
    Ok(result)
}

pub fn parse_version(input: &str) -> Result<Version, ParseError> {
    if input.is_empty() {
        return Err(ParseError::EmptyInput);
    }

    let (before_build, build_text) = match input.split_once('+') {
        Some((left, right)) => (left, Some(right)),
        None => (input, None),
    };
    let (core_text, prerelease_text) = match before_build.split_once('-') {
        Some((left, right)) => (left, Some(right)),
        None => (before_build, None),
    };

    let core_parts: Vec<&str> = core_text.split('.').collect();
    let [major, minor, patch] = parse_core(&core_parts)?;
    let prerelease = match prerelease_text {
        Some(text) => parse_identifiers(text, Section::Prerelease)?,
        None => Vec::new(),
    };
    let build = match build_text {
        Some(text) => parse_identifiers(text, Section::Build)?,
        None => Vec::new(),
    };

    Ok(Version { major, minor, patch, prerelease, build })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_strings(a: &str, b: &str) -> std::cmp::Ordering {
    let a = a.trim_start_matches('0');
    let b = b.trim_start_matches('0');
    let a = if a.is_empty() { "0" } else { a };
    let b = if b.is_empty() { "0" } else { b };
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

pub fn precedence(a: &Version, b: &Version) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.prerelease.is_empty(), b.prerelease.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.prerelease.iter().zip(&b.prerelease) {
                    let ord = match (numeric_identifier(x), numeric_identifier(y)) {
                        (true, true) => compare_numeric_strings(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.cmp(y),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.prerelease.len().cmp(&b.prerelease.len())
            }
        })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Op {
    Lt,
    Le,
    Eq,
    Ge,
    Gt,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Comparator {
    pub op: Op,
    pub version: Version,
    pub allows_prerelease: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Range {
    pub alternatives: Vec<Vec<Comparator>>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RangeError {
    EmptyAlternative { alternative: usize },
    InvalidVersion { alternative: usize, index: usize, error: ParseError },
    InvalidToken { alternative: usize, index: usize },
    BoundOverflow { alternative: usize, index: usize },
}

fn ascii_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\n' | b'\r' | 0x0b | 0x0c)
}

fn tokens(s: &str) -> Vec<&str> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ascii_whitespace(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !ascii_whitespace(bytes[i]) {
            i += 1;
        }
        out.push(&s[start..i]);
    }
    out
}

fn version_from_core(core: [u64; 3], prerelease: Vec<String>) -> Version {
    Version {
        major: core[0], minor: core[1], patch: core[2],
        prerelease, build: Vec::new(),
    }
}

fn upper_bound(
    core: [u64; 3],
    axis: usize,
    alternative: usize,
    index: usize,
) -> Result<Version, RangeError> {
    let mut next = core;
    next[axis] = next[axis].checked_add(1).ok_or(RangeError::BoundOverflow { alternative, index })?;
    for slot in next.iter_mut().skip(axis + 1) {
        *slot = 0;
    }
    Ok(version_from_core(next, vec!["0".to_string()]))
}

struct Endpoint {
    version: Version,
    supplied: usize,
}

fn parse_endpoint(s: &str) -> Option<Endpoint> {
    if s.is_empty() || s.starts_with(['^', '~', '<', '>', '=', '+', '-']) {
        return None;
    }
    let parts: Vec<&str> = s.split('.').collect();
    if !(1..=3).contains(&parts.len()) {
        return None;
    }
    if parts.iter().any(|p| p.is_empty() || p.contains(['*', 'x', 'X', '+'])) {
        return None;
    }

    let (core_text, suffix) = if parts.len() == 3 {
        (s, true)
    } else {
        (s, false)
    };
    let _ = suffix;

    let mut padded: Vec<String> = parts.iter().map(|p| (*p).to_string()).collect();
    while padded.len() < 3 {
        padded.push("0".to_string());
    }
    let refs: Vec<&str> = padded.iter().map(String::as_str).collect();
    let core = parse_core(&refs).ok()?;

    let version = if parts.len() == 3 {
        parse_version(core_text).ok()?
    } else {
        version_from_core(core, Vec::new())
    };
    Some(Endpoint { version, supplied: parts.len() })
}

fn invalid_token(alternative: usize, index: usize) -> RangeError {
    RangeError::InvalidToken { alternative, index }
}

fn add_comparator(out: &mut Vec<Comparator>, op: Op, version: Version, allows: bool) {
    out.push(Comparator { op, version, allows_prerelease: allows });
}

fn expand_token(
    token: &str,
    alternative: usize,
    index: usize,
    out: &mut Vec<Comparator>,
) -> Result<(), RangeError> {
    let explicit = if token.starts_with(">=") {
        Some((Op::Ge, &token[2..]))
    } else if token.starts_with("<=") {
        Some((Op::Le, &token[2..]))
    } else if token.starts_with('>') {
        Some((Op::Gt, &token[1..]))
    } else if token.starts_with('<') {
        Some((Op::Lt, &token[1..]))
    } else if token.starts_with('=') {
        Some((Op::Eq, &token[1..]))
    } else {
        None
    };

    if let Some((op, remainder)) = explicit {
        let version = parse_version(remainder).map_err(|error| RangeError::InvalidVersion {
            alternative, index, error,
        })?;
        let allows = !version.prerelease.is_empty();
        add_comparator(out, op, version, allows);
        return Ok(());
    }

    if token == "*" || token == "x" || token == "X" {
        return Ok(());
    }

    if let Some(rest) = token.strip_prefix('^').or_else(|| token.strip_prefix('~')) {
        let is_caret = token.starts_with('^');
        let endpoint = parse_endpoint(rest).ok_or_else(|| invalid_token(alternative, index))?;
        let core = [endpoint.version.major, endpoint.version.minor, endpoint.version.patch];
        let allows = !endpoint.version.prerelease.is_empty();
        add_comparator(out, Op::Ge, endpoint.version.clone(), allows);

        let axis = if is_caret {
            if core[0] > 0 || endpoint.supplied == 1 {
                0
            } else if core[1] > 0 || endpoint.supplied == 2 {
                1
            } else {
                2
            }
        } else if endpoint.supplied == 1 {
            0
        } else {
            1
        };
        let upper = upper_bound(core, axis, alternative, index)?;
        add_comparator(out, Op::Lt, upper, false);
        return Ok(());
    }

    let parts: Vec<&str> = token.split('.').collect();
    if parts.iter().any(|p| matches!(*p, "*" | "x" | "X")) {
        if parts.len() < 2 || parts.len() > 3 {
            return Err(invalid_token(alternative, index));
        }
        let wild = parts.iter().position(|p| matches!(*p, "*" | "x" | "X"))
            .ok_or_else(|| invalid_token(alternative, index))?;
        if wild != parts.len() - 1 || wild == 0 {
            return Err(invalid_token(alternative, index));
        }
        let numeric = &parts[..wild];
        let mut padded: Vec<String> = numeric.iter().map(|p| (*p).to_string()).collect();
        while padded.len() < 3 {
            padded.push("0".to_string());
        }
        let refs: Vec<&str> = padded.iter().map(String::as_str).collect();
        let core = parse_core(&refs).map_err(|_| invalid_token(alternative, index))?;
        let lower = version_from_core(core, Vec::new());
        add_comparator(out, Op::Ge, lower, false);
        let upper = upper_bound(core, wild - 1, alternative, index)?;
        add_comparator(out, Op::Lt, upper, false);
        return Ok(());
    }

    let endpoint = parse_endpoint(token).ok_or_else(|| invalid_token(alternative, index))?;
    let core = [endpoint.version.major, endpoint.version.minor, endpoint.version.patch];
    let allows = !endpoint.version.prerelease.is_empty();
    if endpoint.supplied == 3 {
        add_comparator(out, Op::Eq, endpoint.version, allows);
    } else {
        add_comparator(out, Op::Ge, endpoint.version, allows);
        let upper = upper_bound(core, endpoint.supplied - 1, alternative, index)?;
        add_comparator(out, Op::Lt, upper, false);
    }
    Ok(())
}

fn expand_hyphen(
    left: &str,
    right: &str,
    alternative: usize,
    out: &mut Vec<Comparator>,
) -> Result<(), RangeError> {
    let left_endpoint = parse_endpoint(left).ok_or_else(|| invalid_token(alternative, 0))?;
    let right_endpoint = parse_endpoint(right).ok_or_else(|| invalid_token(alternative, 2))?;

    let left_allows = !left_endpoint.version.prerelease.is_empty();
    add_comparator(out, Op::Ge, left_endpoint.version, left_allows);

    if right_endpoint.supplied == 3 {
        let allows = !right_endpoint.version.prerelease.is_empty();
        add_comparator(out, Op::Le, right_endpoint.version, allows);
    } else {
        let core = [right_endpoint.version.major, right_endpoint.version.minor, right_endpoint.version.patch];
        let upper = upper_bound(core, right_endpoint.supplied - 1, alternative, 2)?;
        add_comparator(out, Op::Lt, upper, false);
    }
    Ok(())
}

pub fn parse_range(input: &str) -> Result<Range, RangeError> {
    let all_ws = input.bytes().all(ascii_whitespace);
    if input.is_empty() || all_ws {
        return Ok(Range { alternatives: vec![Vec::new()] });
    }

    let mut alternatives = Vec::new();
    for (alternative, text) in input.split("||").enumerate() {
        let ts = tokens(text);
        if ts.is_empty() {
            return Err(RangeError::EmptyAlternative { alternative });
        }
        let mut comparators = Vec::new();
        if ts.len() == 3 && ts[1] == "-" {
            expand_hyphen(ts[0], ts[2], alternative, &mut comparators)?;
        } else {
            for (index, token) in ts.iter().enumerate() {
                expand_token(token, alternative, index, &mut comparators)?;
            }
        }
        alternatives.push(comparators);
    }
    Ok(Range { alternatives })
}

fn relation(candidate: &Version, comparator: &Comparator) -> bool {
    use std::cmp::Ordering;
    let ord = precedence(candidate, &comparator.version);
    match comparator.op {
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Eq => ord == Ordering::Equal,
        Op::Ge => ord != Ordering::Less,
        Op::Gt => ord == Ordering::Greater,
    }
}

pub fn satisfies(version: &Version, range: &Range) -> bool {
    range.alternatives.iter().any(|alternative| {
        if !alternative.iter().all(|c| relation(version, c)) {
            return false;
        }
        if version.prerelease.is_empty() {
            return true;
        }
        alternative.iter().any(|c| {
            c.allows_prerelease
                && (version.major, version.minor, version.patch)
                    == (c.version.major, c.version.minor, c.version.patch)
        })
    })
}

pub fn max_satisfying<'a>(versions: &'a [Version], range: &Range) -> Option<&'a Version> {
    let mut best: Option<&Version> = None;
    for version in versions {
        if satisfies(version, range)
            && best.is_none_or(|current| precedence(version, current).is_gt())
        {
            best = Some(version);
        }
    }
    best
}
