```rust
use std::collections::HashMap;

#[derive(Debug, PartialEq, Eq)]
pub enum Error {
    NoTransaction,
    UnknownSnapshot(u64),
}

pub struct Store {
    values: HashMap<String, String>,
    transactions: Vec<HashMap<String, String>>,
    snapshots: HashMap<u64, HashMap<String, String>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            values: HashMap::new(),
            transactions: Vec::new(),
            snapshots: HashMap::new(),
        }
    }

    pub fn set(&mut self, key: &str, value: &str) {
        self.values.insert(key.to_owned(), value.to_owned());
    }

    pub fn get(&self, key: &str) -> Option<String> {
        self.values.get(key).cloned()
    }

    pub fn delete(&mut self, key: &str) -> bool {
        self.values.remove(key).is_some()
    }

    pub fn count(&self, value: &str) -> usize {
        self.values.values().filter(|v| v.as_str() == value).count()
    }

    pub fn begin(&mut self) {
        self.transactions.push(self.values.clone());
    }

    pub fn commit(&mut self) -> Result<(), Error> {
        if self.transactions.is_empty() {
            return Err(Error::NoTransaction);
        }
        self.transactions.clear();
        Ok(())
    }

    pub fn rollback(&mut self) -> Result<(), Error> {
        match self.transactions.pop() {
            Some(saved) => {
                self.values = saved;
                Ok(())
            }
            None => Err(Error::NoTransaction),
        }
    }

    pub fn snapshot(&mut self, id: u64) {
        self.snapshots.insert(id, self.values.clone());
    }

    pub fn restore(&mut self, id: u64) -> Result<(), Error> {
        match self.snapshots.get(&id) {
            Some(saved) => {
                self.values = saved.clone();
                Ok(())
            }
            None => Err(Error::UnknownSnapshot(id)),
        }
    }
}
```
