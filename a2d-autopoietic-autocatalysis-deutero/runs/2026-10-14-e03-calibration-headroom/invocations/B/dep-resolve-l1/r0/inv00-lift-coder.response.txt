```rust
use std::cmp::Reverse;
use std::collections::{BTreeSet, BinaryHeap, HashMap};

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub dependencies: Vec<String>,
}

pub fn install_order(packages: &[Package]) -> Vec<String> {
    let indices: HashMap<&str, usize> = packages
        .iter()
        .enumerate()
        .map(|(index, package)| (package.name.as_str(), index))
        .collect();

    let mut dependents = vec![BTreeSet::new(); packages.len()];
    let mut remaining_dependencies = vec![0usize; packages.len()];

    for (dependent, package) in packages.iter().enumerate() {
        let unique_dependencies: BTreeSet<&str> =
            package.dependencies.iter().map(String::as_str).collect();

        remaining_dependencies[dependent] = unique_dependencies.len();
        for dependency in unique_dependencies {
            let dependency_index = indices[dependency];
            dependents[dependency_index].insert(dependent);
        }
    }

    let mut ready = BinaryHeap::new();
    for (index, package) in packages.iter().enumerate() {
        if remaining_dependencies[index] == 0 {
            ready.push(Reverse((package.name.as_str(), index)));
        }
    }

    let mut order = Vec::with_capacity(packages.len());
    while let Some(Reverse((_, index))) = ready.pop() {
        order.push(packages[index].name.clone());

        for &dependent in &dependents[index] {
            remaining_dependencies[dependent] -= 1;
            if remaining_dependencies[dependent] == 0 {
                ready.push(Reverse((packages[dependent].name.as_str(), dependent)));
            }
        }
    }

    order
}
```
