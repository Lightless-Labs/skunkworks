#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub ignore_whitespace: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize },
    InvalidPadding { offset: usize },
    InvalidLength { offset: usize },
    TrailingData { offset: usize },
    NonCanonical { offset: usize },
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StreamError {
    Finished,
}

fn alphabet_byte(alphabet: Alphabet, value: u8) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn decode_value(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn is_ignored_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\n' | 0x0b | 0x0c | b'\r')
}

fn encode_full_group(bytes: &[u8], alphabet: Alphabet, output: &mut String) {
    let a = bytes[0];
    let b = bytes[1];
    let c = bytes[2];

    let values = [
        a >> 2,
        ((a & 0x03) << 4) | (b >> 4),
        ((b & 0x0f) << 2) | (c >> 6),
        c & 0x3f,
    ];

    for value in values {
        output.push(alphabet_byte(alphabet, value) as char);
    }
}

fn encode_tail(bytes: &[u8], options: Options, output: &mut String) {
    if bytes.is_empty() {
        return;
    }

    let a = bytes[0];
    let second = if bytes.len() > 1 { bytes[1] } else { 0 };

    output.push(alphabet_byte(options.alphabet, a >> 2) as char);
    output.push(
        alphabet_byte(
            options.alphabet,
            ((a & 0x03) << 4) | (second >> 4),
        ) as char,
    );

    if bytes.len() > 1 {
        output.push(
            alphabet_byte(options.alphabet, (second & 0x0f) << 2) as char,
        );
        if options.padding == Padding::Required {
            output.push('=');
        }
    } else if options.padding == Padding::Required {
        output.push_str("==");
    }
}

pub fn encode(input: &[u8], options: Options) -> String {
    let mut output = String::with_capacity((input.len() + 2) / 3 * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        encode_full_group(chunk, options.alphabet, &mut output);
    }
    encode_tail(chunks.remainder(), options, &mut output);

    output
}

#[derive(Debug, Clone, Copy)]
struct Symbol {
    value: u8,
    offset: usize,
}

#[derive(Debug, Clone, Copy)]
enum QuartetPart {
    Symbol(Symbol),
    Padding(usize),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum PaddingState {
    None,
    NeedSecondEquals,
    NeedFourthEquals,
}

#[derive(Debug, Clone, Copy)]
struct DecodeState {
    quartet: [Option<QuartetPart>; 4],
    len: usize,
    padding_state: PaddingState,
    terminal_padding: bool,
    offset: usize,
    finished: bool,
    error: Option<DecodeError>,
}

impl DecodeState {
    const fn new() -> Self {
        Self {
            quartet: [None; 4],
            len: 0,
            padding_state: PaddingState::None,
            terminal_padding: false,
            offset: 0,
            finished: false,
            error: None,
        }
    }

    fn fail(&mut self, error: DecodeError) -> DecodeError {
        self.error = Some(error);
        error
    }

    fn process_byte(&mut self, byte: u8, options: Options, output: &mut Vec<u8>) -> Result<(), DecodeError> {
        let offset = self.offset;
        self.offset += 1;

        if options.ignore_whitespace && is_ignored_whitespace(byte) {
            return Ok(());
        }

        if self.terminal_padding {
            return Err(self.fail(DecodeError::TrailingData { offset }));
        }

        if byte == b'=' {
            if options.padding == Padding::Forbidden || self.len < 2 {
                return Err(self.fail(DecodeError::InvalidPadding { offset }));
            }

            match self.padding_state {
                PaddingState::NeedSecondEquals => {
                    if self.len != 3 {
                        return Err(self.fail(DecodeError::InvalidPadding { offset }));
                    }
                    self.quartet[3] = Some(QuartetPart::Padding(offset));
                    self.len = 4;
                    self.padding_state = PaddingState::None;
                    self.complete_quartet(output)?;
                    self.terminal_padding = true;
                    return Ok(());
                }
                PaddingState::NeedFourthEquals => {
                    if self.len != 3 {
                        return Err(self.fail(DecodeError::InvalidPadding { offset }));
                    }
                    self.quartet[3] = Some(QuartetPart::Padding(offset));
                    self.len = 4;
                    self.padding_state = PaddingState::None;
                    self.complete_quartet(output)?;
                    self.terminal_padding = true;
                    return Ok(());
                }
                PaddingState::None => {
                    if self.len == 2 {
                        self.quartet[2] = Some(QuartetPart::Padding(offset));
                        self.len = 3;
                        self.padding_state = PaddingState::NeedSecondEquals;
                        return Ok(());
                    }
                    if self.len == 3 {
                        self.quartet[3] = Some(QuartetPart::Padding(offset));
                        self.len = 4;
                        self.complete_quartet(output)?;
                        self.terminal_padding = true;
                        return Ok(());
                    }
                    return Err(self.fail(DecodeError::InvalidPadding { offset }));
                }
            }
        }

        if self.padding_state == PaddingState::NeedSecondEquals {
            return Err(self.fail(DecodeError::InvalidPadding { offset }));
        }

        let Some(value) = decode_value(options.alphabet, byte) else {
            return Err(self.fail(DecodeError::InvalidByte { offset }));
        };

        if self.len >= 4 {
            return Err(self.fail(DecodeError::InvalidLength {
                offset: self.offset,
            }));
        }

        self.quartet[self.len] = Some(QuartetPart::Symbol(Symbol { value, offset }));
        self.len += 1;

        if self.len == 4 {
            self.complete_quartet(output)?;
        }

        Ok(())
    }

    fn complete_quartet(&mut self, output: &mut Vec<u8>) -> Result<(), DecodeError> {
        let part = |index: usize| self.quartet[index].expect("complete quartet");
        let first = match part(0) {
            QuartetPart::Symbol(symbol) => symbol,
            QuartetPart::Padding(offset) => {
                return Err(self.fail(DecodeError::InvalidPadding { offset }));
            }
        };
        let second = match part(1) {
            QuartetPart::Symbol(symbol) => symbol,
            QuartetPart::Padding(offset) => {
                return Err(self.fail(DecodeError::InvalidPadding { offset }));
            }
        };

        match (part(2), part(3)) {
            (QuartetPart::Padding(_), QuartetPart::Padding(_)) => {
                if second.value & 0x0f != 0 {
                    return Err(self.fail(DecodeError::NonCanonical {
                        offset: second.offset,
                    }));
                }
                output.push((first.value << 2) | (second.value >> 4));
            }
            (QuartetPart::Symbol(third), QuartetPart::Padding(_)) => {
                if third.value & 0x03 != 0 {
                    return Err(self.fail(DecodeError::NonCanonical {
                        offset: third.offset,
                    }));
                }
                output.push((first.value << 2) | (second.value >> 4));
                output.push((second.value << 4) | (third.value >> 2));
            }
            (QuartetPart::Symbol(third), QuartetPart::Symbol(fourth)) => {
                output.push((first.value << 2) | (second.value >> 4));
                output.push((second.value << 4) | (third.value >> 2));
                output.push((third.value << 6) | fourth.value);
            }
            (QuartetPart::Padding(offset), QuartetPart::Symbol(_)) => {
                return Err(self.fail(DecodeError::InvalidPadding { offset }));
            }
        }

        self.quartet = [None; 4];
        self.len = 0;
        Ok(())
    }

    fn push(&mut self, input: &[u8], options: Options) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        let mut output = Vec::new();
        for &byte in input {
            if let Err(error) = self.process_byte(byte, options, &mut output) {
                return Err(error);
            }
        }
        Ok(output)
    }

    fn finish(&mut self, options: Options) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        if self.padding_state != PaddingState::None || self.len == 3 && self.quartet[2].is_some_and(|p| matches!(p, QuartetPart::Padding(_))) {
            let error = self.fail(DecodeError::InvalidLength {
                offset: self.offset,
            });
            return Err(error);
        }

        let mut output = Vec::new();
        match options.padding {
            Padding::Required => {
                if self.len != 0 {
                    return Err(self.fail(DecodeError::InvalidLength {
                        offset: self.offset,
                    }));
                }
            }
            Padding::Forbidden => match self.len {
                0 => {}
                1 => {
                    return Err(self.fail(DecodeError::InvalidLength {
                        offset: self.offset,
                    }));
                }
                2 => {
                    let first = match self.quartet[0].unwrap() {
                        QuartetPart::Symbol(symbol) => symbol,
                        QuartetPart::Padding(offset) => {
                            return Err(self.fail(DecodeError::InvalidPadding { offset }));
                        }
                    };
                    let second = match self.quartet[1].unwrap() {
                        QuartetPart::Symbol(symbol) => symbol,
                        QuartetPart::Padding(offset) => {
                            return Err(self.fail(DecodeError::InvalidPadding { offset }));
                        }
                    };
                    if second.value & 0x0f != 0 {
                        return Err(self.fail(DecodeError::NonCanonical {
                            offset: second.offset,
                        }));
                    }
                    output.push((first.value << 2) | (second.value >> 4));
                }
                3 => {
                    let first = match self.quartet[0].unwrap() {
                        QuartetPart::Symbol(symbol) => symbol,
                        QuartetPart::Padding(offset) => {
                            return Err(self.fail(DecodeError::InvalidPadding { offset }));
                        }
                    };
                    let second = match self.quartet[1].unwrap() {
                        QuartetPart::Symbol(symbol) => symbol,
                        QuartetPart::Padding(offset) => {
                            return Err(self.fail(DecodeError::InvalidPadding { offset }));
                        }
                    };
                    let third = match self.quartet[2].unwrap() {
                        QuartetPart::Symbol(symbol) => symbol,
                        QuartetPart::Padding(offset) => {
                            return Err(self.fail(DecodeError::InvalidPadding { offset }));
                        }
                    };
                    if third.value & 0x03 != 0 {
                        return Err(self.fail(DecodeError::NonCanonical {
                            offset: third.offset,
                        }));
                    }
                    output.push((first.value << 2) | (second.value >> 4));
                    output.push((second.value << 4) | (third.value >> 2));
                }
                _ => unreachable!(),
            },
        }

        self.finished = true;
        Ok(output)
    }
}

pub struct Encoder {
    options: Options,
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Self {
        Self {
            options,
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                } else {
                    let group = [self.tail[0], self.tail[1], input[index]];
                    encode_full_group(&group, self.options.alphabet, &mut output);
                    self.tail_len = 0;
                    index += 1;
                    break;
                }
                self.tail_len += 1;
                index += 1;
            }
        }

        while index + 3 <= input.len() {
            encode_full_group(&input[index..index + 3], self.options.alphabet, &mut output);
            index += 3;
        }

        while index < input.len() {
            self.tail[self.tail_len] = input[index];
            self.tail_len += 1;
            index += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }
        self.finished = true;

        let mut output = String::new();
        encode_tail(&self.tail[..self.tail_len], self.options, &mut output);
        self.tail_len = 0;
        Ok(output)
    }
}

pub struct Decoder {
    options: Options,
    state: DecodeState,
}

impl Decoder {
    pub fn new(options: Options) -> Self {
        Self {
            options,
            state: DecodeState::new(),
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        self.state.push(input.as_bytes(), self.options)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, DecodeError> {
        self.state.finish(self.options)
    }
}

pub fn decode(input: &str, options: Options) -> Result<Vec<u8>, DecodeError> {
    let mut decoder = Decoder::new(options);
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
