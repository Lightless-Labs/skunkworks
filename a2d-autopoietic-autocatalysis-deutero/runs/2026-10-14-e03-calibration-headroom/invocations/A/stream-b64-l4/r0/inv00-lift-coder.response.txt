#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub ignore_whitespace: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize },
    InvalidPadding { offset: usize },
    InvalidLength { offset: usize },
    TrailingData { offset: usize },
    NonCanonical { offset: usize },
    Finished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StreamError {
    Finished,
}

fn alphabet_symbol(alphabet: Alphabet, value: u8) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn decode_symbol(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn is_ignored_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\n' | 0x0b | 0x0c | b'\r')
}

fn append_encoded_tail(bytes: &[u8], options: Options, output: &mut String) {
    if bytes.is_empty() {
        return;
    }

    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let values = [
        a >> 2,
        ((a & 0x03) << 4) | (b >> 4),
        (b & 0x0f) << 2,
    ];

    output.push(alphabet_symbol(options.alphabet, values[0]) as char);
    output.push(alphabet_symbol(options.alphabet, values[1]) as char);

    if bytes.len() > 1 {
        output.push(alphabet_symbol(options.alphabet, values[2]) as char);
        if options.padding == Padding::Required {
            output.push('=');
        }
    } else if options.padding == Padding::Required {
        output.push_str("==");
    }
}

pub fn encode(input: &[u8], options: Options) -> String {
    let mut output = String::with_capacity((input.len() + 2) / 3 * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        for value in [
            a >> 2,
            ((a & 0x03) << 4) | (b >> 4),
            ((b & 0x0f) << 2) | (c >> 6),
            c & 0x3f,
        ] {
            output.push(alphabet_symbol(options.alphabet, value) as char);
        }
    }

    append_encoded_tail(chunks.remainder(), options, &mut output);
    output
}

pub fn decode(input: &str, options: Options) -> Result<Vec<u8>, DecodeError> {
    let mut decoder = Decoder::new(options);
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

#[derive(Debug, Clone)]
pub struct Encoder {
    options: Options,
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Self {
        Self {
            options,
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut output = String::with_capacity((input.len() + self.tail_len) / 3 * 4);
        let mut position = 0;

        if self.tail_len > 0 {
            while self.tail_len < 2 && position < input.len() {
                self.tail[self.tail_len] = input[position];
                self.tail_len += 1;
                position += 1;
            }

            if self.tail_len == 2 && position < input.len() {
                let triple = [self.tail[0], self.tail[1], input[position]];
                for value in [
                    triple[0] >> 2,
                    ((triple[0] & 0x03) << 4) | (triple[1] >> 4),
                    ((triple[1] & 0x0f) << 2) | (triple[2] >> 6),
                    triple[2] & 0x3f,
                ] {
                    output.push(alphabet_symbol(self.options.alphabet, value) as char);
                }
                position += 1;
                self.tail_len = 0;
            }
        }

        while position + 3 <= input.len() {
            let a = input[position];
            let b = input[position + 1];
            let c = input[position + 2];
            for value in [
                a >> 2,
                ((a & 0x03) << 4) | (b >> 4),
                ((b & 0x0f) << 2) | (c >> 6),
                c & 0x3f,
            ] {
                output.push(alphabet_symbol(self.options.alphabet, value) as char);
            }
            position += 3;
        }

        self.tail_len = input.len() - position;
        for i in 0..self.tail_len {
            self.tail[i] = input[position + i];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.finished {
            return Err(StreamError::Finished);
        }

        let mut output = String::new();
        append_encoded_tail(&self.tail[..self.tail_len], self.options, &mut output);
        self.tail_len = 0;
        self.finished = true;
        Ok(output)
    }
}

#[derive(Debug, Clone)]
pub struct Decoder {
    options: Options,
    symbols: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    second_padding: bool,
    terminal: bool,
    total_offset: usize,
    finished: bool,
    error: Option<DecodeError>,
}

impl Decoder {
    pub fn new(options: Options) -> Self {
        Self {
            options,
            symbols: [0; 4],
            offsets: [0; 4],
            len: 0,
            second_padding: false,
            terminal: false,
            total_offset: 0,
            finished: false,
            error: None,
        }
    }

    fn fail(&mut self, error: DecodeError) -> Result<Vec<u8>, DecodeError> {
        self.error = Some(error);
        Err(error)
    }

    fn decode_quartet(&mut self, output: &mut Vec<u8>) -> Result<(), DecodeError> {
        let a = self.symbols[0];
        let b = self.symbols[1];

        if self.second_padding {
            if b & 0x0f != 0 {
                return Err(DecodeError::NonCanonical {
                    offset: self.offsets[1],
                });
            }
            output.push((a << 2) | (b >> 4));
            self.terminal = true;
        } else if self.symbols[3] == 0xff {
            let c = self.symbols[2];
            if c & 0x03 != 0 {
                return Err(DecodeError::NonCanonical {
                    offset: self.offsets[2],
                });
            }
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            self.terminal = true;
        } else {
            let c = self.symbols[2];
            let d = self.symbols[3];
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
        }

        self.len = 0;
        self.second_padding = false;
        Ok(())
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        let mut output = Vec::new();
        for (index, &byte) in input.as_bytes().iter().enumerate() {
            let offset = self.total_offset + index;

            if self.options.ignore_whitespace && is_ignored_whitespace(byte) {
                continue;
            }

            if self.terminal {
                return self.fail(DecodeError::TrailingData { offset });
            }

            if byte == b'=' {
                if self.options.padding == Padding::Forbidden || self.len < 2 {
                    return self.fail(DecodeError::InvalidPadding { offset });
                }

                if self.second_padding {
                    if self.len == 3 {
                        self.symbols[3] = 0xff;
                        self.offsets[3] = offset;
                        if let Err(error) = self.decode_quartet(&mut output) {
                            return self.fail(error);
                        }
                    } else {
                        return self.fail(DecodeError::InvalidPadding { offset });
                    }
                } else if self.len == 2 {
                    self.second_padding = true;
                    self.symbols[2] = 0;
                    self.offsets[2] = offset;
                    self.len = 3;
                } else if self.len == 3 {
                    self.symbols[3] = 0xff;
                    self.offsets[3] = offset;
                    if let Err(error) = self.decode_quartet(&mut output) {
                        return self.fail(error);
                    }
                } else {
                    return self.fail(DecodeError::InvalidPadding { offset });
                }
                continue;
            }

            if self.second_padding {
                return self.fail(DecodeError::InvalidPadding { offset });
            }

            let Some(value) = decode_symbol(self.options.alphabet, byte) else {
                return self.fail(DecodeError::InvalidByte { offset });
            };

            self.symbols[self.len] = value;
            self.offsets[self.len] = offset;
            self.len += 1;

            if self.len == 4 {
                if let Err(error) = self.decode_quartet(&mut output) {
                    return self.fail(error);
                }
            }
        }

        self.total_offset += input.len();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, DecodeError> {
        if let Some(error) = self.error {
            return Err(error);
        }
        if self.finished {
            return Err(DecodeError::Finished);
        }

        let eof = self.total_offset;

        if self.options.padding == Padding::Required {
            if self.len != 0 {
                return self.fail(DecodeError::InvalidLength { offset: eof });
            }
            self.finished = true;
            return Ok(Vec::new());
        }

        match self.len {
            0 => {
                self.finished = true;
                Ok(Vec::new())
            }
            1 => self.fail(DecodeError::InvalidLength { offset: eof }),
            2 => {
                if self.symbols[1] & 0x0f != 0 {
                    return self.fail(DecodeError::NonCanonical {
                        offset: self.offsets[1],
                    });
                }
                self.finished = true;
                Ok(vec![(self.symbols[0] << 2) | (self.symbols[1] >> 4)])
            }
            3 => {
                if self.symbols[2] & 0x03 != 0 {
                    return self.fail(DecodeError::NonCanonical {
                        offset: self.offsets[2],
                    });
                }
                self.finished = true;
                Ok(vec![
                    (self.symbols[0] << 2) | (self.symbols[1] >> 4),
                    (self.symbols[1] << 4) | (self.symbols[2] >> 2),
                ])
            }
            _ => unreachable!(),
        }
    }
}
