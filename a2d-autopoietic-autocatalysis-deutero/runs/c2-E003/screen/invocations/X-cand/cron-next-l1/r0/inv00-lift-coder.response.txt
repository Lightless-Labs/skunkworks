#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

const DOMAINS: [(u8, u8); 5] = [(0, 59), (0, 23), (1, 31), (1, 12), (0, 6)];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn error(field: Field, offset: usize, kind: ErrorKind) -> ParseError {
    match kind {
        ErrorKind::EmptyItem => ParseError::EmptyItem { field, offset },
        ErrorKind::InvalidSyntax => ParseError::InvalidSyntax { field, offset },
        ErrorKind::LeadingZero => ParseError::LeadingZero { field, offset },
        ErrorKind::NumberOverflow => ParseError::NumberOverflow { field, offset },
        ErrorKind::OutOfRange => ParseError::OutOfRange { field, offset },
        ErrorKind::ReversedRange => ParseError::ReversedRange { field, offset },
        ErrorKind::BadStep => ParseError::BadStep { field, offset },
    }
}

enum ErrorKind {
    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,
    OutOfRange,
    ReversedRange,
    BadStep,
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    let digits = &bytes[start..end];

    if digits.len() > 1 && digits[0] == b'0' {
        return Err(error(field, start, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in digits {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| error(field, start, ErrorKind::NumberOverflow))?;
    }

    Ok(value)
}

fn parse_field(input: &str, field: Field, min: u8, max: u8) -> Result<Vec<u8>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; (max as usize) + 1];
    let mut item_start = 0;

    for item_end in (0..=bytes.len())
        .filter(|&i| i == bytes.len() || bytes[i] == b',')
    {
        if item_start == item_end {
            return Err(error(field, item_start, ErrorKind::EmptyItem));
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        }

        let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
            (slash, Some(slash + 1))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let hyphen_positions: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            BaseKind::Star
        } else if hyphen_positions.is_empty() && !base.is_empty() {
            if base.iter().all(u8::is_ascii_digit) {
                BaseKind::Number
            } else {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
        } else if hyphen_positions.len() == 1 {
            let hyphen = hyphen_positions[0];
            if hyphen == 0
                || hyphen + 1 == base.len()
                || !base[..hyphen].iter().all(u8::is_ascii_digit)
                || !base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
            BaseKind::Range(hyphen)
        } else {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        };

        if let Some(step_start) = step_start {
            let step_end = item.len();
            if step_start == step_end
                || !item[step_start..step_end]
                    .iter()
                    .all(u8::is_ascii_digit)
            {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
        }

        let (range_start, range_end) = match base_kind {
            BaseKind::Star => (min as u32, max as u32),
            BaseKind::Number => {
                let number = parse_number(base, 0, base.len(), field)?;
                validate_domain(number, item_start, field, min, max)?;
                (number, if step_start.is_some() { max as u32 } else { number })
            }
            BaseKind::Range(hyphen) => {
                let start = parse_number(base, 0, hyphen, field)?;
                validate_domain(start, item_start, field, min, max)?;
                let end_offset = item_start + hyphen + 1;
                let end = parse_number(base, hyphen + 1, base.len(), field)?;
                validate_domain(end, end_offset, field, min, max)?;
                if start > end {
                    return Err(error(field, end_offset, ErrorKind::ReversedRange));
                }
                (start, end)
            }
        };

        let step = if let Some(step_start) = step_start {
            let step_offset = item_start + step_start;
            let step = parse_number(item, step_start, item.len(), field)?;
            let width = (max - min + 1) as u32;
            if step == 0 || step > width {
                return Err(error(field, step_offset, ErrorKind::BadStep));
            }
            step
        } else {
            1
        };

        let mut value = range_start;
        while value <= range_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start = item_end + 1;
    }

    Ok((min..=max).filter(|&v| selected[v as usize]).collect())
}

enum BaseKind {
    Star,
    Number,
    Range(usize),
}

fn validate_domain(
    value: u32,
    offset: usize,
    field: Field,
    min: u8,
    max: u8,
) -> Result<(), ParseError> {
    if value < min as u32 || value > max as u32 {
        Err(error(field, offset, ErrorKind::OutOfRange))
    } else {
        Ok(())
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = Vec::with_capacity(5);
    for (i, part) in parts.iter().enumerate() {
        let (min, max) = DOMAINS[i];
        selected.push(parse_field(part, FIELDS[i], min, max)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}
