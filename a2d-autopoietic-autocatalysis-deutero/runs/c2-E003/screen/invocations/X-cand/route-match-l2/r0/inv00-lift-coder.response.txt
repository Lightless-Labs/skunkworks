#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parsed = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(parsed);
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let parsed = parse_path(&path[..head_end])?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;
        for (id, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(route, &parsed) {
                let replace = match &best {
                    None => true,
                    Some((best_id, _)) => {
                        route.rank > self.routes[*best_id].rank
                            || (route.rank == self.routes[*best_id].rank && id < *best_id)
                    }
                };
                if replace {
                    best = Some((id, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

struct ParsedPath {
    segments: Vec<String>,
}

fn parse_pattern(pattern: &str) -> Result<Route, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Route {
            segments: Vec::new(),
            rank: vec![3],
        });
    }

    let parts = split_non_root(pattern, true)?;
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (i, (start, raw)) in parts.iter().enumerate() {
        let bytes = raw.as_bytes();
        if bytes[0] == b':' || bytes[0] == b'*' {
            let wildcard = bytes[0] == b'*';
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard && i + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard {
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            let decoded = decode_segment(raw, *start, true)?;
            segments.push(Segment::Static(decoded));
        }
    }

    let mut rank = Vec::with_capacity(segments.len() + 1);
    for segment in &segments {
        rank.push(match segment {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        });
    }
    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }

    Ok(Route { segments, rank })
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath { segments: Vec::new() });
    }

    let parts = split_non_root(path, false)?;
    let mut segments = Vec::with_capacity(parts.len());
    for (start, raw) in parts {
        segments.push(decode_segment(raw, start, false)?);
    }
    Ok(ParsedPath { segments })
}

fn split_non_root(input: &str, pattern: bool) -> Result<Vec<(usize, &str)>, Error> {
    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(if pattern {
                    Error::InvalidPattern { at: i }
                } else {
                    Error::InvalidPath { at: i }
                });
            }
            if i == start {
                return Err(if pattern {
                    Error::InvalidPattern { at: start }
                } else {
                    Error::InvalidPath { at: start }
                });
            }
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(if pattern {
            Error::InvalidPattern { at: bytes.len() - 1 }
        } else {
            Error::InvalidPath { at: bytes.len() - 1 }
        });
    }
    parts.push((start, &input[start..]));
    Ok(parts)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, start: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return invalid_at(pattern, start + i);
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            let Some((hi, lo)) = hi.zip(lo) else {
                return invalid_at(pattern, start + i);
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return invalid_at(pattern, start + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return invalid_at(pattern, start + i);
            }
            decoded.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded segments contain ASCII only"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn invalid_at<T>(pattern: bool, at: usize) -> Result<T, Error> {
    Err(if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    })
}

fn match_route(route: &Route, path: &ParsedPath) -> Option<Vec<(String, String)>> {
    let wildcard_at = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));
    let fixed_len = wildcard_at.unwrap_or(route.segments.len());

    if path.segments.len() < fixed_len {
        return None;
    }
    if wildcard_at.is_none() && path.segments.len() != fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        let value = &path.segments[i];
        match &route.segments[i] {
            Segment::Static(expected) if expected != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_at {
        if let Segment::Wildcard(name) = &route.segments[index] {
            let value = path.segments[index..].join("/");
            params.push((name.clone(), value));
        }
    }
    Some(params)
}
