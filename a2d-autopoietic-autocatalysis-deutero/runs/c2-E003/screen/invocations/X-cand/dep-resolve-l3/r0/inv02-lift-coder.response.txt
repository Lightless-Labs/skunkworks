#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| matches(version, *c))
}

struct Search<'a> {
    catalog: &'a [Package],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
    limit_hit: bool,
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn active_constraints(
        &self,
        roots: &[Requirement],
        assigned: &[(String, Version)],
    ) -> std::collections::BTreeMap<String, Vec<Constraint>> {
        let mut active = std::collections::BTreeMap::new();
        for req in roots {
            active
                .entry(req.package.clone())
                .or_insert_with(Vec::new)
                .push(req.constraint);
        }
        for (name, version) in assigned {
            if let Some(package) = self.package(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for req in &release.deps {
                        active
                            .entry(req.package.clone())
                            .or_insert_with(Vec::new)
                            .push(req.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }
        active
    }

    fn inspect(&mut self, roots: &[Requirement], assigned: &mut Vec<(String, Version)>) {
        if self.limit_hit {
            return;
        }
        let active = self.active_constraints(roots, assigned);

        if let Some(name) = active.keys().find(|name| self.package(name).is_none()) {
            if self.first_error.is_none() {
                self.first_error = Some(ResolveError::Missing {
                    package: name.clone(),
                });
            }
            return;
        }

        let assigned_map: std::collections::BTreeMap<_, _> =
            assigned.iter().cloned().collect();

        for (name, constraints) in &active {
            if let Some(version) = assigned_map.get(name) {
                if !satisfies(*version, constraints) {
                    if self.first_error.is_none() {
                        self.first_error = Some(ResolveError::Conflict {
                            package: name.clone(),
                            selected: *version,
                            requirements: constraints.clone(),
                        });
                    }
                    return;
                }
            }
        }

        for (name, constraints) in &active {
            if assigned_map.contains_key(name) {
                continue;
            }
            let package = self.package(name).unwrap();
            if !package
                .releases
                .iter()
                .any(|release| satisfies(release.version, constraints))
            {
                if self.first_error.is_none() {
                    self.first_error = Some(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
                return;
            }
        }

        let Some((name, constraints)) = active
            .iter()
            .find(|(name, _)| !assigned_map.contains_key(*name))
        else {
            let mut selected = assigned
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect::<Vec<_>>();
            selected.sort_by(|a, b| a.package.cmp(&b.package));
            let cost = self
                .locks
                .iter()
                .filter(|lock| {
                    !selected
                        .iter()
                        .any(|s| s.package == lock.package && s.version == lock.version)
                })
                .count();
            let vector = selected.clone();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_selected)) if cost < *best_cost => true,
                Some((best_cost, best_selected)) if cost == *best_cost => {
                    self.compare_vectors(&vector, best_selected) == std::cmp::Ordering::Greater
                }
                _ => false,
            };
            if replace {
                self.best = Some((cost, selected));
            }
            return;
        };

        let package = self.package(name).unwrap();
        let mut candidates = package
            .releases
            .iter()
            .filter(|r| satisfies(r.version, constraints))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == *name) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                self.limit_hit = true;
                return;
            }
            self.attempts += 1;
            assigned.push((name.clone(), candidate));
            self.inspect(roots, assigned);
            assigned.pop();
            if self.limit_hit {
                return;
            }
        }
    }

    fn compare_vectors(&self, a: &[Selection], b: &[Selection]) -> std::cmp::Ordering {
        let mut names = self
            .catalog
            .iter()
            .map(|p| p.name.as_str())
            .collect::<Vec<_>>();
        names.sort_unstable();
        for name in names {
            let av = a.iter().find(|s| s.package == name).map(|s| s.version);
            let bv = b.iter().find(|s| s.package == name).map(|s| s.version);
            let cmp = match (av, bv) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(x), Some(y)) => x.cmp(&y),
            };
            if cmp != std::cmp::Ordering::Equal {
                return cmp;
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    let mut seen = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit_hit: false,
    };
    search.inspect(roots, &mut Vec::new());

    if search.limit_hit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }
    let Some((_, selected)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::Missing {
            package: String::new(),
        }));
    };

    let selected_map = selected
        .iter()
        .map(|s| (s.package.as_str(), s.version))
        .collect::<std::collections::BTreeMap<_, _>>();
    let mut final_constraints = search.active_constraints(roots, &selected.iter().map(|s| (s.package.clone(), s.version)).collect::<Vec<_>>());
    let mut all_names = std::collections::BTreeSet::new();
    for lock in locks {
        all_names.insert(lock.package.clone());
    }
    for selection in &selected {
        all_names.insert(selection.package.clone());
    }

    let mut changes = Vec::new();
    for name in all_names {
        let before = locks.iter().find(|l| l.package == name).map(|l| l.version);
        let after = selected_map.get(name.as_str()).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .package(&name)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if !satisfies(
                    old,
                    final_constraints
                        .get(&name)
                        .map(|v| v.as_slice())
                        .unwrap_or(&[]),
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));

    Ok(Resolution { selected, changes })
}
