#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn valid_name(s: &str) -> bool {
    let bytes = s.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn permits(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn satisfies(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| permits(*c, v))
}

struct Search<'a> {
    packages: &'a [Package],
    by_name: std::collections::BTreeMap<&'a str, &'a Package>,
    locks: std::collections::BTreeMap<&'a str, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
    limit: bool,
}

impl<'a> Search<'a> {
    fn active(&self, selected: &std::collections::BTreeMap<String, Version>, roots: &[Requirement]) -> std::collections::BTreeMap<String, Vec<Constraint>> {
        let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
        for req in roots {
            active.entry(req.package.clone()).or_default().push(req.constraint);
        }
        for (name, version) in selected {
            if let Some(pkg) = self.by_name.get(name.as_str()) {
                if let Some(rel) = pkg.releases.iter().find(|r| r.version == *version) {
                    for dep in &rel.deps {
                        active.entry(dep.package.clone()).or_default().push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical_constraints(constraints);
        }
        active
    }

    fn inspect(
        &mut self,
        roots: &[Requirement],
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) {
        if self.limit {
            return;
        }

        let active = self.active(selected, roots);

        if let Some(name) = active
            .keys()
            .find(|name| !self.by_name.contains_key(name.as_str()))
        {
            if self.first_error.is_none() {
                self.first_error = Some(ResolveError::Missing {
                    package: name.clone(),
                });
            }
            return;
        }

        for (name, version) in selected.iter() {
            if let Some(cs) = active.get(name) {
                if !satisfies(*version, cs) {
                    if self.first_error.is_none() {
                        self.first_error = Some(ResolveError::Conflict {
                            package: name.clone(),
                            selected: *version,
                            requirements: cs.clone(),
                        });
                    }
                    return;
                }
            }
        }

        for (name, cs) in &active {
            if !selected.contains_key(name) {
                let pkg = self.by_name[name.as_str()];
                if !pkg.releases.iter().any(|r| satisfies(r.version, cs)) {
                    if self.first_error.is_none() {
                        self.first_error = Some(ResolveError::NoMatch {
                            package: name.clone(),
                            requirements: cs.clone(),
                        });
                    }
                    return;
                }
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            let solution = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect::<Vec<_>>();
            let cost = self
                .locks
                .iter()
                .filter(|(package, version)| {
                    selected.get(**package).copied() != Some(**version)
                })
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_solution)) => {
                    if cost < *best_cost {
                        true
                    } else if cost > *best_cost {
                        false
                    } else {
                        self.compare_vectors(&solution, best_solution)
                            == std::cmp::Ordering::Greater
                    }
                }
            };
            if replace {
                self.best = Some((cost, solution));
            }
            return;
        };

        let cs = &active[&name];
        let pkg = self.by_name[name.as_str()];
        let mut candidates = pkg
            .releases
            .iter()
            .filter(|r| satisfies(r.version, cs))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(name.as_str()) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            self.inspect(roots, selected);
            selected.remove(&name);
            if self.limit {
                return;
            }
        }
    }

    fn compare_vectors(&self, a: &[Selection], b: &[Selection]) -> std::cmp::Ordering {
        let amap = a
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect::<std::collections::BTreeMap<_, _>>();
        let bmap = b
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect::<std::collections::BTreeMap<_, _>>();
        let names = self
            .by_name
            .keys()
            .copied()
            .collect::<std::collections::BTreeSet<_>>();
        for name in names {
            match (amap.get(name), bmap.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (Some(av), Some(bv)) => {
                    let ord = av.cmp(bv);
                    if ord != std::cmp::Ordering::Equal {
                        return ord;
                    }
                }
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let by_name = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect::<std::collections::BTreeMap<_, _>>();
    let lock_map = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect::<std::collections::BTreeMap<_, _>>();

    let mut search = Search {
        packages: catalog,
        by_name,
        locks: lock_map,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit: false,
    };
    let mut selected = std::collections::BTreeMap::new();
    search.inspect(roots, &mut selected);

    if search.limit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }
    let Some((_, selected)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::Missing {
            package: String::new(),
        }));
    };

    let final_active = search.active(
        &selected
            .iter()
            .map(|s| (s.package.clone(), s.version))
            .collect(),
        roots,
    );
    let selected_map = selected
        .iter()
        .map(|s| (s.package.clone(), s.version))
        .collect::<std::collections::BTreeMap<_, _>>();
    let mut package_names = std::collections::BTreeSet::new();
    package_names.extend(locks.iter().map(|l| l.package.as_str()));
    package_names.extend(selected.iter().map(|s| s.package.as_str()));

    let mut changes = Vec::new();
    for name in package_names {
        let before = lock_map.get(name).copied();
        let after = selected_map.get(name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = by_name
                    .get(name)
                    .is_some_and(|pkg| pkg.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_active
                    .get(name)
                    .is_some_and(|cs| satisfies(old, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name.to_string(),
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
