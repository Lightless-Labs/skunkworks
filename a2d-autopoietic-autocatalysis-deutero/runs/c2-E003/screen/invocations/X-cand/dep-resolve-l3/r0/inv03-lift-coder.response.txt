#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn encoded(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encoded(site), encoded(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encoded(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encoded(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encoded(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encoded(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encoded(package),
            canonical(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",")
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encoded(package),
            version_text(*selected),
            canonical(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn satisfies(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches(v, *c))
}

struct Search<'a> {
    catalog: &'a [Package],
    package_index: std::collections::BTreeMap<String, usize>,
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn active_constraints(
        &self,
        roots: &[Requirement],
        chosen: &std::collections::BTreeMap<String, usize>,
    ) -> std::collections::BTreeMap<String, Vec<Constraint>> {
        let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
        for r in roots {
            active
                .entry(r.package.clone())
                .or_default()
                .push(r.constraint);
        }
        for (name, &release_index) in chosen {
            let package = &self.catalog[self.package_index[name]];
            for dep in &package.releases[release_index].deps {
                active
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
            }
        }
        for cs in active.values_mut() {
            *cs = canonical(std::mem::take(cs));
        }
        active
    }

    fn node_error(
        &self,
        active: &std::collections::BTreeMap<String, Vec<Constraint>>,
        chosen: &std::collections::BTreeMap<String, usize>,
    ) -> Option<ResolveError> {
        for name in active.keys() {
            if !self.package_index.contains_key(name) {
                return Some(ResolveError::Missing {
                    package: name.clone(),
                });
            }
        }

        for (name, cs) in active {
            if let Some(&release_index) = chosen.get(name) {
                let version = self.catalog[self.package_index[name]].releases[release_index].version;
                if !satisfies(version, cs) {
                    return Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for (name, cs) in active {
            if chosen.contains_key(name) {
                continue;
            }
            let package = &self.catalog[self.package_index[name]];
            if !package
                .releases
                .iter()
                .any(|r| satisfies(r.version, cs))
            {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.clone(),
                });
            }
        }
        None
    }

    fn visit(
        &mut self,
        roots: &[Requirement],
        chosen: &mut std::collections::BTreeMap<String, usize>,
    ) -> Result<(), ResolveError> {
        let active = self.active_constraints(roots, chosen);
        if let Some(error) = self.node_error(&active, chosen) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active.keys().find(|name| !chosen.contains_key(*name)).cloned();
        let Some(name) = next else {
            let selected: Vec<Selection> = chosen
                .iter()
                .map(|(name, &idx)| Selection {
                    package: name.clone(),
                    version: self.catalog[self.package_index[name]].releases[idx].version,
                })
                .collect();
            let cost = self
                .locks
                .iter()
                .filter(|lock| {
                    !selected.iter().any(|selection| {
                        selection.package == lock.package && selection.version == lock.version
                    })
                })
                .count();

            let better = match &self.best {
                None => true,
                Some((best_cost, best_selected)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        self.compare_vectors(&selected, best_selected).is_gt()
                    }
                }
            };
            if better {
                self.best = Some((cost, selected));
            }
            return Ok(());
        };

        let package = &self.catalog[self.package_index[&name]];
        let constraints = &active[&name];
        let mut candidates: Vec<usize> = package
            .releases
            .iter()
            .enumerate()
            .filter(|(_, r)| satisfies(r.version, constraints))
            .map(|(i, _)| i)
            .collect();
        candidates.sort_by_key(|&i| std::cmp::Reverse(package.releases[i].version));

        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(pos) = candidates
                .iter()
                .position(|&i| package.releases[i].version == lock.version)
            {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        for idx in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            chosen.insert(name.clone(), idx);
            self.visit(roots, chosen)?;
            chosen.remove(&name);
        }
        Ok(())
    }

    fn compare_vectors(&self, a: &[Selection], b: &[Selection]) -> std::cmp::Ordering {
        let mut names: Vec<&String> = self.package_index.keys().collect();
        names.sort();
        for name in names {
            let av = a.iter().find(|s| &s.package == name).map(|s| s.version);
            let bv = b.iter().find(|s| &s.package == name).map(|s| s.version);
            let ord = match (av, bv) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(x), Some(y)) => x.cmp(&y),
            };
            if !ord.is_eq() {
                return ord;
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_index = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect();
    let mut search = Search {
        catalog,
        package_index,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(roots, &mut std::collections::BTreeMap::new())?;
    let Some((_, selected)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let final_active = search.active_constraints(
        roots,
        &selected
            .iter()
            .map(|s| {
                let idx = search.catalog[search.package_index[&s.package]]
                    .releases
                    .iter()
                    .position(|r| r.version == s.version)
                    .unwrap();
                (s.package.clone(), idx)
            })
            .collect(),
    );

    let before: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let after: std::collections::BTreeMap<String, Version> = selected
        .iter()
        .map(|s| (s.package.clone(), s.version))
        .collect();
    let mut all_names = std::collections::BTreeSet::new();
    all_names.extend(before.keys().cloned());
    all_names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for name in all_names {
        let b = before.get(&name).copied();
        let a = after.get(&name).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .package_index
                    .get(&name)
                    .map(|&i| {
                        search.catalog[i]
                            .releases
                            .iter()
                            .any(|r| r.version == old)
                    })
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_active
                    .get(&name)
                    .map(|cs| satisfies(old, cs))
                    .unwrap_or(true)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
