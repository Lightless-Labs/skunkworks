#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

#[derive(Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing_slash: bool,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            let hi = hex_upper(bytes[i + 1]);
            let lo = hex_upper(bytes[i + 2]);
            let decoded = match (hi, lo) {
                (Some(h), Some(l)) => (h << 4) | l,
                _ => {
                    return Err(if pattern {
                        Error::InvalidPattern { at: offset + i }
                    } else {
                        Error::InvalidPath { at: offset + i }
                    });
                }
            };
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: offset + i }
            } else {
                Error::InvalidPath { at: offset + i }
            });
        }
    }
    Ok(String::from_utf8(out).expect("decoded segment is ASCII"))
}

fn split_head(input: &str, pattern: bool) -> Result<ParsedPath, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let trailing_slash = input.ends_with('/');
    let effective_end = if trailing_slash {
        input.len() - 1
    } else {
        input.len()
    };
    let bytes = input.as_bytes();
    let mut starts = Vec::new();
    let mut start = 1;
    for i in 1..effective_end {
        if bytes[i] == b'/' {
            if i == start {
                return Err(invalid(start));
            }
            starts.push((start, i));
            start = i + 1;
        }
    }
    if start == effective_end {
        return Err(invalid(start));
    }
    starts.push((start, effective_end));

    let mut segments = Vec::with_capacity(starts.len());
    for (start, end) in starts {
        let raw = &input[start..end];
        segments.push(decode_segment(raw, start, pattern)?);
    }
    Ok(ParsedPath {
        segments,
        trailing_slash,
    })
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, false));
    }

    let trailing_slash = pattern.ends_with('/');
    let effective_end = if trailing_slash {
        pattern.len() - 1
    } else {
        pattern.len()
    };
    let bytes = pattern.as_bytes();
    let mut ranges = Vec::new();
    let mut start = 1;
    for i in 1..effective_end {
        if bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }
    if start == effective_end {
        return Err(Error::InvalidPattern { at: start });
    }
    ranges.push((start, effective_end));

    let mut segments = Vec::with_capacity(ranges.len());
    let mut names = std::collections::HashSet::new();
    let mut wildcard = false;

    for (idx, (start, end)) in ranges.iter().copied().enumerate() {
        let raw = &pattern[start..end];
        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            let valid = {
                let nb = name.as_bytes();
                !nb.is_empty()
                    && nb[0].is_ascii_lowercase()
                    && nb[1..]
                        .iter()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
            };
            if !valid || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if marker == b'*' {
                if idx + 1 != ranges.len() {
                    return Err(Error::InvalidPattern { at: start });
                }
                wildcard = true;
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }
    }

    Ok((segments, trailing_slash, wildcard))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing_slash, wildcard) = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing_slash,
            wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = split_head(head, false)?;

        let mut candidates = Vec::new();
        for (id, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != parsed.trailing_slash
            {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((id, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut allowed = std::collections::BTreeSet::new();
        for (id, _) in &candidates {
            allowed.insert(self.routes[*id].method.clone());
        }
        if allowed.contains("GET") {
            allowed.insert("HEAD".to_owned());
        }

        let eligible_method = if method == "HEAD"
            && !candidates
                .iter()
                .any(|(id, _)| self.routes[*id].method == "HEAD")
        {
            "GET"
        } else {
            method
        };

        let mut best: Option<(usize, Vec<(String, String)>, Vec<u8>)> = None;
        for (id, params) in candidates {
            let route = &self.routes[id];
            if route.method != eligible_method {
                continue;
            }
            let rank = rank(route);
            if best.as_ref().is_none_or(|(_, _, best_rank)| rank > *best_rank) {
                best = Some((id, params, rank));
            }
        }

        let (id, params, _) = match best {
            Some(best) => best,
            None => {
                return Err(Error::MethodNotAllowed {
                    allowed: allowed.into_iter().collect(),
                });
            }
        };

        let redirect = if self.policy == SlashPolicy::Redirect
            && !parsed.segments.is_empty()
            && self.routes[id].trailing_slash != parsed.trailing_slash
        {
            let mut target = head.to_owned();
            if self.routes[id].trailing_slash {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: id,
            params,
            redirect,
        })
    }
}

fn rank(route: &Route) -> Vec<u8> {
    let mut rank = Vec::with_capacity(route.segments.len() + 1);
    for segment in &route.segments {
        rank.push(match segment {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        });
    }
    if !route.wildcard {
        rank.push(3);
    }
    rank
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = wildcard_index.unwrap_or(route.segments.len());

    if wildcard_index.is_none() && request.len() != fixed_len {
        return None;
    }
    if wildcard_index.is_some() && request.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(value) if value != &request[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), request[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_index {
        let Segment::Wildcard(name) = &route.segments[index] else {
            unreachable!()
        };
        params.push((name.clone(), request[index..].join("/")));
    }
    Some(params)
}
