#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateLock { package: String },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    Missing { package: String, evidence: Vec<Evidence> },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    MissingFeature {
        package: String,
        selected: Version,
        feature: String,
        path: Vec<String>,
    },
    Limit { attempts: usize },

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(items: &[Constraint]) -> Vec<Constraint> {
    let mut out = items.to_vec();
    out.sort();
    out.dedup();
    out
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

fn sorted_evidence(items: &[Evidence]) -> Vec<Evidence> {
    let mut out = items.to_vec();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn canonical_evidence_lines(items: &[Evidence]) -> String {
    let mut out = String::new();
    for ev in sorted_evidence(items) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(ev.constraint));
        out.push_str(" via ");
        out.push_str(&path_text(&ev.path));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            canonical_evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "no-match|package={}|constraints={}{}",
                encode(package),
                cs,
                canonical_evidence_lines(evidence)
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={}{}",
                encode(package),
                version_text(*selected),
                cs,
                canonical_evidence_lines(evidence)
            )
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={} \n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        )
        .replace("feature={} \n", "feature={}\n"),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn matches_all(v: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| matches_constraint(v, *c))
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn validate_input(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut packages = BTreeSet::new();
    for p in catalog {
        if !packages.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut seen = BTreeSet::new();
            for f in &r.features {
                if !seen.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    Ok(())
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    deps: BTreeMap<String, Vec<(usize, Dependency)>>,
}

fn root_token(root: &Requirement, index: usize) -> String {
    let features = root.features.iter().cloned().collect::<BTreeSet<_>>()
        .into_iter()
        .collect::<Vec<_>>()
        .join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        features
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let features = dep.features.iter().cloned().collect::<BTreeSet<_>>()
        .into_iter()
        .collect::<Vec<_>>()
        .join(",");
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features,
        gate,
        platform
    )
}

fn build_closure(
    catalog_map: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut constraints: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    let mut origins: BTreeMap<String, Vec<Origin>> = BTreeMap::new();
    let mut requests: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();

    for (i, root) in roots.iter().enumerate() {
        constraints.entry(root.package.clone()).or_default().push(root.constraint);
        origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: vec![root_token(root, i)],
        });
        requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
    }

    let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
    for (i, root) in roots.iter().enumerate() {
        let candidate = vec![root_token(root, i)];
        let slot = paths.entry(root.package.clone()).or_insert_with(|| candidate.clone());
        if (candidate.len(), &candidate) < (slot.len(), slot) {
            *slot = candidate;
        }
    }

    let mut activated: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    let mut processed_features: BTreeSet<(String, String)> = BTreeSet::new();
    let mut processed_deps: BTreeSet<(String, usize)> = BTreeSet::new();
    let mut deps_out: BTreeMap<String, Vec<(usize, Dependency)>> = BTreeMap::new();

    loop {
        let mut changed = false;
        let package_names = selected.keys().cloned().collect::<Vec<_>>();
        for name in package_names {
            let Some(version) = selected.get(&name).copied() else {
                continue;
            };
            let Some(package) = catalog_map.get(&name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let current = requests.get(&name).cloned().unwrap_or_default();
            for feature_name in current {
                if !processed_features.insert((name.clone(), feature_name.clone())) {
                    continue;
                }
                activated.entry(name.clone()).or_default().insert(feature_name.clone());
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if requests.entry(name.clone()).or_default().insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
                changed = true;
            }

            for (dep_index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target) {
                    continue;
                }
                if dep.gate.as_ref().is_some_and(|g| {
                    !activated
                        .get(&name)
                        .is_some_and(|set| set.contains(g))
                }) {
                    continue;
                }
                if !processed_deps.insert((name.clone(), dep_index)) {
                    continue;
                }

                deps_out
                    .entry(name.clone())
                    .or_default()
                    .push((dep_index, dep.clone()));
                constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);

                let source_path = paths.get(&name).cloned().unwrap_or_default();
                let mut dep_path = source_path;
                dep_path.push(dependency_token(&name, version, dep_index, dep));
                origins.entry(dep.package.clone()).or_default().push(Origin {
                    constraint: dep.constraint,
                    path: dep_path.clone(),
                });
                let slot = paths
                    .entry(dep.package.clone())
                    .or_insert_with(|| dep_path.clone());
                if (dep_path.len(), &dep_path) < (slot.len(), slot) {
                    *slot = dep_path;
                    changed = true;
                }
                let dest = requests.entry(dep.package.clone()).or_default();
                for feature in &dep.features {
                    if dest.insert(feature.clone()) {
                        changed = true;
                    }
                }
                changed = true;
            }
        }
        if !changed {
            break;
        }
    }

    for values in constraints.values_mut() {
        values.sort();
        values.dedup();
    }

    Closure {
        constraints,
        origins,
        requests,
        activated,
        paths,
        deps: deps_out,
    }
}

fn release_matches(release: &Release, constraints: &[Constraint]) -> bool {
    matches_all(release.version, constraints)
}

fn evidence_for(closure: &Closure, package: &str) -> Vec<Evidence> {
    closure
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect()
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = build_closure(self.catalog, self.roots, selected, self.target);

        let missing = closure
            .constraints
            .keys()
            .filter(|name| !self.catalog.contains_key(*name))
            .next()
            .cloned();
        if let Some(package) = missing {
            self.record_error(ResolveError::Missing {
                evidence: evidence_for(&closure, &package),
                package,
            });
            return Ok(());
        }

        for (package, version) in selected.iter() {
            let constraints = closure.constraints.get(package).cloned().unwrap_or_default();
            if !matches_all(*version, &constraints) {
                self.record_error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: evidence_for(&closure, package),
                });
                return Ok(());
            }
        }

        for (package, version) in selected.iter() {
            let activated = closure.activated.get(package).cloned().unwrap_or_default();
            let Some(catalog_package) = self.catalog.get(package) else {
                continue;
            };
            let Some(release) = catalog_package.releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            for feature in activated {
                if !release.features.iter().any(|f| f.name == feature) {
                    self.record_error(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: *version,
                        feature,
                        path: closure.paths.get(package).cloned().unwrap_or_default(),
                    });
                    return Ok(());
                }
            }
        }

        for package in closure.constraints.keys() {
            if selected.contains_key(package) {
                continue;
            }
            let constraints = closure.constraints.get(package).cloned().unwrap_or_default();
            let has_candidate = self.catalog[*package]
                .releases
                .iter()
                .any(|r| release_matches(r, &constraints));
            if !has_candidate {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints,
                    evidence: evidence_for(&closure, package),
                });
                return Ok(());
            }
        }

        let next = closure
            .constraints
            .keys()
            .find(|package| !selected.contains_key(*package))
            .cloned();
        let Some(package_name) = next else {
            self.record_solution(selected);
            return Ok(());
        };

        let constraints = closure
            .constraints
            .get(&package_name)
            .cloned()
            .unwrap_or_default();
        let mut candidates = self.catalog[&package_name]
            .releases
            .iter()
            .filter(|r| release_matches(r, &constraints))
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.version.cmp(&a.version));

        if let Some(locked) = self.locks.get(&package_name) {
            if let Some(index) = candidates.iter().position(|r| r.version == *locked) {
                let candidate = candidates.remove(index);
                candidates.insert(0, candidate);
            }
        }

        let mut first_local_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), candidate.version);
            self.visit(selected)?;
            selected.remove(&package_name);

            if self.best.is_none() && self.first_error.is_none() {
                if let Some(err) = first_local_error.take() {
                    self.first_error = Some(err);
                }
            }
        }
        Ok(())
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn record_solution(&mut self, selected: &BTreeMap<String, Version>) {
        let cost = self
            .locks
            .iter()
            .filter(|(package, version)| selected.get(*package) != Some(*version))
            .count();
        let better = match &self.best {
            None => true,
            Some(best) => {
                if cost != self.best_cost {
                    cost < self.best_cost
                } else {
                    self.catalog_names.iter().find_map(|name| {
                        let a = selected.get(name);
                        let b = best.get(name);
                        if a == b {
                            None
                        } else {
                            Some(match (a, b) {
                                (None, Some(_)) => false,
                                (Some(_), None) => true,
                                (Some(x), Some(y)) => x > y,
                                (None, None) => false,
                            })
                        }
                    }).unwrap_or(false)
                }
            }
        };
        if better {
            self.best = Some(selected.clone());
            self.best_cost = cost;
        }
    }
}

fn final_constraints(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> BTreeMap<String, Vec<Constraint>> {
    build_closure(catalog, roots, selected, target).constraints
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_input(catalog, roots, locks)?;

    let catalog_map = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();
    let lock_map = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();

    let mut names = catalog_map.keys().cloned().collect::<Vec<_>>();
    names.sort();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
        catalog_names: names,
    };

    let mut selected = BTreeMap::new();
    search.visit(&mut selected)?;

    let Some(solution) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_constraints = final_constraints(&catalog_map, roots, &solution, target);
    let selected_vec = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let all_packages = lock_map
        .keys()
        .chain(solution.keys())
        .cloned()
        .collect::<BTreeSet<_>>();
    let mut changes = Vec::new();
    for package in all_packages {
        let before = lock_map.get(&package).copied();
        let after = solution.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !matches_all(
                    after.unwrap(),
                    final_constraints.get(&package).map(Vec::as_slice).unwrap_or(&[]),
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
