#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct RuntimeJob {
    burst_index: usize,
    remaining_cpu: u32,
    level: usize,
    completion: Option<u64>,
    cpu_total: u64,
    io_total: u64,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut runtime: Vec<RuntimeJob> = jobs
        .iter()
        .map(|job| RuntimeJob {
            burst_index: 0,
            remaining_cpu: job.bursts[0],
            level: job.level,
            completion: None,
            cpu_total: job
                .bursts
                .iter()
                .step_by(2)
                .map(|&duration| u64::from(duration))
                .sum(),
            io_total: job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&duration| u64::from(duration))
                .sum(),
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut time = 0_u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completed_count = 0_usize;

    while completed_count < jobs.len() {
        events.sort_by_key(|event| (event.time, event.kind, event.index));

        let mut event_position = 0;
        while event_position < events.len() && events[event_position].time <= time {
            let event = events[event_position];
            queues[event.level].push_back(event.index);
            event_position += 1;
        }
        events.drain(0..event_position);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if events.is_empty() {
                break;
            }
            let next_time = events[0].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().expect("selected queue is nonempty");
        let job_id = jobs[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != job_id) {
            time += u64::from(switch_cost);
        }

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        let mut event_position = 0;
        while event_position < events.len() && events[event_position].time <= time {
            let event = events[event_position];
            queues[event.level].push_back(event.index);
            event_position += 1;
        }
        events.drain(0..event_position);

        let quantum = quanta[level];
        let consumed = runtime[index].remaining_cpu.min(quantum);
        let start = time;
        time += u64::from(consumed);
        let end = time;
        runtime[index].remaining_cpu -= consumed;

        let burst_finished = runtime[index].remaining_cpu == 0;
        let final_burst = burst_finished && runtime[index].burst_index + 1 == jobs[index].bursts.len();
        runs.push(Run {
            id: job_id,
            start,
            end,
            finished: final_burst,
            level,
        });

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        let mut event_position = 0;
        while event_position < events.len() && events[event_position].time <= end {
            let event = events[event_position];
            queues[event.level].push_back(event.index);
            event_position += 1;
        }
        events.drain(0..event_position);

        if !burst_finished {
            runtime[index].level = (level + 1).min(quanta.len() - 1);
            queues[runtime[index].level].push_back(index);
        } else if final_burst {
            runtime[index].completion = Some(end);
            completed_count += 1;
        } else {
            let io_position = runtime[index].burst_index + 1;
            let wake_time = end + u64::from(jobs[index].bursts[io_position]);
            let wake_level = level.saturating_sub(1);
            runtime[index].burst_index += 2;
            runtime[index].remaining_cpu = jobs[index].bursts[runtime[index].burst_index];
            events.push(Event {
                time: wake_time,
                kind: 1,
                index,
                level: wake_level,
            });
        }

        last_dispatch_id = Some(job_id);
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let completion = runtime[index].completion.unwrap_or(time);
            let turnaround = completion - u64::from(job.arrival);
            let waiting = turnaround - runtime[index].cpu_total - runtime[index].io_total;
            Stat {
                id: job.id,
                completion,
                turnaround,
                waiting,
            }
        })
        .collect();

    let count = jobs.len() as u64;
    let rounded_mean = |total: u64| -> u64 {
        if count == 0 {
            0
        } else {
            (total * 100 + count / 2) / count
        }
    };

    let total_waiting: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean(total_waiting),
        mean_turnaround_x100: rounded_mean(total_turnaround),
        end: time,
    })
}
