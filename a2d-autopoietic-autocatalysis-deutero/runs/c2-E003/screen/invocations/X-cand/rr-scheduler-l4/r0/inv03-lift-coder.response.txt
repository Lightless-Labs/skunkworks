#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    index: usize,
    kind: EventKind,
}

struct State {
    burst_index: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

fn event_key(event: &Event) -> (u64, u8, usize) {
    (
        event.time,
        match event.kind {
            EventKind::Arrival => 0,
            EventKind::Wake { .. } => 1,
        },
        event.index,
    )
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        return 0;
    }
    let numerator = total * 100;
    let divisor = count as u64;
    (numerator + divisor / 2) / divisor
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            index,
            kind: EventKind::Arrival,
        });
    }
    events.sort_by_key(event_key);

    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            burst_index: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut next_event = 0usize;
    let mut now = 0u64;
    let mut last_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    loop {
        while next_event < events.len() && events[next_event].time <= now {
            let event = events[next_event];
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake { level } => queues[level].push_back(event.index),
            }
            next_event += 1;
        }

        let selected = queues.iter_mut().find_map(|queue| queue.pop_front());
        let Some(index) = selected else {
            if next_event >= events.len() {
                break;
            }
            let next_time = events[next_event].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_id = None;
            }
            continue;
        };

        let level = states[index].level;
        let id = jobs[index].id;
        if last_id.is_some_and(|previous| previous != id) {
            now += switch_cost as u64;
        }

        while next_event < events.len() && events[next_event].time <= now {
            let event = events[next_event];
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake { level } => queues[level].push_back(event.index),
            }
            next_event += 1;
        }

        let consumed = states[index].remaining.min(quanta[level]);
        let start = now;
        now += consumed as u64;

        states[index].remaining -= consumed;
        let burst_finished = states[index].remaining == 0;
        let job_finished = burst_finished && states[index].burst_index + 1 == jobs[index].bursts.len();

        runs.push(Run {
            id,
            start,
            end: now,
            finished: job_finished,
            level,
        });

        while next_event < events.len() && events[next_event].time <= now {
            let event = events[next_event];
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake { level } => queues[level].push_back(event.index),
            }
            next_event += 1;
        }

        if job_finished {
            states[index].completion = Some(now);
        } else if !burst_finished {
            states[index].level = (level + 1).min(quanta.len() - 1);
            queues[states[index].level].push_back(index);
        } else {
            let io_delay = jobs[index].bursts[states[index].burst_index + 1] as u64;
            states[index].burst_index += 2;
            states[index].remaining = jobs[index].bursts[states[index].burst_index];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: now + io_delay,
                index,
                kind: EventKind::Wake { level: wake_level },
            });
            events[next_event..].sort_by_key(event_key);
        }

        last_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;
    let mut end = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap_or(0);
        let turnaround = completion - job.arrival as u64;
        let cpu_and_io: u64 = job.bursts.iter().map(|&burst| burst as u64).sum();
        let waiting = turnaround - cpu_and_io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        end = end.max(completion);
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end,
    })
}
