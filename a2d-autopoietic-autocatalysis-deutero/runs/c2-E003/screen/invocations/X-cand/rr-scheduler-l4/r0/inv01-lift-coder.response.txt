```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    index: usize,
    kind: EventKind,
}

struct State {
    burst_position: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        return 0;
    }
    let denominator = count as u64;
    (total * 100 + denominator / 2) / denominator
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut queues: Vec<Vec<usize>> = vec![Vec::new(); quanta.len()];
    let mut queue_heads = vec![0usize; quanta.len()];
    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            index,
            kind: EventKind::Arrival,
        })
        .collect();
    events.sort_by_key(|event| {
        (
            event.time,
            match event.kind {
                EventKind::Arrival => 0u8,
                EventKind::Wake { .. } => 1u8,
            },
            event.index,
        )
    });
    let mut next_event = 0usize;

    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let admit_through = |limit: u64,
                         events: &[Event],
                         next_event: &mut usize,
                         queues: &mut Vec<Vec<usize>>,
                         queue_heads: &[usize]| {
        while *next_event < events.len() && events[*next_event].time <= limit {
            let event = events[*next_event];
            let level = match event.kind {
                EventKind::Arrival => jobs[event.index].level,
                EventKind::Wake { level } => level,
            };
            queues[level].push(event.index);
            *next_event += 1;
        }
        let _ = queue_heads;
    };

    let mut completed = 0usize;
    while completed < jobs.len() {
        admit_through(time, &events, &mut next_event, &mut queues, &queue_heads);

        let selected_level = (0..queues.len())
            .find(|&level| queue_heads[level] < queues[level].len());

        let Some(level) = selected_level else {
            if next_event >= events.len() {
                break;
            }
            let next_time = events[next_event].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let index = queues[level][queue_heads[level]];
        queue_heads[level] += 1;

        let id = jobs[index].id;
        if last_dispatched_id.is_some_and(|last_id| last_id != id) {
            time += switch_cost as u64;
        }
        admit_through(time, &events, &mut next_event, &mut queues, &queue_heads);

        let dispatch_level = states[index].level;
        let quantum = quanta[dispatch_level];
        let consumed = states[index].remaining.min(quantum);
        let start = time;
        time += consumed as u64;
        let end = time;
        states[index].remaining -= consumed;

        let burst_ended = states[index].remaining == 0;
        let finished = burst_ended && states[index].burst_position + 1 == jobs[index].bursts.len();
        runs.push(Run {
            id,
            start,
            end,
            finished,
            level: dispatch_level,
        });

        admit_through(end, &events, &mut next_event, &mut queues, &queue_heads);

        if finished {
            states[index].completion = Some(end);
            completed += 1;
        } else if burst_ended {
            states[index].burst_position += 2;
            let io_delay = jobs[index].bursts[states[index].burst_position - 1] as u64;
            states[index].remaining = jobs[index].bursts[states[index].burst_position];
            let wake_level = dispatch_level.saturating_sub(1);
            events.push(Event {
                time: end + io_delay,
                index,
                kind: EventKind::Wake { level: wake_level },
            });
            events[next_event..].sort_by_key(|event| {
                (
                    event.time,
                    match event.kind {
                        EventKind::Arrival => 0u8,
                        EventKind::Wake { .. } => 1u8,
                    },
                    event.index,
                )
            });
            admit_through(end, &events, &mut next_event, &mut queues, &queue_heads);
        } else {
            states[index].level = (dispatch_level + 1).min(quanta.len() - 1);
            queues[states[index].level].push(index);
        }

        last_dispatched_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;
    let mut schedule_end = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap_or(time);
        let turnaround = completion - job.arrival as u64;
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&length| length as u64)
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&length| length as u64)
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
        total_waiting += waiting;
        total_turnaround += turnaround;
        schedule_end = schedule_end.max(completion);
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end: schedule_end,
    })
}
```
