#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]

    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]

}

}

    pub minimum_adjustment: u64,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
    Month,
    Period,
    Credit { index: usize },
    CreditOrder { index: usize },
    InvoiceDiscount,
    InvalidName { index: usize },
    DuplicateName { index: usize },
    Discount,
    CombinedOverflow,
    Meter { index: usize, source: Box<Error> },
    InvoiceOverflow,
}

fn band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    units
        .checked_mul(unit)
        .and_then(|v| v.checked_add(start))
        .ok_or(Error::Overflow { index })
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().unwrap().up_to.is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn graduated_tiers(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut amount = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                let allocated = remaining.min(width);
                previous = bound;
                allocated
            }
            None => remaining,
        };

        if units > 0 {
            let charge = band_amount(units, tier.unit, tier.start, index)?;
            amount = amount
                .checked_add(charge)
                .ok_or(Error::Overflow { index })?;
            bands.push(Band {
                index,
                units,
                amount: charge,
            });
            remaining -= units;
        }
    }

    Ok(Charge { amount, bands })
}

fn volume_tiers(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let (index, tier) = tiers
        .iter()
        .enumerate()
        .find(|(_, tier)| tier.up_to.map_or(true, |bound| quantity <= bound))
        .unwrap();

    let amount = band_amount(quantity, tier.unit, tier.start, index)?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index,
            units: quantity,
            amount,
        }],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                })
            } else {
                let amount = band_amount(quantity, *unit, *start, 0)?;
                Ok(Charge {
                    amount,
                    bands: vec![Band {
                        index: 0,
                        units: quantity,
                        amount,
                    }],
                })
            }
        }
        Price::Tiers(tiers) => graduated_tiers(quantity, tiers),
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Flat { .. }, _) => quote(quantity, price)?,
        (Price::Tiers(tiers), Mode::Graduated) => graduated_tiers(quantity, tiers)?,
        (Price::Tiers(tiers), Mode::Volume) => volume_tiers(quantity, tiers)?,
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::CombinedOverflow)?;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if !matches!(days, 28 | 29 | 30 | 31) {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut previous_end = start;
    let mut credited = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        credited += credit.end - credit.start;
        previous_end = credit.end;
    }

    let active_days = end - start - credited;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice_remainder = remainder * 2;
            if twice_remainder > denominator
                || (twice_remainder == denominator && quotient % 2 == 1)
            {
                quotient + 1
            } else {
                quotient
            }
        }
    };

    Ok(Proration {
        active_days,
        amount: rounded as u64,
    })
}

fn discount_amount(base: u64, bps: u32) -> u64 {
    ((base as u128) * (bps as u128) / 10_000u128) as u64
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 32
        && !bytes.iter().any(|&byte| {
            byte <= 0x1f || byte == 0x7f || byte == b'|' || byte == b':'
        })
}

pub fn invoice(
    meters: &[Meter],
    invoice_discount_bps: u32,
    credit: u64,
) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 {
        return Err(Error::InvoiceDiscount);
    }

    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) {
            return Err(Error::InvalidName { index });
        }
        if meters[..index]
            .iter()
            .any(|earlier| earlier.name == meter.name)
        {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;

    for (index, meter) in meters.iter().enumerate() {
        if meter.discount_bps > 10_000 {
            return Err(Error::Meter {
                index,
                source: Box::new(Error::Discount),
            });
        }

        let bill = bill(
            meter.quantity,
            &meter.price,
            meter.mode.clone(),
            meter.minimum,
        )
        .map_err(|source| Error::Meter {
            index,
            source: Box::new(source),
        })?;

        let proration = prorate(
            meter.monthly_fee,
            meter.days,
            meter.start,
            meter.end,
            &meter.credits,
            meter.rounding.clone(),
        )
        .map_err(|source| Error::Meter {
            index,
            source: Box::new(source),
        })?;

        let base = bill
            .total
            .checked_add(proration.amount)
            .ok_or_else(|| Error::Meter {
                index,
                source: Box::new(Error::CombinedOverflow),
            })?;
        let discount = discount_amount(base, meter.discount_bps);
        let line_total = base - discount;

        subtotal = subtotal
            .checked_add(line_total)
            .ok_or(Error::InvoiceOverflow)?;

        lines.push(InvoiceLine {
            name: meter.name.clone(),
            usage: bill.charge.amount,
            minimum_adjustment: bill.minimum_adjustment,
            monthly: proration.amount,
            discount,
            total: line_total,
        });
    }

    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));

    let invoice_discount = discount_amount(subtotal, invoice_discount_bps);
    let available = subtotal - invoice_discount;
    let applied_credit = credit.min(available);
    let total = available - applied_credit;

    let mut rendered = String::new();
    for line in &lines {
        rendered.push_str(&format!(
            "{}:{}|{}|{}|{}|{}|{}\n",
            line.name.len(),
            line.name,
            line.usage,
            line.minimum_adjustment,
            line.monthly,
            line.discount,
            line.total
        ));
    }
    rendered.push_str(&format!(
        "TOTAL|{}|{}|{}|{}\n",
        subtotal, invoice_discount, applied_credit, total
    ));

    Ok(Invoice {
        lines,
        subtotal,
        discount: invoice_discount,
        credit: applied_credit,
        total,
        rendered,
    })
}
