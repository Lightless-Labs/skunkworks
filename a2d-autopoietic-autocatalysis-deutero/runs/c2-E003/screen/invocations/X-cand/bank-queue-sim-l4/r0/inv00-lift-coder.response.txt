#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

#[derive(Clone)]
struct Queued {
    index: usize,
    entry: u64,
    remaining_budget: Option<u64>,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Running {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event {
        time,
        customer: id,
        kind,
    });
}

fn best_waiter(queue: &[Queued], customers: &[Customer]) -> Option<usize> {
    queue.iter().enumerate().min_by(|(_, a), (_, b)| {
        let ca = &customers[a.index];
        let cb = &customers[b.index];
        ca.priority
            .cmp(&cb.priority)
            .then_with(|| b.paused.cmp(&a.paused))
            .then_with(|| {
                if a.paused {
                    a.pause_time.cmp(&b.pause_time)
                } else {
                    ca.arrival.cmp(&cb.arrival)
                }
            })
            .then_with(|| {
                if a.paused {
                    ca.id.cmp(&cb.id)
                } else {
                    cb.id.cmp(&ca.id)
                }
            })
    }).map(|(i, _)| i)
}

fn expire_queue(
    queue: &mut Vec<Queued>,
    customers: &[Customer],
    time: u64,
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<Queued> = Vec::new();
    let mut retained = Vec::with_capacity(queue.len());
    for q in queue.drain(..) {
        if q.deadline.map_or(false, |d| d <= time) {
            expired.push(q);
        } else {
            retained.push(q);
        }
    }
    expired.sort_by(|a, b| {
        a.deadline.unwrap().cmp(&b.deadline.unwrap()).then_with(|| {
            customers[a.index].id.cmp(&customers[b.index].id)
        })
    });
    for q in expired {
        let id = customers[q.index].id;
        stats.wait_ticks += q.waited as u128;
        stats.reneged += 1;
        emit(events, time, id, Kind::Renege { waited: q.waited });
    }
    *queue = retained;
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut queue: Vec<Queued> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatch: Option<u32> = None;
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut now = customers
        .get(*arrival_order.first().unwrap_or(&0))
        .map_or(0, |c| c.arrival);

    loop {
        // Phase A: complete the active service before any other phase.
        if running.as_ref().map_or(false, |r| r.finish == now) {
            let r = running.take().unwrap();
            let c = &customers[r.index];
            emit(&mut events, now, c.id, Kind::Finish);
            stats.completed += 1;
            stats.completion_order.push(c.id);
            stats.turnaround_ticks += (now - c.arrival) as u128;
        }

        // Phase B: expire customers who were waiting at the start of the tick.
        expire_queue(&mut queue, customers, now, &mut events, &mut stats);

        // Phase C: process the entire arrival cohort without dispatching mid-cohort.
        while next_arrival < arrival_order.len()
            && customers[arrival_order[next_arrival]].arrival == now
        {
            let index = arrival_order[next_arrival];
            next_arrival += 1;
            let c = &customers[index];
            emit(&mut events, now, c.id, Kind::Arrive);
            let occupied = queue.len();
            let limit = config.waiting_capacity + usize::from(running.is_none());
            if occupied >= limit {
                stats.balked += 1;
                emit(&mut events, now, c.id, Kind::Balk);
            } else {
                stats.accepted += 1;
                queue.push(Queued {
                    index,
                    entry: now,
                    remaining_budget: c.patience,
                    deadline: c.patience.map(|p| now + p),
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
            }
        }

        // Phase D: reconsider preemption once switching protection has ended.
        let can_preempt = running
            .as_ref()
            .map_or(false, |r| now >= r.work_at && !queue.is_empty());
        if can_preempt {
            let bi = best_waiter(&queue, customers).unwrap();
            let running_index = running.as_ref().unwrap().index;
            if customers[queue[bi].index].priority < customers[running_index].priority {
                let r = running.take().unwrap();
                let paused = &customers[r.index];
                let remaining = r.finish - now;
                emit(
                    &mut events,
                    now,
                    paused.id,
                    Kind::Preempt {
                        remaining,
                        by: customers[queue[bi].index].id,
                    },
                );
                stats.preemptions += 1;
                stats.busy_ticks += (now - r.dispatch_time) as u128;
                let deadline = match queue.iter().find(|q| q.index == r.index) {
                    Some(_) => None,
                    None => None,
                };
                let mut q = Queued {
                    index: r.index,
                    entry: now,
                    remaining_budget: paused.patience,
                    deadline,
                    waited: 0,
                    paused: true,
                    pause_time: now,
                };
                // Recover the cumulative unused patience budget from prior waiting.
                let used = stats.wait_ticks;
                let _ = used;
                if let Some(budget) = q.remaining_budget {
                    q.deadline = Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                        customer: paused.id,
                        time: now,
                        stage: OverflowStage::Patience,
                    })?);
                }
                queue.push(q);
            }
        }

        // Phase E: dispatch repeatedly, including inline zero-duration finishes.
        loop {
            if running.is_some() || queue.is_empty() {
                break;
            }
            let qi = best_waiter(&queue, customers).unwrap();
            let mut q = queue.remove(qi);
            let c = &customers[q.index];
            let waited = now - q.entry;
            q.waited = q.waited.saturating_add(waited);
            stats.wait_ticks += waited as u128;
            if let Some(budget) = q.remaining_budget {
                q.remaining_budget = Some(budget.saturating_sub(waited));
            }

            let cost = if last_dispatch.map_or(true, |id| id == c.id) {
                0
            } else {
                config.switch_cost
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: c.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(q.remaining_budget.map_or(c.service, |_| c.service)).ok_or(
                Error::TimeOverflow {
                    customer: c.id,
                    time: now,
                    stage: OverflowStage::Finish,
                },
            )?;

            if q.paused {
                emit(
                    &mut events,
                    now,
                    c.id,
                    Kind::Resume {
                        remaining: c.service,
                        work_at,
                        finish,
                    },
                );
            } else {
                stats.started += 1;
                emit(
                    &mut events,
                    now,
                    c.id,
                    Kind::Start {
                        remaining: c.service,
                        work_at,
                        finish,
                    },
                );
            }
            last_dispatch = Some(c.id);

            if finish == now {
                emit(&mut events, now, c.id, Kind::Finish);
                stats.completed += 1;
                stats.completion_order.push(c.id);
                stats.turnaround_ticks += (now - c.arrival) as u128;
                continue;
            }
            running = Some(Running {
                index: q.index,
                remaining: c.service,
                work_at,
                finish,
                dispatch_time: now,
            });
        }

        // Phase F: expire any customers newly queued during C or D.
        expire_queue(&mut queue, customers, now, &mut events, &mut stats);

        if running.is_none() && queue.is_empty() && next_arrival >= arrival_order.len() {
            break;
        }

        let mut next = u64::MAX;
        let mut has_next = false;
        if next_arrival < arrival_order.len() {
            let t = customers[arrival_order[next_arrival]].arrival;
            next = t;
            has_next = true;
        }
        if let Some(r) = &running {
            for t in [r.work_at, r.finish] {
                if t > now && (!has_next || t < next) {
                    next = t;
                    has_next = true;
                }
            }
        }
        for q in &queue {
            if let Some(d) = q.deadline {
                if d > now && (!has_next || d < next) {
                    next = d;
                    has_next = true;
                }
            }
        }
        if !has_next {
            break;
        }
        now = next;
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
