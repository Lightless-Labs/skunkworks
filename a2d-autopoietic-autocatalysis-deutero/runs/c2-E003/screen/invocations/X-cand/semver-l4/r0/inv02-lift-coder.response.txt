use std::cmp::Ordering;
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn number(s: &str, at: usize, leading_zero: bool) -> Result<u64, VersionError> {
    if leading_zero && s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at });
    }
    let mut n = 0u64;
    for b in s.bytes() {
        let d = (b - b'0') as u64;
        n = n.checked_mul(10).and_then(|x| x.checked_add(d))
            .ok_or(VersionError::Overflow { at })?;
    }
    Ok(n)
}

fn parse_ids(s: &str, base: usize, prerelease: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut start = 0;
    loop {
        let end = bytes[start..].iter().position(|b| *b == b'.')
            .map(|x| start + x).unwrap_or(bytes.len());
        if end == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
        for i in start..end {
            if !is_ident_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: base + i });
            }
        }
        let item = &s[start..end];
        if prerelease && item.bytes().all(is_digit) && item.len() > 1 && item.starts_with('0') {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        out.push(item.to_string());
        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let b = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];
    for (field, slot) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < b.len() && is_digit(b[pos]) {
            pos += 1;
        }
        if start == pos {
            if pos == b.len() || matches!(b[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        let field_str = &s[start..pos];
        *slot = number(field_str, start, true)?;
        if field < 2 {
            if pos < b.len() && b[pos] == b'.' {
                pos += 1;
            } else if pos == b.len() || matches!(b[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < b.len() && b[pos] == b'-' {
        let start = pos + 1;
        let end = b[start..].iter().position(|x| *x == b'+')
            .map(|x| start + x).unwrap_or(b.len());
        pre = parse_ids(&s[start..end], start, true)?;
        pos = end;
    }
    if pos < b.len() && b[pos] == b'+' {
        let start = pos + 1;
        build = parse_ids(&s[start..], start, false)?;
        pos = b.len();
    }
    if pos < b.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major: core[0], minor: core[1], patch: core[2], pre, build })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() {
        return if b.is_empty() { Ordering::Equal } else { Ordering::Greater };
    }
    if b.is_empty() {
        return Ordering::Less;
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let o = match (xn, yn) {
            (true, true) => cmp_numeric(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if o != Ordering::Equal {
            return o;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn is_wild(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

#[derive(Clone)]
struct Partial {
    vals: Vec<u64>,
    count: usize,
    full: Option<Version>,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.is_empty() {
        return Err(RangeError::Syntax { at });
    }
    if s.contains('-') || s.contains('+') {
        return parse_version(s).map(|v| Partial {
            vals: vec![v.major, v.minor, v.patch], count: 3, full: Some(v)
        }).map_err(|error| RangeError::Version { at, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|x| is_wild(x)) {
        return parse_version(s).map(|v| Partial {
            vals: vec![v.major, v.minor, v.patch], count: 3, full: Some(v)
        }).map_err(|error| RangeError::Version { at, error });
    }
    if fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut vals = Vec::new();
    let mut wildcard_seen = false;
    for f in &fields {
        if is_wild(f) {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.bytes().all(is_digit) ||
                (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let mut n = 0u64;
            for b in f.bytes() {
                n = n.checked_mul(10).and_then(|x| x.checked_add((b - b'0') as u64))
                    .ok_or(RangeError::Syntax { at })?;
            }
            vals.push(n);
        }
    }
    let count = vals.len();
    Ok(Partial { vals, count, full: None })
}

fn make_version(m: u64, n: u64, p: u64, pre: Vec<String>) -> Version {
    Version { major: m, minor: n, patch: p, pre, build: Vec::new() }
}

fn padded(p: &Partial) -> Version {
    make_version(
        *p.vals.first().unwrap_or(&0),
        *p.vals.get(1).unwrap_or(&0),
        *p.vals.get(2).unwrap_or(&0),
        p.full.as_ref().map(|v| v.pre.clone()).unwrap_or_default(),
    )
}

fn upper_next(p: &Partial, mode: &str, at: usize) -> Result<Version, RangeError> {
    let mut m = *p.vals.first().unwrap_or(&0);
    let mut n = *p.vals.get(1).unwrap_or(&0);
    let mut patch = *p.vals.get(2).unwrap_or(&0);
    let bump = if mode == "tilde" {
        if p.count == 1 { 0 } else { 1 }
    } else if mode == "caret" {
        if m > 0 || p.count == 1 { 0 }
        else if n > 0 || p.count == 2 { 1 }
        else { 2 }
    } else if p.count == 1 { 0 } else { 1 };
    match bump {
        0 => { m = m.checked_add(1).ok_or(RangeError::Syntax { at })?; n = 0; patch = 0; }
        1 => { n = n.checked_add(1).ok_or(RangeError::Syntax { at })?; patch = 0; }
        _ => { patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?; }
    }
    Ok(make_version(m, n, patch, vec!["0".to_string()]))
}

fn add_cmp(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push(core(&v));
    }
    arm.cmps.push(Cmp { op, version: v, anchor });
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    if s.is_empty() || !s.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn partial_lower(p: &Partial) -> Version {
    padded(p)
}

fn process_partial_token(arm: &mut Arm, token: &str, at: usize, mode: &str) -> Result<(), RangeError> {
    let raw = if mode == "caret" || mode == "tilde" { &token[1..] } else { token };
    let operand_at = at + token.len() - raw.len();
    let p = parse_partial(raw, operand_at)?;
    if mode == "bare" && p.full.is_some() {
        let v = p.full.unwrap();
        add_cmp(arm, Op::Eq, v, true);
        return Ok(());
    }
    if mode != "bare" && p.count == 0 {
        return Err(RangeError::Syntax { at });
    }
    if mode == "bare" && p.count == 0 {
        return Ok(());
    }
    let lower = partial_lower(&p);
    add_cmp(arm, Op::Ge, lower, true);
    if mode != "bare" || p.count < 3 {
        if p.count < 3 || mode != "bare" {
            let upper = upper_next(&p, mode, operand_at)?;
            add_cmp(arm, Op::Lt, upper, false);
        }
    }
    Ok(())
}

fn parse_comparator(token: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, n) = if token.starts_with(">=") { (Op::Ge, 2) }
        else if token.starts_with("<=") { (Op::Le, 2) }
        else if token.starts_with('>') { (Op::Gt, 1) }
        else if token.starts_with('<') { (Op::Lt, 1) }
        else { (Op::Eq, 1) };
    let operand = &token[n..];
    let v = parse_full_operand(operand, at + n)
        .map_err(|e| match e {
            RangeError::Syntax { .. } => RangeError::Syntax { at },
            other => other,
        })?;
    add_cmp(arm, op, v, true);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[(String, usize)], arm: &mut Arm) -> Result<(), RangeError> {
    let hyphens: Vec<usize> = tokens.iter().enumerate()
        .filter_map(|(i, x)| if x.0 == "-" { Some(i) } else { None }).collect();
    if !hyphens.is_empty() {
        if tokens.len() != 3 || hyphens != vec![1] {
            return Err(RangeError::Syntax { at: tokens[hyphens[0]].1 });
        }
        let (ls, la) = &tokens[0];
        let (us, ua) = &tokens[2];
        let l = parse_partial(ls, *la)?;
        let u = parse_partial(us, *ua)?;
        if l.count == 0 { return Err(RangeError::Syntax { at: *la }); }
        if u.count == 0 { return Err(RangeError::Syntax { at: *ua }); }
        add_cmp(arm, Op::Ge, partial_lower(&l), true);
        if let Some(v) = u.full {
            add_cmp(arm, Op::Le, v, true);
        } else {
            let upper = upper_next(&u, "hyphen", *ua)?;
            add_cmp(arm, Op::Lt, upper, false);
        }
        return Ok(());
    }
    for (token, at) in tokens {
        let first = token.as_bytes().first().copied().unwrap_or(0);
        if first == b'^' {
            process_partial_token(arm, token, *at, "caret")?;
        } else if first == b'~' {
            process_partial_token(arm, token, *at, "tilde")?;
        } else if matches!(first, b'<' | b'>' | b'=') {
            parse_comparator(token, *at, arm)?;
        } else if first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X') {
            process_partial_token(arm, token, *at, "bare")?;
        } else {
            return Err(RangeError::Syntax { at: *at });
        }
    }
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { cmps: Vec::new(), anchors: Vec::new() };
    parse_hyphen_arm(tokens, &mut arm)?;
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut tokens = Vec::<(String, usize)>::new();
    let mut i = 0usize;
    let mut sep_positions = Vec::new();
    while i < b.len() {
        while i < b.len() && whitespace(b[i]) { i += 1; }
        if i == b.len() { break; }
        let start = i;
        while i < b.len() && !whitespace(b[i]) { i += 1; }
        let token = &s[start..i];
        if token == "||" {
            sep_positions.push((start, tokens.len()));
            tokens.push((token.to_string(), start));
        } else {
            tokens.push((token.to_string(), start));
        }
    }
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    for (token, at) in tokens {
        if token == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
        } else {
            current.push((token, at));
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    let _ = sep_positions;
    Ok(Range { arms })
}

fn cmp_satisfies(v: &Version, c: &Cmp) -> bool {
    match compare_versions(v, &c.version) {
        Ordering::Less => matches!(c.op, Op::Lt | Op::Le),
        Ordering::Equal => matches!(c.op, Op::Eq | Op::Le | Op::Ge),
        Ordering::Greater => matches!(c.op, Op::Gt | Op::Ge),
    }
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.cmps.iter().all(|c| cmp_satisfies(v, c)) {
        return false;
    }
    if v.pre.is_empty() || include {
        return true;
    }
    arm.anchors.contains(&core(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

fn next_core(c: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if c.2 < u64::MAX { Some((c.0, c.1, c.2 + 1)) }
    else if c.1 < u64::MAX { Some((c.0, c.1 + 1, 0)) }
    else if c.0 < u64::MAX { Some((c.0 + 1, 0, 0)) }
    else { None }
}

fn arm_test_candidates(a: &Arm, b: Option<&Arm>) -> Vec<Version> {
    let mut out = Vec::new();
    for c in a.cmps.iter().chain(b.into_iter().flat_map(|x| x.cmps.iter())) {
        let mut v = c.version.clone();
        v.build.clear();
        out.push(v.clone());
        if !v.pre.is_empty() {
            let mut successor = v.clone();
            successor.pre.push("0".to_string());
            out.push(successor);
            let mut stable = v.clone();
            stable.pre.clear();
            out.push(stable);
        } else if let Some(n) = next_core(core(&v)) {
            out.push(make_version(n.0, n.1, n.2, vec!["0".into()]));
            out.push(make_version(n.0, n.1, n.2, Vec::new()));
        }
        let c0 = core(&v);
        if c0.2 > 0 {
            out.push(make_version(c0.0, c0.1, c0.2 - 1, Vec::new()));
        }
    }
    out.push(make_version(0, 0, 0, Vec::new()));
    out
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            for v in arm_test_candidates(aa, Some(bb)) {
                if arm_matches(&v, aa, include_prerelease) &&
                    arm_matches(&v, bb, include_prerelease) {
                    return true;
                }
            }
            if !include_prerelease {
                for tuple in aa.anchors.iter().filter(|x| bb.anchors.contains(x)) {
                    let v = make_version(tuple.0, tuple.1, tuple.2, vec!["0".into()]);
                    if arm_matches(&v, aa, false) && arm_matches(&v, bb, false) {
                        return true;
                    }
                }
            }
        }
    }
    false
}

fn bound_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn arm_canonical(arm: &Arm) -> String {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.cmps {
        let mut v = c.version.clone();
        v.build.clear();
        match c.op {
            Op::Eq => {
                lower = Some((v.clone(), true));
                upper = Some((v, true));
            }
            Op::Gt | Op::Ge => {
                let inclusive = c.op == Op::Ge;
                let replace = lower.as_ref().map_or(true, |(old, inc)| {
                    compare_versions(&v, old) == Ordering::Greater ||
                        (compare_versions(&v, old) == Ordering::Equal && !inclusive && *inc)
                });
                if replace { lower = Some((v, inclusive)); }
            }
            Op::Lt | Op::Le => {
                let inclusive = c.op == Op::Le;
                let replace = upper.as_ref().map_or(true, |(old, inc)| {
                    compare_versions(&v, old) == Ordering::Less ||
                        (compare_versions(&v, old) == Ordering::Equal && !inclusive && *inc)
                });
                if replace { upper = Some((v, inclusive)); }
            }
        }
    }
    let lo = lower.map(|(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, bound_text(&v))
        .replace(">=", if inc { ">=" } else { ">" }));
    let hi = upper.map(|(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, bound_text(&v)));
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let anchor_text = if anchors.is_empty() { "-".to_string() } else {
        anchors.iter().map(|x| format!("{}.{}.{}", x.0, x.1, x.2)).collect::<Vec<_>>().join(",")
    };
    format!("[{};{};pre={}]", lo.unwrap_or_else(|| "*".into()), hi.unwrap_or_else(|| "*".into()), anchor_text)
}

pub fn canonical(r: &Range) -> String {
    let mut arms: Vec<String> = r.arms.iter().map(arm_canonical).collect();
    arms.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    arms.dedup();
    if arms.is_empty() { "EMPTY".into() } else { arms.join(" || ") }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push((s, v, index)),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<(&String, Version, usize)> = None;
    for (s, v, i) in parsed {
        if !matches(&v, r) { continue; }
        let replace = best.as_ref().map_or(true, |(_, bv, _)| {
            let o = compare_versions(&v, bv);
            if max { o == Ordering::Greater } else { o == Ordering::Less }
        });
        if replace { best = Some((s, v, i)); }
    }
    Ok(best.map(|(s, _, _)| s.clone()))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {

    pub version: String,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    parsed: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => match parse_range(s) {
            Ok(r) => { parsed.push(r); Ok(()) }
            Err(error) => Err(SelectError::InvalidRange {
                package: package.to_string(), label: label.to_string(),
                path: path.clone(), error
            }),
        },
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, parsed)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn expr_matches(expr: &SetExpr, v: &Version, include: bool, leaves: &mut usize, parsed: &[Range]) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let r = &parsed[*leaves];
            *leaves += 1;
            matches_with(v, r, include)
        }
        SetExpr::Union(children) => children.iter().any(|x| expr_matches(x, v, include, leaves, parsed)),
        SetExpr::Intersection(children) => children.iter().all(|x| expr_matches(x, v, include, leaves, parsed)),
    }
}

fn combination_unsat(
    chosen: &[usize],
    candidates: &[Version],
    constraints: &[Constraint],
    ranges: &[Vec<Range>],
    include: bool,
) -> bool {
    candidates.iter().all(|v| {
        chosen.iter().any(|&ci| {
            let mut idx = 0;
            !expr_matches(&constraints[ci].expr, v, include, &mut idx, &ranges[ci])
        })
    })
}

fn find_conflict(
    candidates: &[Version],
    constraints: &[Constraint],
    ranges: &[Vec<Range>],
    include: bool,
) -> Vec<String> {
    if candidates.is_empty() { return Vec::new(); }
    let n = constraints.len();
    for size in 1..=n {
        let mut found: Vec<Vec<String>> = Vec::new();
        fn rec(
            start: usize, need: usize, picked: &mut Vec<usize>,
            n: usize, candidates: &[Version], constraints: &[Constraint],
            ranges: &[Vec<Range>], include: bool, found: &mut Vec<Vec<String>>,
        ) {
            if need == 0 {
                if combination_unsat(picked, candidates, constraints, ranges, include) {
                    let mut labels: Vec<String> = picked.iter().map(|&i| constraints[i].label.clone()).collect();
                    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                    found.push(labels);
                }
                return;
            }
            if n - start < need { return; }
            for i in start..=n - need {
                picked.push(i);
                rec(i + 1, need - 1, picked, n, candidates, constraints, ranges, include, found);
                picked.pop();
            }
        }
        rec(0, size, &mut Vec::new(), n, candidates, constraints, ranges, include, &mut found);
        if !found.is_empty() {
            found.sort_by(|a, b| {
                for (x, y) in a.iter().zip(b) {
                    let o = x.as_bytes().cmp(y.as_bytes());
                    if o != Ordering::Equal { return o; }
                }
                a.len().cmp(&b.len())
            });
            return found.remove(0);
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: p.name.clone(), label: c.label.clone()
                });
            }
        }
    }
    let mut candidates_by_package = Vec::new();
    for p in packages {
        let mut parsed = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            match parse_version(s) {
                Ok(v) => parsed.push(v),
                Err(error) => return Err(SelectError::InvalidCandidate {
                    package: p.name.clone(), index, error
                }),
            }
        }
        candidates_by_package.push(parsed);
    }
    let mut ranges_by_package: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut per_constraint = Vec::new();
        for c in &p.constraints {
            let mut leaves = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut leaves)?;
            per_constraint.push(leaves);
        }
        ranges_by_package.push(per_constraint);
    }

    let mut result = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let candidates = &candidates_by_package[pi];
        let ranges = &ranges_by_package[pi];
        let mut qualifying = Vec::new();
        for (i, v) in candidates.iter().enumerate() {
            let accepted = p.constraints.iter().enumerate().all(|(ci, c)| {
                let mut idx = 0;
                expr_matches(&c.expr, v, include_prerelease, &mut idx, &ranges[ci])
            });
            if accepted { qualifying.push(i); }
        }
        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: find_conflict(candidates, &p.constraints, ranges, include_prerelease),
            });
        }
        let stable_only = matches!(preference, Preference::StableFirst) &&
            qualifying.iter().any(|&i| candidates[i].pre.is_empty());
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            if stable_only && !candidates[i].pre.is_empty() { continue; }
            if stable_only && !candidates[best].pre.is_empty() { best = i; continue; }
            let order = compare_versions(&candidates[i], &candidates[best]);
            let replace = match preference {
                Preference::Lowest => order == Ordering::Less,
                Preference::Highest | Preference::StableFirst => order == Ordering::Greater,
            };
            if replace { best = i; }
        }
        result.push(Selection {
            package: p.name.clone(),
            version: p.candidates[best].clone(),
        });
    }
    Ok(result)
}
