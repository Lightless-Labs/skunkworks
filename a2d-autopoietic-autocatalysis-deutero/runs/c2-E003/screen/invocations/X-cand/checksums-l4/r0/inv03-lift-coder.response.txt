#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Span {
    pub const fn new(start: usize, end: usize) -> Self {
        Self { start, end }
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

fn crc32_update_raw(mut register: u32, data: &[u8]) -> u32 {
    for &byte in data {
        register ^= byte as u32;
        for _ in 0..8 {
            let low = register & 1;
            register >>= 1;
            if low != 0 {
                register ^= 0xedb88320;
            }
        }
    }
    register
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_update_raw(0xffff_ffff, data) ^ 0xffff_ffff
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    const MOD: u32 = 65_521;
    for &byte in data {
        *a = (*a + byte as u32) % MOD;
        *b = (*b + *a) % MOD;
    }
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc_raw = 0xffff_ffff;
    let mut a = 1u32;
    let mut b = 0u32;

    for span in spans {
        let selected = &data[span.start..span.end];
        crc_raw = crc32_update_raw(crc_raw, selected);
        adler_update(&mut a, &mut b, selected);
    }

    Ok(Hashes {
        crc32: crc_raw ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

fn fletcher16_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % 255;
        *b = (*b + *a) % 255;
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    fletcher16_update(&mut a, &mut b, data);
    ((b as u16) << 8) | a as u16
}

fn fletcher32_word(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);

    for pair in &mut chunks {
        let word = u16::from_be_bytes([pair[0], pair[1]]);
        fletcher32_word(&mut a, &mut b, word);
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_word(&mut a, &mut b, (last as u16) << 8);
    }

    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                fletcher32_word(&mut a, &mut b, u16::from_be_bytes([high, byte]));
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_word(&mut a, &mut b, (high as u16) << 8);
    }

    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut register: u16 = match variant {
                Crc16::Xmodem => 0,
                Crc16::False => 0xffff,
                Crc16::X25 => unreachable!(),
            };

            for &byte in data {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if top != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            register
        }
        Crc16::X25 => {
            let mut register = 0xffffu16;
            for &byte in data {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
            register ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc_raw = 0xffff_ffff;
    let mut adler_a = 1u32;
    let mut adler_b = 0u32;
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_pending: Option<u8> = None;
    let mut crc16_register = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        crc_raw = crc32_update_raw(crc_raw, selected);
        adler_update(&mut adler_a, &mut adler_b, selected);
        fletcher16_update(&mut f16_a, &mut f16_b, selected);

        for &byte in selected {
            if let Some(high) = f32_pending.take() {
                fletcher32_word(&mut f32_a, &mut f32_b, u16::from_be_bytes([high, byte]));
            } else {
                f32_pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_register ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = crc16_register & 0x8000;
                        crc16_register = crc16_register.wrapping_shl(1);
                        if top != 0 {
                            crc16_register ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_register ^= byte as u16;
                    for _ in 0..8 {
                        let low = crc16_register & 1;
                        crc16_register >>= 1;
                        if low != 0 {
                            crc16_register ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(high) = f32_pending {
        fletcher32_word(&mut f32_a, &mut f32_b, (high as u16) << 8);
    }

    let crc16_value = match variant {
        Crc16::Xmodem | Crc16::False => crc16_register,
        Crc16::X25 => crc16_register ^ 0xffff,
    };

    Ok(Multi {
        base: Hashes {
            crc32: crc_raw ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_value,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc_raw = crc32_update_raw(self.crc_raw, data);
        adler_update(&mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            let selected = &data[span.start..span.end];
            self.crc_raw = crc32_update_raw(self.crc_raw, selected);
            adler_update(&mut self.adler_a, &mut self.adler_b, selected);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_hex_u32(bytes, 1, 9)?;
        let adler = parse_hex_u32(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn crc32_combine_raw(left: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];

    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut length = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }
    }

    crc
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    crc32_combine_raw(left, right_len) ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }

    Ok(Part {
        crc32: combined,
        len: total,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut len = 0u64;
    let mut crc_raw = 0xffff_ffff;
    let mut a = 1u32;
    let mut b = 0u32;

    for span in spans {
        let selected = &data[span.start..span.end];
        len += selected.len() as u64;
        crc_raw = crc32_update_raw(crc_raw, selected);
        adler_update(&mut a, &mut b, selected);
    }

    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len,
        crc_raw ^ 0xffff_ffff,
        (b << 16) | a
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let len = parse_hex_u64(bytes, 4, 20)?;
    let crc = parse_hex_u32_verify(bytes, 25, 33)?;
    let adler = parse_hex_u32_verify(bytes, 40, 48)?;

    let hashes = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    let selected_len: usize = spans.iter().map(|span| span.end - span.start).sum();
    if len != selected_len as u64 {
        return Err(VerifyError::Length);
    }
    if crc != hashes.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: hashes.crc32,
        });
    }
    if adler != hashes.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: hashes.adler32,
        });
    }
    Ok(())
}

fn parse_hex_u32_verify(bytes: &[u8], start: usize, end: usize) -> Result<u32, VerifyError> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        value = (value << 4) | digit as u64;
    }
    Ok(value)
}
