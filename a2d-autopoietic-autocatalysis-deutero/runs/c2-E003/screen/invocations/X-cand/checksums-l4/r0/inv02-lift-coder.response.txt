    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    pub fletcher32: u32,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },

fn crc32_update_byte(mut state: u32, byte: u8) -> u32 {
    state ^= byte as u32;
    for _ in 0..8 {
        let low = state & 1;
        state >>= 1;
        if low != 0 {
            state ^= 0xedb8_8320;
        }
    }
    state
}

fn adler_update_byte(a: &mut u32, b: &mut u32, byte: u8) {
    *a = (*a + byte as u32) % 65_521;
    *b = (*b + *a) % 65_521;
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut state = 0xffff_ffff;
    for &byte in data {
        state = crc32_update_byte(state, byte);
    }
    state ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1u32, 0u32);
    for &byte in data {
        adler_update_byte(&mut a, &mut b, byte);
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc_state: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_state: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_state: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_state ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc_state = crc32_update_byte(self.crc_state, byte);
            adler_update_byte(&mut self.adler_a, &mut self.adler_b, byte);
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        let mut adler = 0u32;
        for (offset, &byte) in bytes.iter().enumerate() {
            if (1..9).contains(&offset) || (11..19).contains(&offset) {
                let digit = hex_value(byte).ok_or(SnapshotError::Hex { offset })?;
                if offset < 9 {
                    crc = (crc << 4) | digit as u32;
                } else {
                    adler = (adler << 4) | digit as u32;
                }
            }
        }
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

#[derive(Debug, Clone, Copy)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending_high: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending_high: None,
        }
    }

    fn update_byte(&mut self, byte: u8) {
        if let Some(high) = self.pending_high.take() {
            self.update_word(((high as u16) << 8) | byte as u16);
        } else {
            self.pending_high = Some(byte);
        }
    }

    fn update_word(&mut self, word: u16) {
        self.a = (self.a + word as u32) % 65_535;
        self.b = (self.b + self.a) % 65_535;
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending_high.take() {
            self.update_word((high as u16) << 8);
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    for &byte in data {
        state.update_byte(byte);
    }
    state.finish()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Fletcher32State::new();
    for span in spans {
        for &byte in &data[span.start..span.end] {
            state.update_byte(byte);
        }
    }
    Ok(state.finish())
}

fn crc16_update(mut state: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                state ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = state & 0x8000;
                    state = state.wrapping_shl(1);
                    if top != 0 {
                        state ^= 0x1021;
                    }
                }
            }
            state
        }
        Crc16::X25 => {
            for &byte in data {
                state ^= byte as u16;
                for _ in 0..8 {
                    let low = state & 1;
                    state >>= 1;
                    if low != 0 {
                        state ^= 0x8408;
                    }
                }
            }
            state ^ 0xffff
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let initial = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    crc16_update(initial, data, variant)
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32 = Fletcher32State::new();
    let mut c16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);
        for &byte in selected {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;
            f32.update_byte(byte);
        }
        c16 = crc16_update(c16, selected, variant);
    }

    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: f32.finish(),
        crc16: c16,
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }

    let mut even = [0u32; 32];
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut length = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        if part.len != 0 {
            combined = crc32_combine(combined, part.crc32, part.len);
        }
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

fn selected_stats(data: &[u8], spans: &[Span]) -> (u64, Hashes) {
    let mut total = 0u64;
    let mut stream = Stream::new();
    for span in spans {
        let selected = &data[span.start..span.end];
        total += selected.len() as u64;
        stream.update(selected);
    }
    (total, stream.checksum())
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let (len, sums) = selected_stats(data, spans);
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, sums.crc32, sums.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut length = 0u64;
    let mut crc = 0u32;
    let mut adler = 0u32;
    for (offset, &byte) in bytes.iter().enumerate() {
        let digit = if (4..20).contains(&offset)
            || (25..33).contains(&offset)
            || (40..48).contains(&offset)
        {
            Some(hex_value(byte).ok_or(VerifyError::Hex { offset })?)
        } else {
            None
        };
        if let Some(digit) = digit {
            if (4..20).contains(&offset) {
                length = (length << 4) | digit as u64;
            } else if (25..33).contains(&offset) {
                crc = (crc << 4) | digit as u32;
            } else {
                adler = (adler << 4) | digit as u32;
            }
        }
    }

    let (actual_len, sums) = selected_stats(data, spans);
    if length != actual_len {
        return Err(VerifyError::Length);
    }
    if crc != sums.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: sums.crc32,
        });
    }
    if adler != sums.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: sums.adler32,
        });
    }
    Ok(())
}
