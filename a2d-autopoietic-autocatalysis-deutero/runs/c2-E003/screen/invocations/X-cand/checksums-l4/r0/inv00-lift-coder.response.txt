const ADLER_MOD: u32 = 65_521;

pub fn crc32(data: &[u8]) -> u32 {
    let mut reg = 0xffff_ffffu32;
    for &byte in data {
        reg ^= byte as u32;
        for _ in 0..8 {
            let low = reg & 1;
            reg >>= 1;
            if low != 0 {
                reg ^= 0xedb8_8320;
            }
        }
    }
    reg ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % ADLER_MOD;
        b = (b + a) % ADLER_MOD;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

#[derive(Clone, Copy)]
struct Accumulator {
    crc: u32,
    a: u32,
    b: u32,
    f16a: u32,
    f16b: u32,
    f32a: u32,
    f32b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
    crc16_variant: Crc16,
}

impl Accumulator {
    fn new(crc16_variant: Crc16) -> Self {
        let initial_crc16 = match crc16_variant {
            Crc16::Xmodem => 0,
            Crc16::False | Crc16::X25 => 0xffff,
        };
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
            f16a: 0,
            f16b: 0,
            f32a: 0,
            f32b: 0,
            f32_pending: None,
            crc16: initial_crc16,
            crc16_variant,
        }
    }

    fn push(&mut self, byte: u8) {
        self.crc ^= byte as u32;
        for _ in 0..8 {
            let low = self.crc & 1;
            self.crc >>= 1;
            if low != 0 {
                self.crc ^= 0xedb8_8320;
            }
        }

        self.a = (self.a + byte as u32) % ADLER_MOD;
        self.b = (self.b + self.a) % ADLER_MOD;

        self.f16a = (self.f16a + byte as u32) % 255;
        self.f16b = (self.f16b + self.f16a) % 255;

        if let Some(high) = self.f32_pending.take() {
            self.push_fletcher32_word(((high as u16) << 8) | byte as u16);
        } else {
            self.f32_pending = Some(byte);
        }

        match self.crc16_variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = self.crc16 & 0x8000;
                    self.crc16 = self.crc16.wrapping_shl(1);
                    if top != 0 {
                        self.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                self.crc16 ^= byte as u16;
                for _ in 0..8 {
                    let low = self.crc16 & 1;
                    self.crc16 >>= 1;
                    if low != 0 {
                        self.crc16 ^= 0x8408;
                    }
                }
            }
        }
    }

    fn push_fletcher32_word(&mut self, word: u16) {
        self.f32a = (self.f32a + word as u32) % 65_535;
        self.f32b = (self.f32b + self.f32a) % 65_535;
    }

    fn finish(mut self) -> Multi {
        if let Some(high) = self.f32_pending.take() {
            self.push_fletcher32_word((high as u16) << 8);
        }

        let crc = self.crc ^ 0xffff_ffff;
        let crc16 = match self.crc16_variant {
            Crc16::X25 => self.crc16 ^ 0xffff,
            _ => self.crc16,
        };
        Multi {
            base: Hashes {
                crc32: crc,
                adler32: (self.b << 16) | self.a,
            },
            fletcher16: ((self.f16b as u16) << 8) | self.f16a as u16,
            fletcher32: (self.f32b << 16) | self.f32a,
            crc16,
        }
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            acc.push(byte);
        }
    }
    Ok(acc.finish().base)
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc ^= byte as u32;
            for _ in 0..8 {
                let low = self.crc & 1;
                self.crc >>= 1;
                if low != 0 {
                    self.crc ^= 0xedb8_8320;
                }
            }
            self.a = (self.a + byte as u32) % ADLER_MOD;
            self.b = (self.b + self.a) % ADLER_MOD;
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        let mut next = self.clone();
        for span in spans {
            next.update(&data[span.start..span.end]);
        }
        *self = next;
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let Hashes { crc32, adler32 } = self.checksum();
        format!("c{crc32:08x}:a{adler32:08x}")
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let crc = parse_hex(bytes, 1, 9)?;
        let adler = parse_hex(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

fn parse_hex(bytes: &[u8], start: usize, end: usize) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => bytes[offset] - b'0',
            b'a'..=b'f' => bytes[offset] - b'a' + 10,
            _ => return Err(SnapshotError::Hex { offset }),
        };
        value = (value << 4) | digit as u32;
    }
    Ok(value)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (0u32, 0u32);
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = ((pair[0] as u16) << 8) | pair[1] as u16;
        a = (a + word as u32) % 65_535;
        b = (b + a) % 65_535;
    }
    if let Some(&last) = chunks.remainder().first() {
        a = (a + ((last as u32) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                let word = ((high as u16) << 8) | byte as u16;
                a = (a + word as u32) % 65_535;
                b = (b + a) % 65_535;
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        a = (a + ((high as u32) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut acc = Accumulator::new(variant);
    for &byte in data {
        acc.push(byte);
    }
    let mut result = acc.finish().crc16;
    // finish() finalizes X25 and leaves the other variants unchanged.
    if variant == Crc16::X25 {
        result
    } else {
        result
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(variant);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            acc.push(byte);
        }
    }
    Ok(acc.finish())
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    fn zero_byte(mut reg: u32) -> u32 {
        for _ in 0..8 {
            let low = reg & 1;
            reg >>= 1;
            if low != 0 {
                reg ^= 0xedb8_8320;
            }
        }
        reg
    }

    fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
        let mut sum = 0u32;
        let mut index = 0usize;
        while vector != 0 {
            if vector & 1 != 0 {
                sum ^= matrix[index];
            }
            vector >>= 1;
            index += 1;
        }
        sum
    }

    let mut matrix = [0u32; 32];
    for bit in 0..32 {
        matrix[bit] = zero_byte(1u32 << bit);
    }
    let mut value = left;
    let mut length = right_len;
    while length != 0 {
        if length & 1 != 0 {
            value = matrix_times(&matrix, value);
        }
        length >>= 1;
        if length != 0 {
            let old = matrix;
            for bit in 0..32 {
                matrix[bit] = matrix_times(&old, old[bit]);
            }
        }
    }
    value ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }
    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

fn selected_checksums_and_len(
    data: &[u8],
    spans: &[Span],
) -> Result<(Hashes, u64), SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(Crc16::Xmodem);
    let mut len = 0u64;
    for span in spans {
        len += (span.end - span.start) as u64;
        for &byte in &data[span.start..span.end] {
            acc.push(byte);
        }
    }
    Ok((acc.finish().base, len))
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let (hashes, len) = selected_checksums_and_len(data, spans)?;
    Ok(format!(
        "len={len:016x};crc={:08x};adler={:08x}",
        hashes.crc32, hashes.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    let (hashes, len) = selected_checksums_and_len(data, spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let fields = [(4usize, 20usize), (25, 33), (40, 48)];
    let mut values = [0u64; 3];
    for (field, (start, end)) in fields.iter().copied().enumerate() {
        let mut value = 0u64;
        for offset in start..end {
            let digit = match bytes[offset] {
                b'0'..=b'9' => bytes[offset] - b'0',
                b'a'..=b'f' => bytes[offset] - b'a' + 10,
                _ => return Err(VerifyError::Hex { offset }),
            };
            value = (value << 4) | digit as u64;
        }
        values[field] = value;
    }

    if values[0] != len {
        return Err(VerifyError::Length);
    }
    let expected_crc = values[1] as u32;
    if expected_crc != hashes.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: hashes.crc32,
        });
    }
    let expected_adler = values[2] as u32;
    if expected_adler != hashes.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: hashes.adler32,
        });
    }
    Ok(())
}
