#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    text: String,
    pieces: usize,
    gap: usize,
    ended_marker: bool,
    consumed: usize,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn scalar_string(chars: &[char]) -> String {
    chars.iter().collect()
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x.consumed != y.consumed {
            return x.consumed > y.consumed;
        }
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x.text != y.text {
            return x.text < y.text;
        }
    }
    false
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut Vec<Vec<[Option<Candidate>; 2]>>,
) -> Option<Candidate> {
    if wi >= words.len() {
        return Some(Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
        });
    }
    let fi = usize::from(!first);
    if let Some(cached) = &memo[wi][offset][fi] {
        return Some(cached.clone());
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let cap = options.width - indent;
    let mut best: Option<Candidate> = None;

    let mut content = String::new();
    let mut used = 0usize;
    let mut pieces = 0usize;
    let mut consumed = 0usize;

    for widx in wi..words.len() {
        let word = &words[widx];
        let start = if widx == wi { offset } else { 0 };
        let separator_width = usize::from(widx != wi);
        if used + separator_width >= cap {
            break;
        }
        if separator_width != 0 {
            content.push(' ');
            used += 1;
        }
        pieces += 1;

        let available = cap - used;
        let remaining = word.chars.len() - start;

        // Complete this word and end the line here.
        if remaining <= available {
            content.push_str(&scalar_string(&word.chars[start..]));
            let add = remaining;
            let line_used = used + add;
            let line_text = format!("{}{}", " ".repeat(indent), content);
            let line = Line {
                indent,
                text: line_text,
                pieces,
                gap: options.width - indent - line_used,
                ended_marker: false,
                consumed: consumed + add,
            };
            let next = solve(words, widx + 1, 0, false, options, memo);
            if let Some(mut tail) = next {
                let line_cost = (line.gap as u64).pow(2);
                let single_piece_cost = if pieces == 1 { 3 } else { 0 };
                tail.cost += line_cost + single_piece_cost;
                tail.lines.insert(0, line);
                if best.as_ref().is_none_or(|b| better(&tail, b)) {
                    best = Some(tail);
                }
            }
        }

        // End at any marker in the current word that fits.
        for &mark in &word.markers {
            if mark <= start || mark >= word.chars.len() {
                continue;
            }
            let prefix_len = mark - start;
            if prefix_len + 1 > available {
                continue;
            }
            let mut marker_content = content.clone();
            marker_content.push_str(&scalar_string(&word.chars[start..mark]));
            marker_content.push('-');
            let line_used = used + prefix_len + 1;
            let line = Line {
                indent,
                text: format!("{}{}", " ".repeat(indent), marker_content),
                pieces,
                gap: options.width - indent - line_used,
                ended_marker: true,
                consumed: consumed + prefix_len,
            };
            if let Some(mut tail) = solve(words, widx, mark, false, options, memo) {
                tail.cost += (line.gap as u64).pow(2)
                    + options.hyphen_cost as u64
                    + if pieces == 1 { 3 } else { 0 };
                tail.marker_lines += 1;
                tail.lines.insert(0, line);
                if best.as_ref().is_none_or(|b| better(&tail, b)) {
                    best = Some(tail);
                }
            }
        }

        // Raw split is only legal on an otherwise empty line and only if no
        // marker prefix can fit.
        if widx == wi
            && pieces == 1
            && remaining > available
            && !word.markers.iter().any(|&mark| {
                mark > start && mark < word.chars.len() && mark - start + 1 <= available
            })
        {
            let raw = available;
            let raw_text = format!(
                "{}{}",
                " ".repeat(indent),
                scalar_string(&word.chars[start..start + raw])
            );
            let line = Line {
                indent,
                text: raw_text,
                pieces: 1,
                gap: options.width - indent - raw,
                ended_marker: false,
                consumed: raw,
            };
            if let Some(mut tail) = solve(words, wi, start + raw, false, options, memo) {
                tail.cost += (line.gap as u64).pow(2) + (2 * options.width + 1) as u64 + 3;
                tail.lines.insert(0, line);
                if best.as_ref().is_none_or(|b| better(&tail, b)) {
                    best = Some(tail);
                }
            }
        }

        if remaining > available {
            break;
        }
        content.push_str(&scalar_string(&word.chars[start..]));
        used += remaining;
        consumed += remaining;
    }

    if let Some(answer) = &best {
        memo[wi][offset][fi] = Some(answer.clone());
    }
    best
}

fn wrap_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut memo: Vec<Vec<[Option<Candidate>; 2]>> = words
        .iter()
        .map(|w| vec![[None, None]; w.chars.len() + 1])
        .collect();
    let candidate = solve(words, 0, 0, true, options, &mut memo).unwrap();

    let mut result = Vec::with_capacity(candidate.lines.len());
    for (line_index, line) in candidate.lines.iter().enumerate() {
        let is_final = line_index + 1 == candidate.lines.len();
        if !options.justify || is_final || line.pieces < 2 {
            result.push(line.text.clone());
            continue;
        }

        let gaps = line.pieces - 1;
        let extra = line.gap;
        let base = extra / gaps;
        let remainder = extra % gaps;
        let mut out = String::new();
        let mut chars = line.text.chars().peekable();
        let mut seen_indent = 0usize;
        while seen_indent < line.indent {
            out.push(chars.next().unwrap());
            seen_indent += 1;
        }
        let mut gap_index = 0usize;
        while let Some(ch) = chars.next() {
            if ch == ' ' && gap_index < gaps {
                out.push(' ');
                let cyclic_start = line_index % gaps;
                let distance = (gap_index + gaps - cyclic_start) % gaps;
                let bonus = usize::from(distance < remainder);
                out.push_str(&" ".repeat(base + bonus));
                gap_index += 1;
            } else {
                out.push(ch);
            }
        }
        result.push(out);
    }
    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    if original.len() > 512 {
        return Err(WrapError::InputTooLong {
            chars: original.len(),
        });
    }

    for &(index, ch) in &original {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= original.len() {
                return false;
            }
            let ch = original[j as usize].1;
            ch != '~' && !separators(ch)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < original.len() {
        let ch = original[i].1;
        if ch == '\r' {
            normalized.push('\n');
            if i + 1 < original.len() && original[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(ch);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let mut end = p;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - p >= 2 {
                let words = parse_words(&segment);
                output.extend(wrap_paragraph(&words, options));
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(end - p - 1));
                p = end;
                continue;
            }
        }
        segment.push(normalized[p]);
        p += 1;
    }
    let words = parse_words(&segment);
    output.extend(wrap_paragraph(&words, options));
    Ok(output)
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };
    for &ch in segment {
        if separators(ch) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if ch == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(ch);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    words
}
