#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    rendered: String,
    consumed: usize,
    marker_ended: bool,
    raw: bool,
    gap: usize,
    piece_count: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    rendered: Vec<String>,
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.rendered < b.rendered
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn build_line(indent: usize, pieces: &[String], marker: bool) -> String {
    let mut result = " ".repeat(indent);
    for (i, piece) in pieces.iter().enumerate() {
        if i > 0 {
            result.push(' ');
        }
        result.push_str(piece);
    }
    if marker {
        result.push('-');
    }
    result
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    use std::collections::HashMap;

    type State = (usize, usize, bool);
    let mut memo: HashMap<State, Option<Solution>> = HashMap::new();

    fn recurse(
        state: State,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut HashMap<State, Option<Solution>>,
    ) -> Option<Solution> {
        if state.0 == words.len() {
            return Some(Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                rendered: Vec::new(),
            });
        }
        if let Some(value) = memo.get(&state) {
            return value.clone();
        }

        let (word_index, offset, first_line) = state;
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let mut best: Option<Solution> = None;

        let mut consider = |pieces: Vec<String>,
                            next_state: State,
                            consumed: usize,
                            marker: bool,
                            raw: bool| {
            let Some(tail) = recurse(next_state, words, options, memo) else {
                return;
            };
            let rendered = build_line(indent, &pieces, marker);
            let displayed = scalar_len(&rendered);
            let gap = options.width - displayed;
            let nonfinal = !tail.lines.is_empty();
            let mut cost = (gap as u64) * (gap as u64);
            if !nonfinal {
                cost /= 4;
            }
            if marker {
                cost += options.hyphen_cost as u64;
            }
            if raw {
                cost += (2 * options.width + 1) as u64;
            }
            if nonfinal && pieces.len() == 1 {
                cost += 3;
            }

            let mut candidate = Solution {
                cost: cost + tail.cost,
                marker_lines: usize::from(marker) + tail.marker_lines,
                lines: Vec::with_capacity(1 + tail.lines.len()),
                consumed: Vec::with_capacity(1 + tail.consumed.len()),
                rendered: Vec::with_capacity(1 + tail.rendered.len()),
            };
            candidate.lines.push(Line {
                indent,
                pieces,
                rendered: rendered.clone(),
                consumed,
                marker_ended: marker,
                raw,
                gap,
                piece_count: 0,
            });
            candidate.lines.extend(tail.lines);
            candidate.consumed.push(consumed);
            candidate.consumed.extend(tail.consumed);
            candidate.rendered.push(rendered);
            candidate.rendered.extend(tail.rendered);

            if best.as_ref().map_or(true, |current| better(&candidate, current)) {
                best = Some(candidate);
            }
        };

        let word = &words[word_index];
        let remaining = word.chars.len() - offset;

        // End this line at any fitting marker within the current word.
        for &marker_pos in &word.markers {
            if marker_pos > offset && marker_pos < word.chars.len() {
                let prefix_len = marker_pos - offset;
                if prefix_len + 1 <= capacity {
                    let prefix: String = word.chars[offset..marker_pos].iter().collect();
                    consider(
                        vec![prefix],
                        (word_index, marker_pos, false),
                        prefix_len,
                        true,
                        false,
                    );
                }
            }
        }

        // Raw split is allowed only on an empty line when no marker split fits.
        if remaining > capacity {
            let any_marker_fits = word.markers.iter().any(|&marker_pos| {
                marker_pos > offset
                    && marker_pos < word.chars.len()
                    && marker_pos - offset + 1 <= capacity
            });
            if !any_marker_fits {
                let prefix: String = word.chars[offset..offset + capacity].iter().collect();
                let next = if offset + capacity == word.chars.len() {
                    (word_index + 1, 0, false)
                } else {
                    (word_index, offset + capacity, false)
                };
                consider(vec![prefix], next, capacity, false, true);
            }
        }

        // Enumerate lines ending after a completed word, possibly with more words.
        let mut pieces = Vec::new();
        let mut used = 0usize;
        for wi in word_index..words.len() {
            if wi == word_index {
                let suffix: String = words[wi].chars[offset..].iter().collect();
                let n = words[wi].chars.len() - offset;
                if n > capacity {
                    break;
                }
                used = n;
                pieces.push(suffix);
            } else {
                let n = words[wi].chars.len();
                if used + 1 + n > capacity {
                    break;
                }
                used += 1 + n;
                pieces.push(words[wi].chars.iter().collect());
            }

            consider(
                pieces.clone(),
                (wi + 1, 0, false),
                used,
                false,
                false,
            );
        }

        memo.insert(state, best.clone());
        best
    }

    recurse((0, 0, true), words, options, &mut memo)
        .map(|solution| solution.lines)
        .unwrap_or_default()
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };

    for &ch in segment {
        if matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}') {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if ch == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(ch);
        }
    }

    if !current.chars.is_empty() {
        words.push(current);
    }
    words
}

fn justify_line(line: &Line, line_index: usize, width: usize) -> String {
    if line.piece_count < 2 {
        return line.rendered.clone();
    }

    let gaps = line.piece_count - 1;
    let extra = width.saturating_sub(scalar_len(&line.rendered));
    let each = extra / gaps;
    let remainder = extra % gaps;
    let start = line_index % gaps;

    let mut result = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            let gap_index = i - 1;
            let bonus = usize::from((gap_index + gaps - start) % gaps < remainder);
            result.push_str(&" ".repeat(1 + each + bonus));
        }
        result.push_str(piece);
    }
    if line.marker_ended {
        result.push('-');
    }
    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, ch) in &chars {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let is_separator = |ch: char| matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}');
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_side = |j: isize| {
                j >= 0
                    && (j as usize) < chars.len()
                    && chars[j as usize].1 != '~'
                    && !is_separator(chars[j as usize].1)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker {
                    index: chars[i].0,
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let ch = chars[i].1;
        if ch == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(ch);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut pos = 0;
    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let start = pos;
            while pos < normalized.len() && normalized[pos] == '\n' {
                pos += 1;
            }
            let run = pos - start;
            if run >= 2 {
                let words = parse_segment(&segment);
                let lines = solve_paragraph(&words, options);
                for (line_index, line) in lines.iter().enumerate() {
                    output.push(if options.justify && line_index + 1 < lines.len() && line.piece_count >= 2 {
                        justify_line(line, line_index, options.width)
                    } else {
                        line.rendered.clone()
                    });
                }
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(run - 1));
                continue;
            }
            segment.push('\n');
        } else {
            segment.push(normalized[pos]);
        }
        pos += 1;
    }

    let words = parse_segment(&segment);
    let lines = solve_paragraph(&words, options);
    for (line_index, line) in lines.iter().enumerate() {
        output.push(if options.justify && line_index + 1 < lines.len() && line.piece_count >= 2 {
            justify_line(line, line_index, options.width)
        } else {
            line.rendered.clone()
        });
    }

    Ok(output)
}
