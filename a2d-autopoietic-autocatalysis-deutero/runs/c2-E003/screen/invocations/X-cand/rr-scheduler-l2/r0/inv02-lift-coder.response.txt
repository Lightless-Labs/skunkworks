#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&i| (jobs[i].arrival, i));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completions: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut event_pos = 0;
    let mut time = 0u64;
    let mut last_id: Option<u32> = None;

    fn admit(
        jobs: &[Job],
        events: &[usize],
        event_pos: &mut usize,
        time: u64,
        remaining: &mut [u32],
        completions: &mut [Option<u64>],
        ready: &mut std::collections::VecDeque<usize>,
    ) {
        while *event_pos < events.len() {
            let index = events[*event_pos];
            let arrival = jobs[index].arrival as u64;
            if arrival > time {
                break;
            }
            *event_pos += 1;
            if remaining[index] == 0 {
                completions[index] = Some(arrival);
            } else {
                ready.push_back(index);
            }
        }
    }

    loop {
        admit(
            jobs,
            &events,
            &mut event_pos,
            time,
            &mut remaining,
            &mut completions,
            &mut ready,
        );

        if let Some(index) = ready.pop_front() {
            let id = jobs[index].id;
            if last_id.is_some_and(|last| last != id) {
                time += switch_cost as u64;
            }

            admit(
                jobs,
                &events,
                &mut event_pos,
                time,
                &mut remaining,
                &mut completions,
                &mut ready,
            );

            let start = time;
            let consumed = remaining[index].min(quantum);
            time += consumed as u64;
            remaining[index] -= consumed;
            let finished = remaining[index] == 0;

            runs.push(Run {
                id,
                start,
                end: time,
                finished,
            });

            admit(
                jobs,
                &events,
                &mut event_pos,
                time,
                &mut remaining,
                &mut completions,
                &mut ready,
            );

            if finished {
                completions[index] = Some(time);
            } else {
                ready.push_back(index);
            }
            last_id = Some(id);
        } else if event_pos < events.len() {
            let next_time = jobs[events[event_pos]].arrival as u64;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_id = None;
            }
        } else {
            break;
        }
    }

    let completion: Vec<(u32, u64)> = jobs
        .iter()
        .zip(completions)
        .map(|(job, completed_at)| (job.id, completed_at.expect("every job completes")))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
