#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

const ADLER_MOD: u32 = 65_521;

fn crc32_update(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= u32::from(byte);
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + u32::from(byte)) % ADLER_MOD;
        *b = (*b + *a) % ADLER_MOD;
    }
}

fn finalized_hashes(crc: u32, a: u32, b: u32) -> Hashes {
    Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_update(0xffff_ffff, data) ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;

    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }

    Ok(finalized_hashes(crc, a, b))
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        finalized_hashes(self.crc, self.a, self.b)
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;

        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }

        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = crc32_update(self.crc, data);
        adler_update(&mut self.a, &mut self.b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;

        for span in spans {
            self.update(&data[span.start..span.end]);
        }

        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        let mut adler = 0u32;

        for offset in 1..9 {
            let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            crc = (crc << 4) | u32::from(digit);
        }
        for offset in 11..19 {
            let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            adler = (adler << 4) | u32::from(digit);
        }

        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;

    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }

    ((b as u16) << 8) | a as u16
}

fn fletcher32_update(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + u32::from(word)) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);

    for pair in &mut chunks {
        let word = u16::from_be_bytes([pair[0], pair[1]]);
        fletcher32_update(&mut a, &mut b, word);
    }

    if let [last] = chunks.remainder() {
        fletcher32_update(&mut a, &mut b, u16::from(*last) << 8);
    }

    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending_high = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                fletcher32_update(&mut a, &mut b, word);
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_update(&mut a, &mut b, u16::from(high) << 8);
    }

    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc = match variant {
                Crc16::Xmodem => 0u16,
                Crc16::False => 0xffffu16,
                Crc16::X25 => unreachable!(),
            };

            for &byte in data {
                crc ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc = 0xffffu16;
            for &byte in data {
                crc ^= u16::from(byte);
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
            crc ^ 0xffff
        }
    }
}

struct MultiState {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    fletcher16_a: u32,
    fletcher16_b: u32,
    fletcher32_a: u32,
    fletcher32_b: u32,
    fletcher32_high: Option<u8>,
    crc16: u16,
    variant: Crc16,
}

impl MultiState {
    fn new(variant: Crc16) -> Self {
        let crc16 = match variant {
            Crc16::Xmodem => 0,
            Crc16::False | Crc16::X25 => 0xffff,
        };

        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            fletcher16_a: 0,
            fletcher16_b: 0,
            fletcher32_a: 0,
            fletcher32_b: 0,
            fletcher32_high: None,
            crc16,
            variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        self.crc = crc32_update(self.crc, data);
        adler_update(&mut self.adler_a, &mut self.adler_b, data);

        for &byte in data {
            self.fletcher16_a = (self.fletcher16_a + u32::from(byte)) % 255;
            self.fletcher16_b = (self.fletcher16_b + self.fletcher16_a) % 255;

            if let Some(high) = self.fletcher32_high.take() {
                fletcher32_update(
                    &mut self.fletcher32_a,
                    &mut self.fletcher32_b,
                    (u16::from(high) << 8) | u16::from(byte),
                );
            } else {
                self.fletcher32_high = Some(byte);
            }

            match self.variant {
                Crc16::Xmodem | Crc16::False => {
                    self.crc16 ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        let top = self.crc16 & 0x8000;
                        self.crc16 = self.crc16.wrapping_shl(1);
                        if top != 0 {
                            self.crc16 ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    self.crc16 ^= u16::from(byte);
                    for _ in 0..8 {
                        let low = self.crc16 & 1;
                        self.crc16 >>= 1;
                        if low != 0 {
                            self.crc16 ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    fn finish(mut self) -> Multi {
        if let Some(high) = self.fletcher32_high.take() {
            fletcher32_update(
                &mut self.fletcher32_a,
                &mut self.fletcher32_b,
                u16::from(high) << 8,
            );
        }

        let crc16 = match self.variant {
            Crc16::X25 => self.crc16 ^ 0xffff,
            Crc16::Xmodem | Crc16::False => self.crc16,
        };

        Multi {
            base: finalized_hashes(self.crc, self.adler_a, self.adler_b),
            fletcher16: ((self.fletcher16_b as u16) << 8) | self.fletcher16_a as u16,
            fletcher32: (self.fletcher32_b << 16) | self.fletcher32_a,
            crc16,
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut state = MultiState::new(variant);
    for span in spans {
        state.update(&data[span.start..span.end]);
    }

    Ok(state.finish())
}
