#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized = input.replace("\r\n", "\n").replace('\r', "\n");
    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut index = 0;

    while index < chars.len() {
        if chars[index] == '\n' {
            let run_start = index;
            while index < chars.len() && chars[index] == '\n' {
                index += 1;
            }
            if index - run_start >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                for _ in 1..(index - run_start) {
                    output.push(String::new());
                }
                segment_start = index;
            }
        } else {
            index += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(chars: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in chars {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }
    if words.is_empty() {
        return;
    }

    let mut line_number = 0usize;
    let mut line = String::new();
    let mut content_length = 0usize;
    let mut word_index = 0usize;
    let mut suffix = words[0].clone();

    while word_index < words.len() {
        let indent = if line_number == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;

        if suffix.len() <= capacity {
            if line.is_empty() {
                line.extend(std::iter::repeat(' ').take(indent));
                line.extend(suffix.iter());
                content_length = suffix.len();
                word_index += 1;
                if word_index < words.len() {
                    suffix = words[word_index].clone();
                }
            } else if content_length + 1 + suffix.len() <= capacity {
                line.push(' ');
                line.extend(suffix.iter());
                content_length += 1 + suffix.len();
                word_index += 1;
                if word_index < words.len() {
                    suffix = words[word_index].clone();
                }
            } else {
                output.push(std::mem::take(&mut line));
                line_number += 1;
                content_length = 0;
            }
        } else if line.is_empty() {
            line.extend(std::iter::repeat(' ').take(indent));
            line.extend(suffix.iter().take(capacity));
            suffix.drain(..capacity);
            content_length = capacity;
            output.push(std::mem::take(&mut line));
            line_number += 1;
            content_length = 0;
        } else {
            let available = capacity.saturating_sub(content_length + 1);
            if available >= 2 {
                line.push(' ');
                line.extend(suffix.iter().take(available));
                suffix.drain(..available);
                output.push(std::mem::take(&mut line));
                line_number += 1;
                content_length = 0;
            } else {
                output.push(std::mem::take(&mut line));
                line_number += 1;
                content_length = 0;
            }
        }
    }

    if !line.is_empty() {
        output.push(line);
    }
}
