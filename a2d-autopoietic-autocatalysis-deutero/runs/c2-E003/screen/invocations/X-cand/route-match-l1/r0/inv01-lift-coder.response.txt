#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Parameter(String),
}

#[derive(Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Parameter(_) => 1,
            })
            .collect();

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
            .unwrap_or(path.len());
        let segments = parse_path(&path[..head_end], false)?;

        let mut best: Option<(usize, &Route)> = None;
        for (id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != segments.len() {
                continue;
            }
            if !route_matches(route, &segments) {
                continue;
            }

            let replace = match best {
                None => true,
                Some((_, previous)) => route.rank > previous.rank,
            };
            if replace {
                best = Some((id, route));
            }
        }

        let (route_id, route) = best.ok_or(Error::NotFound)?;
        let mut params = Vec::new();
        for (pattern_segment, value) in route.segments.iter().zip(segments.iter()) {
            if let Segment::Parameter(name) = pattern_segment {
                params.push((name.clone(), value.clone()));
            }
        }

        Ok(Match {
            route: route_id,
            params,
        })
    }
}

fn route_matches(route: &Route, values: &[String]) -> bool {
    route
        .segments
        .iter()
        .zip(values)
        .all(|(segment, value)| match segment {
            Segment::Static(expected) => expected == value,
            Segment::Parameter(_) => true,
        })
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    if pattern == "/" {
        return Ok(Vec::new());
    }

    let (parts, trailing) = split_non_root(pattern);
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (part, offset) in parts {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: offset });
        }

        if part.starts_with(':') {
            let name = &part[1..];
            if !valid_capture_name(name) || !names.insert(name) {
                return Err(Error::InvalidPattern { at: offset });
            }
            segments.push(Segment::Parameter(name.to_owned()));
        } else if part.starts_with('*') {
            return Err(Error::InvalidPattern { at: offset });
        } else {
            let decoded = decode_static(part, offset)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    if trailing {
        return Err(Error::InvalidPattern {
            at: pattern.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(path: &str, _unused: bool) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    if path == "/" {
        return Ok(Vec::new());
    }

    let (parts, trailing) = split_non_root(path);
    let mut segments = Vec::with_capacity(parts.len());

    for (part, offset) in parts {
        if part.is_empty() {
            return Err(Error::InvalidPath { at: offset });
        }
        let decoded =
            decode_static(part, offset).map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);
    }

    if trailing {
        return Err(Error::InvalidPath { at: path.len() - 1 });
    }

    Ok(segments)
}

fn split_non_root(input: &str) -> (Vec<(&str, usize)>, bool) {
    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };

    let mut parts = Vec::new();
    let mut start = 1;
    for index in 1..end {
        if bytes[index] == b'/' {
            parts.push((&input[start..index], start));
            start = index + 1;
        }
    }
    parts.push((&input[start..end], start));

    (parts, trailing)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn decode_static(segment: &str, offset: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' {
            if index + 2 >= bytes.len() {
                return Err(offset + index);
            }
            let high = hex_upper(bytes[index + 1]).ok_or(offset + index)?;
            let low = hex_upper(bytes[index + 2]).ok_or(offset + index)?;
            let value = (high << 4) | low;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + index);
            }

            decoded.push(value);
            index += 3;
        } else {
            if !byte.is_ascii() || !is_unreserved(byte) {
                return Err(offset + index);
            }
            decoded.push(byte);
            index += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}
