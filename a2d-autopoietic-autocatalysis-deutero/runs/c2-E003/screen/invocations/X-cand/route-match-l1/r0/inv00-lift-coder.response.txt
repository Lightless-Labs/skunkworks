```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { id, segments });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if route.segments.len() != request_segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matched = true;

            for (pattern_segment, request_segment) in
                route.segments.iter().zip(request_segments.iter())
            {
                match pattern_segment {
                    Segment::Static(value) if value == request_segment => {}
                    Segment::Param(name) => {
                        params.push((name.clone(), request_segment.clone()));
                    }
                    _ => {
                        matched = false;
                        break;
                    }
                }
            }

            if matched {
                match best {
                    None => best = Some((route, params)),
                    Some((current, _)) => {
                        if route_is_more_specific(route, current) {
                            best = Some((route, params));
                        }
                    }
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn route_is_more_specific(candidate: &Route, current: &Route) -> bool {
    for (a, b) in candidate.segments.iter().zip(current.segments.iter()) {
        let a_rank = segment_rank(a);
        let b_rank = segment_rank(b);
        if a_rank != b_rank {
            return a_rank > b_rank;
        }
    }
    false
}

fn segment_rank(segment: &Segment) -> u8 {
    match segment {
        Segment::Static(_) => 2,
        Segment::Param(_) => 1,
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut raw_segments = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(Error::InvalidPattern { at: i });
            }
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            raw_segments.push((start, i));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }
    raw_segments.push((start, bytes.len()));

    let mut seen = std::collections::HashSet::new();
    let mut result = Vec::with_capacity(raw_segments.len());

    for (start, end) in raw_segments {
        let segment = &input[start..end];
        if segment.starts_with(':') {
            let name = &segment[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(Segment::Param(name.to_string()));
        } else if segment.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            result.push(Segment::Static(decode_segment(segment, start, true)?));
        }
    }

    Ok(result)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut raw_segments = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(Error::InvalidPath { at: i });
            }
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            raw_segments.push((start, i));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }
    raw_segments.push((start, bytes.len()));

    let mut result = Vec::with_capacity(raw_segments.len());
    for (start, end) in raw_segments {
        result.push(decode_segment(&input[start..end], start, false)?);
    }
    Ok(result)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(
    segment: &str,
    offset: usize,
    pattern: bool,
) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid_at(pattern, offset + i));
            }
            let high = hex_value(bytes[i + 1]).ok_or_else(|| invalid_at(pattern, offset + i))?;
            let low = hex_value(bytes[i + 2]).ok_or_else(|| invalid_at(pattern, offset + i))?;
            let value = (high << 4) | low;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid_at(pattern, offset + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid_at(pattern, offset + i));
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn invalid_at(pattern: bool, at: usize) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}
```
