use std::cmp::Reverse;
use std::collections::{BTreeMap, BTreeSet, BinaryHeap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EdgeStatus {
    Duplicate,
    Internal,
    Link,
    Merge,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Step {
    pub edge_index: usize,
    pub status: EdgeStatus,
    pub before: (usize, usize),
    pub after: (usize, usize),
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub edge_index: usize,
    pub before: Vec<usize>,
    pub after: usize,
    pub members: Vec<usize>,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Batch {
    pub steps: Vec<Step>,
    pub merges: Vec<Merge>,
    pub graph: Graph,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn validate(n: usize, edges: &[(usize, usize)]) -> Result<(), GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }
    Ok(())
}

fn bfs_route(adj: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let mut parent = vec![usize::MAX; adj.len()];
    let mut seen = vec![false; adj.len()];
    let mut queue = VecDeque::new();
    seen[start] = true;
    queue.push_back(start);

    while let Some(u) = queue.pop_front() {
        if u == goal {
            break;
        }
        let mut neighbors = adj[u].clone();
        neighbors.sort_unstable();
        neighbors.dedup();
        for v in neighbors {
            if !seen[v] {
                seen[v] = true;
                parent[v] = u;
                queue.push_back(v);
            }
        }
    }

    if !seen[goal] {
        return None;
    }
    let mut route = vec![goal];
    let mut cur = goal;
    while cur != start {
        cur = parent[cur];
        route.push(cur);
    }
    route.reverse();
    Some(route)
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;

    let mut adj = vec![Vec::<usize>::new(); n];
    let mut rev = vec![Vec::<usize>::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    // Iterative Kosaraju avoids recursion depth limits on long paths.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((u, next)) = stack.last_mut() {
            if *next < adj[*u].len() {
                let v = adj[*u][*next];
                *next += 1;
                if !seen[v] {
                    seen[v] = true;
                    stack.push((v, 0));
                }
            } else {
                finish.push(*u);
                stack.pop();
            }
        }
    }

    let mut raw_id = vec![usize::MAX; n];
    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    for &root in finish.iter().rev() {
        if raw_id[root] != usize::MAX {
            continue;
        }
        let id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![root];
        raw_id[root] = id;
        while let Some(u) = stack.pop() {
            members.push(u);
            for &v in &rev[u] {
                if raw_id[v] == usize::MAX {
                    raw_id[v] = id;
                    stack.push(v);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    let mut order: Vec<usize> = (0..raw_components.len()).collect();
    order.sort_by(|&a, &b| {
        raw_components[b]
            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
    });
    let mut raw_to_id = vec![0; order.len()];
    let mut components = Vec::with_capacity(order.len());
    for (id, raw) in order.into_iter().enumerate() {
        raw_to_id[raw] = id;
        components.push(raw_components[raw].clone());
    }
    let component_of: Vec<usize> = raw_id.iter().map(|&id| raw_to_id[id]).collect();
    let c = components.len();

    let mut cyclic: Vec<bool> = components.iter().map(|m| m.len() > 1).collect();
    let mut internal_edges = vec![0usize; c];
    let mut arc_map: BTreeMap<(usize, usize), (usize, usize)> = BTreeMap::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_map.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs: Vec<Arc> = arc_map
        .iter()
        .map(|(&(from, to), &(multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut dag = vec![Vec::new(); c];
    let mut indegree = vec![0usize; c];
    for arc in &arcs {
        dag[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    // Reverse heap ordering selects largest internal count, then largest minimum vertex.
    let mut ready: BinaryHeap<(usize, usize, Reverse<usize>)> = BinaryHeap::new();
    for id in 0..c {
        if indegree[id] == 0 {
            ready.push((
                internal_edges[id],
                components[id][0],
                Reverse(id),
            ));
        }
    }
    let mut topo = Vec::with_capacity(c);
    let mut depth = vec![0usize; c];
    while let Some((_, _, Reverse(u))) = ready.pop() {
        topo.push(u);
        for &v in &dag[u] {
            depth[v] = depth[v].max(depth[u] + 1);
            indegree[v] -= 1;
            if indegree[v] == 0 {
                ready.push((
                    internal_edges[v],
                    components[v][0],
                    Reverse(v),
                ));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut dag = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            dag[arc.from].push(arc.to);
        }
        if let Some(route) = bfs_route(&dag, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = bfs_route(&dag, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    history: Vec<Merge>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        validate(n, edges)?;
        Ok(Self {
            n,
            edges: edges.to_vec(),
            history: Vec::new(),
        })
    }

    pub fn snapshot(&self) -> Graph {
        analyze(self.n, &self.edges).expect("stored graph is valid")
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(from, to)) in edges.iter().enumerate() {
            if from >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: from,
                });
            }
            if to >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: to,
                });
            }
        }

        let mut steps = Vec::with_capacity(edges.len());
        let mut merges = Vec::new();
        for &(from, to) in edges {
            let edge_index = self.edges.len();
            let before_graph = self.snapshot();
            let before = (
                before_graph.component_of[from],
                before_graph.component_of[to],
            );
            let duplicate = self.edges.iter().any(|&(u, v)| u == from && v == to);
            let was_internal = before.0 == before.1;

            self.edges.push((from, to));
            let after_graph = self.snapshot();
            let after = (
                after_graph.component_of[from],
                after_graph.component_of[to],
            );

            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if was_internal {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };

            if status == EdgeStatus::Merge {
                let merged_id = after.0;
                let members = after_graph.components[merged_id].clone();
                let member_set: BTreeSet<usize> = members.iter().copied().collect();
                let before_ids: BTreeSet<usize> = before_graph
                    .components
                    .iter()
                    .enumerate()
                    .filter_map(|(id, component)| {
                        component
                            .iter()
                            .any(|v| member_set.contains(v))
                            .then_some(id)
                    })
                    .collect();
                let event = Merge {
                    edge_index,
                    before: before_ids.into_iter().collect(),
                    after: merged_id,
                    members,
                };
                merges.push(event.clone());
                self.history.push(event);
            }

            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.snapshot(),
        })
    }
}
