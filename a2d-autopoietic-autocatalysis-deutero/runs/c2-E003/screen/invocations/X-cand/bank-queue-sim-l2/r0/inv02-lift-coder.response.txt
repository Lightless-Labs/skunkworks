#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    open: bool,
    idle_since: u64,
    finish: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            open: false,
            idle_since: server.opens,
            finish: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut arrival_pos = 0;
    let mut waiting: Vec<usize> = Vec::new();

    loop {
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            next_time = Some(next_time.map_or(time, |current| current.min(time)));
        };

        for state in &states {
            if let Some(finish) = state.finish {
                consider(finish);
            }
            if !state.open {
                consider(state.opens);
            }
        }
        if arrival_pos < arrivals.len() {
            consider(customers[arrivals[arrival_pos]].arrival);
        }

        let Some(time) = next_time else {
            break;
        };

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| (state.finish == Some(time)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| states[i].id);
        for i in finishing {
            let customer = match events.iter().rev().find_map(|event| match event {
                Event::Start {
                    time: start_time,
                    customer,
                    server,
                    finish,
                } if *server == states[i].id && *finish == time && *start_time <= time => {
                    Some(*customer)
                }
                _ => None,
            }) {
                Some(customer) => customer,
                None => unreachable!(),
            };
            events.push(Event::Finish {
                time,
                customer,
                server: states[i].id,
            });
            states[i].finish = None;
            states[i].idle_since = time;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| (!state.open && state.opens == time).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for i in opening {
            states[i].open = true;
            events.push(Event::Open {
                time,
                server: states[i].id,
            });
        }

        while arrival_pos < arrivals.len()
            && customers[arrivals[arrival_pos]].arrival == time
        {
            let customer_index = arrivals[arrival_pos];
            events.push(Event::Arrive {
                time,
                customer: customers[customer_index].id,
            });
            waiting.push(customer_index);
            arrival_pos += 1;
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.open && state.finish.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };
            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index];

            let finish = time
                .checked_add(customer.service)
                .and_then(|value| value.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
            } else {
                states[server_index].finish = Some(finish);
            }
        }
    }

    Ok(events)
}
