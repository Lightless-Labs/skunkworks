```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    idle_since: u64,
    open: bool,
    occupied_until: Option<(u64, u32)>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<&Customer> = customers.iter().collect();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            idle_since: server.opens,
            open: false,
            occupied_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut next_customer = 0;
    let mut finished_customers = 0;

    loop {
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if next_time.map_or(true, |current| time < current) {
                next_time = Some(time);
            }
        };

        if next_customer < ordered_customers.len() {
            consider(ordered_customers[next_customer].arrival);
        }
        for state in &states {
            if !state.open {
                consider(state.opens);
            }
            if let Some((finish, _)) = state.occupied_until {
                consider(finish);
            }
        }

        let Some(time) = next_time else {
            break;
        };

        let mut completions: Vec<(usize, u32, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                state.occupied_until.and_then(|(finish, customer)| {
                    (finish == time).then_some((index, state.id, customer))
                })
            })
            .collect();
        completions.sort_by_key(|entry| entry.1);
        for (index, server, customer) in completions {
            events.push(Event::Finish {
                time,
                customer,
                server,
            });
            states[index].occupied_until = None;
            states[index].idle_since = time;
            finished_customers += 1;
        }

        let mut opening_indices: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.open && state.opens == time).then_some(index)
            })
            .collect();
        opening_indices.sort_by_key(|&index| std::cmp::Reverse(states[index].id));
        for index in opening_indices {
            states[index].open = true;
            events.push(Event::Open {
                time,
                server: states[index].id,
            });
        }

        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == time
        {
            let customer = ordered_customers[next_customer];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer);
            next_customer += 1;
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.open && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting.remove(0);
            let server_id = states[server_index].id;
            let Some(after_service) = time.checked_add(customer.service) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                });
            };
            let Some(finish) = after_service.checked_add(states[server_index].overhead) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                });
            };

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
                finished_customers += 1;
            } else {
                states[server_index].occupied_until = Some((finish, customer.id));
            }
        }

        if next_customer == ordered_customers.len()
            && waiting.is_empty()
            && finished_customers == customers.len()
            && states.iter().all(|state| state.open)
        {
            break;
        }
    }

    Ok(events)
}
```
