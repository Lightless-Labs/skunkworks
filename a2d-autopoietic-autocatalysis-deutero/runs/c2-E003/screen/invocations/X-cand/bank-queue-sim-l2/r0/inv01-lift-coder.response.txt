```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy: Option<(u64, u32)>, // finish time and customer ID
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            busy: None,
            idle_since: server.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0;
    let mut now = 0u64;

    loop {
        // Finish events precede all other events at this tick.
        let mut finished: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                state
                    .busy
                    .filter(|(finish, _)| *finish == now)
                    .map(|(_, customer)| (state.id, index, customer))
            })
            .collect();
        finished.sort_by_key(|&(server_id, _, _)| server_id);

        for (_, index, customer) in finished {
            let state = &mut states[index];
            state.busy = None;
            state.idle_since = now;
            events.push(Event::Finish {
                time: now,
                customer,
                server: state.id,
            });
        }

        // Open events are emitted in descending server ID order.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.opens == now).then_some(index)
            })
            .collect();
        opening.sort_by_key(|&index| std::cmp::Reverse(states[index].id));

        for index in opening {
            let state = &mut states[index];
            state.opened = true;
            events.push(Event::Open {
                time: now,
                server: state.id,
            });
        }

        // Arrivals are emitted in descending customer ID order.
        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == now
        {
            let mut end = next_customer + 1;
            while end < customer_order.len()
                && customers[customer_order[end]].arrival == now
            {
                end += 1;
            }

            for &customer_index in &customer_order[next_customer..end] {
                let customer = &customers[customer_index];
                events.push(Event::Arrive {
                    time: now,
                    customer: customer.id,
                });
                waiting.push(customer_index);
            }
            next_customer = end;
        }

        // Dispatch in FIFO order, recalculating the best server after each start.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting[0];
            let customer = &customers[customer_index];
            let server = &states[server_index];

            let Some(after_service) = now.checked_add(customer.service) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };
            let Some(finish) = after_service.checked_add(server.overhead) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };

            waiting.remove(0);
            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy = Some((finish, customer.id));
            }
        }

        let mut next_time: Option<u64> = None;

        if next_customer < customer_order.len() {
            next_time = Some(customers[customer_order[next_customer]].arrival);
        }
        for state in &states {
            if !state.opened {
                next_time = Some(next_time.map_or(state.opens, |time| time.min(state.opens)));
            }
            if let Some((finish, _)) = state.busy {
                next_time = Some(next_time.map_or(finish, |time| time.min(finish)));
            }
        }

        match next_time {
            Some(time) if time > now => now = time,
            Some(_) => unreachable!("all events at this tick were processed"),
            None => break,
        }
    }

    Ok(events)
}
```
