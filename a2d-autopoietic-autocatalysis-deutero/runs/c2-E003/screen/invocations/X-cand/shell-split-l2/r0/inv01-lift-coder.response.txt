#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn find_braced_end(input: &str, mut i: usize, dollar: usize) -> Result<usize, SplitError> {
    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < input.len() {
                i += input[i..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            return Ok(i);
        } else {
            i += c.len_utf8();
        }
    }
    Err(SplitError::UnterminatedExpansion { pos: dollar })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let end = find_braced_end(input, content_start, dollar)?;
    let content = &input[content_start..end];

    let name_end = content
        .char_indices()
        .find_map(|(i, c)| if !is_name_continue(c) { Some(i) } else { None })
        .unwrap_or(content.len());

    if name_end == 0 || !is_name_start(content.chars().next().unwrap()) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &content[..name_end];
    let rest = &content[name_end..];
    let value = if rest.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default_text) = rest.strip_prefix(":-") {
        let found = lookup(name, env).unwrap_or("");
        if found.is_empty() {
            decode_default(default_text)
        } else {
            found.to_string()
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((end + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i += n;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '"' || next == '\\' || next == '$' || next == '\n' {
                        if next != '\n' {
                            word.push(next);
                        }
                        i += next.len_utf8();
                    } else {
                        word.push('\\');
                    }
                } else if c == '$' {
                    if i + 1 < input.len() {
                        let next = input[i + 1..].chars().next().unwrap();
                        if next == '{' {
                            let (after, value) = parse_braced(input, i, env)?;
                            word.push_str(&value);
                            active = true;
                            i = after;
                        } else if is_name_start(next) {
                            let start = i + 1;
                            let mut end = start;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            word.push_str(lookup(&input[start..end], env).unwrap_or(""));
                            active = true;
                            i = end;
                        } else {
                            word.push('$');
                            i += n;
                        }
                    } else {
                        word.push('$');
                        i += n;
                    }
                } else {
                    word.push(c);
                    i += n;
                }
                let _ = open;
            }
            Quote::None => {
                if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i += n;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        word.push(next);
                        active = true;
                    }
                    i += next.len_utf8();
                } else if c == '$' {
                    if i + 1 < input.len() {
                        let next = input[i + 1..].chars().next().unwrap();
                        if next == '{' {
                            let (after, value) = parse_braced(input, i, env)?;
                            word.push_str(&value);
                            active = true;
                            i = after;
                        } else if is_name_start(next) {
                            let start = i + 1;
                            let mut end = start;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            word.push_str(lookup(&input[start..end], env).unwrap_or(""));
                            active = true;
                            i = end;
                        } else {
                            word.push('$');
                            active = true;
                            i += n;
                        }
                    } else {
                        word.push('$');
                        active = true;
                        i += n;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
