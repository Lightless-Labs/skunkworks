#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        let next_i = i + ch.len_utf8();

        match quote {
            Quote::Single(open) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                i = next_i;
                let _ = open;
            }
            Quote::Double(open) => {
                match ch {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i >= input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let next = input[next_i..].chars().next().unwrap();
                        let after_next = next_i + next.len_utf8();
                        if matches!(next, '"' | '\\' | '$') {
                            word.push(next);
                            i = after_next;
                        } else if next == '\n' {
                            i = after_next;
                        } else {
                            word.push('\\');
                            i = next_i;
                        }
                    }
                    '$' => {
                        let (consumed, value) = expand(input, i, env)?;
                        if let Some((end, value)) = consumed.zip(value) {
                            active = true;
                            word.push_str(&value);
                            i = end;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    }
                    _ => {
                        word.push(ch);
                        i = next_i;
                    }
                }
                let _ = open;
            }
            Quote::None => {
                match ch {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(std::mem::take(&mut word));
                            active = false;
                        }
                        i = next_i;
                    }
                    '\'' => {
                        active = true;
                        quote = Quote::Single(i);
                        i = next_i;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double(i);
                        i = next_i;
                    }
                    '\\' => {
                        if next_i >= input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let next = input[next_i..].chars().next().unwrap();
                        let after_next = next_i + next.len_utf8();
                        if next != '\n' {
                            active = true;
                            word.push(next);
                        }
                        i = after_next;
                    }
                    '$' => {
                        let (consumed, value) = expand(input, i, env)?;
                        if let Some((end, value)) = consumed.zip(value) {
                            active = true;
                            word.push_str(&value);
                            i = end;
                        } else {
                            active = true;
                            word.push('$');
                            i = next_i;
                        }
                    }
                    _ => {
                        active = true;
                        word.push(ch);
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(Option<usize>, Option<String>), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((None, None));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;
        while cursor < input.len() {
            let ch = input[cursor..].chars().next().unwrap();
            let end = cursor + ch.len_utf8();
            if ch == '\\' {
                if end < input.len() {
                    let escaped = input[end..].chars().next().unwrap();
                    cursor = end + escaped.len_utf8();
                } else {
                    cursor = end;
                }
            } else if ch == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor = end;
            }
        }
        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];
        let (name, default) = if let Some(name) = contents.strip_suffix("") {
            let _ = name;
            if let Some(split_at) = contents.find(":-") {
                (&contents[..split_at], Some(&contents[split_at + 2..]))
            } else {
                (contents, None)
            }
        } else {
            unreachable!()
        };

        if !valid_name(name) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        if default.is_none() && contents.contains(':') {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        if let Some(default) = default {
            if name.contains('\\') || name.contains('}') {
                return Err(SplitError::InvalidExpansion { pos: dollar });
            }
            let value = lookup(env, name);
            let result = match value {
                Some(v) if !v.is_empty() => v.to_string(),
                _ => decode_default(default),
            };
            return Ok((Some(close + 1), Some(result)));
        }

        if name.contains('\\') || name.contains(':') {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        return Ok((Some(close + 1), Some(lookup(env, name).unwrap_or("").to_string())));
    }

    if is_name_start(first) {
        let mut end = after_dollar;
        while end < input.len() {
            let ch = input[end..].chars().next().unwrap();
            if is_name_continue(ch) {
                end += ch.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        return Ok((Some(end), Some(lookup(env, name).unwrap_or("").to_string())));
    }

    Ok((None, None))
}

fn valid_name(name: &str) -> bool {
    let mut chars = name.chars();
    match chars.next() {
        Some(ch) if is_name_start(ch) => chars.all(is_name_continue),
        _ => false,
    }
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut output = String::new();
    let mut i = 0;
    while i < text.len() {
        let ch = text[i..].chars().next().unwrap();
        let next = i + ch.len_utf8();
        if ch == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            output.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            output.push(ch);
            i = next;
        }
    }
    output
}
