```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    started: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.started, b.started) {
            (true, false) => std::cmp::Ordering::Less,
            (false, true) => std::cmp::Ordering::Greater,
            (true, true) => a
                .pause_time
                .cmp(&b.pause_time)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (false, false) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn pop_best(waiting: &mut Vec<Waiting>) -> Option<Waiting> {
    let index = waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
        .map(|(index, _)| index)?;
    Some(waiting.remove(index))
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;
    let mut now = arrivals.first().map_or(0, |c| c.arrival);

    while finished_count < customers.len() {
        // A finish at this tick is processed before arrivals.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let done = running.take().unwrap();
            events.push(Event {
                time: now,
                customer: done.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        // Process the entire arrival cohort before making any scheduling choice.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                customer,
                started: false,
                pause_time: 0,
            });
            next_arrival += 1;
        }

        // Reconsider preemption once protection has ended, after arrivals.
        if let Some(active) = running.as_ref() {
            if now >= active.work_at && now < active.finish {
                if let Some(best_index) = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
                    .map(|(index, _)| index)
                {
                    let best_priority = waiting[best_index].customer.priority;
                    if best_priority < active.customer.priority {
                        let by = waiting[best_index].customer.id;
                        let active = running.take().unwrap();
                        let remaining = active.finish - now;
                        events.push(Event {
                            time: now,
                            customer: active.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            customer: active.customer,
                            started: true,
                            pause_time: now,
                        });
                    }
                }
            }
        }

        // Dispatch repeatedly at this tick, including immediate finishes.
        while running.is_none() {
            let Some(mut selected) = pop_best(&mut waiting) else {
                break;
            };
            let switching = last_dispatched.is_some_and(|id| id != selected.customer.id);
            let cost = if switching { switch_cost } else { 0 };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: selected.customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(selected.customer.service).ok_or(
                Error::TimeOverflow {
                    customer: selected.customer.id,
                    time: now,
                },
            )?;

            let was_started = selected.started;
            let remaining = if was_started {
                // Remaining service was recorded at preemption time.
                // Stored in service for resumed customers below.
                selected.customer.service
            } else {
                selected.customer.service
            };

            events.push(Event {
                time: now,
                customer: selected.customer.id,
                kind: if was_started {
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    }
                },
            });
            last_dispatched = Some(selected.customer.id);

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: selected.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
                continue;
            }

            running = Some(Running {
                customer: selected.customer.clone(),
                remaining,
                work_at,
                finish,
            });
        }

        if finished_count >= customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for candidate in [active.work_at, active.finish] {
                if candidate > now && next_time.is_none_or(|t| candidate < t) {
                    next_time = Some(candidate);
                }
            }
        }

        if let Some(time) = next_time {
            now = time;
        } else if running.is_none() && waiting.is_empty() && next_arrival < arrivals.len() {
            now = arrivals[next_arrival].arrival;
        } else {
            break;
        }
    }

    Ok(events)
}
```
