use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn core_missing_byte(b: u8) -> bool {
    matches!(b, b'.' | b'-' | b'+')
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut nums = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if pos == start {
            if pos == bytes.len() || core_missing_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        nums[field] = s[start..pos]
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || core_missing_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let (ids, next) = parse_identifiers(s, pos, true)?;
        pre = ids;
        pos = next;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        let (ids, next) = parse_identifiers(s, pos, false)?;
        build = ids;
        pos = next;
    }
    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    is_pre: bool,
) -> Result<(Vec<String>, usize), VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut pos = start;

    loop {
        let id_start = pos;
        while pos < bytes.len()
            && bytes[pos] != b'.'
            && !(is_pre && bytes[pos] == b'+')
        {
            let b = bytes[pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..pos];
        if is_pre
            && id.len() > 1
            && id.as_bytes()[0].is_ascii_digit()
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        ids.push(id.to_string());

        if pos < bytes.len() && bytes[pos] == b'.' {
            pos += 1;
            continue;
        }
        if is_pre && pos < bytes.len() && bytes[pos] == b'+' {
            break;
        }
        break;
    }

    Ok((ids, pos))
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_ids(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => cmp_numeric_ids(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: Option<(u64, u64, u64)>,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

fn tokenize(s: &str) -> Vec<Token> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !ws(bytes[i]) {
            i += 1;
        }
        out.push(Token {
            text: s[start..i].to_string(),
            start,
        });
    }
    out
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields(Vec<u64>),
}

fn parse_partial(s: &str, start: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parse_version(s)
            .map(Partial::Full)
            .map_err(|error| RangeError::Version { at: start, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f)) {
        return parse_version(s)
            .map(Partial::Full)
            .map_err(|error| RangeError::Version { at: start, error });
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: start });
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if is_wildcard(f) {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: start });
            }
            let n = f
                .parse::<u64>()
                .map_err(|_| RangeError::Syntax { at: start })?;
            nums.push(n);
        }
    }
    Ok(Partial::Fields(nums))
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn version_core(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build: Vec::new(),
    }
}

fn full_anchor(v: &Version) -> Option<(u64, u64, u64)> {
    if v.pre.is_empty() {
        None
    } else {
        Some((v.major, v.minor, v.patch))
    }
}

fn padded(fields: &[u64]) -> Version {
    version_core(
        fields.first().copied().unwrap_or(0),
        fields.get(1).copied().unwrap_or(0),
        fields.get(2).copied().unwrap_or(0),
        Vec::new(),
    )
}

fn increment_endpoint(
    fields: &[u64],
    index: usize,
    start: usize,
) -> Result<Version, RangeError> {
    let mut n = [0u64; 3];
    for (i, v) in fields.iter().enumerate().take(3) {
        n[i] = *v;
    }
    n[index] = n[index]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at: start })?;
    for x in n.iter_mut().skip(index + 1) {
        *x = 0;
    }
    Ok(version_core(n[0], n[1], n[2], vec!["0".to_string()]))
}

fn add_cmp(out: &mut Vec<Comparator>, op: Op, version: Version, anchor: Option<(u64, u64, u64)>) {
    out.push(Comparator { op, version, anchor });
}

fn parse_full_comparator(t: &Token) -> Result<Comparator, RangeError> {
    let (op, off) = if t.text.starts_with("<=") {
        (Op::Le, 2)
    } else if t.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if t.text.starts_with('<') {
        (Op::Lt, 1)
    } else if t.text.starts_with('>') {
        (Op::Gt, 1)
    } else if t.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &t.text[off..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: t.start });
    }
    let v = parse_version(operand).map_err(|error| RangeError::Version {
        at: t.start + off,
        error,
    })?;
    let anchor = full_anchor(&v);
    Ok(Comparator {
        op,
        version: v,
        anchor,
    })
}

fn parse_bare(t: &Token, out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let p = parse_partial(&t.text, t.start)?;
    match p {
        Partial::Full(v) => {
            let anchor = full_anchor(&v);
            add_cmp(out, Op::Eq, v, anchor);
        }
        Partial::Fields(fields) => {
            let k = fields.len();
            if k == 0 {
                return Ok(());
            }
            let low = padded(&fields);
            add_cmp(out, Op::Ge, low, None);
            if k < 3 {
                let upper = increment_endpoint(&fields, k - 1, t.start)?;
                add_cmp(out, Op::Lt, upper, None);
            }
        }
    }
    Ok(())
}

fn parse_caret_tilde(t: &Token, caret: bool, out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let operand = &t.text[1..];
    let operand_start = t.start + 1;
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: t.start });
    }
    let p = parse_partial(operand, operand_start)?;
    match p {
        Partial::Full(v) => {
            let fields = vec![v.major, v.minor, v.patch];
            let low_anchor = full_anchor(&v);
            add_cmp(out, Op::Ge, v, low_anchor);
            let idx = if caret {
                if fields[0] > 0 { 0 } else if fields[1] > 0 { 1 } else { 2 }
            } else {
                1
            };
            let upper = increment_endpoint(&fields, idx, operand_start)?;
            add_cmp(out, Op::Lt, upper, None);
        }
        Partial::Fields(fields) => {
            let k = fields.len();
            if k == 0 {
                return Err(RangeError::Syntax { at: t.start });
            }
            let low = padded(&fields);
            add_cmp(out, Op::Ge, low, None);
            let idx = if caret {
                if fields[0] > 0 || k == 1 {
                    0
                } else if fields.get(1).copied().unwrap_or(0) > 0 || k == 2 {
                    1
                } else {
                    2
                }
            } else if k == 1 {
                0
            } else {
                1
            };
            let upper = increment_endpoint(&fields, idx, operand_start)?;
            add_cmp(out, Op::Lt, upper, None);
        }
    }
    Ok(())
}

fn parse_hyphen(tokens: &[Token], arm_start: usize, out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let minus = tokens.iter().position(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax {
            at: tokens[minus].start,
        });
    }
    let lower = parse_partial(&tokens[0].text, tokens[0].start)?;
    let upper = parse_partial(&tokens[2].text, tokens[2].start)?;

    let lower_fields = match lower {
        Partial::Full(v) => {
            let anchor = full_anchor(&v);
            add_cmp(out, Op::Ge, v.clone(), anchor);
            vec![v.major, v.minor, v.patch]
        }
        Partial::Fields(f) => {
            if f.is_empty() {
                return Err(RangeError::Syntax { at: tokens[0].start });
            }
            let v = padded(&f);
            add_cmp(out, Op::Ge, v, None);
            f
        }
    };
    let _ = lower_fields;

    match upper {
        Partial::Full(v) => {
            let anchor = full_anchor(&v);
            add_cmp(out, Op::Le, v, anchor);
        }
        Partial::Fields(f) => {
            if f.is_empty() {
                return Err(RangeError::Syntax { at: tokens[2].start });
            }
            let v = padded(&f);
            add_cmp(out, Op::Ge, v, None);
            let endpoint = increment_endpoint(&f, f.len() - 1, tokens[2].start)?;
            add_cmp(out, Op::Lt, endpoint, None);
        }
    }
    let _ = arm_start;
    Ok(())
}

fn parse_arm(tokens: &[Token]) -> Result<Vec<Comparator>, RangeError> {
    let mut out = Vec::new();
    if tokens.is_empty() {
        return Ok(out);
    }
    if let Some(minus) = tokens.iter().position(|t| t.text == "-") {
        if tokens.len() != 3 || minus != 1 {
            return Err(RangeError::Syntax {
                at: tokens[minus].start,
            });
        }
        return parse_hyphen(tokens, tokens[0].start, &mut out).map(|_| out);
    }

    for t in tokens {
        let first = t.text.as_bytes().first().copied();
        match first {
            Some(b'^') => parse_caret_tilde(t, true, &mut out)?,
            Some(b'~') => parse_caret_tilde(t, false, &mut out)?,
            Some(b'<') | Some(b'>') | Some(b'=') => {
                out.push(parse_full_comparator(t)?);
            }
            Some(b) if b.is_ascii_digit() || matches!(b, b'*' | b'x' | b'X') => {
                parse_bare(t, &mut out)?;
            }
            _ => return Err(RangeError::Syntax { at: t.start }),
        }
    }
    Ok(out)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    let mut seen_separator = false;

    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty {
                    at: tokens[i].start,
                });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            seen_separator = true;
        } else {
            current.push(tokens[i].clone());
            seen_separator = false;
        }
        i += 1;
    }
    if current.is_empty() {
        if seen_separator {
            return Err(RangeError::Empty { at: s.len() });
        }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admission = (v.major, v.minor, v.patch);
        let admitted = arm.iter().any(|c| c.anchor == Some(admission));
        if !v.pre.is_empty() && !admitted {
            return false;
        }
        arm.iter().all(|c| cmp_matches(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (greatest && ord == Ordering::Greater)
                    || (!greatest && ord == Ordering::Less)
                {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
