use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if start == *pos {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut at = start;
    loop {
        let ident_start = at;
        while at < end && bytes[at] != b'.' {
            if !is_ident_char(bytes[at]) {
                return Err(VersionError::InvalidChar { at });
            }
            at += 1;
        }
        if ident_start == at {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..at];
        if is_pre && ident.bytes().all(is_digit) && ident.len() > 1 && ident.starts_with('0') {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(ident.to_owned());
        if at == end {
            break;
        }
        at += 1;
        if at == end {
            return Err(VersionError::EmptyIdentifier { at });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        let bytes = s.as_bytes();
        if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        if bytes[pos] != b'.' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let n = parse_core(s, &mut pos)?;
        if _ == 0 {
            let minor = n;
            // Parse patch in the next iteration.
            let bytes = s.as_bytes();
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core(s, &mut pos)?;
            return parse_tail(s, pos, major, minor, patch);
        }
        let _ = n;
    }
    unreachable!()
}

fn parse_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let plus = s[pos..].find('+').map(|i| pos + i).unwrap_or(bytes.len());
                pre = parse_identifiers(s, pos, plus, true)?;
                pos = plus;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }
    if pos < bytes.len() {
        if bytes[pos] != b'+' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        build = parse_identifiers(s, pos, bytes.len(), false)?;
        pos = bytes.len();
    }
    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Comparator {
    Compare { op: Op, version: Version, anchor: bool },
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn cmp_matches(v: &Version, op: Op, bound: &Version) -> bool {
    let ord = compare_versions(v, bound);
    match op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn has_anchor(comps: &[Comparator], v: &Version) -> bool {
    comps.iter().any(|c| match c {
        Comparator::Compare { version, anchor, .. } => {
            *anchor && !version.pre.is_empty()
                && (version.major, version.minor, version.patch) == (v.major, v.minor, v.patch)
        }
    })
}

fn arm_matches(v: &Version, comps: &[Comparator]) -> bool {
    if !v.pre.is_empty() && !has_anchor(comps, v) {
        return false;
    }
    comps.iter().all(|c| match c {
        Comparator::Compare { op, version, .. } => cmp_matches(v, *op, version),
    })
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone)]
struct Tok<'a> {
    text: &'a str,
    start: usize,
}

fn tokens(s: &str) -> Vec<Tok<'_>> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && whitespace(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !whitespace(bytes[i]) {
            i += 1;
        }
        out.push(Tok { text: &s[start..i], start });
    }
    out
}

fn full_comparator(tok: &Tok<'_>) -> Result<Comparator, RangeError> {
    let (op, offset) = if tok.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if tok.text.starts_with("<=") {
        (Op::Le, 2)
    } else if tok.text.starts_with('>') {
        (Op::Gt, 1)
    } else if tok.text.starts_with('<') {
        (Op::Lt, 1)
    } else if tok.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &tok.text[offset..];
    let operand_start = tok.start + offset;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let version = parse_version(operand).map_err(|error| RangeError::Version { at: operand_start, error })?;
    let anchor = !version.pre.is_empty();
    Ok(Comparator::Compare { op, version, anchor })
}

#[derive(Clone)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
    wildcard: bool,
}

fn parse_partial(text: &str, at: usize) -> Result<Partial, RangeError> {
    let force_full = text.contains('-') || text.contains('+')
        || (text.split('.').count() >= 3 && !text.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if force_full {
        let v = parse_version(text).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial { nums: vec![v.major, v.minor, v.patch], full: Some(v), wildcard: false });
    }
    let fields: Vec<&str> = text.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut saw_wild = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            saw_wild = true;
        } else {
            if saw_wild || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            match f.parse::<u64>() {
                Ok(n) => nums.push(n),
                Err(_) => return Err(RangeError::Syntax { at }),
            }
        }
    }
    Ok(Partial { nums, full: None, wildcard: saw_wild || nums.len() < 3 })
}

fn version_tuple(nums: &[u64]) -> Version {
    Version {
        major: *nums.get(0).unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn inc_upper(mut v: Version, index: usize, at: usize) -> Result<Version, RangeError> {
    let target = match index {
        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if index <= 1 {
        v.minor = if index == 0 { 0 } else { v.minor };
    }
    if index == 0 {
        v.patch = 0;
    }
    v.pre = vec!["0".to_string()];
    v.build.clear();
    Ok(v)
}

fn lower_partial(p: &Partial) -> Version {
    if let Some(v) = &p.full {
        v.clone()
    } else {
        version_tuple(&p.nums)
    }
}

fn add_cmp(out: &mut Vec<Comparator>, op: Op, version: Version, anchor: bool) {
    out.push(Comparator::Compare { op, version, anchor });
}

fn parse_operand_token(tok: &Tok<'_>, out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let text = tok.text;
    if text.starts_with('^') || text.starts_with('~') {
        let is_caret = text.starts_with('^');
        let operand = &text[1..];
        let at = tok.start + 1;
        let p = parse_partial(operand, at)?;
        if p.nums.is_empty() {
            return Err(RangeError::Syntax { at: tok.start });
        }
        let lower = lower_partial(&p);
        let k = if p.full.is_some() { 3 } else { p.nums.len() };
        let upper_index = if is_caret {
            if lower.major > 0 || k == 1 { 0 } else if lower.minor > 0 || k == 2 { 1 } else { 2 }
        } else if k == 1 { 0 } else { 1 };
        let upper = inc_upper(version_tuple(&[lower.major, lower.minor, lower.patch]), upper_index, at)?;
        add_cmp(out, Op::Ge, lower.clone(), p.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
        add_cmp(out, Op::Lt, upper, false);
        return Ok(());
    }
    let p = parse_partial(text, tok.start)?;
    if let Some(v) = p.full {
        let anchor = !v.pre.is_empty();
        add_cmp(out, Op::Eq, v, anchor);
        return Ok(());
    }
    if p.nums.is_empty() {
        return Ok(());
    }
    let lower = version_tuple(&p.nums);
    let k = p.nums.len();
    add_cmp(out, Op::Ge, lower.clone(), false);
    if k < 3 {
        let upper = inc_upper(lower, k - 1, tok.start)?;
        add_cmp(out, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_hyphen_arm(ts: &[Tok<'_>], out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let minus = ts.iter().position(|t| t.text == "-").unwrap();
    if ts.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax { at: ts[minus].start });
    }
    let lo = parse_partial(ts[0].text, ts[0].start)?;
    let hi = parse_partial(ts[2].text, ts[2].start)?;
    if lo.nums.is_empty() {
        return Err(RangeError::Syntax { at: ts[0].start });
    }
    if hi.nums.is_empty() {
        return Err(RangeError::Syntax { at: ts[2].start });
    }
    let lower = lower_partial(&lo);
    let upper_anchor = hi.full.as_ref().is_some_and(|v| !v.pre.is_empty());
    add_cmp(out, Op::Ge, lower, lo.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
    if let Some(v) = hi.full {
        add_cmp(out, Op::Le, v, upper_anchor);
    } else if hi.nums.len() == 3 {
        add_cmp(out, Op::Le, version_tuple(&hi.nums), false);
    } else {
        let upper = inc_upper(version_tuple(&hi.nums), hi.nums.len() - 1, ts[2].start)?;
        add_cmp(out, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_arm(ts: &[Tok<'_>]) -> Result<Vec<Comparator>, RangeError> {
    if let Some(i) = ts.iter().position(|t| t.text == "-") {
        if ts.len() != 3 || i != 1 {
            return Err(RangeError::Syntax { at: ts[i].start });
        }
        let mut out = Vec::new();
        parse_hyphen_arm(ts, &mut out)?;
        return Ok(out);
    }
    let mut out = Vec::new();
    for tok in ts {
        if tok.text.starts_with(['<', '>', '=']) {
            out.push(full_comparator(tok)?);
        } else {
            let b = tok.text.as_bytes().first().copied();
            if !b.is_some_and(|c| c.is_ascii_digit() || matches!(c, b'*' | b'x' | b'X' | b'^' | b'~')) {
                return Err(RangeError::Syntax { at: tok.start });
            }
            parse_operand_token(tok, &mut out)?;
        }
    }
    Ok(out)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let ts = tokens(s);
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < ts.len() {
        if ts[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: ts[i].start });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            if i + 1 == ts.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(ts[i].clone());
        }
        i += 1;
    }
    if !current.is_empty() {
        arms.push(parse_arm(&current)?);
    } else {
        return Err(RangeError::Empty { at: s.len() });
    }
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| parse_version(s).map_err(|error| CandidateError { index, error }))
        .collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
