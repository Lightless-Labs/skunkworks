#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append(word: &mut Word, c: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_string(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append(word, c, active_glob);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn find_braced_close(input: &str, start: usize) -> Option<usize> {
    let mut iter = input[start..].char_indices().peekable();
    while let Some((rel, c)) = iter.next() {
        if c == '\\' {
            iter.next();
        } else if c == '}' {
            return Some(start + rel);
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut iter = text.chars();
    while let Some(c) = iter.next() {
        if c == '\\' {
            if let Some(next) = iter.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn split_braced<'a>(
    input: &str,
    dollar: usize,
    env: &'a [(&'a str, &'a str)],
) -> Result<(usize, &'a str, Option<String>), SplitError> {
    let content_start = dollar + 2;
    let close = find_braced_close(input, content_start)
        .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let mut name_end = 0;
    for (i, c) in contents.char_indices() {
        if is_name_continue(c) {
            name_end = i + c.len_utf8();
        } else {
            break;
        }
    }

    if name_end == 0 || !is_name_start(contents.chars().next().unwrap()) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let remainder = &contents[name_end..];

    if remainder.is_empty() {
        let value = lookup(env, name).unwrap_or("");
        Ok((close + 1, value, None))
    } else if let Some(default) = remainder.strip_prefix(":-") {
        let value = lookup(env, name).unwrap_or("");
        if value.is_empty() {
            Ok((close + 1, "", Some(decode_default(default))))
        } else {
            Ok((close + 1, value, None))
        }
    } else {
        Err(SplitError::InvalidExpansion { pos: dollar })
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append(&mut current, c, false);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        append(&mut current, next, false);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        append(&mut current, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, value, default) = split_braced(input, i, env)?;
                        if let Some(text) = default {
                            append_string(&mut current, &text, false);
                        } else {
                            append_string(&mut current, value, false);
                        }
                        active = true;
                        i = end;
                    } else if let Some(first) = input[next_i..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_i;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(env, &input[next_i..end]).unwrap_or("");
                            append_string(&mut current, value, false);
                            active = true;
                            i = end;
                        } else {
                            append(&mut current, '$', false);
                            i = next_i;
                        }
                    } else {
                        append(&mut current, '$', false);
                        i = next_i;
                    }
                } else {
                    append(&mut current, c, false);
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        append(&mut current, next, false);
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, value, default) = split_braced(input, i, env)?;
                        if let Some(text) = default {
                            append_string(&mut current, &text, true);
                        } else {
                            append_string(&mut current, value, true);
                        }
                        active = true;
                        i = end;
                    } else if let Some(first) = input[next_i..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_i;
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(env, &input[next_i..end]).unwrap_or("");
                            append_string(&mut current, value, true);
                            active = true;
                            i = end;
                        } else {
                            active = true;
                            append(&mut current, '$', true);
                            i = next_i;
                        }
                    } else {
                        active = true;
                        append(&mut current, '$', true);
                        i = next_i;
                    }
                } else {
                    active = true;
                    append(&mut current, c, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}
