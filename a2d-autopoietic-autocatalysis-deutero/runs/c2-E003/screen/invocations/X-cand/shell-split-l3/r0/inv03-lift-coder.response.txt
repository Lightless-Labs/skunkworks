#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_text(word: &mut Word, text: &str, active_globs: bool) {
    for c in text.chars() {
        if active_globs && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn append_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn char_at(input: &str, pos: usize) -> Option<char> {
    input.get(pos..)?.chars().next()
}

fn next_pos(input: &str, pos: usize) -> usize {
    pos + char_at(input, pos).map_or(0, char::len_utf8)
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar + 2;
    let mut close = content_start;

    while close < input.len() {
        let c = char_at(input, close).expect("valid character boundary");
        if c == '\\' {
            close = next_pos(input, close);
            if close < input.len() {
                close = next_pos(input, close);
            }
        } else if c == '}' {
            break;
        } else {
            close = next_pos(input, close);
        }
    }

    if close >= input.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let contents = &input[content_start..close];
    let mut name_end = 0;
    let mut chars = contents.char_indices();
    match chars.next() {
        Some((_, c)) if is_name_start(c) => name_end = c.len_utf8(),
        _ => return Err(SplitError::InvalidExpansion { pos: dollar }),
    }

    for (offset, c) in chars {
        if is_name_continue(c) {
            name_end = offset + c.len_utf8();
        } else {
            break;
        }
    }

    let name = &contents[..name_end];
    let suffix = &contents[name_end..];
    let value = if suffix.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = suffix.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut pos = 0;

    while pos < text.len() {
        let c = char_at(text, pos).expect("valid character boundary");
        if c == '\\' {
            pos = next_pos(text, pos);
            if pos < text.len() {
                let escaped = char_at(text, pos).expect("valid character boundary");
                decoded.push(escaped);
                pos = next_pos(text, pos);
            } else {
                decoded.push('\\');
            }
        } else {
            decoded.push(c);
            pos = next_pos(text, pos);
        }
    }

    decoded
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut pos = 0;

    while pos < input.len() {
        let c = char_at(input, pos).expect("valid character boundary");

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut word, c, false);
                }
                pos = next_pos(input, pos);
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next_pos(input, pos);
                } else if c == '\\' {
                    let slash = pos;
                    pos = next_pos(input, pos);
                    if pos >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = char_at(input, pos).expect("valid character boundary");
                    if matches!(next, '"' | '\\' | '$') {
                        append_scalar(&mut word, next, false);
                        pos = next_pos(input, pos);
                    } else if next == '\n' {
                        pos = next_pos(input, pos);
                    } else {
                        append_scalar(&mut word, '\\', false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    let after = next_pos(input, pos);
                    if char_at(input, after) == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if char_at(input, after) == Some('{') {
                        let (value, end) = parse_braced(input, pos, env)?;
                        append_text(&mut word, &value, false);
                        active = true;
                        pos = end;
                    } else if char_at(input, after).is_some_and(is_name_start) {
                        let mut end = after;
                        while char_at(input, end).is_some_and(is_name_continue) {
                            end = next_pos(input, end);
                        }
                        let value = lookup(env, &input[after..end]).unwrap_or("");
                        append_text(&mut word, value, false);
                        active = true;
                        pos = end;
                    } else {
                        append_scalar(&mut word, '$', false);
                        pos = after;
                    }
                } else {
                    append_scalar(&mut word, c, false);
                    pos = next_pos(input, pos);
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    pos = next_pos(input, pos);
                } else if c == '\'' {
                    quote = Quote::Single(pos);
                    active = true;
                    pos = next_pos(input, pos);
                } else if c == '"' {
                    quote = Quote::Double(pos);
                    active = true;
                    pos = next_pos(input, pos);
                } else if c == '\\' {
                    let slash = pos;
                    pos = next_pos(input, pos);
                    if pos >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = char_at(input, pos).expect("valid character boundary");
                    if next == '\n' {
                        pos = next_pos(input, pos);
                    } else {
                        append_scalar(&mut word, next, false);
                        active = true;
                        pos = next_pos(input, pos);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    let after = next_pos(input, pos);
                    if char_at(input, after) == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if char_at(input, after) == Some('{') {
                        let (value, end) = parse_braced(input, pos, env)?;
                        append_text(&mut word, &value, true);
                        active = true;
                        pos = end;
                    } else if char_at(input, after).is_some_and(is_name_start) {
                        let mut end = after;
                        while char_at(input, end).is_some_and(is_name_continue) {
                            end = next_pos(input, end);
                        }
                        let value = lookup(env, &input[after..end]).unwrap_or("");
                        append_text(&mut word, value, true);
                        active = true;
                        pos = end;
                    } else {
                        append_scalar(&mut word, '$', true);
                        active = true;
                        pos = after;
                    }
                } else {
                    append_scalar(&mut word, c, true);
                    active = true;
                    pos = next_pos(input, pos);
                }
            }
        }
    }

    match quote {
        Quote::Single(open) => return Err(SplitError::UnterminatedSingle { pos: open }),
        Quote::Double(open) => return Err(SplitError::UnterminatedDouble { pos: open }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
