#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_scalar(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    let mut chars = input[after_dollar..].char_indices();

    if let Some((_, first)) = chars.next() {
        if is_name_start(first) {
            let mut end = after_dollar + first.len_utf8();
            for (offset, c) in chars {
                if is_name_char(c) {
                    end = after_dollar + offset + c.len_utf8();
                } else {
                    break;
                }
            }
            let name = &input[after_dollar..end];
            return Ok((end, lookup(env, name).unwrap_or("").to_string()));
        }
    }

    if !input[after_dollar..].starts_with('{') {
        return Ok((after_dollar, "$".to_string()));
    }

    let content_start = after_dollar + 1;
    let mut cursor = content_start;
    let mut close = None;
    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += 1;
            if cursor < input.len() {
                cursor += input[cursor..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let Some(close_pos) = close else {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    };

    let contents = &input[content_start..close_pos];
    let (name_end, name) = {
        let mut end = 0;
        for (offset, c) in contents.char_indices() {
            if is_name_char(c) {
                if offset == 0 && !is_name_start(c) {
                    break;
                }
                end = offset + c.len_utf8();
            } else {
                break;
            }
        }
        (end, &contents[..end])
    };

    if name.is_empty() || !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &contents[name_end..];
    let value = if remainder.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close_pos + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut cursor = 0usize;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next_pos = cursor + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut word, c, false);
                }
                cursor = next_pos;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next_pos;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[next_pos..].chars().next().unwrap();
                    let after_next = next_pos + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_scalar(&mut word, next, false);
                        }
                        cursor = after_next;
                    } else {
                        append_scalar(&mut word, '\\', false);
                        cursor = next_pos;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_pos..].starts_with('{')
                        || input[next_pos..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, cursor, env)?;
                        append_expansion(&mut word, &value, false);
                        active = true;
                        cursor = end;
                    } else {
                        append_scalar(&mut word, '$', false);
                        cursor = next_pos;
                    }
                } else {
                    append_scalar(&mut word, c, false);
                    cursor = next_pos;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    cursor = next_pos;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_pos = cursor;
                    active = true;
                    cursor = next_pos;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_pos = cursor;
                    active = true;
                    cursor = next_pos;
                } else if c == '\\' {
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_pos..].chars().next().unwrap();
                    let after_next = next_pos + next.len_utf8();
                    if next == '\n' {
                        cursor = after_next;
                    } else {
                        append_scalar(&mut word, next, false);
                        active = true;
                        cursor = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_pos..].starts_with('{')
                        || input[next_pos..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, cursor, env)?;
                        append_expansion(&mut word, &value, true);
                        active = true;
                        cursor = end;
                    } else {
                        append_scalar(&mut word, '$', true);
                        active = true;
                        cursor = next_pos;
                    }
                } else {
                    append_scalar(&mut word, c, true);
                    active = true;
                    cursor = next_pos;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
