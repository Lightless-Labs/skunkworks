#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn width(field: Field) -> u32 {
    match field {
        Field::Minute => 60,
        Field::Hour => 24,
        Field::DayOfMonth => 31,
        Field::Month => 12,
        Field::DayOfWeek => 7,
    }
}

fn names(field: Field, s: &str) -> Option<u32> {
    let list: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    list.iter()
        .position(|name| name.eq_ignore_ascii_case(s))
        .map(|i| i as u32 + if field == Field::Month { 1 } else { 0 })
}

fn is_digits(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn parse_number(
    s: &str,
    field: Field,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if step {
        if value == 0 || value > width(field) {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (lo, hi) = bounds(field);
        if value < lo || value > hi {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(value)
}

fn atom_value(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if is_digits(s) {
        parse_number(s, field, offset, false)
    } else if let Some(value) = names(field, s) {
        Ok(value)
    } else {
        Err(ParseError::InvalidSyntax { field, offset })
    }
}

fn parse_item(
    item: &str,
    field: Field,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
        (
            &item[..slash],
            Some(&item[slash + 1..]),
            Some(item_offset + slash + 1),
        )
    } else {
        (item, None, None)
    };

    if step_text == Some("") {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let hyphen_positions: Vec<usize> = base
        .as_bytes()
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    if hyphen_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (start_text, end_text, range) = if base == "*" {
        (None, None, false)
    } else if hyphen_positions.len() == 1 {
        let hyphen = hyphen_positions[0];
        let left = &base[..hyphen];
        let right = &base[hyphen + 1..];
        if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        (Some(left), Some(right), true)
    } else {
        if base.is_empty() || base == "*" {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        (Some(base), None, false)
    };

    // Validate atom shapes before evaluating any numeric values.
    for text in [start_text, end_text].into_iter().flatten() {
        if !is_digits(text) && names(field, text).is_none() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    }
    if let Some(step) = step_text {
        if !is_digits(step) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    }

    let start = if let Some(text) = start_text {
        let off = item_offset;
        Some(atom_value(text, field, off)?)
    } else {
        None
    };
    let end = if let Some(text) = end_text {
        let off = item_offset + base.len() - text.len();
        Some(atom_value(text, field, off)?)
    } else {
        None
    };

    if let (Some(a), Some(b)) = (start, end) {
        if a > b {
            let end_off = item_offset + base.len() - end_text.unwrap().len();
            return Err(ParseError::ReversedRange {
                field,
                offset: end_off,
            });
        }
    }

    let step = if let Some(text) = step_text {
        Some(parse_number(
            text,
            field,
            step_offset.unwrap(),
            true,
        )?)
    } else {
        None
    };

    let (minimum, maximum) = bounds(field);
    let from = match start {
        Some(v) => v,
        None => minimum,
    };
    let to = if let Some(v) = end {
        v
    } else if step.is_some() {
        maximum
    } else {
        start.unwrap_or(maximum)
    };
    let stride = step.unwrap_or(1);

    let mut value = from;
    while value <= to {
        selected[value as usize] = true;
        match value.checked_add(stride) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    let _ = range;
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (lo, hi) = bounds(FIELDS[i]);
        vec![false; (hi + 1) as usize]
    });

    for (index, (&field, text)) in FIELDS.iter().zip(fields.iter()).enumerate() {
        for (offset, &b) in text.as_bytes().iter().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut item_offset = 0;
        for item in text.split(',') {
            parse_item(item, field, item_offset, &mut sets[index])?;
            item_offset += item.len() + 1;
        }
    }

    Ok(Cron {
        sets,
        day_of_month_unrestricted: fields[2] == "*",
        day_of_week_unrestricted: fields[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.sets[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &selected)| selected.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            current = increment_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.sets[0][at.minute as usize]
            || !self.sets[1][at.hour as usize]
            || !self.sets[3][at.month as usize]
        {
            return false;
        }

        let dom = self.sets[2][at.day as usize];
        let dow = self.sets[4][weekday(at.year, at.month, at.day) as usize];

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn increment_minute(at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        return CivilDateTime {
            minute: at.minute + 1,
            ..at
        };
    }
    if at.hour < 23 {
        return CivilDateTime {
            hour: at.hour + 1,
            minute: 0,
            ..at
        };
    }
    if at.day < days_in_month(at.year, at.month) {
        return CivilDateTime {
            day: at.day + 1,
            hour: 0,
            minute: 0,
            ..at
        };
    }
    if at.month < 12 {
        return CivilDateTime {
            month: at.month + 1,
            day: 1,
            hour: 0,
            minute: 0,
            ..at
        };
    }
    CivilDateTime {
        year: at.year + 1,
        month: 1,
        day: 1,
        hour: 0,
        minute: 0,
    }
}
