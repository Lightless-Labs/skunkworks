use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Clone, Copy)]
struct Domain {
    min: u32,
    max: u32,
}

impl Domain {
    fn width(self) -> u32 {
        self.max - self.min + 1
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn err_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn is_name_byte(b: u8) -> bool {
    b.is_ascii_alphabetic()
}

fn name_value(field: Field, token: &[u8]) -> Option<u32> {
    let upper: Vec<u8> = token.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => match upper.as_slice() {
            b"JAN" => Some(1),
            b"FEB" => Some(2),
            b"MAR" => Some(3),
            b"APR" => Some(4),
            b"MAY" => Some(5),
            b"JUN" => Some(6),
            b"JUL" => Some(7),
            b"AUG" => Some(8),
            b"SEP" => Some(9),
            b"OCT" => Some(10),
            b"NOV" => Some(11),
            b"DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_slice() {
            b"SUN" => Some(0),
            b"MON" => Some(1),
            b"TUE" => Some(2),
            b"WED" => Some(3),
            b"THU" => Some(4),
            b"FRI" => Some(5),
            b"SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

enum AtomKind {
    Number,
    Name(u32),
}

struct ParsedAtom {
    kind: AtomKind,
    value: u32,
    start: usize,
}

fn atom_kind(field: Field, bytes: &[u8]) -> Option<AtomKind> {
    if !bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit) {
        return Some(AtomKind::Number);
    }
    if !bytes.is_empty() && bytes.iter().all(|b| is_name_byte(*b)) {
        return name_value(field, bytes).map(AtomKind::Name);
    }
    None
}

fn parse_number(field: Field, bytes: &[u8], offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value: u32 = 0;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_atom(
    field: Field,
    bytes: &[u8],
    offset: usize,
    bounds_check: bool,
) -> Result<ParsedAtom, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        let value = parse_number(field, bytes, offset)?;
        if bounds_check {
            let d = domain(field);
            if value < d.min || value > d.max {
                return Err(ParseError::OutOfRange { field, offset });
            }
        }
        Ok(ParsedAtom {
            kind: AtomKind::Number,
            value,
            start: offset,
        })
    } else if let Some(value) = name_value(field, bytes) {
        Ok(ParsedAtom {
            kind: AtomKind::Name(value),
            value,
            start: offset,
        })
    } else {
        Err(err_syntax(field, offset))
    }
}

fn parse_field(
    field: Field,
    text: &str,
) -> Result<(Vec<bool>, bool), ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let d = domain(field);
    let mut selected = vec![false; (d.max + 1) as usize];
    let mut exactly_star = false;

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(err_syntax(field, item_start));
        }

        let (base, step_part, step_offset) = if let Some(&slash) = slash_positions.first() {
            if slash == 0 || slash + 1 >= item.len() {
                return Err(err_syntax(field, item_start));
            }
            (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
        } else {
            (item, None, 0)
        };

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if is_star && !hyphens.is_empty() {
            return Err(err_syntax(field, item_start));
        }
        if hyphens.len() > 1 {
            return Err(err_syntax(field, item_start));
        }

        let (start_bytes, end_bytes, start_offset, end_offset) =
            if let Some(&hyphen) = hyphens.first() {
                if hyphen == 0 || hyphen + 1 >= base.len() {
                    return Err(err_syntax(field, item_start));
                }
                (
                    &base[..hyphen],
                    Some(&base[hyphen + 1..]),
                    item_start,
                    item_start + hyphen + 1,
                )
            } else {
                (base, None, item_start, 0)
            };

        if is_star {
            // Star is a distinct complete base.
        } else {
            if atom_kind(field, start_bytes).is_none()
                || end_bytes.is_some_and(|end| atom_kind(field, end).is_none())
            {
                return Err(err_syntax(field, item_start));
            }
        }
        if let Some(step) = step_part {
            if !step.iter().all(u8::is_ascii_digit) {
                return Err(err_syntax(field, item_start));
            }
        }

        let start = if is_star {
            None
        } else {
            Some(parse_atom(field, start_bytes, start_offset, true)?)
        };
        let end = if let Some(end) = end_bytes {
            Some(parse_atom(field, end, end_offset, true)?)
        } else {
            None
        };

        if let (Some(a), Some(b)) = (&start, &end) {
            if a.value > b.value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: b.start,
                });
            }
        }

        let step = if let Some(step_bytes) = step_part {
            let value = parse_number(field, step_bytes, step_offset)?;
            if value < 1 || value > d.width() {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            value
        } else {
            1
        };

        let (range_start, range_end) = if is_star {
            (d.min, d.max)
        } else if let (Some(a), Some(b)) = (&start, &end) {
            (a.value, b.value)
        } else if let Some(a) = &start {
            if step_part.is_some() {
                (a.value, d.max)
            } else {
                (a.value, a.value)
            }
        } else {
            unreachable!()
        };

        let mut value = range_start;
        while value <= range_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        if item_start == 0 && item_end == bytes.len() && text == "*" {
            exactly_star = true;
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok((selected, exactly_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    let mut dom_unrestricted = false;
    let mut dow_unrestricted = false;

    for (i, (field, text)) in fields.iter().copied().zip(pieces.iter().copied()).enumerate() {
        let (values, star) = parse_field(field, text)?;
        selected[i] = values;
        if field == Field::DayOfMonth {
            dom_unrestricted = star;
        } else if field == Field::DayOfWeek {
            dow_unrestricted = star;
        }
    }

    Ok(Cron {
        selected,
        day_of_month_unrestricted: dom_unrestricted,
        day_of_week_unrestricted: dow_unrestricted,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &chosen)| chosen.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut date = after;
        loop {
            let max_day = days_in_month(date.year, date.month);
            let first_minute = if date.day == after.day
                && date.month == after.month
                && date.year == after.year
            {
                date.hour as u16 * 60 + date.minute as u16 + 1
            } else {
                0
            };

            if self.selected[3][date.month as usize] {
                for minute_of_day in first_minute..(24 * 60) {
                    let hour = (minute_of_day / 60) as u8;
                    let minute = (minute_of_day % 60) as u8;
                    let candidate = CivilDateTime {
                        year: date.year,
                        month: date.month,
                        day: date.day,
                        hour,
                        minute,
                    };
                    if self.matches_valid(candidate) {
                        return Ok(Some(candidate));
                    }
                }
            }

            if date.day < max_day {
                date.day += 1;
            } else if date.month < 12 {
                date.month += 1;
                date.day = 1;
            } else if date.year < 2399 {
                date.year += 1;
                date.month = 1;
                date.day = 1;
            } else {
                return Ok(None);
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom = self.selected[2][at.day as usize];
        let dow = weekday(at.year, at.month, at.day) as usize;
        let weekday_selected = self.selected[4][dow];

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => weekday_selected,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && weekday_selected
                } else {
                    dom || weekday_selected
                }
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}
