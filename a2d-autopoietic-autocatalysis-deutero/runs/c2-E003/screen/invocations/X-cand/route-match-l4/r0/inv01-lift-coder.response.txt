#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || matches!(b, b'_' | b'-'))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &[u8], start: usize, pattern: bool) -> Result<String, Error> {
    let fail = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let mut decoded = Vec::with_capacity(raw.len());
    let mut i = 0;
    while i < raw.len() {
        let b = raw[i];
        if b == b'%' {
            if i + 2 >= raw.len() {
                return Err(fail(start + i));
            }
            let hi = hex_value(raw[i + 1]).ok_or_else(|| fail(start + i))?;
            let lo = hex_value(raw[i + 2]).ok_or_else(|| fail(start + i))?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(fail(start + i));
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(fail(start + i));
            }
            decoded.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let bytes = input.as_bytes();
    let fail = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if bytes.first() != Some(&b'/') {
        return Err(fail(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut trailing = false;
    let end = if bytes.last() == Some(&b'/') {
        trailing = true;
        bytes.len() - 1
    } else {
        bytes.len()
    };

    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut start = 1;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(fail(start));
        }

        let raw = &bytes[start..stop];
        if pattern && (raw[0] == b':' || raw[0] == b'*') {
            let name_bytes = &raw[1..];
            let name = std::str::from_utf8(name_bytes).unwrap_or("");
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(fail(start));
            }
            let is_wildcard = raw[0] == b'*';
            if is_wildcard && stop < end {
                return Err(fail(start));
            }
            segments.push(if is_wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let text = decode_segment(raw, start, pattern)?;
            segments.push(Segment::Static(text));
        }

        if stop == end {
            break;
        }
        start = stop + 1;
        if start == end {
            return Err(fail(start));
        }
    }

    Ok((segments, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn encode_component(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let candidate = Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        };
        if let Some(existing) = self
            .routes
            .iter()
            .filter(|r| r.method == method && shape_conflicts(r, &candidate, self.policy))
            .map(|r| r.id)
            .min()
        {
            return Err(Error::Conflict { with: existing });
        }
        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, trailing) = parse_path(head, false)?;

        let mut candidates: Vec<(usize, Vec<(String, String)>, Vec<u8>)> = Vec::new();

        for route in &self.routes {
            let mut params = Vec::new();
            let mut rank = Vec::new();
            let mut ri = 0;
            let mut pi = 0;
            let mut matched = true;

            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Static(expected) => {
                        if pi >= request_segments.len() {
                            matched = false;
                            break;
                        }
                        match &request_segments[pi] {
                            Segment::Static(actual) if actual == expected => {
                                rank.push(2);
                                pi += 1;
                            }
                            _ => {
                                matched = false;
                                break;
                            }
                        }
                    }
                    Segment::Param(name) => {
                        if pi >= request_segments.len() {
                            matched = false;
                            break;
                        }
                        match &request_segments[pi] {
                            Segment::Static(actual) => {
                                rank.push(1);
                                params.push((name.clone(), actual.clone()));
                                pi += 1;
                            }
                            _ => {
                                matched = false;
                                break;
                            }
                        }
                    }
                    Segment::Wildcard(name) => {
                        let tail = request_segments[pi..]
                            .iter()
                            .filter_map(|s| match s {
                                Segment::Static(v) => Some(v.as_str()),
                                _ => None,
                            })
                            .collect::<Vec<_>>()
                            .join("/");
                        rank.push(0);
                        params.push((name.clone(), tail));
                        pi = request_segments.len();
                    }
                }
                ri += 1;
            }

            if matched && pi == request_segments.len() {
                if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                    continue;
                }
                if !matches!(route.segments.last(), Some(Segment::Wildcard(_))) {
                    rank.push(3);
                }
                candidates.push((route.id, params, rank));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_candidates: Vec<_> = candidates
            .iter()
            .filter(|(id, _, _)| self.routes[*id].method == "HEAD")
            .cloned()
            .collect();
        let eligible_method = if method == "HEAD" && head_candidates.is_empty() {
            "GET"
        } else {
            method
        };
        let eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(id, _, _)| self.routes[*id].method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (id, _, _) in self.path_candidate_ids(head, trailing)? {
                let m = self.routes[id].method.clone();
                allowed.insert(m.clone());
                if m == "GET" {
                    allowed.insert("HEAD".to_owned());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let winner = eligible
            .into_iter()
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.0.cmp(&a.0)))
            .expect("eligible is nonempty");

        let route = &self.routes[winner.0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing != trailing
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.0,
            params: winner.1,
            redirect,
        })
    }

    fn path_candidate_ids(&self, head: &str, trailing: bool) -> Result<Vec<usize>, Error> {
        let (request_segments, _) = parse_path(head, false)?;
        let mut ids = Vec::new();
        for route in &self.routes {
            let mut ri = 0;
            let mut pi = 0;
            let mut matched = true;
            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Static(expected) => {
                        if pi >= request_segments.len()
                            || !matches!(&request_segments[pi], Segment::Static(actual) if actual == expected)
                        {
                            matched = false;
                            break;
                        }
                        pi += 1;
                    }
                    Segment::Param(_) => {
                        if pi >= request_segments.len() {
                            matched = false;
                            break;
                        }
                        pi += 1;
                    }
                    Segment::Wildcard(_) => {
                        pi = request_segments.len();
                    }
                }
                ri += 1;
            }
            if matched && pi == request_segments.len()
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
            {
                ids.push(route.id);
            }
        }
        Ok(ids)
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(n) | Segment::Wildcard(n) => capture_names.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &capture_names {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let extra = supplied
            .keys()
            .filter(|key| !capture_names.iter().any(|n| n == *key))
            .min()
            .copied();
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: key.to_owned(),
            });
        }

        for segment in &route.segments {
            let (key, wildcard) = match segment {
                Segment::Param(n) => (n.as_str(), false),
                Segment::Wildcard(n) => (n.as_str(), true),
                Segment::Static(_) => continue,
            };
            let value = supplied[key];
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid_value = if wildcard {
                valid_ascii
                    && (value.is_empty()
                        || value
                            .split('/')
                            .all(|part| !part.is_empty()))
            } else {
                valid_ascii && !value.is_empty() && !value.contains('/')
            };
            if !valid_value {
                return Err(Error::InvalidValue {
                    name: key.to_owned(),
                });
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if let Segment::Wildcard(name) = segment {
                let value = supplied[name.as_str()];
                if !value.is_empty() {
                    if !first {
                        output.push('/');
                    }
                    output.push_str(
                        &value
                            .split('/')
                            .map(encode_component)
                            .collect::<Vec<_>>()
                            .join("/"),
                    );
                }
                first = false;
                continue;
            }

            if !first {
                output.push('/');
            }
            match segment {
                Segment::Static(text) => output.push_str(&encode_component(text)),
                Segment::Param(name) => output.push_str(&encode_component(supplied[name.as_str()])),
                Segment::Wildcard(_) => unreachable!(),
            }
            first = false;
        }

        if output != "/" && route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}
