#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains(&product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: [1, 3, 4, 10][i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !matches!(coin, 1 | 3 | 4 | 10) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_ref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };
        let product = match self.products.iter().find(|p| p.id == *selected_id) {
            Some(product) => product,
            None => return Err(Error::NoSelection),
        };
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change_counts = match choose_change(amount, self.tubes) {
            Some(counts) => counts,
            None => return Err(Error::NoChange { amount }),
        };

        let denominations = [1u16, 3, 4, 10];
        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= change_counts[i];
        }
        for &coin in &self.escrow {
            let i = denomination_index(coin).expect("escrow only contains accepted coins");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: denominations[i],
                });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change_counts[i] {
                change.push(denominations[i]);
            }
        }

        let receipt = Receipt {
            id: selected_id.clone(),
            price,
            change,
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(refunded)
    }
}

fn denomination_index(coin: u16) -> Option<usize> {
    match coin {
        1 => Some(0),
        3 => Some(1),
        4 => Some(2),
        10 => Some(3),
        _ => None,
    }
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let denominations = [1u16, 3, 4, 10];

    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        greedy[i] = tubes[i].min(remaining / denominations[i]);
        remaining -= greedy[i] * denominations[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let target = usize::from(amount);
    let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
    best[0] = Some([0; 4]);

    for i in 0..4 {
        let coin_value = usize::from(denominations[i]);
        let limit = usize::from(tubes[i]);
        for _ in 0..limit {
            for value in (coin_value..=target).rev() {
                if let Some(previous) = best[value - coin_value] {
                    let mut candidate = previous;
                    candidate[i] += 1;
                    if best[value].is_none_or(|current| better(&candidate, &current)) {
                        best[value] = Some(candidate);
                    }
                }
            }
        }
    }

    best[target]
}

fn better(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
    let candidate_total: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
    let current_total: u32 = current.iter().map(|&n| u32::from(n)).sum();
    if candidate_total != current_total {
        return candidate_total < current_total;
    }
    for i in (0..4).rev() {
        if candidate[i] != current[i] {
            return candidate[i] > current[i];
        }
    }
    false
}
