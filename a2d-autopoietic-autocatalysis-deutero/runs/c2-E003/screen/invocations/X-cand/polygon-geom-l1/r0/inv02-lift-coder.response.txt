#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_sqrt(n: u128) -> u128 {
    // Integer square root via binary search; the input is bounded well below u128::MAX.
    let mut low = 0u128;
    let mut high = n.min(u64::MAX as u128 + 1);
    while low < high {
        let mid = low + (high - low + 1) / 2;
        if mid <= n / mid.max(1) {
            low = mid;
        } else {
            high = mid - 1;
        }
    }

    let square = low * low;
    let remainder = n - square;
    if remainder >= low + 1 {
        low + 1
    } else {
        low
    }
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128 - a.x as i128).unsigned_abs();
    let dy = (b.y as i128 - a.y as i128).unsigned_abs();
    let squared = dx * dx + dy * dy;
    rounded_sqrt(squared * 1_000_000) as u64
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.first() == points.last() {
        points.len() - 1
    } else {
        points.len()
    };
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0u64;
    let mut longest_edge = None;
    let mut longest_pair: Option<(Point, Point)> = None;

    for index in 0..effective_len {
        let a = polygon[index];
        let b = polygon[(index + 1) % effective_len];

        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = edge_length_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let (low, high) = if a <= b { (a, b) } else { (b, a) };
        let pair = (low, high);
        match longest_edge {
            None => {
                longest_edge = Some(index);
                longest_pair = Some(pair);
            }
            Some(current) => {
                let current_length = edge_milli[current];
                if length > current_length
                    || (length == current_length && Some(pair) < longest_pair)
                    || (length == current_length && Some(pair) == longest_pair && index < current)
                {
                    longest_edge = Some(index);
                    longest_pair = Some(pair);
                }
            }
        }
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
