#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_sqrt(n: u128) -> u128 {
    // Find floor(sqrt(n)) without multiplication overflow.
    let mut low = 0_u128;
    let mut high = u64::MAX as u128 + 1;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid <= n / mid {
            low = mid;
        } else {
            high = mid;
        }
    }

    // Compare n with (low + 1/2)^2 exactly, using 4*n vs (2*low+1)^2.
    // Values in this module are far below u128's limit.
    let boundary = 2 * low + 1;
    if 4 * n >= boundary * boundary {
        low + 1
    } else {
        low
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let raw_len = points.len();
    if raw_len > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: raw_len });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if raw_len >= 2 && points[raw_len - 1] == points[0] {
        raw_len - 1
    } else {
        raw_len
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0_i128;
    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];

        let cross = (a.x as i128) * (b.y as i128) - (a.y as i128) * (b.x as i128);
        signed_area2 += cross;

        let dx = (a.x as i128) - (b.x as i128);
        let dy = (a.y as i128) - (b.y as i128);
        let d2 = (dx * dx + dy * dy) as u128;
        let rounded = rounded_sqrt(d2 * 1_000_000) as u64;

        edge_milli.push(rounded);
        perimeter_milli += rounded;

        let replace = match longest_edge {
            None => true,
            Some(best) => {
                if rounded > edge_milli[best] {
                    true
                } else if rounded < edge_milli[best] {
                    false
                } else {
                    let best_a = polygon[best];
                    let best_b = polygon[(best + 1) % effective_len];
                    let best_pair = if best_a <= best_b {
                        (best_a, best_b)
                    } else {
                        (best_b, best_a)
                    };
                    let candidate_pair = if a <= b { (a, b) } else { (b, a) };
                    candidate_pair < best_pair
                        || (candidate_pair == best_pair && i < best)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
