```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut order: Vec<usize> = (0..jobs.len()).collect();
    order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion_times = vec![0u64; jobs.len()];
    let mut completed = vec![false; jobs.len()];
    let mut completed_count = 0usize;
    let mut next_event = 0usize;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    while completed_count < jobs.len() {
        while next_event < order.len()
            && u64::from(jobs[order[next_event]].arrival) <= time
        {
            let index = order[next_event];
            next_event += 1;
            if jobs[index].ticks == 0 {
                completed[index] = true;
                completion_times[index] = u64::from(jobs[index].arrival);
                completed_count += 1;
            } else {
                queues[jobs[index].level].push_back(index);
            }
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());

        let Some(level) = selected_level else {
            if next_event < order.len() {
                let next_time = u64::from(jobs[order[next_event]].arrival);
                if next_time > time {
                    if let Some(last) = idle.last_mut() {
                        if last.1 == time {
                            last.1 = next_time;
                        } else {
                            idle.push((time, next_time));
                        }
                    } else {
                        idle.push((time, next_time));
                    }
                    time = next_time;
                    last_dispatched = None;
                }
                continue;
            }
            break;
        };

        let index = queues[level].pop_front().expect("selected queue is nonempty");
        let job = &jobs[index];

        if last_dispatched.is_some_and(|last_id| last_id != job.id) {
            time += u64::from(switch_cost);
        }

        while next_event < order.len()
            && u64::from(jobs[order[next_event]].arrival) <= time
        {
            let arrived = order[next_event];
            next_event += 1;
            if jobs[arrived].ticks == 0 {
                completed[arrived] = true;
                completion_times[arrived] = u64::from(jobs[arrived].arrival);
                completed_count += 1;
            } else {
                queues[jobs[arrived].level].push_back(arrived);
            }
        }

        let start = time;
        let consumed = remaining[index].min(quanta[level]);
        time += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished,
            level,
        });

        if finished {
            completed[index] = true;
            completion_times[index] = time;
            completed_count += 1;
        }

        while next_event < order.len()
            && u64::from(jobs[order[next_event]].arrival) <= time
        {
            let arrived = order[next_event];
            next_event += 1;
            if jobs[arrived].ticks == 0 {
                completed[arrived] = true;
                completion_times[arrived] = u64::from(jobs[arrived].arrival);
                completed_count += 1;
            } else {
                queues[jobs[arrived].level].push_back(arrived);
            }
        }

        if !finished {
            let demoted_level = (level + 1).min(quanta.len() - 1);
            queues[demoted_level].push_back(index);
        }

        last_dispatched = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion_times[index]))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
```
