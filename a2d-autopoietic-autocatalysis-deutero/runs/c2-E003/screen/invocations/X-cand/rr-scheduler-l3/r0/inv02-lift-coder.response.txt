```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let mut now = 0u64;
    let mut event_cursor = 0usize;
    let mut last_dispatched_id: Option<u32> = None;

    fn admit_through(
        timestamp: u64,
        jobs: &[Job],
        events: &[usize],
        event_cursor: &mut usize,
        remaining: &[u32],
        completion: &mut [Option<u64>],
        queues: &mut [std::collections::VecDeque<usize>],
    ) {
        while *event_cursor < events.len()
            && u64::from(jobs[events[*event_cursor]].arrival) <= timestamp
        {
            let index = events[*event_cursor];
            *event_cursor += 1;

            if remaining[index] == 0 {
                completion[index] = Some(u64::from(jobs[index].arrival));
            } else {
                queues[jobs[index].level].push_back(index);
            }
        }
    }

    loop {
        admit_through(
            now,
            jobs,
            &events,
            &mut event_cursor,
            &remaining,
            &mut completion,
            &mut queues,
        );

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(queue_level) = selected_level else {
            if event_cursor == events.len() {
                break;
            }

            let next_time = u64::from(jobs[events[event_cursor]].arrival);
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let index = queues[queue_level].pop_front().unwrap();
        let job = &jobs[index];

        if last_dispatched_id.is_some_and(|id| id != job.id) {
            now += u64::from(switch_cost);
        }

        admit_through(
            now,
            jobs,
            &events,
            &mut event_cursor,
            &remaining,
            &mut completion,
            &mut queues,
        );

        let run_level = levels[index];
        let consumed = remaining[index].min(quanta[run_level]);
        let start = now;
        now += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id: job.id,
            start,
            end: now,
            finished,
            level: run_level,
        });

        admit_through(
            now,
            jobs,
            &events,
            &mut event_cursor,
            &remaining,
            &mut completion,
            &mut queues,
        );

        if finished {
            completion[index] = Some(now);
        } else {
            levels[index] = (run_level + 1).min(quanta.len() - 1);
            queues[levels[index]].push_back(index);
        }

        last_dispatched_id = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}
```
