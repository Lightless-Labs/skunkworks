```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len().saturating_mul(3));
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut arrival_index = 0;
    let mut now = 0u64;
    let mut running: Option<(u32, u64)> = None;
    let mut previous_finish: Option<u64> = None;

    loop {
        let next_arrival = arrivals.get(arrival_index).map(|c| c.arrival);
        let next_finish = running.map(|(_, finish)| finish);

        let tick = match (next_arrival, next_finish) {
            (None, None) => break,
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (Some(a), Some(f)) => a.min(f),
        };
        now = tick;

        if let Some((customer, finish)) = running {
            if finish == tick {
                events.push(Event {
                    time: tick,
                    customer,
                    kind: Kind::Finish,
                });
                running = None;
                previous_finish = Some(tick);
            }
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == tick {
            let customer = arrivals[arrival_index];
            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            arrival_index += 1;
        }

        while running.is_none() {
            let Some(customer) = waiting.pop_front() else {
                break;
            };

            let wake = previous_finish.map_or(true, |finish| tick > finish);
            let service_finish = tick.checked_add(customer.service).ok_or(
                Error::TimeOverflow {
                    customer: customer.id,
                    time: tick,
                },
            )?;
            let finish = service_finish
                .checked_add(if wake { wake_cost } else { 0 })
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time: tick,
                })?;

            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Start { finish, wake },
            });

            if finish == tick {
                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                previous_finish = Some(tick);
            } else {
                running = Some((customer.id, finish));
            }
        }
    }

    let _ = now;
    Ok(events)
}
```
