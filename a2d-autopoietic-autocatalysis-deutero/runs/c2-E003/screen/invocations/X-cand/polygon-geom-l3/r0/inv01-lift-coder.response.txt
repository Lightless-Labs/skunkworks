#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const RAW_LIMIT: usize = 4096;

fn valid_point(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > RAW_LIMIT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !valid_point(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.first() == points.last() {
        points.len() - 1
    } else {
        points.len()
    };
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    Ok(points[..effective_len].to_vec())
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = (b.x as i128 - a.x as i128) as i128;
    let dy = (b.y as i128 - a.y as i128) as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let target = squared_distance(a, b) * 1_000_000;
    let mut lo = 0u128;
    let mut hi = 3_000_000_000u128;

    while lo < hi {
        let mid = lo + (hi - lo) / 2;
        if mid * mid < target {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    let ceil = lo;
    let floor = if ceil * ceil == target { ceil } else { ceil - 1 };

    let floor_sq = floor * floor;
    let ceil_sq = ceil * ceil;
    let dist_floor = target - floor_sq;
    let dist_ceil = ceil_sq - target;
    if dist_floor < dist_ceil || (dist_floor == dist_ceil && floor % 2 == 1) {
        floor as u64
    } else {
        ceil as u64
    }
}

fn edge_lengths(vertices: &[Point]) -> Vec<u64> {
    (0..vertices.len())
        .map(|i| rounded_edge_milli(vertices[i], vertices[(i + 1) % vertices.len()]))
        .collect()
}

fn area_sum(vertices: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..vertices.len() {
        let a = vertices[i];
        let b = vertices[(i + 1) % vertices.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let vertices = prepare_polygon(points)?;
    let signed_area2 = area_sum(&vertices);
    let area2 = signed_area2.unsigned_abs();
    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    let edge_milli = edge_lengths(&vertices);
    let perimeter_milli = edge_milli.iter().copied().sum();

    let mut longest_edge = None;
    for i in 0..vertices.len() {
        longest_edge = match longest_edge {
            None => Some(i),
            Some(best) => {
                if edge_milli[i] > edge_milli[best] {
                    Some(i)
                } else if edge_milli[i] < edge_milli[best] {
                    Some(best)
                } else {
                    let ordered_pair = |index: usize| {
                        let a = vertices[index];
                        let b = vertices[(index + 1) % vertices.len()];
                        if a <= b { (a, b) } else { (b, a) }
                    };
                    if ordered_pair(i) < ordered_pair(best)
                        || (ordered_pair(i) == ordered_pair(best) && i < best)
                    {
                        Some(i)
                    } else {
                        Some(best)
                    }
                }
            }
        };
    }

    Ok(Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn point_on_segment(q: Point, a: Point, b: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let vertices = prepare_polygon(points)?;
    if queries.len() > RAW_LIMIT {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !valid_point(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..vertices.len() {
            if point_on_segment(q, vertices[i], vertices[(i + 1) % vertices.len()]) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..vertices.len() {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let side = cross(a, b, q);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > RAW_LIMIT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !valid_point(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(p, _)| *p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    if unique.is_empty() {
        return Ok(Hull {
            vertices: Vec::new(),
            source_indices: Vec::new(),
            area2: 0,
            perimeter_milli: 0,
        });
    }

    if unique.len() == 1 {
        return Ok(Hull {
            vertices: vec![unique[0].0],
            source_indices: vec![unique[0].1],
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|&(p, _)| cross(unique[0].0, unique[1].0, p) == 0);

    let hull_points: Vec<(Point, usize)> = if all_collinear {
        match mode {
            Collinear::Include => unique,
            Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
        }
    } else {
        let mut lower: Vec<(Point, usize)> = Vec::new();
        for &item in &unique {
            while lower.len() >= 2 {
                let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, item.0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                lower.pop();
            }
            lower.push(item);
        }

        let mut upper: Vec<(Point, usize)> = Vec::new();
        for &item in unique.iter().rev() {
            while upper.len() >= 2 {
                let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, item.0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                upper.pop();
            }
            upper.push(item);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_points.iter().map(|&(p, _)| p).collect();
    let source_indices: Vec<usize> = hull_points.iter().map(|&(_, i)| i).collect();
    let area2 = area_sum(&vertices).unsigned_abs();
    let perimeter_milli = edge_lengths(&vertices).iter().copied().sum();

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}
