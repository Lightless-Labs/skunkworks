use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    if start >= bytes.len() || !bytes[start].is_ascii_digit() {
        return if is_core_terminator(bytes.get(start).copied()) {
            Err(VersionError::MissingCore { at: start })
        } else {
            Err(VersionError::InvalidChar { at: start })
        };
    }

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    let field = &bytes[start..*pos];
    if field.len() > 1 && field[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in field {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((digit - b'0') as u64))
        {
            Some(v) => v,
            None => return Err(VersionError::Overflow { at: start }),
        };
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &s[ident_start..i];
        if is_pre
            && ident.len() > 1
            && ident.as_bytes().iter().all(u8::is_ascii_digit)
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(ident.to_owned());

        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start > end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return if is_core_terminator(s.as_bytes().get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let minor = parse_core_field(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return if is_core_terminator(s.as_bytes().get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let patch = parse_core_field(s, &mut pos)?;
    if pos < s.len() && !matches!(s.as_bytes()[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        let pre_start = pos + 1;
        let build_marker = s.as_bytes()[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pre_start + offset);
        let pre_end = build_marker.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        if let Some(marker) = build_marker {
            build = parse_identifiers(s, marker + 1, s.len(), false)?;
        }
    } else if pos < s.len() && s.as_bytes()[pos] == b'+' {
        build = parse_identifiers(s, pos + 1, s.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);
    match (a_numeric, b_numeric) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (ai, bi) in a.pre.iter().zip(&b.pre) {
                    let order = compare_identifiers(ai, bi);
                    if order != Ordering::Equal {
                        return order;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Operator,
    version: Version,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
        pos += 1;
    }
    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, op_len) = if token.starts_with(">=") {
            (Operator::GreaterEqual, 2)
        } else if token.starts_with("<=") {
            (Operator::LessEqual, 2)
        } else if token.starts_with('>') {
            (Operator::Greater, 1)
        } else if token.starts_with('<') {
            (Operator::Less, 1)
        } else if token.starts_with('=') {
            (Operator::Equal, 1)
        } else {
            (Operator::Equal, 0)
        };

        let operand_start = token_start + op_len;
        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

fn comparator_matches(candidate: &Version, comparator: &Comparator) -> bool {
    match comparator.op {
        Operator::Equal => compare_versions(candidate, &comparator.version) == Ordering::Equal,
        Operator::Less => compare_versions(candidate, &comparator.version) == Ordering::Less,
        Operator::LessEqual => {
            compare_versions(candidate, &comparator.version) != Ordering::Greater
        }
        Operator::Greater => {
            compare_versions(candidate, &comparator.version) == Ordering::Greater
        }
        Operator::GreaterEqual => {
            compare_versions(candidate, &comparator.version) != Ordering::Less
        }
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted = v.pre.is_empty()
            || arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && (v.major, v.minor, v.patch)
                        == (
                            comparator.version.major,
                            comparator.version.minor,
                            comparator.version.patch,
                        )
            });
        admitted && arm.iter().all(|comparator| comparator_matches(v, comparator))
    })
}
