```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    let product = units.checked_mul(unit).ok_or(Error::Overflow { index })?;
    product.checked_add(start).ok_or(Error::Overflow { index })
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().and_then(|tier| tier.up_to).is_some() {
        return Err(Error::MissingFinal);
    }

    Ok(())
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    let mut remaining = quantity;
    let mut previous = 0;
    let mut amount = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                remaining.min(width)
            }
            None => remaining,
        };

        if units > 0 {
            let band_charge = band_amount(units, tier.unit, tier.start, index)?;
            amount = amount
                .checked_add(band_charge)
                .ok_or(Error::Overflow { index })?;
            bands.push(Band {
                index,
                units,
                amount: band_charge,
            });
            remaining -= units;
        }

        if let Some(bound) = tier.up_to {
            previous = bound;
        }
    }

    Ok(Charge { amount, bands })
}

fn volume(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    for (index, tier) in tiers.iter().enumerate() {
        if tier.up_to.is_none_or(|bound| quantity <= bound) {
            let amount = band_amount(quantity, tier.unit, tier.start, index)?;
            return Ok(Charge {
                amount,
                bands: vec![Band {
                    index,
                    units: quantity,
                    amount,
                }],
            });
        }
    }

    unreachable!("validated schedules end with an unbounded tier")
}

fn charge(quantity: u64, price: &Price, mode: Mode) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                })
            } else {
                let amount = band_amount(quantity, *unit, *start, 0)?;
                Ok(Charge {
                    amount,
                    bands: vec![Band {
                        index: 0,
                        units: quantity,
                        amount,
                    }],
                })
            }
        }
        Price::Tiers(tiers) => match mode {
            Mode::Graduated => graduated(quantity, tiers),
            Mode::Volume => volume(quantity, tiers),
        },
    }
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    charge(quantity, price, Mode::Graduated)
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = charge(quantity, price, mode)?;
    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::Overflow { index: 0 })?;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```
