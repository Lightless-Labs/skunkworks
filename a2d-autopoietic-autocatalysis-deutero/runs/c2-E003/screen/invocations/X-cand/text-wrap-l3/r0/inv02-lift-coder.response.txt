#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Debug, Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Debug, Clone)]
struct Piece {
    text: String,
    is_word: bool,
}

#[derive(Debug, Clone)]
struct Line {
    indent: usize,
    pieces: Vec<Piece>,
    content_len: usize,
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn make_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current_chars = Vec::new();
    let mut current_breaks = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !current_chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current_chars),
                    breaks: std::mem::take(&mut current_breaks),
                });
            }
        } else if ch == '~' {
            current_breaks.push(current_chars.len());
        } else {
            current_chars.push(ch);
        }
    }

    if !current_chars.is_empty() {
        words.push(Word {
            chars: current_chars,
            breaks: current_breaks,
        });
    }

    words
}

fn visible_suffix(word: &Word, offset: usize) -> String {
    word.chars[offset..].iter().collect()
}

fn opportunities(word: &Word, offset: usize, room: usize) -> Vec<(usize, usize)> {
    word.breaks
        .iter()
        .copied()
        .filter(|&p| p > offset && p < word.chars.len() && p - offset + 1 <= room)
        .map(|p| (p, p - offset + 1))
        .collect()
}

fn emit_line(lines: &mut Vec<Line>, current: &mut Line) {
    if current.content_len > 0 {
        lines.push(current.clone());
    }
}

fn layout_paragraph(words: &[Word], options: &WrapOptions, lines: &mut Vec<Line>) {
    if words.is_empty() {
        return;
    }

    let mut paragraph_lines = Vec::new();
    let mut line_no = 0usize;
    let mut current = Line {
        indent: options.first_indent,
        pieces: Vec::new(),
        content_len: 0,
    };

    let mut start_new_line = |current: &mut Line, line_no: &mut usize| {
        *line_no += 1;
        *current = Line {
            indent: options.rest_indent,
            pieces: Vec::new(),
            content_len: 0,
        };
    };

    for word in words {
        let mut offset = 0usize;
        while offset < word.chars.len() {
            let capacity = options.width - current.indent;
            let nonempty = current.content_len > 0;
            let gap = usize::from(nonempty);
            let room = capacity.saturating_sub(current.content_len + gap);
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room {
                if nonempty {
                    current.pieces.push(Piece {
                        text: " ".to_string(),
                        is_word: false,
                    });
                    current.content_len += 1;
                }
                let text = visible_suffix(word, offset);
                current.content_len += suffix_len;
                current.pieces.push(Piece { text, is_word: true });
                offset = word.chars.len();
                continue;
            }

            let usable = opportunities(word, offset, room);
            let choice = usable.last().copied();

            if let Some((position, added_len)) = choice {
                let unused = room - added_len;
                let accept = if !nonempty {
                    true
                } else {
                    let baseline = capacity.saturating_sub(current.content_len);
                    unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                        < baseline.saturating_mul(baseline)
                };

                if accept {
                    if nonempty {
                        current.pieces.push(Piece {
                            text: " ".to_string(),
                            is_word: false,
                        });
                        current.content_len += 1;
                    }
                    let fragment: String = word.chars[offset..position].iter().collect();
                    current.pieces.push(Piece {
                        text: format!("{fragment}-"),
                        is_word: true,
                    });
                    current.content_len += added_len;
                    offset = position;
                    emit_line(&mut paragraph_lines, &mut current);
                    start_new_line(&mut current, &mut line_no);
                    continue;
                }
            }

            if nonempty {
                emit_line(&mut paragraph_lines, &mut current);
                start_new_line(&mut current, &mut line_no);
                continue;
            }

            let capacity = options.width - current.indent;
            if suffix_len > capacity {
                let end = offset + capacity;
                let raw: String = word.chars[offset..end].iter().collect();
                current.pieces.push(Piece {
                    text: raw,
                    is_word: true,
                });
                current.content_len = capacity;
                offset = end;
                emit_line(&mut paragraph_lines, &mut current);
                start_new_line(&mut current, &mut line_no);
            } else {
                let text = visible_suffix(word, offset);
                current.content_len += suffix_len;
                current.pieces.push(Piece { text, is_word: true });
                offset = word.chars.len();
            }
        }
    }

    emit_line(&mut paragraph_lines, &mut current);

    let final_line = paragraph_lines.len().saturating_sub(1);
    for (index, line) in paragraph_lines.iter().enumerate() {
        let word_count = line.pieces.iter().filter(|p| p.is_word).count();
        let gaps = word_count.saturating_sub(1);
        let is_final = index == final_line;

        let mut rendered = " ".repeat(line.indent);
        if options.justify && !is_final && gaps >= 1 {
            let extra = options.width.saturating_sub(line.indent + line.content_len);
            let base = extra / gaps;
            let remainder = extra % gaps;
            let mut gap_index = 0usize;
            for piece in &line.pieces {
                if piece.is_word {
                    rendered.push_str(&piece.text);
                } else {
                    let add = base + usize::from((gap_index + gaps - (index % gaps)) % gaps < remainder);
                    rendered.push_str(&" ".repeat(1 + add));
                    gap_index += 1;
                }
            }
        } else {
            for piece in &line.pieces {
                rendered.push_str(&piece.text);
            }
        }
        lines.push(Line {
            indent: 0,
            pieces: vec![Piece {
                text: rendered,
                is_word: true,
            }],
            content_len: 0,
        });
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }
        let valid_neighbor = |j: usize| {
            j < original.len()
                && original[j].1 != '~'
                && !is_separator(original[j].1)
        };
        if i == 0 || i + 1 >= original.len() || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        if ch == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut end = i + 1;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                let words = make_words(&normalized[segment_start..i]);
                layout_paragraph(&words, options, &mut output);
                for _ in 1..(end - i) {
                    output.push(Line {
                        indent: 0,
                        pieces: vec![Piece {
                            text: String::new(),
                            is_word: true,
                        }],
                        content_len: 0,
                    });
                }
                segment_start = end;
                i = end;
                continue;
            }
        }
        i += 1;
    }

    let words = make_words(&normalized[segment_start..]);
    layout_paragraph(&words, options, &mut output);

    Ok(output
        .into_iter()
        .map(|line| line.pieces.into_iter().map(|p| p.text).collect())
        .collect())
}
