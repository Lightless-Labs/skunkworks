#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;

        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }

        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.update_byte(byte);
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;

        for span in spans {
            self.update(&data[span.start..span.end]);
        }

        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();

        if bytes.len() != 19
            || bytes[0] != b'c'
            || bytes[9] != b':'
            || bytes[10] != b'a'
        {
            return Err(SnapshotError::Shape);
        }

        for offset in (1..9).chain(11..19) {
            if hex_value(bytes[offset]).is_none() {
                return Err(SnapshotError::Hex { offset });
            }
        }

        let crc = parse_hex_u32(bytes, 1);
        let adler = parse_hex_u32(bytes, 11);
        Self::from_seed(crc, adler)
    }

    fn update_byte(&mut self, byte: u8) {
        self.crc_raw = crc32_raw_byte(self.crc_raw, byte);

        let modulus = 65_521u32;
        self.adler_a = (self.adler_a + u32::from(byte)) % modulus;
        self.adler_b = (self.adler_b + self.adler_a) % modulus;
    }
}

fn crc32_raw_byte(mut crc: u32, byte: u8) -> u32 {
    crc ^= u32::from(byte);
    for _ in 0..8 {
        let low_bit = crc & 1;
        crc >>= 1;
        if low_bit != 0 {
            crc ^= 0xedb8_8320;
        }
    }
    crc
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

fn selected_len(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |total, span| {
        total + (span.end - span.start) as u64
    })
}

fn update_base(crc_raw: &mut u32, a: &mut u32, b: &mut u32, byte: u8) {
    *crc_raw = crc32_raw_byte(*crc_raw, byte);
    *a = (*a + u32::from(byte)) % 65_521;
    *b = (*b + *a) % 65_521;
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    for &byte in data {
        crc = crc32_raw_byte(crc, byte);
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 65_521;
        b = (b + a) % 65_521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | (a as u16)
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut chunks = data.chunks_exact(2);

    for word_bytes in &mut chunks {
        let word = (u16::from(word_bytes[0]) << 8) | u16::from(word_bytes[1]);
        a = (a + u64::from(word)) % 65_535;
        b = (b + a) % 65_535;
    }

    if let Some(&last) = chunks.remainder().first() {
        let word = u16::from(last) << 8;
        a = (a + u64::from(word)) % 65_535;
        b = (b + a) % 65_535;
    }

    ((b as u32) << 16) | (a as u32)
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut a = 0u64;
    let mut b = 0u64;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                a = (a + u64::from(word)) % 65_535;
                b = (b + a) % 65_535;
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        let word = u16::from(high) << 8;
        a = (a + u64::from(word)) % 65_535;
        b = (b + a) % 65_535;
    }

    Ok(((b as u32) << 16) | (a as u32))
}

fn crc16_byte(mut crc: u16, byte: u8, variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            crc ^= u16::from(byte) << 8;
            for _ in 0..8 {
                let top = crc & 0x8000;
                crc = crc.wrapping_shl(1);
                if top != 0 {
                    crc ^= 0x1021;
                }
            }
        }
        Crc16::X25 => {
            crc ^= u16::from(byte);
            for _ in 0..8 {
                let low = crc & 1;
                crc >>= 1;
                if low != 0 {
                    crc ^= 0x8408;
                }
            }
        }
    }
    crc
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for &byte in data {
        crc = crc16_byte(crc, byte, variant);
    }

    if variant == Crc16::X25 {
        crc ^ 0xffff
    } else {
        crc
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc_raw = 0xffff_ffffu32;
    let mut adler_a = 1u32;
    let mut adler_b = 0u32;
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u64;
    let mut f32_b = 0u64;
    let mut f32_pending: Option<u8> = None;
    let mut crc16_reg = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            update_base(&mut crc_raw, &mut adler_a, &mut adler_b, byte);

            f16_a = (f16_a + u32::from(byte)) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_pending.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                f32_a = (f32_a + u64::from(word)) % 65_535;
                f32_b = (f32_b + f32_a) % 65_535;
            } else {
                f32_pending = Some(byte);
            }

            crc16_reg = crc16_byte(crc16_reg, byte, variant);
        }
    }

    if let Some(high) = f32_pending {
        let word = u16::from(high) << 8;
        f32_a = (f32_a + u64::from(word)) % 65_535;
        f32_b = (f32_b + f32_a) % 65_535;
    }

    if variant == Crc16::X25 {
        crc16_reg ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc_raw ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b as u16) << 8) | (f16_a as u16),
        fletcher32: ((f32_b as u32) << 16) | (f32_a as u32),
        crc16: crc16_reg,
    })
}

fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn matrix_square(matrix: &[u32; 32]) -> [u32; 32] {
    let mut square = [0u32; 32];
    for index in 0..32 {
        square[index] = matrix_times(matrix, matrix[index]);
    }
    square
}

fn crc32_zero_byte_step(mut crc: u32) -> u32 {
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb8_8320;
        }
    }
    crc
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut operator = [0u32; 32];
    for bit in 0..32 {
        operator[bit] = crc32_zero_byte_step(1u32 << bit);
    }

    let mut length = right_len;
    let mut transformed = left;

    while length != 0 {
        if length & 1 != 0 {
            transformed = matrix_times(&operator, transformed);
        }
        length >>= 1;
        if length != 0 {
            operator = matrix_square(&operator);
        }
    }

    transformed ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total_len = 0u64;

    for (index, part) in parts.iter().enumerate() {
        total_len = total_len
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined_crc = 0u32;
    for part in parts {
        combined_crc = crc32_combine(combined_crc, part.crc32, part.len);
    }

    Ok(Part {
        crc32: combined_crc,
        len: total_len,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }

    let hashes = stream.checksum();
    let len = selected_len(spans);

    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(
    data: &[u8],
    spans: &[Span],
    token: &str,
) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let expected_len = parse_hex_u64(bytes, 4);
    let expected_crc = parse_hex_u32(bytes, 25);
    let expected_adler = parse_hex_u32(bytes, 40);

    if expected_len != selected_len(spans) {
        return Err(VerifyError::Length);
    }

    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    let actual = stream.checksum();

    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }

    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }

    Ok(())
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize) -> u32 {
    let mut value = 0u32;
    for &byte in &bytes[start..start + 8] {
        value = (value << 4) | u32::from(hex_value(byte).unwrap_or(0));
    }
    value
}

fn parse_hex_u64(bytes: &[u8], start: usize) -> u64 {
    let mut value = 0u64;
    for &byte in &bytes[start..start + 16] {
        value = (value << 4) | u64::from(hex_value(byte).unwrap_or(0));
    }
    value
}
