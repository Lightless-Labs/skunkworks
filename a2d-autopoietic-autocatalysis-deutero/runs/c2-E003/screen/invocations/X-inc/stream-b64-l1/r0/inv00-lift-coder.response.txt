#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
}

pub fn encode(input: &[u8]) -> String {
    const ALPHABET: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let n = ((chunk[0] as u32) << 16)
            | ((chunk[1] as u32) << 8)
            | (chunk[2] as u32);
        output.push(ALPHABET[((n >> 18) & 0x3f) as usize] as char);
        output.push(ALPHABET[((n >> 12) & 0x3f) as usize] as char);
        output.push(ALPHABET[((n >> 6) & 0x3f) as usize] as char);
        output.push(ALPHABET[(n & 0x3f) as usize] as char);
    }

    let tail = chunks.remainder();
    if tail.len() == 1 {
        output.push(ALPHABET[(tail[0] >> 2) as usize] as char);
        output.push(ALPHABET[((tail[0] & 0x03) << 4) as usize] as char);
        output.push_str("==");
    } else if tail.len() == 2 {
        output.push(ALPHABET[(tail[0] >> 2) as usize] as char);
        output.push(ALPHABET[(((tail[0] & 0x03) << 4) | (tail[1] >> 4)) as usize] as char);
        output.push(ALPHABET[((tail[1] & 0x0f) << 2) as usize] as char);
        output.push('=');
    }

    output
}

fn decode_digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut output = Vec::new();
    let mut group = [0u8; 4];
    let mut offsets = [0usize; 4];
    let mut len = 0usize;
    let mut padded_group = false;

    for (offset, &byte) in input.iter().enumerate() {
        if padded_group {
            return Err(Error::BadPadding { offset });
        }

        if len == 2 && group[2] == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        if byte == b'=' {
            if len < 2 {
                return Err(Error::BadPadding { offset });
            }
            group[len] = byte;
            offsets[len] = offset;
            len += 1;

            if len == 4 {
                if group[2] == b'=' {
                    output.push((group[0] << 2) | (group[1] >> 4));
                } else {
                    output.push((group[0] << 2) | (group[1] >> 4));
                    output.push((group[1] << 4) | (group[2] >> 2));
                }
                padded_group = true;
                len = 0;
            }
            continue;
        }

        let Some(digit) = decode_digit(byte) else {
            return Err(Error::InvalidByte { offset, byte });
        };

        group[len] = digit;
        offsets[len] = offset;
        len += 1;

        if len == 4 {
            output.push((group[0] << 2) | (group[1] >> 4));
            output.push((group[1] << 4) | (group[2] >> 2));
            output.push((group[2] << 6) | group[3]);
            len = 0;
        }
    }

    if len == 0 {
        return Ok(output);
    }

    if len == 3 && group[2] == b'=' {
        return Err(Error::BadPadding {
            offset: offsets[2],
        });
    }

    Err(Error::Truncated {
        offset: offsets[0],
    })
}
