#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct CandidateLine {
    text: String,
    consumed: usize,
    marker_end: bool,
    raw: bool,
    pieces: usize,
    gap: usize,
    next_word: usize,
    next_offset: usize,
}

#[derive(Clone)]
struct Plan {
    cost: u64,
    marker_lines: usize,
    lines: Vec<CandidateLine>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn checked_line_candidates(
    words: &[Word],
    word_index: usize,
    offset: usize,
    first: bool,
    capacity: usize,
    indent: usize,
) -> Vec<CandidateLine> {
    let mut result = Vec::new();
    let mut display = String::new();
    let mut consumed = 0usize;
    let mut pieces = 0usize;
    let mut wi = word_index;
    let mut off = offset;
    let mut current_display_count = 0usize;

    loop {
        if wi >= words.len() {
            break;
        }
        let word = &words[wi];
        let remaining = word.chars.len().saturating_sub(off);
        if remaining == 0 {
            wi += 1;
            off = 0;
            continue;
        }

        let space_cost = if pieces == 0 { 0 } else { 1 };
        if current_display_count + space_cost >= capacity {
            break;
        }

        let available = capacity - current_display_count - space_cost;
        let take = remaining.min(available);
        if take == 0 {
            break;
        }

        if pieces > 0 {
            display.push(' ');
            current_display_count += 1;
        }
        pieces += 1;

        for &c in &word.chars[off..off + take] {
            display.push(c);
        }
        current_display_count += take;
        consumed += take;

        if take == remaining {
            let done_next = wi + 1;
            let mut completed = display.clone();
            let mut completed_count = current_display_count;

            let end_wi = done_next;
            let mut end_wi = end_wi;
            while end_wi < words.len() && words[end_wi].chars.is_empty() {
                end_wi += 1;
            }

            result.push(CandidateLine {
                text: completed,
                consumed,
                marker_end: false,
                raw: false,
                pieces,
                gap: 0,
                next_word: end_wi,
                next_offset: 0,
            });

            // Marker endings may be selected strictly before a word's end.
            // They are considered below for each available prefix.
            let _ = completed_count;

            wi = done_next;
            off = 0;
            if wi >= words.len() {
                break;
            }
            continue;
        }

        // The current word does not fit completely. Consider marker endings
        // within the displayed prefix.
        let prefix_start = off;
        for &marker in &word.markers {
            if marker <= prefix_start || marker >= word.chars.len() {
                continue;
            }
            let prefix_len = marker - prefix_start;
            let line_used = current_display_count - take + prefix_len + 1;
            if line_used <= capacity {
                let mut text = display.clone();
                text.truncate(text.chars().count().saturating_sub(take));
                for &c in &word.chars[prefix_start..marker] {
                    text.push(c);
                }
                text.push('-');

                result.push(CandidateLine {
                    text,
                    consumed: consumed - take + prefix_len,
                    marker_end: true,
                    raw: false,
                    pieces,
                    gap: 0,
                    next_word: wi,
                    next_offset: marker,
                });
            }
        }

        // Raw splitting is legal only if no remaining marker prefix plus
        // hyphen fits in the full capacity, and only on an otherwise empty line.
        if pieces == 1 && prefix_start == offset && current_display_count == take {
            let marker_fits = word.markers.iter().any(|&marker| {
                marker > offset
                    && marker < word.chars.len()
                    && marker - offset + 1 <= capacity
            });
            if remaining > capacity && !marker_fits && take == capacity {
                result.push(CandidateLine {
                    text: display.clone(),
                    consumed,
                    marker_end: false,
                    raw: true,
                    pieces: 1,
                    gap: 0,
                    next_word: wi,
                    next_offset: off + take,
                });
            }
        }
        break;
    }

    // A line may stop after any completed word; candidates above already
    // include each such stopping point. Remove candidates that exceed capacity.
    result.retain(|line| line.text.chars().count() <= capacity);
    let _ = first;
    let _ = indent;
    result
}

fn better(a: &Plan, b: &Plan) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let av: Vec<usize> = a.lines.iter().map(|l| l.consumed).collect();
    let bv: Vec<usize> = b.lines.iter().map(|l| l.consumed).collect();
    if av != bv {
        return av > bv;
    }

    let at: Vec<&str> = a.lines.iter().map(|l| l.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|l| l.text.as_str()).collect();
    at < bt
}

fn optimize_paragraph(words: &[Word], options: &WrapOptions) -> Vec<CandidateLine> {
    use std::collections::HashMap;

    fn solve(
        words: &[Word],
        wi: usize,
        off: usize,
        first: bool,
        options: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Option<Plan>>,
    ) -> Option<Plan> {
        let mut wi = wi;
        let mut off = off;
        while wi < words.len() && off >= words[wi].chars.len() {
            wi += 1;
            off = 0;
        }
        if wi >= words.len() {
            return Some(Plan {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            });
        }

        let key = (wi, off, first);
        if let Some(found) = memo.get(&key) {
            return found.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let candidates =
            checked_line_candidates(words, wi, off, first, capacity, indent);
        let mut best: Option<Plan> = None;

        for mut line in candidates {
            let next_first = false;
            let rest = solve(
                words,
                line.next_word,
                line.next_offset,
                next_first,
                options,
                memo,
            );
            let Some(rest) = rest else { continue };

            let is_final = rest.lines.is_empty();
            let gap = options.width - indent - line.text.chars().count();
            line.gap = gap;
            let square = (gap as u64) * (gap as u64);
            let mut line_cost = if is_final { square / 4 } else { square };
            if line.marker_end {
                line_cost += options.hyphen_cost as u64;
            }
            if line.raw {
                line_cost += (2 * options.width + 1) as u64;
            }
            if !is_final && line.pieces == 1 {
                line_cost += 3;
            }

            let mut lines = Vec::with_capacity(rest.lines.len() + 1);
            lines.push(line.clone());
            lines.extend(rest.lines);
            let candidate = Plan {
                cost: line_cost + rest.cost,
                marker_lines: usize::from(line.marker_end) + rest.marker_lines,
                lines,
            };
            if best.as_ref().map_or(true, |current| better(&candidate, current)) {
                best = Some(candidate);
            }
        }

        memo.insert(key, best.clone());
        best
    }

    solve(words, 0, 0, true, options, &mut HashMap::new())
        .map(|plan| plan.lines)
        .unwrap_or_default()
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    let flush = |words: &mut Vec<Word>,
                 chars: &mut Vec<char>,
                 markers: &mut Vec<usize>| {
        if !chars.is_empty() {
            words.push(Word {
                chars: std::mem::take(chars),
                markers: std::mem::take(markers),
            });
        } else {
            markers.clear();
        }
    };

    for &c in segment {
        if is_separator(c) {
            flush(&mut words, &mut chars, &mut markers);
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    flush(&mut words, &mut chars, &mut markers);
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let left_ok = i > 0 && chars[i - 1].1 != '~' && !is_separator(chars[i - 1].1);
        let right_ok =
            i + 1 < chars.len() && chars[i + 1].1 != '~' && !is_separator(chars[i + 1].1);
        if !left_ok || !right_ok {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut j = 0usize;

    while j < normalized.len() {
        if normalized[j] == '\n' {
            let mut end = j + 1;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - j >= 2 {
                let segment = &normalized[start..j];
                let words = parse_segment(segment);
                let lines = optimize_paragraph(&words, options);
                append_rendered(&mut output, lines, options);

                for _ in 1..(end - j) {
                    output.push(String::new());
                }
                start = end;
                j = end;
                continue;
            }
        }
        j += 1;
    }

    let segment = &normalized[start..];
    let words = parse_segment(segment);
    let lines = optimize_paragraph(&words, options);
    append_rendered(&mut output, lines, options);

    Ok(output)
}

fn append_rendered(
    output: &mut Vec<String>,
    lines: Vec<CandidateLine>,
    options: &WrapOptions,
) {
    let line_count = lines.len();
    for (index, line) in lines.into_iter().enumerate() {
        let indent = if index == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let mut rendered = " ".repeat(indent);

        let justify = options.justify
            && index + 1 < line_count
            && line.pieces >= 2;
        if !justify {
            rendered.push_str(&line.text);
            output.push(rendered);
            continue;
        }

        let gaps = line.pieces - 1;
        let extra = options.width - indent - line.text.chars().count();
        let base = extra / gaps;
        let remainder = extra % gaps;
        let mut gap_index = 0usize;

        // Candidate text contains ordinary one-space interword gaps. Expand
        // them cyclically from the specified paragraph line index.
        for c in line.text.chars() {
            if c == ' ' {
                rendered.push(' ');
                rendered.push_str(&" ".repeat(base));
                let start_gap = index % gaps;
                let distance = (gap_index + gaps - start_gap) % gaps;
                if distance < remainder {
                    rendered.push(' ');
                }
                gap_index += 1;
            } else {
                rendered.push(c);
            }
        }
        output.push(rendered);
    }
}
