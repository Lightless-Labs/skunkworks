use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut ident_start = start;
    loop {
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        let numeric = bytes[ident_start..i].iter().all(|b| is_digit(*b));
        if is_pre && numeric && i - ident_start > 1 && bytes[ident_start] == b'0' {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        out.push(s[ident_start..i].to_string());
        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    if byte_at(s, pos) != Some(b'.') {
        return match byte_at(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let minor = parse_core_number(s, &mut pos)?;
    if byte_at(s, pos) != Some(b'.') {
        return match byte_at(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let patch = parse_core_number(s, &mut pos)?;

    if let Some(b) = byte_at(s, pos) {
        if b != b'-' && b != b'+' {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if byte_at(s, pos) == Some(b'-') {
        let pre_start = pos + 1;
        let build_marker = s[pre_start..].find('+').map(|n| pre_start + n);
        let pre_end = build_marker.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }
    if byte_at(s, pos) == Some(b'+') {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    let an = is_numeric_identifier(a);
    let bn = is_numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_identifier(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparisons: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<(usize, &'_ str)> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && whitespace(bytes[i]) {
            i += 1;
        }
        if i == end {
            break;
        }
        let begin = i;
        while i < end && !whitespace(bytes[i]) {
            i += 1;
        }
        result.push((begin, &s[begin..i]));
    }
    result
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Partial {
        fields: Vec<u64>,
        count: usize,
    },
}

impl Partial {
    fn numeric_count(&self) -> usize {
        match self {
            Partial::Full(_) => 3,
            Partial::Partial { count, .. } => *count,
        }
    }

    fn lower(&self) -> Version {
        match self {
            Partial::Full(v) => v.clone(),
            Partial::Partial { fields, .. } => Version {
                major: fields.first().copied().unwrap_or(0),
                minor: fields.get(1).copied().unwrap_or(0),
                patch: fields.get(2).copied().unwrap_or(0),
                pre: Vec::new(),
                build: Vec::new(),
            },
        }
    }

    fn is_full(&self) -> bool {
        matches!(self, Partial::Full(_))
    }
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let full_like = s.contains('-')
        || s.contains('+')
        || (s.split('.').count() >= 3
            && !s.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if full_like {
        return parse_full_operand(s, at).map(Partial::Full);
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut values = Vec::new();
    let mut saw_wild = false;
    for f in &fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(*f, "*" | "x" | "X") {
            saw_wild = true;
            continue;
        }
        if saw_wild
            || !f.bytes().all(is_digit)
            || (f.len() > 1 && f.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }
        let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
        values.push(n);
    }
    let count = values.len();
    Ok(Partial::Partial {
        fields: values,
        count,
    })
}

fn tuple_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn bump(v: &Version, which: usize, at: usize) -> Result<Version, RangeError> {
    let (major, minor, patch) = match which {
        0 => (
            v.major.checked_add(1).ok_or(RangeError::Syntax { at })?,
            0,
            0,
        ),
        1 => (
            v.major,
            v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?,
            0,
        ),
        _ => (
            v.major,
            v.minor,
            v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?,
        ),
    };
    let mut upper = tuple_version(major, minor, patch);
    upper.pre.push("0".to_string());
    Ok(upper)
}

fn add_cmp(comps: &mut Vec<Comparator>, anchors: &mut Vec<(u64, u64, u64)>, op: Op, v: Version) {
    if !v.pre.is_empty() {
        anchors.push((v.major, v.minor, v.patch));
    }
    comps.push(Comparator { op, version: v });
}

fn expand_partial(
    partial: Partial,
    at: usize,
    comps: &mut Vec<Comparator>,
    anchors: &mut Vec<(u64, u64, u64)>,
) -> Result<(), RangeError> {
    match partial {
        Partial::Full(v) => add_cmp(comps, anchors, Op::Eq, v),
        Partial::Partial { fields, count } => {
            if count == 0 {
                return Ok(());
            }
            let lower = tuple_version(
                fields.first().copied().unwrap_or(0),
                fields.get(1).copied().unwrap_or(0),
                fields.get(2).copied().unwrap_or(0),
            );
            add_cmp(comps, anchors, Op::Ge, lower.clone());
            if count < 3 {
                let upper = bump(&lower, if count == 1 { 0 } else { 1 }, at)?;
                comps.push(Comparator {
                    op: Op::Lt,
                    version: upper,
                });
            }
        }
    }
    Ok(())
}

fn expand_caret_tilde(
    operand: &str,
    at: usize,
    caret: bool,
    comps: &mut Vec<Comparator>,
    anchors: &mut Vec<(u64, u64, u64)>,
) -> Result<(), RangeError> {
    let p = parse_partial(operand, at)?;
    let count = p.numeric_count();
    if count == 0 {
        return Err(RangeError::Syntax { at: at - 1 });
    }
    let lower = p.lower();
    add_cmp(comps, anchors, Op::Ge, lower.clone());
    let which = if caret {
        if lower.major > 0 || count == 1 {
            0
        } else if lower.minor > 0 || count == 2 {
            1
        } else {
            2
        }
    } else if count == 1 {
        0
    } else {
        1
    };
    let upper = bump(&lower, which, at - 1)?;
    comps.push(Comparator {
        op: Op::Lt,
        version: upper,
    });
    Ok(())
}

fn expand_hyphen(
    lower: Partial,
    lower_at: usize,
    upper: Partial,
    upper_at: usize,
    comps: &mut Vec<Comparator>,
    anchors: &mut Vec<(u64, u64, u64)>,
) -> Result<(), RangeError> {
    if lower.numeric_count() == 0 {
        return Err(RangeError::Syntax { at: lower_at });
    }
    if upper.numeric_count() == 0 {
        return Err(RangeError::Syntax { at: upper_at });
    }
    let low = lower.lower();
    add_cmp(comps, anchors, Op::Ge, low);
    match upper {
        Partial::Full(v) => add_cmp(comps, anchors, Op::Le, v),
        Partial::Partial { fields, count } => {
            let high = tuple_version(
                fields.first().copied().unwrap_or(0),
                fields.get(1).copied().unwrap_or(0),
                fields.get(2).copied().unwrap_or(0),
            );
            if count == 3 {
                add_cmp(comps, anchors, Op::Le, high);
            } else {
                let next = bump(&high, if count == 1 { 0 } else { 1 }, upper_at)?;
                comps.push(Comparator {
                    op: Op::Lt,
                    version: next,
                });
            }
        }
    }
    Ok(())
}

fn parse_arm(
    s: &str,
    start: usize,
    end: usize,
) -> Result<Arm, RangeError> {
    let toks = tokens(s, start, end);
    if toks.iter().any(|(_, t)| *t == "-") {
        let minus_at = toks.iter().find(|(_, t)| *t == "-").unwrap().0;
        if toks.len() != 3 || toks[1].1 != "-" {
            return Err(RangeError::Syntax { at: minus_at });
        }
        let (la, lt) = toks[0];
        let (ua, ut) = toks[2];
        let lower = parse_partial(lt, la)?;
        let upper = parse_partial(ut, ua)?;
        let mut comparisons = Vec::new();
        let mut anchors = Vec::new();
        expand_hyphen(lower, la, upper, ua, &mut comparisons, &mut anchors)?;
        return Ok(Arm {
            comparisons,
            anchors,
        });
    }

    let mut comparisons = Vec::new();
    let mut anchors = Vec::new();
    for (at, token) in toks {
        if token.is_empty() {
            continue;
        }
        let first = token.as_bytes()[0];
        if first == b'^' || first == b'~' {
            if token.len() == 1 {
                return Err(RangeError::Syntax { at });
            }
            expand_caret_tilde(
                &token[1..],
                at + 1,
                first == b'^',
                &mut comparisons,
                &mut anchors,
            )?;
            continue;
        }
        let op_prefix = if token.starts_with(">=") {
            Some((Op::Ge, 2))
        } else if token.starts_with("<=") {
            Some((Op::Le, 2))
        } else if token.starts_with('>') {
            Some((Op::Gt, 1))
        } else if token.starts_with('<') {
            Some((Op::Lt, 1))
        } else if token.starts_with('=') {
            Some((Op::Eq, 1))
        } else {
            None
        };
        if let Some((op, len)) = op_prefix {
            let operand = &token[len..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = parse_full_operand(operand, at + len)?;
            add_cmp(&mut comparisons, &mut anchors, op, v);
            continue;
        }
        if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at });
        }
        let p = parse_partial(token, at)?;
        expand_partial(p, at, &mut comparisons, &mut anchors)?;
    }
    Ok(Arm {
        comparisons,
        anchors,
    })
}

fn is_or_token(s: &str, at: usize, end: usize) -> bool {
    s[at..end] == *"||"
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    let mut any_token = false;
    while i < bytes.len() {
        while i < bytes.len() && whitespace(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !whitespace(bytes[i]) {
            i += 1;
        }
        let end = i;
        any_token = true;
        if is_or_token(s, start, end) {
            let arm = parse_arm(s, arm_start, start)?;
            if tokens(s, arm_start, start).is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(arm);
            arm_start = end;
            while arm_start < bytes.len() && whitespace(bytes[arm_start]) {
                arm_start += 1;
            }
            if arm_start == bytes.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        }
    }
    if !any_token {
        return Err(RangeError::Empty { at: 0 });
    }
    let final_tokens = tokens(s, arm_start, s.len());
    if final_tokens.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(s, arm_start, s.len())?);
    Ok(Range {
        arms: arms
            .into_iter()
            .map(|a| a.comparisons)
            .collect(),
    })
}

fn arm_matches(v: &Version, arm: &[Comparator], anchors: &[(u64, u64, u64)]) -> bool {
    if !v.pre.is_empty() && !anchors.contains(&(v.major, v.minor, v.patch)) {
        return false;
    }
    arm.iter().all(|c| {
        let ord = compare_versions(v, &c.version);
        match c.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        // Anchors are reconstructed from explicit comparator operands.
        let anchors: Vec<_> = arm
            .iter()
            .filter(|c| !c.version.pre.is_empty())
            .map(|c| (c.version.major, c.version.minor, c.version.patch))
            .collect();
        arm_matches(v, arm, &anchors)
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    choose_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }
    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(old) => {
                let ord = compare_versions(version, &parsed[old]);
                if (choose_max && ord == Ordering::Greater)
                    || (!choose_max && ord == Ordering::Less)
                {
                    Some(i)
                } else {
                    Some(old)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
