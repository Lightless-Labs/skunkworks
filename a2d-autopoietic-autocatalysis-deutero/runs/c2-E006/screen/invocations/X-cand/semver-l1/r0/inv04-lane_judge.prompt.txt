use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    if *pos == start {
        return Err(if is_core_terminator(bytes.get(start).copied()) {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }

    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in digits {
        let n = u64::from(digit - b'0');
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(n))
            .ok_or(VersionError::Overflow { at: start })?;
    }

    Ok(value)
}

fn valid_identifier_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut identifiers = Vec::new();
    let mut id_start = start;

    loop {
        let mut id_end = id_start;
        while id_end < end && bytes[id_end] != b'.' {
            if !valid_identifier_byte(bytes[id_end]) {
                return Err(VersionError::InvalidChar { at: id_end });
            }
            id_end += 1;
        }

        if id_end == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let identifier = &s[id_start..id_end];
        if is_pre
            && identifier.bytes().all(|b| b.is_ascii_digit())
            && identifier.len() > 1
            && identifier.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        identifiers.push(identifier.to_owned());

        if id_end == end {
            break;
        }
        id_start = id_end + 1;
    }

    Ok(identifiers)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;

    if s.as_bytes().get(pos) != Some(&b'.') {
        let at = pos;
        return Err(if is_core_terminator(s.as_bytes().get(pos).copied()) {
            VersionError::MissingCore { at }
        } else {
            VersionError::InvalidChar { at }
        });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;

    if s.as_bytes().get(pos) != Some(&b'.') {
        let at = pos;
        return Err(if is_core_terminator(s.as_bytes().get(pos).copied()) {
            VersionError::MissingCore { at }
        } else {
            VersionError::InvalidChar { at }
        });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;

    let bytes = s.as_bytes();
    let at = pos;
    match bytes.get(pos).copied() {
        None => Ok(Version {
            major,
            minor,
            patch,
            pre: Vec::new(),
            build: Vec::new(),
        }),
        Some(b'-') => {
            let pre_start = pos + 1;
            let build_marker = bytes[pre_start..]
                .iter()
                .position(|&b| b == b'+')
                .map(|offset| pre_start + offset);
            let pre_end = build_marker.unwrap_or(bytes.len());

            let pre = parse_identifiers(s, pre_start, pre_end, true)?;
            let build = if let Some(plus) = build_marker {
                parse_identifiers(s, plus + 1, bytes.len(), false)?
            } else {
                Vec::new()
            };

            Ok(Version {
                major,
                minor,
                patch,
                pre,
                build,
            })
        }
        Some(b'+') => {
            let build = parse_identifiers(s, pos + 1, bytes.len(), false)?;
            Ok(Version {
                major,
                minor,
                patch,
                pre: Vec::new(),
                build,
            })
        }
        Some(_) => Err(VersionError::InvalidChar { at }),
    }
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (left, right) in a.pre.iter().zip(&b.pre) {
                    let ordering = compare_identifiers(left, right);
                    if ordering != Ordering::Equal {
                        return ordering;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Copy, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();
    let mut prerelease_cores = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (operator, operand_offset) = if token.starts_with("<=") {
            (Operator::LessEqual, 2)
        } else if token.starts_with(">=") {
            (Operator::GreaterEqual, 2)
        } else if token.starts_with('=') {
            (Operator::Equal, 1)
        } else if token.starts_with('<') {
            (Operator::Less, 1)
        } else if token.starts_with('>') {
            (Operator::Greater, 1)
        } else {
            (Operator::Equal, 0)
        };

        let operand_start = token_start + operand_offset;
        let operand = &s[operand_start..token_end];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            prerelease_cores.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { operator, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        comparators,
        prerelease_cores,
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .prerelease_cores
            .contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Equal => ordering == Ordering::Equal,
            Operator::Less => ordering == Ordering::Less,
            Operator::LessEqual => ordering != Ordering::Greater,
            Operator::Greater => ordering == Ordering::Greater,
            Operator::GreaterEqual => ordering != Ordering::Less,
        }
    })
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_core_number(bytes: &[u8], start: usize) -> Result<(u64, usize), VersionError> {
    let mut end = start;
    while end < bytes.len() && is_digit(bytes[end]) {
        end += 1;
    }

    if end == start {
        return Err(match bytes.get(start) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &b in &bytes[start..end] {
        let digit = (b - b'0') as u64;
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(digit))
            .ok_or(VersionError::Overflow { at: start })?;
    }

    Ok((value, end))
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    check_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut identifiers = Vec::new();
    let mut ident_start = start;

    loop {
        let mut pos = ident_start;
        while pos < end && bytes[pos] != b'.' {
            if !is_identifier_char(bytes[pos]) {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        if check_numeric_leading_zero {
            let ident = &bytes[ident_start..pos];
            if ident.iter().all(|b| is_digit(*b))
                && ident.len() > 1
                && ident[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
        }

        identifiers.push(s[ident_start..pos].to_owned());

        if pos == end {
            break;
        }
        ident_start = pos + 1;
    }

    Ok(identifiers)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let (major, mut pos) = parse_core_number(bytes, 0)?;

    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(match bytes.get(pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let (minor, next) = parse_core_number(bytes, pos)?;
    pos = next;

    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(match bytes.get(pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let (patch, mut pos) = parse_core_number(bytes, pos)?;

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let build_sep = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pre_start + offset);

        let pre_end = build_sep.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;

        if pos < bytes.len() {
            // The byte at pos is the plus that terminates prerelease.
            let build_start = pos + 1;
            build = parse_identifiers(s, build_start, bytes.len(), false)?;
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {

}

fn compare_pre_identifier(a: &str, b: &str) -> Ordering {

        (true, true) => compare_numeric_identifiers(a, b),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),

}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {

            prerelease_cores.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { op, version });

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        comparators,
        prerelease_cores,
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .prerelease_cores
            .contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.op {
            Op::Eq => ordering == Ordering::Equal,
            Op::Lt => ordering == Ordering::Less,
            Op::Le => ordering != Ordering::Greater,
            Op::Gt => ordering == Ordering::Greater,
            Op::Ge => ordering != Ordering::Less,
        }
    })
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn missing_core_byte(b: u8) -> bool {
    matches!(b, b'.' | b'-' | b'+')
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for (field, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            return Err(if pos == bytes.len() || missing_core_byte(bytes[pos]) {
                VersionError::MissingCore { at: start }
            } else {
                VersionError::InvalidChar { at: start }
            });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut number = 0u64;
        for &digit in &bytes[start..pos] {
            let digit = (digit - b'0') as u64;
            number = number
                .checked_mul(10)
                .and_then(|n| n.checked_add(digit))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *value = number;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        loop {
            let start = pos;
            while pos < bytes.len() && bytes[pos] != b'.' && bytes[pos] != b'+' {
                if !bytes[pos].is_ascii_alphanumeric() && bytes[pos] != b'-' {
                    return Err(VersionError::InvalidChar { at: pos });
                }
                pos += 1;
            }

            if start == pos {
                return Err(VersionError::EmptyIdentifier { at: start });
            }

            let identifier = &s[start..pos];
            let is_numeric = identifier.bytes().all(|b| b.is_ascii_digit());
            if is_numeric && identifier.len() > 1 && identifier.as_bytes()[0] == b'0' {
                return Err(VersionError::LeadingZero { at: start });
            }
            pre.push(identifier.to_owned());

            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
                continue;
            }
            break;
        }
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        loop {
            let start = pos;
            while pos < bytes.len() && bytes[pos] != b'.' {
                if !bytes[pos].is_ascii_alphanumeric() && bytes[pos] != b'-' {
                    return Err(VersionError::InvalidChar { at: pos });
                }
                pos += 1;
            }

            if start == pos {
                return Err(VersionError::EmptyIdentifier { at: start });
            }

            build.push(s[start..pos].to_owned());

            if pos < bytes.len() {
                pos += 1;
                continue;
            }
            break;
        }
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn is_numeric_identifier(identifier: &str) -> bool {
    !identifier.is_empty() && identifier.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (left, right) in a.pre.iter().zip(&b.pre) {
                    let order = compare_identifiers(left, right);
                    if order != Ordering::Equal {
                        return order;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    prerelease_cores: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Copy, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut comparators = Vec::new();
    let mut prerelease_cores = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];
        let token_bytes = token.as_bytes();

        let (operator, operand_offset) = if token_bytes.starts_with(b">=") {
            (Operator::GreaterEqual, 2)
        } else if token_bytes.starts_with(b"<=") {
            (Operator::LessEqual, 2)
        } else if token_bytes.starts_with(b">") {
            (Operator::Greater, 1)
        } else if token_bytes.starts_with(b"<") {
            (Operator::Less, 1)
        } else if token_bytes.starts_with(b"=") {
            (Operator::Equal, 1)
        } else if token_bytes.first().is_some_and(u8::is_ascii_digit) {
            (Operator::Equal, 0)
        } else {
            return Err(RangeError::Syntax { at: token_start });
        };

        if operand_offset >= token_bytes.len()
            || !token_bytes[operand_offset].is_ascii_digit()
        {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand_start = token_start + operand_offset;
        let version = parse_version(&s[operand_start..token_end]).map_err(|error| {
            RangeError::Version {
                at: operand_start,
                error,
            }
        })?;

        if !version.pre.is_empty() {
            prerelease_cores.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { operator, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        comparators,
        prerelease_cores,
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .prerelease_cores
            .contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Equal => ordering == Ordering::Equal,
            Operator::Less => ordering == Ordering::Less,
            Operator::LessEqual => ordering != Ordering::Greater,
            Operator::Greater => ordering == Ordering::Greater,
            Operator::GreaterEqual => ordering != Ordering::Less,
        }
    })
}
