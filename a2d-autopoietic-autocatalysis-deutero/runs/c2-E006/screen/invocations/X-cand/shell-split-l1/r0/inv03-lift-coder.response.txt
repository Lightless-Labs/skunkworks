#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy)]
    enum Quote {
        None,
        Single(usize),
        Double(usize),
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match quote {
            Quote::None => match ch {
                '\'' => {
                    active = true;
                    quote = Quote::Single(pos);
                }
                '"' => {
                    active = true;
                    quote = Quote::Double(pos);
                }
                '\\' => {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                }
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                _ => {
                    active = true;
                    word.push(ch);
                }
            },
            Quote::Single(open_pos) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    let _ = open_pos;
                    word.push(ch);
                }
            }
            Quote::Double(open_pos) => match ch {
                '"' => quote = Quote::None,
                '\\' => {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    match next {
                        '"' | '\\' | '$' => word.push(next),
                        '\n' => {}
                        _ => {
                            word.push('\\');
                            word.push(next);
                        }
                    }
                    let _ = open_pos;
                }
                _ => word.push(ch),
            },
        }
    }

    match quote {
        Quote::None => {}
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
