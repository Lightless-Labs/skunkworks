fn saturating_mul_add(a: usize, b: usize, c: usize) -> usize {
    a.saturating_mul(b).saturating_add(c)

fn validate_cell(cell: &str, row: usize, column: usize) -> Result<(), TableError> {
    let mut chars = cell.char_indices().peekable();
    while let Some((byte, ch)) = chars.next() {
        if ch == '\r' {
            if matches!(chars.peek(), Some((_, '\n'))) {
                chars.next();
                continue;
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }
        if ch == '\n' {
            continue;
        }
        if ch == '\u{2028}' || ch == '\u{2029}' || ch.is_control() {
            return Err(TableError::InvalidChar { row, column, byte });
        }
    }
    Ok(())
}

fn tokenize_line(line: &str, markdown: bool) -> Vec<Token> {
    let mut tokens = Vec::new();
    for ch in line.chars() {
        match (markdown, ch) {
            (true, '|') => tokens.push(Token {
                text: "\\|".to_owned(),
                width: 2,
            }),
            (true, '\\') => tokens.push(Token {
                text: "\\\\".to_owned(),
                width: 2,
            }),
            _ => tokens.push(Token {
                text: ch.to_string(),
                width: 1,
            }),
        }
    }
    tokens
}

fn split_and_tokenize(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|part| {
            let trimmed = part.trim_end_matches(' ');
            tokenize_line(trimmed, markdown)
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().fold(0usize, |sum, token| sum.saturating_add(token.width))
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.to_vec()];
            }

            let available = width.saturating_sub(1);
            let mut kept = Vec::new();
            let mut used = 0usize;
            for token in tokens {
                if used.saturating_add(token.width) > available {
                    break;
                }
                used += token.width;
                kept.push(token.clone());
            }
            kept.push(Token {
                text: "…".to_owned(),
                width: 1,
            });
            vec![kept]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0usize;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0usize;
                while end < tokens.len()
                    && used.saturating_add(tokens[end].width) <= width
                {
                    used += tokens[end].width;
                    end += 1;
                }
                // All tokens have width at most two and valid mode minima are at least one;
                // Markdown's escaped tokens fit because its minimum width is three.
                if end == start {
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }
            result
        }
    }
}

fn render_fragment(
    tokens: &[Token],
    width: usize,
    alignment: &Alignment,
    omit_right_padding: bool,
) -> String {
    let content_width = token_width(tokens);
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    let mut result = String::new();
    result.extend(std::iter::repeat_n(' ', left));
    for token in tokens {
        result.push_str(&token.text);
    }
    if !omit_right_padding {
        result.extend(std::iter::repeat_n(' ', right));
    }
    result
}

fn append_grid_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width.saturating_add(2)));
        out.push('+');
    }
    out.push('\n');
}

fn append_data_line(
    out: &mut String,
    fragments: &[Vec<Token>],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for i in 0..widths.len() {
                out.push(' ');
                out.push_str(&render_fragment(&fragments[i], widths[i], &alignments[i], false));
                out.push(' ');
                out.push('|');
            }
            out.push('\n');
        }
        Mode::Compact => {
            for i in 0..widths.len() {
                if i > 0 {
                    out.push_str("  ");
                }
                out.push_str(&render_fragment(
                    &fragments[i],
                    widths[i],
                    &alignments[i],
                    i + 1 == widths.len(),
                ));
            }
            out.push('\n');
        }
    }
}

fn append_markdown_marker(out: &mut String, widths: &[usize], alignments: &[Alignment]) {
    out.push('|');
    for (i, &width) in widths.iter().enumerate() {
        out.push(' ');
        match alignments[i] {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width.saturating_sub(1)));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width.saturating_sub(1)));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width.saturating_sub(2)));
                out.push(':');
            }
        }
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column_width = if matches!(options.mode, Mode::Markdown) {
        3
    } else {
        1
    };
    let overhead = match options.mode {
        Mode::Grid | Mode::Markdown => saturating_mul_add(3, n, 1),
        Mode::Compact => n.saturating_sub(1).saturating_mul(2),
    };
    let minimum_total = overhead.saturating_add(n.saturating_mul(minimum_column_width));
    if options.max_width != 0 && options.max_width < minimum_total {
        return Err(TableError::TooNarrow {
            minimum: minimum_total,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            validate_cell(cell, row, column)?;
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let transformed: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| split_and_tokenize(cell, markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_column_width; n];
    for row in &transformed {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let available = options.max_width.saturating_sub(overhead);
        let mut total = widths.iter().fold(0usize, |sum, width| sum.saturating_add(*width));
        while total > available {
            let mut candidate: Option<usize> = None;
            for i in 0..n {
                if widths[i] > minimum_column_width
                    && candidate.map_or(true, |best| widths[i] >= widths[best])
                {
                    candidate = Some(i);
                }
            }
            if let Some(column) = candidate {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut out, &widths);
    }

    for (row_index, row) in transformed.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<Vec<Token>>> = Vec::with_capacity(n);
        let mut row_height = 1usize;
        for column in 0..n {
            let mut combined = Vec::new();
            for line in &row[column] {
                combined.extend(fragments(line, widths[column], &overflows[column]));
            }
            row_height = row_height.max(combined.len());
            cell_fragments.push(combined);
        }

        for fragment_index in 0..row_height {
            let line: Vec<Vec<Token>> = (0..n)
                .map(|column| {
                    cell_fragments[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            append_data_line(&mut out, &line, &widths, &alignments, &options.mode);
        }

        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            append_markdown_marker(&mut out, &widths, &alignments);
        } else if matches!(options.mode, Mode::Grid)
            && options.header
            && row_index == 0
            && rows.len() > 1
        {
            append_grid_border(&mut out, &widths);
        }
    }

    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut out, &widths);
    }

    Ok(out)
}

    tokens.iter().map(|token| token.width).sum()
}

fn fragment_lines(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.iter().map(|token| token.text.as_str()).collect()];
            }

            let mut result = String::new();
            let mut used = 0usize;
            let limit = width.saturating_sub(1);

            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                result.push_str(&token.text);
                used += token.width;
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0usize;

            while start < tokens.len() {
                let mut end = start;
                let mut used = 0usize;

                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }

                let fragment = tokens[start..end]
                    .iter()
                    .map(|token| token.text.as_str())
                    .collect();
                result.push(fragment);
                start = end;
            }

            result
        }
    }
}

fn scalar_len(text: &str) -> usize {
    text.chars().count()
}

fn align(text: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let content_width = scalar_len(text);
    let extra = width.saturating_sub(content_width);

    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    let right = if omit_right { 0 } else { right };
    let mut result = String::with_capacity(text.len() + left + right);
    result.extend(std::iter::repeat_n(' ', left));
    result.push_str(text);
    result.extend(std::iter::repeat_n(' ', right));
    result
}

fn border(widths: &[usize], output: &mut String) {
    output.push('+');
    for &width in widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');
}

fn render_data_line(
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
    output: &mut String,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            output.push('|');
            for column in 0..widths.len() {
                output.push(' ');
                output.push_str(&align(
                    &fragments[column],
                    widths[column],
                    &alignments[column],
                    false,
                ));
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
        Mode::Compact => {
            for column in 0..widths.len() {
                if column > 0 {
                    output.push_str("  ");
                }
                output.push_str(&align(
                    &fragments[column],
                    widths[column],
                    &alignments[column],
                    column + 1 == widths.len(),
                ));
            }
            output.push('\n');
        }
    }
}

fn render_markdown_marker(widths: &[usize], alignments: &[Alignment], output: &mut String) {
    output.push('|');
    for (column, &width) in widths.iter().enumerate() {
        output.push(' ');
        match alignments[column] {
            Alignment::Left => {
                output.push(':');
                output.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                output.extend(std::iter::repeat_n('-', width - 1));
                output.push(':');
            }
            Alignment::Center => {
                output.push(':');
                output.extend(std::iter::repeat_n('-', width - 2));
                output.push(':');
            }
        }
        output.push(' ');
        output.push('|');
    }
    output.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let columns = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != columns {
            return Err(TableError::Ragged {
                row: row_index,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != columns {
        return Err(TableError::OverflowCount {
            expected: columns,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_width = if options.mode == Mode::Markdown { 3 } else { 1 };
    let fixed = overhead(&options.mode, columns);
    let minimum = fixed.saturating_add(columns.saturating_mul(minimum_width));
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut byte = 0usize;
            let mut chars = cell.chars().peekable();
            while let Some(c) = chars.next() {
                if c == '\r' {
                    if chars.peek() == Some(&'\n') {
                        chars.next();
                        byte += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                if is_invalid(c) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                byte += c.len_utf8();
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; columns]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; columns]
    } else {
        options.overflow.clone()
    };

    let normalized: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| {
                    let normalized = cell.replace("\r\n", "\n");
                    split_cell(&normalized, options.mode == Mode::Markdown)
                })
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_width; columns];
    for row in &normalized {
        for (column, logical_lines) in row.iter().enumerate() {
            for line in logical_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        loop {
            let Some(total) = checked_total(&widths, fixed) else {
                break;
            };
            if total <= options.max_width {
                break;
            }

            let candidate = (0..columns)
                .rev()
                .find(|&column| widths[column] > minimum_width)
                .filter(|&column| {
                    widths[column]
                        == (0..columns)
                            .filter(|&i| widths[i] > minimum_width)
                            .map(|i| widths[i])
                            .max()
                            .unwrap_or(0)
                });

            match candidate {
                Some(column) => widths[column] -= 1,
                None => break,
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        border(&widths, &mut output);
    }

    for (row_index, row) in normalized.iter().enumerate() {
        let rendered: Vec<Vec<String>> = row
            .iter()
            .enumerate()
            .map(|(column, logical_lines)| {
                logical_lines
                    .iter()
                    .flat_map(|line| fragment_lines(line, widths[column], &overflows[column]))
                    .collect()
            })
            .collect();

        let height = rendered.iter().map(Vec::len).max().unwrap_or(1);
        for fragment_index in 0..height {
            let fragments: Vec<String> = rendered
                .iter()
                .map(|cell| cell.get(fragment_index).cloned().unwrap_or_default())
                .collect();
            render_data_line(
                &fragments,
                &widths,
                &alignments,
                &options.mode,
                &mut output,
            );
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            border(&widths, &mut output);
        }
        if options.mode == Mode::Markdown && options.header && row_index == 0 {
            render_markdown_marker(&widths, &alignments, &mut output);
        }
    }

    if options.mode == Mode::Grid {
        border(&widths, &mut output);
    }

    Ok(output)
}

fn normalize_and_validate(
    cell: &str,
    row: usize,
    column: usize,
) -> Result<String, TableError> {
    let chars: Vec<(usize, char)> = cell.char_indices().collect();
    let mut normalized = String::with_capacity(cell.len());
    let mut i = 0;

    while i < chars.len() {
        let (byte, ch) = chars[i];
        if ch == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
                continue;
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }

        if ch != '\n' && (ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}') {
            return Err(TableError::InvalidChar { row, column, byte });
        }

        normalized.push(ch);
        i += 1;
    }

    Ok(normalized)
}

fn lines_to_tokens(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    cell.split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for ch in trimmed.chars() {
                match (markdown, ch) {
                    (true, '|') => tokens.push(Token {
                        text: "\\|".to_owned(),
                        width: 2,
                    }),
                    (true, '\\') => tokens.push(Token {
                        text: "\\\\".to_owned(),
                        width: 2,
                    }),
                    _ => tokens.push(Token {
                        text: ch.to_string(),
                        width: 1,
                    }),
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()

fn fragment_line(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }

            let prefix_limit = width.saturating_sub(1);
            let mut used = 0;
            let mut result = String::new();
            for token in tokens {
                if used + token.width > prefix_limit {
                    break;
                }
                used += token.width;
                result.push_str(&token.text);
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn aligned(content: &str, width: usize, alignment: &Alignment) -> (String, usize) {
    let content_width = content.chars().count();
    let extra = width.saturating_sub(content_width);
    let left = match alignment {
        Alignment::Left => 0,
        Alignment::Right => extra,
        Alignment::Center => extra / 2,
    };
    let right = extra - left;
    let mut out = String::new();
    out.extend(std::iter::repeat_n(' ', left));
    out.push_str(content);
    out.extend(std::iter::repeat_n(' ', right));
    (out, right)
}

fn data_line(
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut line = String::new();
    match mode {
        Mode::Grid | Mode::Markdown => {
            line.push('|');
            for i in 0..widths.len() {
                let (content, _) = aligned(&fragments[i], widths[i], &alignments[i]);
                line.push(' ');
                line.push_str(&content);
                line.push(' ');
                line.push('|');
            }
        }
        Mode::Compact => {
            for i in 0..widths.len() {
                if i != 0 {
                    line.push_str("  ");
                }
                let (content, right_padding) = aligned(&fragments[i], widths[i], &alignments[i]);
                if i + 1 == widths.len() {
                    let keep = content.chars().count().saturating_sub(right_padding);
                    line.extend(content.chars().take(keep));
                } else {
                    line.push_str(&content);
                }
            }
        }
    }
    line.push('\n');
    line
}

fn markdown_separator(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut line = String::from("|");

        let width = widths[i];
        let marker = match alignments[i] {
            Alignment::Left => format!(":{} ", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        line.push(' ');
        line.push_str(marker.trim_end_matches(' '));
        line.push(' ');
        line.push('|');
    }
    line.push('\n');
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2usize.saturating_mul(n.saturating_sub(1)),
        Mode::Grid | Mode::Markdown => 3usize.saturating_mul(n).saturating_add(1),
    };
    let minimum = overhead.saturating_add(n.saturating_mul(minimum_column));
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let mut normalized = Vec::with_capacity(rows.len());
    for (row_index, row) in rows.iter().enumerate() {
        let mut normalized_row = Vec::with_capacity(n);
        for (column_index, cell) in row.iter().enumerate() {
            normalized_row.push(normalize_and_validate(cell, row_index, column_index)?);
        }
        normalized.push(normalized_row);
    }

    let cells: Vec<Vec<Vec<Vec<Token>>>> = normalized
        .iter()
        .map(|row| row.iter().map(|cell| lines_to_tokens(cell, markdown)).collect())
        .collect();

    let mut widths = vec![minimum_column; n];
    for row in &cells {
        for (column, logical_lines) in row.iter().enumerate() {
            for line in logical_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total = widths.iter().fold(overhead, |sum, &w| sum.saturating_add(w));
        while total > options.max_width {
            let mut candidate = None;
            let mut largest = 0;
            for i in 0..n {
                if widths[i] > minimum_column && (candidate.is_none() || widths[i] >= largest) {
                    candidate = Some(i);
                    largest = widths[i];
                }
            }
            if let Some(i) = candidate {
                widths[i] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in cells.iter().enumerate() {
        let mut rendered_cells: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut row_height = 1;
        for column in 0..n {
            let mut fragments = Vec::new();
            for logical_line in &row[column] {
                fragments.extend(fragment_line(
                    logical_line,
                    widths[column],
                    &overflows[column],
                ));
            }

            rendered_cells.push(fragments);
        }

        for fragment_index in 0..row_height {
            let fragments: Vec<String> = (0..n)
                .map(|column| {
                    rendered_cells[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();

                &fragments,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

            output.push_str(&border(&widths));
        }
        if options.mode == Mode::Markdown && options.header && row_index == 0 {
            output.push_str(&markdown_separator(&widths, &alignments));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    Ok(output)
}
