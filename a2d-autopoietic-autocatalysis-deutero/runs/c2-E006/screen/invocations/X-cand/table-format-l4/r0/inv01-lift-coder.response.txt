#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn normalize_and_validate(
    cell: &str,
    row: usize,
    column: usize,
) -> Result<String, TableError> {
    let chars: Vec<(usize, char)> = cell.char_indices().collect();
    let mut normalized = String::with_capacity(cell.len());
    let mut i = 0;

    while i < chars.len() {
        let (byte, ch) = chars[i];
        if ch == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
                continue;
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }

        if ch != '\n' && (ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}') {
            return Err(TableError::InvalidChar { row, column, byte });
        }

        normalized.push(ch);
        i += 1;
    }

    Ok(normalized)
}

fn lines_to_tokens(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    cell.split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for ch in trimmed.chars() {
                match (markdown, ch) {
                    (true, '|') => tokens.push(Token {
                        text: "\\|".to_owned(),
                        width: 2,
                    }),
                    (true, '\\') => tokens.push(Token {
                        text: "\\\\".to_owned(),
                        width: 2,
                    }),
                    _ => tokens.push(Token {
                        text: ch.to_string(),
                        width: 1,
                    }),
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragment_line(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }

            let prefix_limit = width.saturating_sub(1);
            let mut used = 0;
            let mut result = String::new();
            for token in tokens {
                if used + token.width > prefix_limit {
                    break;
                }
                used += token.width;
                result.push_str(&token.text);
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn aligned(content: &str, width: usize, alignment: &Alignment) -> (String, usize) {
    let content_width = content.chars().count();
    let extra = width.saturating_sub(content_width);
    let left = match alignment {
        Alignment::Left => 0,
        Alignment::Right => extra,
        Alignment::Center => extra / 2,
    };
    let right = extra - left;
    let mut out = String::new();
    out.extend(std::iter::repeat_n(' ', left));
    out.push_str(content);
    out.extend(std::iter::repeat_n(' ', right));
    (out, right)
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for &width in widths {
        line.extend(std::iter::repeat_n('-', width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

fn data_line(
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut line = String::new();
    match mode {
        Mode::Grid | Mode::Markdown => {
            line.push('|');
            for i in 0..widths.len() {
                let (content, _) = aligned(&fragments[i], widths[i], &alignments[i]);
                line.push(' ');
                line.push_str(&content);
                line.push(' ');
                line.push('|');
            }
        }
        Mode::Compact => {
            for i in 0..widths.len() {
                if i != 0 {
                    line.push_str("  ");
                }
                let (content, right_padding) = aligned(&fragments[i], widths[i], &alignments[i]);
                if i + 1 == widths.len() {
                    let keep = content.chars().count().saturating_sub(right_padding);
                    line.extend(content.chars().take(keep));
                } else {
                    line.push_str(&content);
                }
            }
        }
    }
    line.push('\n');
    line
}

fn markdown_separator(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut line = String::from("|");
    for i in 0..widths.len() {
        let width = widths[i];
        let marker = match alignments[i] {
            Alignment::Left => format!(":{} ", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        line.push(' ');
        line.push_str(marker.trim_end_matches(' '));
        line.push(' ');
        line.push('|');
    }
    line.push('\n');
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2usize.saturating_mul(n.saturating_sub(1)),
        Mode::Grid | Mode::Markdown => 3usize.saturating_mul(n).saturating_add(1),
    };
    let minimum = overhead.saturating_add(n.saturating_mul(minimum_column));
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let mut normalized = Vec::with_capacity(rows.len());
    for (row_index, row) in rows.iter().enumerate() {
        let mut normalized_row = Vec::with_capacity(n);
        for (column_index, cell) in row.iter().enumerate() {
            normalized_row.push(normalize_and_validate(cell, row_index, column_index)?);
        }
        normalized.push(normalized_row);
    }

    let markdown = options.mode == Mode::Markdown;
    let cells: Vec<Vec<Vec<Vec<Token>>>> = normalized
        .iter()
        .map(|row| row.iter().map(|cell| lines_to_tokens(cell, markdown)).collect())
        .collect();

    let mut widths = vec![minimum_column; n];
    for row in &cells {
        for (column, logical_lines) in row.iter().enumerate() {
            for line in logical_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total = widths.iter().fold(overhead, |sum, &w| sum.saturating_add(w));
        while total > options.max_width {
            let mut candidate = None;
            let mut largest = 0;
            for i in 0..n {
                if widths[i] > minimum_column && (candidate.is_none() || widths[i] >= largest) {
                    candidate = Some(i);
                    largest = widths[i];
                }
            }
            if let Some(i) = candidate {
                widths[i] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in cells.iter().enumerate() {
        let mut rendered_cells: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut row_height = 1;
        for column in 0..n {
            let mut fragments = Vec::new();
            for logical_line in &row[column] {
                fragments.extend(fragment_line(
                    logical_line,
                    widths[column],
                    &overflows[column],
                ));
            }
            row_height = row_height.max(fragments.len());
            rendered_cells.push(fragments);
        }

        for fragment_index in 0..row_height {
            let fragments: Vec<String> = (0..n)
                .map(|column| {
                    rendered_cells[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            output.push_str(&data_line(
                &fragments,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
        if options.mode == Mode::Markdown && options.header && row_index == 0 {
            output.push_str(&markdown_separator(&widths, &alignments));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    Ok(output)
}
