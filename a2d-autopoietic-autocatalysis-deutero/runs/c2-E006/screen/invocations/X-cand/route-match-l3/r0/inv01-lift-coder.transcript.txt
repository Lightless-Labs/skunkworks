fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)

}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let body = &input[1..];
    let trailing = body.ends_with('/');
    let content = if trailing {
        &body[..body.len() - 1]
    } else {
        body
    };

    let raw_segments: Vec<&str> = content.split('/').collect();
    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();
    let mut offset = 1usize;

    for (i, raw) in raw_segments.iter().enumerate() {
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: offset });
        }

        if let Some(name) = raw.strip_prefix(':') {
            validate_capture_name(name, offset)?;
            if !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: offset });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            validate_capture_name(name, offset)?;
            if i + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: offset });
            }
            if !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: offset });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            segments.push(Segment::Static(decode_segment(raw, offset, true)?));
        }

        offset += raw.len() + 1;
    }

    Ok((segments, trailing))
}

fn parse_path(input: &str) -> Result<ParsedPath, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let body = &input[1..];
    let trailing = body.ends_with('/');
    let content = if trailing {
        &body[..body.len() - 1]
    } else {
        body
    };

    let mut segments = Vec::new();
    let mut offset = 1usize;
    for raw in content.split('/') {
        if raw.is_empty() {
            return Err(Error::InvalidPath { at: offset });
        }
        segments.push(decode_segment(raw, offset, false)?);
        offset += raw.len() + 1;
    }

    Ok(ParsedPath { segments, trailing })
}

fn validate_capture_name(name: &str, at: usize) -> Result<(), Error> {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || !bytes[0].is_ascii_lowercase()
        || !bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
    {
        return Err(Error::InvalidPattern { at });
    }
    Ok(())
}

fn decode_segment(raw: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return invalid_segment(pattern, base + i);
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return invalid_segment(pattern, base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return invalid_segment(pattern, base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn invalid_segment<T>(pattern: bool, at: usize) -> Result<T, Error> {
    if pattern {
        Err(Error::InvalidPattern { at })
    } else {
        Err(Error::InvalidPath { at })
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0usize;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    (path_index == path.len()).then_some(params)
}
