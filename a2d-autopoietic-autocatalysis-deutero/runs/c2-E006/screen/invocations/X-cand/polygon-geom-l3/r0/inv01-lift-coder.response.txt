#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const MAX_INPUT: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_INPUT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points[points.len() - 1] == points[0] {
        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    Ok(&points[..effective_len])
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_sqrt(n: u128) -> u128 {
    // Find floor(sqrt(n)) by binary search without overflowing in the comparison.
    let mut lo = 0u128;
    let mut hi = 1u128 << 64;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid.max(1) {
            lo = mid;
        } else {
            hi = mid;
        }
    }

    // Round to nearest integer, ties upward. Compare n with (lo + 1/2)^2
    // using 4*n versus (2*lo + 1)^2. Values used here fit u128.
    let threshold = 2 * lo + 1;
    if 4 * n >= threshold * threshold {
        lo + 1
    } else {
        lo
    }
}

fn edge_millimeters(a: Point, b: Point) -> u64 {
    let scaled = squared_distance(a, b) * 1_000_000;
    rounded_sqrt(scaled) as u64
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare_polygon(points)?;
    let signed = signed_area2(points);
    let area = signed.unsigned_abs();

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        let length = edge_millimeters(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(best) => {
                if length != edge_milli[best] {
                    length > edge_milli[best]
                } else {
                    let current_pair = if a <= b { (a, b) } else { (b, a) };
                    let best_a = points[best];
                    let best_b = points[(best + 1) % points.len()];
                    let best_pair = if best_a <= best_b {
                        (best_a, best_b)
                    } else {
                        (best_b, best_a)
                    };
                    current_pair < best_pair
                        || (current_pair == best_pair && i < best)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    let orientation = if signed > 0 {
        Orientation::CounterClockwise
    } else if signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2: signed,
        area2: area,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && (q.x as i128) >= (a.x.min(b.x) as i128)
        && (q.x as i128) <= (a.x.max(b.x) as i128)
        && (q.y as i128) >= (a.y.min(b.y) as i128)
        && (q.y as i128) <= (a.y.max(b.y) as i128)
}

fn classify_one(points: &[Point], query: Point) -> Location {
    let mut owner = None;
    for i in 0..points.len() {
        if on_segment(points[i], points[(i + 1) % points.len()], query) {
            owner = Some(i);
            break;
        }
    }
    if let Some(edge) = owner {
        return Location::Boundary { edge };
    }

    let mut inside = false;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        if (a.y > query.y) != (b.y > query.y) {
            let side = cross(a, b, query);
            if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                inside = !inside;
            }
        }
    }

    if inside {
        Location::Inside
    } else {
        Location::Outside
    }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let points = prepare_polygon(points)?;

    if queries.len() > MAX_INPUT {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    Ok(queries
        .iter()
        .copied()
        .map(|query| classify_one(points, query))
        .collect())
}

fn chain(points: &[(Point, usize)], include_collinear: bool) -> Vec<(Point, usize)> {
    let mut result: Vec<(Point, usize)> = Vec::new();

    for &point in points {
        while result.len() >= 2 {
            let turn = cross(result[result.len() - 2].0, result[result.len() - 1].0, point.0);
            if turn > 0 || (include_collinear && turn == 0) {
                break;
            }
            result.pop();
        }
        result.push(point);
    }

    result
}

fn hull_metrics(vertices: &[Point]) -> (u128, u64) {
    if vertices.len() < 2 {
        return (0, 0);
    }

    let area = if vertices.len() >= 3 {
        signed_area2(vertices).unsigned_abs()
    } else {
        0
    };

    let mut perimeter = 0u64;
    for i in 0..vertices.len() {
        perimeter += edge_millimeters(vertices[i], vertices[(i + 1) % vertices.len()]);
    }
    (area, perimeter)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_INPUT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = unique.iter().find(|entry| entry.0 == point) {
            let _ = existing;
        } else {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|entry| entry.0);

    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|entry| entry.0).collect();
        let source_indices: Vec<usize> = unique.iter().map(|entry| entry.1).collect();
        let (area2, perimeter_milli) = hull_metrics(&vertices);
        return Ok(Hull {
            vertices,
            source_indices,
            area2,
            perimeter_milli,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|entry| cross(unique[0].0, unique[1].0, entry.0) == 0);

    let hull_points: Vec<(Point, usize)> = if all_collinear {
        match mode {
            Collinear::Include => unique,
            Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
        }
    } else {
        let include = mode == Collinear::Include;
        let mut lower = chain(&unique, include);
        let mut upper_input = unique.clone();
        upper_input.reverse();
        let mut upper = chain(&upper_input, include);

        lower.pop();
        upper.pop();
        lower.extend(upper);

        // For Include, the monotone-chain construction with collinear turns
        // can retain interior collinear points twice at the ends; remove repeats.
        let mut seen = Vec::new();
        lower.retain(|entry| {
            if seen.contains(&entry.0) {
                false
            } else {
                seen.push(entry.0);
                true
            }
        });
        lower
    };

    let vertices: Vec<Point> = hull_points.iter().map(|entry| entry.0).collect();
    let source_indices: Vec<usize> = hull_points.iter().map(|entry| entry.1).collect();
    let (area2, perimeter_milli) = hull_metrics(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}
