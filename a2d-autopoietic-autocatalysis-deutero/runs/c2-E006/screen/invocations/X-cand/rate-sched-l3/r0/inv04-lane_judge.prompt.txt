```rust
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

pub struct Runner {
    jobs: Vec<Job>,
    dependency_indices: Vec<Vec<usize>>,
    states: Vec<State>,
    attempts: Vec<u8>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (index, job) in jobs.iter().enumerate() {
            let valid_id = !job.id.is_empty()
                && job.id.len() <= 16
                && job
                    .id
                    .bytes()
                    .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-');
            if !valid_id {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if ids.insert(job.id.as_str(), index).is_some() {
                return Err(Error::Duplicate(index));
            }
        }

        let mut dependency_indices = vec![Vec::new(); jobs.len()];
        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                match ids.get(dependency.as_str()) {
                    Some(&index) => dependency_indices[job_index].push(index),
                    None => return Err(Error::MissingDependency(job_index, dependency_index)),
                }
            }
        }

        fn visit(
            node: usize,
            graph: &[Vec<usize>],
            colors: &mut [u8],
        ) -> bool {
            if colors[node] == 1 {
                return true;
            }
            if colors[node] == 2 {
                return false;
            }
            colors[node] = 1;
            for &dependency in &graph[node] {
                if visit(dependency, graph, colors) {
                    return true;
                }
            }
            colors[node] = 2;
            false
        }

        let mut colors = vec![0; jobs.len()];
        for index in 0..jobs.len() {
            if visit(index, &dependency_indices, &mut colors) {
                return Err(Error::Cycle);
            }
        }

        let count = jobs.len();
        Ok(Self {
            jobs,
            dependency_indices,
            states: vec![State::Succeeded; count],
            attempts: vec![0; count],
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.states.iter().any(|state| {
            matches!(state, State::Waiting { .. } | State::Running { .. })
        }) {
            return Err(Error::Busy);
        }

        self.states.fill(State::Waiting { ready_at: now });
        self.attempts.fill(0);
        self.last_time = now;
        self.cycle_started = true;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for index in 0..self.jobs.len() {
            let State::Waiting { ready_at } = self.states[index] else {
                continue;
            };
            if ready_at > now
                || self.dependency_indices[index]
                    .iter()
                    .any(|&dependency| self.states[dependency] != State::Succeeded)
            {
                continue;
            }

            let priority = if self.aging_ms == 0 {
                self.jobs[index].priority
            } else {
                let elapsed_steps = (now - ready_at) / self.aging_ms;
                let added = elapsed_steps.min(9) as u8;
                self.jobs[index].priority.saturating_add(added).min(9)
            };

            if best.is_none_or(|(_, best_priority)| priority > best_priority) {
                best = Some((index, priority));
            }
        }

        self.last_time = now;
        let Some((index, _)) = best else {
            return Ok(None);
        };

        self.attempts[index] += 1;
        let attempt = self.attempts[index];
        self.states[index] = State::Running { attempt };
        Ok(Some(Ticket {
            id: self.jobs[index].id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;
        if !matches!(self.states[index], State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.states[index] = State::Succeeded;
        } else if self.attempts[index] == self.jobs[index].max_attempts {
            self.states[index] = State::Failed;
            self.propagate_blocked();
        } else {
            let shift = self.attempts[index] - 1;
            let delay = if shift >= 64 {
                self.retry_cap
            } else {
                self.retry_base
                    .checked_mul(1u64 << shift)
                    .unwrap_or(u64::MAX)
                    .min(self.retry_cap)
            };
            self.states[index] = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;
        Ok(self.states[index].clone())
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for index in 0..self.jobs.len() {
                if !matches!(self.states[index], State::Waiting { .. }) {
                    continue;
                }
                if self.dependency_indices[index].iter().any(|&dependency| {
                    matches!(self.states[dependency], State::Failed | State::Blocked)
                }) {
                    self.states[index] = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }
}
```

```rust
use std::collections::{HashMap, HashSet, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobState {
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<Job>,
    dependency_indices: Vec<Vec<usize>>,
    id_to_index: HashMap<String, usize>,
    states: Vec<JobState>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    watermark: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut id_to_index = HashMap::new();

        for (index, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if id_to_index.contains_key(&job.id) {
                return Err(Error::Duplicate(index));
            }
            id_to_index.insert(job.id.clone(), index);
        }

        let mut dependency_indices = Vec::with_capacity(jobs.len());

        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            let mut dependencies = Vec::with_capacity(job.dependencies.len());

            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }

                let Some(&dependency_job_index) = id_to_index.get(dependency) else {
                    return Err(Error::MissingDependency(job_index, dependency_index));
                };
                dependencies.push(dependency_job_index);
            }

            dependency_indices.push(dependencies);
        }

        if has_cycle(&dependency_indices) {
            return Err(Error::Cycle);
        }

        let states = (0..jobs.len())
            .map(|_| JobState {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            dependency_indices,
            id_to_index,
            states,
            aging_ms,
            retry_base,
            retry_cap,
            watermark: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;

        if self.cycle_started
            && self.states.iter().any(|job_state| {
                matches!(job_state.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for job_state in &mut self.states {
            job_state.state = State::Waiting { ready_at: now };
            job_state.attempts = 0;
        }

        self.cycle_started = true;
        self.watermark = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;

        for index in 0..self.jobs.len() {
            let State::Waiting { ready_at } = self.states[index].state else {
                continue;
            };
            if ready_at > now {
                continue;
            }
            if !self.dependency_indices[index]
                .iter()
                .all(|&dependency| matches!(self.states[dependency].state, State::Succeeded))
            {
                continue;
            }

            let effective_priority = if self.aging_ms == 0 {
                self.jobs[index].priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                let aged_priority = u64::from(self.jobs[index].priority).saturating_add(age);
                aged_priority.min(9) as u8
            };

            match selected {
                None => selected = Some((index, effective_priority)),
                Some((_, best_priority)) if effective_priority > best_priority => {
                    selected = Some((index, effective_priority));
                }
                _ => {}
            }
        }

        let result = if let Some((index, _)) = selected {
            let attempt = self.states[index].attempts + 1;
            self.states[index].attempts = attempt;
            self.states[index].state = State::Running { attempt };
            Some(Ticket {
                id: self.jobs[index].id.clone(),
                attempt,
            })
        } else {
            None
        };

        self.watermark = now;
        Ok(result)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let Some(&index) = self.id_to_index.get(id) else {
            return Err(Error::Unknown);
        };

        let State::Running { attempt } = self.states[index].state else {
            return Err(Error::NotRunning);
        };

        if success {
            self.states[index].state = State::Succeeded;
        } else if attempt == self.jobs[index].max_attempts {
            self.states[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let delay = retry_delay(self.retry_base, self.retry_cap, attempt);
            self.states[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.watermark = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(&index) = self.id_to_index.get(id) else {
            return Err(Error::Unknown);
        };
        Ok(self.states[index].state.clone())
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.watermark {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();

            for index in 0..self.jobs.len() {
                if !matches!(self.states[index].state, State::Waiting { .. }) {
                    continue;
                }
                if self.dependency_indices[index].iter().any(|&dependency| {
                    matches!(
                        self.states[dependency].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    newly_blocked.push(index);
                }
            }

            if newly_blocked.is_empty() {
                break;
            }

            for index in newly_blocked {
                self.states[index].state = State::Blocked;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    let count = dependencies.len();
    let mut indegree = vec![0usize; count];
    let mut dependents = vec![Vec::new(); count];

    for (job, deps) in dependencies.iter().enumerate() {
        indegree[job] = deps.len();
        for &dependency in deps {
            dependents[dependency].push(job);
        }
    }

    let mut ready = VecDeque::new();
    for (index, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            ready.push_back(index);
        }
    }

    let mut visited = 0usize;
    while let Some(job) = ready.pop_front() {
        visited += 1;
        for &dependent in &dependents[job] {
            indegree[dependent] -= 1;
            if indegree[dependent] == 0 {
                ready.push_back(dependent);
            }
        }
    }

    visited != count
}

fn retry_delay(retry_base: u64, retry_cap: u64, attempt: u8) -> u64 {
    let mut delay = retry_base;
    for _ in 1..attempt {
        if delay >= retry_cap || delay > retry_cap / 2 {
            return retry_cap;
        }
        delay *= 2;
    }
    delay.min(retry_cap)
}
```

### C.rs
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    attempts: u8,
}

pub struct Runner {
    entries: Vec<Entry>,
    id_to_index: std::collections::HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut id_to_index = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            let valid_id = !job.id.is_empty()
                && job.id.len() <= 16
                && job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
            if !valid_id {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if id_to_index.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            id_to_index.insert(job.id.clone(), i);
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match id_to_index.get(dep) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(resolved);
        }

        let mut marks = vec![0u8; jobs.len()];
        fn has_cycle(node: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[node] == 1 {
                return true;
            }
            if marks[node] == 2 {
                return false;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if has_cycle(dep, deps, marks) {
                    return true;
                }
            }
            marks[node] = 2;
            false
        }
        for i in 0..jobs.len() {
            if has_cycle(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .zip(dependencies)
            .map(|(job, dependencies)| Entry {
                job,
                dependencies,
                state: State::Blocked,
                attempts: 0,
            })
            .collect();

        Ok(Self {
            entries,
            id_to_index,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.entries.iter().any(|entry| {
            matches!(entry.state, State::Waiting { .. } | State::Running { .. })
        }) {
            return Err(Error::Busy);
        }

        for entry in &mut self.entries {
            entry.state = State::Waiting { ready_at: now };
            entry.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, entry) in self.entries.iter().enumerate() {
            let ready_at = match entry.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !entry
                .dependencies
                .iter()
                .all(|&dep| matches!(self.entries[dep].state, State::Succeeded))
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (entry.job.priority as u64 + age).min(9) as u8
            };

            if best.is_none_or(|(best_i, best_priority)| {
                effective > best_priority || (effective == best_priority && i < best_i)
            }) {
                best = Some((i, effective));
            }
        }

        let result = if let Some((i, _)) = best {
            let entry = &mut self.entries[i];
            entry.attempts += 1;
            entry.state = State::Running {
                attempt: entry.attempts,
            };
            Some(Ticket {
                id: entry.job.id.clone(),
                attempt: entry.attempts,
            })
        } else {
            None
        };

        self.last_time = now;
        Ok(result)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        if !matches!(self.entries[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.entries[index].state = State::Succeeded;
        } else if self.entries[index].attempts == self.entries[index].job.max_attempts {
            self.entries[index].state = State::Failed;
        } else {
            let attempt = self.entries[index].attempts;
            let mut delay = self.retry_base;
            for _ in 1..attempt {
                if delay >= self.retry_cap || delay > self.retry_cap / 2 {
                    delay = self.retry_cap;
                    break;
                }
                delay *= 2;
            }
            delay = delay.min(self.retry_cap);
            self.entries[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        if matches!(self.entries[index].state, State::Failed) {
            loop {
                let mut changed = false;
                for i in 0..self.entries.len() {
                    if !matches!(self.entries[i].state, State::Waiting { .. }) {
                        continue;
                    }
                    if self.entries[i].dependencies.iter().any(|&dep| {
                        matches!(
                            self.entries[dep].state,
                            State::Failed | State::Blocked
                        )
                    }) {
                        self.entries[i].state = State::Blocked;
                        changed = true;
                    }
                }
                if !changed {
                    break;
                }
            }
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        Ok(self.entries[index].state.clone())
    }
}
```
