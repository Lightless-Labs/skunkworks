```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut seen = std::collections::HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::with_capacity(customers.len().saturating_mul(3));
    let mut waiting = std::collections::VecDeque::new();
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut active: Option<(usize, u64)> = None;
    let mut previous_finish: Option<u64> = None;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        let next_arrival_time = order
            .get(next_arrival)
            .map(|&index| customers[index].arrival);

        let tick = match (active.map(|(_, finish)| finish), next_arrival_time) {
            (Some(finish), Some(arrival)) => finish.min(arrival),
            (Some(finish), None) => finish,
            (None, Some(arrival)) => arrival,
            (None, None) => break,
        };
        now = tick;

        if let Some((index, finish)) = active {
            if finish == now {
                events.push(Event {
                    time: now,
                    customer: customers[index].id,
                    kind: Kind::Finish,
                });
                active = None;
                finished_count += 1;
                previous_finish = Some(now);
            }
        }

        while let Some(&index) = order.get(next_arrival) {
            if customers[index].arrival != now {
                break;
            }
            events.push(Event {
                time: now,
                customer: customers[index].id,
                kind: Kind::Arrive,
            });
            waiting.push_back(index);
            next_arrival += 1;
        }

        if active.is_none() {
            if let Some(index) = waiting.pop_front() {
                let customer = &customers[index];
                let wake = previous_finish.map_or(true, |finish| now > finish);
                let service_finish = now.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    },
                )?;
                let finish = if wake {
                    service_finish.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    })?
                } else {
                    service_finish
                };

                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == now {
                    events.push(Event {
                        time: now,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    finished_count += 1;
                    previous_finish = Some(now);
                } else {
                    active = Some((index, finish));
                }
            }
        }
    }

    Ok(events)
}
```
