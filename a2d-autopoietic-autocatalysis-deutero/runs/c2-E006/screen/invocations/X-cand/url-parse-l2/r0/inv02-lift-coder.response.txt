#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = offsets.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if is_safe_byte(component, byte) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());

    let authority = &bytes[authority_start..authority_end];
    let mut ats = authority
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i));
    let first_at = ats.next();
    if let Some(second_at) = ats.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, host_start) = if let Some(at) = first_at {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (
            Some(input[authority_start..authority_start + at].to_owned()),
            authority_start + at + 1,
        )
    } else {
        (None, authority_start)
    };

    let host_bytes = &bytes[host_start..authority_end];
    let (host, port, ipv6) = if host_bytes.first() == Some(&b'[') {
        let closing = host_bytes.iter().position(|&b| b == b']');
        let closing = match closing {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &input[host_start + 1..host_start + closing];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let after = host_start + closing + 1;
        let suffix = &bytes[after..authority_end];
        let port = if suffix.is_empty() {
            None
        } else if suffix[0] == b':' {
            Some(parse_port(&bytes[after + 1..authority_end], after + 1)?)
        } else {
            return Err(ParseError::InvalidHost { at: after });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let host_port_colon = host_bytes.iter().position(|&b| b == b':');
        let host_end = host_port_colon
            .map(|offset| host_start + offset)
            .unwrap_or(authority_end);
        if host_end == host_start {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (at, &byte) in bytes[host_start..host_end].iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + at,
                });
            }
        }
        let port = match host_port_colon {
            Some(offset) => Some(parse_port(
                &bytes[host_start + offset + 1..authority_end],
                host_start + offset + 1,
            )?),
            None => None,
        };
        (input[host_start..host_end].to_ascii_lowercase(), port, false)
    };

    let fragment_pos = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());

    let query_pos = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_pos.map(|at| input[at + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|at| input[at + 1..].to_owned());

    if let Some(ref value) = userinfo {
        validate_url_component(Component::UserInfo, value, authority_start)?;
    }
    if path_end != authority_end {
        validate_url_component(Component::Path, &path, authority_end)?;
    }
    if let (Some(ref value), Some(at)) = (&query, query_pos) {
        validate_url_component(Component::Query, value, at + 1)?;
    }
    if let (Some(ref value), Some(at)) = (&fragment, fragment_pos) {
        validate_url_component(Component::Fragment, value, at + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(bytes: &[u8], start: usize) -> Result<u16, ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + offset });
        }
    }
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for &byte in bytes {
        value = match value.checked_mul(10).and_then(|n| n.checked_add((byte - b'0') as u32)) {
            Some(n) if n <= 65535 => n,
            _ => return Err(ParseError::InvalidPort { at: start }),
        };
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn validate_url_component(
    component: Component,
    value: &str,
    original_start: usize,
) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape {
            at: original_start + at,
        }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 {
            at: original_start + at,
        }),
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_safe_byte(component: Component, byte: u8) -> bool {
    if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
        return true;
    }
    match component {
        Component::UserInfo => byte == b':',
        Component::Path => byte == b'/',
        Component::Query => matches!(byte, b'/' | b'?'),
        Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
    }
}
