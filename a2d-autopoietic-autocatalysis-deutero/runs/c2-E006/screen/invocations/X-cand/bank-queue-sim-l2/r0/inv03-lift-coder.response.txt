#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy: bool,
    idle_since: u64,
    finish_time: u64,
    customer_id: u32,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy: false,
            finish_time: 0,
            customer_id: 0,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut current_time = 0u64;

    loop {
        let mut finished: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (state.busy && state.finish_time == current_time).then_some(i)
            })
            .collect();
        finished.sort_by_key(|&i| states[i].server.id);
        for i in finished {
            let customer_id = states[i].customer_id;
            let server_id = states[i].server.id;
            events.push(Event::Finish {
                time: current_time,
                customer: customer_id,
                server: server_id,
            });
            states[i].busy = false;
            states[i].idle_since = current_time;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.server.opens == current_time).then_some(i)
            })
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: current_time,
                server: states[i].server.id,
            });
        }

        let cohort_start = next_customer;
        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == current_time
        {
            next_customer += 1;
        }
        for &customer_index in &customer_order[cohort_start..next_customer] {
            events.push(Event::Arrive {
                time: current_time,
                customer: customers[customer_index].id,
            });
            waiting.push(customer_index);
        }

        while !waiting.is_empty() {
            let selected_server = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && !state.busy)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected_server else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index].server;

            let Some(after_service) = current_time.checked_add(customer.service) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: current_time,
                });
            };
            let Some(finish_time) = after_service.checked_add(server.overhead) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: current_time,
                });
            };

            events.push(Event::Start {
                time: current_time,
                customer: customer.id,
                server: server.id,
                finish: finish_time,
            });

            if finish_time == current_time {
                events.push(Event::Finish {
                    time: current_time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].busy = false;
                states[server_index].idle_since = current_time;
            } else {
                states[server_index].busy = true;
                states[server_index].finish_time = finish_time;
                states[server_index].customer_id = customer.id;
            }
        }

        if next_customer == customer_order.len()
            && states.iter().all(|state| state.opened && !state.busy)
        {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > current_time && next_time.is_none_or(|next| time < next) {
                next_time = Some(time);
            }
        };

        if next_customer < customer_order.len() {
            consider(customers[customer_order[next_customer]].arrival);
        }
        for state in &states {
            if !state.opened {
                consider(state.server.opens);
            }
            if state.busy {
                consider(state.finish_time);
            }
        }

        match next_time {
            Some(time) => current_time = time,
            None => break,
        }
    }

    Ok(events)
}
