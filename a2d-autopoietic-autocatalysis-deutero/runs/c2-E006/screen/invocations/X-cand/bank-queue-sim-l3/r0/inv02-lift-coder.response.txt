```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: usize,
    paused_at: Option<u64>,
    remaining: u64,
}

struct Active {
    customer: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting, customers: &[Customer]) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    let ca = &customers[a.customer];
    let cb = &customers[b.customer];

    ca.priority
        .cmp(&cb.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(ta), Some(tb)) => ta.cmp(&tb).then_with(|| ca.id.cmp(&cb.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => ca.arrival.cmp(&cb.arrival).then_with(|| cb.id.cmp(&ca.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::cmp::Ordering;

    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0;
    let mut finished_count = 0;
    let mut ever_started = vec![false; customers.len()];
    let mut last_dispatched: Option<u32> = None;
    let mut now: Option<u64> = None;

    while finished_count < customers.len() {
        let mut tick: Option<u64> = None;

        if next_arrival < arrivals.len() {
            tick = Some(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(running) = &active {
            tick = Some(match tick {
                Some(t) => t.min(running.finish),
                None => running.finish,
            });
            if running.work_at > now.unwrap_or(0) {
                tick = Some(match tick {
                    Some(t) => t.min(running.work_at),
                    None => running.work_at,
                });
            }
        }

        let time = match (now, tick) {
            (Some(previous), Some(next)) => next.max(previous),
            (None, Some(next)) => next,
            _ => break,
        };
        now = Some(time);

        if active.as_ref().is_some_and(|running| running.finish == time) {
            let running = active.take().unwrap();
            trace.push(Event {
                time,
                customer: customers[running.customer].id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == time
        {
            let customer = arrivals[next_arrival];
            trace.push(Event {
                time,
                customer: customers[customer].id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                customer,
                paused_at: None,
                remaining: customers[customer].service,
            });
            next_arrival += 1;
        }

        if let Some(running) = &active {
            if time >= running.work_at {
                waiting.sort_by(|a, b| waiting_cmp(a, b, customers));
                if let Some(best) = waiting.first() {
                    let running_priority = customers[running.customer].priority;
                    let best_customer = best.customer;
                    if customers[best_customer].priority < running_priority {
                        let running = active.take().unwrap();
                        let remaining = running.finish - time;
                        trace.push(Event {
                            time,
                            customer: customers[running.customer].id,
                            kind: Kind::Preempt {
                                remaining,
                                by: customers[best_customer].id,
                            },
                        });
                        waiting.push(Waiting {
                            customer: running.customer,
                            paused_at: Some(time),
                            remaining,
                        });
                    }
                }
            }
        }

        if active.is_none() {
            loop {
                if waiting.is_empty() {
                    break;
                }
                waiting.sort_by(|a, b| waiting_cmp(a, b, customers));
                let selected = waiting.remove(0);
                let customer = selected.customer;
                let id = customers[customer].id;
                let cost = if last_dispatched.is_some_and(|last| last != id) {
                    switch_cost
                } else {
                    0
                };

                let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: id,
                    time,
                })?;
                let finish = work_at
                    .checked_add(selected.remaining)
                    .ok_or(Error::TimeOverflow {
                        customer: id,
                        time,
                    })?;

                last_dispatched = Some(id);
                let kind = if ever_started[customer] {
                    Kind::Resume {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    ever_started[customer] = true;
                    Kind::Start {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                };
                trace.push(Event {
                    time,
                    customer: id,
                    kind,
                });

                if finish == time {
                    trace.push(Event {
                        time,
                        customer: id,
                        kind: Kind::Finish,
                    });
                    finished_count += 1;
                    continue;
                }

                active = Some(Active {
                    customer,
                    remaining: selected.remaining,
                    work_at,
                    finish,
                });
                break;
            }
        }

        // Avoid selecting the same already-processed tick when only a past
        // boundary could otherwise be considered.
        if active.is_none() && waiting.is_empty() && next_arrival == arrivals.len() {
            break;
        }

        let _ = Ordering::Equal;
    }

    Ok(trace)
}
```
