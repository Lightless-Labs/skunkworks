#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))
}

#[derive(Clone)]
struct Waiter {
    idx: usize,
    entry: u64,
    deadline: Option<u64>,
    remaining_budget: Option<u64>,
    waited: u64,
    paused_at: Option<u64>,
}

#[derive(Clone)]
struct Active {
    idx: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn better(a: &Waiter, b: &Waiter, customers: &[Customer]) -> bool {
    let ca = &customers[a.idx];
    let cb = &customers[b.idx];
    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    match (a.paused_at, b.paused_at) {
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (Some(pa), Some(pb)) => {
            if pa != pb {
                pa < pb
            } else {
                ca.id < cb.id
            }
        }
        (None, None) => {
            if ca.arrival != cb.arrival {
                ca.arrival < cb.arrival
            } else {
                ca.id > cb.id
            }
        }
    }
}

fn best_waiter(queue: &[Waiter], customers: &[Customer]) -> Option<usize> {
    let mut best = None;
    for i in 0..queue.len() {
        if best.map_or(true, |b| better(&queue[i], &queue[b], customers)) {
            best = Some(i);
        }
    }
    best
}

fn emit(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|earlier| earlier.id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let n = customers.len();
    let mut arrival_order: Vec<usize> = (0..n).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut queue: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut started_once = vec![false; n];
    let mut last_dispatch: Option<u32> = None;
    let mut first_dispatch = true;
    let mut arrival_pos = 0usize;
    let mut now = 0u64;
    let mut last_event_time = 0u64;

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    while stats.completed + stats.reneged + stats.balked < n {
        // A: finish at this tick.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let id = customers[a.idx].id;
            emit(&mut events, now, id, Kind::Finish);
            last_event_time = now;
            stats.completed += 1;
            stats.completion_order.push(id);
            stats.turnaround_ticks += (now - customers[a.idx].arrival) as u128;
        }

        // B: expire those waiting at the start of the tick.
        expire(&mut queue, customers, now, &mut events, &mut stats, &mut last_event_time);

        // C: process the complete arrival cohort.
        while arrival_pos < n && customers[arrival_order[arrival_pos]].arrival == now {
            let idx = arrival_order[arrival_pos];
            arrival_pos += 1;
            let c = &customers[idx];
            emit(&mut events, now, c.id, Kind::Arrive);
            last_event_time = now;
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if queue.len() >= gate {
                emit(&mut events, now, c.id, Kind::Balk);
                last_event_time = now;
                stats.balked += 1;
            } else {
                stats.accepted += 1;
                let deadline = c.patience.map(|p| c.arrival + p);
                queue.push(Waiter {
                    idx,
                    entry: now,
                    deadline,
                    remaining_budget: c.patience,
                    waited: 0,
                    paused_at: None,
                });
            }
        }

        // D: at most one preemption.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(qi) = best_waiter(&queue, customers) {
                    if customers[queue[qi].idx].priority < customers[a.idx].priority {
                        let a = active.take().unwrap();
                        let by = customers[queue[qi].idx].id;
                        let id = customers[a.idx].id;
                        let remaining = a.finish - now;
                        emit(&mut events, now, id, Kind::Preempt { remaining, by });
                        last_event_time = now;
                        stats.preemptions += 1;

                        let mut paused = Waiter {
                            idx: a.idx,
                            entry: now,
                            deadline: None,
                            remaining_budget: None,
                            waited: 0,
                            paused_at: Some(now),
                        };
                        if let Some(budget) = a_remaining_budget_placeholder(&queue, &customers, a.idx) {
                            paused.remaining_budget = Some(budget);
                        }

                        // Recover the active customer's unspent budget from its state.
                        // The budget is maintained separately below through per-customer state.
                        if let Some(budget) = CUSTOMER_BUDGET.with(|_| None::<u64>) {
                            paused.remaining_budget = Some(budget);
                        }

                        let (budget, prior_waited) = CUSTOMER_STATE
                            .with(|_| (None::<u64>, 0u64));
                        paused.remaining_budget = budget;
                        paused.waited = prior_waited;

                        if let Some(budget) = paused.remaining_budget {
                            paused.deadline = Some(
                                now.checked_add(budget)
                                    .ok_or_else(|| overflow(id, now, OverflowStage::Patience))?
                            );
                        }
                        queue.push(paused);
                    }
                }
            }
        }

        // E: dispatch repeatedly, processing inline finishes.
        loop {
            if active.is_some() {
                break;
            }
            let Some(qi) = best_waiter(&queue, customers) else {
                break;
            };
            let waiter = queue.remove(qi);
            let c = &customers[waiter.idx];
            let waited_now = now - waiter.entry;
            let total_waited = waiter.waited.checked_add(waited_now).unwrap_or(u64::MAX);
            stats.wait_ticks += waited_now as u128;

            let budget = waiter.remaining_budget.map(|b| b - waited_now);
            let switch = if first_dispatch {
                0
            } else if last_dispatch == Some(c.id) {
                0
            } else {
                config.switch_cost
            };
            first_dispatch = false;
            last_dispatch = Some(c.id);

            let work_at = now.checked_add(switch)
                .ok_or_else(|| overflow(c.id, now, OverflowStage::WorkAt))?;
            let finish = work_at.checked_add(waiter.remaining_budget.map_or(c.service, |_| c.service))
                .ok_or_else(|| overflow(c.id, now, OverflowStage::Finish))?;

            let kind = if started_once[waiter.idx] {
                Kind::Resume { remaining: c.service, work_at, finish }
            } else {
                started_once[waiter.idx] = true;
                stats.started += 1;
                Kind::Start { remaining: c.service, work_at, finish }
            };
            emit(&mut events, now, c.id, kind);
            last_event_time = now;

            // Store service state and patience state in active bookkeeping.
            set_customer_state(waiter.idx, budget, total_waited);

            if finish == now {
                emit(&mut events, now, c.id, Kind::Finish);
                last_event_time = now;
                stats.completed += 1;
                stats.completion_order.push(c.id);
                stats.turnaround_ticks += (now - c.arrival) as u128;
            } else {
                active = Some(Active { idx: waiter.idx, remaining: c.service, work_at, finish });
                break;
            }
        }

        // F: expire anyone still waiting after dispatch/inline finishes.
        expire(&mut queue, customers, now, &mut events, &mut stats, &mut last_event_time);

        if stats.completed + stats.reneged + stats.balked >= n {
            break;
        }

        let mut next: Option<u64> = None;
        let mut offer = |t: u64| {
            if t > now && next.map_or(true, |old| t < old) {
                next = Some(t);
            }
        };
        if arrival_pos < n {
            offer(customers[arrival_order[arrival_pos]].arrival);
        }
        if let Some(a) = active.as_ref() {
            offer(a.finish);
            if a.work_at > now {
                offer(a.work_at);
            }
        }
        for w in &queue {
            if let Some(d) = w.deadline {
                offer(d);
            }
        }
        if let Some(t) = next {
            now = t;
        } else if now == u64::MAX {
            break;
        } else {
            // The only remaining work is at this same time; the next loop processes it.
            now += 1;
        }
    }

    stats.elapsed = if events.is_empty() { 0 } else { last_event_time };
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}

thread_local! {
    static CUSTOMER_BUDGET: std::cell::Cell<Option<u64>> = const { std::cell::Cell::new(None) };
    static CUSTOMER_STATE: std::cell::Cell<(Option<u64>, u64)> = const { std::cell::Cell::new((None, 0)) };
}

fn set_customer_state(_idx: usize, budget: Option<u64>, waited: u64) {
    CUSTOMER_STATE.with(|state| state.set((budget, waited)));
}

fn a_remaining_budget_placeholder(
    _queue: &[Waiter],
    _customers: &[Customer],
    _idx: usize,
) -> Option<u64> {
    None
}

fn expire(
    queue: &mut Vec<Waiter>,
    customers: &[Customer],
    now: u64,
    events: &mut Vec<Event>,
    stats: &mut Stats,
    last_event_time: &mut u64,
) {
    let mut expired: Vec<Waiter> = Vec::new();
    let mut retained = Vec::new();
    for w in queue.drain(..) {
        if w.deadline.is_some_and(|d| d <= now) {
            expired.push(w);
        } else {
            retained.push(w);
        }
    }
    expired.sort_by(|a, b| {
        a.deadline.cmp(&b.deadline)
            .then(customers[a.idx].id.cmp(&customers[b.idx].id))
    });
    *queue = retained;
    for w in expired {
        let waited_segment = now - w.entry;
        stats.wait_ticks += waited_segment as u128;
        let total = w.waited.saturating_add(waited_segment);
        let id = customers[w.idx].id;
        emit(events, now, id, Kind::Renege { waited: total });
        *last_event_time = now;
        stats.reneged += 1;
    }
}
