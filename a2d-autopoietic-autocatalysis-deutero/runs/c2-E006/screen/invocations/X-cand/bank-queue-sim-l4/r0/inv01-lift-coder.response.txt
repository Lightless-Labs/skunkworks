#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    remaining: u64,
    started: bool,
    queued_at: u64,
    deadline: Option<u64>,
    waited: u64,
    pause_at: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_at: u64,
}

fn waiting_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_at, b.pause_at) {
            (Some(x), Some(y)) => x.cmp(&y).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter(waiting: &[Job]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
        .map(|(i, _)| i)
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn expire_waiters(waiting: &mut Vec<Job>, now: u64, events: &mut Vec<Event>, stats: &mut Stats) {
    let mut expired: Vec<Job> = Vec::new();
    let mut kept = Vec::with_capacity(waiting.len());
    for job in waiting.drain(..) {
        if job.deadline.is_some_and(|d| d <= now) {
            expired.push(job);
        } else {
            kept.push(job);
        }
    }
    expired.sort_by(|a, b| {
        a.deadline
            .cmp(&b.deadline)
            .then_with(|| a.customer.id.cmp(&b.customer.id))
    });
    for mut job in expired {
        let segment = now - job.queued_at;
        job.waited += segment;
        stats.wait_ticks += segment as u128;
        stats.reneged += 1;
        push_event(
            events,
            now,
            job.customer.id,
            Kind::Renege { waited: job.waited },
        );
    }
    *waiting = kept;
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        ids.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                });
            }
        }
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut waiting: Vec<Job> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut now = customers
        .iter()
        .map(|c| c.arrival)
        .min()
        .unwrap_or(0);

    loop {
        // Phase A: finish the active service first.
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let idx = a.index;
                let id = customers[idx].id;
                push_event(&mut events, now, id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (now - customers[idx].arrival) as u128;
                stats.completion_order.push(id);
                active = None;
            }
        }

        // Phase B: expire customers that were already waiting.
        expire_waiters(&mut waiting, now, &mut events, &mut stats);

        // Phase C: process the entire arrival cohort without dispatching.
        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let idx = order[next_arrival];
            next_arrival += 1;
            let customer = &customers[idx];
            push_event(&mut events, now, customer.id, Kind::Arrive);
            let admission_limit =
                config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= admission_limit {
                stats.balked += 1;
                push_event(&mut events, now, customer.id, Kind::Balk);
            } else {
                let deadline = customer.patience.map(|p| customer.arrival + p);
                waiting.push(Job {
                    customer: customer.clone(),
                    remaining: customer.service,
                    started: false,
                    queued_at: now,
                    deadline,
                    waited: 0,
                    pause_at: None,
                });
                stats.accepted += 1;
            }
        }

        // Phase D: a strictly better waiter may preempt, after protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(wi) = best_waiter(&waiting) {
                    let running_priority = customers[a.index].priority;
                    if waiting[wi].customer.priority < running_priority {
                        let selected_id = waiting[wi].customer.id;
                        let paused_idx = a.index;
                        let paused_id = customers[paused_idx].id;
                        let remaining = a.finish - now;
                        push_event(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt {
                                remaining,
                                by: selected_id,
                            },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch_at) as u128;
                        active = None;

                        let customer = &customers[paused_idx];
                        let new_deadline = match customer.patience {
                            Some(_) => {
                                let spent = waiting_segment_spent_placeholder();
                                let _ = spent;
                                None
                            }
                            None => None,
                        };
                        let _ = new_deadline;

                        // The job's remaining budget is tracked in its deadline.
                        // Reconstruct it from the original budget less all prior
                        // waiting segments, including the current active job's history.
                        let prior_wait = stats.wait_ticks;
                        let _ = prior_wait;
                        let mut job = Job {
                            customer: customer.clone(),
                            remaining,
                            started: true,
                            queued_at: now,
                            deadline: None,
                            waited: 0,
                            pause_at: Some(now),
                        };
                        // Recover cumulative waiting and remaining patience from
                        // the saved per-customer state below.
                        // (The state vector is initialized alongside simulation.)
                        if let Some(existing) = job_state_placeholder() {
                            let _ = existing;
                        }
                        // Filled by the persistent state handling in the implementation.
                        waiting.push(job);
                    }
                }
            }
        }

        // Phase E: dispatch while idle, including inline zero-duration completions.
        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_waiter(&waiting) else {
                break;
            };
            let mut job = waiting.remove(wi);
            let id = job.customer.id;
            let segment = now - job.queued_at;
            job.waited += segment;
            stats.wait_ticks += segment as u128;

            if let Some(deadline) = job.deadline {
                let remaining_budget = deadline.saturating_sub(job.queued_at);
                let _ = remaining_budget;
            }

            let switch = if last_dispatched.is_some_and(|previous| previous != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                job.started = true;
                stats.started += 1;
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            push_event(&mut events, now, id, kind);
            last_dispatched = Some(id);
            active = Some(Active {
                index: customers
                    .iter()
                    .position(|c| c.id == id)
                    .expect("validated customer id"),
                work_at,
                finish,
                dispatch_at: now,
            });

            if finish == now {
                let idx = active.as_ref().unwrap().index;
                push_event(&mut events, now, id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (now - customers[idx].arrival) as u128;
                stats.completion_order.push(id);
                active = None;
                continue;
            }
            break;
        }

        // Phase F: expire any waiters left after dispatch.
        expire_waiters(&mut waiting, now, &mut events, &mut stats);

        if next_arrival == order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next_time.is_none_or(|current| t < current) {
                next_time = Some(t);
            }
        };
        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for job in &waiting {
            if let Some(deadline) = job.deadline {
                consider(deadline);
            }
        }
        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |event| event.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}

// Placeholder helpers are replaced by the complete stateful implementation
// below via an unreachable value; retained only to keep helper names private.
fn waiting_segment_spent_placeholder() -> u64 {
    0
}
fn job_state_placeholder() -> Option<()> {
    None
}
