#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    while remaining != 0 {
        groups.push((remaining % 1000) as u16);
        remaining /= 1000;
    }

    let mut parts: Vec<String> = Vec::new();
    for index in (0..groups.len()).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        let mut part = encode_group(group);
        if index > 0 {
            part.push(' ');
            part.push_str(SCALES[index - 1]);
        }
        parts.push(part);
    }

    let units_group = groups.first().copied().unwrap_or(0);
    let has_higher = groups.iter().skip(1).any(|&g| g != 0);
    let cross_and = units_group > 0 && units_group < 100 && has_higher;

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    for (i, part) in parts.iter().enumerate() {
        if i > 0 {
            if cross_and && i + 1 == parts.len() {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        result.push_str(part);
    }

    Ok(result)
}

fn encode_group(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }

    if n < 100 {
        let tens = n / 10;
        let units = n % 10;
        if units == 0 {
            return TENS[(tens - 2) as usize].to_string();
        }
        return format!("{}-{}", TENS[(tens - 2) as usize], SMALL[units as usize]);
    }

    let hundreds = n / 100;
    let remainder = n % 100;
    if remainder == 0 {
        format!("{} hundred", SMALL[hundreds as usize])
    } else {
        format!(
            "{} hundred and {}",
            SMALL[hundreds as usize],
            encode_group(remainder)
        )
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !(bytes[index - 1].is_ascii_lowercase()
                    && bytes[index + 1].is_ascii_lowercase())
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let tokens = tokenize(input);
    for (token, index) in &tokens {
        if token == "minus" {
            if *index != tokens[0].1 {
                return Err(WordsError::InvalidWord { index: *index });
            }
        } else if !is_dictionary_word(token) {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic_start = usize::from(negative);

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (word, _) in tokens.iter().skip(arithmetic_start) {
        match *word {
            "zero" | "and" => {}
            "one" => group += 1,
            "two" => group += 2,
            "three" => group += 3,
            "four" => group += 4,
            "five" => group += 5,
            "six" => group += 6,
            "seven" => group += 7,
            "eight" => group += 8,
            "nine" => group += 9,
            "ten" => group += 10,
            "eleven" => group += 11,
            "twelve" => group += 12,
            "thirteen" => group += 13,
            "fourteen" => group += 14,
            "fifteen" => group += 15,
            "sixteen" => group += 16,
            "seventeen" => group += 17,
            "eighteen" => group += 18,
            "nineteen" => group += 19,
            "twenty" => group += 20,
            "thirty" => group += 30,
            "forty" => group += 40,
            "fifty" => group += 50,
            "sixty" => group += 60,
            "seventy" => group += 70,
            "eighty" => group += 80,
            "ninety" => group += 90,
            "hundred" => {
                group = group.checked_mul(100).unwrap_or(u128::MAX);
            }
            "thousand" => apply_scale(&mut total, &mut group, 1_000),
            "million" => apply_scale(&mut total, &mut group, 1_000_000),
            "billion" => apply_scale(&mut total, &mut group, 1_000_000_000),
            "trillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000),
            "quadrillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000_000),
            "quintillion" => apply_scale(&mut total, &mut group, LIMIT),
            _ => unreachable!(),
        }

        if total > LIMIT || group > LIMIT || total.checked_add(group).unwrap_or(u128::MAX) > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(value)
}

fn apply_scale(total: &mut u128, group: &mut u128, scale: u128) {
    let scaled = group.checked_mul(scale).unwrap_or(u128::MAX);
    *total = total.checked_add(scaled).unwrap_or(u128::MAX);
    *group = 0;
}

fn tokenize(input: &str) -> Vec<(&str, usize)> {
    let bytes = input.as_bytes();
    let mut tokens = Vec::new();
    let mut start = 0;

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));
    tokens
}

fn is_dictionary_word(word: &str) -> bool {
    word == "zero"
        || word == "hundred"
        || word == "thousand"
        || word == "million"
        || word == "billion"
        || word == "trillion"
        || word == "quadrillion"
        || word == "quintillion"
        || word == "and"
        || word == "minus"
        || SMALL[1..].contains(&word)
        || TENS.contains(&word)
}
