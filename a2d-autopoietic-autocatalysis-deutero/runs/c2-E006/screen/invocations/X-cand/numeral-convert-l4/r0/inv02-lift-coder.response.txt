#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const ONES: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "quintillion",
    "quadrillion",
    "trillion",
    "billion",
    "million",
    "thousand",
    "",
];

fn spell_under_thousand(n: u16) -> String {
    if n < 20 {
        return ONES[n as usize].to_string();
    }
    if n < 100 {
        let tens = TENS[(n / 10) as usize];
        let units = n % 10;
        if units == 0 {
            tens.to_string()
        } else {
            format!("{tens}-{}", ONES[units as usize])
        }
    } else {
        let hundreds = n / 100;
        let remainder = n % 100;
        if remainder == 0 {
            format!("{} hundred", ONES[hundreds as usize])
        } else {
            format!(
                "{} hundred and {}",
                ONES[hundreds as usize],
                spell_under_thousand(remainder)
            )
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    if value.unsigned_abs() > LIMIT {
        return Err(WordsError::Range);
    }
    if value == 0 {
        return Ok("zero".to_string());
    }

    let negative = value < 0;
    let mut magnitude = value.unsigned_abs();
    let mut groups = [0u16; 7];
    for i in (0..7).rev() {
        groups[i] = (magnitude % 1000) as u16;
        magnitude /= 1000;
    }

    let units = groups[6];
    let cross_and = units > 0 && units < 100 && groups[..6].iter().any(|&g| g != 0);
    let mut parts = Vec::new();

    for i in 0..7 {
        let group = groups[i];
        if group == 0 {
            continue;
        }
        if i == 6 && cross_and {
            if let Some(last) = parts.last_mut() {
                last.push_str(" and ");
            }
        }
        let mut part = spell_under_thousand(group);
        if !SCALES[i].is_empty() {
            part.push(' ');
            part.push_str(SCALES[i]);
        }
        parts.push(part);
    }

    let mut result = parts.join(" ");
    if negative {
        result.insert_str(0, "minus ");
    }
    Ok(result)
}

fn word_value(word: &str) -> Option<u128> {
    match word {
        "zero" => Some(0),
        "one" => Some(1),
        "two" => Some(2),
        "three" => Some(3),
        "four" => Some(4),
        "five" => Some(5),
        "six" => Some(6),
        "seven" => Some(7),
        "eight" => Some(8),
        "nine" => Some(9),
        "ten" => Some(10),
        "eleven" => Some(11),
        "twelve" => Some(12),
        "thirteen" => Some(13),
        "fourteen" => Some(14),
        "fifteen" => Some(15),
        "sixteen" => Some(16),
        "seventeen" => Some(17),
        "eighteen" => Some(18),
        "nineteen" => Some(19),
        "twenty" => Some(20),
        "thirty" => Some(30),
        "forty" => Some(40),
        "fifty" => Some(50),
        "sixty" => Some(60),
        "seventy" => Some(70),
        "eighty" => Some(80),
        "ninety" => Some(90),
        _ => None,
    }
}

fn scale_value(word: &str) -> Option<u128> {
    match word {
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        _ => None,
    }
}

fn is_dictionary_word(word: &str) -> bool {
    word == "minus"
        || word == "hundred"
        || word == "and"
        || word_value(word).is_some()
        || scale_value(word).is_some()
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let before_is_letter = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let after_is_letter =
                index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !before_is_letter || !after_is_letter {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(usize, &str)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((start, &input[start..index]));
            start = index + 1;
        }
    }
    tokens.push((start, &input[start..]));

    for (position, word) in &tokens {
        if !is_dictionary_word(word) {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let negative = tokens.first().map(|(_, word)| *word) == Some("minus");
    for (i, (position, word)) in tokens.iter().enumerate() {
        if *word == "minus" && i != 0 {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (_, word) in arithmetic {
        if let Some(value) = word_value(word) {
            group = group.checked_add(value).ok_or(WordsError::Range)?;
        } else if *word == "hundred" {
            group = group.checked_mul(100).ok_or(WordsError::Range)?;
        } else if *word == "and" {
        } else if let Some(scale) = scale_value(word) {
            let scaled = group.checked_mul(scale).ok_or(WordsError::Range)?;
            total = total.checked_add(scaled).ok_or(WordsError::Range)?;
            group = 0;
        }

        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }

    Ok(value)
}
