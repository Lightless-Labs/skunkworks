```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
            cwd: "/".to_owned(),
        }
    }

    fn validate_and_normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd
                .split('/')
                .filter(|part| !part.is_empty())
                .map(str::to_owned)
                .collect()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                _ => components.push(part.to_owned()),
            }
        }
        Ok(components)
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn lookup<'a>(&'a self, components: &[String]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut prefix = Vec::new();

        for component in components {
            match current {
                Entry::Dir(children) => {
                    prefix.push(component.clone());
                    current = children
                        .get(component)
                        .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
                }
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(&prefix))),
            }
        }
        Ok(current)
    }

    fn lookup_parent_mut<'a>(
        &'a mut self,
        components: &[String],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let parent_components = &components[..components.len() - 1];
        let mut current = &mut self.root;
        let mut prefix = Vec::new();

        for component in parent_components {
            match current {
                Entry::Dir(children) => {
                    prefix.push(component.clone());
                    current = children
                        .get_mut(component)
                        .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
                }
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(&prefix))),
            }
        }

        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::canonical(parent_components))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.validate_and_normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let final_path = Self::canonical(&components);
        let name = components.last().expect("nonempty path").clone();
        let parent = self.lookup_parent_mut(&components)?;
        if parent.contains_key(&name) {
            return Err(Error::Exists(final_path));
        }
        parent.insert(name, Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.validate_and_normalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let final_path = Self::canonical(&components);
        let name = components.last().expect("nonempty path").clone();
        let parent = self.lookup_parent_mut(&components)?;
        match parent.get_mut(&name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(final_path)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.validate_and_normalize(path)?;
        let final_path = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(final_path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.validate_and_normalize(path)?;
        let final_path = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(final_path)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.validate_and_normalize(path)?;
        let final_path = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::Dir(_) => {
                self.cwd = final_path;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(final_path)),
        }
    }
}
```
