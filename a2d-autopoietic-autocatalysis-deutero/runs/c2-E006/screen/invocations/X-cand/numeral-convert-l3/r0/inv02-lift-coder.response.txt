#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1i128 << 120) - 1;

fn checksum(payload: &str) -> u8 {
    payload.bytes().enumerate().fold(0u8, |sum, (i, byte)| {
        let code = match byte {
            b'T' => 2,
            b'1' => 1,
            _ => 0,
        };
        (sum + (((i as u8).wrapping_add(1) % 7) * code) % 7) % 7
    })
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    if value < -LIMIT || value > LIMIT {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut reversed = Vec::new();

    if n == 0 {
        reversed.push(b'0');
    } else {
        while n != 0 {
            match system {
                SignedSystem::BalancedTernary => {
                    let rem = n.rem_euclid(3);
                    let digit = if rem == 2 { -1 } else { rem };
                    reversed.push(match digit {
                        -1 => b'T',
                        0 => b'0',
                        _ => b'1',
                    });
                    n = (n - digit) / 3;
                }
                SignedSystem::Negabinary => {
                    let digit = n.rem_euclid(2);
                    reversed.push(if digit == 0 { b'0' } else { b'1' });
                    n = (n - digit) / -2;
                }
            }
        }
    }

    reversed.reverse();
    let payload = String::from_utf8(reversed).expect("encoded digits are ASCII");
    if with_checksum {
        let check = checksum(&payload);
        Ok(format!("{payload}:{check}"))
    } else {
        Ok(payload)
    }
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    if input.is_empty() {
        return Err(SignedError::Empty);
    }

    let colon = input.bytes().position(|b| b == b':');
    let payload_end = colon.unwrap_or(input.len());
    if payload_end == 0 {
        return Err(SignedError::Empty);
    }

    let payload = &input[..payload_end];
    for (index, byte) in payload.bytes().enumerate() {
        let valid = match system {
            SignedSystem::BalancedTernary => matches!(byte, b'T' | b'0' | b'1'),
            SignedSystem::Negabinary => matches!(byte, b'0' | b'1'),
        };
        if !valid {
            return Err(SignedError::InvalidByte { index });
        }
    }

    let supplied = if let Some(colon_index) = colon {
        let suffix = &input[colon_index + 1..];
        let bytes = suffix.as_bytes();
        if bytes.len() != 1 || !(b'0'..=b'6').contains(&bytes[0]) {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
        Some(bytes[0] - b'0')
    } else {
        None
    };

    if payload.len() > 1 && payload.as_bytes()[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let (base, min_digit) = match system {
        SignedSystem::BalancedTernary => (3i128, true),
        SignedSystem::Negabinary => (-2i128, false),
    };

    let mut accumulator = 0i128;
    for (index, byte) in payload.bytes().enumerate() {
        let digit = match byte {
            b'T' => -1,
            b'0' => 0,
            b'1' => 1,
            _ => unreachable!(),
        };
        let _ = min_digit;
        accumulator = accumulator * base + digit;
        if !(-LIMIT..=LIMIT).contains(&accumulator) {
            return Err(SignedError::Range { index });
        }
    }

    if let Some(found) = supplied {
        let expected = checksum(payload);
        if found != expected {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(accumulator)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, with_checksum)
}
