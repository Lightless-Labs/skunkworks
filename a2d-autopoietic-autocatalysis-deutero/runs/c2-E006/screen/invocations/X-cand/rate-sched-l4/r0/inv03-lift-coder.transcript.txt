                    b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-'
                })
        };

                return Err(Error::BadId(i));

                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));

                }

        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);

            if should_process && time <= until {

                _ => break,

            .iter()
            .enumerate()

                _ => None,

                _ => continue,

            let mut did_start = false;
            for candidate in candidates {
                let candidate_tag = self.entries[candidate].job.tag.as_deref();
                let running: Vec<usize> = (0..self.entries.len())
                    .filter(|&i| matches!(self.entries[i].state, State::Running { .. }))
                    .collect();
                let slot_free = running.len() < self.capacity;
                let tag_free = candidate_tag.map_or(true, |tag| {
                    !running.iter().any(|&i| self.entries[i].job.tag.as_deref() == Some(tag))
                });

                if slot_free && tag_free {
                    self.start_job(candidate, time, trace);
                    did_start = true;
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&victim| {
                        self.entries[candidate].job.priority
                            >= self.entries[victim].job.priority.saturating_add(2)
                            && (running.len() - 1 < self.capacity)
                            && candidate_tag.map_or(true, |tag| {
                                !running.iter().any(|&other| {
                                    other != victim
                                        && self.entries[other].job.tag.as_deref() == Some(tag)
                                })
                            })
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            self.entries[b]
                                .segment_started_at
                                .cmp(&self.entries[a].segment_started_at)
                        })
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });

                if let Some(victim) = victims.first().copied() {
                    let victim_id = self.entries[victim].job.id.clone();
                    let remaining = match self.entries[victim].state {
                        State::Running { finish_at, .. } => finish_at.saturating_sub(time),
                        _ => 0,
                    };
                    let attempt = match self.entries[victim].state {
                        State::Running { attempt, .. } => attempt,
                        _ => 0,
                    };
                    trace.push(format!(
                        "{time} preempt {victim_id} by {}",
                        self.entries[candidate].job.id
                    ));
                    self.entries[victim].state = State::Waiting {
                        ready_at: time,
                        attempt,
                    };
                    self.entries[victim].age_from = time;
                    self.entries[victim].remaining = Some(remaining);
                    self.start_job(candidate, time, trace);
                    did_start = true;
                    break;
                }
            }
            if !did_start {
                break;
            }
        }
    }

    fn start_job(&mut self, i: usize, time: u64, trace: &mut Vec<String>) {
        let (attempt, remaining) = match self.entries[i].state {
            State::Waiting { attempt, .. } => match self.entries[i].remaining.take() {
                Some(rem) => (attempt, rem),
                None => (attempt.saturating_add(1), self.entries[i].job.duration),
            },
            _ => return,
        };
        let finish_at = time.saturating_add(remaining);
        let id = self.entries[i].job.id.clone();
        self.entries[i].state = State::Running { attempt, finish_at };
        self.entries[i].segment_started_at = time;
        trace.push(format!("{time} start {id} {attempt} {remaining}"));
    }
}
