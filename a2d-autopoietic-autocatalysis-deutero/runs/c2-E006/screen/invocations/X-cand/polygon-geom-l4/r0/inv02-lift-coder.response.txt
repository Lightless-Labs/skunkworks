#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const MAX_COORD: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;
type Wide = i128;

fn in_range(p: Point) -> bool {
    (-MAX_COORD..=MAX_COORD).contains(&p.x) && (-MAX_COORD..=MAX_COORD).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }
    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[0] == effective[effective.len() - 1] {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> Wide {
    (b.x as Wide - a.x as Wide) * (c.y as Wide - a.y as Wide)
        - (b.y as Wide - a.y as Wide) * (c.x as Wide - a.x as Wide)
}

fn shoelace(points: &[Point]) -> Wide {
    let mut sum = 0_i128;
    if points.len() < 2 {
        return sum;
    }
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as Wide * b.y as Wide - a.y as Wide * b.x as Wide;
    }
    sum
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn edge_mill(a: Point, b: Point) -> u64 {
    let dx = a.x as i128 - b.x as i128;
    let dy = a.y as i128 - b.y as i128;
    let scaled_sq = (dx * dx + dy * dy) as u128 * 1_000_000;
    let root = isqrt(scaled_sq);
    let rounded = if 4 * scaled_sq >= 4 * root * root + 4 * root + 1 {
        root + 1
    } else {
        root
    };
    rounded as u64
}

fn endpoint_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

fn measures_for(points: &[Point]) -> Measures {
    let signed_area2 = shoelace(points);
    let area2 = signed_area2.unsigned_abs();
    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };
    let n = points.len();
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge = None;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        let len = edge_mill(a, b);
        edge_milli.push(len);
        perimeter_milli += len;
        longest_edge = match longest_edge {
            None => Some(i),
            Some(old) => {
                if len > edge_milli[old] {
                    Some(i)
                } else if len < edge_milli[old] {
                    Some(old)
                } else {
                    let old_pair = endpoint_pair(points[old], points[(old + 1) % n]);
                    let new_pair = endpoint_pair(a, b);
                    if new_pair < old_pair || (new_pair == old_pair && i < old) {
                        Some(i)
                    } else {
                        Some(old)
                    }
                }
            }
        };
    }
    Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let effective = prepare_polygon(points)?;
    Ok(measures_for(&effective))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &q) in queries.iter().enumerate() {
        if !in_range(q) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if on_segment(a, b, q) {
                boundary = Some(i);
                break;
            }
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(match boundary {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }
    Ok(results)
}

fn orientation(a: Point, b: Point, c: Point) -> Wide {
    cross(a, b, c)
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let o1 = orientation(a, b, c);
    let o2 = orientation(a, b, d);
    let o3 = orientation(c, d, a);
    let o4 = orientation(c, d, b);
    if o1 == 0 && on_segment(a, b, c) { return true; }
    if o2 == 0 && on_segment(a, b, d) { return true; }
    if o3 == 0 && on_segment(c, d, a) { return true; }
    if o4 == 0 && on_segment(c, d, b) { return true; }
    (o1 < 0) != (o2 < 0) && (o3 < 0) != (o4 < 0)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (i, &p) in points.iter().enumerate() {
        if !unique.iter().any(|(q, _)| *q == p) {
            unique.push((p, i));
        }
    }
    unique.sort_by_key(|(p, _)| *p);
    if unique.len() <= 1 {
        return Ok(hull_from_pairs(unique));
    }

    let all_collinear = unique.iter().skip(2).all(|(p, _)| {
        cross(unique[0].0, unique[1].0, *p) == 0
    });
    if all_collinear {
        let selected = match mode {
            Collinear::Include => unique,
            Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
        };
        return Ok(hull_from_pairs(selected));
    }

    let mut lower: Vec<(Point, usize)> = Vec::new();
    for &item in &unique {
        while lower.len() >= 2 {
            let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, item.0);
            let remove = match mode {
                Collinear::Exclude => turn <= 0,
                Collinear::Include => turn < 0,
            };
            if remove { lower.pop(); } else { break; }
        }
        lower.push(item);
    }
    let mut upper: Vec<(Point, usize)> = Vec::new();
    for &item in unique.iter().rev() {
        while upper.len() >= 2 {
            let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, item.0);
            let remove = match mode {
                Collinear::Exclude => turn <= 0,
                Collinear::Include => turn < 0,
            };
            if remove { upper.pop(); } else { break; }
        }
        upper.push(item);
    }
    lower.pop();
    upper.pop();
    lower.extend(upper);
    Ok(hull_from_pairs(lower))
}

fn hull_from_pairs(pairs: Vec<(Point, usize)>) -> Hull {
    let vertices: Vec<Point> = pairs.iter().map(|(p, _)| *p).collect();
    let source_indices: Vec<usize> = pairs.iter().map(|(_, i)| *i).collect();
    let area2 = shoelace(&vertices).unsigned_abs();
    let perimeter_milli = if vertices.len() < 2 {
        0
    } else {
        (0..vertices.len())
            .map(|i| edge_mill(vertices[i], vertices[(i + 1) % vertices.len()]))
            .sum()
    };
    Hull { vertices, source_indices, area2, perimeter_milli }
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let effective = prepare_polygon(points)?;
    let n = effective.len();

    let mut duplicate = None;
    for i in 0..n {
        for j in i + 1..n {
            if effective[i] == effective[j] {
                let pair = (i, j);
                if duplicate.map_or(true, |old| pair < old) {
                    duplicate = Some(pair);
                }
            }
        }
    }
    if let Some((first, second)) = duplicate {
        return Err(GeomError::Duplicate { first, second });
    }

    let area = shoelace(&effective);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = effective[(i + n - 1) % n];
        let curr = effective[i];
        let next = effective[(i + 1) % n];
        if cross(prev, curr, next) == 0 {
            let ax = prev.x as i128 - curr.x as i128;
            let ay = prev.y as i128 - curr.y as i128;
            let bx = next.x as i128 - curr.x as i128;
            let by = next.y as i128 - curr.y as i128;
            if ax * bx + ay * by > 0 {
                return Err(GeomError::AdjacentOverlap { vertex: i });
            }
        }
    }

    for i in 0..n {
        for j in i + 1..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                effective[i],
                effective[(i + 1) % n],
                effective[j],
                effective[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection { edge_a: i, edge_b: j });
            }
        }
    }

    let min_index = (0..n).min_by_key(|&i| effective[i]).unwrap();
    let forward = area > 0;
    let mut vertices = Vec::with_capacity(n);
    let mut source_indices = Vec::with_capacity(n);
    for k in 0..n {
        let idx = if forward { (min_index + k) % n } else { (min_index + n - k) % n };
        vertices.push(effective[idx]);
        source_indices.push(idx);
    }
    let measures = measures_for(&vertices);
    Ok(Validated { vertices, source_indices, measures })
}
