```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }

    if board.cells.len() != board.width * board.height {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0
            && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two())
        {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

fn process_line(
    board: &Board,
    output: &mut [u32],
    merges: &mut Vec<Merge>,
    score: &mut u64,
    direction: Direction,
    line_number: usize,
) -> Result<(), Error> {
    let mut indices = Vec::new();

    match direction {
        Direction::Left | Direction::Right => {
            for step in 0..board.width {
                let col = if direction == Direction::Left {
                    step
                } else {
                    board.width - 1 - step
                };
                indices.push(line_number * board.width + col);
            }
        }
        Direction::Up | Direction::Down => {
            for step in 0..board.height {
                let row = if direction == Direction::Up {
                    step
                } else {
                    board.height - 1 - step
                };
                indices.push(row * board.width + line_number);
            }
        }
    }

    let tiles: Vec<(usize, u32)> = indices
        .into_iter()
        .filter_map(|index| {
            let value = board.cells[index];
            (value != 0).then_some((index, value))
        })
        .collect();

    let mut packed = Vec::with_capacity(tiles.len());
    let mut i = 0;
    while i < tiles.len() {
        if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
            let value = tiles[i].1;
            if value == 2_147_483_648 {
                return Err(Error::MergeOverflow {
                    first: tiles[i].0,
                    second: tiles[i + 1].0,
                });
            }

            let merged_value = value * 2;
            let to = indices_for_destination(board, direction, line_number, packed.len());
            merges.push(Merge {
                from: [tiles[i].0, tiles[i + 1].0],
                to,
                value: merged_value,
            });
            *score += u64::from(merged_value);
            packed.push(merged_value);
            i += 2;
        } else {
            packed.push(tiles[i].1);
            i += 1;
        }
    }

    for (position, value) in packed.into_iter().enumerate() {
        let destination = indices_for_destination(board, direction, line_number, position);
        output[destination] = value;
    }

    Ok(())
}

fn indices_for_destination(
    board: &Board,
    direction: Direction,
    line_number: usize,
    position: usize,
) -> usize {
    match direction {
        Direction::Left => line_number * board.width + position,
        Direction::Right => line_number * board.width + (board.width - 1 - position),
        Direction::Up => position * board.width + line_number,
        Direction::Down => (board.height - 1 - position) * board.width + line_number,
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut output = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line_number in 0..line_count {
        process_line(
            board,
            &mut output,
            &mut merges,
            &mut score,
            direction,
            line_number,
        )?;
    }

    let changed = output != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells: output,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let directions = [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ];
    let mut legal = Vec::new();

    for direction in directions {
        if let Ok(result) = move_board(board, direction) {
            if result.changed {
                legal.push(direction);
            }
        }
    }

    Ok(legal)
}
```
