#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_literal(word: &mut Word, c: char, wildcard_active: bool) {
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, wildcard_active: bool) {
    for c in value.chars() {
        append_literal(word, c, wildcard_active);
    }
}

fn first_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let bytes = input.as_bytes();
    let mut close = dollar + 2;

    while close < bytes.len() {
        if bytes[close] == b'\\' {
            close += 1;
            if close < bytes.len() {
                let c = input[close..].chars().next().unwrap();
                close += c.len_utf8();
            }
        } else if bytes[close] == b'}' {
            break;
        } else {
            let c = input[close..].chars().next().unwrap();
            close += c.len_utf8();
        }
    }

    if close >= bytes.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let contents = &input[dollar + 2..close];
    let content_bytes = contents.as_bytes();
    let mut name_end = 0;

    if content_bytes.is_empty() || !is_name_start(contents.chars().next().unwrap()) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let first = contents.chars().next().unwrap();
    name_end += first.len_utf8();
    while name_end < content_bytes.len() {
        let c = contents[name_end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        name_end += c.len_utf8();
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];
    let value = if rest.is_empty() {
        first_env(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match first_env(env, name) {
            Some(v) if !v.is_empty() => v.to_string(),
            _ => {
                let mut decoded = String::new();
                let mut i = 0;
                while i < default.len() {
                    let c = default[i..].chars().next().unwrap();
                    if c == '\\' {
                        i += 1;
                        if i < default.len() {
                            let next = default[i..].chars().next().unwrap();
                            decoded.push(next);
                            i += next.len_utf8();
                        } else {
                            decoded.push('\\');
                        }
                    } else {
                        decoded.push(c);
                        i += c.len_utf8();
                    }
                }
                decoded
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;
    let bytes = input.as_bytes();

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let width = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut current, c, false);
                }
                i += width;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i += width;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    match next {
                        '"' | '\\' | '$' => {
                            append_literal(&mut current, next, false);
                            i += next.len_utf8();
                        }
                        '\n' => {
                            i += 1;
                        }
                        _ => {
                            append_literal(&mut current, '\\', false);
                        }
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if bytes.get(i + 1) == Some(&b'(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if bytes.get(i + 1) == Some(&b'{') {
                        let (next_i, value) = parse_expansion(input, i, env)?;
                        append_expansion(&mut current, &value, false);
                        active = true;
                        i = next_i;
                    } else if i + 1 < input.len()
                        && is_name_start(input[i + 1..].chars().next().unwrap())
                    {
                        let mut end = i + 1;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[i + 1..end];
                        append_expansion(&mut current, first_env(env, name).unwrap_or(""), false);
                        active = true;
                        i = end;
                    } else {
                        append_literal(&mut current, '$', false);
                        i += width;
                    }
                } else {
                    append_literal(&mut current, c, false);
                    i += width;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += width;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = i;
                    active = true;
                    i += width;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = i;
                    active = true;
                    i += width;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '\n' {
                        i += 1;
                    } else {
                        append_literal(&mut current, next, false);
                        active = true;
                        i += next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if bytes.get(i + 1) == Some(&b'(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if bytes.get(i + 1) == Some(&b'{') {
                        let (next_i, value) = parse_expansion(input, i, env)?;
                        append_expansion(&mut current, &value, true);
                        active = true;
                        i = next_i;
                    } else if i + 1 < input.len()
                        && is_name_start(input[i + 1..].chars().next().unwrap())
                    {
                        let mut end = i + 1;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[i + 1..end];
                        append_expansion(&mut current, first_env(env, name).unwrap_or(""), true);
                        active = true;
                        i = end;
                    } else {
                        append_literal(&mut current, '$', true);
                        active = true;
                        i += width;
                    }
                } else {
                    append_literal(&mut current, c, true);
                    active = true;
                    i += width;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        let chosen = candidates
            .into_iter()
            .flatten()
            .min_by_key(|candidate| candidate.len());

        let candidate = chosen.unwrap_or_else(|| {
            let mut out = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    out.push('\'');
                    out.push('\\');
                    out.push('\'');
                    out.push('\'');
                } else {
                    out.push(c);
                }
            }
            out.push('\'');
            out
        });
        encoded.push(candidate);
    }

    Ok(encoded.join(" "))
}
