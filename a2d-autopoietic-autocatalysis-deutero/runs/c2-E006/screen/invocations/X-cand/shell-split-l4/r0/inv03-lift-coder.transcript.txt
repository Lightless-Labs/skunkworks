fn parse_braced(input: &str, dollar: usize) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += 1;
            if cursor < input.len() {
                cursor += input[cursor..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            let contents = &input[content_start..cursor];
            let end = cursor + 1;

            let name_len = contents
                .char_indices()
                .take_while(|(i, c)| {
                    if *i == 0 {
                        is_name_start(*c)
                    } else {
                        is_name_continue(*c)
                    }
                })
                .map(|(_, c)| c.len_utf8())
                .sum::<usize>();

            if name_len == 0 {
                return Err(SplitError::InvalidExpansion { pos: dollar });
            }

            let name = &contents[..name_len];
            let rest = &contents[name_len..];

            if rest.is_empty() {
                return Ok((
                    end,
                    first_env(&[], name).unwrap_or("").to_string(),

            }

            if let Some(default_text) = rest.strip_prefix(":-") {
                return Ok((end, default_text.to_string()));
            }

            return Err(SplitError::InvalidExpansion { pos: dollar });
        } else {
            cursor += c.len_utf8();
        }
    }

    Err(SplitError::UnterminatedExpansion { pos: dollar })
}

            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        return None;
    }
    Some(format!("'{}'", word))
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }

    let mut result = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

fn encode_word(word: &str) -> String {
    let candidates = [
        bare_candidate(word),
        single_candidate(word),
        double_candidate(word),
    ];

    let mut best: Option<(usize, String)> = None;
    for (priority, candidate) in candidates.into_iter().enumerate() {
        if let Some(candidate) = candidate {
            let len = candidate.len();
            if best.as_ref().is_none_or(|(best_len, _)| len < *best_len) {
                best = Some((len, candidate));
            } else if best.as_ref().is_some_and(|(best_len, _)| len == *best_len) {
                let _ = priority;
            }
        }
    }

    if let Some((_, candidate)) = best {
        return candidate;
    }

    let mut result = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            result.push('\'');
            result.push('\\');
            result.push('\'');
            result.push('\'');
        } else {
            result.push(c);
        }
    }
    result.push('\'');
    result
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul { word: index, pos });
        }
    }

    Ok(words
        .iter()
        .map(|word| encode_word(word))
        .collect::<Vec<_>>()
        .join(" "))
}
