#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn append_char(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_str(word: &mut Word, s: &str, active_glob: bool) {
    for c in s.chars() {
        append_char(word, c, active_glob);
    }
}

fn first_env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }

    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().expect("valid UTF-8 boundary");
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut current, c, false);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    let after_next = next_i + next.len_utf8();
                    match next {
                        '"' | '\\' | '$' => {
                            append_char(&mut current, next, false);
                            i = after_next;
                        }
                        '\n' => {
                            i = after_next;
                        }
                        _ => {
                            append_char(&mut current, '\\', false);
                            i = next_i;
                        }
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let after_dollar = next_i;
                    let following = input[after_dollar..].chars().next();
                    if following == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if following == Some('{') {
                        let content_start = after_dollar + 1;
                        let mut cursor = content_start;
                        let mut closing = None;
                        while cursor < input.len() {
                            let ch = input[cursor..].chars().next().expect("valid UTF-8 boundary");
                            let ch_end = cursor + ch.len_utf8();
                            if ch == '\\' {
                                if ch_end < input.len() {
                                    let skipped = input[ch_end..].chars().next().expect("valid UTF-8 boundary");
                                    cursor = ch_end + skipped.len_utf8();
                                } else {
                                    cursor = ch_end;
                                }
                            } else if ch == '}' {
                                closing = Some(cursor);
                                break;
                            } else {
                                cursor = ch_end;
                            }
                        }
                        let Some(close_pos) = closing else {
                            return Err(SplitError::UnterminatedExpansion { pos: i });
                        };
                        let contents = &input[content_start..close_pos];
                        let (name, default) = if let Some(op_pos) = contents.find(":-") {
                            (&contents[..op_pos], Some(&contents[op_pos + 2..]))
                        } else {
                            (contents, None)
                        };
                        let mut name_chars = name.chars();
                        let valid_name = name_chars.next().is_some_and(is_name_start)
                            && name_chars.all(is_name_continue);
                        let valid_form = valid_name
                            && (default.is_some()
                                || contents.chars().all(is_name_continue));
                        if !valid_form {
                            return Err(SplitError::InvalidExpansion { pos: i });
                        }

                        let value = first_env_value(env, name);
                        let selected = match default {
                            Some(text) if value.is_none_or(str::is_empty) => {
                                decode_default(text)
                            }
                            _ => value.unwrap_or("").to_string(),
                        };
                        append_str(&mut current, &selected, false);
                        active = true;
                        i = close_pos + 1;
                    } else if let Some(name_start) = following.filter(|ch| is_name_start(*ch)) {
                        let _ = name_start;
                        let mut end = after_dollar;
                        while end < input.len() {
                            let ch = input[end..].chars().next().expect("valid UTF-8 boundary");
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[after_dollar..end];
                        let value = first_env_value(env, name).unwrap_or("");
                        append_str(&mut current, value, false);
                        active = true;
                        i = end;
                    } else {
                        append_char(&mut current, '$', false);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut current, c, false);
                    active = true;
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        append_char(&mut current, next, false);
                        active = true;
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let following = input[next_i..].chars().next();
                    if following == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if following == Some('{') {
                        let content_start = next_i + 1;
                        let mut cursor = content_start;
                        let mut closing = None;
                        while cursor < input.len() {
                            let ch = input[cursor..].chars().next().expect("valid UTF-8 boundary");
                            let ch_end = cursor + ch.len_utf8();
                            if ch == '\\' {
                                if ch_end < input.len() {
                                    let skipped = input[ch_end..].chars().next().expect("valid UTF-8 boundary");
                                    cursor = ch_end + skipped.len_utf8();
                                } else {
                                    cursor = ch_end;
                                }
                            } else if ch == '}' {
                                closing = Some(cursor);
                                break;
                            } else {
                                cursor = ch_end;
                            }
                        }
                        let Some(close_pos) = closing else {
                            return Err(SplitError::UnterminatedExpansion { pos: i });
                        };
                        let contents = &input[content_start..close_pos];
                        let (name, default) = if let Some(op_pos) = contents.find(":-") {
                            (&contents[..op_pos], Some(&contents[op_pos + 2..]))
                        } else {
                            (contents, None)
                        };

                        let mut name_chars = name.chars();
                        let valid_name = name_chars.next().is_some_and(is_name_start)
                            && name_chars.all(is_name_continue);
                        let valid_form = valid_name
                            && (default.is_some()
                                || contents.chars().all(is_name_continue));
                        if !valid_form {
                            return Err(SplitError::InvalidExpansion { pos: i });
                        }

                        let value = first_env_value(env, name);
                        let selected = match default {
                            Some(text) if value.is_none_or(str::is_empty) => {
                                decode_default(text)
                            }
                            _ => value.unwrap_or("").to_string(),
                        };
                        append_str(&mut current, &selected, true);
                        active = true;
                        i = close_pos + 1;
                    } else if following.is_some_and(is_name_start) {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().expect("valid UTF-8 boundary");
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_i..end];
                        let value = first_env_value(env, name).unwrap_or("");
                        append_str(&mut current, value, true);
                        active = true;
                        i = end;
                    } else {
                        append_char(&mut current, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut current, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        return None;
    }
    Some(format!("'{}'", word))
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::new();
    out.push('"');
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn encode_word(word: &str) -> String {
    let mut candidates = Vec::new();
    if let Some(candidate) = bare_candidate(word) {
        candidates.push((candidate, 0usize));
    }
    if let Some(candidate) = single_candidate(word) {
        candidates.push((candidate, 1usize));
    }
    if let Some(candidate) = double_candidate(word) {
        candidates.push((candidate, 2usize));
    }

    if let Some((candidate, _)) = candidates
        .into_iter()
        .min_by_key(|(candidate, priority)| (candidate.len(), *priority))
    {
        return candidate;
    }

    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    Ok(words
        .iter()
        .map(|word| encode_word(word))
        .collect::<Vec<_>>()
        .join(" "))
}
