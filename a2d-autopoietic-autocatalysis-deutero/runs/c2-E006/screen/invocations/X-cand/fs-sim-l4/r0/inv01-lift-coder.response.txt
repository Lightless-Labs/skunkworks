```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize_components<I, S>(components: I) -> Vec<String>
where
    I: IntoIterator<Item = S>,
    S: AsRef<str>,
{
    let mut out = Vec::new();
    for part in components {
        let part = part.as_ref();
        match part {
            "" | "." => {}
            ".." => {
                out.pop();
            }
            _ => out.push(part.to_owned()),
        }
    }
    out
}

fn split_canonical(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn join_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn canonical(path: &str, cwd: &str) -> String {
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        split_canonical(cwd)
    };
    parts.extend(path.split('/').map(str::to_owned));
    join_components(&normalize_components(parts))
}

fn parent_and_name(path: &str) -> (String, String) {
    let parts = split_canonical(path);
    let name = parts.last().cloned().unwrap_or_default();
    let parent = join_components(&parts[..parts.len().saturating_sub(1)]);
    (parent, name)
}

fn is_within(path: &str, dir: &str) -> bool {
    path == dir || dir == "/" || path.starts_with(&format!("{}/", dir))
}

fn child_path(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn entry_kind(entry: &Entry) -> Kind {
        match entry {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }

    fn resolve(&self, path: &str, create_final: bool) -> Result<String, Error> {
        let mut pending: Vec<String> = split_canonical(path);
        let mut physical: Vec<String> = Vec::new();
        let mut seen = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(join_components(&physical));
            }

            let component = pending.remove(0);
            let candidate_parts = {
                let mut v = physical.clone();
                v.push(component.clone());
                v
            };
            let candidate = join_components(&candidate_parts);

            let current = join_components(&physical);
            match self.entries.get(&current) {
                Some(Entry::Dir) => {}
                Some(_) => return Err(Error::NotDir(current)),
                None => return Err(Error::Missing(current)),
            }

            match self.entries.get(&candidate) {
                None if create_final && pending.is_empty() => return Ok(candidate),
                None => return Err(Error::Missing(candidate)),
                Some(Entry::Link(target)) => {
                    if !seen.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let mut target_parts = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        physical.clone()
                    };
                    target_parts.extend(target.split('/').map(str::to_owned));
                    let mut expanded = normalize_components(target_parts);
                    expanded.extend(pending.into_iter());
                    pending = normalize_components(expanded);
                    physical.clear();
                }
                Some(_) => {
                    if !pending.is_empty() && !matches!(self.entries.get(&candidate), Some(Entry::Dir)) {
                        return Err(Error::NotDir(candidate));
                    }
                    physical.push(component);
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = parent_and_name(path);
        let resolved_parent = self.resolve(&parent, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn direct_children(&self, path: &str) -> Vec<(String, String)> {
        self.entries
            .iter()
            .filter_map(|(candidate, entry)| {
                if candidate == path {
                    return None;
                }
                let (parent, name) = parent_and_name(candidate);
                if parent == path {
                    Some((name, candidate.clone()))
                } else {
                    let _ = entry;
                    None
                }
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        let (parent, name) = self.resolve_parent(&path)?;
        let destination = child_path(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        let destination = self.resolve(&path, true)?;
        match self.entries.get(&destination) {
            Some(Entry::Dir) => return Err(Error::IsDir(destination)),
            Some(Entry::File(_)) => {}
            Some(Entry::Link(_)) => unreachable!(),
            None => {}
        }
        self.entries
            .insert(destination, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        let resolved = self.resolve(&path, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        let resolved = self.resolve(&path, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut result = Vec::new();
        for (name, child) in self.direct_children(&resolved) {
            if let Some(entry) = self.entries.get(&child) {
                result.push((name, Self::entry_kind(entry)));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        let resolved = self.resolve(&path, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        validate_path(from)?;
        let from = canonical(from, &self.cwd);
        validate_path(to)?;
        let to = canonical(to, &self.cwd);

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&from)?;
        let source = child_path(&src_parent, &src_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&to)?;
        let destination = child_path(&dst_parent, &dst_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let dest_entry = self.entries.get(&destination).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }
        if let Some(ref dest_entry) = dest_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.direct_children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| is_within(p, &source))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if let Some(dest_entry) = dest_entry {
            if matches!(dest_entry, Entry::Dir) {
                let keys: Vec<String> = self
                    .entries
                    .keys()
                    .filter(|p| is_within(p, &destination))
                    .cloned()
                    .collect();
                for key in keys {
                    self.entries.remove(&key);
                }
            } else {
                self.entries.remove(&destination);
            }
        }

        let keys: Vec<String> = moved.iter().map(|(p, _)| p.clone()).collect();
        for key in keys {
            self.entries.remove(&key);
        }
        for (old, entry) in moved {
            let suffix = &old[source.len()..];
            self.entries.insert(format!("{destination}{suffix}"), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        if path == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&path)?;
        let target = child_path(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.direct_children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        if matches!(entry, Entry::Dir) && recursive {
            let keys: Vec<String> = self
                .entries
                .keys()
                .filter(|p| is_within(p, &target))
                .cloned()
                .collect();
            for key in keys {
                self.entries.remove(&key);
            }
        } else {
            self.entries.remove(&target);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        validate_path(link)?;
        let link = canonical(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (parent, name) = self.resolve_parent(&link)?;
        let destination = child_path(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        let (parent, name) = parent_and_name(&path);
        let resolved_parent = self.resolve(&parent, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        let final_path = child_path(&resolved_parent, &name);
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        validate_path(path)?;
        let path = canonical(path, &self.cwd);
        let resolved = self.resolve(&path, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut rows = Vec::new();
        for (candidate, entry) in &self.entries {
            if !is_within(candidate, &resolved) {
                continue;
            }
            let size = match entry {
                Entry::Dir => self
                    .entries
                    .iter()
                    .filter(|(p, _)| is_within(p, candidate))
                    .map(|(p, e)| {
                        if p == candidate {
                            0
                        } else {
                            match e {
                                Entry::File(data) => data.len() as u64,
                                Entry::Link(target) => target.len() as u64,
                                Entry::Dir => 0,
                            }
                        }
                    })
                    .sum(),
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            rows.push((candidate.clone(), size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}
```
