```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|c| !c.is_empty() && *c != ".")
            .fold(Vec::new(), |mut acc, c| {
                if c == ".." {
                    acc.pop();
                } else {
                    acc.push(c.to_string());
                }
                acc
            })
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for c in path.split('/') {
            if c.is_empty() || c == "." {
                continue;
            }
            if c == ".." {
                parts.pop();
            } else {
                parts.push(c.to_string());
            }
        }
        Ok(Self::canonical(&parts))
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            _ => "/".to_string(),
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            String::new()
        } else {
            path.rsplit('/').next().unwrap_or("").to_string()
        }
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        path == dir || (dir == "/" && path.starts_with('/'))
            || (dir != "/" && path.strip_prefix(dir).is_some_and(|s| s.starts_with('/')))
    }

    fn children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| *p != dir && Self::parent(p) == dir)
            .cloned()
            .collect()
    }

    fn resolve(&self, initial: &str, allow_missing_final: bool) -> Result<String, Error> {
        let mut todo = Self::components(initial);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if todo.is_empty() {
                return Ok(Self::canonical(&resolved));
            }

            let name = todo.remove(0);
            let candidate_parts = {
                let mut p = resolved.clone();
                p.push(name);
                p
            };
            let candidate = Self::canonical(&candidate_parts);
            let is_final = todo.is_empty();

            match self.entries.get(&candidate) {
                None if is_final && allow_missing_final => return Ok(candidate),
                None => return Err(Error::Missing(candidate)),
                Some(Entry::Dir) => resolved.push(Self::basename(&candidate)),
                Some(Entry::File(_)) if !is_final => return Err(Error::NotDir(candidate)),
                Some(Entry::File(_)) => return Ok(candidate),
                Some(Entry::Link(target)) => {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let mut target_parts = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        resolved.clone()
                    };
                    for c in target.split('/') {
                        if c.is_empty() || c == "." {
                            continue;
                        }
                        if c == ".." {
                            target_parts.pop();
                        } else {
                            target_parts.push(c.to_string());
                        }
                    }
                    let mut expanded = target_parts;
                    expanded.extend(todo.into_iter());
                    let expanded_path = Self::canonical(&Self::components(
                        &format!("/{}", expanded.join("/")),
                    ));
                    todo = Self::components(&expanded_path);
                    resolved.clear();
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let name = Self::basename(path);
        let resolved = self.resolve(&parent, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Ok((resolved, name)),
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    fn resolved_child(parent: &str, name: &str) -> String {
        Self::join(parent, name)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = Self::resolved_child(&parent, &name);
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        let target = self.resolve(&canonical, true)?;
        match self.entries.get(&target) {
            Some(Entry::Dir) => Err(Error::IsDir(target)),
            Some(Entry::File(_)) => {
                self.entries.insert(target, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let parent = Self::parent(&target);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {
                        self.entries.insert(target, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(parent)),
                    None => Err(Error::Missing(parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        let target = self.resolve(&canonical, false)?;
        match self.entries.get(&target) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(target)),
            _ => Err(Error::Missing(target)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        let target = self.resolve(&canonical, false)?;
        if !matches!(self.entries.get(&target), Some(Entry::Dir)) {
            return Err(Error::NotDir(target));
        }
        let mut result: Vec<_> = self
            .children(&target)
            .into_iter()
            .map(|p| {
                let kind = match self.entries.get(&p).unwrap() {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                    Entry::Link(_) => Kind::Link,
                };
                (Self::basename(&p), kind)
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        let target = self.resolve(&canonical, false)?;
        if !matches!(self.entries.get(&target), Some(Entry::Dir)) {
            return Err(Error::NotDir(target));
        }
        self.cwd = target;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = Self::normalize(from, &self.cwd)?;
        let to = Self::normalize(to, &self.cwd)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&from)?;
        let src = Self::join(&src_parent, &src_name);
        let source = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(source, Entry::Dir) && Self::is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&to)?;
        let dst = Self::join(&dst_parent, &dst_name);

        if matches!(self.entries.get(&dst), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && Self::is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        if let Some(dest) = self.entries.get(&dst) {
            if !overwrite {
                return Err(Error::Exists(dst));
            }
            match (&source, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&dst).is_empty() {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| **p == src || Self::is_within(p, &src))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if self.entries.contains_key(&dst) {
            let victims: Vec<String> = self
                .entries
                .keys()
                .filter(|p| **p == dst || Self::is_within(p, &dst))
                .cloned()
                .collect();
            for p in victims {
                self.entries.remove(&p);
            }
        }

        for (p, _) in &moving {
            self.entries.remove(p);
        }
        for (old, entry) in moving {
            let suffix = if old == src { String::new() } else { old[src.len()..].to_string() };
            let new_path = format!("{dst}{suffix}");
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = Self::join(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        let victims: Vec<String> = self
            .entries
            .keys()
            .filter(|p| **p == target || (recursive && Self::is_within(p, &target)))
            .cloned()
            .collect();
        for p in victims {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical = Self::normalize(link, &self.cwd)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let path = Self::join(&parent, &name);
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = Self::join(&parent, &name);
        match self.entries.get(&target) {
            Some(Entry::Link(text)) => Ok(text.clone()),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = Self::normalize(path, &self.cwd)?;
        let target = self.resolve(&canonical, false)?;
        let entry = self.entries.get(&target).ok_or_else(|| Error::Missing(target.clone()))?;
        if !matches!(entry, Entry::Dir) {
            let size = match entry {
                Entry::File(data) => data.len() as u64,
                Entry::Link(text) => text.len() as u64,
                Entry::Dir => 0,
            };
            return Ok(vec![(target, size)]);
        }

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| **p == target || Self::is_within(p, &target))
            .cloned()
            .collect();

        let mut sizes = BTreeMap::new();
        for p in &paths {
            let size = match self.entries.get(p).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(text) => text.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p.clone(), size);
        }

        for p in paths.iter().rev() {
            let size = *sizes.get(p).unwrap_or(&0);
            if p != &target {
                let parent = Self::parent(p);
                if let Some(total) = sizes.get_mut(&parent) {
                    *total = total.saturating_add(size);
                }
            }
        }

        Ok(sizes.into_iter().collect())
    }
}
```
