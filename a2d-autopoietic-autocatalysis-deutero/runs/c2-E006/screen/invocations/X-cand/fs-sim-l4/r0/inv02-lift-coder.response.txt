use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize_path(path)?;
        let parent = parent_path(&canonical);
        let resolved_parent = self.resolve_full(&parent)?;
        self.require_dir(&resolved_parent)?;
        let final_path = join_path(&resolved_parent, basename(&canonical));
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize_path(path)?;
        let resolved = self.resolve_write(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let parent = parent_path(&resolved);
                self.require_dir(&parent)?;
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize_path(path)?;
        let resolved = self.resolve_full(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize_path(path)?;
        let resolved = self.resolve_full(&canonical)?;
        self.require_dir(&resolved)?;
        let prefix = if resolved == "/" {
            "/".to_string()
        } else {
            format!("{}/", resolved)
        };
        let mut rows = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == &resolved || !entry_path.starts_with(&prefix) {
                continue;
            }
            let rest = &entry_path[prefix.len()..];
            if rest.is_empty() || rest.contains('/') {
                continue;
            }
            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
                Entry::Link(_) => Kind::Link,
            };
            rows.push((rest.to_string(), kind));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize_path(path)?;
        let resolved = self.resolve_full(&canonical)?;
        self.require_dir(&resolved)?;
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize_path(from)?;
        let destination = self.normalize_path(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let source_parent = self.resolve_full(&parent_path(&source))?;
        self.require_dir(&source_parent)?;
        let source_path = join_path(&source_parent, basename(&source));
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && contains_path(&source_path, &self.cwd) {
            return Err(Error::Busy(source_path));
        }

        let destination_parent = self.resolve_full(&parent_path(&destination))?;
        self.require_dir(&destination_parent)?;
        let destination_path = join_path(&destination_parent, basename(&destination));

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && contains_path(&destination_path, &self.cwd)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && contains_path(&source_path, &destination_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if overwrite {
            if let Some(ref target_entry) = destination_entry {
                match (&source_entry, target_entry) {
                    (Entry::Dir, Entry::Dir) => {
                        if self.has_children(&destination_path) {
                            return Err(Error::NotEmpty(destination_path));
                        }
                    }
                    (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                    (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                    _ => {}
                }
            }
        }

        let moved = self.collect_subtree(&source_path);
        if destination_entry.is_some() {
            self.remove_subtree(&destination_path);
        }
        self.remove_subtree(&source_path);
        for (old_path, entry) in moved {
            let suffix = &old_path[source_path.len()..];
            let new_path = format!("{}{}", destination_path, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize_path(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let parent = self.resolve_full(&parent_path(&canonical))?;
        self.require_dir(&parent)?;
        let resolved = join_path(&parent, basename(&canonical));
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && contains_path(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir) && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        self.remove_subtree(&resolved);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = self.normalize_path(link)?;
        validate_target(target)?;

        let parent = self.resolve_full(&parent_path(&canonical_link))?;
        self.require_dir(&parent)?;
        let resolved_link = join_path(&parent, basename(&canonical_link));
        if self.entries.contains_key(&resolved_link) {
            return Err(Error::Exists(resolved_link));
        }
        self.entries
            .insert(resolved_link, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize_path(path)?;
        let resolved = self.resolve_parent_and_final(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize_path(path)?;
        let resolved = self.resolve_full(&canonical)?;
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let paths = self.collect_subtree(&resolved);
        let mut rows = Vec::new();
        for (entry_path, entry) in paths {
            let size = match entry {
                Entry::Dir => {
                    if entry_path == resolved {
                        self.subtree_payload_size(&resolved)
                    } else {
                        self.subtree_payload_size(&entry_path)
                    }
                }
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            rows.push((entry_path, size));
        }
        if !rows.iter().any(|(p, _)| p == &resolved) {
            let size = match entry {
                Entry::Dir => self.subtree_payload_size(&resolved),
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            rows.push((resolved, size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    fn normalize_path(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(normalize_from(path, &self.cwd))
    }

    fn require_dir(&self, path: &str) -> Result<(), Error> {
        match self.entries.get(path) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_string())),
            None => Err(Error::Missing(path.to_string())),
        }
    }

    fn resolve_full(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, true, false)
    }

    fn resolve_write(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, true, true)
    }

    fn resolve_parent_and_final(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, false, false)
    }

    fn resolve(&self, path: &str, follow_final: bool, allow_missing_final: bool) -> Result<String, Error> {
        let mut current = path.to_string();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            let components = components(&current);
            let mut physical = String::from("/");
            let mut restarted = false;

            for (index, component) in components.iter().enumerate() {
                let next = join_path(&physical, component);
                let final_component = index + 1 == components.len();

                if !final_component {
                    match self.entries.get(&physical) {
                        Some(Entry::Dir) => {}
                        Some(_) => return Err(Error::NotDir(physical)),
                        None => return Err(Error::Missing(physical)),
                    }
                }

                let should_follow = !final_component || follow_final;
                match self.entries.get(&next) {
                    Some(Entry::Link(target)) if should_follow => {
                        if visited.contains(&next) {
                            return Err(Error::Loop(next));
                        }
                        if expansions >= 8 {
                            return Err(Error::Depth(next));
                        }
                        visited.insert(next.clone());
                        expansions += 1;

                        let suffix = components[index + 1..].to_vec();
                        let base = if target.starts_with('/') {
                            normalize_from(target, "/")
                        } else {
                            normalize_from(target, &physical)
                        };
                        let mut combined = base;
                        for part in suffix {
                            combined = join_path(&combined, &part);
                        }
                        current = normalize_from(&combined, "/");
                        restarted = true;
                        break;
                    }
                    Some(_) => {
                        if !final_component {
                            match self.entries.get(&next) {
                                Some(Entry::Dir) => {}
                                Some(_) => return Err(Error::NotDir(next)),
                                None => return Err(Error::Missing(next)),
                            }
                        }
                        physical = next;
                    }
                    None => {
                        if final_component && allow_missing_final {
                            physical = next;
                        } else {
                            return Err(Error::Missing(next));
                        }
                    }
                }
            }

            if restarted {
                continue;
            }

            if components.is_empty() {
                return Ok("/".to_string());
            }

            if allow_missing_final && !self.entries.contains_key(&physical) {
                let parent = parent_path(&physical);
                self.require_dir(&parent)?;
            }
            return Ok(physical);
        }
    }

    fn has_children(&self, path: &str) -> bool {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };
        self.entries
            .keys()
            .any(|candidate| candidate != path && candidate.starts_with(&prefix))
    }

    fn collect_subtree(&self, path: &str) -> Vec<(String, Entry)> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };
        self.entries
            .iter()
            .filter(|(candidate, _)| **candidate == path || candidate.starts_with(&prefix))
            .map(|(candidate, entry)| (candidate.clone(), entry.clone()))
            .collect()
    }

    fn remove_subtree(&mut self, path: &str) {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };
        let to_remove: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| **candidate == path || candidate.starts_with(&prefix))
            .cloned()
            .collect();
        for candidate in to_remove {
            self.entries.remove(&candidate);
        }
    }

    fn subtree_payload_size(&self, path: &str) -> u64 {
        self.collect_subtree(path)
            .into_iter()
            .map(|(_, entry)| match entry {
                Entry::Dir => 0,
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
            })
            .sum()
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidTarget(i));
    }
    Ok(())
}

fn normalize_from(path: &str, base: &str) -> String {
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(base)
    };
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_string()),
        }
    }
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .fold(Vec::new(), |mut parts, part| {
            if part == ".." {
                parts.pop();
            } else {
                parts.push(part.to_string());
            }
            parts
        })
}

fn parent_path(path: &str) -> String {
    if path == "/" {
        return "/".to_string();
    }
    match path.rfind('/') {
        Some(0) | None => "/".to_string(),
        Some(index) => path[..index].to_string(),
    }
}

fn basename(path: &str) -> &str {
    path.rsplit('/').next().unwrap_or("")
}

fn join_path(parent: &str, child: &str) -> String {
    if parent == "/" {
        format!("/{}", child)
    } else if child.is_empty() {
        parent.to_string()
    } else {
        format!("{}/{}", parent, child)
    }
}

fn contains_path(directory: &str, candidate: &str) -> bool {
    candidate == directory
        || (directory == "/" && candidate.starts_with('/'))
        || candidate
            .strip_prefix(directory)
            .is_some_and(|suffix| suffix.starts_with('/'))
}
