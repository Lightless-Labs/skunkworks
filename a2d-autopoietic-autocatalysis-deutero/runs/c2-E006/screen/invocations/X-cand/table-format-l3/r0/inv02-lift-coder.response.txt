#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    let minimum = n.saturating_mul(4).saturating_add(1);
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let trimmed: Vec<Vec<String>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| cell.trim_end_matches(' ').to_owned())
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; n];
    for row in &trimmed {
        for (column, cell) in row.iter().enumerate() {
            widths[column] = widths[column].max(cell.chars().count());
        }
    }

    let overhead = n.saturating_mul(3).saturating_add(1);
    if options.max_width != 0 {
        let mut total = overhead.saturating_add(widths.iter().copied().fold(0usize, usize::saturating_add));
        while total > options.max_width {
            let mut widest: Option<usize> = None;
            for column in 0..n {
                if widths[column] > 1
                    && widest.map_or(true, |current| widths[column] >= widths[current])
                {
                    widest = Some(column);
                }
            }
            if let Some(column) = widest {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    fn border(widths: &[usize], out: &mut String) {
        out.push('+');
        for &width in widths {
            for _ in 0..width.saturating_add(2) {
                out.push('-');
            }
            out.push('+');
        }
        out.push('\n');
    }

    fn data_line(
        row: &[String],
        widths: &[usize],
        alignments: &[Alignment],
        out: &mut String,
    ) {
        out.push('|');
        for (column, cell) in row.iter().enumerate() {
            let width = widths[column];
            let chars: Vec<char> = cell.chars().collect();
            let content: String = if chars.len() > width {
                chars.iter().take(width - 1).collect::<String>() + "…"
            } else {
                cell.clone()
            };
            let content_width = content.chars().count();
            let extra = width - content_width;
            let alignment = alignments.get(column).unwrap_or(&Alignment::Left);
            let (left, right) = match alignment {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            out.push(' ');
            for _ in 0..left {
                out.push(' ');
            }
            out.push_str(&content);
            for _ in 0..right {
                out.push(' ');
            }
            out.push(' ');
            out.push('|');
        }
        out.push('\n');
    }

    let mut output = String::new();
    let alignments = &options.alignments;
    border(&widths, &mut output);

    for (row_index, row) in trimmed.iter().enumerate() {
        data_line(row, &widths, alignments, &mut output);
        if options.header && row_index == 0 && trimmed.len() > 1 {
            border(&widths, &mut output);
        }
    }

    border(&widths, &mut output);
    Ok(output)
}
