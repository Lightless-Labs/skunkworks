pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Status {
    Queued,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u128,
    status: Status,
}

pub struct Scheduler {
    aging_ms: u64,
    last_successful_time: u64,
    next_sequence: u128,
    entries: Vec<Entry>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_successful_time: 0,
            next_sequence: 0,
            entries: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        if now < self.last_successful_time {
            return Err(Error::TimeReversed);
        }
        if !Self::valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.entries.iter().any(|entry| entry.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self
            .next_sequence
            .checked_add(1)
            .expect("scheduler insertion sequence exhausted");

        self.entries.push(Entry {
            job,
            sequence,
            status: Status::Queued,
        });
        self.last_successful_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_successful_time {
            return Err(Error::TimeReversed);
        }

        let Some(index) = self.entries.iter().position(|entry| entry.job.id == id) else {
            return Err(Error::Unknown);
        };

        match self.entries[index].status {
            Status::Queued => self.entries[index].status = Status::Cancelled,
            Status::Cancelled => return Err(Error::Cancelled),
            Status::Completed => return Err(Error::Completed),
        }

        self.last_successful_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_successful_time {
            return Err(Error::TimeReversed);
        }

        let mut selected: Option<(usize, u8, u128)> = None;

        for (index, entry) in self.entries.iter().enumerate() {
            if entry.status != Status::Queued || entry.job.ready_at > now {
                continue;
            }

            let effective_priority = self.effective_priority(&entry.job, now);
            match selected {
                None => selected = Some((index, effective_priority, entry.sequence)),
                Some((_, best_priority, best_sequence))
                    if effective_priority > best_priority
                        || (effective_priority == best_priority
                            && entry.sequence < best_sequence) =>
                {
                    selected = Some((index, effective_priority, entry.sequence));
                }
                _ => {}
            }
        }

        self.last_successful_time = now;

        if let Some((index, effective_priority, _)) = selected {
            self.entries[index].status = Status::Completed;
            Ok(Some(Dispatch {
                id: self.entries[index].job.id.clone(),
                effective_priority,
            }))
        } else {
            Ok(None)
        }
    }

    fn valid_id(id: &str) -> bool {
        let bytes = id.as_bytes();
        (1..=16).contains(&bytes.len())
            && bytes.iter().all(|&byte| {
                byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-'
            })
    }

    fn effective_priority(&self, job: &Job, now: u64) -> u8 {
        if self.aging_ms == 0 {
            return job.priority;
        }

        let intervals = (now - job.ready_at) / self.aging_ms;
        let remaining = (9 - job.priority) as u64;
        if intervals >= remaining {
            9
        } else {
            job.priority + intervals as u8
        }
    }
}
