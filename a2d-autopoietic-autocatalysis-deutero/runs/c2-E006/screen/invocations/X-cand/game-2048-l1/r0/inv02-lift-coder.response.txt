```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Vec<usize> = match side {
        Side::Left => (0..row.len()).collect(),
        Side::Right => (0..row.len()).rev().collect(),
    };
    let nonempty: Vec<usize> = indices.into_iter().filter(|&i| row[i] != 0).collect();

    let mut cells = vec![0; row.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut cursor = 0usize;
    let mut output_offset = 0usize;

    while cursor < nonempty.len() {
        let first = nonempty[cursor];

        if cursor + 1 < nonempty.len() && row[first] == row[nonempty[cursor + 1]] {
            let second = nonempty[cursor + 1];
            let value = row[first]
                .checked_mul(2)
                .ok_or(Error::MergeOverflow { first, second })?;
            let to = match side {
                Side::Left => output_offset,
                Side::Right => row.len() - 1 - output_offset,
            };
            cells[to] = value;
            merges.push(Merge {
                from: [first, second],
                to,
                value,
            });
            score += u64::from(value);
            cursor += 2;
        } else {
            let to = match side {
                Side::Left => output_offset,
                Side::Right => row.len() - 1 - output_offset,
            };
            cells[to] = row[first];
            cursor += 1;
        }

        output_offset += 1;
    }

    let changed = cells.as_slice() != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
```
