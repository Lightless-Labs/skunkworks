#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();

    if let Some(at) = bytes.iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = bytes.iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = bytes.iter().position(|&b| (1..=31).contains(&b) || b == 127) {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/').find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> impl Iterator<Item = (usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, component| {
            let start = *offset;
            *offset += component.len() + 1;
            Some((start, component))
        })
        .filter(|(_, component)| !component.is_empty())
}

fn reduce(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    let mut stack: Vec<String> = Vec::new();

    for (at, component) in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_string());
                }
            }
            other => stack.push(other.to_string()),
        }
    }

    Ok(stack)
}

fn format_path(parts: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        if parts.is_empty() {
            return "/".to_string();
        }
        let mut result = format!("/{}", parts.join("/"));
        if intent {
            result.push('/');
        }
        result
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        let mut result = parts.join("/");
        if intent {
            result.push('/');
        }
        result
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let parts = reduce(path, absolute)?;
    Ok(format_path(&parts, absolute, intent))
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => {
            let parts = normalized
                .split('/')
                .filter(|component| !component.is_empty())
                .map(str::to_string)
                .collect();
            Ok(parts)
        }
        Err(_) => Err(error),
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde_request = bytes.first() == Some(&b'~');
    let (mut stack, remainder, offset, absolute) = if tilde_request {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let base = home_parts.ok_or(PathError::MissingHome)?;
        if bytes.len() == 1 {
            (base, "", path.len(), true)
        } else {
            (base, &path[2..], 2, true)
        }
    } else if bytes.first() == Some(&b'/') {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_parts, path, 0, true)
    };

    for (relative_at, component) in components(remainder) {
        let at = offset + relative_at;
        match component {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            other => stack.push(other.to_string()),
        }
    }

    Ok(format_path(&stack, absolute, directory_intent(path)))
}
