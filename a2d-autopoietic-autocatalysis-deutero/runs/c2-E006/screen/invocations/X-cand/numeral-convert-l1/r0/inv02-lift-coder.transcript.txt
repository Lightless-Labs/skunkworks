#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

fn encode_in_range(value: u32, case: &RomanCase) -> String {
    if value == 0 {
        return match case {
            RomanCase::Upper => "N".to_string(),
            RomanCase::Lower => "n".to_string(),
        };
    }

    const TOKENS: &[(u32, &str)] = &[
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    let mut remaining = value;
    let mut output = String::new();

    for &(amount, token) in TOKENS {
        while remaining >= amount {
            remaining -= amount;
            match case {
                RomanCase::Upper => output.push_str(token),
                RomanCase::Lower => output.push_str(&token.to_ascii_lowercase()),
            }
        }
    }

    output
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }
    Ok(encode_in_range(value, &case))
}

fn symbol_value(byte: u8) -> u32 {
    match byte.to_ascii_uppercase() {
        b'I' => 1,
        b'V' => 5,
        b'X' => 10,
        b'L' => 50,
        b'C' => 100,
        b'D' => 500,
        b'M' => 1000,
        _ => 0,
    }
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(RomanError::Empty);
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if !matches!(
            byte,
            b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
                | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n'
        ) {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let first_is_lower = bytes[0].is_ascii_lowercase();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte.is_ascii_alphabetic() && byte.is_ascii_lowercase() != first_is_lower {
            return Err(RomanError::MixedCase { index });
        }
    }

    if bytes == b"N" || bytes == b"n" {
        return Ok(0);
    }

    if bytes.iter().any(|&byte| byte == b'N' || byte == b'n') {
        return Err(RomanError::NonCanonical { index: 0 });
    }

    let mut value = 0u32;
    let mut index = 0;
    while index < bytes.len() {
        let current = symbol_value(bytes[index]);
        if index + 1 < bytes.len() {
            let next = symbol_value(bytes[index + 1]);
            if next > current {
                value = value.saturating_add(next - current).min(4000);
                index += 2;
                continue;
            }
        }
        value = value.saturating_add(current).min(4000);
        index += 1;
    }

    if value > 3999 {
        return Err(RomanError::Range);
    }

    let case = if first_is_lower {
        RomanCase::Lower
    } else {
        RomanCase::Upper
    };
    let canonical = encode_in_range(value, &case);
    let canonical_bytes = canonical.as_bytes();

    let differing_index = bytes
        .iter()
        .zip(canonical_bytes.iter())
        .position(|(left, right)| left != right)
        .unwrap_or_else(|| bytes.len().min(canonical_bytes.len()));

    if bytes != canonical_bytes {
        return Err(RomanError::NonCanonical {
            index: differing_index,
        });
    }

    Ok(value)
}
