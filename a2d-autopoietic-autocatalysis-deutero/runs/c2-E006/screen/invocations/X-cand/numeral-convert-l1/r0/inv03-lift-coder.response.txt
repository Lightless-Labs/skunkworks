#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

const TOKENS: [(u32, &str); 13] = [
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
];

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }

    if value == 0 {
        return Ok(match case {
            RomanCase::Upper => "N".to_owned(),
            RomanCase::Lower => "n".to_owned(),
        });
    }

    let mut remainder = value;
    let mut result = String::new();

    for &(token_value, token) in &TOKENS {
        while remainder >= token_value {
            remainder -= token_value;
            match case {
                RomanCase::Upper => result.push_str(token),
                RomanCase::Lower => result.push_str(&token.to_ascii_lowercase()),
            }
        }
    }

    Ok(result)
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !matches!(byte, b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
            | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n')
        {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    let case = if bytes[0].is_ascii_uppercase() {
        RomanCase::Upper
    } else {
        RomanCase::Lower
    };

    for (index, &byte) in bytes.iter().enumerate() {
        let opposite_case = match case {
            RomanCase::Upper => byte.is_ascii_lowercase(),
            RomanCase::Lower => byte.is_ascii_uppercase(),
        };
        if opposite_case {
            return Err(RomanError::MixedCase { index });
        }
    }

    if input == "N" || input == "n" {
        return Ok(0);
    }

    if bytes.iter().any(|&byte| byte == b'N' || byte == b'n') {
        return Err(RomanError::NonCanonical { index: 0 });
    }

    let mut total = 0u32;
    let mut index = 0;
    while index < bytes.len() {
        let current = roman_symbol_value(bytes[index]);
        if index + 1 < bytes.len() {
            let next = roman_symbol_value(bytes[index + 1]);
            if next > current {
                total = total.saturating_add(next - current).min(4000);
                index += 2;
                if total > 3999 {
                    return Err(RomanError::Range);
                }
                continue;
            }
        }
        total = total.saturating_add(current).min(4000);
        index += 1;
        if total > 3999 {
            return Err(RomanError::Range);
        }
    }

    let canonical = roman_encode(total, case)?;
    let canonical_bytes = canonical.as_bytes();
    let first_difference = bytes
        .iter()
        .zip(canonical_bytes.iter())
        .position(|(left, right)| left != right)
        .unwrap_or_else(|| bytes.len().min(canonical_bytes.len()));

    if input != canonical {
        return Err(RomanError::NonCanonical {
            index: first_difference,
        });
    }

    Ok(total)
}

fn roman_symbol_value(byte: u8) -> u32 {
    match byte.to_ascii_uppercase() {
        b'I' => 1,
        b'V' => 5,
        b'X' => 10,
        b'L' => 50,
        b'C' => 100,
        b'D' => 500,
        b'M' => 1000,
        _ => 0,
    }
}
