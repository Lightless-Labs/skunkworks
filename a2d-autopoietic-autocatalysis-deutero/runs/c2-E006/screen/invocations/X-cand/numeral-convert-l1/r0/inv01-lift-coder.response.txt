#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

fn tokens(case: &RomanCase) -> [(u32, &'static str); 13] {
    let upper = [
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    match case {
        RomanCase::Upper => upper,
        RomanCase::Lower => upper.map(|(value, spelling)| {
            let lower = match spelling {
                "CM" => "cm",
                "CD" => "cd",
                "XC" => "xc",
                "XL" => "xl",
                "IX" => "ix",
                "IV" => "iv",
                "M" => "m",
                "D" => "d",
                "C" => "c",
                "L" => "l",
                "X" => "x",
                "V" => "v",
                "I" => "i",
                _ => unreachable!(),
            };
            (value, lower)
        }),
    }
}

fn encode_in_range(value: u32, case: &RomanCase) -> String {
    if value == 0 {
        return match case {
            RomanCase::Upper => "N".to_owned(),
            RomanCase::Lower => "n".to_owned(),
        };
    }

    let mut remaining = value;
    let mut output = String::new();

    for (amount, spelling) in tokens(case) {
        while remaining >= amount {
            output.push_str(spelling);
            remaining -= amount;
        }
    }

    output
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }
    Ok(encode_in_range(value, &case))
}

fn symbol_value(byte: u8) -> u32 {
    match byte {
        b'I' | b'i' => 1,
        b'V' | b'v' => 5,
        b'X' | b'x' => 10,
        b'L' | b'l' => 50,
        b'C' | b'c' => 100,
        b'D' | b'd' => 500,
        b'M' | b'm' => 1000,
        _ => 0,
    }
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    let bytes = input.as_bytes();

    // Validate every byte before considering case.
    for (index, &byte) in bytes.iter().enumerate() {
        if !matches!(
            byte,
            b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
                | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n'
        ) {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let lower_case = bytes[0].is_ascii_lowercase();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte != b'N' && byte != b'n' && byte.is_ascii_lowercase() != lower_case {
            return Err(RomanError::MixedCase { index });
        }
    }

    if bytes == b"N" || bytes == b"n" {
        return Ok(0);
    }

    if bytes.contains(&b'N') || bytes.contains(&b'n') {
        return Err(RomanError::NonCanonical { index: 0 });
    }

    let mut provisional = 0u32;
    let mut index = 0;

    while index < bytes.len() {
        let current = symbol_value(bytes[index]);
        if index + 1 < bytes.len() {
            let next = symbol_value(bytes[index + 1]);
            if next > current {
                provisional = provisional.saturating_add(next - current).min(4000);
                index += 2;
                continue;
            }
        }

        provisional = provisional.saturating_add(current).min(4000);
        index += 1;
    }

    if provisional > 3999 {
        return Err(RomanError::Range);
    }

    let case = if lower_case {
        RomanCase::Lower
    } else {
        RomanCase::Upper
    };
    let expected = encode_in_range(provisional, &case);
    let expected_bytes = expected.as_bytes();

    let first_difference = bytes
        .iter()
        .zip(expected_bytes.iter())
        .position(|(actual, expected)| actual != expected)
        .unwrap_or_else(|| bytes.len().min(expected_bytes.len()));

    if bytes != expected_bytes {
        return Err(RomanError::NonCanonical {
            index: first_difference,
        });
    }

    Ok(provisional)
}
