#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        if let Some((i, _)) = bytes
            .iter()
            .enumerate()
            .find(|(_, b)| b.is_ascii_whitespace() && **b != b' ')
        {
            return Err(CronError::Layout(i));
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut cron = Self {
            minutes: [false; 60],
            hours: [false; 24],
        };

        parse_field(&bytes[..split], 0, 59, &mut cron.minutes)?;
        parse_field(
            &bytes[split + 1..],
            split + 1,
            23,
            &mut cron.hours,
        )?;

        Ok(cron)
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &[u8],
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .iter()
            .position(|&b| b == b',')
            .map_or(field.len(), |p| start + p);

        parse_item(&field[start..end], offset + start, max, selected)?;

        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn parse_item(
    item: &[u8],
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    if item.is_empty() {
        return Err(CronError::Syntax(offset));
    }

    let slashes: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slashes.len() > 1 {
        return Err(CronError::Syntax(offset + slashes[1]));
    }

    let (base_end, step_start) = if let Some(&slash) = slashes.first() {
        (slash, Some(slash + 1))
    } else {
        (item.len(), None)
    };
    let base = &item[..base_end];

    let wildcard = base == b"*";
    let dash = base.iter().position(|&b| b == b'-');

    if step_start.is_some() && !wildcard && dash.is_none() {
        return Err(CronError::Syntax(offset + base_end));
    }

    let (range_start, range_end, dash_offset) = if wildcard {
        (0usize, max, None)
    } else if let Some(dash_pos) = dash {
        let left = parse_number(&base[..dash_pos], offset, max)?;
        let right = parse_number(
            &base[dash_pos + 1..],
            offset + dash_pos + 1,
            max,
        )?;
        if left > right {
            return Err(CronError::Descending(offset + dash_pos));
        }
        (left, right, Some(dash_pos))
    } else {
        let value = parse_number(base, offset, max)?;
        (value, value, None)
    };

    let step = if let Some(step_begin) = step_start {
        let value = parse_digits(
            &item[step_begin..],
            offset + step_begin,
            None,
        )?;
        let step_max = if max == 59 { 60 } else { 24 };
        if value == 0 || value > step_max {
            return Err(CronError::Step(offset + step_begin));
        }
        value
    } else {
        1
    };

    let _ = dash_offset;
    let mut n = range_start;
    while n <= range_end {
        selected[n] = true;
        match n.checked_add(step) {
            Some(next) if next > n => n = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_number(bytes: &[u8], offset: usize, max: usize) -> Result<usize, CronError> {
    parse_digits(bytes, offset, Some(max))
}

fn parse_digits(
    bytes: &[u8],
    offset: usize,
    max: Option<usize>,
) -> Result<usize, CronError> {
    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in bytes {
        let digit = (b - b'0') as usize;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => return Err(max.map_or(CronError::Step(offset), |_| CronError::Bounds(offset))),
        };
    }

    if let Some(limit) = max {
        if value > limit {
            return Err(CronError::Bounds(offset));
        }
    }
    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        let id = id.to_owned();
        self.reserved.push(id.clone());
        self.jobs.push(Job {
            id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll;
        let end_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            match job.policy {
                Policy::All => {
                    let first = lower.map_or(0, |t| t / 60_000 + 1);
                    if first <= end_minute {
                        for minute in first..=end_minute {
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
                Policy::Once => {
                    let first = lower.map_or(0, |t| t / 60_000 + 1);
                    if first <= end_minute {
                        if let Some(minute) = (first..=end_minute)
                            .rev()
                            .find(|&minute| job.cron.matches(minute))
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0
                        && lower.map_or(true, |t| now > t)
                        && job.cron.matches(end_minute)
                    {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
