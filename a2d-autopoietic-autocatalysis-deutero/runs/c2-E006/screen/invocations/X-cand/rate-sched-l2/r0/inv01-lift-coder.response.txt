#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    priority: u8,
    policy: Policy,
    cron: Cron,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        // Layout errors involving non-space ASCII whitespace take precedence.
        if let Some((offset, _)) = bytes
            .iter()
            .enumerate()
            .find(|(_, b)| **b != b' ' && b.is_ascii_whitespace())
        {
            return Err(CronError::Layout(offset));
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, b)| (*b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let separator = spaces[0];
        if separator == 0 || separator + 1 == bytes.len() {
            return Err(CronError::Layout(separator));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(
            &text[..separator],
            0,
            59,
            &mut minutes,
        )?;
        parse_field(
            &text[separator + 1..],
            separator + 1,
            23,
            &mut hours,
        )?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    maximum: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .find(',')
            .map(|n| start + n)
            .unwrap_or(field.len());
        let item = &field[start..end];
        if item.is_empty() {
            return Err(CronError::Syntax(offset + start));
        }
        parse_item(item, offset + start, maximum, selected)?;

        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    maximum: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, b)| (*b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(CronError::Syntax(offset + slash_positions[1]));
    }

    let slash = slash_positions.first().copied();
    let base_end = slash.unwrap_or(bytes.len());
    let base = &item[..base_end];

    if slash.is_some() && base != "*" && !base.contains('-') {
        return Err(CronError::Syntax(offset + slash.unwrap()));
    }

    let (range_start, range_end, dash) = if base == "*" {
        (0, maximum, None)
    } else if let Some(dash) = base.as_bytes().iter().position(|b| *b == b'-') {
        let left = parse_number(&base[..dash], offset, maximum)?;
        let right = parse_number(
            &base[dash + 1..],
            offset + dash + 1,
            maximum,
        )?;
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        (left, right, Some(dash))
    } else {
        let number = parse_number(base, offset, maximum)?;
        (number, number, None)
    };
    let _ = dash;

    let step = if let Some(slash) = slash {
        let step_start = offset + slash + 1;
        let value = parse_number(&item[slash + 1..], step_start, usize::MAX)?;
        let step_max = if maximum == 59 { 60 } else { 24 };
        if value == 0 || value > step_max {
            return Err(CronError::Step(step_start));
        }
        value
    } else {
        1
    };

    let mut value = range_start;
    while value <= range_end {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_number(
    fragment: &str,
    offset: usize,
    maximum: usize,
) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();
    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some((index, _)) = bytes.iter().enumerate().find(|(_, b)| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + index));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for byte in bytes {
        let digit = (byte - b'0') as usize;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(offset)),
        };
    }
    if value > maximum {
        return Err(CronError::Bounds(offset));
    }
    Ok(value)
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            priority,
            policy,
            cron,
            order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(previous) = self.last_poll {
            if now < previous {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let first_minute = match lower_exclusive {
            None => 0,
            Some(previous) => previous / 60_000 + 1,
        };
        let last_minute = now / 60_000;

        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();
        for job in &self.jobs {
            match job.policy {
                Policy::All => {
                    if first_minute <= last_minute {
                        for minute in first_minute..=last_minute {
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
                Policy::Once => {
                    if first_minute <= last_minute {
                        if let Some(minute) = (first_minute..=last_minute)
                            .rev()
                            .find(|minute| job.cron.matches(*minute))
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        let eligible = lower_exclusive
                            .map(|previous| minute * 60_000 > previous)
                            .unwrap_or(true);
                        if eligible && job.cron.matches(minute) {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
