use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_id_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_core_num(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(match bytes.get(start).copied() {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut id_start = start;
    loop {
        let mut p = id_start;
        while p < end && bytes[p] != b'.' {
            if !is_id_char(bytes[p]) {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if pre && id.bytes().all(is_digit) && id.len() > 1 && id.starts_with('0') {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if p == end {
            break;
        }
        id_start = p + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_num(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(match s.as_bytes().get(pos).copied() {
                None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }
        pos += 1;
        if _ == 0 {
            // handled below through the ordered core fields
        }
    }
    // Reparse in the required left-to-right sequence.
    let mut pos = 0;
    let major = parse_core_num(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(match s.as_bytes().get(pos).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;
    let minor = parse_core_num(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(match s.as_bytes().get(pos).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;
    let patch = parse_core_num(s, &mut pos)?;

    if pos < s.len() && s.as_bytes()[pos] != b'-' && s.as_bytes()[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = s[pre_start..].find('+').map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }
    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let c = core(a).cmp(&core(b));
    if c != Ordering::Equal {
        return c;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => cmp_numeric_str(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    pub arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
pub struct Arm {
    pub comparisons: Vec<Comparison>,
    pub anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Comparison {
    pub op: Op,
    pub version: Version,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone)]
struct Tok {
    text: String,
    at: usize,
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<Tok> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    while p < end {
        while p < end && ws(bytes[p]) {
            p += 1;
        }
        if p == end {
            break;
        }
        let a = p;
        while p < end && !ws(bytes[p]) {
            p += 1;
        }
        out.push(Tok {
            text: s[a..p].to_string(),
            at: a,
        });
    }
    out
}

fn version_operand(s: &str, at: usize, text: &str) -> Result<Version, RangeError> {
    parse_version(text).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone)]
struct Partial {
    values: Vec<u64>,
    numeric: usize,
    full: Option<Version>,
}

fn parse_partial(s: &str, at: usize, text: &str) -> Result<Partial, RangeError> {
    let fields: Vec<&str> = text.split('.').collect();
    let force_full = text.contains('-')
        || text.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")));
    if force_full {
        return Ok(Partial {
            values: Vec::new(),
            numeric: 3,
            full: Some(version_operand(s, at, text)?),
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut values = Vec::new();
    let mut wild = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            wild = true;
        } else {
            if wild || !field.bytes().all(is_digit) || (field.len() > 1 && field.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            values.push(n);
        }
    }
    Ok(Partial {
        numeric: values.len(),
        values,
        full: None,
    })
}

fn padded(p: &Partial) -> Version {
    let mut vals = p.values.clone();
    while vals.len() < 3 {
        vals.push(0);
    }
    Version {
        major: vals[0],
        minor: vals[1],
        patch: vals[2],
        pre: p.full.as_ref().map(|v| v.pre.clone()).unwrap_or_default(),
        build: p.full.as_ref().map(|v| v.build.clone()).unwrap_or_default(),
    }
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut n = v.clone();
    n.pre = vec!["0".to_string()];
    n.build.clear();
    let target = match field {
        0 => &mut n.major,
        1 => &mut n.minor,
        _ => &mut n.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field == 0 {
        n.minor = 0;
        n.patch = 0;
    } else if field == 1 {
        n.patch = 0;
    }
    Ok(n)
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.comparisons.push(Comparison { op, version });
}

fn parse_bare(s: &str, tok: &Tok, arm: &mut Arm) -> Result<(), RangeError> {
    let p = parse_partial(s, tok.at, &tok.text)?;
    if let Some(v) = p.full {
        add_cmp(arm, Op::Eq, v, true);
        return Ok(());
    }
    if p.numeric == 0 {
        return Ok(());
    }
    let lo = padded(&p);
    add_cmp(arm, Op::Ge, lo.clone(), true);
    if p.numeric < 3 {
        let field = if p.numeric == 1 { 0 } else { 1 };
        let hi = bump(&lo, field, tok.at)?;
        add_cmp(arm, Op::Lt, hi, false);
    }
    Ok(())
}

fn parse_caret_tilde(s: &str, tok: &Tok, arm: &mut Arm, caret: bool) -> Result<(), RangeError> {
    let operand_at = tok.at + 1;
    let text = &tok.text[1..];
    let p = parse_partial(s, operand_at, text)?;
    if p.numeric == 0 {
        return Err(RangeError::Syntax { at: tok.at });
    }
    let lo = padded(&p);
    add_cmp(arm, Op::Ge, lo.clone(), true);
    let field = if caret {
        if lo.major > 0 || p.numeric == 1 {
            0
        } else if lo.minor > 0 || p.numeric == 2 {
            1
        } else {
            2
        }
    } else if p.numeric == 1 {
        0
    } else {
        1
    };
    let hi = bump(&lo, field, operand_at)?;
    add_cmp(arm, Op::Lt, hi, false);
    Ok(())
}

fn parse_comparator(s: &str, tok: &Tok, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, n) = if tok.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if tok.text.starts_with("<=") {
        (Op::Le, 2)
    } else if tok.text.starts_with('>') {
        (Op::Gt, 1)
    } else if tok.text.starts_with('<') {
        (Op::Lt, 1)
    } else {
        (Op::Eq, 1)
    };
    let operand = &tok.text[n..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: tok.at });
    }
    let v = version_operand(s, tok.at + n, operand)?;
    add_cmp(arm, op, v, true);
    Ok(())
}

fn parse_hyphen(s: &str, ts: &[Tok], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = ts.iter().find(|t| t.text == "-").unwrap();
    if ts.len() != 3 || ts[1].text != "-" {
        return Err(RangeError::Syntax { at: minus.at });
    }
    let lo = parse_partial(s, ts[0].at, &ts[0].text)?;
    let hi = parse_partial(s, ts[2].at, &ts[2].text)?;
    if lo.numeric == 0 {
        return Err(RangeError::Syntax { at: ts[0].at });
    }
    if hi.numeric == 0 {
        return Err(RangeError::Syntax { at: ts[2].at });
    }
    let lower = padded(&lo);
    add_cmp(arm, Op::Ge, lower, true);
    if let Some(v) = hi.full {
        add_cmp(arm, Op::Le, v, true);
    } else {
        let upper = padded(&hi);
        if hi.numeric == 3 {
            add_cmp(arm, Op::Le, upper, true);
        } else {
            let field = if hi.numeric == 1 { 0 } else { 1 };
            add_cmp(arm, Op::Lt, bump(&upper, field, ts[2].at)?, false);
        }
    }
    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let ts = tokens(s, start, end);
    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };
    if ts.iter().any(|t| t.text == "-") {
        parse_hyphen(s, &ts, &mut arm)?;
        return Ok(arm);
    }
    for tok in &ts {
        let first = tok.text.as_bytes().first().copied();
        match first {
            Some(b'<' | b'>' | b'=') => parse_comparator(s, tok, &mut arm)?,
            Some(b'^') => parse_caret_tilde(s, tok, &mut arm, true)?,
            Some(b'~') => parse_caret_tilde(s, tok, &mut arm, false)?,
            Some(b'0'..=b'9' | b'*' | b'x' | b'X') => parse_bare(s, tok, &mut arm)?,
            _ => return Err(RangeError::Syntax { at: tok.at }),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut spans = Vec::new();
    let mut p = 0;
    while p < bytes.len() && ws(bytes[p]) {
        p += 1;
    }
    if p == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arm_start = p;
    loop {
        while p < bytes.len() && !ws(bytes[p]) {
            p += 1;
        }
        let token_end = p;
        let is_or = &s[arm_start..token_end] == "||";
        if is_or {
            let sep = arm_start;
            let arm_end = sep;
            if tokens(s, spans.last().map(|x: &(usize, usize)| x.1).unwrap_or(0), arm_end).is_empty() {
                return Err(RangeError::Empty { at: sep });
            }
            spans.push((spans.last().map(|x: &(usize, usize)| x.1).unwrap_or(0), arm_end));
            p = token_end;
            while p < bytes.len() && ws(bytes[p]) {
                p += 1;
            }
            if p == bytes.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
            arm_start = p;
            continue;
        }
        if p == bytes.len() {
            break;
        }
        while p < bytes.len() && ws(bytes[p]) {
            p += 1;
        }
        if p == bytes.len() {
            break;
        }
        arm_start = p;
    }
    let first_non_ws = s.find(|c: char| !ws(c as u8)).unwrap_or(0);
    let mut arms = Vec::new();
    let mut segment_start = first_non_ws;
    let mut i = first_non_ws;
    while i < s.len() {
        if s.as_bytes()[i..].starts_with(b"||") {
            let before_ws = i > 0 && ws(s.as_bytes()[i - 1]);
            let after = i + 2;
            let after_ws = after < s.len() && ws(s.as_bytes()[after]);
            if before_ws && after_ws {
                let mut end = i;
                while end > segment_start && ws(s.as_bytes()[end - 1]) {
                    end -= 1;
                }
                arms.push(parse_arm(s, segment_start, end)?);
                i = after;
                while i < s.len() && ws(s.as_bytes()[i]) {
                    i += 1;
                }
                if i == s.len() {
                    return Err(RangeError::Empty { at: s.len() });
                }
                segment_start = i;
                continue;
            }
        }
        i += 1;
    }
    let mut end = s.len();
    while end > segment_start && ws(s.as_bytes()[end - 1]) {
        end -= 1;
    }
    arms.push(parse_arm(s, segment_start, end)?);
    Ok(Range { arms })
}

fn cmp_one(v: &Version, c: &Comparison) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

fn matches_arm(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.comparisons.iter().all(|c| cmp_one(v, c)) {
        return false;
    }
    v.pre.is_empty()
        || include
        || arm.anchors.iter().any(|&t| t == core(v))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| matches_arm(v, arm, include_prerelease))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.comparisons {
        match c.op {
            Op::Eq => {
                let l = (c.version.clone(), true);
                let u = (c.version.clone(), true);
                if lower.as_ref().map_or(true, |x| {
                    compare_versions(&l.0, &x.0) == Ordering::Greater
                        || (compare_versions(&l.0, &x.0) == Ordering::Equal && !x.1)
                }) {
                    lower = Some(l);
                }
                if upper.as_ref().map_or(true, |x| {
                    compare_versions(&u.0, &x.0) == Ordering::Less
                        || (compare_versions(&u.0, &x.0) == Ordering::Equal && !x.1)
                }) {
                    upper = Some(u);
                }
            }
            Op::Gt | Op::Ge => {
                let inclusive = c.op == Op::Ge;
                if lower.as_ref().map_or(true, |x| {
                    compare_versions(&c.version, &x.0) == Ordering::Greater
                        || (compare_versions(&c.version, &x.0) == Ordering::Equal && !inclusive && x.1)
                }) {
                    lower = Some((c.version.clone(), inclusive));
                }
            }
            Op::Lt | Op::Le => {
                let inclusive = c.op == Op::Le;
                if upper.as_ref().map_or(true, |x| {
                    compare_versions(&c.version, &x.0) == Ordering::Less
                        || (compare_versions(&c.version, &x.0) == Ordering::Equal && !inclusive && x.1)
                }) {
                    upper = Some((c.version.clone(), inclusive));
                }
            }
        }
    }
    (lower, upper)
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn arm_nonempty(arm: &Arm) -> bool {
    let (lo, hi) = bounds(arm);
    if let (Some((l, li)), Some((h, hi))) = (&lo, &hi) {
        let c = compare_versions(l, h);
        if c == Ordering::Greater || (c == Ordering::Equal && (!li || !hi)) {
            return false;
        }
        if c == Ordering::Equal {
            return l.pre.is_empty() || arm.anchors.contains(&core(l));
        }
    }
    true
}

pub fn canonical(r: &Range) -> String {
    let mut outputs = Vec::new();
    for arm in &r.arms {
        if !arm_nonempty(arm) {
            continue;
        }
        let (lower, upper) = bounds(arm);
        let lo = lower.map_or("*".to_string(), |(mut v, inclusive)| {
            v.build.clear();
            format!("{}{}", if inclusive { ">=" } else { ">" }, display_version(&v))
        });
        let hi = upper.map_or("*".to_string(), |(mut v, inclusive)| {
            v.build.clear();
            format!("{}{}", if inclusive { "<=" } else { "<" }, display_version(&v))
        });
        let mut anchors = arm.anchors.clone();
        anchors.sort_unstable();
        anchors.dedup();
        let a = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(x, y, z)| format!("{x}.{y}.{z}"))
                .collect::<Vec<_>>()
                .join(",")
        };
        outputs.push(format!("[{lo};{hi};pre={a}]"));
    }
    outputs.sort();
    outputs.dedup();
    if outputs.is_empty() {
        "EMPTY".to_string()
    } else {
        outputs.join(" || ")
    }
}

fn stable_successor(v: &Version) -> Option<Version> {
    if v.patch < u64::MAX {
        return Some(Version {
            major: v.major,
            minor: v.minor,
            patch: v.patch + 1,
            pre: vec!["0".into()],
            build: Vec::new(),
        });
    }
    if v.minor < u64::MAX {
        return Some(Version {
            major: v.major,
            minor: v.minor + 1,
            patch: 0,
            pre: vec!["0".into()],
            build: Vec::new(),
        });
    }
    if v.major < u64::MAX {
        return Some(Version {
            major: v.major + 1,
            minor: 0,
            patch: 0,
            pre: vec!["0".into()],
            build: Vec::new(),
        });
    }
    None
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let (al, au) = bounds(aa);
            let (bl, bu) = bounds(bb);
            let lower = match (al, bl) {
                (Some(x), Some(y)) => {
                    let c = compare_versions(&x.0, &y.0);
                    Some(if c == Ordering::Greater || (c == Ordering::Equal && !x.1 && y.1) { x } else { y })
                }
                (Some(x), None) | (None, Some(x)) => Some(x),
                _ => None,
            };
            let upper = match (au, bu) {
                (Some(x), Some(y)) => {
                    let c = compare_versions(&x.0, &y.0);
                    Some(if c == Ordering::Less || (c == Ordering::Equal && !x.1 && y.1) { x } else { y })
                }
                (Some(x), None) | (None, Some(x)) => Some(x),
                _ => None,
            };
            if let (Some((l, li)), Some((h, hi))) = (&lower, &upper) {
                let c = compare_versions(l, h);
                if c == Ordering::Greater || (c == Ordering::Equal && (!li || !hi)) {
                    continue;
                }
            }
            if include_prerelease {
                let mut candidate = lower.as_ref().map(|x| x.0.clone()).unwrap_or(Version {
                    major: 0, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new()
                });
                if lower.as_ref().is_some_and(|x| !x.1 && candidate.pre.is_empty()) {
                    if let Some(next) = stable_successor(&candidate) {
                        candidate = next;
                    } else {
                        continue;
                    }
                }
                if upper.as_ref().map_or(true, |(u, incl)| {
                    let c = compare_versions(&candidate, u);
                    c == Ordering::Less || (c == Ordering::Equal && *incl)
                }) {
                    return true;
                }
            } else {
                for &(m, n, p) in &aa.anchors {
                    if !bb.anchors.contains(&(m, n, p)) {
                        continue;
                    }
                    let candidate = Version {
                        major: m, minor: n, patch: p, pre: vec!["0".into()], build: Vec::new()
                    };
                    if aa.comparisons.iter().all(|c| cmp_one(&candidate, c))
                        && bb.comparisons.iter().all(|c| cmp_one(&candidate, c))
                    {
                        return true;
                    }
                }
                let mut stable = lower.as_ref().map(|x| x.0.clone()).unwrap_or(Version {
                    major: 0, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new()
                });
                if !stable.pre.is_empty() || lower.as_ref().is_some_and(|x| !x.1) {
                    if let Some(next) = stable_successor(&stable) {
                        stable = next;
                    } else {
                        continue;
                    }
                }
                if stable.pre.is_empty()
                    && aa.comparisons.iter().all(|c| cmp_one(&stable, c))
                    && bb.comparisons.iter().all(|c| cmp_one(&stable, c))
                    && upper.as_ref().map_or(true, |(u, incl)| {
                        let c = compare_versions(&stable, u);
                        c == Ordering::Less || (c == Ordering::Equal && *incl)
                    })
                {
                    return true;
                }
            }
        }
    }
    false
}
