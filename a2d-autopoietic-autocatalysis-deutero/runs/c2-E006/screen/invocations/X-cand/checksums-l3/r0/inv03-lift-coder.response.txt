#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc_state: u32,
    adler_a: u32,
    adler_b: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

fn update_crc32(mut state: u32, data: &[u8]) -> u32 {
    for &byte in data {
        state ^= u32::from(byte);
        for _ in 0..8 {
            let low_bit = state & 1;
            state >>= 1;
            if low_bit != 0 {
                state ^= 0xedb88320;
            }
        }
    }
    state
}

fn update_adler(mut a: u32, mut b: u32, data: &[u8]) -> (u32, u32) {
    const MOD: u32 = 65_521;
    for &byte in data {
        a = (a + u32::from(byte)) % MOD;
        b = (b + a) % MOD;
    }
    (a, b)
}

pub fn crc32(data: &[u8]) -> u32 {
    update_crc32(0xffff_ffff, data) ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let (a, b) = update_adler(1, 0, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn update_fletcher16(mut a: u32, mut b: u32, data: &[u8]) -> (u32, u32) {
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    (a, b)
}

fn update_fletcher32_words(mut a: u32, mut b: u32, data: &[u8]) -> (u32, u32) {
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = (u32::from(pair[0]) << 8) | u32::from(pair[1]);
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    let remainder = chunks.remainder();
    if let Some(&byte) = remainder.first() {
        let word = u32::from(byte) << 8;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    (a, b)
}

fn update_crc16(mut state: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                state ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let top_bit = state & 0x8000;
                    state = state.wrapping_shl(1);
                    if top_bit != 0 {
                        state ^= 0x1021;
                    }
                }
            }
            state
        }
        Crc16::X25 => {
            for &byte in data {
                state ^= u16::from(byte);
                for _ in 0..8 {
                    let low_bit = state & 1;
                    state >>= 1;
                    if low_bit != 0 {
                        state ^= 0x8408;
                    }
                }
            }
            state
        }
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (a, b) = update_fletcher16(0, 0, data);
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (a, b) = update_fletcher32_words(0, 0, data);
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                let word = (u32::from(high) << 8) | u32::from(byte);
                a = (a + word) % 65_535;
                b = (b + a) % 65_535;
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        let word = u32::from(high) << 8;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }

    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let initial = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let state = update_crc16(initial, data, variant);
    match variant {
        Crc16::X25 => state ^ 0xffff,
        Crc16::Xmodem | Crc16::False => state,
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc32_state = 0xffff_ffff;
    let mut adler_a = 1u32;
    let mut adler_b = 0u32;
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_pending: Option<u8> = None;
    let crc16_initial = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let mut crc16_state = crc16_initial;

    for span in spans {
        let selected = &data[span.start..span.end];
        crc32_state = update_crc32(crc32_state, selected);
        (adler_a, adler_b) = update_adler(adler_a, adler_b, selected);
        (f16_a, f16_b) = update_fletcher16(f16_a, f16_b, selected);
        crc16_state = update_crc16(crc16_state, selected, variant);

        for &byte in selected {
            if let Some(high) = f32_pending.take() {
                let word = (u32::from(high) << 8) | u32::from(byte);
                f32_a = (f32_a + word) % 65_535;
                f32_b = (f32_b + f32_a) % 65_535;
            } else {
                f32_pending = Some(byte);
            }
        }
    }

    if let Some(high) = f32_pending {
        let word = u32::from(high) << 8;
        f32_a = (f32_a + word) % 65_535;
        f32_b = (f32_b + f32_a) % 65_535;
    }

    let crc16_result = match variant {
        Crc16::X25 => crc16_state ^ 0xffff,
        Crc16::Xmodem | Crc16::False => crc16_state,
    };

    Ok(Multi {
        base: Hashes {
            crc32: crc32_state ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_result,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_state: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_state: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_state ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc_state = update_crc32(self.crc_state, data);
        (self.adler_a, self.adler_b) =
            update_adler(self.adler_a, self.adler_b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksums = self.checksum();
        format!("c{:08x}:a{:08x}", checksums.crc32, checksums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        let mut adler = 0u32;
        for offset in 1..9 {
            let digit = hex_digit(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            crc = (crc << 4) | u32::from(digit);
        }
        for offset in 11..19 {
            let digit = hex_digit(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            adler = (adler << 4) | u32::from(digit);
        }

        Self::from_seed(crc, adler)
    }
}

fn hex_digit(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}
