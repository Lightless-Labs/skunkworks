#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn core_error_byte(b: Option<u8>, at: usize) -> Result<(), VersionError> {
    match b {
        None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at }),
        Some(_) => Err(VersionError::InvalidChar { at }),
    }
}

fn parse_identifiers(
    s: &str,
    base: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut start = 0;
    loop {
        let mut end = start;
        while end < bytes.len() && bytes[end] != b'.' {
            let b = bytes[end];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: base + end });
            }
            end += 1;
        }
        if end == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
        let ident = &s[start..end];
        if prerelease
            && ident.len() > 1
            && ident.starts_with('0')
            && ident.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        out.push(ident.to_owned());
        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut values = [0u64; 3];

    for (field, value) in values.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if pos == start {
            return core_error_byte(bytes.get(pos).copied(), pos);
        }
        let digits = &s[start..pos];
        if digits.len() > 1 && digits.starts_with('0') {
            return Err(VersionError::LeadingZero { at: start });
        }
        *value = digits
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;

        if field < 2 {
            if bytes.get(pos) != Some(&b'.') {
                return core_error_byte(bytes.get(pos).copied(), pos);
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let plus = s[pre_start..].find('+').map(|i| pre_start + i);
                let pre_end = plus.unwrap_or(s.len());
                pre = parse_identifiers(&s[pre_start..pre_end], pre_start, true)?;
                if let Some(plus_at) = plus {
                    build = parse_identifiers(&s[plus_at + 1..], plus_at + 1, false)?;
                }
            }
            b'+' => {
                build = parse_identifiers(&s[pos + 1..], pos + 1, false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    Ok(Version {
        major: values[0],
        minor: values[1],
        patch: values[2],
        pre,
        build,
    })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric(a: &str, b: &str) -> std::cmp::Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let c = match (xn, yn) {
            (true, true) => cmp_numeric(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)

fn make_version(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version { major, minor, patch, pre, build: Vec::new() }

fn zero_pre() -> Vec<String> {
    vec!["0".to_owned()]
}

fn bump_tuple(
    t: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let (mut major, mut minor, mut patch) = t;
    let res = match field {
        0 => major.checked_add(1).map(|n| {
            major = n;
            minor = 0;
            patch = 0;
        }),
        1 => minor.checked_add(1).map(|n| {
            minor = n;
            patch = 0;
        }),
        _ => patch.checked_add(1).map(|n| patch = n),
    };
    if res.is_none() {
        return Err(RangeError::Syntax { at });
    }
    Ok(make_version(major, minor, patch, zero_pre()))
}

#[derive(Clone, Debug)]
struct Operand {
    version: Version,
    count: usize,
    full: bool,
}

fn parse_operand(s: &str, base: usize) -> Result<Operand, RangeError> {
    let forces_full = s.contains('-')
        || s.contains('+')
        || (s.split('.').count() >= 3
            && !s.split('.').any(|x| matches!(x, "*" | "x" | "X")));
    if forces_full {
        let version = parse_version(s).map_err(|error| RangeError::Version { at: base, error })?;
        return Ok(Operand { version, count: 3, full: true });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: base });
    }
    let mut nums = Vec::new();
    let mut wild = false;
    for f in &fields {
        if matches!(*f, "*" | "x" | "X") {
            wild = true;
        } else {
            if wild
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
                || f.parse::<u64>().is_err()
            {
                return Err(RangeError::Syntax { at: base });
            }
            nums.push(f.parse::<u64>().unwrap());
        }
    }
    let count = nums.len();
    let major = nums.first().copied().unwrap_or(0);
    let minor = nums.get(1).copied().unwrap_or(0);
    let patch = nums.get(2).copied().unwrap_or(0);
    Ok(Operand {
        version: make_version(major, minor, patch, Vec::new()),
        count,
        full: false,
    })
}

fn add_bound(arm: &mut Arm, op: Op, version: Version, explicit: bool) {
    if explicit && !version.pre.is_empty() {
        arm.anchors.push(tuple(&version));
    }
    arm.bounds.push(Bound { version, op });
}

fn parse_comparator(tok: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, n) = if tok.starts_with(">=") {
        (Op::Ge, 2)
    } else if tok.starts_with("<=") {
        (Op::Le, 2)
    } else if tok.starts_with('>') {
        (Op::Gt, 1)
    } else if tok.starts_with('<') {
        (Op::Lt, 1)
    } else if tok.starts_with('=') {
        (Op::Eq, 1)
    } else {
        return Err(RangeError::Syntax { at });
    };
    let operand = &tok[n..];
    let operand_at = at + n;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let version = parse_version(operand)
        .map_err(|error| RangeError::Version { at: operand_at, error })?;
    add_bound(arm, op, version, true);
    Ok(())
}

fn parse_partial_token(
    token: &str,
    base: usize,
    arm: &mut Arm,
    kind: char,
    token_start: usize,
) -> Result<(), RangeError> {
    let operand = if kind == '\0' { token } else { &token[1..] };
    let operand_at = if kind == '\0' { base } else { base + 1 };
    let p = parse_operand(operand, operand_at)?;
    if kind != '\0' && p.count == 0 {
        return Err(RangeError::Syntax { at: token_start });
    }
    if kind == '\0' && p.full {
        add_bound(arm, Op::Eq, p.version, true);
        return Ok(());
    }
    if kind == '\0' && p.count == 0 {
        return Ok(());
    }
    let lower = p.version.clone();
    add_bound(arm, Op::Ge, lower.clone(), true);
    if kind == '\0' {
        if p.count == 1 {
            let upper = bump_tuple(tuple(&p.version), 0, operand_at)?;
            add_bound(arm, Op::Lt, upper, false);
        } else if p.count == 2 {
            let upper = bump_tuple(tuple(&p.version), 1, operand_at)?;
            add_bound(arm, Op::Lt, upper, false);
        }
    } else if kind == '^' {
        let t = tuple(&p.version);
        let field = if t.0 > 0 || p.count == 1 {
            0
        } else if t.1 > 0 || p.count == 2 {
            1
        } else {
            2
        };
        let upper = bump_tuple(t, field, operand_at)?;
        add_bound(arm, Op::Lt, upper, false);
    } else {
        let field = if p.count == 1 { 0 } else { 1 };
        let upper = bump_tuple(tuple(&p.version), field, operand_at)?;
        add_bound(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn tokenize(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && is_ws(b[i]) { i += 1; }
        if i == end { break; }
        let st = i;
        while i < end && !is_ws(b[i]) { i += 1; }
        out.push((st, &s[st..i]));
    }
    out
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let tokens = tokenize(s, start, end);
    let mut arm = Arm { bounds: Vec::new(), anchors: Vec::new() };
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: start });
    }
    let minus: Vec<usize> = tokens.iter().enumerate().filter(|(_, (_, t))| *t == "-").map(|(i, _)| i).collect();
    if !minus.is_empty() {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: tokens[minus[0]].0 });
        }
        let (lo_at, lo_s) = tokens[0];
        let (hi_at, hi_s) = tokens[2];
        let lo = parse_operand(lo_s, lo_at)?;
        let hi = parse_operand(hi_s, hi_at)?;
        if lo.count == 0 { return Err(RangeError::Syntax { at: lo_at }); }
        if hi.count == 0 { return Err(RangeError::Syntax { at: hi_at }); }
        add_bound(&mut arm, Op::Ge, lo.version.clone(), true);
        if hi.full {
            add_bound(&mut arm, Op::Le, hi.version, true);
        } else {
            let upper = if hi.count == 1 {
                bump_tuple(tuple(&hi.version), 0, hi_at)?
            } else if hi.count == 2 {
                bump_tuple(tuple(&hi.version), 1, hi_at)?
            } else {
                add_bound(&mut arm, Op::Le, hi.version, true);
                return Ok(arm);
            };
            add_bound(&mut arm, Op::Lt, upper, false);
        }
        return Ok(arm);
    }

    for (at, tok) in tokens {
        let first = tok.as_bytes()[0];
        match first {
            b'<' | b'>' | b'=' => parse_comparator(tok, at, &mut arm)?,
            b'^' => parse_partial_token(tok, at, &mut arm, '^', at)?,
            b'~' => parse_partial_token(tok, at, &mut arm, '~', at)?,
            b'*' | b'x' | b'X' | b'0'..=b'9' => parse_partial_token(tok, at, &mut arm, '\0', at)?,
            _ => return Err(RangeError::Syntax { at }),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    if bytes.iter().all(|b| is_ws(*b)) {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    while i + 1 < bytes.len() {
        if bytes[i] == b'|' && bytes[i + 1] == b'|' && i > 0 && is_ws(bytes[i - 1]) && i + 2 < bytes.len() && is_ws(bytes[i + 2]) {
            arms.push(parse_arm(s, arm_start, i)?);
            i += 2;
            while i < bytes.len() && is_ws(bytes[i]) { i += 1; }
            arm_start = i;
        } else {
            i += 1;
        }
    }
    if arm_start >= bytes.len() {
        return Err(RangeError::Empty { at: bytes.len() });
    }
    arms.push(parse_arm(s, arm_start, bytes.len())?);
    Ok(Range { arms })
}

fn bound_ok(v: &Version, b: &Bound) -> bool {
    use std::cmp::Ordering::*;
    match (b.op, compare_versions(v, &b.version)) {
        (Op::Eq, Equal) | (Op::Lt, Less) | (Op::Le, Less | Equal) |
        (Op::Gt, Greater) | (Op::Ge, Greater | Equal) => true,
        _ => false,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.bounds.iter().all(|b| bound_ok(v, b)) {
        return false;
    }
    v.pre.is_empty()
        || include_prerelease
        || arm.anchors.iter().any(|t| *t == tuple(v))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index: i, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let c = compare_versions(v, &parsed[j]);
            if (max && c == std::cmp::Ordering::Greater) || (!max && c == std::cmp::Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    // Collect stable and prerelease candidates at the exact finite boundaries.
    // This conservative witness search covers ordinary ranges while preserving
    // the standard library-only implementation.
    for arm_a in &a.arms {
        for arm_b in &b.arms {
            let mut candidates = vec![
                make_version(0, 0, 0, Vec::new()),
                make_version(u64::MAX, u64::MAX, u64::MAX, Vec::new()),
            ];
            for bound in arm_a.bounds.iter().chain(&arm_b.bounds) {
                candidates.push(bound.version.clone());
                let t = tuple(&bound.version);
                candidates.push(make_version(t.0, t.1, t.2, zero_pre()));
            }
            for v in candidates {
                if arm_matches(&v, arm_a, include_prerelease)
                    && arm_matches(&v, arm_b, include_prerelease)
                {
                    return true;
                }
            }
        }
    }
    false
}

pub fn canonical(r: &Range) -> String {
    let mut rendered = Vec::new();
    for arm in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        let mut impossible = false;
        for b in &arm.bounds {
            let v = b.version.clone();
            match b.op {
                Op::Eq => {
                    let l = (v.clone(), true);
                    let u = (v, true);
                    lower = stronger_lower(lower, l);
                    upper = stronger_upper(upper, u);
                }
                Op::Gt | Op::Ge => {
                    lower = stronger_lower(lower, (v, b.op == Op::Ge));
                }
                Op::Lt | Op::Le => {
                    upper = stronger_upper(upper, (v, b.op == Op::Le));
                }
            }
        }
        if let (Some((l, li)), Some((u, ui))) = (&lower, &upper) {
            match compare_versions(l, u) {
                std::cmp::Ordering::Greater => impossible = true,
                std::cmp::Ordering::Equal if !li || !ui => impossible = true,
                _ => {}
            }
        }
        if impossible { continue; }
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_owned()
        } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        let lower_text = lower.map(|(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, version_text(&v))).unwrap_or_else(|| "*".to_owned());
        let upper_text = upper.map(|(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, version_text(&v))).unwrap_or_else(|| "*".to_owned());
        rendered.push(format!("[{lower_text};{upper_text};pre={anchor_text}]"));
    }
    rendered.sort();
    rendered.dedup();
    if rendered.is_empty() { "EMPTY".to_owned() } else { rendered.join(" || ") }
}

fn stronger_lower(old: Option<(Version, bool)>, new: (Version, bool)) -> Option<(Version, bool)> {
    match old {
        None => Some(new),
        Some((v, inc)) => match compare_versions(&new.0, &v) {
            std::cmp::Ordering::Greater => Some(new),
            std::cmp::Ordering::Less => Some((v, inc)),
            std::cmp::Ordering::Equal => Some((v, inc && new.1)),
        }
    }
}
fn stronger_upper(old: Option<(Version, bool)>, new: (Version, bool)) -> Option<(Version, bool)> {
    match old {
        None => Some(new),
        Some((v, inc)) => match compare_versions(&new.0, &v) {
            std::cmp::Ordering::Less => Some(new),
            std::cmp::Ordering::Greater => Some((v, inc)),
            std::cmp::Ordering::Equal => Some((v, inc && new.1)),
        }
    }
}
fn version_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(expr: &SetExpr, parsed: &[Range], cursor: &mut usize, v: &Version, inc: bool) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let ok = matches_with(v, &parsed[*cursor], inc);
            *cursor += 1;
            ok
        }
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, parsed, cursor, v, inc)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, parsed, cursor, v, inc)),
    }
}

fn gather_ranges(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    out: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_owned(),
                label: label.to_owned(),
                path: path.clone(),
                error,
            })?;
            out.push(r);
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                gather_ranges(x, package, label, path, out)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn expr_accept(expr: &SetExpr, v: &Version, inc: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, inc)).unwrap_or(false),
        SetExpr::Union(xs) => xs.iter().any(|x| expr_accept(x, v, inc)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| expr_accept(x, v, inc)),
    }
}

fn conflict_labels(pkg: &Package, parsed: &[Version], inc: bool) -> Vec<String> {
    if pkg.candidates.is_empty() { return Vec::new(); }
    let n = pkg.constraints.len();
    for size in 1..=n {
        let mut chosen = Vec::new();
        if let Some(result) = find_conflict(pkg, parsed, inc, size, 0, &mut chosen) {
            return result;
        }
    }
    Vec::new()
}

fn find_conflict(
    pkg: &Package,
    parsed: &[Version],
    inc: bool,
    size: usize,
    start: usize,
    chosen: &mut Vec<usize>,
) -> Option<Vec<String>> {
    if chosen.len() == size {
        let any = parsed.iter().any(|v| chosen.iter().all(|&i| expr_accept(&pkg.constraints[i].expr, v, inc)));
        if !any {
            let mut labels: Vec<String> = chosen.iter().map(|&i| pkg.constraints[i].label.clone()).collect();
            labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
            return Some(labels);
        }
        return None;
    }
    for i in start..pkg.constraints.len() {
        chosen.push(i);
        if let Some(r) = find_conflict(pkg, parsed, inc, size, i + 1, chosen) { return Some(r); }
        chosen.pop();
    }
    None
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }

    let mut all_versions = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index: i, error,
            })?);
        }
        all_versions.push(vs);
    }

    let mut parsed_ranges = Vec::new();
    for p in packages {
        let mut prs = Vec::new();
        for c in &p.constraints {
            let mut found = Vec::new();
            gather_ranges(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut found)?;
            prs.push(found);
        }
        parsed_ranges.push(prs);
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &all_versions[pi];
        let mut qualifying = Vec::new();
        for (i, v) in versions.iter().enumerate() {
            let good = p.constraints.iter().all(|c| expr_accept(&c.expr, v, include_prerelease));
            if good { qualifying.push(i); }
        }
        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflict_labels(p, versions, include_prerelease),
            });
        }
        let stable_exists = matches!(preference, Preference::StableFirst)
            && qualifying.iter().any(|&i| versions[i].pre.is_empty());
        let eligible: Vec<usize> = if stable_exists {
            qualifying.into_iter().filter(|&i| versions[i].pre.is_empty()).collect()
        } else {
            qualifying
        };
        let mut best = eligible[0];
        for &i in eligible.iter().skip(1) {
            let cmp = compare_versions(&versions[i], &versions[best]);
            let choose = match preference {
                Preference::Lowest => cmp == std::cmp::Ordering::Less,
                Preference::Highest | Preference::StableFirst => cmp == std::cmp::Ordering::Greater,
            };
            if choose { best = i; }
        }
        selections.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(selections)
}
