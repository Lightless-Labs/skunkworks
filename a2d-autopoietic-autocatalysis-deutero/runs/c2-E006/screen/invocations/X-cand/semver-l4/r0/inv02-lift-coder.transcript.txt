            }
            let patch = core(s, &mut i)?;
            let mut pre = Vec::new();
            let mut build = Vec::new();
            if byte(s, i) == Some(b'-') {
                i += 1;
                let pre_start = i;
                while i < s.len() && s.as_bytes()[i] != b'+' {
                    i += 1;
                }
                pre = identifiers(s, pre_start, i, true)?;
            }
            if byte(s, i) == Some(b'+') {
                i += 1;
                build = identifiers(s, i, s.len(), false)?;
                i = s.len();
            }
            if i != s.len() {
                return Err(VersionError::InvalidChar { at: i });
            }
            return Ok(Version { major, minor, patch, pre, build });
        }
    }
    unreachable!()
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> std::cmp::Ordering {
    use std::cmp::Ordering::*;
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Equal,
        (true, false) => return Greater,
        (false, true) => return Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Less,
            (false, true) => Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> std::cmp::Ordering {
    a.major
        .cmp(&b.major)
        .then_with(|| a.minor.cmp(&b.minor))
        .then_with(|| a.patch.cmp(&b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let mut i = 0;
    while i < s.len() {
        while i < s.len() && ws(s.as_bytes()[i]) { i += 1; }
        if i == s.len() { break; }
        let start = i;
        while i < s.len() && !ws(s.as_bytes()[i]) { i += 1; }
        out.push((start, &s[start..i]));
    }
    out
}

fn is_wild(s: &str) -> bool { matches!(s, "*" | "x" | "X") }

#[derive(Clone, Debug)]
enum ParsedOperand {
    Full(Version),
    Partial { fields: [Option<u64>; 3], count: usize, numeric: usize },
}

fn parse_operand(s: &str, at: usize) -> Result<ParsedOperand, RangeError> {
    if s.contains('-') || s.contains('+') ||
        (s.split('.').count() >= 3 && !s.split('.').any(is_wild)) {
        return parse_version(s)
            .map(ParsedOperand::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    let parts: Vec<&str> = s.split('.').collect();
    if parts.is_empty() || parts.len() > 3 || parts.iter().any(|p| p.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = [None; 3];
    let mut wildcard_seen = false;
    let mut numeric = 0;
    for (i, p) in parts.iter().enumerate() {
        if is_wild(p) {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !p.bytes().all(|b| b.is_ascii_digit())
                || (p.len() > 1 && p.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            fields[i] = Some(n);
            numeric += 1;
        }
    }
    Ok(ParsedOperand::Partial { fields, count: parts.len(), numeric })
}

fn lower_of(p: &ParsedOperand) -> Version {
    match p {
        ParsedOperand::Full(v) => v.clone(),
        ParsedOperand::Partial { fields, .. } => Version {
            major: fields[0].unwrap_or(0),
            minor: fields[1].unwrap_or(0),
            patch: fields[2].unwrap_or(0),
            pre: Vec::new(),
            build: Vec::new(),
        },
    }
}

fn upper_for(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut u = Version { major:v.major, minor:v.minor, patch:v.patch, pre:vec!["0".to_string()], build:Vec::new() };
    match field {
        0 => { u.major = u.major.checked_add(1).ok_or(RangeError::Syntax { at })?; u.minor=0;u.patch=0; }
        1 => { u.minor = u.minor.checked_add(1).ok_or(RangeError::Syntax { at })?; u.patch=0; }
        _ => { u.patch = u.patch.checked_add(1).ok_or(RangeError::Syntax { at })?; }
    }
    Ok(u)
}

fn push_partial_bounds(
    comps: &mut Vec<Comparator>, anchors: &mut Vec<Anchor>,
    p: ParsedOperand, at: usize, mode: u8,
) -> Result<(), RangeError> {
    match p {
        ParsedOperand::Full(v) => {
            if !v.pre.is_empty() { anchors.push(Anchor(v.major,v.minor,v.patch)); }
            match mode {
                0 => comps.push(Comparator::Eq(v)),
                1 => { comps.push(Comparator::Ge(v.clone())); comps.push(Comparator::Lt(upper_for(&v, 0, at)?)); }
                _ => comps.push(Comparator::Ge(v)),
            }
        }
        ParsedOperand::Partial { fields, count, numeric } => {
            if mode != 0 && numeric == 0 { return Err(RangeError::Syntax { at }); }
            if numeric == 0 { return Ok(()); }
            let v = lower_of(&ParsedOperand::Partial { fields, count, numeric });
            comps.push(Comparator::Ge(v.clone()));
            let bump = match mode {
                1 => {
                    if v.major > 0 || numeric == 1 { 0 }
                    else if v.minor > 0 || numeric == 2 { 1 }
                    else { 2 }
                }
                2 => if numeric == 1 { 0 } else { 1 },
                _ => if numeric == 1 { 0 } else { 1 },
            };
            comps.push(Comparator::Lt(upper_for(&v, bump, at)?));
        }
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = split_tokens(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut begin = 0;
    let mut i = 0;
    while i <= tokens.len() {
        if i == tokens.len() || tokens[i].1 != "||" {
            i += 1;
            continue;
        }
        if i == begin { return Err(RangeError::Empty { at: tokens[i].0 }); }
        arms.push(parse_arm(&tokens[begin..i])?);
        begin = i + 1;
        i += 1;
    }
    if begin == tokens.len() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&tokens[begin..])?);
    Ok(Range { arms })
}

fn parse_arm(tokens: &[(usize, &str)]) -> Result<Vec<Comparator>, RangeError> {
    let mut comps = Vec::new();
    let mut anchors = Vec::new();
    if tokens.iter().any(|(_, t)| *t == "-") {
        let minus = tokens.iter().find(|(_, t)| *t == "-").unwrap().0;
        if tokens.len() != 3 || tokens[1].1 != "-" { return Err(RangeError::Syntax { at: minus }); }
        let (la, lt) = tokens[0];
        let (ua, ut) = tokens[2];
        let lp = parse_operand(lt, la)?;
        let up = parse_operand(ut, ua)?;
        if matches!(&lp, ParsedOperand::Partial{numeric:0,..}) { return Err(RangeError::Syntax{at:la}); }
        if matches!(&up, ParsedOperand::Partial{numeric:0,..}) { return Err(RangeError::Syntax{at:ua}); }
        push_partial_bounds(&mut comps,&mut anchors,lp,la,3)?;
        match up {
            ParsedOperand::Full(v) => {
                if !v.pre.is_empty() { anchors.push(Anchor(v.major,v.minor,v.patch)); }
                comps.push(Comparator::Le(v));
            }
            ParsedOperand::Partial{fields,count,numeric} => {
                let v=lower_of(&ParsedOperand::Partial{fields,count,numeric});
                let bump=if numeric==1 {0} else {1};
                comps.push(Comparator::Lt(upper_for(&v,bump,ua)?));
            }
        }
        return Ok(comps);
    }

    for &(at, token) in tokens {
        let (op, operand, offset) = if let Some(rest)=token.strip_prefix(">=") { (Some(4),rest,at+2) }
        else if let Some(rest)=token.strip_prefix("<=") { (Some(2),rest,at+2) }
        else if let Some(rest)=token.strip_prefix(">") { (Some(5),rest,at+1) }
        else if let Some(rest)=token.strip_prefix("<") { (Some(1),rest,at+1) }
        else if let Some(rest)=token.strip_prefix("=") { (Some(0),rest,at+1) }
        else { (None,token,at) };

        if let Some(op) = op {
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = parse_version(operand)
                .map_err(|error| RangeError::Version { at: offset, error })?;
            if !v.pre.is_empty() { anchors.push(Anchor(v.major,v.minor,v.patch)); }
            comps.push(match op {
                0 => Comparator::Eq(v), 1 => Comparator::Lt(v), 2 => Comparator::Le(v),
                4 => Comparator::Ge(v), _ => Comparator::Gt(v),
            });
            continue;
        }

        let (mode, opnd, op_at) = match token.strip_prefix('^') {
            Some(x) => (1, x, at+1),
            None => match token.strip_prefix('~') {
                Some(x) => (2,x,at+1),
                None => (0,token,at),
            }
        };
        if opnd.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if !(opnd.as_bytes()[0].is_ascii_digit() || is_wild(opnd)) {
            return Err(RangeError::Syntax { at });
        }
        let p = parse_operand(opnd, op_at)?;
        push_partial_bounds(&mut comps, &mut anchors, p, op_at, mode)?;
    }
    Ok(comps)
}

fn admitted(v: &Version, anchors: &[Anchor]) -> bool {
    v.pre.is_empty() || anchors.iter().any(|a| a.0==v.major && a.1==v.minor && a.2==v.patch)
}

fn eval_arm(v: &Version, comps: &[Comparator]) -> bool {
    comps.iter().all(|c| {
        let o = match c {
            Comparator::Eq(x) => compare_versions(v,x)==std::cmp::Ordering::Equal,
            Comparator::Lt(x) => compare_versions(v,x)==std::cmp::Ordering::Less,
            Comparator::Le(x) => compare_versions(v,x)!=std::cmp::Ordering::Greater,
            Comparator::Gt(x) => compare_versions(v,x)==std::cmp::Ordering::Greater,
            Comparator::Ge(x) => compare_versions(v,x)!=std::cmp::Ordering::Less,
        };
        o
    })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| {
        eval_arm(v, arm) && (include_prerelease || v.pre.is_empty())
    })
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v,r,false) }

fn extrema(candidates:&[String], r:&Range, max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in candidates.iter().enumerate(){
        parsed.push(parse_version(s).map_err(|error|CandidateError{index:i,error})?);
    }
    let mut best:Option<usize>=None;
    for i in 0..parsed.len(){
        if !matches(&parsed[i],r){continue}
        if let Some(j)=best {
            let o=compare_versions(&parsed[i],&parsed[j]);
            if (max && o==std::cmp::Ordering::Greater)||(!max&&o==std::cmp::Ordering::Less){best=Some(i)}
        }else{best=Some(i)}
    }
    Ok(best.map(|i|candidates[i].clone()))
}

)->Result<Vec<Selection>,SelectError>{

            return Err(SelectError::DuplicatePackage{name:packages[i].name.clone()});
        }
    }
    for p in packages {
        for i in 0..p.constraints.len(){
            if p.constraints[..i].iter().any(|c|c.label==p.constraints[i].label){
                return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:p.constraints[i].label.clone()});
            }
        }
    }
    let mut parsed=Vec::new();
    for p in packages{
        let mut v=Vec::new();
        for (i,s) in p.candidates.iter().enumerate(){
            v.push(parse_version(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);
        }
        parsed.push(v);
    }
    fn validate_expr(expr:&SetExpr,path:&mut Vec<usize>,p:&Package,c:&Constraint)->Result<(),SelectError>{
        match expr{
            SetExpr::Range(s)=>{parse_range(s).map_err(|error|SelectError::InvalidRange{package:p.name.clone(),label:c.label.clone(),path:path.clone(),error})?;}
            SetExpr::Union(xs)|SetExpr::Intersection(xs)=>{
                for (i,x) in xs.iter().enumerate(){path.push(i);validate_expr(x,path,p,c)?;path.pop();}
            }
        }
        Ok(())
    }
    for p in packages{for c in &p.constraints{validate_expr(&c.expr,&mut Vec::new(),p,c)?;}}
    fn accepts(e:&SetExpr,v:&Version,include:bool)->bool{
        match e{
            SetExpr::Range(s)=>parse_range(s).map(|r|matches_with(v,&r,include)).unwrap_or(false),
            SetExpr::Union(xs)=>xs.iter().any(|x|accepts(x,v,include)),
            SetExpr::Intersection(xs)=>xs.iter().all(|x|accepts(x,v,include)),
        }
    }
    let mut out=Vec::new();
    for (pi,p) in packages.iter().enumerate(){
        let qualifies:Vec<usize>=(0..parsed[pi].len()).filter(|&i|p.constraints.iter().all(|c|accepts(&c.expr,&parsed[pi][i],include_prerelease))).collect();

            return Err(SelectError::NoSolution{package:p.name.clone(),labels});

            qualifies.iter().copied().filter(|&i|parsed[pi][i].pre.is_empty()).collect()
        }else{qualifies};
        let mut best=pool[0];

        }
        out.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }
    Ok(out)
}
