#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();

    if let Some(at) = bytes.iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = bytes.iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = bytes
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = bytes[0] == b'/';
    let directory_intent = bytes.last() == Some(&b'/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..");

    let mut components: Vec<&str> = Vec::new();
    let mut offset = 0usize;

    for component in path.split('/') {
        if !component.is_empty() {
            match component {
                "." => {}
                ".." => {
                    if components.last().is_some_and(|last| *last != "..") {
                        components.pop();
                    } else if absolute {
                        return Err(PathError::Escape { at: offset });
                    } else {
                        components.push(component);
                    }
                }
                _ => components.push(component),
            }
        }
        offset += component.len() + 1;
    }

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    if !components.is_empty() {
        if absolute {
            result.push_str(&components.join("/"));
        } else {
            result = components.join("/");
        }

        if directory_intent {
            result.push('/');
        }
    } else if !absolute {
        result.push('.');
    }

    Ok(result)
}
