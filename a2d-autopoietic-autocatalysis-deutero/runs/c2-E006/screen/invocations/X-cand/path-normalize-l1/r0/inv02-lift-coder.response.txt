#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.bytes().position(|byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .bytes()
        .position(|byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..");

    let mut components: Vec<&str> = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        if !component.is_empty() {
            match component {
                "." => {}
                ".." => {
                    if components.last().is_some_and(|previous| *previous != "..") {
                        components.pop();
                    } else if absolute {
                        return Err(PathError::Escape { at: offset });
                    } else {
                        components.push(component);
                    }
                }
                _ => components.push(component),
            }
        }
        offset += component.len() + 1;
    }

    if components.is_empty() {
        return Ok(if absolute { "/".into() } else { ".".into() });
    }

    let mut result = String::new();
    if absolute {
        result.push('/');
    }
    result.push_str(&components.join("/"));

    if directory_intent {
        result.push('/');
    }

    Ok(result)
}
