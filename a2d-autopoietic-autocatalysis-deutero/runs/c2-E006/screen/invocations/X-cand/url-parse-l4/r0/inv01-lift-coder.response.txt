fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let src = input.as_bytes();
    let mut decoded = Vec::with_capacity(src.len());
    let mut origins = Vec::with_capacity(src.len());
    let mut i = 0;

    while i < src.len() {
        if src[i] == b'%' {
            if i + 2 >= src.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(src[i + 1]);
            let lo = hex_value(src[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(src[i]);
            origins.push(i);
            i += 1;
        }
    }

    match String::from_utf8(decoded) {
        Ok(s) => Ok(s),
        Err(err) => {
            let at = err.utf8_error().valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(at).copied().unwrap_or(input.len()),
            })
        }
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &b in text.as_bytes() {
        let safe = is_unreserved(b)
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn scan_url_bytes(input: &str) -> Result<(), ParseError> {
    for (at, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn validate_url_component(
    component: Component,
    text: &str,
    offset: usize,
) -> Result<(), ParseError> {
    match decode_component(component, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: offset + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: offset + at }),
    }
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: port_start + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for b in port.bytes() {
        value = match value.checked_mul(10).and_then(|n| n.checked_add((b - b'0') as u32)) {
            Some(n) if n <= 65535 => n,
            _ => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_url_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let after_marker = colon + 3;
    let tail = &input[after_marker..];
    let authority_len = tail
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(tail.len());
    let authority_start = after_marker;
    let authority_end = authority_start + authority_len;
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| if b == b'@' { Some(i) } else { None })
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_string()), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_abs = authority_start + hostport_start;

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let open_at = hostport_abs;
        let close_rel = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: open_at }),
        };
        let interior = &hostport[1..close_rel];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }
        let suffix = &hostport[close_rel + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_abs + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_abs + close_rel + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let host_colon = hostport.bytes().position(|b| b == b':');
        let (host_text, port) = match host_colon {
            Some(i) => {
                let port = parse_port(&hostport[i + 1..], hostport_abs + i + 1)?;
                (&hostport[..i], Some(port))
            }
            None => (hostport, None),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_abs });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hostport_abs + i });
            }
        }
        (host_text.to_ascii_lowercase(), port, false)
    };

    let fragment_rel = input[authority_end..]
        .bytes()
        .position(|b| b == b'#')
        .map(|i| authority_end + i);
    let before_fragment_end = fragment_rel.unwrap_or(input.len());
    let query_rel = input[authority_end..before_fragment_end]
        .bytes()
        .position(|b| b == b'?')
        .map(|i| authority_end + i);

    let path_end = query_rel.unwrap_or(before_fragment_end);
    let raw_path = &input[authority_end..path_end];
    let path = if raw_path.is_empty() {
        "/".to_string()
    } else {
        raw_path.to_string()
    };

    let query = query_rel.map(|q| input[q + 1..before_fragment_end].to_string());
    let fragment = fragment_rel.map(|f| input[f + 1..].to_string());

    if let Some(ref u) = userinfo {
        let u_start = authority_start;
        validate_url_component(Component::UserInfo, u, u_start)?;
    }
    if !raw_path.is_empty() {
        validate_url_component(Component::Path, raw_path, authority_end)?;
    }
    if let Some(q) = query_rel {
        validate_url_component(
            Component::Query,
            query.as_deref().unwrap_or(""),
            q + 1,
        )?;
    }
    if let Some(f) = fragment_rel {
        validate_url_component(
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            f + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn path_error_from_codec(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(path_error_from_codec)?;

    let mut stack: Vec<String> = Vec::new();
    let mut terminal_navigation = false;

    for segment in path[1..].split('/') {
        let decoded =
            decode_component(Component::Path, segment).map_err(path_error_from_codec)?;
        if decoded == "." {
            terminal_navigation = true;
        } else if decoded == ".." {
            terminal_navigation = true;
            stack.pop();
        } else {
            terminal_navigation = false;
            stack.push(segment.to_string());
        }
    }

    if terminal_navigation {
        stack.push(String::new());
    }
    if stack.is_empty() {
        Ok("/".to_string())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn global_reference_scan(input: &str) -> Result<(), ParseError> {
    scan_url_bytes(input)
}

fn split_reference_components(input: &str) -> (&str, Option<&str>, Option<&str>) {
    let frag_pos = input.find('#');
    let before_fragment = frag_pos.map(|i| &input[..i]).unwrap_or(input);
    let fragment = frag_pos.map(|i| &input[i + 1..]);
    let query_pos = before_fragment.find('?');
    let path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| &before_fragment[i + 1..]);
    (path, query, fragment)
}

fn normalize_resolved(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    global_reference_scan(reference).map_err(ResolveError::Reference)?;

    if reference
        .find(|c| matches!(c, '/' | '?' | '#'))
        .map(|stop| reference[..stop].contains(':'))
        .unwrap_or_else(|| reference.contains(':'))
    {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_resolved(url);
    }

    if reference.starts_with("//") {
        let raw = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&raw).map_err(ResolveError::Reference)?;
        return normalize_resolved(url);
    }

    let (reference_path, reference_query, reference_fragment) =
        split_reference_components(reference);
    let selected_path;
    let selected_query;

    if reference_path.is_empty() {
        selected_path = base_path.clone();
        selected_query = if reference_query.is_some() {
            reference_query.map(str::to_string)
        } else {
            base_url.query.clone()
        };
    } else {
        selected_query = reference_query.map(str::to_string);
        if reference_path.starts_with('/') {
            selected_path = reference_path.to_string();
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            selected_path = format!("{}{}", &base_path[..slash], reference_path);
        }
    }

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        raw.push_str(userinfo);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        raw.push(':');
        raw.push_str(&port.to_string());
    }
    raw.push_str(&selected_path);
    if let Some(query) = &selected_query {
        raw.push('?');
        raw.push_str(query);
    }
    if let Some(fragment) = reference_fragment {
        raw.push('#');
        raw.push_str(fragment);
    }

    let url = parse_url(&raw).map_err(ResolveError::Reference)?;
    normalize_resolved(url)
}

fn query_decode(input: &str, offset: usize) -> Result<String, QueryError> {
    let plus_replaced = input.replace('+', " ");
    match decode_component(Component::Query, &plus_replaced) {
        Ok(s) => Ok(s),
        Err(CodecError::BadEscape { at }) => Err(QueryError::BadEscape { at: offset + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(QueryError::InvalidUtf8 { at: offset + at }),
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut start = 0;

    for field in input.split('&') {
        let field_start = start;
        start += field.len() + 1;
        if field.is_empty() {
            continue;
        }

        let equal = field.find('=');
        let (raw_key, raw_value, value_offset) = match equal {
            Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
            None => ("", "", field_start),
        };
        let key = query_decode(raw_key, field_start)?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value = query_decode(raw_value, value_offset)?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else if b == b' ' {
            out.push('+');
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, dest)) = merged.iter_mut().find(|(existing, _)| existing == key) {
            dest.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(&key), form_encode(&value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(raw_userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, raw_userinfo).map_err(|e| {
            CanonicalError::Parse(match e {
                CodecError::BadEscape { at } => ParseError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at },
            })
        })?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host)
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let default = (url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443);
        if !default {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        let decoded =
            decode_component(Component::Path, segment).map_err(|e| match e {
                CodecError::BadEscape { at } => {
                    CanonicalError::Path(PathError::BadEscape { at })
                }
                CodecError::InvalidUtf8 { at } => {
                    CanonicalError::Path(PathError::InvalidUtf8 { at })
                }
            })?;
        canonical_segments.push(encode_unreserved_only(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded =
            decode_component(Component::Fragment, fragment).map_err(|e| match e {
                CodecError::BadEscape { at } => {
                    CanonicalError::Parse(ParseError::BadEscape { at })
                }
                CodecError::InvalidUtf8 { at } => {
                    CanonicalError::Parse(ParseError::InvalidUtf8 { at })
                }
            })?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(out)
}

fn encode_unreserved_only(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}
