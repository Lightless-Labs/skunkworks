#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|c| !c.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|c| *c == "." || *c == "..")
}

fn finish_absolute(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return "/".to_string();
    }
    let mut out = format!("/{}", stack.join("/"));
    if intent {
        out.push('/');
    }
    out
}

fn finish_relative(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return ".".to_string();
    }
    let mut out = stack.join("/");
    if intent {
        out.push('/');
    }
    out
}

fn normalize_impl(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    for component in path.split('/') {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|c| c != "..") {
                stack.pop();
            } else if absolute {
                let at = path.find(component).unwrap_or(0);
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        } else {
            stack.push(component.to_string());
        }
    }

    if absolute {
        Ok(finish_absolute(&stack, intent))
    } else {
        Ok(finish_relative(&stack, intent))
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    normalize_impl(path)
}

fn valid_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize_impl(path).map_err(|_| ())?;
    Ok(components(&normalized).into_iter().map(str::to_string).collect())
}

fn tilde_start<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<(bool, &'a str, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, path, 0));
    }
    let bytes = path.as_bytes();
    if bytes.len() == 1 {
        let _ = home.ok_or(PathError::MissingHome)?;
        return Ok((true, "", 1));
    }
    if bytes[1] == b'/' {
        let _ = home.ok_or(PathError::MissingHome)?;
        return Ok((true, &path[2..], 2));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

fn reduce_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<bool, PathError> {
    if component.is_empty() || component == "." {
        return Ok(false);
    }
    if component == ".." {
        if stack.last().is_some_and(|c| c != "..") {
            stack.pop();
        } else if absolute {
            return Err(PathError::Escape { at });
        } else {
            stack.push("..".to_string());
        }
        Ok(false)
    } else {
        stack.push(component.to_string());
        Ok(true)
    }
}

fn resolve_core(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (expanded, remainder, consumed) = tilde_start(path, home)?;

    let absolute_request = !expanded && path.starts_with('/');
    let mut stack = if absolute_request {
        Vec::new()
    } else if expanded {
        home.unwrap_or(&[]).to_vec()
    } else {
        cwd.to_vec()
    };

    let mut count = 0usize;
    let mut active: Vec<usize> = Vec::new();

    enum Source<'a> {
        Context(Vec<String>),
        Raw(&'a str, usize),
    }

    let mut work: Vec<Source<'_>> = Vec::new();
    let request_offset = if expanded { consumed } else { 0 };
    work.push(Source::Raw(remainder, request_offset));

    while let Some(source) = work.pop() {
        match source {
            Source::Context(parts) => {
                for component in parts {
                    if component == "." || component.is_empty() {
                        continue;
                    }
                    if component == ".." {
                        if !stack.is_empty() {
                            stack.pop();
                        }
                    } else {
                        stack.push(component);
                        if let Some(index) = links.iter().position(|l| {
                            let key = finish_absolute(&stack, false);
                            l.path == key
                        }) {
                            if active.contains(&index) {
                                return Err(PathError::Loop { index });
                            }
                            if count >= max_expansions {
                                return Err(PathError::Limit { index });
                            }
                            count += 1;
                            active.push(index);
                            let target = &links[index].target;
                            let target_absolute = target.starts_with('/');
                            if target_absolute {
                                stack.clear();
                            } else {
                                stack.pop();
                            }
                            work.push(Source::Context(Vec::new()));
                            work.push(Source::Raw(target, 0));
                            // Active entries are retired by the marker below.
                            work.push(Source::Context(vec![format!("\0ACTIVE:{index}")]));
                        }
                    }
                }
            }
            Source::Raw(raw, base_offset) => {
                let mut offset = base_offset;
                let mut iter = raw.split('/');
                for component in iter.by_ref() {
                    let start = offset;
                    offset += component.len() + 1;
                    if component.is_empty() || component == "." {
                        continue;
                    }
                    if component.starts_with('\0') && component.starts_with("\0ACTIVE:") {
                        continue;
                    }
                    let pushed =
                        reduce_component(&mut stack, component, true, start)?;
                    if pushed {
                        let key = finish_absolute(&stack, false);
                        if let Some(index) = links.iter().position(|l| l.path == key) {
                            if active.contains(&index) {
                                return Err(PathError::Loop { index });
                            }
                            if count >= max_expansions {
                                return Err(PathError::Limit { index });
                            }
                            count += 1;
                            active.push(index);
                            stack.pop();
                            let target = &links[index].target;
                            if target.starts_with('/') {
                                stack.clear();
                            }
                            work.push(Source::Context(vec![format!("\0ACTIVE:{index}")]));
                            work.push(Source::Raw(target, 0));
                        }
                    }
                }
            }
        }

        if let Some(Source::Context(parts)) = work.last() {
            if parts.len() == 1 && parts[0].starts_with("\0ACTIVE:") {
                let marker = &parts[0][8..];
                if let Ok(index) = marker.parse::<usize>() {
                    active.retain(|&i| i != index);
                }
                work.pop();
            }
        }
    }

    Ok(finish_absolute(&stack, intent))
}

fn validate_link_table(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize_impl(&link.path).is_ok_and(|n| n == link.path)
            && link.path != "/";
        let target_valid = validate_raw(&link.target).is_ok();
        if !path_valid || !target_valid {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(p, _)| *p == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

fn resolve_prepared(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home, &[], 0)
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = valid_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(valid_context(h).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    validate_raw(path)?;
    resolve_prepared(path, &cwd_parts, home_parts.as_deref())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = valid_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(valid_context(h).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    validate_link_table(links)?;
    validate_raw(path)?;
    resolve_core(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        max_expansions,
    )
}

fn route(base: &[String], target: &[String]) -> String {
    let mut common = 0usize;
    while common < base.len()
        && common < target.len()
        && base[common] == target[common]
    {
        common += 1;
    }
    let mut parts = vec!["..".to_string(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

fn abs_parts(path: &str) -> Vec<String> {
    components(path).into_iter().map(str::to_string).collect()
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let b = abs_parts(&resolved_base);
    let t = abs_parts(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&b, &t)];
    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let a = abs_parts(&probe);
        if a.len() <= t.len() && a.iter().zip(&t).all(|(x, y)| x == y) {
            let mut candidate = route(&b, &abs_parts(&link.path));
            let suffix = &t[a.len()..];
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }
        if let Ok(result) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if abs_parts(&result) == t {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Empty)
}
