use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|part| !part.is_empty())
            .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn render_absolute(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return "/".to_owned();
    }
    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn render_normalized(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            return "/".to_owned();
        }
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        return ".".to_owned();
    } else {
        parts.join("/")
    };
    if intent {
        result.push('/');
    }
    result
}

fn reduce_plain(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    let mut stack: Vec<String> = Vec::new();
    for (at, part) in path.split('/').scan(0usize, |offset, part| {
        let current = *offset;
        *offset += part.len() + 1;
        Some((current, part))
    }) {
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_owned());
            }
        } else {
            stack.push(part.to_owned());
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let parts = reduce_plain(path, absolute)?;
    Ok(render_normalized(&parts, absolute, intent))
}

fn validated_context(path: &str, cwd: bool) -> Result<Vec<String>, PathError> {
    let invalid = if cwd {
        PathError::InvalidCwd
    } else {
        PathError::InvalidHome
    };
    if !path.starts_with('/') {
        return Err(invalid);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized)
            .into_iter()
            .map(str::to_owned)
            .collect()),
        Err(_) => Err(invalid),
    }
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        Some(path) => validated_context(path, false).map(Some),
        None => Ok(None),
    }
}

fn expand_request<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<(&'a str, Option<&'a str>), PathError> {
    if !path.starts_with('~') {
        return Ok((path, None));
    }
    if path.len() == 1 {
        if home.is_none() {
            return Err(PathError::MissingHome);
        }
        return Ok(("", Some("bare")));
    }
    if path.as_bytes()[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if home.is_none() {
        return Err(PathError::MissingHome);
    }
    Ok((&path[2..], Some("slash")))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, true)?;
    let home_parts = validate_home(home)?;
    validate_raw(path)?;
    let (request, tilde_kind) = expand_request(path, home_parts.as_deref())?;
    let intent = if tilde_kind == Some("bare") {
        false
    } else {
        directory_intent(path)
    };

    let mut stack = if path.starts_with('/') {
        Vec::new()
    } else if tilde_kind.is_some() {
        home_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    for (at, part) in request.split('/').scan(
        if tilde_kind == Some("slash") { 2usize } else { 0usize },
        |offset, part| {
            let current = *offset;
            *offset += part.len() + 1;
            Some((current, part))
        },
    ) {
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            stack.push(part.to_owned());
        }
    }
    Ok(render_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let key_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let target_valid = validate_raw(&link.target).is_ok();
        if !key_valid || !target_valid {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

#[derive(Clone)]
struct Work {
    raw: String,
    offset: usize,
    is_request: bool,
    finish_link: Option<usize>,
}

fn lookup_link<'a>(path: &str, links: &'a [Link]) -> Option<(usize, &'a Link)> {
    links
        .iter()
        .enumerate()
        .find(|(_, link)| link.path == path)
}

fn walk_links(
    initial: Vec<String>,
    request: &str,
    request_offset: usize,
    links: &[Link],
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    let mut work = vec![Work {
        raw: request.to_owned(),
        offset: request_offset,
        is_request: true,
        finish_link: None,
    }];
    let mut active: HashSet<usize> = HashSet::new();
    let mut expansions = 0usize;

    while let Some(mut current) = work.pop() {
        if let Some(index) = current.finish_link.take() {
            active.remove(&index);
            continue;
        }

        let mut cursor = 0usize;
        while cursor <= current.raw.len() {
            let start = cursor;
            let end = current.raw[start..]
                .find('/')
                .map(|n| start + n)
                .unwrap_or(current.raw.len());
            let part = &current.raw[start..end];
            cursor = end.saturating_add(1);
            if part.is_empty() || part == "." {
                if end == current.raw.len() {
                    break;
                }
                continue;
            }
            if part == ".." {
                if stack.pop().is_none() {
                    return Err(PathError::Escape {
                        at: current.offset + start,
                    });
                }
                if end == current.raw.len() {
                    break;
                }
                continue;
            }

            stack.push(part.to_owned());
            let absolute = render_absolute(&stack, false);
            if let Some((index, link)) = lookup_link(&absolute, links) {
                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if expansions >= max_expansions {
                    return Err(PathError::Limit { index });
                }
                expansions += 1;
                active.insert(index);

                let target = link.target.clone();
                let target_absolute = target.starts_with('/');
                if target_absolute {
                    stack.clear();
                } else {
                    stack.pop();
                }

                let target_work = Work {
                    raw: target,
                    offset: 0,
                    is_request: false,
                    finish_link: None,
                };
                work.push(Work {
                    raw: String::new(),
                    offset: 0,
                    is_request: false,
                    finish_link: Some(index),
                });
                if end < current.raw.len() {
                    work.push(Work {
                        raw: current.raw[end..].to_owned(),
                        offset: current.offset + end + 1,
                        is_request: current.is_request,
                        finish_link: None,
                    });
                }
                work.push(target_work);
                break;
            }
            if end == current.raw.len() {
                break;
            }
        }
    }
    Ok(stack)
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(&'a str, usize, Option<&'static str>), PathError> {
    if !path.starts_with('~') {
        return Ok((path, 0, None));
    }
    if path.len() == 1 {
        if home.is_none() {
            return Err(PathError::MissingHome);
        }
        return Ok(("", 0, Some("bare")));
    }
    if path.as_bytes()[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if home.is_none() {
        return Err(PathError::MissingHome);
    }
    Ok((&path[2..], 2, Some("slash")))
}

fn resolve_links_validated(
    path: &str,
    cwd_parts: Vec<String>,
    home_parts: Option<Vec<String>>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let (request, offset, tilde_kind) = prepare_request(path, home_parts.as_deref())?;
    let intent = if tilde_kind == Some("bare") {
        false
    } else {
        directory_intent(path)
    };
    let initial = if path.starts_with('/') {
        Vec::new()
    } else if tilde_kind.is_some() {
        home_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    let result = walk_links(initial, request, offset, links, max_expansions)?;
    Ok(render_absolute(&result, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, true)?;
    let home_parts = validate_home(home)?;
    validate_links(links)?;
    resolve_links_validated(path, cwd_parts, home_parts, links, max_expansions)
}

fn split_absolute(path: &str) -> Vec<String> {
    components(path)
        .into_iter()
        .map(str::to_owned)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_owned()));
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = split_absolute(&resolved_base);
    let target_parts = split_absolute(&resolved_target);

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        if let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let alias_parts = split_absolute(&probe);
            if alias_parts.len() <= target_parts.len()
                && alias_parts
                    .iter()
                    .zip(target_parts.iter())
                    .all(|(a, b)| a == b)
            {
                let mut candidate = route(&base_parts, &split_absolute(&link.path));
                let suffix = &target_parts[alias_parts.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let target_directory = resolved_target.ends_with('/');
    let mut best: Option<String> = None;

    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }
        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if split_absolute(&resolved) != target_parts {
                continue;
            }
            let replace = best.as_ref().is_none_or(|current| {
                candidate.len() < current.len()
                    || (candidate.len() == current.len()
                        && candidate.as_bytes() < current.as_bytes())
            });
            if replace {
                best = Some(candidate);
            }
        }
    }

    best.ok_or(PathError::Escape { at: 0 })
}
