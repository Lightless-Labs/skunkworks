#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(base + i);
            }
            out.push(value);
            i += 3;
        } else {
            if !unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).expect("decoded segments are ASCII"))
}

fn parse_path(
    input: &str,
    pattern: bool,
) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut start = 1;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(start);
        }

        let raw = &input[start..stop];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(start);
            }
            if marker == b'*' && stop != end {
                return Err(start);
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start)?;
            segments.push(Segment::Static(decoded));
        }

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if trailing && segments.is_empty() {
        return Ok((Vec::new(), false));
    }
    Ok((segments, trailing))
}

fn rank_for(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        rank.push(match segment {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        });
        if matches!(segment, Segment::Wildcard(_)) {
            return rank;
        }
    }
    rank.push(3);
    rank
}

fn shapes_conflict(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let candidate = Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            rank: rank_for(&segments),
            segments,
            trailing,
        };
        if let Some(with) = self
            .routes
            .iter()
            .filter(|r| shapes_conflict(r, &candidate, self.policy))
            .map(|r| r.id)
            .min()
        {
            return Err(Error::Conflict { with });
        }
        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) =
            parse_path(head, false).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some(params) = match_route(&route.segments, &request_segments) {
                candidates.push((route, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };
        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(r, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(r, _)| r.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank
                .cmp(&a.rank)
                .then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible[0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|r| r.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::<&str, &str>::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let required: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|s| match s {
                Segment::Param(n) | Segment::Wildcard(n) => Some(n.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !required.contains(key))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for key in &required {
            let value = supplied[key];
            if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
                return Err(Error::InvalidValue {
                    name: (*key).to_owned(),
                });
            }
            match route.segments.iter().find(|s| match s {
                Segment::Param(n) | Segment::Wildcard(n) => n == key,
                Segment::Static(_) => false,
            }) {
                Some(Segment::Param(_)) => {
                    if value.is_empty() || value.contains('/') {
                        return Err(Error::InvalidValue {
                            name: (*key).to_owned(),
                        });
                    }
                }
                Some(Segment::Wildcard(_)) => {
                    if !value.is_empty()
                        && value
                            .split('/')
                            .any(|part| part.is_empty())
                    {
                        return Err(Error::InvalidValue {
                            name: (*key).to_owned(),
                        });
                    }
                }
                _ => unreachable!(),
            }
        }

        let mut output = String::from("/");
        let mut emitted = false;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_component(value.as_bytes()));
                    emitted = true;
                }
                Segment::Param(key) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_component(supplied[key].as_bytes()));
                    emitted = true;
                }
                Segment::Wildcard(key) => {
                    let value = supplied[key];
                    if !value.is_empty() {
                        if emitted {
                            output.push('/');
                        }
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part.as_bytes()));
                        }
                        emitted = true;
                    }
                }
            }
        }
        if !emitted {
            return Ok("/".to_owned());
        }
        if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn match_route(
    pattern: &[Segment],
    request: &[Segment],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;
    for segment in pattern {
        match segment {
            Segment::Wildcard(name) => {
                let tail = request[i..]
                    .iter()
                    .map(|s| match s {
                        Segment::Static(v) => v.as_str(),
                        _ => unreachable!(),
                    })
                    .collect::<Vec<_>>()
                    .join("/");
                params.push((name.clone(), tail));
                return Some(params);
            }
            Segment::Static(expected) => match request.get(i) {
                Some(Segment::Static(actual)) if expected == actual => {}
                _ => return None,
            },
            Segment::Param(name) => match request.get(i) {
                Some(Segment::Static(value)) => params.push((name.clone(), value.clone())),
                _ => return None,
            },
        }
        i += 1;
    }
    if i == request.len() {
        Some(params)
    } else {
        None
    }
}

fn encode_component(bytes: &[u8]) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in bytes {
        if unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}
