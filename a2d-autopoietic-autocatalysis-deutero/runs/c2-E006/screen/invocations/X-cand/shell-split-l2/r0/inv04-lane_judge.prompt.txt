    env.iter()

fn find_closing_brace(input: &str, mut i: usize) -> Option<usize> {
    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                i += input[i..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().expect("valid character boundary");
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            word.push(escaped);
                        }
                        i = after;
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    let (end, value) = parse_expansion(input, i, env)?;
                    word.push_str(&value);
                    i = end;
                } else {
                    word.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if escaped != '\n' {
                        active = true;
                        word.push(escaped);
                    }
                    i = after;
                } else if c == '$' {
                    let (end, value, did_expand) = parse_dollar(input, i, env)?;
                    if did_expand {
                        active = true;
                        word.push_str(&value);
                    } else {
                        word.push('$');
                    }
                    i = end;
                } else {
                    active = true;
                    word.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_dollar(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String, bool), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar == input.len() {
        return Ok((after_dollar, String::new(), false));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let (end, value) = parse_braced(input, dollar, env)?;
        return Ok((end, value, true));
    }
    if !is_name_start(first) {
        return Ok((after_dollar, String::new(), false));
    }

    let mut end = after_dollar;
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if !is_name_char(c) {
            break;
        }
        end += c.len_utf8();
    }
    let name = &input[after_dollar..end];
    Ok((end, lookup(env, name).unwrap_or("").to_string(), true))
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let (end, value, _) = parse_dollar(input, dollar, env)?;
    Ok((end, value))
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;
    let mut close = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += 1;
            if cursor < input.len() {
                let escaped = input[cursor..].chars().next().unwrap();
                cursor += escaped.len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let name_len = contents
        .char_indices()
        .take_while(|(_, c)| is_name_char(*c))
        .last()
        .map_or(0, |(idx, c)| idx + c.len_utf8());

    if name_len == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let name = &contents[..name_len];
    let remainder = &contents[name_len..];

    if !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let selected = if remainder.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, selected))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let escaped = text[i..].chars().next().unwrap();
                result.push(escaped);
                i += escaped.len_utf8();
            }
        } else {
            result.push(c);
            i += c.len_utf8();
        }
    }
    result
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_open = dollar + 2;
    let tail = &input[after_open..];
    let mut close = None;
    let mut iter = tail.char_indices().peekable();

    while let Some((offset, c)) = iter.next() {
        if c == '\\' {
            // The search rule makes a backslash skip the following scalar.
            iter.next();
        } else if c == '}' {
            close = Some(after_open + offset);
            break;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let mut name_end = 0;
    let mut chars = contents.char_indices();
    match chars.next() {
        Some((_, c)) if is_name_start(c) => name_end = c.len_utf8(),
        _ => return Err(SplitError::InvalidExpansion { pos: dollar }),
    }
    for (offset, c) in chars {
        if is_name_continue(c) {
            name_end = offset + c.len_utf8();
        } else {
            break;
        }
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];

    let value = if rest.is_empty() {
        first_env_value(name, env).unwrap_or("").to_string()
    } else if let Some(text) = rest.strip_prefix(":-") {
        if let Some(value) = first_env_value(name, env).filter(|value| !value.is_empty()) {
            value.to_string()
        } else {
            let mut decoded = String::new();
            let mut text_chars = text.chars();
            while let Some(c) = text_chars.next() {
                if c == '\\' {
                    if let Some(next) = text_chars.next() {
                        decoded.push(next);
                    } else {
                        decoded.push('\\');
                    }
                } else {
                    decoded.push(c);
                }
            }
            decoded
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut iter = input.char_indices().peekable();

    while let Some((pos, c)) = iter.next() {
        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
            }
            Quote::Double => match c {
                '"' => quote = Quote::None,
                '\\' => match iter.next() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, next)) if matches!(next, '"' | '\\' | '$') => {
                        word.push(next);
                    }
                    Some((_, '\n')) => {}
                    Some((_, next)) => {
                        word.push('\\');
                        word.push(next);
                    }
                },
                '$' => {
                    active = true;
                    if let Some((next_pos, next)) = iter.peek().copied() {
                        if next == '{' {
                            let (end, value) = parse_expansion(input, pos, env)?;
                            while iter.peek().is_some_and(|(p, _)| *p < end) {
                                iter.next();
                            }
                            word.push_str(&value);
                        } else if is_name_start(next) {
                            let mut name = String::new();
                            while let Some((_, ch)) = iter.peek().copied() {
                                if is_name_continue(ch) {
                                    name.push(ch);
                                    iter.next();
                                } else {
                                    break;
                                }
                            }
                            word.push_str(first_env_value(&name, env).unwrap_or(""));
                        } else {
                            let _ = next_pos;
                            word.push('$');
                        }
                    } else {
                        word.push('$');
                    }
                }
                _ => word.push(c),
            },
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                } else {
                    match c {
                        '\'' => {
                            active = true;
                            quote = Quote::Single;
                            quote_pos = pos;
                        }
                        '"' => {
                            active = true;
                            quote = Quote::Double;
                            quote_pos = pos;
                        }
                        '\\' => match iter.next() {
                            None => return Err(SplitError::TrailingEscape { pos }),
                            Some((_, '\n')) => {}
                            Some((_, escaped)) => {
                                active = true;
                                word.push(escaped);
                            }
                        },
                        '$' => {
                            active = true;
                            if let Some((_, next)) = iter.peek().copied() {
                                if next == '{' {
                                    let (end, value) = parse_expansion(input, pos, env)?;
                                    while iter.peek().is_some_and(|(p, _)| *p < end) {
                                        iter.next();
                                    }
                                    word.push_str(&value);
                                } else if is_name_start(next) {
                                    let mut name = String::new();
                                    while let Some((_, ch)) = iter.peek().copied() {
                                        if is_name_continue(ch) {
                                            name.push(ch);
                                            iter.next();
                                        } else {
                                            break;
                                        }
                                    }
                                    word.push_str(first_env_value(&name, env).unwrap_or(""));
                                } else {
                                    word.push('$');
                                }
                            } else {
                                word.push('$');
                            }
                        }
                        _ => {
                            active = true;
                            word.push(c);
                        }
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
