#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn find_closing_brace(input: &str, mut i: usize) -> Option<usize> {
    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                i += input[i..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if matches!(n, '"' | '\\' | '$') {
                        word.push(n);
                        i = after_n;
                    } else if n == '\n' {
                        i = after_n;
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if is_name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            if let Some(value) = lookup(name, env) {
                                word.push_str(value);
                            }
                            i = end;
                        } else if n == '{' {
                            let content_start = next_i + 1;
                            let Some(close) = find_closing_brace(input, content_start) else {
                                return Err(SplitError::UnterminatedExpansion { pos: i });
                            };
                            let content = &input[content_start..close];
                            let (name, default) = if let Some(op) = content.find(":-") {
                                (&content[..op], Some(&content[op + 2..]))
                            } else {
                                (content, None)
                            };
                            if name.is_empty()
                                || !name.chars().next().is_some_and(is_name_start)
                                || !name.chars().all(is_name_continue)
                            {
                                return Err(SplitError::InvalidExpansion { pos: i });
                            }
                            let value = lookup(name, env);
                            if let Some(default_text) = default {
                                if let Some(value) = value.filter(|v| !v.is_empty()) {
                                    word.push_str(value);
                                } else {
                                    word.push_str(&decode_default(default_text));
                                }
                            } else if let Some(value) = value {
                                word.push_str(value);
                            }
                            i = close + 1;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    if n == '\n' {
                        i = next_i + n.len_utf8();
                    } else {
                        active = true;
                        word.push(n);
                        i = next_i + n.len_utf8();
                    }
                } else if c == '$' {
                    active = true;
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if is_name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            if let Some(value) = lookup(name, env) {
                                word.push_str(value);
                            }
                            i = end;
                        } else if n == '{' {
                            let content_start = next_i + 1;
                            let Some(close) = find_closing_brace(input, content_start) else {
                                return Err(SplitError::UnterminatedExpansion { pos: i });
                            };
                            let content = &input[content_start..close];
                            let (name, default) = if let Some(op) = content.find(":-") {
                                (&content[..op], Some(&content[op + 2..]))
                            } else {
                                (content, None)
                            };
                            if name.is_empty()
                                || !name.chars().next().is_some_and(is_name_start)
                                || !name.chars().all(is_name_continue)
                            {
                                return Err(SplitError::InvalidExpansion { pos: i });
                            }
                            let value = lookup(name, env);
                            if let Some(default_text) = default {
                                if let Some(value) = value.filter(|v| !v.is_empty()) {
                                    word.push_str(value);
                                } else {
                                    word.push_str(&decode_default(default_text));
                                }
                            } else if let Some(value) = value {
                                word.push_str(value);
                            }
                            i = close + 1;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    word.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
