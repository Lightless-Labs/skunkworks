```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

pub struct IndexedHeap {
    items: Vec<Entry>,
}

struct Entry {
    item: Item,
    value_len: usize,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self { items: Vec::new() }
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let canonical_key = canonicalize_key(key)?;

        if self.items.iter().any(|entry| entry.item.key == canonical_key) {
            return Err(Error::Duplicate { key: canonical_key });
        }

        let entry = Entry {
            item: Item {
                key: canonical_key,
                priority,
                value: value.to_owned(),
            },
            value_len: value.len(),
        };
        self.items.push(entry);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_index()
            .map(|index| self.items[index].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let index = self.best_index()?;
        Some(self.items.remove(index).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical_key = canonicalize_key(key)?;
        Ok(self
            .items
            .iter()
            .find(|entry| entry.item.key == canonical_key)
            .map(|entry| entry.item.clone()))
    }

    fn best_index(&self) -> Option<usize> {
        self.items
            .iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| {
                a.item
                    .priority
                    .cmp(&b.item.priority)
                    .then_with(|| a.value_len.cmp(&b.value_len))
                    .then_with(|| a_index_age_placeholder())
            })
            .map(|(index, _)| index)
    }
}

fn a_index_age_placeholder() -> std::cmp::Ordering {
    std::cmp::Ordering::Equal
}

fn canonicalize_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    let trimmed = &bytes[start..end];
    for (offset, &byte) in trimmed.iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    let mut canonical = input[start..end].to_owned();
    canonical.make_ascii_lowercase();
    Ok(canonical)
}
```
