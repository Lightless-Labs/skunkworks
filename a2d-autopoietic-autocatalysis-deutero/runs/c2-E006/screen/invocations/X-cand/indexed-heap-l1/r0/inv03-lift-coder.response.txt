```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    heap: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            heap: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.heap.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        // Rebase ages before the counter could overflow. The heap's current
        // order preserves relative ages among retained entries.
        if self.next_age == u128::MAX {
            let mut order: Vec<usize> = (0..self.heap.len()).collect();
            order.sort_unstable_by_key(|&index| self.heap[index].age);
            for (age, index) in order.into_iter().enumerate() {
                self.heap[index].age = age as u128;
            }
            self.next_age = self.heap.len() as u128;
        }

        let entry = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };
        self.next_age += 1;

        let index = self.heap.len();
        self.heap.push(entry);
        self.indices.insert(key, index);
        self.sift_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.heap.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.heap.is_empty() {
            return None;
        }

        let last = self.heap.len() - 1;
        self.swap_entries(0, last);
        let entry = self.heap.pop().expect("heap was nonempty");
        self.indices.remove(&entry.item.key);

        if !self.heap.is_empty() {
            self.sift_down(0);
        }

        Some(entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self
            .indices
            .get(&key)
            .map(|&index| self.heap[index].item.clone()))
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !entry_is_better(&self.heap[index], &self.heap[parent]) {
                break;
            }
            self.swap_entries(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        let len = self.heap.len();
        loop {
            let left = index * 2 + 1;
            if left >= len {
                break;
            }

            let right = left + 1;
            let best_child = if right < len
                && entry_is_better(&self.heap[right], &self.heap[left])
            {
                right
            } else {
                left
            };

            if !entry_is_better(&self.heap[best_child], &self.heap[index]) {
                break;
            }
            self.swap_entries(index, best_child);
            index = best_child;
        }
    }

    fn swap_entries(&mut self, first: usize, second: usize) {
        if first == second {
            return;
        }

        self.heap.swap(first, second);
        self.indices
            .insert(self.heap[first].item.key.clone(), first);
        self.indices
            .insert(self.heap[second].item.key.clone(), second);
    }
}

fn entry_is_better(left: &Entry, right: &Entry) -> bool {
    left.item.priority < right.item.priority
        || (left.item.priority == right.item.priority
            && (left.item.value.len() < right.item.value.len()
                || (left.item.value.len() == right.item.value.len()
                    && left.age < right.age)))
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };

        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}
```
