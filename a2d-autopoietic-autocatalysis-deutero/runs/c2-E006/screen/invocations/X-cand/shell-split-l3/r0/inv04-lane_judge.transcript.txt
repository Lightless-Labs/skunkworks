pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_text(word: &mut Word, value: &str, active_globs: bool) {
    for c in value.chars() {
        if active_globs && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn append_default(word: &mut Word, raw: &str, active_globs: bool) {
    let mut chars = raw.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                append_text(word, &next.to_string(), active_globs);
            } else {
                word.text.push('\\');
            }
        } else {
            append_text(word, &c.to_string(), active_globs);
        }
    }
}

fn lookup_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn parse_expansion(
    input: &str,
    bytes: &[u8],
    dollar: usize,
    env: &[(&str, &str)],
    word: &mut Word,
    active_globs: bool,
) -> Result<usize, SplitError> {
    let mut close = None;
    let mut i = dollar + 2;

    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let c = input[i..].chars().next().expect("valid UTF-8 boundary");
                i += c.len_utf8();
            }
        } else if bytes[i] == b'}' {
            close = Some(i);
            break;
        } else {
            let c = input[i..].chars().next().expect("valid UTF-8 boundary");
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[dollar + 2..close];

    let name_end = contents
        .char_indices()
        .find_map(|(idx, c)| (!is_name_continue(c)).then_some(idx))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    let valid_name = !name.is_empty()
        && name.chars().next().is_some_and(is_name_start)
        && name.chars().all(is_name_continue);

    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let rest = &contents[name_end..];
    let default = if rest.is_empty() {
        None
    } else if let Some(text) = rest.strip_prefix(":-") {
        Some(text)
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    match default {
        None => {
            if let Some(value) = lookup_env(env, name) {
                append_text(word, value, active_globs);
            }
        }
        Some(text) => match lookup_env(env, name) {
            Some(value) if !value.is_empty() => append_text(word, value, active_globs),
            _ => append_default(word, text, active_globs),
        },
    }

    Ok(close + 1)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn append_literal(word: &mut Word, c: char, wildcard_active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_expansion(word: &mut Word, text: &str, wildcard_active: bool) {
    for c in text.chars() {
        append_literal(word, c, wildcard_active);
    }
}

fn first_env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.chars();

    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn parse_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let bytes = input.as_bytes();
    let mut close = None;
    let mut i = dollar_pos + 2;

    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let c = input[i..].chars().next().unwrap();
                i += c.len_utf8();
            }
        } else if bytes[i] == b'}' {
            close = Some(i);
            break;
        } else {
            let c = input[i..].chars().next().unwrap();
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let contents = &input[dollar_pos + 2..close];

    let mut name_end = 0;
    let mut chars = contents.char_indices();
    match chars.next() {
        Some((_, c)) if is_name_start(c) => {
            name_end = c.len_utf8();
        }
        _ => return Err(SplitError::InvalidExpansion { pos: dollar_pos }),
    }

    for (offset, c) in chars {
        if !is_name_continue(c) {
            break;
        }
        name_end = offset + c.len_utf8();
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];

    let result = if rest.is_empty() {
        first_env_value(env, name).unwrap_or("").to_string()
    } else if let Some(default_text) = rest.strip_prefix(":-") {
        match first_env_value(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default_text),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((result, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut current, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    let slash_pos = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let following = input[next_i..].chars().next().unwrap();
                    let after_following = next_i + following.len_utf8();
                    if matches!(following, '"' | '\\' | '$') {
                        append_literal(&mut current, following, false);
                        i = after_following;
                    } else if following == '\n' {
                        i = after_following;
                    } else {
                        append_literal(&mut current, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() && input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }

                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let (value, after) = parse_braced(input, i, env)?;
                        append_expansion(&mut current, &value, false);
                        active = true;
                        i = after;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            let value = first_env_value(env, name).unwrap_or("");
                            append_expansion(&mut current, value, false);
                            active = true;
                            i = end;
                        } else {
                            append_literal(&mut current, '$', false);
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut current, '$', false);
                        i = next_i;
                    }
                } else {
                    append_literal(&mut current, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    let slash_pos = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let following = input[next_i..].chars().next().unwrap();
                    let after_following = next_i + following.len_utf8();
                    if following == '\n' {
                        i = after_following;
                    } else {
                        append_literal(&mut current, following, false);
                        active = true;
                        i = after_following;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() && input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }

                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let (value, after) = parse_braced(input, i, env)?;
                        append_expansion(&mut current, &value, true);
                        active = true;
                        i = after;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            let value = first_env_value(env, name).unwrap_or("");
                            append_expansion(&mut current, value, true);
                            active = true;
                            i = end;
                        } else {
                            append_literal(&mut current, '$', true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut current, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_literal(&mut current, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }

    Ok(words)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn append_text(word: &mut Word, value: &str, active_globs: bool) {
    for c in value.chars() {
        if active_globs && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn append_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn find_braced_end(input: &str, start: usize) -> Option<usize> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next().unwrap();
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    let first = input[after_dollar..].chars().next().unwrap();

    if first == '{' {
        let contents_start = after_dollar + 1;
        let end = find_braced_end(input, contents_start)
            .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[contents_start..end];

        let mut name_end = 0;
        for c in contents.chars() {
            if (name_end == 0 && is_name_start(c)) || (name_end > 0 && is_name_continue(c)) {
                name_end += c.len_utf8();
            } else {
                break;
            }
        }

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let remainder = &contents[name_end..];
        let value = if remainder.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((end + 1, value));
    }

    if is_name_start(first) {
        let mut end = after_dollar;
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if is_name_continue(c) {
                end += c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((end, lookup(name, env).unwrap_or("").to_string()))
    } else {
        Ok((after_dollar, String::new()))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut current, c, false);
                }
                i = next_i;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        append_scalar(&mut current, next, false);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        append_scalar(&mut current, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let following = input[next_i..].chars().next();
                    if following.is_some_and(is_name_start) || following == Some('{') {
                        let (end, value) = parse_expansion(input, i, env)?;
                        active = true;
                        append_text(&mut current, &value, false);
                        i = end;
                    } else {
                        append_scalar(&mut current, '$', false);
                        i = next_i;
                    }
                } else {
                    append_scalar(&mut current, c, false);
                    i = next_i;
                }
                if i == input.len() {
                    let _ = open;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next != '\n' {
                        active = true;
                        append_scalar(&mut current, next, false);
                    }
                    i = after_next;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let following = input[next_i..].chars().next();
                    if following.is_some_and(is_name_start) || following == Some('{') {
                        let (end, value) = parse_expansion(input, i, env)?;
                        active = true;
                        append_text(&mut current, &value, true);
                        i = end;
                    } else {
                        active = true;
                        append_scalar(&mut current, '$', false);
                        i = next_i;
                    }
                } else {
                    active = true;
                    append_scalar(&mut current, c, matches!(c, '*' | '?' | '['));
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}
