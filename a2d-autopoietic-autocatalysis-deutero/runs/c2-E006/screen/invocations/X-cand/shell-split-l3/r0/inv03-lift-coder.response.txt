#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_text(word: &mut Word, value: &str, active_globs: bool) {
    for c in value.chars() {
        if active_globs && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn append_default(word: &mut Word, raw: &str, active_globs: bool) {
    let mut chars = raw.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                append_text(word, &next.to_string(), active_globs);
            } else {
                word.text.push('\\');
            }
        } else {
            append_text(word, &c.to_string(), active_globs);
        }
    }
}

fn lookup_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn parse_expansion(
    input: &str,
    bytes: &[u8],
    dollar: usize,
    env: &[(&str, &str)],
    word: &mut Word,
    active_globs: bool,
) -> Result<usize, SplitError> {
    let mut close = None;
    let mut i = dollar + 2;

    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let c = input[i..].chars().next().expect("valid UTF-8 boundary");
                i += c.len_utf8();
            }
        } else if bytes[i] == b'}' {
            close = Some(i);
            break;
        } else {
            let c = input[i..].chars().next().expect("valid UTF-8 boundary");
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[dollar + 2..close];

    let name_end = contents
        .char_indices()
        .find_map(|(idx, c)| (!is_name_continue(c)).then_some(idx))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    let valid_name = !name.is_empty()
        && name.chars().next().is_some_and(is_name_start)
        && name.chars().all(is_name_continue);

    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let rest = &contents[name_end..];
    let default = if rest.is_empty() {
        None
    } else if let Some(text) = rest.strip_prefix(":-") {
        Some(text)
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    match default {
        None => {
            if let Some(value) = lookup_env(env, name) {
                append_text(word, value, active_globs);
            }
        }
        Some(text) => match lookup_env(env, name) {
            Some(value) if !value.is_empty() => append_text(word, value, active_globs),
            _ => append_default(word, text, active_globs),
        },
    }

    Ok(close + 1)
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let bytes = input.as_bytes();
    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_open = 0;
    let mut i = 0;

    while i < bytes.len() {
        let c = input[i..].chars().next().expect("valid UTF-8 boundary");
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    current.text.push(c);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().expect("valid UTF-8 boundary");
                    if matches!(next, '"' | '\\' | '$') {
                        current.text.push(next);
                        i = next_i + next.len_utf8();
                    } else if next == '\n' {
                        i = next_i + next.len_utf8();
                    } else {
                        current.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if bytes.get(next_i) == Some(&b'(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if bytes.get(next_i) == Some(&b'{') {
                        active = true;
                        i = parse_expansion(
                            input,
                            bytes,
                            i,
                            env,
                            &mut current,
                            false,
                        )?;
                    } else if next_i < bytes.len() {
                        let next = input[next_i..].chars().next().expect("valid UTF-8 boundary");
                        if is_name_start(next) {
                            active = true;
                            let mut end = next_i + next.len_utf8();
                            while end < bytes.len() {
                                let ch = input[end..].chars().next().expect("valid UTF-8 boundary");
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            if let Some(value) = lookup_env(env, name) {
                                append_text(&mut current, value, false);
                            }
                            i = end;
                        } else {
                            current.text.push('$');
                            i = next_i;
                        }
                    } else {
                        current.text.push('$');
                        i = next_i;
                    }
                } else {
                    current.text.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_open = i;
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_open = i;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().expect("valid UTF-8 boundary");
                    if next == '\n' {
                        i = next_i + next.len_utf8();
                    } else {
                        active = true;
                        current.text.push(next);
                        i = next_i + next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if bytes.get(next_i) == Some(&b'(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if bytes.get(next_i) == Some(&b'{') {
                        active = true;
                        i = parse_expansion(
                            input,
                            bytes,
                            i,
                            env,
                            &mut current,
                            true,
                        )?;
                    } else if next_i < bytes.len() {
                        let next = input[next_i..].chars().next().expect("valid UTF-8 boundary");
                        if is_name_start(next) {
                            active = true;
                            let mut end = next_i + next.len_utf8();
                            while end < bytes.len() {
                                let ch = input[end..].chars().next().expect("valid UTF-8 boundary");
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            if let Some(value) = lookup_env(env, name) {
                                append_text(&mut current, value, true);
                            }
                            i = end;
                        } else {
                            active = true;
                            current.text.push('$');
                            i = next_i;
                        }
                    } else {
                        active = true;
                        current.text.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    append_text(&mut current, &c.to_string(), true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_open }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_open }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }

    Ok(words)
}
