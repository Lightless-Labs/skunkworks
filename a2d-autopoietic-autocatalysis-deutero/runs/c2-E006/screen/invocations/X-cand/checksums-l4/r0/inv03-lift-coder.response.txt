#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        update_base(&mut self.crc_raw, &mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        for &i in &[1usize, 2, 3, 4, 5, 6, 7, 8, 11, 12, 13, 14, 15, 16, 17, 18] {
            if hex_value(bytes[i]).is_none() {
                return Err(SnapshotError::Hex { offset: i });
            }
        }
        let crc = parse_hex_u32(bytes, 1);
        let adler = parse_hex_u32(bytes, 11);
        Self::from_seed(crc, adler)
    }
}

fn update_base(crc: &mut u32, a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *crc ^= byte as u32;
        for _ in 0..8 {
            *crc = (*crc >> 1) ^ (if *crc & 1 != 0 { 0xedb8_8320 } else { 0 });
        }
        *a = (*a + byte as u32) % 65521;
        *b = (*b + *a) % 65521;
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut raw = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    update_base(&mut raw, &mut a, &mut b, data);
    raw ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_update(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65535;
    *b = (*b + *a) % 65535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut i = 0;
    while i + 1 < data.len() {
        fletcher32_update(&mut a, &mut b, u16::from_be_bytes([data[i], data[i + 1]]));
        i += 2;
    }
    if i < data.len() {
        fletcher32_update(&mut a, &mut b, (data[i] as u16) << 8);
    }
    (b << 16) | a
}

fn fletcher32_spans_inner(data: &[u8], spans: &[Span]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending: Option<u8> = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                fletcher32_update(&mut a, &mut b, u16::from_be_bytes([high, byte]));
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        fletcher32_update(&mut a, &mut b, (high as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    Ok(fletcher32_spans_inner(data, spans))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

fn crc16_update(mut reg: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                reg ^= (byte as u16) << 8;
                for _ in 0..8 {
                    reg = if reg & 0x8000 != 0 {
                        (reg << 1) ^ 0x1021
                    } else {
                        reg << 1
                    };
                }
            }
            reg
        }
        Crc16::X25 => {
            for &byte in data {
                reg ^= byte as u16;
                for _ in 0..8 {
                    reg = if reg & 1 != 0 {
                        (reg >> 1) ^ 0x8408
                    } else {
                        reg >> 1
                    };
                }
            }
            reg
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let initial = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let reg = crc16_update(initial, data, variant);
    if variant == Crc16::X25 { reg ^ 0xffff } else { reg }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut f16a = 0u32;
    let mut f16b = 0u32;
    let mut f32a = 0u32;
    let mut f32b = 0u32;
    let mut pending = None;
    let initial16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let mut c16 = initial16;

    for span in spans {
        let slice = &data[span.start..span.end];
        stream.update(slice);
        c16 = crc16_update(c16, slice, variant);
        for &byte in slice {
            f16a = (f16a + byte as u32) % 255;
            f16b = (f16b + f16a) % 255;
            if let Some(high) = pending.take() {
                fletcher32_update(
                    &mut f32a,
                    &mut f32b,
                    u16::from_be_bytes([high, byte]),
                );
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        fletcher32_update(&mut f32a, &mut f32b, (high as u16) << 8);
    }
    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }
    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16b as u16) << 8) | f16a as u16,
        fletcher32: (f32b << 16) | f32a,
        crc16: c16,
    })
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    fn zero_byte(mut value: u32) -> u32 {
        for _ in 0..8 {
            value = (value >> 1) ^ if value & 1 != 0 { 0xedb8_8320 } else { 0 };
        }
        value
    }

    let mut power = [0u32; 32];
    for i in 0..32 {
        power[i] = zero_byte(1u32 << i);
    }

    fn apply(matrix: &[u32; 32], value: u32) -> u32 {
        let mut result = 0u32;
        for i in 0..32 {
            if value & (1u32 << i) != 0 {
                result ^= matrix[i];
            }
        }
        result
    }

    let mut len = right_len;
    let mut transformed = left;
    while len != 0 {
        if len & 1 != 0 {
            transformed = apply(&power, transformed);
        }
        len >>= 1;
        if len != 0 {
            let old = power;
            for i in 0..32 {
                power[i] = apply(&old, old[i]);
            }
        }
    }
    transformed ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }
    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut len = 0u64;
    for span in spans {
        let slice = &data[span.start..span.end];
        stream.update(slice);
        len += slice.len() as u64;
    }
    let sums = stream.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, sums.crc32, sums.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch { field: usize, expected: u32, actual: u32 },
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize) -> u32 {
    let mut value = 0u32;
    for &byte in &bytes[start..start + 8] {
        value = (value << 4) | hex_value(byte).unwrap();
    }
    value
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let mut length = 0u64;
    for &byte in &bytes[4..20] {
        length = (length << 4) | hex_value(byte).unwrap() as u64;
    }
    let expected_crc = parse_hex_u32(bytes, 25);
    let expected_adler = parse_hex_u32(bytes, 40);

    let mut stream = Stream::new();
    let mut selected_len = 0u64;
    for span in spans {
        let slice = &data[span.start..span.end];
        selected_len += slice.len() as u64;
        stream.update(slice);
    }
    if length != selected_len {
        return Err(VerifyError::Length);
    }
    let actual = stream.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
