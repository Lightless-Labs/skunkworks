#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        update_base(&mut self.crc, &mut self.a, &mut self.b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_hex_u32(bytes, 1..9)?;
        let adler = parse_hex_u32(bytes, 11..19)?;
        Self::from_seed(crc, adler)
    }
}

fn update_base(crc: &mut u32, a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *crc ^= byte as u32;
        for _ in 0..8 {
            *crc = if *crc & 1 != 0 {
                (*crc >> 1) ^ 0xedb8_8320
            } else {
                *crc >> 1
            };
        }

        *a = (*a + byte as u32) % 65521;
        *b = (*b + *a) % 65521;
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            crc = if crc & 1 != 0 {
                (crc >> 1) ^ 0xedb8_8320
            } else {
                crc >> 1
            };
        }
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

fn fletcher32_update(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65535;
    *b = (*b + *a) % 65535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        fletcher32_update(&mut a, &mut b, u16::from_be_bytes([pair[0], pair[1]]));
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_update(&mut a, &mut b, (last as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                fletcher32_update(&mut a, &mut b, ((high as u16) << 8) | byte as u16);
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        fletcher32_update(&mut a, &mut b, (high as u16) << 8);
    }
    Ok((b << 16) | a)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc: u16 = if variant == Crc16::Xmodem { 0 } else { 0xffff };
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    crc = if crc & 0x8000 != 0 {
                        (crc << 1) ^ 0x1021
                    } else {
                        crc << 1
                    };
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc = 0xffffu16;
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    crc = if crc & 1 != 0 {
                        (crc >> 1) ^ 0x8408
                    } else {
                        crc >> 1
                    };
                }
            }
            crc ^ 0xffff
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

struct MultiAccumulator {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    f16_a: u32,
    f16_b: u32,
    f32_a: u32,
    f32_b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
    variant: Crc16,
}

impl MultiAccumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            f16_a: 0,
            f16_b: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc ^= byte as u32;
            for _ in 0..8 {
                self.crc = if self.crc & 1 != 0 {
                    (self.crc >> 1) ^ 0xedb8_8320
                } else {
                    self.crc >> 1
                };
            }

            self.adler_a = (self.adler_a + byte as u32) % 65521;
            self.adler_b = (self.adler_b + self.adler_a) % 65521;

            self.f16_a = (self.f16_a + byte as u32) % 255;
            self.f16_b = (self.f16_b + self.f16_a) % 255;

            if let Some(high) = self.f32_pending.take() {
                fletcher32_update(
                    &mut self.f32_a,
                    &mut self.f32_b,
                    ((high as u16) << 8) | byte as u16,
                );
            } else {
                self.f32_pending = Some(byte);
            }

            match self.variant {
                Crc16::Xmodem | Crc16::False => {
                    self.crc16 ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        self.crc16 = if self.crc16 & 0x8000 != 0 {
                            (self.crc16 << 1) ^ 0x1021
                        } else {
                            self.crc16 << 1
                        };
                    }
                }
                Crc16::X25 => {
                    self.crc16 ^= byte as u16;
                    for _ in 0..8 {
                        self.crc16 = if self.crc16 & 1 != 0 {
                            (self.crc16 >> 1) ^ 0x8408
                        } else {
                            self.crc16 >> 1
                        };
                    }
                }
            }
        }
    }

    fn finish(self) -> Multi {
        let f32_a = self.f32_a;
        let f32_b = self.f32_b;
        let (f32_a, f32_b) = if let Some(high) = self.f32_pending {
            let mut a = f32_a;
            let mut b = f32_b;
            fletcher32_update(&mut a, &mut b, (high as u16) << 8);
            (a, b)
        } else {
            (f32_a, f32_b)
        };

        Multi {
            base: Hashes {
                crc32: self.crc ^ 0xffff_ffff,
                adler32: (self.adler_b << 16) | self.adler_a,
            },
            fletcher16: ((self.f16_b << 8) | self.f16_a) as u16,
            fletcher32: (f32_b << 16) | f32_a,
            crc16: if self.variant == Crc16::X25 {
                self.crc16 ^ 0xffff
            } else {
                self.crc16
            },
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = MultiAccumulator::new(variant);
    for span in spans {
        acc.update(&data[span.start..span.end]);
    }
    Ok(acc.finish())
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
        let mut sum = 0u32;
        let mut index = 0usize;
        while vector != 0 {
            if vector & 1 != 0 {
                sum ^= matrix[index];
            }
            vector >>= 1;
            index += 1;
        }
        sum
    }

    fn matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
        for i in 0..32 {
            square[i] = matrix_times(matrix, matrix[i]);
        }
    }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    let mut even = [0u32; 32];
    matrix_square(&mut even, &odd);
    matrix_square(&mut odd, &even);

    let mut length = right_len;
    let mut crc = left;
    loop {
        matrix_square(&mut even, &odd);
        if length & 1 != 0 {
            crc = matrix_times(&even, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }

        matrix_square(&mut odd, &even);
        if length & 1 != 0 {
            crc = matrix_times(&odd, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }
    }

    crc ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn validate_and_count(data_len: usize, spans: &[Span]) -> Result<u64, SpanError> {
    validate_spans(data_len, spans)?;
    let mut total = 0u64;
    for span in spans {
        let length = span.end - span.start;
        total = total.saturating_add(length as u64);
    }
    Ok(total)
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let length = validate_and_count(data.len(), spans)?;
    let hashes = checksums_spans(data, spans)?;
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length, hashes.crc32, hashes.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8], range: std::ops::Range<usize>) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in range {
        let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

fn parse_hex_u64_verify(bytes: &[u8], range: std::ops::Range<usize>) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in range {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        value = (value << 4) | digit as u64;
    }
    Ok(value)
}

fn parse_hex_u32_verify(
    bytes: &[u8],
    range: std::ops::Range<usize>,
) -> Result<u32, VerifyError> {
    let mut value = 0u32;
    for offset in range {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    let selected_len =
        validate_and_count(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();

    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let length = parse_hex_u64_verify(bytes, 4..20)?;
    let expected_crc = parse_hex_u32_verify(bytes, 25..33)?;
    let expected_adler = parse_hex_u32_verify(bytes, 40..48)?;

    if length != selected_len {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
