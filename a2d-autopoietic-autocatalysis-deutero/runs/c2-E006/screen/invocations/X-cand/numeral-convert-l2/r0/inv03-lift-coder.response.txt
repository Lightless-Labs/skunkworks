#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = base.to_string();
        if bytes[..hash] != *expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    // Invalid digits take precedence over separator errors.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < base => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    // Separators must have a digit immediately on both sides.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let valid_left = offset > 0 && payload[offset - 1] != b'_';
            let valid_right = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !valid_left || !valid_right {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let digits: Vec<u8> = payload
        .iter()
        .filter_map(|&byte| if byte == b'_' { None } else { digit_value(byte) })
        .collect();

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Big unsigned integers are stored little-endian in the given radix.
fn convert_digits(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value: Vec<u8> = Vec::new();

    for &digit in digits {
        let mut carry = digit as u16;
        for place in &mut value {
            let current = (*place as u16) * (from as u16) + carry;
            *place = (current % to as u16) as u8;
            carry = current / to as u16;
        }
        while carry != 0 {
            value.push((carry % to as u16) as u8);
            carry /= to as u16;
        }
    }

    if value.is_empty() {
        value.push(0);
    }
    value
}

fn add_digits(left: &[u8], right: &[u8], base: u8) -> Vec<u8> {
    let mut sum = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    let length = left.len().max(right.len());

    for index in 0..length {
        let a = left.get(index).copied().unwrap_or(0) as u16;
        let b = right.get(index).copied().unwrap_or(0) as u16;
        let current = a + b + carry;
        sum.push((current % base as u16) as u8);
        carry = current / base as u16;
    }

    if carry != 0 {
        sum.push(carry as u8);
    }

    while sum.len() > 1 && sum.last() == Some(&0) {
        sum.pop();
    }
    sum
}

fn render(mut digits: Vec<u8>, style: BaseStyle) -> String {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }

    let width = digits.len().max(style.min_width);
    let padding = width - digits.len();
    let mut output = Vec::with_capacity(width + width / style.group.max(1));

    for _ in 0..padding {
        output.push(b'0');
    }

    for digit in digits.iter().rev() {
        output.push(match *digit {
            0..=9 => b'0' + *digit,
            10..=35 if style.upper => b'A' + (*digit - 10),
            10..=35 => b'a' + (*digit - 10),
            _ => unreachable!(),
        });
    }

    if style.group == 0 {
        return String::from_utf8(output).expect("output contains ASCII only");
    }

    let mut grouped = Vec::with_capacity(output.len() + output.len() / style.group);
    let first_group = match output.len() % style.group {
        0 => style.group,
        remainder => remainder,
    };

    for (index, &byte) in output.iter().enumerate() {
        if index != 0 && (index - first_group) % style.group == 0 {
            grouped.push(b'_');
        }
        grouped.push(byte);
    }

    String::from_utf8(grouped).expect("output contains ASCII only")
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;

    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    Ok(render(convert_digits(&digits, from, to), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;

    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    Ok(render(add_digits(&left_digits, &right_digits, base), style))
}
