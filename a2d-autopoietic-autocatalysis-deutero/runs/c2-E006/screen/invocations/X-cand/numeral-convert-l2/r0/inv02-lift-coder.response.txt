#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut values = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => values.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if values.len() >= 2 && values[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(values)
}

fn multiply_add(number: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for value in number.iter_mut() {
        let current = *value as u16 * radix as u16 + carry;
        *value = (current % 256) as u8;
        carry = current / 256;
    }
    while carry != 0 {
        number.push((carry % 256) as u8);
        carry /= 256;
    }
}

fn add_numbers(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for i in 0..left.len().max(right.len()) {
        let sum = left.get(i).copied().unwrap_or(0) as u16
            + right.get(i).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum % 256) as u8);
        carry = sum / 256;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn render(number: &[u8], radix: u8, style: BaseStyle) -> String {
    let mut value = number.to_vec();
    while value.last() == Some(&0) {
        value.pop();
    }

    let mut digits = Vec::new();
    if value.is_empty() {
        digits.push(0);
    } else {
        while !value.is_empty() {
            let mut remainder = 0u16;
            for byte in value.iter_mut().rev() {
                let current = remainder * 256 + *byte as u16;
                *byte = (current / radix as u16) as u8;
                remainder = current % radix as u16;
            }
            digits.push(remainder as u8);
            while value.last() == Some(&0) {
                value.pop();
            }
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let mut padded = vec![0; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::with_capacity(digits.len() + digits.len() / style.group.max(1));
    for (i, digit) in digits.iter().copied().enumerate() {
        if style.group != 0 && i > 0 && (digits.len() - i) % style.group == 0 {
            output.push('_');
        }
        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(byte as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    if !(2..=36).contains(&from) {
        return Err(BaseError::BadBase {
            which: 0,
            value: from,
        });
    }
    if !(2..=36).contains(&to) {
        return Err(BaseError::BadBase {
            which: 1,
            value: to,
        });
    }
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let mut number = Vec::new();
    for digit in digits {
        multiply_add(&mut number, from, digit);
    }
    Ok(render(&number, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    if !(2..=36).contains(&base) {
        return Err(BaseError::BadBase {
            which: 0,
            value: base,
        });
    }
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_number = Vec::new();
    for digit in left_digits {
        multiply_add(&mut left_number, base, digit);
    }

    let mut right_number = Vec::new();
    for digit in right_digits {
        multiply_add(&mut right_number, base, digit);
    }

    let sum = add_numbers(&left_number, &right_number);
    Ok(render(&sum, base, style))
}
