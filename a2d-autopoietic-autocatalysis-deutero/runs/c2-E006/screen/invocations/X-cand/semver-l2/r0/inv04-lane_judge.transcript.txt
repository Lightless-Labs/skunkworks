#[derive(Clone, Debug, PartialEq, Eq)]

    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && digit(bytes[*pos]) {
        *pos += 1;
    }

    if start == *pos {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok(value)
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(digit)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &s[ident_start..i];
        if prerelease && is_numeric(ident) && ident.len() > 1 && ident.starts_with('0') {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(ident.to_string());

        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start > end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = core_number(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }

    pos += 1;
    let minor = core_number(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }

    pos += 1;
    let patch = core_number(s, &mut pos)?;

    if pos < s.len() && !matches!(s.as_bytes()[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        let pre_start = pos + 1;
        let build_sep = s.as_bytes()[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pre_start + i);
        let pre_end = build_sep.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }

    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        build = parse_identifiers(s, pos + 1, s.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    a.major
        .cmp(&b.major)
        .then_with(|| a.minor.cmp(&b.minor))
        .then_with(|| a.patch.cmp(&b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn parse_partial_operand(
    s: &str,
    at: usize,
) -> Result<PartialOperand, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parse_version(s)
            .map(PartialOperand::Full)
            .map_err(|error| RangeError::Version { at, error });
    }

    let fields: Vec<&str> = s.split('.').collect();
    let has_wildcard = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if fields.len() >= 3 && !has_wildcard {
        return parse_version(s)
            .map(PartialOperand::Full)
            .map_err(|error| RangeError::Version { at, error });
    }

    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for field in fields {
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !field.bytes().all(digit)
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }

    Ok(PartialOperand::Partial {
        numeric_count: nums.len(),
        nums,
    })
}

fn padded(nums: &[u64]) -> Version {
    Version {
        major: *nums.first().unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_version(mut v: Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let target = match field {
        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field {
        0 => {
            v.minor = 0;
            v.patch = 0;
        }
        1 => v.patch = 0,
        _ => {}
    }
    v.pre = vec!["0".to_string()];
    v.build.clear();
    Ok(v)
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors
            .push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator { op, version });
}

fn operand_version(op: PartialOperand) -> Version {
    match op {
        PartialOperand::Full(v) => v,
        PartialOperand::Partial { nums, .. } => padded(&nums),
    }
}

fn parse_expanded_token(

            }
        } else if k == 1 {
            0
        } else {
            1
        };
        let upper = upper_version(lower, field, operand_start)?;
        add_cmp(arm, Op::Lt, upper, false);
        return Ok(());
    }

}

fn parse_comparator(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let bytes = token.text.as_bytes();
    let (op, n) = if token.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.text.starts_with("<=") {
        (Op::Le, 2)
    } else if token.text.starts_with('>') {
        (Op::Gt, 1)
    } else if token.text.starts_with('<') {
        (Op::Lt, 1)
    } else {
        (Op::Eq, 1)
    };

    let explicit = matches!(bytes[0], b'<' | b'>' | b'=');
    let operand_start = token.start + if explicit { n } else { 0 };
    let operand = if explicit { &token.text[n..] } else { &token.text };
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let version = parse_version(operand)
        .map_err(|error| RangeError::Version { at: operand_start, error })?;
    add_cmp(arm, op, version, true);
    Ok(())
}

    }
    if upper_k == 0 {
        return Err(RangeError::Syntax { at: ts[2].start });
    }

    let lower_v = operand_version(lower);
    add_cmp(arm, Op::Ge, lower_v, true);

    match upper {
        PartialOperand::Full(v) => add_cmp(arm, Op::Le, v, true),
        PartialOperand::Partial { nums, numeric_count } => {
            let base = padded(&nums);
            let field = if numeric_count == 1 { 0 } else { 1 };
            let upper_v = upper_version(base, field, ts[2].start)?;
            add_cmp(arm, Op::Lt, upper_v, false);
        }
    }
    Ok(())
}

fn parse_arm(ts: &[Token]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    if ts.iter().any(|t| t.text == "-") {
        parse_hyphen_arm(ts, &mut arm)?;
        return Ok(arm);
    }

    for token in ts {
        let first = token.text.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_comparator(token, &mut arm)?;
        } else if first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X' | b'^' | b'~') {
            parse_expanded_token(token, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: token.start });
        }
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let ts = tokens(s);
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut expect_arm = true;
    let mut first_empty_separator = None;

    for token in ts {
        if token.text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: token.start });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            expect_arm = true;
            first_empty_separator = Some(token.start);
        } else {
            current.push(token);
            expect_arm = false;
        }
    }

    if !current.is_empty() {
        arms.push(parse_arm(&current)?);
    } else if expect_arm {
        return Err(RangeError::Empty {
            at: s.len(),
        });
    }

    let _ = first_empty_separator;
    Ok(Range { arms })
}

fn comparison_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm) -> bool {
    if !v.pre.is_empty()
        && !arm.anchors.iter().any(|&(major, minor, patch)| {
            (v.major, v.minor, v.patch) == (major, minor, patch)
        })
    {
        return false;
    }
    arm.comparators
        .iter()
        .all(|c| comparison_matches(v, c))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }

                best = Some(index);
            }
        } else {
            best = Some(index);
        }
    }

    Ok(best.map(|index| candidates[index].clone()))
}

}

pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }

    if *pos == start {
        return Err(match bytes.get(start).copied() {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }

    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &b in &bytes[start..*pos] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut id_start = start;

    loop {
        let mut id_end = id_start;
        while id_end < end && bytes[id_end] != b'.' {
            id_end += 1;
        }

        if id_end == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        for i in id_start..id_end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }

        if is_pre
            && id_end - id_start > 1
            && bytes[id_start] == b'0'
            && bytes[id_start..id_end].iter().all(|b| is_digit(*b))
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        out.push(s[id_start..id_end].to_string());
        if id_end == end {
            break;
        }
        id_start = id_end + 1;
        if id_start == end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }

    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;

    let major = parse_core_number(bytes, &mut pos)?;
    if bytes.get(pos) != Some(&b'.') {
        return Err(match bytes.get(pos).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let minor = parse_core_number(bytes, &mut pos)?;
    if bytes.get(pos) != Some(&b'.') {
        return Err(match bytes.get(pos).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let patch = parse_core_number(bytes, &mut pos)?;
    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let build_sep = bytes[pre_start..]
            .iter()
            .position(|b| *b == b'+')
            .map(|offset| pre_start + offset);
        let pre_end = build_sep.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;

        if pos < bytes.len() {
            let build_start = pos + 1;
            build = parse_identifiers(s, build_start, bytes.len(), false)?;
            pos = bytes.len();
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }

    debug_assert_eq!(pos, bytes.len());
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn compare_pre_id(a: &str, b: &str) -> Ordering {
    let an = is_numeric_identifier(a);
    let bn = is_numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ordering = compare_pre_id(x, y);
                    if ordering != Ordering::Equal {
                        return ordering;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn parse_partial_or_full(s: &str, at: usize) -> Result<Partial, RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let force_full = s.contains('-')
        || s.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f)));

    if force_full {
        let v = parse_version(s).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            fields: [v.major, v.minor, v.patch],
            numeric: 3,
            full: Some(v),
        });
    }

    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }

    let mut values = [0u64; 3];
    let mut numeric = 0;
    let mut saw_wildcard = false;
    for (i, f) in fields.iter().enumerate() {
        if is_wildcard(f) {
            saw_wildcard = true;
        } else {
            if saw_wildcard
                || !f.as_bytes().iter().all(|b| is_digit(*b))
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let Ok(n) = f.parse::<u64>() else {
                return Err(RangeError::Syntax { at });
            };
            values[i] = n;
            numeric += 1;
        }
    }

    Ok(Partial {
        fields: values,
        numeric,
        full: None,
    })
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn make_version(fields: [u64; 3], pre: Vec<String>) -> Version {
    Version {
        major: fields[0],
        minor: fields[1],
        patch: fields[2],
        pre,
        build: Vec::new(),
    }
}

fn bump(fields: [u64; 3], index: usize, at: usize) -> Result<[u64; 3], RangeError> {
    let mut out = fields;
    out[index] = out[index]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at })?;
    for x in out.iter_mut().skip(index + 1) {
        *x = 0;
    }
    Ok(out)
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(version_tuple(&version));
    }
    arm.comparisons.push(Comparison { op, version });
}

fn parse_full_comparator(token: &str, token_at: usize) -> Result<(Op, Version), RangeError> {
    let (op, skip) = if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &token[skip..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token_at });
    }
    let v = parse_version(operand)
        .map_err(|error| RangeError::Version { at: token_at + skip, error })?;
    Ok((op, v))
}

fn parse_hyphen_arm(tokens: &[(String, usize)], minus_index: usize) -> Result<Arm, RangeError> {
    if tokens.len() != 3 || minus_index != 1 || tokens[1].0 != "-" {
        return Err(RangeError::Syntax {
            at: tokens[minus_index].1,
        });
    }

    let lower = parse_partial_or_full(&tokens[0].0, tokens[0].1)?;
    let upper = parse_partial_or_full(&tokens[2].0, tokens[2].1)?;

    if lower.numeric == 0 {
        return Err(RangeError::Syntax { at: tokens[0].1 });
    }
    if upper.numeric == 0 {
        return Err(RangeError::Syntax { at: tokens[2].1 });
    }

    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };
    let low_version = lower.full.clone().unwrap_or_else(|| make_version(lower.fields, Vec::new()));
    add_cmp(&mut arm, Op::Ge, low_version, true);

    if let Some(v) = upper.full {
        add_cmp(&mut arm, Op::Le, v, true);
    } else {
        let upper_fields = match upper.numeric {
            1 => bump(upper.fields, 0, tokens[2].1)?,
            _ => bump(upper.fields, 1, tokens[2].1)?,
        };
        add_cmp(
            &mut arm,
            Op::Lt,
            make_version(upper_fields, vec!["0".to_string()]),
            false,
        );
    }
    Ok(arm)
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    if let Some(i) = tokens.iter().position(|t| t.0 == "-") {
        return parse_hyphen_arm(tokens, i);
    }

    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };

    for (token, at) in tokens {
        let first = token.as_bytes().first().copied();
        match first {
            Some(b'<' | b'>' | b'=') => {
                let (op, v) = parse_full_comparator(token, *at)?;
                add_cmp(&mut arm, op, v, true);
            }
            Some(b'^' | b'~') => {
                let caret = token.as_bytes()[0] == b'^';
                let operand = &token[1..];
                let operand_at = at + 1;
                let partial = parse_partial_or_full(operand, operand_at)?;
                if partial.numeric == 0 {
                    return Err(RangeError::Syntax { at: *at });
                }
                let lower = partial
                    .full
                    .clone()
                    .unwrap_or_else(|| make_version(partial.fields, Vec::new()));
                add_cmp(&mut arm, Op::Ge, lower, true);

                let k = partial.numeric;
                let idx = if caret {
                    if partial.fields[0] > 0 || k == 1 {
                        0
                    } else if partial.fields[1] > 0 || k == 2 {
                        1
                    } else {
                        2
                    }
                } else if k == 1 {
                    0
                } else {
                    1
                };
                let upper_fields = bump(partial.fields, idx, operand_at)?;
                add_cmp(
                    &mut arm,
                    Op::Lt,
                    make_version(upper_fields, vec!["0".to_string()]),
                    false,
                );
            }
            Some(b'*' | b'x' | b'X') | Some(b'0'..=b'9') => {
                if token.starts_with('*') || token.starts_with('x') || token.starts_with('X') {
                    let partial = parse_partial_or_full(token, *at)?;
                    if partial.numeric == 0 {
                        continue;
                    }
                    let low = make_version(partial.fields, Vec::new());
                    add_cmp(&mut arm, Op::Ge, low, true);
                    let idx = if partial.numeric == 1 { 0 } else { 1 };
                    let upper_fields = bump(partial.fields, idx, *at)?;
                    add_cmp(
                        &mut arm,
                        Op::Lt,
                        make_version(upper_fields, vec!["0".to_string()]),
                        false,
                    );
                } else {
                    let partial = parse_partial_or_full(token, *at)?;
                    if let Some(v) = partial.full {
                        add_cmp(&mut arm, Op::Eq, v, true);
                    } else if partial.numeric > 0 {
                        let low = make_version(partial.fields, Vec::new());
                        add_cmp(&mut arm, Op::Ge, low, true);
                        let idx = if partial.numeric == 1 { 0 } else { 1 };
                        let upper_fields = bump(partial.fields, idx, *at)?;
                        add_cmp(
                            &mut arm,
                            Op::Lt,
                            make_version(upper_fields, vec!["0".to_string()]),
                            false,
                        );
                    }
                }
            }
            _ => return Err(RangeError::Syntax { at: *at }),
        }
    }

    Ok(arm)
}

fn is_range_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str, start: usize, end: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    while p < end {
        while p < end && is_range_ws(bytes[p]) {
            p += 1;
        }
        if p >= end {
            break;
        }
        let token_start = p;
        while p < end && !is_range_ws(bytes[p]) {
            p += 1;
        }
        out.push((s[token_start..p].to_string(), token_start));
    }
    out
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut scan = 0;
    let mut separator_starts = Vec::new();

    while scan < bytes.len() {
        if bytes[scan] == b'|' && bytes.get(scan + 1) == Some(&b'|') {
            let tokens = tokenize(s, arm_start, scan);
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: scan });
            }
            arms.push(parse_arm(&tokens)?);
            separator_starts.push(scan);
            scan += 2;
            arm_start = scan;
        } else {
            scan += 1;
        }
    }

    let tail_tokens = tokenize(s, arm_start, bytes.len());
    if tail_tokens.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: bytes.len() });
    }
    arms.push(parse_arm(&tail_tokens)?);

    Ok(Range { arms })
}

    match c.op {
        Op::Eq => ordering == Ordering::Equal,
        Op::Lt => ordering == Ordering::Less,
        Op::Le => ordering != Ordering::Greater,
        Op::Gt => ordering == Ordering::Greater,
        Op::Ge => ordering != Ordering::Less,

    })
