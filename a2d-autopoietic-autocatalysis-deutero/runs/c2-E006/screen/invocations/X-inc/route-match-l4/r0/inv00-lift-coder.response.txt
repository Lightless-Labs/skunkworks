use std::collections::{BTreeSet, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method == method
                && same_shape(&route.segments, &segments)
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_request(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>, Vec<u8>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &parsed.segments) {
                candidates.push((route, params, rank(&route.segments)));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates.iter().any(|(r, _, _)| r.method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(r, _, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _, _) in &candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_owned());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|a, b| {
            b.2.cmp(&a.2)
                .then_with(|| a.0.id.cmp(&b.0.id))
        });
        let (route, params, _) = eligible[0];

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != parsed.trailing
            && !parsed.segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        let mut seen = HashSet::new();
        for (key, value) in params {
            if !seen.insert(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let captures: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|s| match s {
                Segment::Param(n) | Segment::Wildcard(n) => Some(n.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &captures {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let capture_set: HashSet<&str> = captures.iter().copied().collect();
        if let Some(extra) = supplied
            .keys()
            .filter(|k| !capture_set.contains(**k))
            .min_by(|a, b| a.as_bytes().cmp(b.as_bytes()))
        {
            return Err(Error::ExtraParam {
                name: (**extra).to_owned(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(n) => {
                    let value = supplied[n.as_str()];
                    if !valid_param_value(value) {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Wildcard(n) => {
                    let value = supplied[n.as_str()];
                    if !valid_wildcard_value(value) {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        if route.segments.is_empty() {
            return Ok("/".to_owned());
        }

        let mut output = String::new();
        for segment in &route.segments {
            output.push('/');
            match segment {
                Segment::Static(value) => output.push_str(&encode_segment(value)),
                Segment::Param(name) => {
                    output.push_str(&encode_segment(supplied[name.as_str()]));
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        for (i, component) in value.split('/').enumerate() {
                            if i != 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(component));
                        }
                    }
                }
            }
        }
        if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if method.is_empty() || !method.bytes().all(|b| b.is_ascii_uppercase()) {
        Err(Error::InvalidMethod)
    } else {
        Ok(())
    }
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = pattern.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|b| *b == b'/')
            .map(|i| start + i)
            .unwrap_or(end);
        if segment_end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..segment_end];
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if segment_end != end {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            let decoded = decode_segment(raw, start, true).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
        start = segment_end + 1;
    }

    if segments.is_empty() {
        return Err(Error::InvalidPattern { at: 1 });
    }
    Ok((segments, trailing))
}

fn parse_request(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|b| *b == b'/')
            .map(|i| start + i)
            .unwrap_or(end);
        if segment_end == start {
            return Err(Error::InvalidPath { at: start });
        }
        let decoded =
            decode_segment(&path[start..segment_end], start, false).map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);
        start = segment_end + 1;
    }

    if segments.is_empty() {
        return Err(Error::InvalidPath { at: 1 });
    }
    Ok(ParsedPath { segments, trailing })
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(offset + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(offset + i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(offset + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !b.is_ascii() || !is_unreserved(b) {
                return Err(offset + i);
            }
            out.push(b);
            i += 1;
        }
    }
    if pattern && raw.starts_with(':') || pattern && raw.starts_with('*') {
        return Err(offset);
    }
    Ok(String::from_utf8(out).unwrap())
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(x, y)| match (x, y) {
            (Segment::Static(a), Segment::Static(b)) => a == b,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn match_segments(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;
    for segment in pattern {
        match segment {
            Segment::Static(text) => {
                if path.get(i)? != text {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                let value = path.get(i)?;
                params.push((name.clone(), value.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[i..].join("/");
                params.push((name.clone(), value));
                i = path.len();
                break;
            }
        }
    }
    if i == path.len() {
        Some(params)
    } else {
        None
    }
}

fn rank(segments: &[Segment]) -> Vec<u8> {
    let mut result = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => result.push(2),
            Segment::Param(_) => result.push(1),
            Segment::Wildcard(_) => {
                result.push(0);
                return result;
            }
        }
    }
    result.push(3);
    result
}

fn valid_param_value(value: &str) -> bool {
    !value.is_empty()
        && value
            .bytes()
            .all(|b| (0x20..=0x7e).contains(&b) && b != b'/')
}

fn valid_wildcard_value(value: &str) -> bool {
    if value.is_empty() {
        return true;
    }
    value
        .split('/')
        .all(|part| !part.is_empty() && part.bytes().all(|b| (0x20..=0x7e).contains(&b)))
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}
