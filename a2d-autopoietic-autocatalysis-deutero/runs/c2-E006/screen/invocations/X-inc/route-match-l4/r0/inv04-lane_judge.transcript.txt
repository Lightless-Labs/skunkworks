pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },

    MissingParam { name: String },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(s: &str) -> bool {
    let mut bytes = s.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(s: &str) -> bool {
    let mut bytes = s.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(h) = hex_upper(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(l) = hex_upper(bytes[i + 2]) else {
                return Err(base + i);
            };
            let decoded = (h << 4) | l;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut end = bytes.len();
    let trailing = bytes.last() == Some(&b'/');
    if trailing {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut seen = BTreeSet::new();
    let mut start = 1;
    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(start);
        }

        let raw = &input[start..stop];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(start);
            }
            if marker == b'*' && stop < end {
                return Err(start);
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_static(raw, start)?;
            segments.push(Segment::Static(decoded));
        }

        if stop < end {
            start = stop + 1;
            if start == end {
                return Err(start);
            }
        } else {
            start = end;
        }
    }

    Ok((segments, trailing))
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn same_shape(a: &Route, b: &[Segment], trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != b.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != trailing {
        return false;
    }
    a.segments.iter().zip(b).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn parse_request(input: &str) -> Result<(Vec<String>, bool), usize> {
    let (segments, trailing) = parse_path(input, false)?;
    let decoded = segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(v) => v,
            _ => unreachable!(),
        })
        .collect();
    Ok((decoded, trailing))
}

fn encode_segment(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        if let Some(route) = self.routes.iter().find(|r| {
            r.method == method && same_shape(r, &segments, trailing, self.policy)

        let rank = make_rank(&segments);
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map_or(path.len(), |(i, _)| i);
        let head = &path[..head_end];
        let (request_segments, request_trailing) =
            parse_request(head).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            let mut params = Vec::new();
            let mut ri = 0;
            let mut matched = true;
            for segment in &route.segments {
                match segment {
                    Segment::Static(s) => {
                        if request_segments.get(ri) != Some(s) {
                            matched = false;
                            break;
                        }
                        ri += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = request_segments.get(ri) {
                            params.push((name.clone(), value.clone()));
                            ri += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = request_segments[ri..].join("/");
                        params.push((name.clone(), value));
                        ri = request_segments.len();
                    }
                }
            }
            if matched && ri == request_segments.len() {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(r, _)| r.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(r, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                a.rank
                    .cmp(&b.rank)
                    .then_with(|| b.id.cmp(&a.id))
            })
            .expect("eligible candidates are nonempty");

            && winner.trailing != request_trailing
            && !request_segments.is_empty()

            let mut target = head.to_owned();
            if winner.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params: params.clone(),
            redirect,
        })
    }

        let route = self
            .routes
            .iter()
            .find(|r| r.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied: BTreeMap<&str, &str> = BTreeMap::new();
        let mut repeated = BTreeSet::new();
        for (key, value) in params {
            if supplied.contains_key(*key) || !repeated.insert(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(n) | Segment::Wildcard(n) => capture_names.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &capture_names {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }
        if let Some(key) = supplied.keys().find(|k| !capture_names.contains(k)) {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let (name, wildcard) = match segment {
                Segment::Param(n) => (n, false),
                Segment::Wildcard(n) => (n, true),
                Segment::Static(_) => continue,

            } else if wildcard {
                value.is_empty()
                    || value
                        .split('/')
                        .all(|part| !part.is_empty())
            } else {
                !value.is_empty() && !value.contains('/')
            };
            if !valid {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut first = true;
        let mut wildcard_empty = false;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if !first {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(value));
                    first = false;
                }
                Segment::Param(name) => {
                    if !first {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(supplied[name.as_str()]));
                    first = false;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if value.is_empty() {
                        wildcard_empty = true;

                            output.push('/');

                            }
                            output.push_str(&encode_segment(part));
                        }
                        first = false;
                    }
                }
            }
        }

            output.push('/');
        }
        if output.is_empty() {
            output.push('/');
        }
        Ok(output)
    }

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn uppercase_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(segment: &str) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(i);
            }
            let (Some(hi), Some(lo)) =
                (uppercase_hex(bytes[i + 1]), uppercase_hex(bytes[i + 2]))
            else {
                return Err(i);
            };
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(i);
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn split_segments(input: &str) -> Result<(Vec<(&str, usize)>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let end = if trailing {
        input.len() - 1
    } else {
        input.len()
    };

    let mut segments = Vec::new();
    let mut start = 1;
    let bytes = input.as_bytes();

    for i in 1..end {
        if bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            segments.push((&input[start..i], start));
            start = i + 1;
        }
    }

    if start == end {
        return Err(start);
    }

    segments.push((&input[start..end], start));
    Ok((segments, trailing))
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    let (parts, trailing) =
        split_segments(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = HashSet::new();
    let mut rank = Vec::with_capacity(parts.len() + 1);

    for (index, (raw, at)) in parts.iter().enumerate() {
        if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];

            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: *at });
            }
            if !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: *at });
            }
            if wildcard && index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *at });
            }

            if wildcard {
                segments.push(Segment::Wildcard(name.to_string()));
                rank.push(0);
            } else {
                segments.push(Segment::Param(name.to_string()));
                rank.push(1);
            }
        } else {
            let decoded = decode_segment(raw)
                .map_err(|offset| Error::InvalidPattern { at: *at + offset })?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if !segments
        .iter()
        .any(|segment| matches!(segment, Segment::Wildcard(_)))
    {
        rank.push(3);
    }

    Ok((segments, trailing, rank))
}

fn parse_path_head(path: &str) -> Result<(Vec<String>, bool), Error> {
    let end = path
        .char_indices()
        .find(|(_, ch)| *ch == '?' || *ch == '#')
        .map(|(index, _)| index)
        .unwrap_or(path.len());
    let head = &path[..end];

    let (parts, trailing) =
        split_segments(head).map_err(|at| Error::InvalidPath { at })?;
    let mut decoded = Vec::with_capacity(parts.len());

    for (raw, at) in parts {
        decoded.push(
            decode_segment(raw)
                .map_err(|offset| Error::InvalidPath { at: at + offset })?,
        );
    }

    Ok((decoded, trailing))
}

fn same_shape(
    existing: &Route,
    segments: &[Segment],
    trailing: bool,
    policy: SlashPolicy,
) -> bool {
    if existing.segments.len() != segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && existing.trailing != trailing {
        return false;
    }

    existing
        .segments
        .iter()
        .zip(segments)
        .all(|(left, right)| match (left, right) {
            (Segment::Static(a), Segment::Static(b)) => a == b,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn path_candidate(
    route: &Route,
    path: &[String],
    trailing: bool,
    policy: SlashPolicy,
) -> Option<Vec<(String, String)>> {
    if policy == SlashPolicy::Strict && route.trailing != trailing {
        return None;
    }

    let wildcard = route.segments.last().and_then(|segment| match segment {
        Segment::Wildcard(name) => Some(name),
        _ => None,
    });
    let fixed_len = if wildcard.is_some() {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };

    if wildcard.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard.is_some() && path.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();

    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(value) if value == &path[i] => {}
            Segment::Static(_) => return None,
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if let Some(name) = wildcard {
        params.push((name.clone(), path[fixed_len..].join("/")));
    }

    Some(params)
}

fn encode_component(bytes: &[u8]) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut encoded = String::new();

    for &byte in bytes {
        if is_unreserved(byte) {
            encoded.push(byte as char);
        } else {
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }

    encoded
}

            "HEAD"

            "GET"
        } else {
            method

            .iter()

            .map(|(route, params)| (*route, params.clone()))
            .collect();

            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });

            b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id))

            .copied()
            .filter(|key| !captures.contains(key))
            .collect();
        extras.sort();
        if let Some(key) = extras.first() {

            let wildcard = route
                .segments
                .iter()

            let shape_valid = if wildcard {
                value.is_empty() || value.split('/').all(|part| !part.is_empty())
            } else {
                !value.is_empty() && !value.contains('/')
            };

            if !ascii || !shape_valid {
                return Err(Error::InvalidValue {
                    name: key.to_string(),
                });
            }
        }

            match segment {
                Segment::Static(value) => {
                    output.push('/');

                Segment::Param(name) => {
                    output.push('/');
                    output.push_str(&encode_component(supplied[name.as_str()].as_bytes()));

                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        for part in value.split('/') {
                            output.push('/');
                            output.push_str(&encode_component(part.as_bytes()));
                        }
                    }
                }
            }
        }

        if output.is_empty() {
            output.push('/');
        }
        if route.trailing && output != "/" && !output.ends_with('/') {
            output.push('/');
        }

        Ok(output)
    }

    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, routes: Vec::new() }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for (i, route) in self.routes.iter().enumerate() {
            if route.method == method
                && same_shape(&route.pattern, &segments)
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
            {
                return Err(Error::Conflict { with: i });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            name: name.to_owned(),
            method: method.to_owned(),
            pattern: segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path.find(['?', '#']).unwrap_or(path.len());
        let head = &path[..head_end];
        let (request, trailing) = parse_path(head)?;

        let mut candidates: Vec<(usize, Vec<(String, String)>, Vec<u8>)> = Vec::new();
        for (i, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_route(&route.pattern, &request) {
                candidates.push((i, params, rank(&route.pattern)));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates.iter().any(|(i, _, _)| self.routes[*i].method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };

        let allowed: Vec<String> = {
            let mut methods: Vec<String> = candidates
                .iter()
                .map(|(i, _, _)| self.routes[*i].method.clone())
                .collect();
            if methods.iter().any(|m| m == "GET") {
                methods.push("HEAD".to_owned());
            }
            methods.sort();
            methods.dedup();
            methods
        };

        candidates.retain(|(i, _, _)| self.routes[*i].method == eligible_method);
        if candidates.is_empty() {
            return Err(Error::MethodNotAllowed { allowed });
        }

        candidates.sort_by(|a, b| b.2.cmp(&a.2).then_with(|| a.0.cmp(&b.0)));
        let (id, params, _) = candidates.remove(0);
        let redirect = if self.policy == SlashPolicy::Redirect
            && self.routes[id].trailing != trailing
            && !request.is_empty()
        {
            let mut target = head.to_owned();
            if self.routes[id].trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match { route: id, params, redirect })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for &(key, value) in params {
            if supplied.contains_key(key) {
                return Err(Error::DuplicateParam { name: key.to_owned() });
            }
            supplied.insert(key, value);
        }

        let mut required = Vec::new();
        for seg in &route.pattern {
            match seg {
                Segment::Param(n) | Segment::Wildcard(n) => required.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam { name: (*key).to_owned() });
            }
        }

        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !required.contains(key))
            .collect();
        extras.sort();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam { name: (*key).to_owned() });
        }

        for key in &required {
            let value = supplied[key];
            let wildcard = route.pattern.iter().any(
                |seg| matches!(seg, Segment::Wildcard(n) if n == key),
            );
            if !valid_build_value(value, wildcard) {
                return Err(Error::InvalidValue { name: (*key).to_owned() });
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for seg in &route.pattern {
            if let Segment::Wildcard(name) = seg {
                let value = supplied[name.as_str()];
                if value.is_empty() {
                    break;
                }
                for component in value.split('/') {
                    if !first {
                        output.push('/');
                    }
                    first = false;
                    output.push_str(&encode_component(component));
                }
                break;
            }

            if !first {
                output.push('/');
            }
            first = false;
            match seg {
                Segment::Static(s) => output.push_str(&encode_component(s)),
                Segment::Param(name) => {
                    output.push_str(&encode_component(supplied[name.as_str()]));
                }
                Segment::Wildcard(_) => unreachable!(),
            }
        }

        if output.len() > 1 && route.trailing {
            output.push('/');
        }
        Ok(output)
    }

fn parse_segments(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut start = 1;
    let mut trailing = false;

    while start <= bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|b| *b == b'/')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if end == start {
            if end == bytes.len() {
                trailing = true;
                break;
            }
            return Err(start);
        }

        let raw = &input[start..end];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(start);
            }
            let segment = if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                if end != bytes.len() && bytes[end..].iter().any(|b| *b == b'/') {
                    return Err(start);
                }
                Segment::Wildcard(name.to_owned())
            };
            segments.push(segment);
        } else {
            let decoded = decode_segment(raw, start)?;
            segments.push(Segment::Static(decoded));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            trailing = true;
            break;
        }
    }

    Ok((segments, trailing))
}

fn decode_segment(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(offset + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(offset + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(offset + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(offset + i);
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(x, y)| match (x, y) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn match_route(pattern: &[Segment], path: &[Segment]) -> Option<Vec<(String, String)>> {
    let wildcard_index = pattern.iter().position(|s| matches!(s, Segment::Wildcard(_)));
    let fixed = wildcard_index.unwrap_or(pattern.len());
    if (wildcard_index.is_none() && path.len() != pattern.len())
        || (wildcard_index.is_some() && path.len() < fixed)
    {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed {
        match &pattern[i] {
            Segment::Static(expected) => match &path[i] {
                Segment::Static(actual) if actual == expected => {}
                _ => return None,
            },
            Segment::Param(name) => {
                let Segment::Static(value) = &path[i] else { return None };
                params.push((name.clone(), value.clone()));
            }
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_index {
        let Segment::Wildcard(name) = &pattern[index] else { unreachable!() };
        let tail = path[index..]
            .iter()
            .map(|s| match s {
                Segment::Static(value) => value.as_str(),
                _ => "",
            })
            .collect::<Vec<_>>()
            .join("/");
        params.push((name.clone(), tail));
    }
    Some(params)
}

fn rank(pattern: &[Segment]) -> Vec<u8> {
    let mut out = Vec::new();
    for seg in pattern {
        match seg {
            Segment::Static(_) => out.push(2),
            Segment::Param(_) => out.push(1),
            Segment::Wildcard(_) => {
                out.push(0);
                return out;
            }
        }
    }
    out.push(3);
    out
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
        return false;
    }
    if wildcard {
        if value.is_empty() {
            return true;
        }
        value
            .split('/')
            .all(|part| !part.is_empty() && !part.contains('/'))
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn encode_component(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}
