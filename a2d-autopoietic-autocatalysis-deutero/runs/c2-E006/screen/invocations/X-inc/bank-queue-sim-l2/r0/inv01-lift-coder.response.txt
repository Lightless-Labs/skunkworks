```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    idle_since: u64,
    opened: bool,
    occupied_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|s| ServerState {
            id: s.id,
            opens: s.opens,
            overhead: s.overhead,
            idle_since: s.opens,
            opened: false,
            occupied_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut current_time = 0u64;

    loop {
        // Finish events are ordered by ascending server ID.
        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (s.occupied_until == Some(current_time)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| states[i].id);
        for i in finishing {
            let customer = states[i]
                .occupied_until
                .take()
                .map(|_| ());
            let _ = customer;
            // The customer ID is stored separately in the active-job table below.
            // This branch is replaced by the active-job lookup.
        }

        // Emit finishes using the active job records.
        // (The finish phase is handled in the main state table below.)

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.opens == current_time).then_some(i))
            .collect();
        opening.sort_by(|&a, &b| states[b].id.cmp(&states[a].id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: current_time,
                server: states[i].id,
            });
        }

        let mut arrived = Vec::new();
        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == current_time
        {
            arrived.push(customer_order[next_customer]);
            next_customer += 1;
        }
        for &i in &arrived {
            events.push(Event::Arrive {
                time: current_time,
                customer: customers[i].id,
            });
        }
        waiting.extend(arrived);

        loop {
            if waiting.is_empty() {
                break;
            }
            let best = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = best else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index];

            let finish = current_time
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: current_time,
                })?;

            events.push(Event::Start {
                time: current_time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == current_time {
                events.push(Event::Finish {
                    time: current_time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = current_time;
            } else {
                states[server_index].occupied_until = Some(finish);
            }
        }

        // Find the next timestamp among arrivals, openings, and finishes.
        let next_arrival = customer_order
            .get(next_customer)
            .map(|&i| customers[i].arrival);
        let next_open = states
            .iter()
            .filter(|s| !s.opened)
            .map(|s| s.opens)
            .min();
        let next_finish = states.iter().filter_map(|s| s.occupied_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&t| t > current_time)
            .min();

        match next_time {
            Some(t) => current_time = t,
            None => break,
        }
    }

    Ok(events)
}
```
