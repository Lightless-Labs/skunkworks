```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|s| ServerState {
            id: s.id,
            opens: s.opens,
            overhead: s.overhead,
            opened: false,
            busy_until: None,
            idle_since: s.opens,
        })
        .collect();

    let mut waiting: Vec<Customer> = Vec::new();
    let mut next_arrival = 0usize;
    let mut events = Vec::new();
    let mut now = 0u64;

    loop {
        let mut finishing: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| {
                s.busy_until
                    .filter(|&t| t == now)
                    .map(|_| {
                        let customer = match events.iter().rev().find_map(|e| match e {
                            Event::Start {
                                customer,
                                server,
                                finish,
                                ..
                            } if *server == s.id && *finish == now => Some(*customer),
                            _ => None,
                        }) {
                            Some(id) => id,
                            None => 0,
                        };
                        (s.id, i, customer)
                    })
            })
            .collect();
        finishing.sort_by_key(|x| x.0);

        for (server_id, index, customer_id) in finishing {
            events.push(Event::Finish {
                time: now,
                customer: customer_id,
                server: server_id,
            });
            states[index].busy_until = None;
            states[index].idle_since = now;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.opens == now).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[index].id,
            });
        }

        let begin = next_arrival;
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            next_arrival += 1;
        }
        for customer in arrivals[begin..next_arrival].iter() {
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(customer.clone());
        }

        loop {
            if waiting.is_empty() {
                break;
            }
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };
            let customer = waiting.remove(0);
            let server_id = states[server_index].id;
            let finish = now
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(states[server_index].overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].busy_until = None;
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }

        if next_arrival == arrivals.len()
            && states.iter().all(|s| s.opened)
            && states.iter().all(|s| s.busy_until.is_none())
            && waiting.is_empty()
        {
            break;
        }

        let next = arrivals
            .get(next_arrival)
            .map(|c| c.arrival)
            .into_iter()
            .chain(
                states
                    .iter()
                    .filter(|s| !s.opened)
                    .map(|s| s.opens),
            )
            .chain(states.iter().filter_map(|s| s.busy_until))
            .filter(|&t| t > now)
            .min();

        match next {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(events)
}
```
