```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        if !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        for (i, component) in path[1..].split('/').enumerate() {
            let offset = start;
            if component.is_empty() || component == "." || component == ".." {
                let error_offset = if component.is_empty() {
                    if offset == path.len() {
                        path.len() - 1
                    } else {
                        offset
                    }
                } else {
                    offset
                };
                return Err(Error::InvalidPath(error_offset));
            }
            components.push(component);
            start += component.len() + 1;
            let _ = i;
        }
        Ok(components)
    }

    fn prefix(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn lookup<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut traversed = Vec::new();
        for (index, component) in components.iter().enumerate() {
            match current {
                Entry::Dir(children) => {
                    traversed.push(*component);
                    current = children
                        .get(*component)
                        .ok_or_else(|| Error::Missing(Self::prefix(&traversed)))?;
                }
                Entry::File(_) => return Err(Error::NotDir(Self::prefix(&traversed))),
            }
            if index + 1 < components.len() && matches!(current, Entry::File(_)) {
                return Err(Error::NotDir(Self::prefix(&traversed)));
            }
        }
        Ok(current)
    }

    fn parent_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut current = &mut self.root;
        let mut traversed = Vec::new();
        for component in components {
            match current {
                Entry::Dir(children) => {
                    traversed.push(*component);
                    current = children
                        .get_mut(*component)
                        .ok_or_else(|| Error::Missing(Self::prefix(&traversed)))?;
                }
                Entry::File(_) => return Err(Error::NotDir(Self::prefix(&traversed))),
            }
        }
        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::prefix(components))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let name = components[components.len() - 1];
        let parent = self.parent_mut(parent_components)?;
        let canonical = Self::prefix(&components);
        if parent.contains_key(name) {
            return Err(Error::Exists(canonical));
        }
        parent.insert(name.to_owned(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let name = components[components.len() - 1];
        let parent = self.parent_mut(parent_components)?;
        match parent.get_mut(name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(Self::prefix(&components))),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name.to_owned(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        match self.lookup(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::prefix(&components))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        match self.lookup(&components)? {
            Entry::File(_) => Err(Error::NotDir(Self::prefix(&components))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
