#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn append(word: &mut Word, value: &str, wildcards_active: bool) {
    for c in value.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn append_literal(word: &mut Word, c: char, wildcards_active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if wildcards_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut iter = input.char_indices().peekable();

    while let Some((pos, c)) = iter.next() {
        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                continue;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    continue;
                }
                if c == '\\' {
                    let Some(&(next_pos, next)) = iter.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        iter.next();
                        if next != '\n' {
                            append_literal(&mut word, next, false);
                        }
                    } else {
                        append_literal(&mut word, '\\', false);
                    }
                    continue;
                }
                if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                }
                if c == '$' {
                    if let Some(&(next_pos, '(')) = iter.peek() {
                        let _ = next_pos;
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if let Some(&(next_pos, next)) = iter.peek() {
                        if is_name_start(next) {
                            active = true;
                            let mut name = String::new();
                            name.push(next);
                            iter.next();
                            while let Some(&(_, ch)) = iter.peek() {
                                if is_name_char(ch) {
                                    name.push(ch);
                                    iter.next();
                                } else {
                                    break;
                                }
                            }
                            if let Some(value) = lookup(&name, env) {
                                append(&mut word, value, false);
                            }
                            let _ = next_pos;
                            continue;
                        } else if next == '{' {
                            active = true;
                            let expansion =
                                parse_braced(input, pos, env, &mut iter)?;
                            append(&mut word, &expansion, false);
                            continue;
                        }
                    }
                    active = true;
                    append_literal(&mut word, '$', false);
                    continue;
                }
                append_literal(&mut word, c, false);
                continue;
            }
            Quote::None => {}
        }

        match c {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(word);
                    word = Word {
                        text: String::new(),
                        glob: Vec::new(),
                    };
                    active = false;
                }
            }
            '\'' => {
                active = true;
                quote = Quote::Single(pos);
            }
            '"' => {
                active = true;
                quote = Quote::Double(pos);
            }
            '\\' => {
                let Some(&(_, next)) = iter.peek() else {
                    return Err(SplitError::TrailingEscape { pos });
                };
                iter.next();
                if next != '\n' {
                    active = true;
                    append_literal(&mut word, next, false);
                }
            }
            '`' => return Err(SplitError::CommandSubstitution { pos }),
            '$' => {
                if let Some(&(_, '(')) = iter.peek() {
                    return Err(SplitError::CommandSubstitution { pos });
                }
                if let Some(&(_, next)) = iter.peek() {
                    if is_name_start(next) {
                        active = true;
                        let mut name = String::new();
                        name.push(next);
                        iter.next();
                        while let Some(&(_, ch)) = iter.peek() {
                            if is_name_char(ch) {
                                name.push(ch);
                                iter.next();
                            } else {
                                break;
                            }
                        }
                        if let Some(value) = lookup(&name, env) {
                            append(&mut word, value, true);
                        }
                        continue;
                    } else if next == '{' {
                        active = true;
                        let expansion = parse_braced(input, pos, env, &mut iter)?;
                        append(&mut word, &expansion, true);
                        continue;
                    }
                }
                active = true;
                append_literal(&mut word, '$', true);
            }
            _ => {
                active = true;
                append_literal(&mut word, c, true);
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
    iter: &mut std::iter::Peekable<std::str::CharIndices<'_>>,
) -> Result<String, SplitError> {
    // Consume the opening brace.
    iter.next();

    let content_start = iter.peek().map(|(p, _)| *p).unwrap_or(input.len());
    let mut escaped = false;
    let mut close_pos = None;

    while let Some(&(pos, c)) = iter.peek() {
        if escaped {
            escaped = false;
            iter.next();
            continue;
        }
        if c == '\\' {
            escaped = true;
            iter.next();
            continue;
        }
        if c == '}' {
            close_pos = Some(pos);
            break;
        }
        iter.next();
    }

    let Some(end) = close_pos else {
        return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
    };

    let contents = &input[content_start..end];
    let (name, default) = if let Some(idx) = contents.find(":-") {
        (&contents[..idx], Some(&contents[idx + 2..]))
    } else {
        (contents, None)
    };

    let mut chars = name.chars();
    let valid = chars.next().is_some_and(is_name_start) && chars.all(is_name_char);
    if !valid {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    // Advance over closing brace.
    iter.next();

    let value = lookup(name, env).unwrap_or("");
    match default {
        None => Ok(value.to_string()),
        Some(text) if !value.is_empty() => Ok(value.to_string()),
        Some(text) => Ok(decode_default(text)),
    }
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        return None;
    }
    Some(format!("'{}'", word))
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut result = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

fn fallback_candidate(word: &str) -> String {
    let mut result = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            result.push('\'');
            result.push('\\');
            result.push('\'');
            result.push('\'');
        } else {
            result.push(c);
        }
    }
    result.push('\'');
    result
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        let selected = candidates
            .into_iter()
            .enumerate()
            .filter_map(|(priority, candidate)| {
                candidate.map(|candidate| (candidate.len(), priority, candidate))
            })
            .min_by_key(|(len, priority, _)| (*len, *priority))
            .map(|(_, _, candidate)| candidate)
            .unwrap_or_else(|| fallback_candidate(word));
        encoded.push(selected);
    }
    Ok(encoded.join(" "))
}
