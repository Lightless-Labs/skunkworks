use std::fmt::Write;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_text(word: &mut Word, text: &str, active_glob: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let mut cursor = dollar + 2;
    let mut close = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += 1;
            if cursor < input.len() {
                cursor += input[cursor..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[dollar + 2..close];

    let chars: Vec<(usize, char)> = contents.char_indices().collect();
    if chars.is_empty() || !is_name_start(chars[0].1) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let mut name_end = chars[0].0 + chars[0].1.len_utf8();
    let mut i = 1;
    while i < chars.len() && is_name_char(chars[i].1) {
        name_end = chars[i].0 + chars[i].1.len_utf8();
        i += 1;
    }

    let name = &contents[..name_end];
    let suffix = &contents[name_end..];

    let value = if suffix.is_empty() {
        lookup(env, name).unwrap_or("").to_owned()
    } else if let Some(default) = suffix.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_owned(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let len = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_text(&mut current, &input[cursor..cursor + len], false);
                }
                cursor += len;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    cursor += len;
                } else if c == '\\' {
                    let slash = cursor;
                    cursor += 1;
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_text(
                                &mut current,
                                &input[cursor..cursor + next.len_utf8()],
                                false,
                            );
                        }
                        cursor += next.len_utf8();
                    } else {
                        append_text(&mut current, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    let next_pos = cursor + 1;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_pos..].starts_with('{') {
                        let (value, end) = parse_braced(input, cursor, env)?;
                        append_text(&mut current, &value, false);
                        active = true;
                        cursor = end;
                    } else if next_pos < input.len()
                        && is_name_start(input[next_pos..].chars().next().unwrap())
                    {
                        let mut end = next_pos;
                        while end < input.len() {
                            let next = input[end..].chars().next().unwrap();
                            if !is_name_char(next) {
                                break;
                            }
                            end += next.len_utf8();
                        }
                        let value = lookup(env, &input[next_pos..end]).unwrap_or("");
                        append_text(&mut current, value, false);
                        active = true;
                        cursor = end;
                    } else {
                        append_text(&mut current, "$", false);
                        active = true;
                        cursor += 1;
                    }
                } else {
                    append_text(&mut current, &input[cursor..cursor + len], false);
                    cursor += len;
                }
                let _ = open;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    cursor += len;
                } else if c == '\'' {
                    quote = Quote::Single(cursor);
                    active = true;
                    cursor += len;
                } else if c == '"' {
                    quote = Quote::Double(cursor);
                    active = true;
                    cursor += len;
                } else if c == '\\' {
                    let slash = cursor;
                    cursor += 1;
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    if next != '\n' {
                        append_text(
                            &mut current,
                            &input[cursor..cursor + next.len_utf8()],
                            false,
                        );
                        active = true;
                    }
                    cursor += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    let next_pos = cursor + 1;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_pos..].starts_with('{') {
                        let (value, end) = parse_braced(input, cursor, env)?;
                        append_text(&mut current, &value, true);
                        active = true;
                        cursor = end;
                    } else if next_pos < input.len()
                        && is_name_start(input[next_pos..].chars().next().unwrap())
                    {
                        let mut end = next_pos;
                        while end < input.len() {
                            let next = input[end..].chars().next().unwrap();
                            if !is_name_char(next) {
                                break;
                            }
                            end += next.len_utf8();
                        }
                        let value = lookup(env, &input[next_pos..end]).unwrap_or("");
                        append_text(&mut current, value, true);
                        active = true;
                        cursor = end;
                    } else {
                        append_text(&mut current, "$", false);
                        active = true;
                        cursor += 1;
                    }
                } else {
                    append_text(&mut current, &input[cursor..cursor + len], true);
                    active = true;
                    cursor += len;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        return None;
    }

    let mut out = String::with_capacity(word.len() + 2);
    out.push('\'');
    out.push_str(word);
    out.push('\'');
    Some(out)
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }

    let mut out = String::with_capacity(word.len() + 2);
    out.push('"');
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::new();
    out.push('\'');
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate, 0usize));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate, 1usize));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate, 2usize));
        }

        let chosen = if candidates.is_empty() {
            fallback_candidate(word)
        } else {
            candidates
                .into_iter()
                .min_by_key(|(candidate, priority)| (candidate.len(), *priority))
                .unwrap()
                .0
        };
        encoded.push(chosen);
    }

    let mut result = String::new();
    for (i, word) in encoded.iter().enumerate() {
        if i != 0 {
            result.push(' ');
        }
        result.push_str(word);
    }
    Ok(result)
}
