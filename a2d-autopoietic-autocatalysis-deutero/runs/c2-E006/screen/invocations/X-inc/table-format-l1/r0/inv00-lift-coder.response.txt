```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged {
        row: usize,
        expected: usize,
        actual: usize,
    },
    InvalidChar {
        row: usize,
        column: usize,
        byte: usize,
    },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let expected = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != expected {
            return Err(TableError::Ragged {
                row,
                expected,
                actual: cells.len(),
            });
        }
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, character) in cell.char_indices() {
                if !(' '..='~').contains(&character) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|cells| {
            cells
                .iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let widths: Vec<usize> = (0..expected)
        .map(|column| {
            trimmed
                .iter()
                .map(|row| row[column].len())
                .max()
                .unwrap_or(0)
                .max(1)
        })
        .collect();

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut output = String::new();

    let append_border = |output: &mut String| {
        output.push('+');
        for &width in &widths {
            output.extend(std::iter::repeat_n('-', width + 2));
            output.push('+');
        }
        output.push('\n');
    };

    append_border(&mut output);

    for row in &trimmed {
        output.push('|');
        for (column, content) in row.iter().enumerate() {
            let width = widths[column];
            output.push(' ');
            output.push_str(content);
            output.extend(std::iter::repeat_n(' ', width - content.len()));
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    append_border(&mut output);
    Ok(output)
}
```
