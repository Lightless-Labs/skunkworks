```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len().saturating_mul(3));
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut occupied: Option<(u64, u32)> = None;
    let mut has_dispatched = false;
    let mut previous_finish = 0u64;

    while next_arrival < arrivals.len() || occupied.is_some() || !waiting.is_empty() {
        let next_arrival_time = arrivals
            .get(next_arrival)
            .map(|customer| customer.arrival);

        let tick = match (occupied, next_arrival_time) {
            (Some((finish, _)), Some(arrival)) => finish.min(arrival),
            (Some((finish, _)), None) => finish,
            (None, Some(arrival)) => arrival,
            (None, None) => break,
        };

        if let Some((finish, customer_id)) = occupied {
            if finish == tick {
                events.push(Event {
                    time: tick,
                    customer: customer_id,
                    kind: Kind::Finish,
                });
                occupied = None;
                previous_finish = tick;
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == tick {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if occupied.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = !has_dispatched || tick > previous_finish;
                let service_finish = tick.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: tick,
                    },
                )?;
                let finish = service_finish.checked_add(if wake { wake_cost } else { 0 }).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: tick,
                    },
                )?;

                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });
                has_dispatched = true;

                if finish == tick {
                    events.push(Event {
                        time: tick,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = tick;
                } else {
                    occupied = Some((finish, customer.id));
                }
            }
        }
    }

    Ok(events)
}
```
