```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn canonicalize(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();
        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }
        if start == end {
            return Err(Error::EmptyKey);
        }

        for i in start..end {
            let b = bytes[i];
            let valid = if i == start {
                b.is_ascii_alphabetic()
            } else {
                b.is_ascii_alphanumeric() || b == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: i });
            }
        }

        Ok(input[start..end].to_ascii_lowercase())
    }

    fn cmp_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn swap_slots(&mut self, a: usize, b: usize) {
        self.entries.swap(a, b);
        let key_a = self.entries[a].item.key.clone();
        let key_b = self.entries[b].item.key.clone();
        self.indices.insert(key_a, a);
        self.indices.insert(key_b, b);
    }

    fn sift_up(&mut self, mut i: usize) {
        while i > 0 {
            let parent = (i - 1) / 2;
            if Self::cmp_entries(&self.entries[i], &self.entries[parent]).is_lt() {
                self.swap_slots(i, parent);
                i = parent;
            } else {
                break;
            }
        }
    }

    fn sift_down(&mut self, mut i: usize) {
        let n = self.entries.len();
        loop {
            let left = i * 2 + 1;
            if left >= n {
                break;
            }
            let right = left + 1;
            let mut best = left;
            if right < n && Self::cmp_entries(&self.entries[right], &self.entries[left]).is_lt() {
                best = right;
            }
            if Self::cmp_entries(&self.entries[best], &self.entries[i]).is_lt() {
                self.swap_slots(i, best);
                i = best;
            } else {
                break;
            }
        }
    }

    fn repair(&mut self, i: usize) {
        if i > 0 && Self::cmp_entries(&self.entries[i], &self.entries[(i - 1) / 2]).is_lt() {
            self.sift_up(i);
        } else {
            self.sift_down(i);
        }
    }

    fn remove_at(&mut self, i: usize) -> Entry {
        let removed_key = self.entries[i].item.key.clone();
        self.indices.remove(&removed_key);
        let last = self.entries.len() - 1;
        if i != last {
            self.entries.swap(i, last);
            let moved_key = self.entries[i].item.key.clone();
            self.indices.insert(moved_key, i);
        }
        let removed = self.entries.pop().expect("valid heap slot");
        if i < self.entries.len() {
            self.repair(i);
        }
        removed
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = Self::canonicalize(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        let age = self.next_age;
        self.next_age = self.next_age.wrapping_add(1);
        let i = self.entries.len();
        self.entries.push(Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age,
        });
        self.indices.insert(key, i);
        self.sift_up(i);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|e| e.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            None
        } else {
            Some(self.remove_at(0).item)
        }
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonicalize(key)?;
        Ok(self.indices.get(&key).map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonicalize(key)?;
        let Some(&i) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[i].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[i].item.priority = priority;
        self.repair(i);
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonicalize(key)?;
        let Some(&i) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[i].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[i].item.priority = priority;
        self.repair(i);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonicalize(key)?;
        let Some(&i) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let old = std::mem::replace(&mut self.entries[i].item.value, value.to_owned());
        self.repair(i);
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonicalize(key)?;
        let Some(&i) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.remove_at(i).item)
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source_order: Vec<Entry> = other.entries.iter().cloned().collect();
        source_order.sort_by(Self::cmp_entries);

        if policy == CollisionPolicy::Reject {
            for entry in &source_order {
                if self.indices.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
        };

        for source in source_order {
            let key = source.item.key.clone();
            if let Some(&i) = self.indices.get(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[i].item.priority = source.item.priority;
                        self.entries[i].item.value = source.item.value;
                        self.repair(i);
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.next_age;
                self.next_age = self.next_age.wrapping_add(1);
                let i = self.entries.len();
                self.entries.push(Entry {
                    item: source.item,
                    age,
                });
                self.indices.insert(key, i);
                self.sift_up(i);
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.indices.clear();
        other.next_age = 0;
        Ok(report)
    }
}
```
