use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn missing_core_byte(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    if !bytes.get(*pos).is_some_and(u8::is_ascii_digit) {
        return if missing_core_byte(bytes.get(*pos).copied()) {
            Err(VersionError::MissingCore { at: start })
        } else {
            Err(VersionError::InvalidChar { at: start })
        };
    }

    while bytes.get(*pos).is_some_and(u8::is_ascii_digit) {
        *pos += 1;
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }

    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &s[ident_start..i];
        if prerelease
            && ident.len() > 1
            && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        result.push(ident.to_owned());
        if i == end {
            break;
        }
        ident_start = i + 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return if missing_core_byte(s.as_bytes().get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return if missing_core_byte(s.as_bytes().get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    match s.as_bytes().get(pos).copied() {
        None => {
            return Ok(Version {
                major,
                minor,
                patch,
                pre: Vec::new(),
                build: Vec::new(),
            });
        }
        Some(b'-' | b'+') => {}
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if s.as_bytes()[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = s.as_bytes()[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pre_start + offset);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        if let Some(plus_pos) = plus {
            let build_start = plus_pos + 1;
            build = parse_identifiers(s, build_start, s.len(), false)?;
        }
    } else {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let order = compare_identifiers(x, y);
                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        let token_end = pos;

        let (operator, operand_start) = match bytes[token_start] {
            b'=' => (Operator::Equal, token_start + 1),
            b'<' if bytes.get(token_start + 1) == Some(&b'=') => {
                (Operator::LessEqual, token_start + 2)
            }
            b'<' => (Operator::Less, token_start + 1),
            b'>' if bytes.get(token_start + 1) == Some(&b'=') => {
                (Operator::GreaterEqual, token_start + 2)
            }
            b'>' => (Operator::Greater, token_start + 1),
            b'0'..=b'9' => (Operator::Equal, token_start),
            _ => return Err(RangeError::Syntax { at: token_start }),
        };

        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let version = parse_version(&s[operand_start..token_end]).map_err(|error| {
            RangeError::Version {
                at: operand_start,
                error,
            }
        })?;
        comparators.push(Comparator { operator, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { comparators })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let candidate_core = (v.major, v.minor, v.patch);

    if !v.pre.is_empty()
        && !r.comparators.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && (
                    comparator.version.major,
                    comparator.version.minor,
                    comparator.version.patch,
                ) == candidate_core
        })
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Equal => ordering == Ordering::Equal,
            Operator::Less => ordering == Ordering::Less,
            Operator::LessEqual => ordering != Ordering::Greater,
            Operator::Greater => ordering == Ordering::Greater,
            Operator::GreaterEqual => ordering != Ordering::Less,
        }
    })
}
