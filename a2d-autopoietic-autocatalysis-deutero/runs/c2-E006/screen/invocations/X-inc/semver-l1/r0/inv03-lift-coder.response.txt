use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(b: u8) -> bool {
    matches!(b, b'.' | b'-' | b'+')
}

fn parse_core_number(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let mut end = start;
    while end < bytes.len() && bytes[end].is_ascii_digit() {
        end += 1;
    }

    if end == start {
        if start == bytes.len() || is_core_missing_byte(bytes[start]) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let digits = &bytes[start..end];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value: u64 = 0;
    for &digit in digits {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }

    *pos = end;
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut ident_end = ident_start;
        while ident_end < end && bytes[ident_end] != b'.' {
            let b = bytes[ident_end];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: ident_end });
            }
            ident_end += 1;
        }

        if ident_end == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        if prerelease {
            let ident = &bytes[ident_start..ident_end];
            if ident.iter().all(u8::is_ascii_digit)
                && ident.len() > 1
                && ident[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
        }

        result.push(s[ident_start..ident_end].to_owned());

        if ident_end == end {
            break;
        }
        ident_start = ident_end + 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;

    let major = parse_core_number(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos >= bytes.len() || is_core_missing_byte(bytes[pos]) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core_number(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos >= bytes.len() || is_core_missing_byte(bytes[pos]) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core_number(bytes, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let plus = bytes[pre_start..]
                    .iter()
                    .position(|&b| b == b'+')
                    .map(|offset| pre_start + offset);
                let pre_end = plus.unwrap_or(bytes.len());
                pre = parse_identifiers(s, pre_start, pre_end, true)?;
                pos = pre_end;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    if pos < bytes.len() {
        if bytes[pos] != b'+' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len()
        .cmp(&b.len())
        .then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.iter().zip(b.iter()) {
        let left_numeric = is_numeric_identifier(left);
        let right_numeric = is_numeric_identifier(right);

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => compare_numeric_identifiers(left, right),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CompareOp,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum CompareOp {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    prerelease_cores: Vec<(u64, u64, u64)>,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();
    let mut prerelease_cores = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let (op, operand_start) = match bytes[token_start] {
            b'=' => (CompareOp::Eq, token_start + 1),
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                (CompareOp::Le, token_start + 2)
            }
            b'<' => (CompareOp::Lt, token_start + 1),
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                (CompareOp::Ge, token_start + 2)
            }
            b'>' => (CompareOp::Gt, token_start + 1),
            b'0'..=b'9' => (CompareOp::Eq, token_start),
            _ => return Err(RangeError::Syntax { at: token_start }),
        };

        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            prerelease_cores.push((version.major, version.minor, version.patch));
        }

        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        comparators,
        prerelease_cores,
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .prerelease_cores
            .iter()
            .any(|core| *core == (v.major, v.minor, v.patch))
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.op {
            CompareOp::Eq => ordering == Ordering::Equal,
            CompareOp::Lt => ordering == Ordering::Less,
            CompareOp::Le => ordering != Ordering::Greater,
            CompareOp::Gt => ordering == Ordering::Greater,
            CompareOp::Ge => ordering != Ordering::Less,
        }
    })
}
