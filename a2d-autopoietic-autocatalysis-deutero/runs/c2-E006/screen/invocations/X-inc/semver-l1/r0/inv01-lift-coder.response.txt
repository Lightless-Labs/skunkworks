use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut core = [0u64; 3];

    for (field, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        let digits = &bytes[start..pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut number = 0u64;
        for &digit in digits {
            number = number
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *value = number;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let plus = bytes[pos..].iter().position(|&b| b == b'+').map(|i| pos + i);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(bytes, pre_start, pre_end, true)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(bytes, pos, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut identifiers = Vec::new();
    let mut ident_start = start;

    loop {
        let ident_end = (ident_start..end)
            .find(|&i| bytes[i] == b'.')
            .unwrap_or(end);

        if ident_start == ident_end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        for i in ident_start..ident_end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }

        let ident = &bytes[ident_start..ident_end];
        if prerelease
            && ident.len() > 1
            && ident[0] == b'0'
            && ident.iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        identifiers.push(String::from_utf8(ident.to_vec()).expect("ASCII identifier"));

        if ident_end == end {
            break;
        }
        ident_start = ident_end + 1;
    }

    Ok(identifiers)
}

fn is_numeric_identifier(identifier: &str) -> bool {
    !identifier.is_empty() && identifier.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (left, right) in a.pre.iter().zip(&b.pre) {
                    let left_numeric = is_numeric_identifier(left);
                    let right_numeric = is_numeric_identifier(right);
                    let ordering = match (left_numeric, right_numeric) {
                        (true, true) => compare_numeric_identifiers(left, right),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => left.as_bytes().cmp(right.as_bytes()),
                    };
                    if ordering != Ordering::Equal {
                        return ordering;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Operator,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let mut operand_start = token_start;
        let op = match bytes[token_start] {
            b'=' => {
                operand_start += 1;
                Operator::Equal
            }
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Operator::LessEqual
            }
            b'<' => {
                operand_start += 1;
                Operator::Less
            }
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Operator::GreaterEqual
            }
            b'>' => {
                operand_start += 1;
                Operator::Greater
            }
            _ => Operator::Equal,
        };

        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted = v.pre.is_empty()
            || arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && comparator.version.major == v.major
                    && comparator.version.minor == v.minor
                    && comparator.version.patch == v.patch
            });
        if !admitted {
            return false;
        }

        arm.iter().all(|comparator| {
            let ordering = compare_versions(v, &comparator.version);
            match comparator.op {
                Operator::Equal => ordering == Ordering::Equal,
                Operator::Less => ordering == Ordering::Less,
                Operator::LessEqual => ordering != Ordering::Greater,
                Operator::Greater => ordering == Ordering::Greater,
                Operator::GreaterEqual => ordering != Ordering::Less,
            }
        })
    })
}
