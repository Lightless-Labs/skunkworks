#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];
const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];
const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut n = magnitude;
    while n > 0 {
        groups.push((n % 1000) as u16);
        n /= 1000;
    }

    let mut parts = Vec::new();
    for i in (0..groups.len()).rev() {
        let group = groups[i];
        if group == 0 {
            continue;
        }
        let mut spelling = spell_group(group);
        if i > 0 {
            spelling.push(' ');
            spelling.push_str(SCALES[i - 1]);
        }
        parts.push((i, group, spelling));
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }
    for (idx, (scale_index, group, spelling)) in parts.iter().enumerate() {
        if idx > 0 {
            result.push(' ');
            let is_units = *scale_index == 0;
            let has_higher = parts[..idx].iter().any(|(i, _, _)| *i > 0);
            if is_units && has_higher && *group < 100 {
                result.push_str("and ");
            }
        }
        result.push_str(spelling);
    }
    Ok(result)
}

fn spell_group(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }
    if n < 100 {
        let tens = n / 10;
        let units = n % 10;
        if units == 0 {
            TENS[tens as usize].to_string()
        } else {
            format!("{}-{}", TENS[tens as usize], SMALL[units as usize])
        }
    } else {
        let hundreds = n / 100;
        let remainder = n % 100;
        if remainder == 0 {
            format!("{} hundred", SMALL[hundreds as usize])
        } else {
            format!("{} hundred and {}", SMALL[hundreds as usize], spell_group(remainder))
        }
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_lowercase() || b == b' ' || b == b'-') {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<(usize, &str)> = Vec::new();
    let mut start = 0;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            tokens.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    tokens.push((start, &input[start..]));

    for (i, (pos, token)) in tokens.iter().enumerate() {
        if *token == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: *pos });
            }
        } else if !is_dictionary_word(token) {
            return Err(WordsError::InvalidWord { index: *pos });
        }
    }

    let negative = tokens.first().is_some_and(|(_, token)| *token == "minus");
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for (_, token) in arithmetic {
        match *token {
            "zero" => {}
            "one" => group += 1,
            "two" => group += 2,
            "three" => group += 3,
            "four" => group += 4,
            "five" => group += 5,
            "six" => group += 6,
            "seven" => group += 7,
            "eight" => group += 8,
            "nine" => group += 9,
            "ten" => group += 10,
            "eleven" => group += 11,
            "twelve" => group += 12,
            "thirteen" => group += 13,
            "fourteen" => group += 14,
            "fifteen" => group += 15,
            "sixteen" => group += 16,
            "seventeen" => group += 17,
            "eighteen" => group += 18,
            "nineteen" => group += 19,
            "twenty" => group += 20,
            "thirty" => group += 30,
            "forty" => group += 40,
            "fifty" => group += 50,
            "sixty" => group += 60,
            "seventy" => group += 70,
            "eighty" => group += 80,
            "ninety" => group += 90,
            "hundred" => group *= 100,
            "thousand" => {
                total += group * 1_000;
                group = 0;
            }
            "million" => {
                total += group * 1_000_000;
                group = 0;
            }
            "billion" => {
                total += group * 1_000_000_000;
                group = 0;
            }
            "trillion" => {
                total += group * 1_000_000_000_000;
                group = 0;
            }
            "quadrillion" => {
                total += group * 1_000_000_000_000_000;
                group = 0;
            }
            "quintillion" => {
                total += group * LIMIT;
                group = 0;
            }
            "and" => {}
            _ => unreachable!(),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(value)
}

fn is_dictionary_word(word: &str) -> bool {
    matches!(
        word,
        "zero"
            | "one"
            | "two"
            | "three"
            | "four"
            | "five"
            | "six"
            | "seven"
            | "eight"
            | "nine"
            | "ten"
            | "eleven"
            | "twelve"
            | "thirteen"
            | "fourteen"
            | "fifteen"
            | "sixteen"
            | "seventeen"
            | "eighteen"
            | "nineteen"
            | "twenty"
            | "thirty"
            | "forty"
            | "fifty"
            | "sixty"
            | "seventy"
            | "eighty"
            | "ninety"
            | "hundred"
            | "thousand"
            | "million"
            | "billion"
            | "trillion"
            | "quadrillion"
            | "quintillion"
            | "and"
            | "minus"
    )
}
