#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "thousand",
    "million",
    "billion",
    "trillion",
    "quadrillion",
    "quintillion",
    "",
];

fn append_two(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_two(rem, out);
        }
    } else {
        append_two(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut groups = [0u16; 7];
    let mut rest = magnitude;
    for group in groups.iter_mut().rev() {
        *group = (rest % 1000) as u16;
        rest /= 1000;
    }

    let highest_present = groups.iter().position(|&g| g != 0).unwrap();
    let has_higher = groups[..6].iter().any(|&g| g != 0);
    let units = groups[6];

    for i in highest_present..7 {
        let n = groups[i];
        if n == 0 {
            continue;
        }
        if !out.is_empty() && out != "minus " {
            out.push(' ');
        }

        if i == 6 && has_higher && units < 100 {
            out.push_str("and ");
        }

        append_group(n, &mut out);
        if i < 6 {
            out.push(' ');
            out.push_str(SCALES[5 - i]);
        }
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }
    for (i, w) in SMALL.iter().enumerate().skip(1) {
        if word == *w {
            return Some(i as u128);
        }
    }
    for (i, w) in TENS.iter().enumerate() {
        if word == *w {
            return Some(((i + 2) * 10) as u128);
        }
    }
    match word {
        "hundred" => Some(100),
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        "and" | "minus" => Some(0),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_lowercase() || b == b' ' || b == b'-') {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            tokens.push((&input[start..i], start));
            start = i + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (idx, (token, pos)) in tokens.iter().enumerate() {
        if *token == "minus" && idx == 0 {
            continue;
        }
        if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: *pos });
        }
    }

    let negative = tokens.first().map(|(t, _)| *t == "minus").unwrap_or(false);
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (token, _) in arithmetic {
        match *token {
            "zero" => {}
            "hundred" => group *= 100,
            "thousand" => {
                total += group * 1_000;
                group = 0;
            }
            "million" => {
                total += group * 1_000_000;
                group = 0;
            }
            "billion" => {
                total += group * 1_000_000_000;
                group = 0;
            }
            "trillion" => {
                total += group * 1_000_000_000_000;
                group = 0;
            }
            "quadrillion" => {
                total += group * 1_000_000_000_000_000;
                group = 0;
            }
            "quintillion" => {
                total += group * 1_000_000_000_000_000_000;
                group = 0;
            }
            "and" => {}
            _ => group += word_value(token).unwrap(),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let a = canonical.as_bytes();
        let b = input.as_bytes();
        let common = a.len().min(b.len());
        let index = (0..common).find(|&i| a[i] != b[i]).unwrap_or(common);
        return Err(WordsError::NonCanonical { index });
    }

    Ok(value)
}
