#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn append_group(n: u16, out: &mut String) {
    let mut parts = Vec::new();
    let hundreds = n / 100;
    let remainder = n % 100;

    if hundreds != 0 {
        parts.push(SMALL[hundreds as usize].to_string());
        parts.push("hundred".to_string());
        if remainder != 0 {
            parts.push("and".to_string());
            parts.extend(two_digit_parts(remainder));
        }
    } else {
        parts.extend(two_digit_parts(remainder));
    }

    out.push_str(&parts.join(" "));
}

fn two_digit_parts(n: u16) -> Vec<String> {
    if n == 0 {
        Vec::new()
    } else if n < 20 {
        vec![SMALL[n as usize].to_string()]
    } else {
        let tens = TENS[(n / 10) as usize].to_string();
        let units = n % 10;
        if units == 0 {
            vec![tens]
        } else {
            vec![format!("{}-{}", tens, SMALL[units as usize])]
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    while remaining > 0 {
        groups.push((remaining % 1000) as u16);
        remaining /= 1000;
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut emitted_higher = false;
    for idx in (0..groups.len()).rev() {
        let group = groups[idx];
        if group == 0 {
            continue;
        }

        if !out.is_empty() && !(idx == 0 && group < 100 && emitted_higher) {
            out.push(' ');
        } else if idx == 0 && group < 100 && emitted_higher {
            out.push_str(" and ");
        }

        append_group(group, &mut out);
        if idx > 0 {
            out.push(' ');
            out.push_str(SCALES[idx - 1]);
            emitted_higher = true;
        }
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }
    for (i, value) in SMALL.iter().enumerate().skip(1) {
        if word == *value {
            return Some(i as u128);
        }
    }
    for (i, value) in TENS.iter().enumerate().skip(2) {
        if word == *value {
            return Some((i * 10) as u128);
        }
    }
    None
}

fn scale_value(word: &str) -> Option<u128> {
    match word {
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !(bytes[index - 1].is_ascii_lowercase()
                    && bytes[index + 1].is_ascii_lowercase())
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let tokens: Vec<(&str, usize)> = input
        .split(|c| c == ' ' || c == '-')
        .scan(0usize, |offset, token| {
            let current = *offset;
            *offset += token.len() + 1;
            Some((token, current))
        })
        .collect();

    for (i, (word, index)) in tokens.iter().enumerate() {
        let recognized = *word == "minus"
            || *word == "hundred"
            || *word == "and"
            || scale_value(word).is_some()
            || word_value(word).is_some();
        if !recognized || (*word == "minus" && i != 0) {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (word, _) in arithmetic {
        if let Some(value) = word_value(word) {
            group = group.checked_add(value).ok_or(WordsError::Range)?;
        } else if *word == "hundred" {
            group = group.checked_mul(100).ok_or(WordsError::Range)?;
        } else if *word == "and" {
            // No arithmetic effect.
        } else if let Some(scale) = scale_value(word) {
            let scaled = group.checked_mul(scale).ok_or(WordsError::Range)?;
            total = total.checked_add(scaled).ok_or(WordsError::Range)?;
            group = 0;
        }

        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(value)
}
