#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
    (1, ""),
];

fn encode_under_thousand(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            encode_under_hundred(remainder, out);
        }
    } else {
        encode_under_hundred(n, out);
    }
}

fn encode_under_hundred(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let groups = [
        (1_000_000_000_000_000_000u128, "quintillion"),
        (1_000_000_000_000_000u128, "quadrillion"),
        (1_000_000_000_000u128, "trillion"),
        (1_000_000_000u128, "billion"),
        (1_000_000u128, "million"),
        (1_000u128, "thousand"),
        (1u128, ""),
    ];

    let mut parts: Vec<String> = Vec::new();
    let mut remainder = magnitude;
    for (scale, name) in groups {
        let group = if scale == 1 {
            remainder as u16
        } else {
            let g = (remainder / scale) as u16;
            remainder %= scale;
            g
        };
        if group == 0 {
            continue;
        }
        let mut part = String::new();
        encode_under_thousand(group, &mut part);
        if !name.is_empty() {
            part.push(' ');
            part.push_str(name);
        }
        parts.push(part);
    }

    for (i, part) in parts.iter().enumerate() {
        if i > 0 {
            out.push(' ');
        }
        out.push_str(part);
    }

    // Cross-group "and" joins the final higher group to a units value in 1..99.
    let units = (magnitude % 1000) as u16;
    if parts.len() > 1 && units > 0 && units < 100 {
        let suffix = {
            let mut s = String::new();
            encode_under_hundred(units, &mut s);
            s
        };
        if let Some(pos) = out.rfind(&suffix) {
            if pos > 0 && out.as_bytes()[pos - 1] == b' ' {
                out.replace_range(pos - 1..pos, " and ");
            }
        }
    }

    Ok(out)
}

fn dictionary_value(word: &str) -> Option<(u128, TokenKind)> {
    if word == "zero" {
        return Some((0, TokenKind::Number));
    }
    for (i, w) in SMALL.iter().enumerate().skip(1) {
        if word == *w {
            return Some((i as u128, TokenKind::Number));
        }
    }
    for (i, w) in TENS.iter().enumerate() {
        if word == *w {
            return Some(((i as u128 + 2) * 10, TokenKind::Number));
        }
    }
    if word == "hundred" {
        return Some((100, TokenKind::Hundred));
    }
    for (value, name) in [
        (1_000u128, "thousand"),
        (1_000_000u128, "million"),
        (1_000_000_000u128, "billion"),
        (1_000_000_000_000u128, "trillion"),
        (1_000_000_000_000_000u128, "quadrillion"),
        (1_000_000_000_000_000_000u128, "quintillion"),
    ] {
        if word == name {
            return Some((value, TokenKind::Scale));
        }
    }
    if word == "and" {
        return Some((0, TokenKind::And));
    }
    if word == "minus" {
        return Some((0, TokenKind::Minus));
    }
    None
}

#[derive(Clone, Copy)]
enum TokenKind {
    Number,
    Hundred,
    Scale,
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(WordsError::Empty);
    }

    for (i, &b) in bytes.iter().enumerate() {
        if !(b'a'..=b'z').contains(&b) && b != b' ' && b != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            let before = i > 0 && bytes[i - 1].is_ascii_lowercase();
            let after = i + 1 < bytes.len() && bytes[i + 1].is_ascii_lowercase();
            if !before || !after {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<(&str, usize, (u128, TokenKind))> = Vec::new();
    let mut start = 0usize;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b' ' || bytes[i] == b'-' {
            let word = &input[start..i];
            let Some(value) = dictionary_value(word) else {
                return Err(WordsError::InvalidWord { index: start });
            };
            tokens.push((word, start, value));
            start = i + 1;
        }
    }

    let mut negative = false;
    let mut first_arithmetic = 0usize;
    if matches!(tokens[0].2.1, TokenKind::Minus) {
        negative = true;
        first_arithmetic = 1;
    }
    for (word, index, (_, kind)) in tokens.iter().skip(first_arithmetic) {
        if matches!(kind, TokenKind::Minus) {
            let _ = word;
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for (_, _, (value, kind)) in tokens.iter().skip(first_arithmetic) {
        match kind {
            TokenKind::Number => group += *value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale => {
                total += group * *value;
                group = 0;
            }
            TokenKind::And | TokenKind::Minus => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let decoded = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(decoded)?;
    if canonical.as_bytes() != bytes {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(decoded)
}
