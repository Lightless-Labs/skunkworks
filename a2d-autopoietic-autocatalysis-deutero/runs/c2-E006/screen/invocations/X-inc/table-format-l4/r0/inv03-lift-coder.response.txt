#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type LogicalCell = Vec<Vec<Token>>;

fn minimum_column_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn overhead(mode: &Mode, columns: usize) -> usize {
    match mode {
        Mode::Grid | Mode::Markdown => 3 * columns + 1,
        Mode::Compact => 2 * columns.saturating_sub(1),
    }
}

fn invalid_scalar(c: char) -> bool {
    matches!(c, '\r' | '\u{2028}' | '\u{2029}')
        || (c.is_control() && c != '\n')
}

fn normalize_and_tokenize(s: &str, markdown: bool) -> Vec<Vec<Token>> {
    let mut lines: Vec<Vec<Token>> = vec![Vec::new()];
    let mut chars = s.chars().peekable();

    while let Some(c) = chars.next() {
        match c {
            '\r' => {
                // Validation guarantees this is followed by LF.
                let _ = chars.next();
                lines.push(Vec::new());
            }
            '\n' => lines.push(Vec::new()),
            '|' if markdown => lines.last_mut().unwrap().push(Token {
                text: "\\|".to_owned(),
                width: 2,
            }),
            '\\' if markdown => lines.last_mut().unwrap().push(Token {
                text: "\\\\".to_owned(),
                width: 2,
            }),
            _ => lines.last_mut().unwrap().push(Token {
                text: c.to_string(),
                width: 1,
            }),
        }
    }

    for line in &mut lines {
        while line.last().is_some_and(|t| t.text == " ") {
            line.pop();
        }
    }
    lines
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn make_fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }

            let mut out = String::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                out.push_str(&token.text);
                used += token.width;
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // All token widths are at most two and Markdown has minimum width 3.
                let fragment: String = tokens[start..end]
                    .iter()
                    .map(|t| t.text.as_str())
                    .collect();
                result.push(fragment);
                start = end;
            }
            result
        }
    }
}

fn align(content: &str, content_width: usize, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::with_capacity(content.len() + extra);
    out.push_str(&" ".repeat(left));
    out.push_str(content);
    out.push_str(&" ".repeat(right));
    out
}

fn emit_data_line(
    out: &mut String,
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for i in 0..widths.len() {
                let content = &fragments[i];
                let aligned = align(content, content.chars().count(), widths[i], &alignments[i]);
                out.push(' ');
                out.push_str(&aligned);
                out.push(' ');
                out.push('|');
            }
            out.push('\n');
        }
        Mode::Compact => {
            for i in 0..widths.len() {
                if i > 0 {
                    out.push_str("  ");
                }
                let content = &fragments[i];
                let extra = widths[i].saturating_sub(content.chars().count());
                let (left, right) = match alignments[i] {
                    Alignment::Left => (0, extra),
                    Alignment::Right => (extra, 0),
                    Alignment::Center => (extra / 2, extra - extra / 2),
                };
                out.push_str(&" ".repeat(left));
                out.push_str(content);
                if i + 1 < widths.len() {
                    out.push_str(&" ".repeat(right));
                }
            }
            out.push('\n');
        }
    }
}

fn emit_grid_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for width in widths {
        out.push_str(&"-".repeat(width + 2));
        out.push('+');
    }
    out.push('\n');
}

fn emit_markdown_separator(out: &mut String, widths: &[usize], alignments: &[Alignment]) {
    out.push('|');
    for i in 0..widths.len() {
        let marker = match alignments[i] {
            Alignment::Left => format!(":{}", "-".repeat(widths[i] - 1)),
            Alignment::Right => format!("{}:", "-".repeat(widths[i] - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(widths[i] - 2)),
        };
        out.push(' ');
        out.push_str(&marker);
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_width = minimum_column_width(&options.mode);
    let minimum = overhead(&options.mode, n) + n * min_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            let mut byte = 0;
            let mut iter = cell.char_indices().peekable();
            while let Some((offset, c)) = iter.next() {
                if c == '\r' {
                    if !matches!(iter.peek(), Some((_, '\n'))) {
                        return Err(TableError::InvalidChar {
                            row,
                            column,
                            byte: offset,
                        });
                    }
                    let _ = iter.next();
                    byte = offset + 2;
                    continue;
                }
                if invalid_scalar(c) {
                    return Err(TableError::InvalidChar {
                        row,
                        column,
                        byte: offset,
                    });
                }
                byte = offset + c.len_utf8();
            }
            let _ = byte;
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = options.mode == Mode::Markdown;
    let mut cells: Vec<Vec<LogicalCell>> = Vec::with_capacity(rows.len());
    let mut widths = vec![min_width; n];

    for row in rows {
        let mut processed_row = Vec::with_capacity(n);
        for (column, cell) in row.iter().enumerate() {
            let lines = normalize_and_tokenize(cell, markdown);
            for line in &lines {
                widths[column] = widths[column].max(token_width(line));
            }
            processed_row.push(lines);
        }
        cells.push(processed_row);
    }

    if options.max_width != 0 {
        let available = options.max_width - overhead(&options.mode, n);
        let mut sum: usize = widths.iter().sum();
        while sum > available {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_width)
                .max_by_key(|&i| (widths[i], i));
            let Some(i) = candidate else { break };
            widths[i] -= 1;
            sum -= 1;
        }
    }

    let mut out = String::new();
    if options.mode == Mode::Grid {
        emit_grid_border(&mut out, &widths);
    }

    for (row_index, row) in cells.iter().enumerate() {
        let mut rendered: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let mut fragments = Vec::new();
            for line in &row[column] {
                fragments.extend(make_fragments(line, widths[column], &overflows[column]));
            }
            height = height.max(fragments.len());
            rendered.push(fragments);
        }

        for fragment_index in 0..height {
            let fragments: Vec<String> = (0..n)
                .map(|column| {
                    rendered[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            emit_data_line(
                &mut out,
                &fragments,
                &widths,
                &alignments,
                &options.mode,
            );
        }

        if options.mode == Mode::Grid {
            if options.header && row_index == 0 && rows.len() > 1 {
                emit_grid_border(&mut out, &widths);
            } else if row_index + 1 == rows.len() {
                emit_grid_border(&mut out, &widths);
            }
        } else if options.mode == Mode::Markdown && options.header && row_index == 0 {
            emit_markdown_separator(&mut out, &widths, &alignments);
        }
    }

    Ok(out)
}
