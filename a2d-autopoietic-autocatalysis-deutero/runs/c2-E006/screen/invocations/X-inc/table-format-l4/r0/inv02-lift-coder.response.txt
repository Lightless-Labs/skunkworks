#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn fixed_overhead(mode: &Mode, columns: usize) -> usize {
    match mode {
        Mode::Compact => 2 * columns.saturating_sub(1),
        Mode::Grid | Mode::Markdown => 3 * columns + 1,
    }
}

fn min_column_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn is_invalid(c: char) -> bool {
    c != '\n' && c.is_control() || c == '\u{2028}' || c == '\u{2029}'
}

fn transform_cell(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for c in trimmed.chars() {
                if markdown && c == '|' {
                    tokens.push(Token {
                        text: "\\|".to_owned(),
                        width: 2,
                    });
                } else if markdown && c == '\\' {
                    tokens.push(Token {
                        text: "\\\\".to_owned(),
                        width: 2,
                    });
                } else {
                    tokens.push(Token {
                        text: c.to_string(),
                        width: 1,
                    });
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let available = width.saturating_sub(1);
            let mut used = 0;
            let mut out = String::new();
            for token in tokens {
                if used + token.width > available {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                if end == start {
                    // All supported modes have sufficient minimum width for every token.
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn align(content: &str, width: usize, alignment: &Alignment) -> String {
    let used = content.chars().count();
    let extra = width.saturating_sub(used);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), content, " ".repeat(right))
}

fn append_grid_line(out: &mut String, cells: &[String]) {
    out.push('|');
    for cell in cells {
        out.push(' ');
        out.push_str(cell);
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
}

fn append_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for &width in widths {
        out.push_str(&"-".repeat(width + 2));
        out.push('+');
    }
    out.push('\n');
}

fn append_markdown_marker(out: &mut String, widths: &[usize], alignments: &[Alignment]) {
    out.push('|');
    for (i, &width) in widths.iter().enumerate() {
        let marker = match alignments[i] {
            Alignment::Left => format!(":{}", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        out.push(' ');
        out.push_str(&marker);
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_width = min_column_width(&options.mode);
    let overhead = fixed_overhead(&options.mode, n);
    let minimum = overhead + n * min_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut byte = 0;
            let chars: Vec<(usize, char)> = cell.char_indices().collect();
            let mut i = 0;
            while i < chars.len() {
                let (offset, c) = chars[i];
                if c == '\r' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                        i += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte: offset,
                    });
                }
                if is_invalid(c) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte: offset,
                    });
                }
                byte = offset + c.len_utf8();
                i += 1;
            }
            let _ = byte;
        }
    }

    let markdown = matches!(options.mode, Mode::Markdown);
    let mut transformed = vec![vec![Vec::<Vec<Token>>::new(); n]; rows.len()];
    let mut widths = vec![min_width; n];

    for (r, row) in rows.iter().enumerate() {
        for (c, cell) in row.iter().enumerate() {
            let logical_lines = transform_cell(cell, markdown);
            for line in &logical_lines {
                widths[c] = widths[c].max(token_width(line));
            }
            transformed[r][c] = logical_lines;
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&c| widths[c] > min_width)
                .max_by_key(|&c| (widths[c], c));
            match candidate {
                Some(c) => widths[c] -= 1,
                None => break,
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        append_border(&mut out, &widths);
    }

    for (row_index, row) in transformed.iter().enumerate() {
        let mut cell_fragments = Vec::with_capacity(n);
        let mut height = 1;
        for c in 0..n {
            let mut all = Vec::new();
            for logical_line in &row[c] {
                all.extend(fragments(logical_line, widths[c], &overflows[c]));
            }
            height = height.max(all.len());
            cell_fragments.push(all);
        }

        for line_index in 0..height {
            let mut rendered = Vec::with_capacity(n);
            for c in 0..n {
                let content = cell_fragments[c]
                    .get(line_index)
                    .map(String::as_str)
                    .unwrap_or("");
                let aligned = align(content, widths[c], &alignments[c]);
                if matches!(options.mode, Mode::Compact) && c == n - 1 {
                    let used = content.chars().count();
                    let extra = widths[c].saturating_sub(used);
                    let right_padding = match alignments[c] {
                        Alignment::Left => extra,
                        Alignment::Right => 0,
                        Alignment::Center => extra - extra / 2,
                    };
                    rendered.push(aligned.chars().take(widths[c] - right_padding).collect());
                } else {
                    rendered.push(aligned);
                }
            }

            match options.mode {
                Mode::Grid | Mode::Markdown => append_grid_line(&mut out, &rendered),
                Mode::Compact => {
                    for (c, cell) in rendered.iter().enumerate() {
                        if c > 0 {
                            out.push_str("  ");
                        }
                        out.push_str(cell);
                    }
                    out.push('\n');
                }
            }
        }

        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            append_markdown_marker(&mut out, &widths, &alignments);
        }

        if matches!(options.mode, Mode::Grid) {
            if row_index == 0 && options.header && row_index + 1 < rows.len() {
                append_border(&mut out, &widths);
            }
            if row_index + 1 == rows.len() {
                append_border(&mut out, &widths);
            }
        }
    }

    Ok(out)
}
