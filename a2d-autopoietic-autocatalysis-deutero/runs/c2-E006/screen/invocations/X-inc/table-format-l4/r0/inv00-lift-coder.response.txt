#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn fixed_overhead(mode: &Mode, n: usize) -> usize {
    match mode {
        Mode::Compact => 2 * n.saturating_sub(1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    }
}

fn minimum_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn validate_characters(rows: &[Vec<String>]) -> Result<(), TableError> {
    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut chars = cell.char_indices().peekable();
            while let Some((byte, ch)) = chars.next() {
                if ch == '\r' {
                    if matches!(chars.peek(), Some((_, '\n'))) {
                        chars.next();
                    } else {
                        return Err(TableError::InvalidChar {
                            row: row_index,
                            column: column_index,
                            byte,
                        });
                    }
                } else if ch != '\n' && (ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}') {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }
    Ok(())
}

fn logical_lines(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for ch in trimmed.chars() {
                if markdown && ch == '|' {
                    tokens.push(Token {
                        text: "\\|".to_owned(),
                        width: 2,
                    });
                } else if markdown && ch == '\\' {
                    tokens.push(Token {
                        text: "\\\\".to_owned(),
                        width: 2,
                    });
                } else {
                    tokens.push(Token {
                        text: ch.to_string(),
                        width: 1,
                    });
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|token| token.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }

            let mut used = 0;
            let mut out = String::new();
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align(text: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let extra = width.saturating_sub(scalar_len(text));
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let right = if omit_right { 0 } else { right };
    format!("{}{}{}", " ".repeat(left), text, " ".repeat(right))
}

fn render_data_line(
    cells: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    match mode {
        Mode::Compact => {
            let mut parts = Vec::with_capacity(cells.len());
            for i in 0..cells.len() {
                parts.push(align(
                    &cells[i],
                    widths[i],
                    &alignments[i],
                    i + 1 == cells.len(),
                ));
            }
            parts.join("  ")
        }
        Mode::Grid | Mode::Markdown => {
            let mut line = String::from("|");
            for i in 0..cells.len() {
                line.push(' ');
                line.push_str(&align(&cells[i], widths[i], &alignments[i], false));
                line.push_str(" |");
            }
            line
        }
    }
}

fn grid_border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line
}

fn markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut line = String::from("|");
    for (width, alignment) in widths.iter().zip(alignments) {
        let marker = match alignment {
            Alignment::Left => format!(":{}", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        line.push(' ');
        line.push_str(&marker);
        line.push_str(" |");
    }
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_width = minimum_width(&options.mode);
    let minimum = fixed_overhead(&options.mode, n) + n * min_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    validate_characters(rows)?;

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let transformed: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| logical_lines(cell, markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![min_width; n];
    for row in &transformed {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let overhead = fixed_overhead(&options.mode, n);
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let candidate = (0..n)
                .filter(|&column| widths[column] > min_width)
                .max_by_key(|&column| (widths[column], column));
            match candidate {
                Some(column) => {
                    widths[column] -= 1;
                    total -= 1;
                }
                None => break,
            }
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        output.push_str(&grid_border(&widths));
        output.push('\n');
    }

    for (row_index, row) in transformed.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<Vec<String>>> = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let mut lines_for_cell = Vec::new();
            for logical_line in &row[column] {
                let parts = fragments(logical_line, widths[column], &overflows[column]);
                height = height.max(parts.len());
                lines_for_cell.push(parts);
            }

            let mut all = Vec::new();
            for parts in lines_for_cell {
                all.extend(parts);
            }
            cell_fragments.push(all);
        }

        for fragment_index in 0..height {
            let mut line_cells = Vec::with_capacity(n);
            for column in 0..n {
                line_cells.push(
                    cell_fragments[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default(),
                );
            }
            output.push_str(&render_data_line(
                &line_cells,
                &widths,
                &alignments,
                &options.mode,
            ));
            output.push('\n');
        }

        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            output.push_str(&markdown_marker(&widths, &alignments));
            output.push('\n');
        }

        if matches!(options.mode, Mode::Grid) && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&grid_border(&widths));
            output.push('\n');
        }
    }

    if matches!(options.mode, Mode::Grid) {
        output.push_str(&grid_border(&widths));
        output.push('\n');
    }

    Ok(output)
}
