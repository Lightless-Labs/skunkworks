#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;
    if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = base.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = tag_end + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    // Invalid digits take precedence over separator and leading-zero errors.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < base => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let left_is_digit = offset > 0 && payload[offset - 1] != b'_';
            let right_is_digit = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !left_is_digit || !right_is_digit {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Little-endian base-256 representation.
fn to_big_bytes(digits: &[u8], base: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        let mut carry = digit as u16;
        for byte in &mut value {
            let n = (*byte as u16) * (base as u16) + carry;
            *byte = n as u8;
            carry = n >> 8;
        }
        while carry != 0 {
            value.push(carry as u8);
            carry >>= 8;
        }
    }
    value
}

fn from_big_bytes(value: &[u8], base: u8) -> Vec<u8> {
    let mut work = value.to_vec();
    while work.len() > 1 && work.last() == Some(&0) {
        work.pop();
    }

    let mut digits = Vec::new();
    loop {
        let mut remainder = 0u16;
        let mut nonzero = false;
        for byte in work.iter_mut().rev() {
            let n = (remainder << 8) | (*byte as u16);
            *byte = (n / base as u16) as u8;
            remainder = n % base as u16;
            if *byte != 0 {
                nonzero = true;
            }
        }
        digits.push(remainder as u8);
        if !nonzero {
            break;
        }
    }
    digits.reverse();
    digits
}

fn format_result(mut digits: Vec<u8>, style: BaseStyle) -> String {
    let zero = digits.len() == 1 && digits[0] == 0;
    let pad = if digits.len() < style.min_width {
        style.min_width - digits.len()
    } else {
        0
    };

    let digit_count = digits.len() + pad;
    let mut output = String::with_capacity(digit_count + digit_count / 2);
    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };

    let total = digit_count;
    for position in 0..total {
        if style.group != 0 && position != 0 && (total - position) % style.group == 0 {
            output.push('_');
        }
        let digit = if position < pad {
            0
        } else {
            digits[position - pad]
        };
        output.push(alphabet[digit as usize] as char);
    }

    if zero && style.min_width == 0 {
        // The regular rendering path already emits the single zero.
        return output;
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let value = to_big_bytes(&digits, from);
    let result_digits = from_big_bytes(&value, to);
    Ok(format_result(result_digits, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let mut left_value = to_big_bytes(&left_digits, base);
    let right_value = to_big_bytes(&right_digits, base);

    let max_len = left_value.len().max(right_value.len());
    left_value.resize(max_len, 0);
    let mut carry = 0u16;
    for i in 0..max_len {
        let sum = left_value[i] as u16 + right_value.get(i).copied().unwrap_or(0) as u16 + carry;
        left_value[i] = sum as u8;
        carry = sum >> 8;
    }
    if carry != 0 {
        left_value.push(carry as u8);
    }

    let result_digits = from_big_bytes(&left_value, base);
    Ok(format_result(result_digits, style))
}
