        hash + 1

    };

                operand,

            });
        }

    let mut digits = Vec::new();
    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = i > 0 && payload[i - 1] != b'_';
            let has_digit_after = i + 1 < payload.len() && payload[i + 1] != b'_';

                    operand,

        }

            operand,
            index: payload_start,

        number.pop();

        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < base => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let left_is_digit = offset > 0 && payload[offset - 1] != b'_';
            let right_is_digit = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !left_is_digit || !right_is_digit {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Little-endian base-256 representation.
fn to_big_bytes(digits: &[u8], base: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        let mut carry = digit as u16;
        for byte in &mut value {
            let n = (*byte as u16) * (base as u16) + carry;
            *byte = n as u8;
            carry = n >> 8;
        }
        while carry != 0 {
            value.push(carry as u8);
            carry >>= 8;
        }
    }
    value
}

fn from_big_bytes(value: &[u8], base: u8) -> Vec<u8> {
    let mut work = value.to_vec();
    while work.len() > 1 && work.last() == Some(&0) {
        work.pop();
    }

    let mut digits = Vec::new();
    loop {
        let mut remainder = 0u16;
        let mut nonzero = false;
        for byte in work.iter_mut().rev() {
            let n = (remainder << 8) | (*byte as u16);
            *byte = (n / base as u16) as u8;
            remainder = n % base as u16;
            if *byte != 0 {
                nonzero = true;
            }
        }
        digits.push(remainder as u8);
        if !nonzero {
            break;
        }
    }
    digits.reverse();
    digits
}

fn format_result(mut digits: Vec<u8>, style: BaseStyle) -> String {
    let zero = digits.len() == 1 && digits[0] == 0;
    let pad = if digits.len() < style.min_width {
        style.min_width - digits.len()
    } else {
        0
    };

    let digit_count = digits.len() + pad;
    let mut output = String::with_capacity(digit_count + digit_count / 2);
    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };

    let total = digit_count;
    for position in 0..total {
        if style.group != 0 && position != 0 && (total - position) % style.group == 0 {
            output.push('_');
        }
        let digit = if position < pad {
            0
        } else {
            digits[position - pad]
        };
        output.push(alphabet[digit as usize] as char);
    }

    if zero && style.min_width == 0 {
        // The regular rendering path already emits the single zero.
        return output;
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let value = to_big_bytes(&digits, from);
    let result_digits = from_big_bytes(&value, to);
    Ok(format_result(result_digits, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let mut left_value = to_big_bytes(&left_digits, base);
    let right_value = to_big_bytes(&right_digits, base);

    let max_len = left_value.len().max(right_value.len());
    left_value.resize(max_len, 0);
    let mut carry = 0u16;
    for i in 0..max_len {
        let sum = left_value[i] as u16 + right_value.get(i).copied().unwrap_or(0) as u16 + carry;
        left_value[i] = sum as u8;
        carry = sum >> 8;
    }
    if carry != 0 {
        left_value.push(carry as u8);
    }

    let result_digits = from_big_bytes(&left_value, base);
    Ok(format_result(result_digits, style))
}

fn add_small(number: &mut Vec<u8>, value: u8, radix: u8) {
    let mut carry = value as u16;
    let mut index = number.len();

    while carry > 0 && index > 0 {
        index -= 1;
        let sum = number[index] as u16 + carry;
        number[index] = (sum % radix as u16) as u8;
        carry = sum / radix as u16;
    }

    while carry > 0 {
        number.insert(0, (carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

fn multiply_small(number: &mut Vec<u8>, factor: u8, radix: u8) {

    for digit in number.iter_mut().rev() {
        let product = *digit as u16 * factor as u16 + carry;
        *digit = (product % radix as u16) as u8;
        carry = product / radix as u16;
    }
    while carry > 0 {
        number.insert(0, (carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

    value

fn add_decimal(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u16;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry > 0 {
        let a = if i > 0 {
            i -= 1;
            left[i] as u16
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            right[j] as u16
        } else {
            0
        };
        let sum = a + b + carry;
        result.push((sum % 10) as u8);
        carry = sum / 10;
    }

    if result.is_empty() {
        result.push(0);
    }
    result.reverse();
    result
}

fn divide_decimal(number: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for digit in number.iter_mut() {
        let current = remainder * 10 + *digit as u16;
        *digit = (current / divisor as u16) as u8;
        remainder = current % divisor as u16;
    }

    let first_nonzero = number.iter().position(|&d| d != 0).unwrap_or(number.len() - 1);
    if first_nonzero > 0 {
        number.drain(..first_nonzero);
    }
    remainder as u8
}

fn render(value: &[u8], radix: u8, style: BaseStyle) -> String {
    let mut quotient = value.to_vec();
    let mut digits = Vec::new();

    if quotient.iter().all(|&d| d == 0) {
        digits.push(0);
    } else {
        while quotient.iter().any(|&d| d != 0) {
            digits.push(divide_decimal(&mut quotient, radix));
        }
        digits.reverse();
    }

    while digits.len() < style.min_width {
        digits.insert(0, 0);
    }

    let mut output = String::new();
    for (index, digit) in digits.iter().enumerate() {
        if style.group != 0
            && index > 0
            && (digits.len() - index) % style.group == 0
        {
            output.push('_');
        }

        let character = match *digit {
            0..=9 => (b'0' + *digit) as char,
            value if style.upper => (b'A' + value - 10) as char,
            value => (b'a' + value - 10) as char,
        };
        output.push(character);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(render(&value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let left_value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    let sum = add_decimal(&left_value, &right_value);
    Ok(render(&sum, base, style))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input[..hash] != expected {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    // Invalid digits take precedence over separator and leading-zero errors.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Big-endian base-256 integer helpers.
fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for byte in value.iter_mut().rev() {
        let result = (*byte as u16) * (multiplier as u16) + carry;
        *byte = result as u8;
        carry = result >> 8;
    }
    while carry != 0 {
        value.insert(0, carry as u8);
        carry >>= 8;
    }
    while value.len() > 1 && value[0] == 0 {
        value.remove(0);
    }
}

fn add_values(left: &[u8], right: &[u8]) -> Vec<u8> {
    let max_len = left.len().max(right.len());
    let mut result = Vec::with_capacity(max_len + 1);
    let mut carry = 0u16;

    for i in 0..max_len {
        let a = left
            .len()
            .checked_sub(i + 1)
            .map_or(0, |index| left[index] as u16);
        let b = right
            .len()
            .checked_sub(i + 1)
            .map_or(0, |index| right[index] as u16);
        let sum = a + b + carry;
        result.push(sum as u8);
        carry = sum >> 8;
    }

    if carry != 0 {
        result.push(carry as u8);
    }
    result.reverse();

    while result.len() > 1 && result[0] == 0 {
        result.remove(0);
    }
    result
}

fn divide_value(value: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for byte in value.iter_mut() {
        let current = (remainder << 8) | (*byte as u16);
        *byte = (current / divisor as u16) as u8;
        remainder = current % divisor as u16;
    }
    while value.len() > 1 && value[0] == 0 {
        value.remove(0);
    }
    remainder as u8
}

fn to_radix_digits(value: &[u8], radix: u8) -> Vec<u8> {
    let mut quotient = value.to_vec();
    let mut digits = Vec::new();

    while quotient.iter().any(|&byte| byte != 0) {
        digits.push(divide_value(&mut quotient, radix));
    }

    if digits.is_empty() {
        digits.push(0);
    }
    digits.reverse();
    digits
}

fn render(value: &[u8], radix: u8, style: BaseStyle) -> String {
    let mut digits = to_radix_digits(value, radix);
    if digits.len() < style.min_width {
        let mut padded = vec![0; style.min_width - digits.len()];
        padded.append(&mut digits);
        digits = padded;
    }

    let mut output = String::with_capacity(digits.len() + digits.len() / 2);
    for (index, &digit) in digits.iter().enumerate() {
        if style.group != 0
            && index != 0
            && (digits.len() - index) % style.group == 0
        {
            output.push('_');
        }

        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(byte as char);
    }
    output
}

fn parse_to_integer(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    value
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let value = parse_to_integer(&digits, from);
    Ok(render(&value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
