```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && payload.iter().find(|&&b| b != b'_') == Some(&b'0') {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn add_small(number: &mut Vec<u8>, value: u8, radix: u8) {
    let mut carry = value as u16;
    let mut index = number.len();

    while carry > 0 && index > 0 {
        index -= 1;
        let sum = number[index] as u16 + carry;
        number[index] = (sum % radix as u16) as u8;
        carry = sum / radix as u16;
    }

    while carry > 0 {
        number.insert(0, (carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

fn multiply_small(number: &mut Vec<u8>, factor: u8, radix: u8) {
    let mut carry = 0u16;
    for digit in number.iter_mut().rev() {
        let product = *digit as u16 * factor as u16 + carry;
        *digit = (product % radix as u16) as u8;
        carry = product / radix as u16;
    }
    while carry > 0 {
        number.insert(0, (carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

fn parse_value(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_small(&mut value, radix, 10);
        add_small(&mut value, digit, 10);
    }
    value
}

fn add_decimal(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u16;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry > 0 {
        let a = if i > 0 {
            i -= 1;
            left[i] as u16
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            right[j] as u16
        } else {
            0
        };
        let sum = a + b + carry;
        result.push((sum % 10) as u8);
        carry = sum / 10;
    }

    if result.is_empty() {
        result.push(0);
    }
    result.reverse();
    result
}

fn divide_decimal(number: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for digit in number.iter_mut() {
        let current = remainder * 10 + *digit as u16;
        *digit = (current / divisor as u16) as u8;
        remainder = current % divisor as u16;
    }

    let first_nonzero = number.iter().position(|&d| d != 0).unwrap_or(number.len() - 1);
    if first_nonzero > 0 {
        number.drain(..first_nonzero);
    }
    remainder as u8
}

fn render(value: &[u8], radix: u8, style: BaseStyle) -> String {
    let mut quotient = value.to_vec();
    let mut digits = Vec::new();

    if quotient.iter().all(|&d| d == 0) {
        digits.push(0);
    } else {
        while quotient.iter().any(|&d| d != 0) {
            digits.push(divide_decimal(&mut quotient, radix));
        }
        digits.reverse();
    }

    while digits.len() < style.min_width {
        digits.insert(0, 0);
    }

    let mut output = String::new();
    for (index, digit) in digits.iter().enumerate() {
        if style.group != 0
            && index > 0
            && (digits.len() - index) % style.group == 0
        {
            output.push('_');
        }

        let character = match *digit {
            0..=9 => (b'0' + *digit) as char,
            value if style.upper => (b'A' + value - 10) as char,
            value => (b'a' + value - 10) as char,
        };
        output.push(character);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(render(&value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let left_value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    let sum = add_decimal(&left_value, &right_value);
    Ok(render(&sum, base, style))
}
```
