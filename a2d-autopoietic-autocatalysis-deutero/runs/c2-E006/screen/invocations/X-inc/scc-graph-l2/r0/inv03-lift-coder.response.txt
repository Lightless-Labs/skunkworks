pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (index, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 1,
                vertex: to,
            });
        }
    }

    // Iterative Kosaraju avoids recursion depth limits on long paths.
    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.last_mut() {
            if *next_index < adjacency[*vertex].len() {
                let next = adjacency[*vertex][*next_index];
                *next_index += 1;
                if !seen[next] {
                    seen[next] = true;
                    stack.push((next, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                order.push(finished);
            }
        }
    }

    let mut raw_component_of = vec![usize::MAX; n];
    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    for &start in order.iter().rev() {
        if raw_component_of[start] != usize::MAX {
            continue;
        }
        let id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        raw_component_of[start] = id;
        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &next in &reverse[vertex] {
                if raw_component_of[next] == usize::MAX {
                    raw_component_of[next] = id;
                    stack.push(next);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    let mut raw_ids: Vec<usize> = (0..raw_components.len()).collect();
    raw_ids.sort_by(|&a, &b| {
        raw_components[b]
            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
    });

    let mut canonical_id = vec![0usize; raw_components.len()];
    let mut components = Vec::with_capacity(raw_components.len());
    for (id, raw_id) in raw_ids.into_iter().enumerate() {
        canonical_id[raw_id] = id;
        components.push(raw_components[raw_id].clone());
    }

    let component_of: Vec<usize> = raw_component_of
        .iter()
        .map(|&raw_id| canonical_id[raw_id])
        .collect();

    let component_count = components.len();
    let mut cyclic: Vec<bool> = components.iter().map(|members| members.len() > 1).collect();
    let mut internal_edges = vec![0usize; component_count];
    let mut arc_data: Vec<(usize, usize, usize, usize)> = Vec::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let from_component = component_of[from];
        let to_component = component_of[to];
        if from_component == to_component {
            internal_edges[from_component] += 1;
            if from == to {
                cyclic[from_component] = true;
            }
        } else {
            arc_data.push((from_component, to_component, edge_index, 1));
        }
    }

    arc_data.sort_unstable_by_key(|&(from, to, _, _)| (from, to));
    let mut arcs = Vec::new();
    let mut index = 0;
    while index < arc_data.len() {
        let (from, to, first_edge, _) = arc_data[index];
        let mut multiplicity = 0usize;
        let mut first = first_edge;
        while index < arc_data.len() && arc_data[index].0 == from && arc_data[index].1 == to {
            multiplicity += 1;
            first = first.min(arc_data[index].2);
            index += 1;
        }
        arcs.push(Arc {
            from,
            to,
            multiplicity,
            first_edge: first,
        });
    }
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
