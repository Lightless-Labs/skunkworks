use std::cmp::Reverse;
use std::collections::{BinaryHeap, HashMap, HashSet, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EdgeStatus {
    Duplicate,
    Internal,
    Link,
    Merge,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Step {
    pub edge_index: usize,
    pub status: EdgeStatus,
    pub before: (usize, usize),
    pub after: (usize, usize),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub edge_index: usize,
    pub before: Vec<usize>,
    pub after: usize,
    pub members: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Batch {
    pub steps: Vec<Step>,
    pub merges: Vec<Merge>,
    pub graph: Graph,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn validate(n: usize, edges: &[(usize, usize)]) -> Result<(), GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(u, v)) in edges.iter().enumerate() {
        if u >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: u,
            });
        }
        if v >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: v,
            });
        }
    }
    Ok(())
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;

    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next)) = stack.pop() {
            if next < adj[v].len() {
                stack.push((v, next + 1));
                let w = adj[v][next];
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                order.push(v);
            }
        }
    }

    let mut assigned = vec![false; n];
    let mut comps = Vec::<Vec<usize>>::new();
    for &root in order.iter().rev() {
        if assigned[root] {
            continue;
        }
        assigned[root] = true;
        let mut members = Vec::new();
        let mut stack = vec![root];
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if !assigned[w] {
                    assigned[w] = true;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        comps.push(members);
    }
    comps.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));

    let mut component_of = vec![0; n];
    for (id, c) in comps.iter().enumerate() {
        for &v in c {
            component_of[v] = id;
        }
    }

    let k = comps.len();
    let mut cyclic = vec![false; k];
    for (id, c) in comps.iter().enumerate() {
        cyclic[id] = c.len() > 1;
    }
    let mut internal_edges = vec![0usize; k];
    let mut arc_data = HashMap::<(usize, usize), (usize, usize)>::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }
    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); k];
    let mut indegree = vec![0usize; k];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = BinaryHeap::<(usize, usize, Reverse<usize>)>::new();
    for id in 0..k {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], comps[id][0], Reverse(id)));
        }
    }
    let mut topo = Vec::with_capacity(k);
    let mut depth = vec![0usize; k];
    while let Some((_, _, Reverse(id))) = ready.pop() {
        topo.push(id);
        for &to in &outgoing[id] {
            depth[to] = depth[to].max(depth[id] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.push((internal_edges[to], comps[to][0], Reverse(to)));
            }
        }
    }

    Ok(Graph {
        components: comps,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            return Ok(if self.cyclic[a] {
                Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                }
            } else {
                Query {
                    kind: Relation::Trivial,
                    route: vec![a],
                }
            });
        }

        fn shortest(g: &Graph, start: usize, goal: usize) -> Option<Vec<usize>> {
            let n = g.components.len();
            let mut outgoing = vec![Vec::new(); n];
            for arc in &g.arcs {
                outgoing[arc.from].push(arc.to);
            }
            for v in &mut outgoing {
                v.sort_unstable();
            }
            let mut dist = vec![usize::MAX; n];
            dist[goal] = 0;
            let mut q = VecDeque::from([goal]);
            while let Some(v) = q.pop_front() {
                for arc in &g.arcs {
                    if arc.to == v && dist[arc.from] == usize::MAX {
                        dist[arc.from] = dist[v] + 1;
                        q.push_back(arc.from);
                    }
                }
            }
            if dist[start] == usize::MAX {
                return None;
            }
            let mut route = vec![start];
            let mut cur = start;
            while cur != goal {
                let next = outgoing[cur]
                    .iter()
                    .copied()
                    .filter(|&v| dist[v].checked_add(1) == Some(dist[cur]))
                    .min()?;
                route.push(next);
                cur = next;
            }
            Some(route)
        }

        if let Some(route) = shortest(self, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest(self, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    history: Vec<Merge>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        analyze(n, edges)?;
        Ok(Self {
            n,
            edges: edges.to_vec(),
            history: Vec::new(),
        })
    }

    pub fn snapshot(&self) -> Graph {
        analyze(self.n, &self.edges).expect("validated retained edges")
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len().saturating_add(edges.len()) > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(u, v)) in edges.iter().enumerate() {
            if u >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: u,
                });
            }
            if v >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: v,
                });
            }
        }

        let mut steps = Vec::with_capacity(edges.len());
        let mut merges = Vec::new();
        let mut pairs: HashSet<(usize, usize)> = self.edges.iter().copied().collect();

        for &(u, v) in edges {
            let edge_index = self.edges.len();
            let before_graph = self.snapshot();
            let before = (
                before_graph.component_of[u],
                before_graph.component_of[v],
            );
            let duplicate = pairs.contains(&(u, v));

            self.edges.push((u, v));
            pairs.insert((u, v));
            let after_graph = self.snapshot();
            let after = (after_graph.component_of[u], after_graph.component_of[v]);

            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };

            if status == EdgeStatus::Merge {
                let mut united = Vec::new();
                for (id, comp) in before_graph.components.iter().enumerate() {
                    if reaches(&before_graph, before.0, id)
                        && reaches(&before_graph, id, before.1)
                    {
                        united.extend(comp.iter().copied());
                    }
                }
                united.sort_unstable();
                let mut old_ids: Vec<usize> = united
                    .iter()
                    .map(|&x| before_graph.component_of[x])
                    .collect();
                old_ids.sort_unstable();
                old_ids.dedup();
                let event = Merge {
                    edge_index,
                    before: old_ids,
                    after: after_graph.component_of[u],
                    members: united,
                };
                self.history.push(event.clone());
                merges.push(event);
            }

            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.snapshot(),
        })
    }
}

fn reaches(g: &Graph, from: usize, to: usize) -> bool {
    if from == to {
        return true;
    }
    let mut seen = vec![false; g.components.len()];
    let mut stack = vec![from];
    seen[from] = true;
    while let Some(v) = stack.pop() {
        for arc in g.arcs.iter().filter(|arc| arc.from == v) {
            if arc.to == to {
                return true;
            }
            if !seen[arc.to] {
                seen[arc.to] = true;
                stack.push(arc.to);
            }
        }
    }
    false
}
