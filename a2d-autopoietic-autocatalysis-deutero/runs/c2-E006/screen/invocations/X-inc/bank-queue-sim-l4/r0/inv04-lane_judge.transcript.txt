pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;

    let round_up = remainder > complement || (remainder == complement && quotient % 2 == 1);
    Some(if round_up { quotient + 1 } else { quotient })
}

fn job_better(a: &Job, b: &Job) -> bool {
    if a.priority != b.priority {
        return a.priority < b.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        return a.id < b.id;
    }
    if a.arrival != b.arrival {
        return a.arrival < b.arrival;
    }
    a.id > b.id
}

fn best_waiter(waiting: &[Job]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.map_or(true, |j| job_better(&waiting[i], &waiting[j])) {
            best = Some(i);
        }
    }
    best
}

fn expire_waiters(waiting: &mut Vec<Job>, now: u64, events: &mut Vec<Event>, stats: &mut Stats) {
    let mut expired: Vec<Job> = Vec::new();
    let mut retained = Vec::with_capacity(waiting.len());

    for job in waiting.drain(..) {
        if job.deadline.map_or(false, |deadline| deadline <= now) {
            expired.push(job);
        } else {
            retained.push(job);
        }
    }

    expired.sort_by(|a, b| {
        a.deadline
            .cmp(&b.deadline)
            .then_with(|| a.id.cmp(&b.id))
    });

    for mut job in expired {
        let segment = now - job.entry_time;
        job.waited = job.waited.saturating_add(segment);
        stats.wait_ticks += segment as u128;
        stats.reneged += 1;
        emit(
            events,
            now,
            job.id,
            Kind::Renege {
                waited: job.waited,
            },
        );
    }

    *waiting = retained;
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                });
            }
        }
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut last_event_time = 0u64;

    loop {
        // Advance directly to the next externally observable or internal boundary.
        if events.is_empty() && active.is_none() && waiting.is_empty() {
            if next_arrival >= arrivals.len() {
                break;
            }
            now = customers[arrivals[next_arrival]].arrival;
        }

        // Phase A: a finish at this boundary precedes all other work.
        if let Some(current) = active.as_ref() {
            if current.finish == now {
                let finished = active.take().unwrap();
                let id = finished.job.id;
                stats.busy_ticks += (now - finished.dispatch_time) as u128;
                stats.completed += 1;
                stats.turnaround_ticks += (now - finished.job.arrival) as u128;
                stats.completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                last_event_time = now;
            }
        }

        // Phase B: expire customers that were already waiting at the start of the tick.
        expire_waiters(&mut waiting, now, &mut events, &mut stats);
        if events.last().map(|e| e.time == now).unwrap_or(false) {
            last_event_time = now;
        }

        // Phase C: process the entire arrival cohort without dispatching between arrivals.
        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == now
        {
            let customer = &customers[arrivals[next_arrival]];
            next_arrival += 1;
            emit(&mut events, now, customer.id, Kind::Arrive);
            last_event_time = now;

            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                stats.balked += 1;
                emit(&mut events, now, customer.id, Kind::Balk);
                last_event_time = now;
                continue;
            }

            stats.accepted += 1;
            let deadline = customer
                .patience
                .map(|p| customer.arrival + p);
            waiting.push(Job {
                id: customer.id,
                arrival: customer.arrival,
                priority: customer.priority,
                remaining: customer.service,
                patience_left: customer.patience,
                deadline,
                entry_time: now,
                waited: 0,
                started: false,
                paused: false,
                pause_time: now,
            });
        }

        // Phase D: reconsider the active customer once protection has ended.
        if let Some(current) = active.as_ref() {
            if now >= current.work_at {
                if let Some(index) = best_waiter(&waiting) {
                    if waiting[index].priority < current.job.priority {
                        let waiter_id = waiting[index].id;
                        let mut paused = active.take().unwrap();
                        let remaining = paused.finish - now;
                        stats.busy_ticks += (now - paused.dispatch_time) as u128;
                        stats.preemptions += 1;
                        emit(
                            &mut events,
                            now,
                            paused.job.id,
                            Kind::Preempt {
                                remaining,
                                by: waiter_id,
                            },
                        );
                        last_event_time = now;

                        paused.job.remaining = remaining;
                        paused.job.paused = true;
                        paused.job.pause_time = now;
                        paused.job.entry_time = now;
                        paused.job.deadline = match paused.job.patience_left {
                            Some(budget) => Some(now.checked_add(budget).ok_or(
                                Error::TimeOverflow {
                                    customer: paused.job.id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?),
                            None => None,
                        };
                        waiting.push(paused.job);
                    }
                }
            }
        }

        // Phase E: dispatch while idle. Inline zero-length completions remain in this phase.
        loop {
            if active.is_some() {
                break;
            }
            let Some(index) = best_waiter(&waiting) else {
                break;
            };

            let mut job = waiting.remove(index);
            let segment = now - job.entry_time;
            job.waited = job.waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;
            if let Some(left) = job.patience_left.as_mut() {
                *left -= segment;
            }

            let switch = if last_dispatched == Some(job.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                config.switch_cost
            };

            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: job.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(job.remaining)
                .ok_or(Error::TimeOverflow {
                    customer: job.id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let was_started = job.started;
            let kind = if was_started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            emit(&mut events, now, job.id, kind);
            last_event_time = now;
            if !was_started {
                stats.started += 1;
                job.started = true;
            }
            last_dispatched = Some(job.id);

            active = Some(Active {
                job,
                dispatch_time: now,
                work_at,
                finish,
            });

            if finish == now {
                let finished = active.take().unwrap();
                let id = finished.job.id;
                stats.completed += 1;
                stats.turnaround_ticks += (now - finished.job.arrival) as u128;
                stats.completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                continue;
            }
            break;
        }

        // Phase F: expire newly queued customers, including zero-budget arrivals/pauses.
        expire_waiters(&mut waiting, now, &mut events, &mut stats);
        if events.last().map(|e| e.time == now).unwrap_or(false) {
            last_event_time = now;
        }

        if next_arrival >= arrivals.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now && next_time.map_or(true, |old| candidate < old) {
                next_time = Some(candidate);
            }
        };

        if next_arrival < arrivals.len() {
            consider(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(current) = active.as_ref() {
            consider(current.finish);
            if current.work_at > now {
                consider(current.work_at);
            }
        }
        for job in &waiting {
            if let Some(deadline) = job.deadline {
                consider(deadline);
            }
        }

        if let Some(next) = next_time {
            now = next;
        } else {
            // Any remaining waiter can be dispatched at this same tick on the next pass.
            // This fallback also prevents a malformed state from causing tick-by-tick work.
            if active.is_none() && !waiting.is_empty() {
                continue;
            }
            break;
        }
    }

    stats.elapsed = last_event_time;
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks.saturating_mul(1000), stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(
            stats.turnaround_ticks.saturating_mul(1000),
            stats.completed as u128,
        )
        .unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks.saturating_mul(10000), stats.elapsed as u128)
            .unwrap() as u32
    };

    Ok(Report { events, stats })
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))

fn waiting_cmp(a: &CustomerState, b: &CustomerState) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(x), Some(y)) => x.cmp(&y).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn consume_wait(state: &mut CustomerState, now: u64) {
    let segment = now - state.queue_since;
    state.waited = state.waited.saturating_add(segment);
    if let Some(budget) = state.budget.as_mut() {
        *budget -= segment;
    }
}

fn expire_waiters(
    states: &mut [CustomerState],
    queue: &mut Vec<usize>,
    now: u64,
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<usize> = queue
        .iter()
        .copied()
        .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
        .collect();
    expired.sort_by_key(|&i| (states[i].deadline.unwrap(), states[i].customer.id));
    for i in expired {
        if let Some(pos) = queue.iter().position(|&j| j == i) {
            queue.remove(pos);
        }
        consume_wait(&mut states[i], now);
        event(
            events,
            now,
            states[i].customer.id,
            Kind::Renege { waited: states[i].waited },
        );
        stats.reneged += 1;
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index,
                id: c.id,
            })?),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut states: Vec<CustomerState> = customers
        .iter()
        .cloned()
        .zip(deadlines)
        .map(|(customer, initial_deadline)| CustomerState {
            remaining: customer.service,
            budget: customer.patience,
            deadline: initial_deadline,
            queue_since: customer.arrival,
            customer,
            initial_deadline,
            accepted: false,
            finished: false,
            first_started: false,
            waited: 0,
            pause_time: None,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..states.len()).collect();
    arrivals.sort_by(|&a, &b| {
        states[a]
            .customer
            .arrival
            .cmp(&states[b].customer.arrival)
            .then_with(|| states[a].customer.priority.cmp(&states[b].customer.priority))
            .then_with(|| states[b].customer.id.cmp(&states[a].customer.id))
    });

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut queue: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut now = 0u64;
    let mut arrival_pos = 0usize;
    let mut last_event_time = 0u64;

    while stats.completed + stats.reneged + stats.balked < states.len() {
        let next_arrival = arrivals
            .get(arrival_pos)
            .map(|&i| states[i].customer.arrival);
        let next_finish = active.as_ref().map(|a| a.finish);
        let next_work = active
            .as_ref()
            .and_then(|a| (a.work_at > now).then_some(a.work_at));
        let next_deadline = queue
            .iter()
            .filter_map(|&i| states[i].deadline)
            .filter(|&d| d > now)
            .min();
        let next = [next_arrival, next_finish, next_work, next_deadline]
            .into_iter()
            .flatten()
            .min();

        if next.is_none() {
            break;
        }
        now = next.unwrap();

        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            stats.busy_ticks += (now - a.start) as u128;
            let s = &mut states[a.index];
            s.finished = true;
            event(&mut events, now, s.customer.id, Kind::Finish);
            stats.completed += 1;
            stats.turnaround_ticks += (now - s.customer.arrival) as u128;
            stats.completion_order.push(s.customer.id);
            last_event_time = now;
        }

        expire_waiters(&mut states, &mut queue, now, &mut events, &mut stats);
        if !events.is_empty() && events.last().unwrap().time == now {
            last_event_time = now;
        }

        while let Some(&i) = arrivals.get(arrival_pos) {
            if states[i].customer.arrival != now {
                break;
            }
            arrival_pos += 1;
            let id = states[i].customer.id;
            event(&mut events, now, id, Kind::Arrive);
            last_event_time = now;
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if queue.len() >= gate {
                event(&mut events, now, id, Kind::Balk);
                stats.balked += 1;
                last_event_time = now;
            } else {
                states[i].accepted = true;
                stats.accepted += 1;
                states[i].queue_since = now;
                states[i].budget = states[i].customer.patience;
                states[i].deadline = states[i].initial_deadline;
                queue.push(i);
            }
        }

        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(waiter) = best_waiter(&states, &queue) {
                    if states[waiter].customer.priority < states[a.index].customer.priority {
                        let a = active.take().unwrap();
                        stats.busy_ticks += (now - a.start) as u128;
                        let remaining = a.finish - now;
                        let paused_id = states[a.index].customer.id;
                        event(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt {
                                remaining,
                                by: states[waiter].customer.id,
                            },
                        );
                        stats.preemptions += 1;
                        last_event_time = now;
                        let s = &mut states[a.index];
                        s.remaining = remaining;
                        s.queue_since = now;
                        s.pause_time = Some(now);
                        s.deadline = match s.budget {
                            Some(b) => Some(now.checked_add(b).ok_or(Error::TimeOverflow {
                                customer: paused_id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?),
                            None => None,
                        };
                        queue.push(a.index);
                    }
                }
            }
        }

        while active.is_none() {
            let Some(i) = best_waiter(&states, &queue) else {
                break;
            };
            queue.retain(|&j| j != i);
            consume_wait(&mut states[i], now);
            let segment = now - states[i].queue_since;
            stats.wait_ticks += segment as u128;

            let id = states[i].customer.id;
            let switch = if last_dispatched.is_some_and(|last| last != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[i].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let remaining = states[i].remaining;
            let first = !states[i].first_started;
            if first {
                states[i].first_started = true;
                stats.started += 1;
                event(
                    &mut events,
                    now,
                    id,
                    Kind::Start { remaining, work_at, finish },
                );
            } else {
                event(
                    &mut events,
                    now,
                    id,
                    Kind::Resume { remaining, work_at, finish },
                );
            }
            last_event_time = now;
            last_dispatched = Some(id);
            active = Some(Active { index: i, start: now, work_at, finish, remaining });

            if finish == now {
                let a = active.take().unwrap();
                stats.busy_ticks += (now - a.start) as u128;
                states[i].finished = true;
                event(&mut events, now, id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[i].customer.arrival) as u128;
                stats.completion_order.push(id);
                last_event_time = now;
            }
        }

        expire_waiters(&mut states, &mut queue, now, &mut events, &mut stats);
        if !events.is_empty() && events.last().unwrap().time == now {
            last_event_time = now;
        }

        if stats.completed + stats.reneged + stats.balked >= states.len() {
            break;
        }
    }

    stats.elapsed = last_event_time;
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}

    accepted: bool,
    terminal: bool,
    started: bool,
    remaining: u64,
    budget: Option<u64>,
    deadline: Option<u64>,

    waited: u64,

    pause_time: u64,

impl Runtime {
    fn new(customer: &Customer) -> Self {
        Self {
            accepted: false,
            terminal: false,
            started: false,
            remaining: customer.service,
            budget: customer.patience,
            deadline: customer.patience.map(|p| customer.arrival + p),
            entry: customer.arrival,
            waited: 0,
            paused: false,

fn best_waiter(queue: &[usize], customers: &[Customer], states: &[Runtime]) -> Option<usize> {
    queue.iter().copied().min_by(|&a, &b| {
        let sa = &states[a];
        let sb = &states[b];
        let ca = &customers[a];
        let cb = &customers[b];

        ca.priority
            .cmp(&cb.priority)
            .then_with(|| sb.paused.cmp(&sa.paused))
            .then_with(|| {
                if sa.paused && sb.paused {
                    sa.pause_time
                        .cmp(&sb.pause_time)
                        .then_with(|| ca.id.cmp(&cb.id))
                } else if !sa.paused && !sb.paused {
                    ca.arrival
                        .cmp(&cb.arrival)
                        .then_with(|| cb.id.cmp(&ca.id))
                } else {
                    std::cmp::Ordering::Equal
                }
            })
    })
}

fn consume_wait(state: &mut Runtime, time: u64) -> u64 {
    let segment = time - state.entry;
    state.waited += segment;
    if let Some(budget) = state.budget {
        state.budget = Some(budget - segment);

fn expire_waiters(
    time: u64,
    queue: &mut Vec<usize>,
    customers: &[Customer],
    states: &mut [Runtime],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<usize> = queue
        .iter()
        .copied()
        .filter(|&i| states[i].deadline.is_some_and(|deadline| deadline <= time))
        .collect();

    expired.sort_by(|&a, &b| {
        states[a]
            .deadline
            .cmp(&states[b].deadline)
            .then_with(|| customers[a].id.cmp(&customers[b].id))
    });

    for index in expired {
        if let Some(position) = queue.iter().position(|&queued| queued == index) {
            queue.remove(position);
            consume_wait(&mut states[index], time);
            states[index].terminal = true;
            stats.reneged += 1;
            emit(
                events,
                time,
                customers[index].id,
                Kind::Renege {
                    waited: states[index].waited,
                },
            );
        }
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {

        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

            return Err(Error::ZeroId { index });

        if customers[..index]
            .iter()

        {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });

        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,

        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                });
            }
        }
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<Runtime> = customers.iter().map(Runtime::new).collect();
    let mut queue: Vec<usize> = Vec::new();
    let mut arrival_pos = 0usize;
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut has_dispatched = false;
    let mut now: Option<u64> = None;
    let mut events = Vec::new();

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    while stats.completed + stats.reneged + stats.balked < customers.len() {
        let time = match now {
            Some(time) => time,
            None => {
                let Some(&index) = arrivals.get(arrival_pos) else {
                    break;
                };
                customers[index].arrival
            }
        };

        // Phase A: finish the active customer before processing anything else.
        if active.as_ref().is_some_and(|server| server.finish == time) {
            let server = active.take().unwrap();
            let index = server.index;
            stats.busy_ticks += (time - server.interval_start) as u128;
            stats.completed += 1;
            stats.turnaround_ticks += (time - customers[index].arrival) as u128;
            stats.completion_order.push(customers[index].id);
            states[index].terminal = true;
            emit(&mut events, time, customers[index].id, Kind::Finish);
        }

        // Phase B: expire customers who were already waiting at this tick.
        expire_waiters(
            time,
            &mut queue,
            customers,
            &mut states,
            &mut events,
            &mut stats,
        );

        // Phase C: process the entire arrival cohort without dispatching.
        while arrival_pos < arrivals.len()
            && customers[arrivals[arrival_pos]].arrival == time
        {
            let index = arrivals[arrival_pos];
            arrival_pos += 1;
            emit(&mut events, time, customers[index].id, Kind::Arrive);

            let admission_limit =
                config.waiting_capacity + usize::from(active.is_none());
            if queue.len() >= admission_limit {
                stats.balked += 1;
                emit(&mut events, time, customers[index].id, Kind::Balk);
            } else {
                states[index].accepted = true;
                stats.accepted += 1;
                queue.push(index);
            }
        }

        // Phase D: at and after work_at, a strictly higher priority waiter may
        // preempt the active customer.
        if active
            .as_ref()
            .is_some_and(|server| time >= server.work_at && server.finish > time)
        {
            if let Some(waiter) = best_waiter(&queue, customers, &states) {
                let running_index = active.as_ref().unwrap().index;
                if customers[waiter].priority < customers[running_index].priority {
                    let server = active.take().unwrap();
                    stats.busy_ticks += (time - server.interval_start) as u128;

                    let paused_index = server.index;
                    let remaining = server.finish - time;
                    emit(
                        &mut events,
                        time,
                        customers[paused_index].id,
                        Kind::Preempt {
                            remaining,
                            by: customers[waiter].id,
                        },
                    );
                    stats.preemptions += 1;

                    let deadline = match states[paused_index].budget {
                        Some(budget) => Some(time.checked_add(budget).ok_or(
                            Error::TimeOverflow {
                                customer: customers[paused_index].id,
                                time,
                                stage: OverflowStage::Patience,
                            },
                        )?),
                        None => None,
                    };

                    states[paused_index].remaining = remaining;
                    states[paused_index].entry = time;
                    states[paused_index].deadline = deadline;
                    states[paused_index].paused = true;
                    states[paused_index].pause_time = time;
                    queue.push(paused_index);
                }
            }
        }

        // Phase E: dispatch until a customer occupies the server, or the queue
        // is empty. Zero-duration service finishes inline.
        loop {
            if active.is_some() {
                break;
            }
            let Some(index) = best_waiter(&queue, customers, &states) else {
                break;
            };
            let position = queue.iter().position(|&queued| queued == index).unwrap();
            queue.remove(position);

            let segment = consume_wait(&mut states[index], time);
            stats.wait_ticks += segment as u128;
            states[index].deadline = match states[index].budget {
                Some(budget) => Some(time.checked_add(budget).ok_or(Error::TimeOverflow {
                    customer: customers[index].id,
                    time,
                    stage: OverflowStage::Patience,
                })?),
                None => None,
            };

            let switch = if !has_dispatched {
                0
            } else if last_dispatched == Some(customers[index].id) {
                0
            } else {
                config.switch_cost
            };

            let work_at = time.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: customers[index].id,
                time,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[index].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: customers[index].id,
                    time,
                    stage: OverflowStage::Finish,
                })?;

            let was_started = states[index].started;
            let remaining = states[index].remaining;
            if was_started {
                emit(
                    &mut events,
                    time,
                    customers[index].id,
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                states[index].started = true;
                stats.started += 1;
                emit(
                    &mut events,
                    time,
                    customers[index].id,
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    },
                );
            }

            has_dispatched = true;
            last_dispatched = Some(customers[index].id);
            active = Some(Active {
                index,
                work_at,
                finish,
                interval_start: time,
            });

            if finish != time {
                break;
            }

            let server = active.take().unwrap();
            stats.busy_ticks += (time - server.interval_start) as u128;
            stats.completed += 1;
            stats.turnaround_ticks += (time - customers[index].arrival) as u128;
            stats.completion_order.push(customers[index].id);
            states[index].terminal = true;
            emit(&mut events, time, customers[index].id, Kind::Finish);
        }

        // Phase F: expire zero-budget arrivals and pauses that survived Phase E.
        expire_waiters(
            time,
            &mut queue,
            customers,
            &mut states,
            &mut events,
            &mut stats,
        );

        if let Some(event) = events.last() {
            stats.elapsed = event.time;
        }

        if stats.completed + stats.reneged + stats.balked >= customers.len() {
            break;
        }

        let mut next: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time && next.is_none_or(|current| candidate < current) {
                next = Some(candidate);
            }
        };

        if arrival_pos < arrivals.len() {
            consider(customers[arrivals[arrival_pos]].arrival);
        }
        if let Some(server) = &active {
            consider(server.finish);
            if server.work_at > time {
                consider(server.work_at);
            }
        }
        for &index in &queue {
            if let Some(deadline) = states[index].deadline {
                consider(deadline);
            }
        }

        now = next;
    }

    if stats.accepted > 0 {
        stats.mean_wait_milli =
            rounded_ratio(1000 * stats.wait_ticks, stats.accepted as u128).unwrap_or(0);
    }
    if stats.completed > 0 {
        stats.mean_turnaround_milli =
            rounded_ratio(1000 * stats.turnaround_ticks, stats.completed as u128).unwrap_or(0);
    }
    if stats.elapsed > 0 {
        stats.utilization_bps = rounded_ratio(10000 * stats.busy_ticks, stats.elapsed as u128)
            .unwrap_or(0) as u32;
    }

    Ok(Report { events, stats })
}
