#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;
    Some(if remainder > complement || (remainder == complement && quotient % 2 == 1) {
        quotient + 1
    } else {
        quotient
    })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    waited: u64,
    queue_at: u64,
    pause_at: Option<u64>,
    started: bool,
    accepted: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_at: u64,
}

fn push(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn best_waiter(waiting: &[usize], jobs: &[Job]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| {
        let x = &jobs[a];
        let y = &jobs[b];
        x.customer.priority
            .cmp(&y.customer.priority)
            .then_with(|| match (x.pause_at, y.pause_at) {
                (Some(px), Some(py)) => px.cmp(&py).then(x.customer.id.cmp(&y.customer.id)),
                (Some(_), None) => std::cmp::Ordering::Less,
                (None, Some(_)) => std::cmp::Ordering::Greater,
                (None, None) => x.customer.arrival.cmp(&y.customer.arrival)
                    .then(y.customer.id.cmp(&x.customer.id)),
            })
    })
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|p| p.id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow { index, id: c.id })?),
            None => None,
        };
        ids.push(c.id);
        deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers.iter().enumerate().map(|(i, c)| Job {
        customer: c.clone(),
        deadline: deadlines[i],
        remaining: c.service,
        waited: 0,
        queue_at: c.arrival,
        pause_at: None,
        started: false,
        accepted: false,
    }).collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut stats = Stats {
        accepted: 0, balked: 0, reneged: 0, completed: 0, started: 0,
        preemptions: 0, elapsed: 0, busy_ticks: 0, wait_ticks: 0,
        turnaround_ticks: 0, mean_wait_milli: 0, mean_turnaround_milli: 0,
        utilization_bps: 0, completion_order: Vec::new(),
    };

    loop {
        let next_arrival = order.get(arrival_pos).map(|&i| customers[i].arrival);
        let mut next = next_arrival;
        if let Some(a) = &active {
            for t in [Some(a.finish), (a.work_at > now).then_some(a.work_at)].into_iter().flatten() {
                next = Some(next.map_or(t, |n| n.min(t)));
            }
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].deadline.filter(|&d| d > now) {
                next = Some(next.map_or(d, |n| n.min(d)));
            }
        }
        let Some(t) = next else { break };
        now = t;

        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let j = &jobs[a.index];
            stats.busy_ticks += (now - a.dispatch_at) as u128;
            stats.completed += 1;
            stats.turnaround_ticks += (now - j.customer.arrival) as u128;
            stats.completion_order.push(j.customer.id);
            push(&mut events, now, j.customer.id, Kind::Finish);
        }

        expire_waiters(now, &mut waiting, &mut jobs, &mut events, &mut stats);

        while arrival_pos < order.len() && customers[order[arrival_pos]].arrival == now {
            let i = order[arrival_pos];
            arrival_pos += 1;
            let c = &customers[i];
            push(&mut events, now, c.id, Kind::Arrive);
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                stats.balked += 1;
                push(&mut events, now, c.id, Kind::Balk);
            } else {
                jobs[i].accepted = true;
                jobs[i].deadline = deadlines[i];
                waiting.push(i);
                stats.accepted += 1;
            }
        }

        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, &jobs) {
                    if jobs[w].customer.priority < jobs[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let paused = &mut jobs[a.index];
                        let remaining = a.finish - now;
                        stats.busy_ticks += (now - a.dispatch_at) as u128;
                        push(&mut events, now, paused.customer.id, Kind::Preempt {
                            remaining, by: jobs[w].customer.id,
                        });
                        stats.preemptions += 1;
                        paused.remaining = remaining;
                        paused.queue_at = now;
                        paused.pause_at = Some(now);
                        if let Some(budget) = paused.customer.patience {
                            paused.deadline = Some(now.checked_add(budget).ok_or_else(|| {
                                time_overflow(paused.customer.id, now, OverflowStage::Patience)
                            })?);
                        }
                        waiting.push(a.index);
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, &jobs) else { break };
            waiting.retain(|&x| x != i);
            let waited = now - jobs[i].queue_at;
            jobs[i].waited += waited;
            stats.wait_ticks += waited as u128;
            if let Some(d) = jobs[i].deadline {
                jobs[i].deadline = Some(d);
            }
            let id = jobs[i].customer.id;
            let switch = if last_dispatch.is_some_and(|last| last != id) {
                config.switch_cost
            } else { 0 };
            let work_at = now.checked_add(switch).ok_or_else(|| time_overflow(id, now, OverflowStage::WorkAt))?;
            let finish = work_at.checked_add(jobs[i].remaining).ok_or_else(|| time_overflow(id, now, OverflowStage::Finish))?;
            let kind = if jobs[i].started {
                Kind::Resume { remaining: jobs[i].remaining, work_at, finish }
            } else {
                jobs[i].started = true;
                stats.started += 1;
                Kind::Start { remaining: jobs[i].remaining, work_at, finish }
            };
            push(&mut events, now, id, kind);
            last_dispatch = Some(id);
            active = Some(Active { index: i, work_at, finish, dispatch_at: now });
            if finish == now {
                let a = active.take().unwrap();
                stats.completed += 1;
                stats.turnaround_ticks += (now - jobs[i].customer.arrival) as u128;
                stats.completion_order.push(id);
                push(&mut events, now, id, Kind::Finish);
            } else {
                break;
            }
        }

        expire_waiters(now, &mut waiting, &mut jobs, &mut events, &mut stats);
        stats.elapsed = events.last().map_or(0, |e| e.time);
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 { 0 } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 { 0 } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 { 0 } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}

fn expire_waiters(
    now: u64,
    waiting: &mut Vec<usize>,
    jobs: &mut [Job],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<usize> = waiting.iter().copied()
        .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
        .collect();
    expired.sort_by(|&a, &b| {
        jobs[a].deadline.cmp(&jobs[b].deadline)
            .then(jobs[a].customer.id.cmp(&jobs[b].customer.id))
    });
    for i in expired {
        waiting.retain(|&x| x != i);
        let segment = now - jobs[i].queue_at;
        jobs[i].waited += segment;
        stats.wait_ticks += segment as u128;
        stats.reneged += 1;
        push(events, now, jobs[i].customer.id, Kind::Renege { waited: jobs[i].waited });
    }
}
