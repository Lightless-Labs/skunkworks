```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    entries: Vec<Entry>,
    states: std::collections::HashMap<String, State>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            entries: Vec::new(),
            states: std::collections::HashMap::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !Self::valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.states.contains_key(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.wrapping_add(1);
        self.states.insert(job.id.clone(), State::Queued);
        self.entries.push(Entry { job, sequence });
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        match self.states.get(id) {
            None => return Err(Error::Unknown),
            Some(State::Cancelled) => return Err(Error::Cancelled),
            Some(State::Completed) => return Err(Error::Completed),
            Some(State::Queued) => {}
        }

        self.states.insert(id.to_owned(), State::Cancelled);
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let selected = self
            .entries
            .iter()
            .enumerate()
            .filter(|(_, entry)| {
                entry.job.ready_at <= now
                    && self.states.get(&entry.job.id) == Some(&State::Queued)
            })
            .max_by(|(_, a), (_, b)| {
                let pa = self.effective_priority(&a.job, now);
                let pb = self.effective_priority(&b.job, now);
                pa.cmp(&pb).then_with(|| b.sequence.cmp(&a.sequence))
            })
            .map(|(index, _)| index);

        self.last_time = now;

        let Some(index) = selected else {
            return Ok(None);
        };

        let entry = self.entries.remove(index);
        self.states.insert(entry.job.id.clone(), State::Completed);
        Ok(Some(Dispatch {
            id: entry.job.id,
            effective_priority: self.effective_priority(&entry.job, now),
        }))
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn valid_id(id: &str) -> bool {
        (1..=16).contains(&id.len())
            && id
                .bytes()
                .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
    }

    fn effective_priority(&self, job: &Job, now: u64) -> u8 {
        if self.aging_ms == 0 {
            return job.priority;
        }

        let age_steps = (now - job.ready_at) / self.aging_ms;
        let remaining = 9 - job.priority as u64;
        if age_steps >= remaining {
            9
        } else {
            job.priority + age_steps as u8
        }
    }
}
```
