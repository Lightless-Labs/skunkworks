use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len()
            || matches!(bytes[start], b'.' | b'-' | b'+')
        {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(s: &str, start: usize, end: usize, prerelease: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    loop {
        let id_start = i;
        while i < end && bytes[i] != b'.' {
            if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..i];
        if prerelease && id.bytes().all(|b| b.is_ascii_digit()) && id.len() > 1 && id.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if i == end {
            break;
        }
        i += 1;
        if i == end {
            return Err(VersionError::EmptyIdentifier { at: i });
        }
    }
    out.shrink_to_fit();
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            return Err(if matches!(s.as_bytes()[pos], b'-' | b'+') {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        let n = parse_core(s, &mut pos)?;
        if expected == 0 {
            let minor = n;
            if pos >= s.len() || s.as_bytes()[pos] != b'.' {
                return Err(if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                    VersionError::MissingCore { at: pos }
                } else {
                    VersionError::InvalidChar { at: pos }
                });
            }
            pos += 1;
            let patch = parse_core(s, &mut pos)?;
            let (pre, build) = parse_version_tail(s, pos)?;
            return Ok(Version { major, minor, patch, pre, build });
        }
    }
    unreachable!()
}

fn parse_version_tail(s: &str, pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    let bytes = s.as_bytes();
    if pos == bytes.len() {
        return Ok((Vec::new(), Vec::new()));
    }
    if bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..].iter().position(|&b| b == b'+').map(|x| pre_start + x);
        let pre_end = plus.unwrap_or(bytes.len());
        let pre = parse_identifiers(s, pre_start, pre_end, true)?;
        let build = if let Some(p) = plus {
            parse_identifiers(s, p + 1, bytes.len(), false)?
        } else {
            Vec::new()
        };
        Ok((pre, build))
    } else if bytes[pos] == b'+' {
        let build = parse_identifiers(s, pos + 1, bytes.len(), false)?;
        Ok((Vec::new(), build))
    } else {
        Err(VersionError::InvalidChar { at: pos })
    }
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn cmp_token(tok: &str, offset: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, n) = if tok.starts_with(">=") { (">=",2) } else if tok.starts_with("<=") { ("<=",2) } else if tok.starts_with('>') { (">",1) } else if tok.starts_with('<') { ("<",1) } else if tok.starts_with('=') { ("=",1) } else { ("=",0) };

    Ok(())

fn partial(s: &str, offset: usize) -> Result<(Vec<u64>, bool, Option<Version>), RangeError> {
    let fullish = s.contains('-') || s.contains('+') || (s.split('.').count() >= 3 && !s.split('.').any(|x| matches!(x,"*"|"x"|"X")));
    if fullish {
        let v = parse_version(s).map_err(|error| RangeError::Version { at: offset, error })?;
        return Ok((vec![v.major,v.minor,v.patch], true, Some(v)));
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|x| x.is_empty()) { return Err(RangeError::Syntax { at: offset }); }

    for f in fields {
        if matches!(f,"*"|"x"|"X") { wildcard=true; }

            nums.push(n);
        }
    }
    Ok((nums, false, None))
}
fn expand_bare(tok: &str, offset: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (nums, full, v) = partial(tok, offset)?;
    if full { add_cmp(arm,Cmp::Eq(v.unwrap())); return Ok(()); }
    if nums.len() == 0 { return Ok(()); }
    let lo=padded(&nums);
    add_cmp(arm,Cmp::Ge(lo.clone()));

        let hi=next_bump(lo, nums.len()-1).ok_or(RangeError::Syntax{at:offset})?;
        arm.cmps.push(Cmp::Lt(hi));
    }
    Ok(())
}
fn expand_tilde_caret(tok: &str, offset: usize, caret: bool, arm: &mut Arm) -> Result<(), RangeError> {
    let (nums, full, v) = partial(tok, offset)?;
    if nums.is_empty() { return Err(RangeError::Syntax { at: offset-1 }); }
    let lo = if full { v.unwrap() } else { padded(&nums) };
    add_cmp(arm,Cmp::Ge(lo.clone()));
    let field = if caret {
        if lo.major > 0 || nums.len()==1 { 0 } else if lo.minor > 0 || nums.len()==2 { 1 } else { 2 }
    } else if nums.len()==1 { 0 } else { 1 };
    let hi=next_bump(lo,field).ok_or(RangeError::Syntax{at:offset-1})?;
    arm.cmps.push(Cmp::Lt(hi));
    Ok(())
}

fn parse_arm(s:&str,base:usize,arm:&mut Arm)->Result<(),RangeError>{

    }
    Ok(())
}

    for (pi,p) in packages.iter().enumerate(){let vs=&parsed[pi];let qualifies:Vec<usize>=(0..vs.len()).filter(|&i|p.constraints.iter().all(|c|eval_expr(&c.expr,&vs[i],include_prerelease))).collect();if qualifies.is_empty(){let mut diag=Vec::new();if !p.candidates.is_empty(){let n=p.constraints.len();'outer:for k in 1..=n{let mut sets=Vec::new();combinations(n,k,0,&mut Vec::new(),&mut sets);let mut choices=Vec::new();for set in sets{if !(0..vs.len()).any(|i|set.iter().all(|&j|eval_expr(&p.constraints[j].expr,&vs[i],include_prerelease))){let mut ls=set.iter().map(|&j|p.constraints[j].label.clone()).collect::<Vec<_>>();ls.sort();choices.push(ls);}}if !choices.is_empty(){choices.sort();diag=choices.remove(0);break 'outer;}}}return Err(SelectError::NoSolution{package:p.name.clone(),labels:diag});}
        let stable_exists=preference==Preference::StableFirst&&qualifies.iter().any(|&i|vs[i].pre.is_empty());
        let mut best=qualifies[0];for &i in &qualifies[1..]{if stable_exists&&vs[i].pre.is_empty()!=vs[best].pre.is_empty(){if vs[i].pre.is_empty(){best=i;}continue;}let o=compare_versions(&vs[i],&vs[best]);let better=match preference{Preference::Lowest=>o==Ordering::Less,_=>o==Ordering::Greater};if better{best=i;}}
        result.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }Ok(result)
}
