use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut n = 0u64;
    for &b in &bytes[start..*pos] {
        n = n.checked_mul(10).and_then(|x| x.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(n)
}

fn parse_identifiers(s: &str, base: usize, prerelease: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut start = 0usize;
    loop {
        let end = bytes[start..].iter().position(|&b| b == b'.')
            .map(|i| start + i).unwrap_or(bytes.len());
        if end == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
        let mut numeric = true;
        for i in start..end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: base + i });
            }
            if !is_digit(b) {
                numeric = false;
            }
        }
        if prerelease && numeric && end - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        out.push(s[start..end].to_string());
        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0;
    let major = parse_core_number(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let minor = parse_core_number(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let patch = parse_core_number(bytes, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let start = pos + 1;
        let end = bytes[start..].iter().position(|&b| b == b'+')
            .map(|i| start + i).unwrap_or(bytes.len());
        pre = parse_identifiers(&s[start..end], start, true)?;
        pos = end;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        let start = pos + 1;
        build = parse_identifiers(&s[start..], start, false)?;
        pos = bytes.len();
    }
    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch)).then_with(|| {
        match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric_identifier(x);
                    let yn = is_numeric_identifier(y);
                    let c = match (xn, yn) {
                        (true, true) => cmp_numeric_str(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if c != Ordering::Equal { return c; }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        }
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

fn spaces(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn parse_full_at(op: Op, token: &str, at: usize, operand: usize) -> Result<Comparator, RangeError> {
    if operand >= token.len() || !token.as_bytes()[operand].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let v = parse_version(&token[operand..]).map_err(|error| RangeError::Version { at: at + operand, error })?;
    Ok(Comparator { op, anchor: !v.pre.is_empty(), version: v })
}

#[derive(Clone)]
enum Operand { Full(Version), Partial(Vec<Option<u64>>, usize) }

fn parse_operand(s: &str, at: usize) -> Result<Operand, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parse_version(s).map(Operand::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")) {
        return parse_version(s).map(Operand::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut vals = Vec::new();
    let mut wildcard = false;
    for f in fields {
        if f.is_empty() { return Err(RangeError::Syntax { at }); }
        if matches!(f, "*" | "x" | "X") {
            wildcard = true;
            vals.push(None);
        } else {
            if wildcard || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            vals.push(Some(n));
        }
    }
    let k = vals.iter().take_while(|x| x.is_some()).count();
    Ok(Operand::Partial(vals, k))
}

fn core_version(core: (u64,u64,u64), pre: Vec<String>) -> Version {
    Version { major: core.0, minor: core.1, patch: core.2, pre, build: Vec::new() }
}
fn add_cmp(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !arm.anchors.contains(&(v.major,v.minor,v.patch)) {
        arm.anchors.push((v.major,v.minor,v.patch));
    }
    arm.comps.push(Comparator { op, version: v, anchor });
}
fn next_upper(core: (u64,u64,u64), field: usize, at: usize) -> Result<Version, RangeError> {
    let mut c = core;
    let target = match field { 0 => &mut c.0, 1 => &mut c.1, _ => &mut c.2 };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field { 0 => { c.1=0; c.2=0; }, 1 => c.2=0, _=>{} }
    Ok(core_version(c, vec!["0".into()]))
}
fn padded(vals: &[Option<u64>]) -> (u64,u64,u64) {
    (vals.get(0).and_then(|x|*x).unwrap_or(0),
     vals.get(1).and_then(|x|*x).unwrap_or(0),
     vals.get(2).and_then(|x|*x).unwrap_or(0))
}
fn expand_simple(arm: &mut Arm, token: &str, at: usize) -> Result<(), RangeError> {
    let (kind, operand, off) = if token.starts_with('^') { ('^', &token[1..], 1) }
        else if token.starts_with('~') { ('~', &token[1..], 1) } else { (' ', token, 0) };
    let parsed = parse_operand(operand, at + off)?;
    match parsed {
        Operand::Full(v) if kind == ' ' => add_cmp(arm, Op::Eq, v.clone(), !v.pre.is_empty()),
        Operand::Full(v) => {
            let c=(v.major,v.minor,v.patch);
            add_cmp(arm, Op::Ge, v.clone(), !v.pre.is_empty());
            let k=3;
            let field=if kind=='^' { if v.major>0 {0} else if v.minor>0 {1} else {2} }
                else {1};
            let upper=next_upper(c,field,at+off)?;
            add_cmp(arm,Op::Lt,upper,false);
            let _=k;
        }
        Operand::Partial(vals,k) => {
            if kind != ' ' && k==0 { return Err(RangeError::Syntax { at }); }
            if k==0 { return Ok(()); }
            let c=padded(&vals);
            add_cmp(arm,Op::Ge,core_version(c,vec![]),false);
            let field=if kind=='^' {
                if c.0>0 || k==1 {0} else if c.1>0 || k==2 {1} else {2}
            } else if kind=='~' {
                if k==1 {0} else {1}
            } else if k==1 {0} else {1};
            let upper=next_upper(c,field,at+off)?;
            add_cmp(arm,Op::Lt,upper,false);
        }
    }
    Ok(())
}

fn parse_arm(tokens: &[(String,usize)]) -> Result<Arm,RangeError> {
    let mut arm=Arm { comps:Vec::new(), anchors:Vec::new() };
    if let Some((i,_))=tokens.iter().enumerate().find(|(_,t)| t.0=="-") {
        if tokens.len()!=3 || i!=1 || tokens[0].0=="-" || tokens[2].0=="-" {
            return Err(RangeError::Syntax { at: tokens[i].1 });
        }
        let (lo,la)=(&tokens[0].0,tokens[0].1);
        let (hi,ha)=(&tokens[2].0,tokens[2].1);
        let l=parse_operand(lo,la)?;
        let h=parse_operand(hi,ha)?;
        let (lv,lk)=match l { Operand::Full(v)=>(v,3), Operand::Partial(vals,k)=>(core_version(padded(&vals),vec![]),k) };
        let (hv,hk)=match h { Operand::Full(v)=>(v,3), Operand::Partial(vals,k)=>(core_version(padded(&vals),k),k) };
        if lk==0 || hk==0 { return Err(RangeError::Syntax { at: if lk==0 {la} else {ha} }); }
        add_cmp(&mut arm,Op::Ge,lv.clone(),!lv.pre.is_empty());
        if matches!(parse_operand(hi,ha)?,Operand::Full(_)) {
            add_cmp(&mut arm,Op::Le,hv.clone(),!hv.pre.is_empty());
        } else {
            let field=if hk==1 {0}else{1};
            add_cmp(&mut arm,Op::Lt,next_upper((hv.major,hv.minor,hv.patch),field,ha)?,false);
        }
        return Ok(arm);
    }
    for (t,at) in tokens {
        if t=="||" { return Err(RangeError::Syntax { at:*at }); }
        let b=t.as_bytes();
        if b.is_empty() { continue; }
        if matches!(b[0],b'<'|b'>'|b'=') {
            let (op,n)=if b.starts_with(b"<="){(Op::Le,2)}else if b.starts_with(b">="){(Op::Ge,2)}
                else if b[0]==b'<' {(Op::Lt,1)} else if b[0]==b'>' {(Op::Gt,1)} else {(Op::Eq,1)};
            let c=parse_full_at(op,t,*at,n)?;
            if c.anchor { arm.anchors.push((c.version.major,c.version.minor,c.version.patch)); }
            arm.comps.push(c);
        } else if matches!(b[0],b'^'|b'~'|b'*'|b'x'|b'X') || b[0].is_ascii_digit() {
            expand_simple(&mut arm,t,*at)?;
        } else {
            return Err(RangeError::Syntax { at:*at });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes=s.as_bytes();
    let mut tokens=Vec::new();
    let mut i=0;
    while i<bytes.len() {
        if spaces(bytes[i]) { i+=1; continue; }
        let start=i;
        while i<bytes.len() && !spaces(bytes[i]) { i+=1; }
        tokens.push((s[start..i].to_string(),start));
    }
    if tokens.is_empty() { return Err(RangeError::Empty { at:0 }); }
    let mut arms=Vec::new();
    let mut current=Vec::new();
    let mut j=0;
    while j<tokens.len() {
        if tokens[j].0=="||" {
            if current.is_empty() { return Err(RangeError::Empty { at:tokens[j].1 }); }
            arms.push(parse_arm(&current)?);
            current.clear();
            j+=1;
            if j==tokens.len() { return Err(RangeError::Empty { at:s.len() }); }
        } else {
            current.push(tokens[j].clone()); j+=1;
        }
    }
    if current.is_empty() { return Err(RangeError::Empty { at:s.len() }); }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn cmp_satisfies(v:&Version,c:&Comparator)->bool {
    let o=compare_versions(v,&c.version);
    match c.op { Op::Eq=>o==Ordering::Equal, Op::Lt=>o==Ordering::Less, Op::Le=>o!=Ordering::Greater, Op::Gt=>o==Ordering::Greater, Op::Ge=>o!=Ordering::Less }
}
fn arm_matches(v:&Version,a:&Arm,include:bool)->bool {
    (include || v.pre.is_empty() || a.anchors.contains(&(v.major,v.minor,v.patch)))
        && a.comps.iter().all(|c|cmp_satisfies(v,c))
}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool {
    r.arms.iter().any(|a|arm_matches(v,a,include_prerelease))
}
pub fn matches(v:&Version,r:&Range)->bool { matches_with(v,r,false) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }
fn extrema(candidates:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error|CandidateError{index:i,error})?);
    }
    let mut best:Option<usize>=None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i],r) { continue; }
        if let Some(j)=best {
            let cmp=compare_versions(&parsed[i],&parsed[j]);
            if (max && cmp==Ordering::Greater)||(!max&&cmp==Ordering::Less) {best=Some(i);}
        } else {best=Some(i);}
    }
    Ok(best.map(|i|candidates[i].clone()))
}
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extrema(candidates,r,true)}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extrema(candidates,r,false)}

pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool {
    let mut pool=Vec::new();
    for x in 0..=2 { let _=x; }
    for aa in &a.arms { for bb in &b.arms {
        for c in [&aa.comps,&bb.comps] { for cmp in c { pool.push(cmp.version.clone()); } }
        let candidates=pool.clone();
        for v in candidates {
            if arm_matches(&v,aa,include_prerelease)&&arm_matches(&v,bb,include_prerelease){return true;}
            if !v.pre.is_empty() { continue; }
            let mut p=v.clone(); p.pre=vec!["0".into()];
            if arm_matches(&p,aa,include_prerelease)&&arm_matches(&p,bb,include_prerelease){return true;}
        }
    }}
    false
}

pub fn canonical(r:&Range)->String {
    let mut out=Vec::new();
    for a in &r.arms {
        let mut lower:Option<(Version,bool)>=None;
        let mut upper:Option<(Version,bool)>=None;
        for c in &a.comps {
            let v=c.version.clone();
            match c.op {
                Op::Eq => { lower=Some((v.clone(),true)); upper=Some((v,true)); }
                Op::Ge|Op::Gt => {
                    let inc=matches!(c.op,Op::Ge);
                    if lower.as_ref().map_or(true,|(x,xi)| compare_versions(&v,x)==Ordering::Greater || (compare_versions(&v,x)==Ordering::Equal && !inc && *xi)) { lower=Some((v,inc)); }
                }
                Op::Le|Op::Lt => {
                    let inc=matches!(c.op,Op::Le);
                    if upper.as_ref().map_or(true,|(x,xi)| compare_versions(&v,x)==Ordering::Less || (compare_versions(&v,x)==Ordering::Equal && !inc && *xi)) { upper=Some((v,inc)); }
                }
            }
        }
        let mut anchors=a.anchors.clone(); anchors.sort();
        anchors.dedup();
        let astr=if anchors.is_empty(){"-".into()}else{anchors.iter().map(|(x,y,z)|format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")};
        let bound=|b:Option<(Version,bool)>,lo:bool| -> String {
            match b { None=>"*".into(), Some((mut v,inc)) => {
                v.build.clear();
                let text=format!("{}{}{}",v.major,format!(".{}",v.minor),format!(".{}",v.patch))
                    + if v.pre.is_empty(){""}else{&format!("-{}",v.pre.join("."))};
                format!("{}{}",if lo {if inc{">="}else{">"}}else{if inc{"<="}else{"<"}},text)
            }}
        };
        out.push(format!("[{};{};pre={}]",bound(lower,true),bound(upper,false),astr));
    }
    out.sort(); out.dedup(); out.join(" || ")
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label:String, pub expr:SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name:String, pub candidates:Vec<String>, pub constraints:Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package:String, pub version:String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name:String },
    DuplicateLabel { package:String, label:String },
    InvalidCandidate { package:String, index:usize, error:VersionError },
    InvalidRange { package:String, label:String, path:Vec<usize>, error:RangeError },
    NoSolution { package:String, labels:Vec<String> },
}

fn validate_expr(expr:&SetExpr,package:&str,label:&str,path:&mut Vec<usize>,out:&mut Vec<Range>)->Result<(),SelectError>{
    match expr {
        SetExpr::Range(s)=>out.push(parse_range(s).map_err(|error|SelectError::InvalidRange{package:package.into(),label:label.into(),path:path.clone(),error})?),
        SetExpr::Union(xs)|SetExpr::Intersection(xs)=>for (i,x) in xs.iter().enumerate(){path.push(i);validate_expr(x,package,label,path,out)?;path.pop();}
    }
    Ok(())
}
fn expr_matches(expr:&SetExpr,ranges:&mut std::vec::IntoIter<Range>,v:&Version,include:bool)->bool{
    match expr {
        SetExpr::Range(_)=>ranges.next().map_or(false,|r|matches_with(v,&r,include)),
        SetExpr::Union(xs)=>xs.iter().any(|x|expr_matches(x,ranges,v,include)),
        SetExpr::Intersection(xs)=>xs.iter().all(|x|expr_matches(x,ranges,v,include)),
    }
}
fn conflict_labels(pkg:&Package,vers:&[Version],ranges:&[Vec<Range>],include:bool)->Vec<String>{
    let n=pkg.constraints.len();
    if vers.is_empty(){return vec![];}
    for size in 1..=n {
        let mut comb=Vec::new();
        fn search(start:usize,size:usize,comb:&mut Vec<usize>,n:usize,pkg:&Package,vers:&[Version],ranges:&[Vec<Range>],include:bool,best:&mut Option<Vec<String>>){
            if comb.len()==size {
                let satisfiable=vers.iter().enumerate().any(|(vi,v)|comb.iter().all(|&ci|{
                    let mut it=ranges[ci].clone().into_iter();
                    expr_matches(&pkg.constraints[ci].expr,&mut it,v,include)
                }));
                if !satisfiable {
                    let mut labels=comb.iter().map(|&i|pkg.constraints[i].label.clone()).collect::<Vec<_>>();
                    labels.sort_by(|a,b|a.as_bytes().cmp(b.as_bytes()));
                    if best.as_ref().map_or(true,|x|labels.as_slice()<x.as_slice()){*best=Some(labels);}
                }
                return;
            }
            for i in start..n { comb.push(i);search(i+1,size,comb,n,pkg,vers,ranges,include,best);comb.pop(); }
        }
        let mut best=None; search(0,size,&mut comb,n,pkg,vers,ranges,include,&mut best);
        if let Some(x)=best{return x;}
    }
    Vec::new()
}
pub fn select_packages(packages:&[Package],preference:Preference,include_prerelease:bool)->Result<Vec<Selection>,SelectError>{
    let mut names=std::collections::HashSet::new();
    for p in packages {if !names.insert(p.name.clone()){return Err(SelectError::DuplicatePackage{name:p.name.clone()});}}
    for p in packages {
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints {if !labels.insert(c.label.clone()){return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()});}}
    }
    let mut all_versions=Vec::new();
    for p in packages {
        let mut vs=Vec::new();
        for (i,s) in p.candidates.iter().enumerate(){vs.push(parse_version(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);}
        all_versions.push(vs);
    }
    let mut all_ranges=Vec::new();
    for p in packages {
        let mut pranges=Vec::new();
        for c in &p.constraints {
            let mut rs=Vec::new(); let mut path=Vec::new();
            validate_expr(&c.expr,&p.name,&c.label,&mut path,&mut rs)?;
            pranges.push(rs);
        }
        all_ranges.push(pranges);
    }
    let mut result=Vec::new();
    for (pi,p) in packages.iter().enumerate() {
        let vers=&all_versions[pi]; let ranges=&all_ranges[pi];
        let mut best:Option<usize>=None;
        for i in 0..vers.len() {
            let accepted=p.constraints.iter().enumerate().all(|(ci,c)|{
                let mut it=ranges[ci].clone().into_iter();
                expr_matches(&c.expr,&mut it,&vers[i],include_prerelease)
            });
            if !accepted {continue;}
            if let Some(j)=best {
                let cmp=compare_versions(&vers[i],&vers[j]);
                let replace=match preference {
                    Preference::Highest=>cmp==Ordering::Greater,
                    Preference::Lowest=>cmp==Ordering::Less,
                    Preference::StableFirst=>{
                        let si=vers[i].pre.is_empty(); let sj=vers[j].pre.is_empty();
                        (si&&!sj)||(si==sj&&cmp==Ordering::Greater)
                    }
                };
                if replace {best=Some(i);}
            } else {best=Some(i);}
        }
        if let Some(i)=best {result.push(Selection{package:p.name.clone(),version:p.candidates[i].clone()});}
        else {return Err(SelectError::NoSolution{package:p.name.clone(),labels:conflict_labels(p,vers,ranges,include_prerelease)});}
    }
    Ok(result)
}
