```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy)]
    enum Quote {
        None,
        Single(usize),
        Double(usize),
    }

    fn is_separator(ch: char) -> bool {
        matches!(ch, ' ' | '\t' | '\r' | '\n')
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    current.push(ch);
                }
            }
            Quote::Double(_) => match ch {
                '"' => quote = Quote::None,
                '\\' => match chars.next() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, next)) if matches!(next, '"' | '\\' | '$') => current.push(next),
                    Some((_, '\n')) => {}
                    Some((_, next)) => {
                        current.push('\\');
                        current.push(next);
                    }
                },
                _ => current.push(ch),
            },
            Quote::None => match ch {
                '\'' => {
                    active = true;
                    quote = Quote::Single(pos);
                }
                '"' => {
                    active = true;
                    quote = Quote::Double(pos);
                }
                '\\' => match chars.next() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, '\n')) => {}
                    Some((_, next)) => {
                        active = true;
                        current.push(next);
                    }
                },
                ch if is_separator(ch) => {
                    if active {
                        words.push(std::mem::take(&mut current));
                        active = false;
                    }
                }
                _ => {
                    active = true;
                    current.push(ch);
                }
            },
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }

    Ok(words)
}
```
