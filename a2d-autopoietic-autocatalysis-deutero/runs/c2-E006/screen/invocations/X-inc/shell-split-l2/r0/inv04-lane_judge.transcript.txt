fn expand_name(input: &str, name_start: usize, env: &[(&str, &str)]) -> (String, usize) {
    let mut end = name_start;
    while let Some((c, next)) = find_char(input, end) {
        if is_name_continue(c) {
            end = next;
        } else {
            break;
        }
    }
    let name = &input[name_start..end];
    (lookup(name, env).unwrap_or("").to_string(), end)
}

fn expand_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar_pos + 2; // "${" are ASCII.
    let mut scan = content_start;
    let mut closing = None;

    while scan < input.len() {
        let (c, next) = find_char(input, scan).expect("valid UTF-8 boundary");
        if c == '\\' {
            if next < input.len() {
                let (_, after) = find_char(input, next).expect("valid UTF-8 boundary");
                scan = after;
            } else {
                scan = next;
            }
        } else if c == '}' {
            closing = Some(scan);
            break;
        } else {
            scan = next;
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let content = &input[content_start..close];

    let name_end = content
        .char_indices()
        .find_map(|(offset, c)| {
            if offset == 0 {
                if is_name_start(c) {
                    None
                } else {
                    Some(offset)
                }
            } else if is_name_continue(c) {
                None
            } else {
                Some(offset)
            }
        })
        .unwrap_or(content.len());

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name = &content[..name_end];
    let rest = &content[name_end..];

    let result = if rest.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((result, close + 1))
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut pos = 0;

    while pos < text.len() {
        let (c, next) = find_char(text, pos).expect("valid UTF-8 boundary");
        if c == '\\' && next < text.len() {
            let (escaped, after) = find_char(text, next).expect("valid UTF-8 boundary");
            decoded.push(escaped);
            pos = after;
        } else {
            decoded.push(c);
            pos = next;
        }
    }

    decoded
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut pos = 0;

    while pos < input.len() {
        let (c, next) = next_char(input, pos).unwrap();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                pos = next;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next;
                } else if c == '\\' {
                    let slash_pos = pos;
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let (escaped, after) = next_char(input, next).unwrap();
                    match escaped {
                        '"' | '\\' | '$' => word.push(escaped),
                        '\n' => {}
                        _ => {
                            word.push('\\');
                            word.push(escaped);
                        }
                    }
                    pos = after;
                } else if c == '$' {
                    let (expanded, after) = parse_expansion(input, pos, env)?;
                    word.push_str(&expanded);
                    active = true;
                    pos = after;
                } else {
                    word.push(c);
                    pos = next;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    pos = next;
                } else if c == '\'' {
                    quote = Quote::Single(pos);
                    active = true;
                    pos = next;
                } else if c == '"' {
                    quote = Quote::Double(pos);
                    active = true;
                    pos = next;
                } else if c == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let (escaped, after) = next_char(input, next).unwrap();
                    if escaped != '\n' {
                        word.push(escaped);
                        active = true;
                    }
                    pos = after;
                } else if c == '$' {
                    let (expanded, after) = parse_expansion(input, pos, env)?;

                    pos = after;
                } else {
                    word.push(c);

    }

        words.push(word);
    }
    Ok(words)

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok(("$".to_string(), after_dollar));
    }

    let (first, first_end) = next_char(input, after_dollar).unwrap();

    if first == '{' {
        let mut cursor = first_end;
        let mut close = None;
        while cursor < input.len() {
            let (c, end) = next_char(input, cursor).unwrap();
            if c == '\\' {
                if end < input.len() {
                    let (_, after_escaped) = next_char(input, end).unwrap();
                    cursor = after_escaped;
                } else {
                    cursor = end;
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor = end;
            }
        }

        let close_pos = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[first_end..close_pos];

        let (name_end, name) = parse_braced_name(contents)
            .ok_or(SplitError::InvalidExpansion { pos: dollar })?;

        let selected = if name_end == contents.len() {
            lookup(name, env).unwrap_or("")
        } else if contents[name_end..].starts_with(":-") {
            let default_start = name_end + 2;
            let value = lookup(name, env).unwrap_or("");
            if !value.is_empty() {
                value
            } else {
                let decoded = decode_default(&contents[default_start..]);
                return Ok((decoded, close_pos + 1));
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((selected.to_string(), close_pos + 1))
    } else if is_name_start(first) {
        let mut end = first_end;
        while end < input.len() {
            let (c, after) = next_char(input, end).unwrap();
            if is_name_continue(c) {
                end = after;
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((lookup(name, env).unwrap_or("").to_string(), end))
    } else {
        Ok(("$".to_string(), after_dollar))
    }
}

fn parse_braced_name(contents: &str) -> Option<(usize, &str)> {
    let mut chars = contents.char_indices();
    let (_, first) = chars.next()?;
    if !is_name_start(first) {
        return None;
    }

    let mut end = first.len_utf8();
    for (offset, c) in chars {
        if is_name_continue(c) {
            end = offset + c.len_utf8();
        } else {
            break;
        }
    }

    Some((end, &contents[..end]))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut pos = 0;

    while pos < text.len() {
        let (c, next) = next_char(text, pos).unwrap();
        if c == '\\' && next < text.len() {
            let (escaped, after) = next_char(text, next).unwrap();
            result.push(escaped);
            pos = after;
        } else {
            result.push(c);
            pos = next;
        }
    }

    result
}

fn parse_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let bytes = input.as_bytes();
    let mut cursor = dollar_pos + 2;

    // Locate the first unescaped closing brace. Backslash skips one scalar.
    let close = loop {
        if cursor >= bytes.len() {
            return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
        }

        let c = input[cursor..].chars().next().expect("valid UTF-8 boundary");
        if c == '}' {
            break cursor;
        }
        if c == '\\' {
            cursor += 1;
            if cursor < bytes.len() {
                let next = input[cursor..].chars().next().expect("valid UTF-8 boundary");
                cursor += next.len_utf8();
            }
        } else {
            cursor += c.len_utf8();
        }
    };

    let contents = &input[dollar_pos + 2..close];
    let mut name_end = 0;
    let mut chars = contents.char_indices();

    match chars.next() {
        Some((_, c)) if is_name_start(c) => {
            name_end = c.len_utf8();
        }
        _ => return Err(SplitError::InvalidExpansion { pos: dollar_pos }),
    }

    for (offset, c) in contents[name_end..].char_indices() {
        if is_name_continue(c) {
            name_end += c.len_utf8();
        } else {
            let _ = offset;
            break;
        }
    }

    let name = &contents[..name_end];
    let remainder = &contents[name_end..];
    let value = match remainder {
        "" => lookup(name, env).unwrap_or("").to_string(),
        _ if remainder.starts_with(":-") => {
            let current = lookup(name, env).unwrap_or("");
            if !current.is_empty() {
                current.to_string()
            } else {
                decode_default(&remainder[2..])
            }
        }
        _ => return Err(SplitError::InvalidExpansion { pos: dollar_pos }),
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let bytes = input.as_bytes();
    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < bytes.len() {
        let c = input[cursor..].chars().next().expect("valid UTF-8 boundary");
        let width = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                cursor += width;
            }
            Quote::Double(open_pos) => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        cursor += width;
                    }
                    '\\' => {
                        let escape_pos = cursor;
                        cursor += 1;
                        if cursor >= bytes.len() {
                            return Err(SplitError::TrailingEscape { pos: escape_pos });
                        }
                        let next = input[cursor..].chars().next().expect("valid UTF-8 boundary");
                        match next {
                            '"' | '\\' | '$' => {
                                word.push(next);
                                cursor += next.len_utf8();
                            }
                            '\n' => cursor += 1,
                            _ => word.push('\\'),
                        }
                    }
                    '$' => {
                        if cursor + 1 < bytes.len() && bytes[cursor + 1] == b'{' {
                            let (value, next) = parse_braced(input, cursor, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = next;
                        } else if cursor + 1 < bytes.len() {
                            let next = input[cursor + 1..]
                                .chars()
                                .next()
                                .expect("valid UTF-8 boundary");
                            if is_name_start(next) {
                                let start = cursor + 1;
                                let mut end = start + next.len_utf8();
                                while end < bytes.len() {
                                    let ch = input[end..].chars().next().expect("valid UTF-8 boundary");
                                    if is_name_continue(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[start..end];
                                word.push_str(lookup(name, env).unwrap_or(""));
                                active = true;
                                cursor = end;
                            } else {
                                word.push('$');
                                cursor += 1;
                            }
                        } else {
                            word.push('$');
                            cursor += 1;
                        }
                    }
                    _ => {
                        word.push(c);
                        cursor += width;
                    }
                }
                let _ = open_pos;
            }
            Quote::None => {
                match c {
                    '\'' => {
                        active = true;
                        quote = Quote::Single(cursor);
                        cursor += width;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double(cursor);
                        cursor += width;
                    }
                    '\\' => {
                        let escape_pos = cursor;
                        cursor += 1;
                        if cursor >= bytes.len() {
                            return Err(SplitError::TrailingEscape { pos: escape_pos });
                        }
                        let next = input[cursor..].chars().next().expect("valid UTF-8 boundary");
                        if next == '\n' {
                            cursor += 1;
                        } else {
                            word.push(next);
                            active = true;
                            cursor += next.len_utf8();
                        }
                    }
                    '$' => {
                        if cursor + 1 < bytes.len() && bytes[cursor + 1] == b'{' {
                            let (value, next) = parse_braced(input, cursor, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = next;
                        } else if cursor + 1 < bytes.len() {
                            let next = input[cursor + 1..]
                                .chars()
                                .next()
                                .expect("valid UTF-8 boundary");
                            if is_name_start(next) {
                                let start = cursor + 1;
                                let mut end = start + next.len_utf8();
                                while end < bytes.len() {
                                    let ch = input[end..].chars().next().expect("valid UTF-8 boundary");
                                    if is_name_continue(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[start..end];
                                word.push_str(lookup(name, env).unwrap_or(""));
                                active = true;
                                cursor = end;
                            } else {
                                word.push('$');
                                active = true;
                                cursor += 1;
                            }
                        } else {
                            word.push('$');
                            active = true;
                            cursor += 1;
                        }
                    }
                    _ if is_separator(c) => {
                        if active {
                            words.push(std::mem::take(&mut word));
                            active = false;
                        }
                        cursor += width;
                    }
                    _ => {
                        word.push(c);
                        active = true;
                        cursor += width;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
