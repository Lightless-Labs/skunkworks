#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { start: usize },
    Double { start: usize },
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn find_char(input: &str, mut pos: usize) -> Option<(char, usize)> {
    let c = input.get(pos..)?.chars().next()?;
    let next = pos + c.len_utf8();
    Some((c, next))
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find_map(|(key, value)| {
        if *key == name {
            Some(*value)
        } else {
            None
        }
    })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut pos = 0;

    while pos < input.len() {
        let (c, next) = find_char(input, pos).expect("valid UTF-8 boundary");

        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                pos = next;
            }
            Quote::Double { .. } => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next;
                } else if c == '\\' {
                    let escape_pos = pos;
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let (escaped, after) = find_char(input, next).expect("valid UTF-8 boundary");
                    if escaped == '"' || escaped == '\\' || escaped == '$' {
                        word.push(escaped);
                        pos = after;
                    } else if escaped == '\n' {
                        pos = after;
                    } else {
                        word.push('\\');
                        pos = next;
                    }
                } else if c == '$' {
                    active = true;
                    if next < input.len() {
                        let (following, _) = find_char(input, next).expect("valid UTF-8 boundary");
                        if following == '{' {
                            let (expanded, after) = expand_braced(input, pos, env)?;
                            word.push_str(&expanded);
                            pos = after;
                        } else if is_name_start(following) {
                            let (expanded, after) = expand_name(input, next, env);
                            word.push_str(&expanded);
                            pos = after;
                        } else {
                            word.push('$');
                            pos = next;
                        }
                    } else {
                        word.push('$');
                        pos = next;
                    }
                } else {
                    word.push(c);
                    pos = next;
                }
            }
            Quote::None => {
                if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    pos = next;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single { start: pos };
                    pos = next;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double { start: pos };
                    pos = next;
                } else if c == '\\' {
                    let escape_pos = pos;
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let (escaped, after) = find_char(input, next).expect("valid UTF-8 boundary");
                    if escaped == '\n' {
                        pos = after;
                    } else {
                        active = true;
                        word.push(escaped);
                        pos = after;
                    }
                } else if c == '$' {
                    active = true;
                    if next < input.len() {
                        let (following, _) = find_char(input, next).expect("valid UTF-8 boundary");
                        if following == '{' {
                            let (expanded, after) = expand_braced(input, pos, env)?;
                            word.push_str(&expanded);
                            pos = after;
                        } else if is_name_start(following) {
                            let (expanded, after) = expand_name(input, next, env);
                            word.push_str(&expanded);
                            pos = after;
                        } else {
                            word.push('$');
                            pos = next;
                        }
                    } else {
                        word.push('$');
                        pos = next;
                    }
                } else {
                    active = true;
                    word.push(c);
                    pos = next;
                }
            }
        }
    }

    match quote {
        Quote::Single { start } => return Err(SplitError::UnterminatedSingle { pos: start }),
        Quote::Double { start } => return Err(SplitError::UnterminatedDouble { pos: start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand_name(input: &str, name_start: usize, env: &[(&str, &str)]) -> (String, usize) {
    let mut end = name_start;
    while let Some((c, next)) = find_char(input, end) {
        if is_name_continue(c) {
            end = next;
        } else {
            break;
        }
    }
    let name = &input[name_start..end];
    (lookup(name, env).unwrap_or("").to_string(), end)
}

fn expand_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar_pos + 2; // "${" are ASCII.
    let mut scan = content_start;
    let mut closing = None;

    while scan < input.len() {
        let (c, next) = find_char(input, scan).expect("valid UTF-8 boundary");
        if c == '\\' {
            if next < input.len() {
                let (_, after) = find_char(input, next).expect("valid UTF-8 boundary");
                scan = after;
            } else {
                scan = next;
            }
        } else if c == '}' {
            closing = Some(scan);
            break;
        } else {
            scan = next;
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let content = &input[content_start..close];

    let name_end = content
        .char_indices()
        .find_map(|(offset, c)| {
            if offset == 0 {
                if is_name_start(c) {
                    None
                } else {
                    Some(offset)
                }
            } else if is_name_continue(c) {
                None
            } else {
                Some(offset)
            }
        })
        .unwrap_or(content.len());

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name = &content[..name_end];
    let rest = &content[name_end..];

    let result = if rest.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((result, close + 1))
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut pos = 0;

    while pos < text.len() {
        let (c, next) = find_char(text, pos).expect("valid UTF-8 boundary");
        if c == '\\' && next < text.len() {
            let (escaped, after) = find_char(text, next).expect("valid UTF-8 boundary");
            decoded.push(escaped);
            pos = after;
        } else {
            decoded.push(c);
            pos = next;
        }
    }

    decoded
}
