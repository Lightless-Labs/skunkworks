#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    added_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved_ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(&text[..split], 0, 60, &mut minutes)?;
        parse_field(&text[split + 1..], split + 1, 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved_ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved_ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let added_after = self.last_poll;
        let order = self.jobs.len();

        self.reserved_ids.push(id.to_string());
        self.jobs.push(Job {
            id: id.to_string(),
            cron,
            priority,
            policy,
            order,
            added_after,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let start_exclusive = self.last_poll;
        let mut out: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let interval_start = match (start_exclusive, job.added_after) {
                (Some(last), _) => Some(last),
                (None, Some(added_at)) => Some(added_at),
                (None, None) => None,
            };

            let first_minute = match interval_start {
                Some(t) => t / 60_000 + 1,
                None => 0,
            };
            let last_minute = now / 60_000;
            if first_minute > last_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    for minute in first_minute..=last_minute {
                        if job.cron.matches(minute) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    let latest = (first_minute..=last_minute)
                        .rev()
                        .find(|&minute| job.cron.matches(minute));
                    if let Some(minute) = latest {
                        out.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        if minute >= first_minute && minute <= last_minute && job.cron.matches(minute) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        out.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(out.into_iter().map(|item| item.0).collect())
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = field.as_bytes();
    let mut start = 0;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }

        parse_item(&field[start..end], offset + start, max, selected)?;

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let b = item.as_bytes();

    let slash = b.iter().position(|&c| c == b'/');
    if let Some(first) = slash {
        if let Some(second_rel) = b[first + 1..].iter().position(|&c| c == b'/') {
            return Err(CronError::Syntax(offset + first + 1 + second_rel));
        }
    }

    let base_end = slash.unwrap_or(b.len());
    let base = &item[..base_end];

    if let Some(slash_pos) = slash {
        let is_star_or_range = base == "*" || base.contains('-');
        if !is_star_or_range {
            return Err(CronError::Syntax(offset + slash_pos));
        }
    }

    let (base_start, base_finish, dash_pos) = if base == "*" {
        (0, max - 1, None)
    } else if let Some(dash) = base.as_bytes().iter().position(|&c| c == b'-') {
        let left = parse_number(&base[..dash], offset, max)?;
        let right = parse_number(&base[dash + 1..], offset + dash + 1, max)?;
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        (left, right, Some(dash))
    } else {
        let n = parse_number(base, offset, max)?;
        (n, n, None)
    };

    let step = if let Some(slash_pos) = slash {
        parse_step(
            &item[slash_pos + 1..],
            offset + slash_pos + 1,
            max,
        )?
    } else {
        1
    };

    let _ = dash_pos;
    let mut value = base_start;
    while value <= base_finish {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_step(text: &str, offset: usize, max: usize) -> Result<usize, CronError> {
    let value = parse_digits(text, offset)?;
    if value == 0 || value > max {
        return Err(CronError::Step(offset));
    }
    Ok(value)
}

fn parse_number(text: &str, offset: usize, max: usize) -> Result<usize, CronError> {
    let value = parse_digits(text, offset)?;
    if value > max - 1 {
        return Err(CronError::Bounds(offset));
    }
    Ok(value)
}

fn parse_digits(text: &str, offset: usize) -> Result<usize, CronError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(offset + i));
        }
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(offset)),
        };
    }
    Ok(value)
}
