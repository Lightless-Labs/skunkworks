#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged {
        row: usize,
        expected: usize,
        actual: usize,
    },
    InvalidChar {
        row: usize,
        column: usize,
        byte: usize,
    },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    if let Some(row) = rows.iter().position(Vec::is_empty) {
        return Err(TableError::EmptyRow { row });
    }

    let columns = rows[0].len();
    if let Some((row, cells)) = rows
        .iter()
        .enumerate()
        .find(|(_, cells)| cells.len() != columns)
    {
        return Err(TableError::Ragged {
            row,
            expected: columns,
            actual: cells.len(),
        });
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let fixed_overhead = 3 * columns + 1;
    let minimum = fixed_overhead + columns;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            let mut byte = 0;
            for ch in cell.chars() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
                byte += ch.len_utf8();
            }
        }
    }

    let trimmed: Vec<Vec<String>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| cell.trim_end_matches(' ').to_owned())
                .collect()
        })
        .collect();

    let mut widths: Vec<usize> = (0..columns)
        .map(|column| {
            trimmed
                .iter()
                .map(|row| row[column].chars().count())
                .max()
                .unwrap_or(1)
                .max(1)
        })
        .collect();

    if options.max_width != 0 {
        let mut total = fixed_overhead + widths.iter().sum::<usize>();
        while total > options.max_width {
            let widest = widths
                .iter()
                .enumerate()
                .filter(|(_, width)| **width > 1)
                .max_by_key(|(column, width)| (**width, *column))
                .map(|(column, _)| column);

            if let Some(column) = widest {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    append_border(&mut output, &widths);

    for (row_index, row) in trimmed.iter().enumerate() {
        output.push('|');
        for column in 0..columns {
            let content = truncate(&row[column], widths[column]);
            let alignment = options
                .alignments
                .get(column)
                .unwrap_or(&Alignment::Left);
            append_cell(&mut output, &content, widths[column], alignment);
            output.push('|');
        }
        output.push('\n');

        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output, &widths);
        }
    }

    append_border(&mut output, &widths);
    Ok(output)
}

fn append_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for &width in widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');
}

fn truncate(text: &str, width: usize) -> String {
    let count = text.chars().count();
    if count <= width {
        return text.to_owned();
    }

    let mut result: String = text.chars().take(width.saturating_sub(1)).collect();
    result.push('…');
    result
}

fn append_cell(output: &mut String, content: &str, width: usize, alignment: &Alignment) {
    output.push(' ');
    let content_width = content.chars().count();
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    output.extend(std::iter::repeat_n(' ', left));
    output.push_str(content);
    output.extend(std::iter::repeat_n(' ', right));
    output.push(' ');
}
