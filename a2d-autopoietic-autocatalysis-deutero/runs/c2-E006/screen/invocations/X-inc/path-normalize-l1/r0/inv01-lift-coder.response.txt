#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.bytes().position(|byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .bytes()
        .position(|byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.starts_with('/');
    let directory_intent = path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..");

    let mut components: Vec<(String, usize)> = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        if component.is_empty() {
            offset += 1;
            continue;
        }

        match component {
            "." => {}
            ".." => {
                if components
                    .last()
                    .is_some_and(|(previous, _)| previous != "..")
                {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset });
                } else {
                    components.push((component.to_owned(), offset));
                }
            }
            _ => components.push((component.to_owned(), offset)),
        }

        offset += component.len() + 1;
    }

    let joined = components
        .iter()
        .map(|(component, _)| component.as_str())
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
