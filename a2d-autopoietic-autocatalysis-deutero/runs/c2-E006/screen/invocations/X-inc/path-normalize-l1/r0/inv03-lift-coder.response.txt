#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    let bytes = path.as_bytes();
    let absolute = bytes[0] == b'/';
    let directory_intent = bytes[bytes.len() - 1] == b'/';

    let mut components: Vec<(&str, usize)> = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        while index < bytes.len() && bytes[index] == b'/' {
            index += 1;
        }
        if index == bytes.len() {
            break;
        }

        let start = index;
        while index < bytes.len() && bytes[index] != b'/' {
            index += 1;
        }

        let component = &path[start..index];
        if component == "." || component == ".." {
            // Directory intent also applies when the final nonempty component
            // is dot or parent.
        }

        if component == "." {
            // Discard dot components.
        } else if component == ".." {
            if components.last().is_some_and(|(name, _)| *name != "..") {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape { at: start });
            } else {
                components.push((component, start));
            }
        } else {
            components.push((component, start));
        }
    }

    let last_nonempty_is_dot_or_parent = {
        let mut end = bytes.len();
        while end > 0 && bytes[end - 1] == b'/' {
            end -= 1;
        }
        if end == 0 {
            false
        } else {
            let start = bytes[..end]
                .iter()
                .rposition(|&b| b == b'/')
                .map_or(0, |i| i + 1);
            let last = &path[start..end];
            last == "." || last == ".."
        }
    };
    let directory_intent = directory_intent || last_nonempty_is_dot_or_parent;

    let mut result = String::new();
    if absolute {
        result.push('/');
    }
    for (i, (component, _)) in components.iter().enumerate() {
        if i > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
