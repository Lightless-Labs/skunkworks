#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    attempts: u8,
}

pub struct Runner {
    entries: Vec<Entry>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (index, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if ids.insert(job.id.as_str(), index).is_some() {
                return Err(Error::Duplicate(index));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (dep_index, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(job_index, dep_index));
                }
                match ids.get(dep.as_str()) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(job_index, dep_index)),
                }
            }
            dependencies.push(resolved);
        }

        if has_cycle(&dependencies) {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .zip(dependencies)
            .map(|(job, dependencies)| Entry {
                job,
                dependencies,
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            entries,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.entries.iter().any(|entry| {
            matches!(entry.state, State::Waiting { .. } | State::Running { .. })
        }) {
            return Err(Error::Busy);
        }

        for entry in &mut self.entries {
            entry.state = State::Waiting { ready_at: now };
            entry.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (index, entry) in self.entries.iter().enumerate() {
            let ready_at = match entry.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !entry.dependencies.iter().all(|&dep| {
                matches!(self.entries[dep].state, State::Succeeded)
            }) {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (entry.job.priority as u64 + age).min(9) as u8
            };

            if best.map_or(true, |(best_index, best_priority)| {
                effective > best_priority
                    || (effective == best_priority && index < best_index)
            }) {
                best = Some((index, effective));
            }
        }

        let result = if let Some((index, _)) = best {
            let entry = &mut self.entries[index];
            entry.attempts += 1;
            entry.state = State::Running {
                attempt: entry.attempts,
            };
            Some(Ticket {
                id: entry.job.id.clone(),
                attempt: entry.attempts,
            })
        } else {
            None
        };

        self.last_time = now;
        Ok(result)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let index = self
            .entries
            .iter()
            .position(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;

        if !matches!(self.entries[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.entries[index].state = State::Succeeded;
        } else if self.entries[index].attempts == self.entries[index].job.max_attempts {
            self.entries[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let delay = self.retry_delay(self.entries[index].attempts);
            self.entries[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        self.entries
            .iter()
            .find(|entry| entry.job.id == id)
            .map(|entry| entry.state.clone())
            .ok_or(Error::Unknown)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn retry_delay(&self, attempt: u8) -> u64 {
        let mut delay = self.retry_base;
        for _ in 1..attempt {
            if delay >= self.retry_cap {
                return self.retry_cap;
            }
            delay = delay.saturating_mul(2).min(self.retry_cap);
        }
        delay.min(self.retry_cap)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();
            for (index, entry) in self.entries.iter().enumerate() {
                if matches!(entry.state, State::Waiting { .. })
                    && entry.dependencies.iter().any(|&dep| {
                        matches!(self.entries[dep].state, State::Failed | State::Blocked)
                    })
                {
                    newly_blocked.push(index);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for index in newly_blocked {
                self.entries[index].state = State::Blocked;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    (1..=16).contains(&id.len())
        && id.bytes().all(|byte| {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-'
        })
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, dependencies: &[Vec<usize>], marks: &mut [u8]) -> bool {
        if marks[node] == 1 {
            return true;
        }
        if marks[node] == 2 {
            return false;
        }
        marks[node] = 1;
        for &dependency in &dependencies[node] {
            if visit(dependency, dependencies, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    (0..dependencies.len()).any(|node| visit(node, dependencies, &mut marks))
}
