```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|part| !part.is_empty() && *part != ".")
            .map(str::to_string)
            .collect()
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }

        Self::canonical(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            _ => "/".to_string(),
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            String::new()
        } else {
            path.rsplit('/').next().unwrap_or("").to_string()
        }
    }

    fn join_parts(base: &str, suffix: &[String]) -> String {
        let mut parts = Self::components(base);
        parts.extend_from_slice(suffix);
        Self::canonical(&parts)
    }

    fn is_within(path: &str, directory: &str) -> bool {
        if directory == "/" {
            return path.starts_with('/');
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|rest| rest.starts_with('/'))
    }

    fn children(&self, directory: &str) -> Vec<String> {
        let prefix = if directory == "/" {
            "/".to_string()
        } else {
            format!("{directory}/")
        };

        self.entries
            .keys()
            .filter_map(|path| {
                if path == directory || !path.starts_with(&prefix) {
                    return None;
                }
                let remainder = &path[prefix.len()..];
                if remainder.is_empty() || remainder.contains('/') {
                    return None;
                }
                Some(path.clone())
            })
            .collect()
    }

    fn resolve(&self, path: &str, follow_final: bool) -> Result<String, Error> {
        let initial = Self::components(path);
        let mut pending = initial;
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(Self::canonical(&resolved));
            }

            let component = pending.remove(0);
            resolved.push(component);
            let current = Self::canonical(&resolved);
            let is_final = pending.is_empty();

            let entry = match self.entries.get(&current) {
                Some(entry) => entry,
                None => return Err(Error::Missing(current)),
            };

            if let Entry::Link(target) = entry {
                if follow_final || !is_final {
                    if visited.contains(&current) {
                        return Err(Error::Loop(current));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(current));
                    }
                    visited.insert(current.clone());
                    expansions += 1;

                    let mut target_parts = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        Self::components(&Self::parent(&current))
                    };
                    for part in target.split('/') {
                        match part {
                            "" | "." => {}
                            ".." => {
                                target_parts.pop();
                            }
                            _ => target_parts.push(part.to_string()),
                        }
                    }
                    target_parts.extend(pending.into_iter());
                    pending = target_parts;
                    resolved.clear();
                    continue;
                }
            }

            if !is_final && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(current));
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let resolved_parent = self.resolve(&parent, true)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, Self::basename(path))),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn resolved_child(parent: &str, basename: &str) -> String {
        if parent == "/" {
            format!("/{basename}")
        } else {
            format!("{parent}/{basename}")
        }
    }

    fn subtree_paths(&self, root: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| Self::is_within(path, root))
            .cloned()
            .collect()
    }

    fn subtree_size(&self, root: &str) -> u64 {
        self.subtree_paths(root)
            .iter()
            .filter_map(|path| self.entries.get(path))
            .map(|entry| match entry {
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            })
            .sum()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }

        let (parent, basename) = self.resolve_parent(&canonical)?;
        let destination = Self::resolved_child(&parent, &basename);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = match self.resolve(&canonical, true) {
            Ok(path) => path,
            Err(Error::Missing(_)) => {
                let (parent, basename) = self.resolve_parent(&canonical)?;
                let destination = Self::resolved_child(&parent, &basename);
                if destination == "/" {
                    return Err(Error::IsDir("/".to_string()));
                }
                self.entries
                    .insert(destination, Entry::File(data.to_vec()));
                return Ok(());
            }
            Err(error) => return Err(error),
        };

        match self.entries.get_mut(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(bytes)) => {
                *bytes = data.to_vec();
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, basename) = self.resolve_parent(&canonical)?;
                let destination = Self::resolved_child(&parent, &basename);
                self.entries
                    .insert(destination, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut rows: Vec<(String, Kind)> = self
                    .children(&resolved)
                    .iter()
                    .filter_map(|child| {
                        self.entries
                            .get(child)
                            .map(|entry| (Self::basename(child), entry.kind()))
                    })
                    .collect();
                rows.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(rows)
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        Self::validate(to)?;
        let source = Self::normalize(from, &self.cwd);
        let destination = Self::normalize(to, &self.cwd);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::resolved_child(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = Self::resolved_child(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(existing) = &destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moved = self.subtree_paths(&source_path);
        if destination_entry.is_some() {
            self.entries.remove(&destination_path);
        }

        let replacements: Vec<(String, Entry)> = moved
            .iter()
            .filter_map(|old_path| {
                self.entries.get(old_path).cloned().map(|entry| {
                    let suffix = &old_path[source_path.len()..];
                    (format!("{destination_path}{suffix}"), entry)
                })
            })
            .collect();

        for old_path in moved {
            self.entries.remove(&old_path);
        }
        for (new_path, entry) in replacements {
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, basename) = self.resolve_parent(&canonical)?;
        let target = Self::resolved_child(&parent, &basename);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        let descendants = self.children(&target);
        if matches!(entry, Entry::Dir) && !recursive && !descendants.is_empty() {
            return Err(Error::NotEmpty(target));
        }

        if recursive && matches!(entry, Entry::Dir) {
            for path in self.subtree_paths(&target) {
                self.entries.remove(&path);
            }
        } else {
            self.entries.remove(&target);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let canonical = Self::normalize(link, &self.cwd);
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, basename) = self.resolve_parent(&canonical)?;
        let destination = Self::resolved_child(&parent, &basename);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut paths = self.subtree_paths(&resolved);
        paths.sort();
        Ok(paths
            .into_iter()
            .map(|entry_path| {
                let size = match self.entries.get(&entry_path) {
                    Some(Entry::Dir) => self.subtree_size(&entry_path),
                    Some(Entry::File(bytes)) => bytes.len() as u64,
                    Some(Entry::Link(target)) => target.len() as u64,
                    None => 0,
                };
                (entry_path, size)
            })
            .collect())
    }
}
```
