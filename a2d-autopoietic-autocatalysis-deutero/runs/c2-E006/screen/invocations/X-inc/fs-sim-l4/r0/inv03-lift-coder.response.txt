use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|c| !c.is_empty() && *c != ".")
        .fold(Vec::new(), |mut out, c| {
            if c == ".." {
                out.pop();
            } else {
                out.push(c.to_string());
            }
            out
        })
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn path_parts(path: &str) -> Vec<String> {
    components(path)
}

fn join_from(base: &[String], path: &str) -> Vec<String> {
    let mut out = if path.starts_with('/') {
        Vec::new()
    } else {
        base.to_vec()
    };
    for c in path.split('/') {
        if c.is_empty() || c == "." {
            continue;
        }
        if c == ".." {
            out.pop();
        } else {
            out.push(c.to_string());
        }
    }
    out
}

fn parent_parts(parts: &[String]) -> Vec<String> {
    parts[..parts.len().saturating_sub(1)].to_vec()
}

fn contains_path(parent: &str, child: &str) -> bool {
    parent == "/"
        || child == parent
        || child
            .strip_prefix(parent)
            .is_some_and(|rest| rest.starts_with('/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        validate_path(path)?;
        let cwd_parts = path_parts(&self.cwd);
        Ok(join_from(&cwd_parts, path))
    }

    fn resolve_parts(&self, initial: Vec<String>, allow_missing_final: bool) -> Result<Vec<String>, Error> {
        let mut todo = initial;
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;
        let mut idx = 0usize;

        while idx < todo.len() {
            let part = todo[idx].clone();
            resolved.push(part);
            let current = canonical(&resolved);
            let is_final = idx + 1 == todo.len();

            match self.entries.get(&current) {
                None => {
                    if allow_missing_final && is_final {
                        return Ok(resolved);
                    }
                    return Err(Error::Missing(current));
                }
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => {
                    if !is_final {
                        return Err(Error::NotDir(current));
                    }
                }
                Some(Entry::Link(target)) => {
                    if visited.contains(&current) {
                        return Err(Error::Loop(current));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(current));
                    }
                    visited.insert(current.clone());
                    expansions += 1;

                    let physical_parent = parent_parts(&resolved);
                    let mut expanded = join_from(&physical_parent, target);
                    expanded.extend(todo[idx + 1..].iter().cloned());
                    todo = expanded;
                    resolved.clear();
                    idx = 0;
                    continue;
                }
            }
            idx += 1;
        }
        Ok(resolved)
    }

    fn resolve_parent(&self, parts: &[String]) -> Result<Vec<String>, Error> {
        self.resolve_parts(parent_parts(parts), false)
    }

    fn resolve_full(&self, parts: Vec<String>) -> Result<Vec<String>, Error> {
        self.resolve_parts(parts, false)
    }

    fn existing_kind(&self, parts: &[String]) -> Option<Kind> {
        self.entries.get(&canonical(parts)).map(Entry::kind)
    }

    fn ensure_parent_dir(&self, parent: &[String]) -> Result<(), Error> {
        let path = canonical(parent);
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        let parent = self.resolve_parent(&parts)?;
        self.ensure_parent_dir(&parent)?;
        let name = parts.last().cloned();
        if name.is_none() {
            return Err(Error::Exists("/".to_string()));
        }
        let mut dst = parent;
        dst.push(name.unwrap());
        let dst_path = canonical(&dst);
        if self.entries.contains_key(&dst_path) {
            return Err(Error::Exists(dst_path));
        }
        self.entries.insert(dst_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        let resolved = self.resolve_parts(parts, true)?;
        let dst_path = canonical(&resolved);
        match self.entries.get(&dst_path) {
            Some(Entry::Dir) => return Err(Error::IsDir(dst_path)),
            Some(Entry::File(_)) | None => {}
            Some(Entry::Link(_)) => unreachable!(),
        }
        let parent = parent_parts(&resolved);
        self.ensure_parent_dir(&parent)?;
        self.entries.insert(dst_path, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = self.normalize(path)?;
        let resolved = self.resolve_full(parts)?;
        let p = canonical(&resolved);
        match self.entries.get(&p) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(p)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = self.normalize(path)?;
        let resolved = self.resolve_full(parts)?;
        let p = canonical(&resolved);
        match self.entries.get(&p) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(p)),
            None => return Err(Error::Missing(p)),
        }
        let prefix = if p == "/" { "/".to_string() } else { format!("{}/", p) };
        let mut result = Vec::new();
        for (child, entry) in &self.entries {
            if child == &p || !child.starts_with(&prefix) {
                continue;
            }
            let rest = &child[prefix.len()..];
            if !rest.is_empty() && !rest.contains('/') {
                result.push((rest.to_string(), entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        let resolved = self.resolve_full(parts)?;
        let p = canonical(&resolved);
        match self.entries.get(&p) {
            Some(Entry::Dir) => {
                self.cwd = p;
                Ok(())
            }
            _ => Err(Error::NotDir(p)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let src_parts = self.normalize(from)?;
        let dst_parts = self.normalize(to)?;

        let src_parent = self.resolve_parent(&src_parts)?;
        self.ensure_parent_dir(&src_parent)?;
        let src_name = src_parts.last().cloned().ok_or(Error::Root)?;
        let mut src_resolved = src_parent.clone();
        src_resolved.push(src_name);
        let src_path = canonical(&src_resolved);
        let source = self
            .entries
            .get(&src_path)
            .cloned()
            .ok_or_else(|| Error::Missing(src_path.clone()))?;

        if matches!(source, Entry::Dir) && contains_path(&src_path, &self.cwd) {
            return Err(Error::Busy(src_path));
        }

        let dst_parent = self.resolve_parent(&dst_parts)?;
        self.ensure_parent_dir(&dst_parent)?;
        let dst_name = dst_parts.last().cloned().ok_or(Error::Root)?;
        let mut dst_resolved = dst_parent;
        dst_resolved.push(dst_name);
        let dst_path = canonical(&dst_resolved);

        if matches!(self.entries.get(&dst_path), Some(Entry::Dir))
            && contains_path(&dst_path, &self.cwd)
        {
            return Err(Error::Busy(dst_path));
        }

        if src_path == dst_path {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && contains_path(&src_path, &dst_path) {
            return Err(Error::Cycle(dst_path));
        }

        let destination = self.entries.get(&dst_path).cloned();
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(dst_path));
        }

        if let Some(dest) = &destination {
            let src_dir = matches!(source, Entry::Dir);
            let dst_dir = matches!(dest, Entry::Dir);
            if src_dir && !dst_dir {
                return Err(Error::NotDir(dst_path));
            }
            if !src_dir && dst_dir {
                return Err(Error::IsDir(dst_path));
            }
            if src_dir && dst_dir {
                let prefix = format!("{}/", dst_path);
                if self.entries.keys().any(|p| p.starts_with(&prefix)) {
                    return Err(Error::NotEmpty(dst_path));
                }
            }
        }

        let src_prefix = format!("{}/", src_path);
        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| *p == &src_path || p.starts_with(&src_prefix))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if let Some(dest) = destination {
            if matches!(dest, Entry::Dir) {
                self.entries.remove(&dst_path);
            } else {
                self.entries.remove(&dst_path);
            }
        }
        for (p, _) in &moved {
            self.entries.remove(p);
        }
        for (old, entry) in moved {
            let suffix = &old[src_path.len()..];
            self.entries.insert(format!("{}{}", dst_path, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        if parts.is_empty() {
            return Err(Error::Root);
        }
        let parent = self.resolve_parent(&parts)?;
        self.ensure_parent_dir(&parent)?;
        let name = parts.last().cloned().unwrap();
        let mut resolved = parent;
        resolved.push(name);
        let p = canonical(&resolved);
        let entry = self
            .entries
            .get(&p)
            .cloned()
            .ok_or_else(|| Error::Missing(p.clone()))?;

        if matches!(entry, Entry::Dir) && contains_path(&p, &self.cwd) {
            return Err(Error::Busy(p));
        }
        if matches!(entry, Entry::Dir) {
            let prefix = format!("{}/", p);
            let has_children = self.entries.keys().any(|child| child.starts_with(&prefix));
            if has_children && !recursive {
                return Err(Error::NotEmpty(p));
            }
            if recursive {
                self.entries.retain(|child, _| {
                    child != &p && !child.starts_with(&prefix)
                });
                return Ok(());
            }
        }
        self.entries.remove(&p);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link_parts = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let parent = self.resolve_parent(&link_parts)?;
        self.ensure_parent_dir(&parent)?;
        let Some(name) = link_parts.last().cloned() else {
            return Err(Error::Exists("/".to_string()));
        };
        let mut resolved = parent;
        resolved.push(name);
        let p = canonical(&resolved);
        if self.entries.contains_key(&p) {
            return Err(Error::Exists(p));
        }
        self.entries.insert(p, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let parts = self.normalize(path)?;
        let parent = self.resolve_parent(&parts)?;
        let Some(name) = parts.last().cloned() else {
            return Err(Error::InvalidTarget(0));
        };
        let mut resolved = parent;
        resolved.push(name);
        let p = canonical(&resolved);
        match self.entries.get(&p) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let parts = self.normalize(path)?;
        let resolved = self.resolve_full(parts)?;
        let root = canonical(&resolved);
        let entry = self
            .entries
            .get(&root)
            .ok_or_else(|| Error::Missing(root.clone()))?;

        let prefix = if root == "/" {
            "/".to_string()
        } else {
            format!("{}/", root)
        };
        let mut selected: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| *p == &root || p.starts_with(&prefix))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();
        selected.sort_by(|a, b| a.0.cmp(&b.0));

        let mut sizes = BTreeMap::<String, u64>::new();
        for (p, e) in &selected {
            let size = match e {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p.clone(), size);
        }

        for (p, e) in selected.iter().rev() {
            if matches!(e, Entry::Dir) {
                let child_prefix = if p == "/" {
                    "/".to_string()
                } else {
                    format!("{}/", p)
                };
                let total = selected
                    .iter()
                    .filter(|(candidate, _)| {
                        candidate.starts_with(&child_prefix)
                            && candidate[child_prefix.len()..].find('/').is_none()
                    })
                    .map(|(candidate, _)| *sizes.get(candidate).unwrap_or(&0))
                    .sum();
                sizes.insert(p.clone(), total);
            }
        }

        let _ = entry;
        Ok(selected
            .into_iter()
            .map(|(p, _)| {
                let size = *sizes.get(&p).unwrap_or(&0);
                (p, size)
            })
            .collect())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
