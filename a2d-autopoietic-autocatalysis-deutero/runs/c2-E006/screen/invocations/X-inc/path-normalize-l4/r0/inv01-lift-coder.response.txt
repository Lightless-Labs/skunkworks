use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|part| !part.is_empty())
            .is_some_and(|part| part == "." || part == "..")
}

fn raw_components(path: &str, offset: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        out.push((path[start..i].to_string(), offset + start));
    }
    out
}

fn finish_absolute(components: &[String], intent: bool) -> String {
    if components.is_empty() {
        return "/".to_string();
    }
    let mut result = format!("/{}", components.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn reduce_components(
    components: &mut Vec<String>,
    input: &[(String, usize)],
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in input {
        match part.as_str() {
            "." => {}
            ".." => {
                if components.last().is_some_and(|last| last != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: *at });
                } else {
                    components.push("..".to_string());
                }
            }
            _ => components.push(part.clone()),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut components = Vec::new();
    reduce_components(&mut components, &raw_components(path, 0), absolute)?;

    if absolute {
        Ok(finish_absolute(&components, intent))
    } else {
        let mut result = if components.is_empty() {
            ".".to_string()
        } else {
            components.join("/")
        };
        if intent && result != "." {
            result.push('/');
        }
        Ok(result)
    }
}

fn normalized_context(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }
    match normalize(path) {
        Ok(value) => Ok(value
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()),
        Err(_) => Err(invalid),
    }
}

fn expand_tilde<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(Option<&'a str>, usize, Option<Vec<String>>), PathError> {
    if !path.starts_with('~') {
        return Ok((None, 0, None));
    }
    let bytes = path.as_bytes();
    if bytes.len() > 1 && bytes[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let home_components = match home {
        Some(value) => value
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect(),
        None => return Err(PathError::MissingHome),
    };
    let consumed = if bytes.len() > 1 { 2 } else { 1 };
    Ok((Some(&path[consumed..]), consumed, Some(home_components)))
}

fn resolve_core(
    path: &str,
    cwd_components: &[String],
    home_components: Option<&[String]>,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let is_absolute = path.starts_with('/');
    let (request, consumed, tilde_home) = if path.starts_with('~') {
        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home_components.ok_or(PathError::MissingHome)?;
        let used = if path.len() > 1 { 2 } else { 1 };
        (Some(&path[used..]), used, Some(home))
    } else {
        (None, 0, None)
    };

    let mut stack = if is_absolute {
        Vec::new()
    } else if let Some(home) = tilde_home {
        home.to_vec()
    } else {
        cwd_components.to_vec()
    };

    if let Some(rest) = request {
        let parts = raw_components(rest, consumed);
        reduce_components(&mut stack, &parts, true)?;
    } else {
        let parts = raw_components(path, 0);
        reduce_components(&mut stack, &parts, true)?;
    }

    let mut result = finish_absolute(&stack, intent);
    if stack.is_empty() {
        result = "/".to_string();
    }
    Ok(result)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(normalized_context(value, PathError::InvalidHome)?),
        None => None,
    };
    resolve_core(path, &cwd_components, home_components.as_deref())
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut keys = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path)
            && link.path != "/";
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = keys.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        keys.insert(link.path.clone(), index);
    }
    Ok(keys)
}

#[derive(Clone)]
enum Source {
    Context,
    Request,
    Target(String),
}

#[derive(Clone)]
struct WorkItem {
    part: String,
    at: usize,
    source: Source,
}

struct Traversal<'a> {
    links: &'a [Link],
    key_indices: &'a HashMap<String, usize>,
    stack: Vec<String>,
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl<'a> Traversal<'a> {
    fn current_path(&self) -> String {
        if self.stack.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", self.stack.join("/"))
        }
    }

    fn push_components(
        &mut self,
        items: Vec<WorkItem>,
        queue: &mut Vec<WorkItem>,
    ) -> Result<(), PathError> {
        for item in items {
            match item.part.as_str() {
                "." => {}
                ".." => {
                    if self.stack.pop().is_none() {
                        return Err(PathError::Escape { at: item.at });
                    }
                }
                _ => {
                    self.stack.push(item.part);
                    let matched = self.key_indices.get(&self.current_path()).copied();
                    if let Some(index) = matched {
                        if self.active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if self.expansions >= self.max_expansions {
                            return Err(PathError::Limit { index });
                        }
                        self.expansions += 1;
                        self.active.push(index);

                        if !self.links[index].target.starts_with('/') {
                            self.stack.pop();
                        } else {
                            self.stack.clear();
                        }

                        let target = &self.links[index].target;
                        let target_items = raw_components(target, 0)
                            .into_iter()
                            .map(|(part, at)| WorkItem {
                                part,
                                at,
                                source: Source::Target(target.clone()),
                            })
                            .collect::<Vec<_>>();

                        // Process target to completion before resuming this item stream.
                        let remaining = queue.split_off(0);
                        let mut target_queue = target_items;
                        target_queue.extend(remaining);
                        *queue = target_queue;
                        // A marker ends the link's active lifetime after its target.
                        queue.insert(
                            0,
                            WorkItem {
                                part: String::new(),
                                at: index,
                                source: Source::Target(String::new()),
                            },
                        );
                    }
                }
            }
        }
        Ok(())
    }

    fn run(&mut self, mut queue: Vec<WorkItem>) -> Result<(), PathError> {
        while !queue.is_empty() {
            let item = queue.remove(0);
            if item.part.is_empty() {
                if let Some(pos) = self.active.iter().rposition(|&i| i == item.at) {
                    self.active.remove(pos);
                }
                continue;
            }
            let mut remaining = queue;
            self.push_components(vec![item], &mut remaining)?;
            queue = remaining;
        }
        Ok(())
    }
}

fn work_items(parts: Vec<(String, usize)>, source: Source) -> Vec<WorkItem> {
    parts
        .into_iter()
        .map(|(part, at)| WorkItem {
            part,
            at,
            source: source.clone(),
        })
        .collect()
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(normalized_context(value, PathError::InvalidHome)?),
        None => None,
    };
    let key_indices = validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (absolute, request, request_offset, tilde) = if path.starts_with('~') {
        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home_components.as_ref().ok_or(PathError::MissingHome)?;
        let consumed = if path.len() > 1 { 2 } else { 1 };
        (false, &path[consumed..], consumed, Some(home.as_slice()))
    } else {
        (path.starts_with('/'), path, 0, None)
    };

    let mut traversal = Traversal {
        links,
        key_indices: &key_indices,
        stack: Vec::new(),
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    if absolute {
        // Absolute requests start at root.
    } else if let Some(home) = tilde {
        let items = work_items(
            home.iter().cloned().map(|x| (x, 0)).collect(),
            Source::Context,
        );
        traversal.run(items)?;
    } else {
        let items = work_items(
            cwd_components.iter().cloned().map(|x| (x, 0)).collect(),
            Source::Context,
        );
        traversal.run(items)?;
    }

    let request_items = work_items(raw_components(request, request_offset), Source::Request);
    traversal.run(request_items)?;

    Ok(finish_absolute(&traversal.stack, intent))
}

fn components_of_absolute(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_string()));
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = components_of_absolute(&resolved_base);
    let target_components = components_of_absolute(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let alias_components = components_of_absolute(&probe);
        if alias_components.len() > target_components.len()
            || alias_components
                .iter()
                .zip(&target_components)
                .any(|(a, b)| a != b)
        {
            continue;
        }
        let mut candidate = route(&base_components, &components_of_absolute(&link.path));
        let suffix = &target_components[alias_components.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) =
            resolve_links(&candidate, &resolved_base, home, links, max_expansions)
        {
            if components_of_absolute(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid
        .into_iter()
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())))
        .ok_or(PathError::Escape { at: 0 })
}
