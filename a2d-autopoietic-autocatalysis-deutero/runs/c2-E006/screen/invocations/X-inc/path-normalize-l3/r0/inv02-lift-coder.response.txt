#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|part| !part.is_empty())
        .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<(String, usize)> {
    let mut out = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        out.push((path[start..i].to_owned(), start));
    }
    out
}

fn finish_absolute(stack: &[String], intent: bool) -> String {
    let mut result = if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    };
    if intent && result != "/" {
        result.push('/');
    }
    result
}

fn finish_relative(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return ".".to_owned();
    }
    let mut result = stack.join("/");
    if intent {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    for (part, at) in components(path) {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|s| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }

    if absolute {
        Ok(finish_absolute(&stack, intent))
    } else {
        Ok(finish_relative(&stack, intent))
    }
}

fn validated_context(value: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !value.starts_with('/') {
        return Err(error);
    }
    match normalize(value) {
        Ok(normalized) => Ok(components(&normalized)
            .into_iter()
            .map(|(part, _)| part)
            .collect()),
        Err(_) => Err(error),
    }
}

fn validate_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = validated_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value, PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

fn tilde_start<'a>(
    path: &'a str,
    home_parts: Option<&[String]>,
) -> Result<(bool, &'a str, usize, Option<Vec<String>>), PathError> {
    if !path.starts_with('~') {
        return Ok((path.starts_with('/'), path, 0, None));
    }
    if path.len() > 1 && !path[1..].starts_with('/') {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let parts = home_parts.ok_or(PathError::MissingHome)?;
    let consumed = if path.len() > 1 { 2 } else { 1 };
    let remainder = &path[consumed..];
    Ok((true, remainder, consumed, Some(parts.to_vec())))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_raw(path)?;
    let intent = directory_intent(path);

    let (absolute, remainder, offset, tilde_parts) =
        tilde_start(path, home_parts.as_deref())?;
    let mut stack = if absolute {
        if let Some(parts) = tilde_parts {
            parts
        } else {
            Vec::new()
        }
    } else {
        cwd_parts
    };

    for (part, local_at) in components(remainder) {
        let at = local_at + offset;
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(finish_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        let canonical = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !canonical || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(first) = links[..index]
            .iter()
            .position(|earlier| earlier.path == link.path)
        {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
    }
    Ok(())
}

struct WorkItem {
    part: String,
    at: usize,
    source: String,
}

fn push_component(stack: &mut Vec<String>, part: String) {
    stack.push(part);
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_links(links)?;
    validate_raw(path)?;
    let intent = directory_intent(path);

    let (absolute, remainder, offset, tilde_parts) =
        tilde_start(path, home_parts.as_deref())?;
    let mut stack = if absolute {
        tilde_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    let mut queue: std::collections::VecDeque<WorkItem> = std::collections::VecDeque::new();

    if !absolute {
        for part in stack.clone() {
            queue.push_back(WorkItem {
                part,
                at: 0,
                source: String::new(),
            });
        }
        stack.clear();
    }

    for (part, at) in components(remainder) {
        queue.push_back(WorkItem {
            part,
            at: at + offset,
            source: path.to_owned(),
        });
    }

    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    while let Some(item) = queue.pop_front() {
        match item.part.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: item.at });
                }
                continue;
            }
            _ => {}
        }

        push_component(&mut stack, item.part);

        let current = format!("/{}", stack.join("/"));
        if let Some(index) = links.iter().position(|link| link.path == current) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);

            let link = &links[index];
            let parent_len = stack.len().saturating_sub(1);
            let mut target_stack = if link.target.starts_with('/') {
                Vec::new()
            } else {
                stack[..parent_len].to_vec()
            };

            let target_parts = components(&link.target);
            let mut nested: std::collections::VecDeque<WorkItem> = target_parts
                .into_iter()
                .map(|(part, at)| WorkItem {
                    part,
                    at,
                    source: link.target.clone(),
                })
                .collect();

            while let Some(target_item) = nested.pop_front() {
                match target_item.part.as_str() {
                    "." => continue,
                    ".." => {
                        if target_stack.pop().is_none() {
                            return Err(PathError::Escape {
                                at: target_item.at,
                            });
                        }
                        continue;
                    }
                    _ => {}
                }
                target_stack.push(target_item.part);
                let target_current = format!("/{}", target_stack.join("/"));
                if let Some(nested_index) =
                    links.iter().position(|candidate| candidate.path == target_current)
                {
                    if active.contains(&nested_index) {
                        return Err(PathError::Loop {
                            index: nested_index,
                        });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit {
                            index: nested_index,
                        });
                    }
                    expansions += 1;
                    active.push(nested_index);

                    let nested_link = &links[nested_index];
                    let parent_len = target_stack.len().saturating_sub(1);
                    let base = if nested_link.target.starts_with('/') {
                        Vec::new()
                    } else {
                        target_stack[..parent_len].to_vec()
                    };

                    let mut target_components = components(&nested_link.target);
                    target_components.reverse();
                    for (part, at) in target_components {
                        nested.push_front(WorkItem {
                            part,
                            at,
                            source: nested_link.target.clone(),
                        });
                    }
                    target_stack = base;
                    active.pop();
                }
            }

            stack = target_stack;
            active.pop();
        }
    }

    Ok(finish_absolute(&stack, intent))
}
