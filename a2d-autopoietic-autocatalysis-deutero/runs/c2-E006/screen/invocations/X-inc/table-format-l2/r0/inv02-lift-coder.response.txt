#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let columns = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != columns {
            return Err(TableError::Ragged {
                row,
                expected: columns,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let mut trimmed = Vec::with_capacity(rows.len());
    for (row_index, cells) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(columns);
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column,
                        byte,
                    });
                }
            }
            trimmed_row.push(cell.trim_end_matches(' ').to_owned());
        }
        trimmed.push(trimmed_row);
    }

    let mut widths = vec![1usize; columns];
    for row in &trimmed {
        for (column, cell) in row.iter().enumerate() {
            widths[column] = widths[column].max(cell.len());
        }
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let alignments: Vec<&Alignment> = if options.alignments.is_empty() {
        (0..columns).map(|_| &Alignment::Left).collect()
    } else {
        options.alignments.iter().collect()
    };

    fn border(widths: &[usize], output: &mut String) {
        output.push('+');
        for &width in widths {
            for _ in 0..width + 2 {
                output.push('-');
            }
            output.push('+');
        }
        output.push('\n');
    }

    fn data_line(
        row: &[String],
        widths: &[usize],
        alignments: &[&Alignment],
        output: &mut String,
    ) {
        output.push('|');
        for ((cell, &width), alignment) in row.iter().zip(widths).zip(alignments) {
            let extra = width - cell.len();
            let (left, right) = match alignment {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };
            output.push(' ');
            for _ in 0..left {
                output.push(' ');
            }
            output.push_str(cell);
            for _ in 0..right {
                output.push(' ');
            }
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    let mut output = String::new();
    border(&widths, &mut output);

    for (row_index, row) in trimmed.iter().enumerate() {
        data_line(row, &widths, &alignments, &mut output);
        if options.header && row_index == 0 && rows.len() > 1 {
            border(&widths, &mut output);
        }
    }

    border(&widths, &mut output);
    Ok(output)
}
