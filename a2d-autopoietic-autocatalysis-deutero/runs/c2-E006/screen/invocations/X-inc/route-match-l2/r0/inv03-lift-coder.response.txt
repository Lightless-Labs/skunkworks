#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find(|(_, ch)| *ch == '?' || *ch == '#')
            .map(|(index, _)| index)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &request_segments) {
                let replace = match &best {
                    None => true,
                    Some((current, _)) => {
                        route.rank > current.rank
                            || (route.rank == current.rank && route.id < current.id)
                    }
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut end = bytes.len();
    if bytes[end - 1] == b'/' {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|&byte| byte == b'/')
            .map(|offset| start + offset)
            .unwrap_or(end);

        if segment_end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &input[start..segment_end];
        let first = bytes[start];

        if first == b':' || first == b'*' {
            let is_wildcard = first == b'*';
            let name = &raw[1..];

            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if is_wildcard && segment_end != end {
                return Err(Error::InvalidPattern { at: start });
            }

            names.push(name.to_owned());
            segments.push(if is_wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start, ErrorKind::Pattern)?;
            segments.push(Segment::Static(decoded));
        }

        if segment_end == end {
            break;
        }
        start = segment_end + 1;
    }

    if bytes[bytes.len() - 1] == b'/' {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let end = bytes.len();
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|&byte| byte == b'/')
            .map(|offset| start + offset)
            .unwrap_or(end);

        if segment_end == start {
            return Err(Error::InvalidPath { at: start });
        }

        segments.push(decode_segment(
            &input[start..segment_end],
            start,
            ErrorKind::Path,
        )?);

        if segment_end == end {
            break;
        }
        start = segment_end + 1;
    }

    if bytes[bytes.len() - 1] == b'/' {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

#[derive(Clone, Copy)]
enum ErrorKind {
    Pattern,
    Path,
}

impl ErrorKind {
    fn make(self, at: usize) -> Error {
        match self {
            Self::Pattern => Error::InvalidPattern { at },
            Self::Path => Error::InvalidPath { at },
        }
    }
}

fn decode_segment(raw: &str, offset: usize, kind: ErrorKind) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' {
            if index + 2 >= bytes.len() {
                return Err(kind.make(offset + index));
            }
            let high = hex_value(bytes[index + 1]);
            let low = hex_value(bytes[index + 2]);
            let Some(high) = high else {
                return Err(kind.make(offset + index));
            };
            let Some(low) = low else {
                return Err(kind.make(offset + index));
            };

            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(kind.make(offset + index));
            }
            decoded.push(value);
            index += 3;
        } else {
            if !is_unreserved(byte) {
                return Err(kind.make(offset + index));
            }
            decoded.push(byte);
            index += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    matches!(
        byte,
        b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'.' | b'_' | b'~' | b'-'
    )
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn match_route(
    pattern: &[Segment],
    path: &[String],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
