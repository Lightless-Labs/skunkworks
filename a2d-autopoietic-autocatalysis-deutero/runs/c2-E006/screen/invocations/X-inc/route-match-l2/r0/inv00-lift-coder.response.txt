#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, _) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let (path_segments, _) = parse_path(path)?;
        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(route, &path_segments) {
                let replace = match &best {
                    None => true,
                    Some((best_route, _)) => route.rank > best_route.rank,
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);

        if stop == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &input[start..stop];
        let first = bytes[start];
        let segment = if first == b':' || first == b'*' {
            let wildcard = first == b'*';
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && stop != end {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            }
        } else {
            Segment::Static(decode_segment(raw, start, ErrorKind::Pattern)?)
        };
        segments.push(segment);

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if trailing {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }
    Ok((segments, trailing))
}

fn parse_path(input: &str) -> Result<(Vec<String>, bool), Error> {
    let head_end = input
        .as_bytes()
        .iter()
        .position(|&b| b == b'?' || b == b'#')
        .unwrap_or(input.len());
    let head = &input[..head_end];

    if !head.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if head == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = head.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);

        if stop == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(
            &head[start..stop],
            start,
            ErrorKind::Path,
        )?);

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if trailing {
        return Err(Error::InvalidPath { at: head_end - 1 });
    }
    Ok((segments, trailing))
}

#[derive(Clone, Copy)]
enum ErrorKind {
    Pattern,
    Path,
}

fn decode_segment(raw: &str, base: usize, kind: ErrorKind) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(make_error(kind, base + i));
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(make_error(kind, base + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(make_error(kind, base + i));
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segments are ASCII"))
}

fn make_error(kind: ErrorKind, at: usize) -> Error {
    match kind {
        ErrorKind::Pattern => Error::InvalidPattern { at },
        ErrorKind::Path => Error::InvalidPath { at },
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|&byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), path[path_index..].join("/")));
                return Some(params);
            }
        }
    }

    (path_index == path.len()).then_some(params)
}
