const ADLER_MOD: u32 = 65_521;

fn crc32_raw_update(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            crc = (crc >> 1) ^ if crc & 1 != 0 { 0xedb8_8320 } else { 0 };
        }
    }
    crc
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_raw_update(0xffff_ffff, data) ^ 0xffff_ffff
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % ADLER_MOD;
        *b = (*b + *a) % ADLER_MOD;
    }
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1, 0);
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn update_hashes(hashes: &mut Hashes, data: &[u8]) {
    hashes.crc32 = crc32_raw_update(hashes.crc32, data);
    let mut a = hashes.adler32 & 0xffff;
    let mut b = hashes.adler32 >> 16;
    adler_update(&mut a, &mut b, data);
    hashes.adler32 = (b << 16) | a;
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut hashes = Hashes {
        crc32: 0xffff_ffff,
        adler32: 1,
    };
    for span in spans {
        update_hashes(&mut hashes, &data[span.start..span.end]);
    }
    hashes.crc32 ^= 0xffff_ffff;
    Ok(hashes)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_update(a: &mut u32, b: &mut u32, data: &[u8], pending: &mut Option<u8>) {
    let mut i = 0;
    if let Some(high) = pending.take() {
        if let Some(&low) = data.first() {
            let word = ((high as u32) << 8) | low as u32;
            *a = (*a + word) % 65_535;
            *b = (*b + *a) % 65_535;
            i = 1;
        } else {
            *pending = Some(high);
            return;
        }
    }

    while i + 1 < data.len() {
        let word = ((data[i] as u32) << 8) | data[i + 1] as u32;
        *a = (*a + word) % 65_535;
        *b = (*b + *a) % 65_535;
        i += 2;
    }
    if i < data.len() {
        *pending = Some(data[i]);
    }
}

fn fletcher32_finish(a: u32, b: u32, pending: Option<u8>) -> u32 {
    let (mut a, mut b) = (a, b);
    if let Some(high) = pending {
        a = (a + ((high as u32) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    (b << 16) | a
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (0, 0);
    let mut pending = None;
    fletcher32_update(&mut a, &mut b, data, &mut pending);
    fletcher32_finish(a, b, pending)
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let (mut a, mut b) = (0, 0);
    let mut pending = None;
    for span in spans {
        fletcher32_update(
            &mut a,
            &mut b,
            &data[span.start..span.end],
            &mut pending,
        );
    }
    Ok(fletcher32_finish(a, b, pending))
}

fn crc16_update(mut crc: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    crc = (crc << 1) ^ if crc & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            crc
        }
        Crc16::X25 => {
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    crc = (crc >> 1) ^ if crc & 1 != 0 { 0x8408 } else { 0 };
                }
            }
            crc
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem => crc16_update(0, data, variant),
        Crc16::False | Crc16::X25 => crc16_update(0xffff, data, variant)
            ^ if variant == Crc16::X25 { 0xffff } else { 0 },
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let (mut base, mut f16_a, mut f16_b, mut f32_a, mut f32_b) =
        (Hashes { crc32: 0xffff_ffff, adler32: 1 }, 0u32, 0u32, 0u32, 0u32);
    let mut f32_pending = None;
    let mut c16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let slice = &data[span.start..span.end];
        update_hashes(&mut base, slice);
        for &byte in slice {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;
        }
        fletcher32_update(&mut f32_a, &mut f32_b, slice, &mut f32_pending);
        c16 = crc16_update(c16, slice, variant);
    }

    base.crc32 ^= 0xffff_ffff;
    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }
    Ok(Multi {
        base,
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: fletcher32_finish(f32_a, f32_b, f32_pending),
        crc16: c16,
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = gf2_matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = crc32_raw_update(self.crc, data);
        adler_update(&mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        let mut adler = 0u32;
        for offset in 1..9 {
            let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            crc = (crc << 4) | digit as u32;
        }
        for offset in 11..19 {
            let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            adler = (adler << 4) | digit as u32;
        }
        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn selected_len(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |sum, span| {
        sum + (span.end - span.start) as u64
    })
}

fn selected_hashes(data: &[u8], spans: &[Span]) -> Hashes {
    let mut hashes = Hashes {
        crc32: 0xffff_ffff,
        adler32: 1,
    };
    for span in spans {
        update_hashes(&mut hashes, &data[span.start..span.end]);
    }
    hashes.crc32 ^= 0xffff_ffff;
    hashes
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let hashes = selected_hashes(data, spans);
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        selected_len(spans),
        hashes.crc32,
        hashes.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let length = parse_hex_u64(bytes, 4, 20);
    let expected_crc = parse_hex_u32(bytes, 25, 33);
    let expected_adler = parse_hex_u32(bytes, 40, 48);
    if length != selected_len(spans) {
        return Err(VerifyError::Length);
    }

    let actual = selected_hashes(data, spans);
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> u64 {
    let mut value = 0u64;
    for &byte in &bytes[start..end] {
        value = (value << 4) | hex_value(byte).unwrap() as u64;
    }
    value
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> u32 {
    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = (value << 4) | hex_value(byte).unwrap() as u32;
    }
    value
}
