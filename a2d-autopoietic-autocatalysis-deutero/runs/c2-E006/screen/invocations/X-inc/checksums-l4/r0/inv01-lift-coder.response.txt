#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn crc32_update(mut reg: u32, data: &[u8]) -> u32 {
    for &byte in data {
        reg ^= byte as u32;
        for _ in 0..8 {
            let low = reg & 1;
            reg >>= 1;
            if low != 0 {
                reg ^= 0xedb88320;
            }
        }
    }
    reg
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_update(0xffff_ffff, data) ^ 0xffff_ffff
}

fn adler_update(mut a: u32, mut b: u32, data: &[u8]) -> (u32, u32) {
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (a, b)
}

pub fn adler32(data: &[u8]) -> u32 {
    let (a, b) = adler_update(1, 0, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        crc = crc32_update(crc, &data[span.start..span.end]);
        (a, b) = adler_update(a, b, &data[span.start..span.end]);
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = crc32_update(self.crc, data);
        (self.a, self.b) = adler_update(self.a, self.b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        let mut adler = 0u32;
        for offset in (1..9).chain(11..19) {
            let digit = match bytes[offset] {
                b'0'..=b'9' => bytes[offset] - b'0',
                b'a'..=b'f' => bytes[offset] - b'a' + 10,
                _ => return Err(SnapshotError::Hex { offset }),
            } as u32;
            if offset < 9 {
                crc = (crc << 4) | digit;
            } else {
                adler = (adler << 4) | digit;
            }
        }
        Self::from_seed(crc, adler)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut reg: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                reg ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = reg & 0x8000;
                    reg = reg.wrapping_shl(1);
                    if top != 0 {
                        reg ^= 0x1021;
                    }
                }
            }
            reg
        }
        Crc16::X25 => {
            for &byte in data {
                reg ^= byte as u16;
                for _ in 0..8 {
                    let low = reg & 1;
                    reg >>= 1;
                    if low != 0 {
                        reg ^= 0x8408;
                    }
                }
            }
            reg ^ 0xffff
        }
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &x in data {
        a = (a + x as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_update(mut a: u32, mut b: u32, data: &[u8], pending: &mut Option<u8>) -> (u32, u32) {
    let mut i = 0;
    if let Some(high) = pending.take() {
        if !data.is_empty() {
            let word = ((high as u32) << 8) | data[0] as u32;
            a = (a + word) % 65535;
            b = (b + a) % 65535;
            i = 1;
        } else {
            *pending = Some(high);
            return (a, b);
        }
    }
    while i + 1 < data.len() {
        let word = ((data[i] as u32) << 8) | data[i + 1] as u32;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
        i += 2;
    }
    if i < data.len() {
        *pending = Some(data[i]);
    }
    (a, b)
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0;
    let mut b = 0;
    let mut pending = None;
    (a, b) = fletcher32_update(a, b, data, &mut pending);
    if let Some(high) = pending {
        a = (a + ((high as u32) << 8)) % 65535;
        b = (b + a) % 65535;
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0;
    let mut b = 0;
    let mut pending = None;
    for span in spans {
        (a, b) = fletcher32_update(a, b, &data[span.start..span.end], &mut pending);
    }
    if let Some(high) = pending {
        a = (a + ((high as u32) << 8)) % 65535;
        b = (b + a) % 65535;
    }
    Ok((b << 16) | a)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let (mut ad_a, mut ad_b) = (1, 0);
    let (mut f16a, mut f16b) = (0u32, 0u32);
    let (mut f32a, mut f32b) = (0u32, 0u32);
    let mut pending = None;
    let mut c16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update(crc, selected);
        (ad_a, ad_b) = adler_update(ad_a, ad_b, selected);
        for &x in selected {
            f16a = (f16a + x as u32) % 255;
            f16b = (f16b + f16a) % 255;
        }
        (f32a, f32b) = fletcher32_update(f32a, f32b, selected, &mut pending);
        match variant {
            Crc16::Xmodem | Crc16::False => {
                for &byte in selected {
                    c16 ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = c16 & 0x8000;
                        c16 = c16.wrapping_shl(1);
                        if top != 0 {
                            c16 ^= 0x1021;
                        }
                    }
                }
            }
            Crc16::X25 => {
                for &byte in selected {
                    c16 ^= byte as u16;
                    for _ in 0..8 {
                        let low = c16 & 1;
                        c16 >>= 1;
                        if low != 0 {
                            c16 ^= 0x8408;
                        }
                    }
                }
            }
        }
    }
    if let Some(high) = pending {
        f32a = (f32a + ((high as u32) << 8)) % 65535;
        f32b = (f32b + f32a) % 65535;
    }
    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }
    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (ad_b << 16) | ad_a,
        },
        fletcher16: ((f16b as u16) << 8) | f16a as u16,
        fletcher32: (f32b << 16) | f32a,
        crc16: c16,
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut i = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[i];
        }
        vector >>= 1;
        i += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = gf2_matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb88320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn selected_stats(data: &[u8], spans: &[Span]) -> Result<(u64, Hashes), SpanError> {
    validate_spans(data.len(), spans)?;
    let mut length = 0u64;
    let mut crc = 0xffff_ffff;
    let (mut a, mut b) = (1, 0);
    for span in spans {
        let selected = &data[span.start..span.end];
        length += selected.len() as u64;
        crc = crc32_update(crc, selected);
        (a, b) = adler_update(a, b, selected);
    }
    Ok((
        length,
        Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (b << 16) | a,
        },
    ))
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let (length, hashes) = selected_stats(data, spans)?;
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length, hashes.crc32, hashes.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    let (length, hashes) = selected_stats(data, spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut parsed_len = 0u64;
    let mut parsed_crc = 0u32;
    let mut parsed_adler = 0u32;
    for offset in 4..48 {
        if (20..25).contains(&offset) || (33..40).contains(&offset) {
            continue;
        }
        let value = match bytes[offset] {
            b'0'..=b'9' => bytes[offset] - b'0',
            b'a'..=b'f' => bytes[offset] - b'a' + 10,
            _ => return Err(VerifyError::Hex { offset }),
        } as u32;

        if offset < 20 {
            parsed_len = (parsed_len << 4) | value as u64;
        } else if offset < 33 {
            parsed_crc = (parsed_crc << 4) | value;
        } else {
            parsed_adler = (parsed_adler << 4) | value;
        }
    }

    if parsed_len != length {
        return Err(VerifyError::Length);
    }
    if parsed_crc != hashes.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: parsed_crc,
            actual: hashes.crc32,
        });
    }
    if parsed_adler != hashes.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: parsed_adler,
            actual: hashes.adler32,
        });
    }
    Ok(())
}
