#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> impl Iterator<Item = &str> {
    path.split('/').filter(|part| !part.is_empty())
}

fn reduce(
    stack: &mut Vec<String>,
    path: &str,
    absolute: bool,
    offset: usize,
) -> Result<(), PathError> {
    let mut search_from = 0;

    for part in components(path) {
        let at = path[search_from..]
            .find(part)
            .map(|n| search_from + n)
            .unwrap_or(search_from);
        search_from = at + part.len();

        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset + at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(part.to_owned()),
        }
    }

    Ok(())
}

fn compose(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    reduce(&mut stack, path, absolute, 0)?;
    Ok(compose(&stack, absolute, intent))
}

fn validated_context(path: &str) -> Option<Vec<String>> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return None;
    }

    let normalized = normalize(path).ok()?;
    Some(
        components(&normalized)
            .map(str::to_owned)
            .collect::<Vec<_>>(),
    )
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd).ok_or(PathError::InvalidCwd)?;
    let home_components = match home {
        Some(path) => Some(validated_context(path).ok_or(PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let (absolute, mut stack, remainder, offset) = if path.starts_with('/') {
        (true, Vec::new(), path, 0)
    } else if path.starts_with('~') {
        if path.len() > 1 && !path.as_bytes()[1..].starts_with(b"/") {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }

        let stack = home_components.ok_or(PathError::MissingHome)?;
        if path == "~" {
            let result = compose(&stack, true, false);
            return Ok(result);
        }

        (true, stack, &path[2..], 2)
    } else {
        (true, cwd_components, path, 0)
    };

    let intent = directory_intent(path);
    reduce(&mut stack, remainder, absolute, offset)?;
    Ok(compose(&stack, true, intent))
}
