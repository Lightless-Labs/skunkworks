#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .filter(|(_, part)| !part.is_empty())
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, part)| *part == "." || *part == "..")
}

fn reduce(
    stack: &mut Vec<String>,
    parts: impl IntoIterator<Item = (usize, String)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, part) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, dir_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if dir_intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn normalized_absolute(path: &str) -> Result<Vec<String>, PathError> {
    validate_raw(path)?;
    if !path.starts_with('/') {
        return Err(PathError::Escape { at: 0 });
    }
    let mut stack = Vec::new();
    reduce(
        &mut stack,
        components(path).into_iter().map(|(at, part)| (at, part.to_string())),
        true,
    )?;
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce(
        &mut stack,
        components(path).into_iter().map(|(at, part)| (at, part.to_string())),
        absolute,
    )?;

    Ok(render(&stack, absolute, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let mut cwd_stack = normalized_absolute(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(home) => Some(normalized_absolute(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let intent = directory_intent(path);
    let (mut stack, remainder, base_offset) = if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else if path.starts_with('~') {
        if path.len() > 1 && !path.as_bytes()[1..].starts_with(b"/") {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let Some(home_stack) = home_stack else {
            return Err(PathError::MissingHome);
        };
        let remainder = if path == "~" { "" } else { &path[2..] };
        (home_stack, remainder, if path == "~" { path.len() } else { 2 })
    } else {
        (std::mem::take(&mut cwd_stack), path, 0)
    };

    reduce(
        &mut stack,
        components(remainder)
            .into_iter()
            .map(|(at, part)| (at + base_offset, part.to_string())),
        true,
    )?;

    Ok(render(&stack, true, intent))
}
