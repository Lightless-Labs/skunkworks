#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    started: bool,
    paused_at: Option<u64>,
}

struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| b.started.cmp(&a.started))
        .then_with(|| match (a.started, b.started) {
            (true, true) => a
                .paused_at
                .unwrap_or(0)
                .cmp(&b.paused_at.unwrap_or(0))
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (false, false) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
            _ => Ordering::Equal,
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut arrival_index = 0;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0;

    while finished_count < customers.len() {
        let next_arrival = arrivals.get(arrival_index).map(|c| c.arrival);
        let next_finish = running.as_ref().map(|r| r.finish);
        let next_work = running
            .as_ref()
            .filter(|r| r.work_at > 0)
            .map(|r| r.work_at);

        let time = [next_arrival, next_finish, next_work]
            .into_iter()
            .flatten()
            .min()
            .expect("unfinished customers must have a pending event");

        // Scheduled finish happens before arrivals at the same tick.
        if running.as_ref().is_some_and(|r| r.finish == time) {
            let done = running.take().unwrap();
            trace.push(Event {
                time,
                customer: done.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        // Process the entire arrival cohort before making dispatch decisions.
        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            let customer = arrivals[arrival_index].clone();
            trace.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                customer,
                started: false,
                paused_at: None,
            });
            arrival_index += 1;
        }

        // Reconsider preemption once protection has ended.
        if let Some(active) = running.as_ref() {
            if time >= active.work_at && time < active.finish {
                waiting.sort_by(waiting_cmp);
                if let Some(best) = waiting.first() {
                    if best.customer.priority < active.customer.priority {
                        let by = best.customer.id;
                        let active = running.take().unwrap();
                        let remaining = active.finish - time;
                        trace.push(Event {
                            time,
                            customer: active.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            customer: active.customer,
                            started: true,
                            paused_at: Some(time),
                        });
                    }
                }
            }
        }

        // Dispatch repeatedly, including immediate zero-duration finishes.
        while running.is_none() && !waiting.is_empty() {
            waiting.sort_by(waiting_cmp);
            let selected = waiting.remove(0);
            let customer = selected.customer;
            let is_resume = selected.started;
            let cost = if last_dispatched == Some(customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };

            let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: customer.id,
                time,
            })?;
            let finish = work_at
                .checked_add(selected_if_remaining(&customer, selected.started, &waiting, &running))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time,
                })?;

            // Remaining service is stored for paused jobs; for fresh jobs it is
            // the original service duration.
            let remaining = if is_resume {
                // The paused remainder is carried separately below via lookup.
                // This branch is replaced by the value held in the queue record.
                0
            } else {
                customer.service
            };

            let _ = remaining;
            unreachable!();
        }
    }

    Ok(trace)
}

fn selected_if_remaining(
    _customer: &Customer,
    _started: bool,
    _waiting: &[Waiting],
    _running: &Option<Running>,
) -> u64 {
    0
}
