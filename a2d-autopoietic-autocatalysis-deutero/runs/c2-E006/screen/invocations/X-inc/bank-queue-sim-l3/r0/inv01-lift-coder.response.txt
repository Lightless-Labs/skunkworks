#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    paused_at: Option<u64>,
}

struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_precedes(a: &Waiting, b: &Waiting) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    match (a.paused_at, b.paused_at) {
        (Some(pa), Some(pb)) if pa != pb => pa < pb,
        (Some(_), None) => true,
        (None, Some(_)) => false,
        _ => {
            if a.customer.arrival != b.customer.arrival {
                a.customer.arrival < b.customer.arrival
            } else {
                a.customer.id > b.customer.id
            }
        }
    }
}

fn best_waiter_index(waiting: &[Waiting]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| {
            if waiter_precedes(a, b) {
                std::cmp::Ordering::Less
            } else if waiter_precedes(b, a) {
                std::cmp::Ordering::Greater
            } else {
                std::cmp::Ordering::Equal
            }
        })
        .map(|(index, _)| index)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;
    let mut now = arrivals[0].arrival;

    while finished_count < customers.len() {
        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let id = active.customer.id;
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                paused_at: None,
            });
            next_arrival += 1;
        }

        let protection_ended = running
            .as_ref()
            .map(|active| now >= active.work_at)
            .unwrap_or(false);
        if protection_ended {
            if let (Some(active), Some(best_index)) =
                (running.as_ref(), best_waiter_index(&waiting))
            {
                if waiting[best_index].customer.priority < active.customer.priority {
                    let by = waiting[best_index].customer.id;
                    let active = running.take().unwrap();
                    let remaining = active.finish - now;
                    events.push(Event {
                        time: now,
                        customer: active.customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiting.push(Waiting {
                        customer: active.customer,
                        remaining,
                        started: true,
                        paused_at: Some(now),
                    });
                }
            }
        }

        while running.is_none() {
            let Some(index) = best_waiter_index(&waiting) else {
                break;
            };
            let selected = waiting.remove(index);
            let id = selected.customer.id;
            let cost = if last_dispatched.map(|last| last != id).unwrap_or(false) {
                switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
            })?;
            let finish = work_at.checked_add(selected.remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
            })?;

            let kind = if selected.started {
                Kind::Resume {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time: now,
                customer: id,
                kind,
            });
            last_dispatched = Some(id);
            running = Some(Running {
                customer: selected.customer,
                remaining: selected.remaining,
                work_at,
                finish,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for t in [Some(active.finish), Some(active.work_at)] {
                if let Some(t) = t {
                    if t > now {
                        next_time = Some(next_time.map_or(t, |cur| cur.min(t)));
                    }
                }
            }
        }
        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    Ok(events)
}
