```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let absolute = path.starts_with('/');
        let mut components: Vec<&str> = Vec::new();

        if !absolute && self.cwd != "/" {
            components.extend(self.cwd[1..].split('/'));
        }

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn parent_and_name(path: &str) -> Option<(String, String)> {
        if path == "/" {
            return None;
        }
        match path.rfind('/') {
            Some(0) => Some(("/".to_string(), path[1..].to_string())),
            Some(i) => Some((path[..i].to_string(), path[i + 1..].to_string())),
            None => None,
        }
    }

    fn resolve_dir(&self, path: &str) -> Result<(), Error> {
        if path == "/" {
            return Ok(());
        }

        let mut prefix = String::new();
        for component in path[1..].split('/') {
            prefix.push('/');
            prefix.push_str(component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent_and_name(path).expect("non-root path");
        self.resolve_dir(&parent)?;
        Ok((parent, name))
    }

    fn contains_path(parent: &str, candidate: &str) -> bool {
        if parent == "/" {
            candidate.starts_with('/')
        } else {
            candidate == parent
                || candidate
                    .strip_prefix(parent)
                    .is_some_and(|suffix| suffix.starts_with('/'))
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };

        self.entries
            .keys()
            .filter_map(|candidate| {
                let rest = candidate.strip_prefix(&prefix)?;
                if !rest.is_empty() && !rest.contains('/') {
                    Some(candidate.clone())
                } else {
                    None
                }
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }
        let (parent, _) = self.resolve_parent(&path)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        let _ = parent;
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        self.resolve_parent(&path)?;

        match self.entries.get_mut(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(existing)) => {
                existing.clear();
                existing.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(data)) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Ok(self
                .children(&path)
                .into_iter()
                .map(|child| {
                    let name = child[1..].to_string();
                    let kind = self.entries[&child].kind();
                    (name, kind)
                })
                .collect());
        }

        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            Some(Entry::Dir) => Ok(self
                .children(&path)
                .into_iter()
                .map(|child| {
                    let name = child[child.rfind('/').unwrap() + 1..].to_string();
                    let kind = self.entries[&child].kind();
                    (name, kind)
                })
                .collect()),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            self.cwd = path;
            return Ok(());
        }
        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            Some(Entry::Dir) => {
                self.cwd = path;
                Ok(())
            }
        }
    }

    pub fn rename(
        &mut self,
        from: &str,
        to: &str,
        overwrite: bool,
    ) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&source)?;
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        self.resolve_parent(&destination)?;

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(existing) = &destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let source_prefix = format!("{}/", source);
        let mut moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter_map(|(key, value)| {
                if key == &source || key.starts_with(&source_prefix) {
                    let suffix = &key[source.len()..];
                    Some((format!("{}{}", destination, suffix), value.clone()))
                } else {
                    None
                }
            })
            .collect();

        let source_keys: Vec<String> = self
            .entries
            .keys()
            .filter(|key| *key == &source || key.starts_with(&source_prefix))
            .cloned()
            .collect();

        if destination_entry.is_some() {
            let destination_prefix = format!("{}/", destination);
            let destination_keys: Vec<String> = self
                .entries
                .keys()
                .filter(|key| *key == &destination || key.starts_with(&destination_prefix))
                .cloned()
                .collect();
            for key in destination_keys {
                self.entries.remove(&key);
            }
        }

        for key in source_keys {
            self.entries.remove(&key);
        }
        for (key, value) in moved.drain(..) {
            self.entries.insert(key, value);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&path)?;
        let entry = self
            .entries
            .get(&path)
            .cloned()
            .ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains_path(&path, &self.cwd) {
            return Err(Error::Busy(path));
        }

        let children = self.children(&path);
        if matches!(entry, Entry::Dir) && !children.is_empty() && !recursive {
            return Err(Error::NotEmpty(path));
        }

        let prefix = format!("{}/", path);
        let keys: Vec<String> = self
            .entries
            .keys()
            .filter(|key| {
                *key == &path || (recursive && matches!(entry, Entry::Dir) && key.starts_with(&prefix))
            })
            .cloned()
            .collect();

        for key in keys {
            self.entries.remove(&key);
        }
        Ok(())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
