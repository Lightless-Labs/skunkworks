#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (parts, trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parts) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let allowed = allowed_methods(&candidates);
        let eligible: Vec<usize> = if method == "HEAD" {
            let has_head = candidates.iter().any(|(r, _)| r.method == "HEAD");
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| {
                    if (has_head && r.method == "HEAD") || (!has_head && r.method == "GET") {
                        Some(i)
                    } else {
                        None
                    }
                })
                .collect()
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| (r.method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            return Err(Error::MethodNotAllowed { allowed });
        }

        let winner_index = *eligible
            .iter()
            .max_by(|&&a, &&b| {
                candidates[a]
                    .0
                    .rank
                    .cmp(&candidates[b].0.rank)
                    .then_with(|| candidates[b].0.id.cmp(&candidates[a].0.id))
            })
            .unwrap();

        let (route, params) = &candidates[winner_index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && route.trailing != trailing
        {
            Some(if route.trailing {
                format!("{head}/")
            } else {
                head.to_owned()
            })
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|n| start + n)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &input[start..next];
        if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && next != end {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start, true).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }

        if next == end {
            break;
        }
        start = next + 1;
        if start == end {
            break;
        }
    }

    Ok((segments, trailing))
}

fn parse_path(input: &str) -> Result<(Vec<String>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut parts = Vec::new();
    let mut start = 1;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|n| start + n)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPath { at: start });
        }
        let decoded = decode_segment(&input[start..next], start, false)
            .map_err(|at| Error::InvalidPath { at })?;
        parts.push(decoded);
        if next == end {
            break;
        }
        start = next + 1;
        if start == end {
            break;
        }
    }

    Ok((parts, trailing))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize, pattern: bool) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            let _ = pattern;
            return Err(base + i);
        }
    }

    String::from_utf8(out).map_err(|_| base)
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, parts: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if parts.get(i)? != value {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), parts.get(i)?.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), parts[i..].join("/")));
                i = parts.len();
                break;
            }
        }
    }

    if i == parts.len() {
        Some(params)
    } else {
        None
    }
}

fn allowed_methods(candidates: &[(&Route, Vec<(String, String)>)]) -> Vec<String> {
    let mut methods = std::collections::BTreeSet::new();
    for (route, _) in candidates {
        methods.insert(route.method.clone());
    }
    if methods.contains("GET") {
        methods.insert("HEAD".to_owned());
    }
    methods.into_iter().collect()
}
