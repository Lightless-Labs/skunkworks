#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_wildcard(c: char) -> bool {
    matches!(c, '*' | '?' | '[')
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && is_wildcard(c) {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn lookup_env<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn expand_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar_pos + 2;
    let mut cursor = content_start;
    let mut closing = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if c == '}' {
            closing = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let closing = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let content = &input[content_start..closing];

    let name_end = content
        .char_indices()
        .find_map(|(i, c)| (!is_name_continue(c)).then_some(i))
        .unwrap_or(content.len());

    let name = &content[..name_end];
    if name.is_empty() || !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let remainder = &content[name_end..];
    let value = if remainder.is_empty() {
        lookup_env(name, env).unwrap_or("").to_string()
    } else if let Some(default_text) = remainder.strip_prefix(":-") {
        match lookup_env(name, env) {
            Some(found) if !found.is_empty() => found.to_string(),
            _ => decode_default(default_text),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((value, closing + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let width = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                cursor += width;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    cursor += width;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    cursor += width;
                    if cursor >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_literal(&mut word, next, false);
                        }
                        cursor += next.len_utf8();
                    } else {
                        append_literal(&mut word, '\\', false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    let dollar_pos = cursor;
                    let next_pos = cursor + width;
                    if next_pos < input.len() {
                        let next = input[next_pos..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: dollar_pos });
                        } else if next == '{' {
                            let (value, end) = expand_braced(input, dollar_pos, env)?;
                            append_expansion(&mut word, &value, false);
                            active = true;
                            cursor = end;
                        } else if is_name_start(next) {
                            let mut end = next_pos + next.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_pos..end];
                            append_expansion(
                                &mut word,
                                lookup_env(name, env).unwrap_or(""),
                                false,
                            );
                            active = true;
                            cursor = end;
                        } else {
                            append_literal(&mut word, '$', false);
                            active = true;
                            cursor += width;
                        }
                    } else {
                        append_literal(&mut word, '$', false);
                        active = true;
                        cursor += width;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    cursor += width;
                }

                let _ = open_pos;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    cursor += width;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(cursor);
                    cursor += width;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(cursor);
                    cursor += width;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    cursor += width;
                    if cursor >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    cursor += next.len_utf8();
                    if next == '\n' {
                        // A line continuation contributes neither text nor word activity.
                    } else {
                        append_literal(&mut word, next, false);
                        active = true;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    let dollar_pos = cursor;
                    let next_pos = cursor + width;
                    if next_pos < input.len() {
                        let next = input[next_pos..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: dollar_pos });
                        } else if next == '{' {
                            let (value, end) = expand_braced(input, dollar_pos, env)?;
                            append_expansion(&mut word, &value, true);
                            active = true;
                            cursor = end;
                        } else if is_name_start(next) {
                            let mut end = next_pos + next.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_pos..end];
                            append_expansion(
                                &mut word,
                                lookup_env(name, env).unwrap_or(""),
                                true,
                            );
                            active = true;
                            cursor = end;
                        } else {
                            append_literal(&mut word, '$', false);
                            active = true;
                            cursor += width;
                        }
                    } else {
                        append_literal(&mut word, '$', false);
                        active = true;
                        cursor += width;
                    }
                } else {
                    append_literal(&mut word, c, true);
                    active = true;
                    cursor += width;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
