#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_text(word: &mut Word, text: &str, active_glob: bool) {
    for c in text.chars() {
        append_scalar(word, c, active_glob);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn split_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let bytes = input.as_bytes();
    let mut close = dollar + 2;

    while close < bytes.len() {
        if bytes[close] == b'\\' {
            close += 1;
            if close < bytes.len() {
                let c = input[close..].chars().next().unwrap();
                close += c.len_utf8();
            }
        } else if bytes[close] == b'}' {
            break;
        } else {
            let c = input[close..].chars().next().unwrap();
            close += c.len_utf8();
        }
    }

    if close >= bytes.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let contents = &input[dollar + 2..close];
    let chars: Vec<(usize, char)> = contents.char_indices().collect();
    let mut name_end = 0;

    if chars.is_empty() || !is_name_start(chars[0].1) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    name_end = chars[0].1.len_utf8();
    for &(offset, c) in chars.iter().skip(1) {
        if is_name_continue(c) {
            name_end = offset + c.len_utf8();
        } else {
            break;
        }
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];

    let value = if rest.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => {
                let mut decoded = String::new();
                let mut i = 0;
                while i < default.len() {
                    let c = default[i..].chars().next().unwrap();
                    if c == '\\' {
                        i += 1;
                        if i < default.len() {
                            let escaped = default[i..].chars().next().unwrap();
                            decoded.push(escaped);
                            i += escaped.len_utf8();
                        } else {
                            decoded.push('\\');
                        }
                    } else {
                        decoded.push(c);
                        i += c.len_utf8();
                    }
                }
                decoded
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            append_scalar(&mut word, escaped, false);
                        }
                        i = next_i + escaped.len_utf8();
                    } else {
                        append_scalar(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (value, end) = split_braced(input, i, env)?;
                        append_text(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(env, &input[next_i..end]).unwrap_or("");
                            append_text(&mut word, value, false);
                            active = true;
                            i = end;
                        } else {
                            append_scalar(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        append_scalar(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_scalar(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    if escaped != '\n' {
                        append_scalar(&mut word, escaped, false);
                        active = true;
                    }
                    i = next_i + escaped.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (value, end) = split_braced(input, i, env)?;
                        append_text(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(env, &input[next_i..end]).unwrap_or("");
                            append_text(&mut word, value, true);
                            active = true;
                            i = end;
                        } else {
                            append_scalar(&mut word, '$', true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_scalar(&mut word, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_scalar(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
