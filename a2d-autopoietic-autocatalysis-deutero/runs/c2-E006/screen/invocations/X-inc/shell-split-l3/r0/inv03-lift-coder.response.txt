#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_char(word: &mut Word, c: char, wildcard_active: bool) {
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_str(word: &mut Word, text: &str, wildcard_active: bool) {
    for c in text.chars() {
        append_char(word, c, wildcard_active);
    }
}

fn find_unescaped_closing_brace(input: &str, start: usize) -> Option<usize> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next()?;
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next()?;
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find_map(|(key, value)| (*key == name).then_some(*value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let d = input[next_i..].chars().next().unwrap();
                    let after_d = next_i + d.len_utf8();
                    if matches!(d, '"' | '\\' | '$') {
                        append_char(&mut word, d, false);
                        i = after_d;
                    } else if d == '\n' {
                        i = after_d;
                    } else {
                        append_char(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let close = find_unescaped_closing_brace(input, next_i + 1)
                            .ok_or(SplitError::UnterminatedExpansion { pos: i })?;
                        let body = &input[next_i + 1..close];
                        let (name, default) = if let Some(split_at) = body.find(":-") {
                            (&body[..split_at], Some(&body[split_at + 2..]))
                        } else {
                            (body, None)
                        };
                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: i });
                        }
                        let value = match (lookup(name, env), default) {
                            (Some(v), Some(d)) if !v.is_empty() => v.to_string(),
                            (_, Some(d)) => decode_default(d),
                            (Some(v), None) => v.to_string(),
                            (None, None) => String::new(),
                        };
                        append_str(&mut word, &value, false);
                        active = true;
                        i = close + 1;
                    } else if let Some(first) = input[next_i..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            if let Some(value) = lookup(name, env) {
                                append_str(&mut word, value, false);
                            }
                            active = true;
                            i = end;
                        } else {
                            append_char(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        append_char(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_char(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let d = input[next_i..].chars().next().unwrap();
                    let after_d = next_i + d.len_utf8();
                    if d == '\n' {
                        i = after_d;
                    } else {
                        append_char(&mut word, d, false);
                        active = true;
                        i = after_d;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let close = find_unescaped_closing_brace(input, next_i + 1)
                            .ok_or(SplitError::UnterminatedExpansion { pos: i })?;
                        let body = &input[next_i + 1..close];
                        let (name, default) = if let Some(split_at) = body.find(":-") {
                            (&body[..split_at], Some(&body[split_at + 2..]))
                        } else {
                            (body, None)
                        };
                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: i });
                        }
                        let value = match (lookup(name, env), default) {
                            (Some(v), Some(_)) if !v.is_empty() => v.to_string(),
                            (_, Some(d)) => decode_default(d),
                            (Some(v), None) => v.to_string(),
                            (None, None) => String::new(),
                        };
                        append_str(&mut word, &value, true);
                        active = true;
                        i = close + 1;
                    } else if let Some(first) = input[next_i..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            if let Some(value) = lookup(name, env) {
                                append_str(&mut word, value, true);
                            }
                            active = true;
                            i = end;
                        } else {
                            append_char(&mut word, '$', true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_char(&mut word, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
