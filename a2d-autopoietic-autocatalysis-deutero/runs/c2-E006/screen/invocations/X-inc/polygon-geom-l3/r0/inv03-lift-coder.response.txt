#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_POINTS: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_POINTS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points[0] == points[points.len() - 1] {
        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    Ok(&points[..effective_len])
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = a.x as i128 - b.x as i128;
    let dy = a.y as i128 - b.y as i128;
    let squared = (dx * dx + dy * dy) as u128;
    let scaled = squared * 1_000_000u128;

    let mut low = 0u128;
    let mut high = 2_828_428_428u128;
    while low < high {
        let mid = low + (high - low + 1) / 2;
        if mid * mid <= scaled {
            low = mid;
        } else {
            high = mid - 1;
        }
    }

    let lower_square = low * low;
    let upper_square = (low + 1) * (low + 1);
    if scaled - lower_square >= upper_square - scaled {
        (low + 1) as u64
    } else {
        low as u64
    }
}

fn perimeter_and_edges(points: &[Point]) -> (Vec<u64>, u64) {
    if points.is_empty() {
        return (Vec::new(), 0);
    }
    let mut edges = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    for i in 0..points.len() {
        let length = rounded_edge_milli(points[i], points[(i + 1) % points.len()]);
        edges.push(length);
        perimeter += length;
    }
    (edges, perimeter)
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare_polygon(points)?;
    let signed = signed_area2(points);
    let orientation = if signed > 0 {
        Orientation::CounterClockwise
    } else if signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };
    let (edge_milli, perimeter_milli) = perimeter_and_edges(points);

    let mut longest_edge = None;
    for i in 0..edge_milli.len() {
        match longest_edge {
            None => longest_edge = Some(i),
            Some(best) => {
                if edge_milli[i] > edge_milli[best]
                    || (edge_milli[i] == edge_milli[best]
                        && {
                            let pair = ordered_pair(points[i], points[(i + 1) % points.len()]);
                            let best_pair = ordered_pair(
                                points[best],
                                points[(best + 1) % points.len()],
                            );
                            pair < best_pair || (pair == best_pair && i < best)
                        })
                {
                    longest_edge = Some(i);
                }
            }
        }
    }

    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn ordered_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;

    if queries.len() > MAX_POINTS {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary_edge = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];

            if on_segment(a, b, query) {
                boundary_edge = Some(i);
                break;
            }

            if (a.y > query.y) != (b.y > query.y) {
                let side = cross(a, b, query);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }

        results.push(if let Some(edge) = boundary_edge {
            Location::Boundary { edge }
        } else if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_POINTS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = unique.iter_mut().find(|entry| entry.0 == point) {
            existing.1 = existing.1.min(index);
        } else {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|entry| entry.0);

    let n = unique.len();
    let (vertices, source_indices) = if n <= 1 {
        (
            unique.iter().map(|entry| entry.0).collect(),
            unique.iter().map(|entry| entry.1).collect(),
        )
    } else {
        let all_collinear = (2..n).all(|i| {
            cross(unique[0].0, unique[n - 1].0, unique[i].0) == 0
        });

        if all_collinear {
            match mode {
                Collinear::Include => (
                    unique.iter().map(|entry| entry.0).collect(),
                    unique.iter().map(|entry| entry.1).collect(),
                ),
                Collinear::Exclude => {
                    let chosen = [unique[0], unique[n - 1]];
                    (
                        chosen.iter().map(|entry| entry.0).collect(),
                        chosen.iter().map(|entry| entry.1).collect(),
                    )
                }
            }
        } else {
            let include_collinear = mode == Collinear::Include;
            let mut lower: Vec<usize> = Vec::new();
            for i in 0..n {
                while lower.len() >= 2 {
                    let turn = cross(
                        unique[lower[lower.len() - 2]].0,
                        unique[lower[lower.len() - 1]].0,
                        unique[i].0,
                    );
                    if if include_collinear { turn < 0 } else { turn <= 0 } {
                        lower.pop();
                    } else {
                        break;
                    }
                }
                lower.push(i);
            }

            let mut upper: Vec<usize> = Vec::new();
            for i in (0..n).rev() {
                while upper.len() >= 2 {
                    let turn = cross(
                        unique[upper[upper.len() - 2]].0,
                        unique[upper[upper.len() - 1]].0,
                        unique[i].0,
                    );
                    if if include_collinear { turn < 0 } else { turn <= 0 } {
                        upper.pop();
                    } else {
                        break;
                    }
                }
                upper.push(i);
            }

            lower.pop();
            upper.pop();
            lower.extend(upper);

            let mut seen = vec![false; n];
            let mut indices = Vec::with_capacity(lower.len());
            for index in lower {
                if !seen[index] {
                    seen[index] = true;
                    indices.push(index);
                }
            }
            (
                indices.iter().map(|&i| unique[i].0).collect(),
                indices.iter().map(|&i| unique[i].1).collect(),
            )
        }
    };

    let area = if vertices.len() < 3 {
        0
    } else {
        signed_area2(&vertices).unsigned_abs()
    };
    let (_, perimeter_milli) = perimeter_and_edges(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli,
    })
}
