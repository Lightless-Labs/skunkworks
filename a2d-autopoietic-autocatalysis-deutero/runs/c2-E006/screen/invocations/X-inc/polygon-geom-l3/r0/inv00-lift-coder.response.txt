#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const MAX_COORD: i64 = 1_000_000_000;
const MAX_INPUT: usize = 4096;

fn in_range(p: Point) -> bool {
    (-MAX_COORD..=MAX_COORD).contains(&p.x) && (-MAX_COORD..=MAX_COORD).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_INPUT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective.first() == effective.last() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices { len: effective.len() });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let scaled = squared_distance(a, b) * 1_000_000;
    let root = isqrt(scaled);
    let remainder = scaled - root * root;
    let rounded = if remainder >= root + 1 { root + 1 } else { root };
    rounded as u64
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut lo = 1u128;
    let mut hi = n.min(1u128 << 64);
    while lo < hi {
        let mid = lo + (hi - lo + 1) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    lo
}

fn signed_area2(points: &[Point]) -> i128 {
    let n = points.len();
    let mut sum = 0i128;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn edge_lengths(points: &[Point]) -> Vec<u64> {
    let n = points.len();
    (0..n)
        .map(|i| rounded_edge_milli(points[i], points[(i + 1) % n]))
        .collect()
}

fn perimeter(edge_milli: &[u64]) -> u64 {
    edge_milli.iter().copied().sum()
}

fn longest_edge(points: &[Point], lengths: &[u64]) -> Option<usize> {
    let mut best: Option<usize> = None;
    for i in 0..lengths.len() {
        let Some(current) = best else {
            best = Some(i);
            continue;
        };
        if lengths[i] > lengths[current] {
            best = Some(i);
        } else if lengths[i] == lengths[current] {
            let (lo_i, hi_i) = ordered_pair(points[i], points[(i + 1) % points.len()]);
            let (lo_c, hi_c) = ordered_pair(points[current], points[(current + 1) % points.len()]);
            if (lo_i, hi_i, i) < (lo_c, hi_c, current) {
                best = Some(i);
            }
        }
    }
    best
}

fn ordered_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed = signed_area2(&polygon);
    let lengths = edge_lengths(&polygon);
    let orientation = match signed.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };
    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation,
        perimeter_milli: perimeter(&lengths),
        longest_edge: longest_edge(&polygon, &lengths),
        edge_milli: lengths,
    })
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_INPUT {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let n = polygon.len();
    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        for i in 0..n {
            if on_segment(polygon[i], polygon[(i + 1) % n], query) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..n {
            let a = polygon[i];
            let b = polygon[(i + 1) % n];
            if (a.y > query.y) != (b.y > query.y) {
                let c = cross(a, b, query);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside { Location::Inside } else { Location::Outside });
    }
    Ok(results)
}

fn monotone_hull(points: &[Point], include_collinear: bool) -> Vec<Point> {
    if points.len() <= 1 {
        return points.to_vec();
    }

    let all_collinear = points
        .iter()
        .skip(2)
        .all(|&p| cross(points[0], points[1], p) == 0);
    if all_collinear {
        if include_collinear {
            return points.to_vec();
        }
        return vec![points[0], *points.last().unwrap()];
    }

    let mut lower = Vec::new();
    for &point in points {
        while lower.len() >= 2 {
            let turn = cross(lower[lower.len() - 2], lower[lower.len() - 1], point);
            if if include_collinear { turn < 0 } else { turn <= 0 } {
                lower.pop();
            } else {
                break;
            }
        }
        lower.push(point);
    }

    let mut upper = Vec::new();
    for &point in points.iter().rev() {
        while upper.len() >= 2 {
            let turn = cross(upper[upper.len() - 2], upper[upper.len() - 1], point);
            if if include_collinear { turn < 0 } else { turn <= 0 } {
                upper.pop();
            } else {
                break;
            }
        }
        upper.push(point);
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);
    lower
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_INPUT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut first_indices = std::collections::BTreeMap::<Point, usize>::new();
    for (index, &point) in points.iter().enumerate() {
        first_indices.entry(point).or_insert(index);
    }
    let unique: Vec<Point> = first_indices.keys().copied().collect();
    let vertices = monotone_hull(&unique, mode == Collinear::Include);
    let source_indices = vertices.iter().map(|p| first_indices[p]).collect::<Vec<_>>();

    let area = if vertices.len() >= 3 {
        signed_area2(&vertices).unsigned_abs()
    } else {
        0
    };
    let lengths = edge_lengths(&vertices);
    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli: perimeter(&lengths),
    })
}
