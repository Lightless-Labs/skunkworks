pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    emitted_in_row: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffered: Vec::new(),
            emitted_in_row: 0,
            closed: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut result = String::new();
        let mut bytes = Vec::with_capacity(self.buffered.len() + input.len());
        bytes.extend_from_slice(&self.buffered);
        bytes.extend_from_slice(input);
        self.buffered.clear();

        let complete_len = bytes.len() / 3 * 3;
        for chunk in bytes[..complete_len].chunks_exact(3) {
            self.encode_group(chunk, false, &mut result);
        }
        self.buffered.extend_from_slice(&bytes[complete_len..]);
        Ok(result)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let mut result = String::new();
        if !self.buffered.is_empty() {
            let tail = self.buffered.clone();
            self.encode_group(&tail, true, &mut result);
            self.buffered.clear();
        }
        Ok(result)
    }

    fn encode_group(&mut self, chunk: &[u8], tail: bool, out: &mut String) {
        let table = alphabet_bytes(self.options.alphabet);
        let a = chunk[0];
        let b = if chunk.len() > 1 { chunk[1] } else { 0 };
        let c = if chunk.len() > 2 { chunk[2] } else { 0 };
        let mut symbols = [0u8; 4];
        symbols[0] = table[(a >> 2) as usize];
        symbols[1] = table[(((a & 0x03) << 4) | (b >> 4)) as usize];
        symbols[2] = table[(((b & 0x0f) << 2) | (c >> 6)) as usize];
        symbols[3] = table[(c & 0x3f) as usize];

        let count = if chunk.len() == 1 {
            2
        } else if chunk.len() == 2 {
            3
        } else {
            4
        };
        let padded = tail && self.options.padding != Padding::Forbidden;
        let total = if padded && count < 4 { 4 } else { count };

        for i in 0..total {
            if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
                if self.emitted_in_row == width {
                    match ending {
                        LineEnding::Lf => out.push('\n'),
                        LineEnding::CrLf => out.push_str("\r\n"),
                    }
                    self.emitted_in_row = 0;
                }
            }
            if i < count {
                out.push(symbols[i] as char);
            } else {
                out.push('=');
            }
            self.emitted_in_row += 1;
        }
    }
}

#[derive(Clone, Copy)]
enum FramingState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: Vec<(u8, usize)>,
    offset: usize,
    terminal: bool,
    pending_second_pad: Option<usize>,
    sticky: Option<Error>,
    closed: bool,
    row_symbols: usize,
    framing: FramingState,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            offset: 0,
            terminal: false,
            pending_second_pad: None,
            sticky: None,
            closed: false,
            row_symbols: 0,
            framing: FramingState::Normal,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(err) = &self.sticky {
            return Err(err.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let here = self.offset;
            self.offset += 1;
            if let Err(err) = self.process_byte(byte, here, &mut output) {
                self.sticky = Some(err.clone());
                return Err(err);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(err) = &self.sticky {
            return Err(err.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let result = self.finish_inner();
        match result {
            Ok(output) => {
                self.closed = true;
                Ok(output)
            }
            Err(err) => {
                self.sticky = Some(err.clone());
                Err(err)
            }
        }
    }

    fn process_byte(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.process_symbol(byte, offset, output),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.process_symbol(byte, offset, output)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.framing {
                    FramingState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.framing = FramingState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        Ok(())
                    }
                    FramingState::NeedSymbol { .. } => {
                        self.framing = FramingState::Normal;
                        self.row_symbols = 0;
                        self.process_wrapped_symbol(byte, offset, width, ending, output)
                    }
                    FramingState::Normal => {
                        if self.row_symbols == width {
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.framing = FramingState::NeedSymbol {
                                        separator_offset: offset,
                                    };
                                    Ok(())
                                }
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.framing = FramingState::NeedLf {
                                        cr_offset: offset,
                                    };
                                    Ok(())
                                }
                                _ => Err(Error::BadWrap { offset }),
                            }
                        } else if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                            Err(Error::BadWrap { offset })
                        } else {
                            self.process_wrapped_symbol(byte, offset, width, ending, output)
                        }
                    }
                }
            }
        }
    }

    fn process_wrapped_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        _width: usize,
        _ending: LineEnding,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        self.row_symbols += 1;
        self.process_symbol(byte, offset, output)
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal {
            return Err(Error::BadPadding { offset });
        }

        if let Some(pad_offset) = self.pending_second_pad {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.group.push((byte, offset));
            self.pending_second_pad = None;
            self.emit_group(output)?;
            self.terminal = true;
            let _ = pad_offset;
            return Ok(());
        }

        let position = self.group.len();
        if byte == b'=' {
            if position < 2 {
                return Err(Error::BadPadding { offset });
            }
            if position == 2 {
                if self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset });
                }
                self.pending_second_pad = Some(offset);
                self.group.push((byte, offset));
                return Ok(());
            }
            if position == 3 {
                if self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset });
                }
                self.group.push((byte, offset));
                self.emit_group(output)?;
                self.terminal = true;
                return Ok(());
            }
        }

        let value = decode_digit(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset, byte })?;
        self.group.push((value, offset));
        if self.group.len() == 4 {
            self.emit_group(output)?;
        }
        Ok(())
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.group.len() != 4 {
            return Ok(());
        }
        let a = self.group[0].0;
        let b = self.group[1].0;
        let third = self.group[2].0;
        let fourth = self.group[3].0;

        if self.group[2].0 == b'=' {
            output.push((a << 2) | (b >> 4));
        } else if self.group[3].0 == b'=' {
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (third >> 2));
        } else {
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (third >> 2));
            output.push((third << 6) | fourth);
        }
        self.group.clear();
        Ok(())
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        let mut output = Vec::new();
        match self.framing {
            FramingState::NeedLf { cr_offset } => {
                return Err(Error::BadWrap { offset: cr_offset });
            }
            FramingState::NeedSymbol { separator_offset } => {
                return Err(Error::BadWrap {
                    offset: separator_offset,
                });
            }
            FramingState::Normal => {}
        }

        if let Some(offset) = self.pending_second_pad {
            return Err(Error::BadPadding { offset });
        }

        if self.group.is_empty() {
            return Ok(output);
        }

        match self.options.padding {
            Padding::Required => Err(Error::Truncated {
                offset: self.group[0].1,
            }),
            Padding::Forbidden | Padding::Optional => {
                if self.group.iter().any(|(value, _)| *value == b'=') {
                    return Err(Error::BadPadding {
                        offset: self.group[0].1,
                    });
                }
                match self.group.len() {
                    1 => Err(Error::Truncated {
                        offset: self.group[0].1,
                    }),
                    2 => {
                        output.push((self.group[0].0 << 2) | (self.group[1].0 >> 4));
                        Ok(output)
                    }
                    3 => {
                        output.push((self.group[0].0 << 2) | (self.group[1].0 >> 4));
                        output.push((self.group[1].0 << 4) | (self.group[2].0 >> 2));
                        Ok(output)
                    }
                    _ => Ok(output),
                }
            }
        }
    }
}

fn decode_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    alphabet_bytes(alphabet)
        .iter()
        .position(|&candidate| candidate == byte)
        .map(|n| n as u8)
}

fn digit_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,

pub struct Encoder {
    options: Options,
    tail: [u8; 2],
    tail_len: usize,
    column: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: [0; 2],
            tail_len: 0,
            column: 0,
            closed: false,
        })
    }

    fn emit_symbol(&mut self, output: &mut String, symbol: u8) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            if self.column == width {
                match ending {
                    LineEnding::Lf => output.push('\n'),
                    LineEnding::CrLf => output.push_str("\r\n"),
                }
                self.column = 0;
            }
        }
        output.push(symbol as char);
        self.column += 1;
    }

    fn emit_group(&mut self, output: &mut String, a: u8, b: u8, c: u8) {
        let alphabet = alphabet_bytes(self.options.alphabet);
        self.emit_symbol(output, alphabet[(a >> 2) as usize]);
        self.emit_symbol(output, alphabet[(((a & 3) << 4) | (b >> 4)) as usize]);
        self.emit_symbol(output, alphabet[(((b & 15) << 2) | (c >> 6)) as usize]);
        self.emit_symbol(output, alphabet[(c & 63) as usize]);
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 2 && index < input.len() {
                self.tail[self.tail_len] = input[index];
                self.tail_len += 1;
                index += 1;
            }
            if self.tail_len == 2 && index < input.len() {
                self.emit_group(&mut output, self.tail[0], self.tail[1], input[index]);
                self.tail_len = 0;
                index += 1;
            } else if self.tail_len < 2 || index == input.len() {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            self.emit_group(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        self.tail_len = input.len() - index;
        for i in 0..self.tail_len {
            self.tail[i] = input[index + i];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;

        let mut output = String::new();
        if self.tail_len == 1 {
            let alphabet = alphabet_bytes(self.options.alphabet);
            let a = self.tail[0];
            self.emit_symbol(&mut output, alphabet[(a >> 2) as usize]);
            self.emit_symbol(&mut output, alphabet[((a & 3) << 4) as usize]);
            if self.options.padding != Padding::Forbidden {
                self.emit_symbol(&mut output, b'=');
                self.emit_symbol(&mut output, b'=');
            }
        } else if self.tail_len == 2 {
            let alphabet = alphabet_bytes(self.options.alphabet);
            let a = self.tail[0];
            let b = self.tail[1];
            self.emit_symbol(&mut output, alphabet[(a >> 2) as usize]);
            self.emit_symbol(&mut output, alphabet[(((a & 3) << 4) | (b >> 4)) as usize]);
            self.emit_symbol(&mut output, alphabet[((b & 15) << 2) as usize]);
            if self.options.padding != Padding::Forbidden {
                self.emit_symbol(&mut output, b'=');
            }
        }
        self.tail_len = 0;
        Ok(output)
    }
}

enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: [u8; 4],
    group_offsets: [usize; 4],
    group_len: usize,
    pending_second_pad: Option<usize>,
    terminal_padding: bool,
    offset: usize,
    column: usize,
    frame: FrameState,
    closed: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: [0; 4],
            group_offsets: [0; 4],
            group_len: 0,
            pending_second_pad: None,
            terminal_padding: false,
            offset: 0,
            column: 0,
            frame: FrameState::Normal,
            closed: false,
            sticky_error: None,
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn decode_complete_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let a = digit_value(self.group[0], self.options.alphabet).unwrap();
        let b = digit_value(self.group[1], self.options.alphabet).unwrap();

        if self.group[2] == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding {
                    offset: self.group_offsets[2],
                });
            }
            output.push((a << 2) | (b >> 4));
            self.terminal_padding = true;
        } else {
            let c = digit_value(self.group[2], self.options.alphabet).unwrap();
            if self.group[3] == b'=' {
                if self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding {
                        offset: self.group_offsets[3],
                    });
                }
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                self.terminal_padding = true;
            } else {
                let d = digit_value(self.group[3], self.options.alphabet).unwrap();
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
            }
        }

        self.group_len = 0;
        Ok(())
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if self.pending_second_pad.is_some() {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.group[3] = b'=';
            self.group_offsets[3] = offset;
            self.pending_second_pad = None;
            self.decode_complete_group(output)?;
            return Ok(());
        }

        let position = self.group_len;
        if byte == b'=' {
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.group[2] = b'=';
                    self.group_offsets[2] = offset;
                    self.group_len = 3;
                    self.pending_second_pad = Some(offset);
                    return Ok(());
                }
                3 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.group[3] = b'=';
                    self.group_offsets[3] = offset;
                    self.decode_complete_group(output)?;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = digit_value(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset, byte })?;
        self.group[position] = byte;
        self.group_offsets[position] = offset;
        self.group_len += 1;
        if self.group_len == 4 {
            let _ = value;
            self.decode_complete_group(output)?;
        }
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            match self.frame {
                FrameState::NeedLf { cr_offset } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.frame = FrameState::NeedSymbol {
                        separator_offset: cr_offset,
                    };
                    return Ok(());
                }
                FrameState::NeedSymbol { .. } => {
                    self.frame = FrameState::Normal;
                }
                FrameState::Normal => {
                    if self.column == width {
                        match ending {
                            LineEnding::Lf if byte == b'\n' => {
                                self.frame = FrameState::NeedSymbol {
                                    separator_offset: offset,
                                };
                                self.column = 0;
                                return Ok(());
                            }
                            LineEnding::CrLf if byte == b'\r' => {
                                self.frame = FrameState::NeedLf { cr_offset: offset };
                                self.column = 0;
                                return Ok(());
                            }
                            _ => return Err(Error::BadWrap { offset }),
                        }
                    } else if byte == b'\r' || byte == b'\n' || byte == 9 || byte == 11 || byte == 12 || byte == 32 {
                        return Err(Error::BadWrap { offset });
                    }
                }
            }

            if let FrameState::NeedSymbol { separator_offset } = self.frame {
                if byte == b'\r' || byte == b'\n' || byte == 9 || byte == 11 || byte == 12 || byte == 32 {
                    return Err(Error::BadWrap { offset });
                }
                self.frame = FrameState::Normal;
                let _ = separator_offset;
            }

            if matches!(self.frame, FrameState::Normal) {
                self.column += 1;
            }
        } else if self.options.whitespace == Whitespace::IgnoreAscii
            && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
        {
            return Ok(());
        }

        self.process_symbol(byte, offset, output)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut output) {
                return Err(self.fail(error));
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        if let FrameState::NeedLf { cr_offset } = self.frame {
            let error = self.fail(Error::BadWrap { offset: cr_offset });
            return Err(error);
        }
        if let FrameState::NeedSymbol { separator_offset } = self.frame {
            let error = self.fail(Error::BadWrap {
                offset: separator_offset,
            });
            return Err(error);
        }
        if let Some(offset) = self.pending_second_pad {
            let error = self.fail(Error::BadPadding { offset });
            return Err(error);
        }

        let mut output = Vec::new();
        if self.group_len == 0 {
            self.closed = true;
            return Ok(output);
        }

        if self.options.padding == Padding::Required {
            let error = self.fail(Error::Truncated {
                offset: self.group_offsets[0],
            });
            return Err(error);
        }

        match self.group_len {
            1 => {
                let error = self.fail(Error::Truncated {
                    offset: self.group_offsets[0],
                });
                Err(error)
            }
            2 => {
                let a = digit_value(self.group[0], self.options.alphabet).unwrap();
                let b = digit_value(self.group[1], self.options.alphabet).unwrap();
                output.push((a << 2) | (b >> 4));
                self.closed = true;
                Ok(output)
            }
            3 => {
                let a = digit_value(self.group[0], self.options.alphabet).unwrap();
                let b = digit_value(self.group[1], self.options.alphabet).unwrap();
                let c = digit_value(self.group[2], self.options.alphabet).unwrap();
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                self.closed = true;
                Ok(output)
            }
            _ => unreachable!(),
        }
    }
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    row_symbols: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            pending: Vec::new(),
            row_symbols: 0,
            closed: false,
        })
    }

    fn append_symbols(&mut self, output: &mut String, symbols: &[u8]) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            for &symbol in symbols {
                if self.row_symbols == width {
                    match ending {
                        LineEnding::Lf => output.push('\n'),
                        LineEnding::CrLf => output.push_str("\r\n"),
                    }
                    self.row_symbols = 0;
                }
                output.push(symbol as char);
                self.row_symbols += 1;
            }
        } else {
            for &symbol in symbols {
                output.push(symbol as char);
            }
        }
    }

    fn encode_block(&mut self, block: &[u8], output: &mut String) {
        let alphabet = alphabet_bytes(self.options.alphabet);
        let a = block[0];
        let b = block[1];
        let c = block[2];
        let symbols = [
            alphabet[(a >> 2) as usize],
            alphabet[(((a & 0x03) << 4) | (b >> 4)) as usize],
            alphabet[(((b & 0x0f) << 2) | (c >> 6)) as usize],
            alphabet[(c & 0x3f) as usize],
        ];
        self.append_symbols(output, &symbols);
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        for &byte in input {
            self.pending.push(byte);
            if self.pending.len() == 3 {
                let block = [
                    self.pending[0],
                    self.pending[1],
                    self.pending[2],
                ];
                self.encode_block(&block, &mut output);
                self.pending.clear();
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let alphabet = alphabet_bytes(self.options.alphabet);
        match self.pending.len() {
            0 => {}
            1 => {
                let a = self.pending[0];
                let mut symbols = vec![
                    alphabet[(a >> 2) as usize],
                    alphabet[((a & 0x03) << 4) as usize],
                ];
                if self.options.padding != Padding::Forbidden {
                    symbols.extend_from_slice(b"==");
                }
                self.append_symbols(&mut output, &symbols);
            }
            2 => {
                let a = self.pending[0];
                let b = self.pending[1];
                let mut symbols = vec![
                    alphabet[(a >> 2) as usize],
                    alphabet[(((a & 0x03) << 4) | (b >> 4)) as usize],
                    alphabet[((b & 0x0f) << 2) as usize],
                ];
                if self.options.padding != Padding::Forbidden {
                    symbols.push(b'=');
                }
                self.append_symbols(&mut output, &symbols);
            }
            _ => unreachable!(),
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

#[derive(Clone, Copy)]
enum WrapState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    symbols: Vec<(u8, usize)>,
    absolute_offset: usize,
    row_symbols: usize,
    wrap_state: WrapState,
    padded_finished: bool,
    closed: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            symbols: Vec::with_capacity(4),
            absolute_offset: 0,
            row_symbols: 0,
            wrap_state: WrapState::Normal,
            padded_finished: false,
            closed: false,
            sticky_error: None,
        })
    }

    fn scan_symbol(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.padded_finished {
            return Err(Error::BadPadding { offset });
        }

        if self.symbols.len() == 2 && self.symbols[1].0 == b'=' {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.symbols.push((byte, offset));
            self.emit_group(output)?;
            return Ok(());
        }

        let position = self.symbols.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.symbols.push((byte, offset));
                    return Ok(());
                }
                3 => {
                    self.symbols.push((byte, offset));
                    self.emit_group(output)?;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        if alphabet_digit(byte, self.options.alphabet).is_none() {
            return Err(Error::InvalidByte { offset, byte });
        }
        self.symbols.push((byte, offset));

        if self.symbols.len() == 4 {
            self.emit_group(output)?;
        }
        Ok(())
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let digits: Vec<Option<u8>> = self
            .symbols
            .iter()
            .map(|(byte, _)| {
                if *byte == b'=' {
                    None
                } else {
                    alphabet_digit(*byte, self.options.alphabet)
                }
            })
            .collect();

        let a = digits[0].ok_or_else(|| Error::BadPadding {
            offset: self.symbols[0].1,
        })?;
        let b = digits[1].ok_or_else(|| Error::BadPadding {
            offset: self.symbols[1].1,
        })?;

        match (digits[2], digits[3]) {
            (Some(c), Some(d)) => {
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
            }
            (None, None) => {
                output.push((a << 2) | (b >> 4));
                self.padded_finished = true;
            }
            (Some(c), None) => {
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                self.padded_finished = true;
            }
            (None, Some(_)) => {
                return Err(Error::BadPadding {
                    offset: self.symbols[3].1,
                });
            }
        }
        self.symbols.clear();
        Ok(())
    }

    fn frame_byte(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        match self.options.whitespace {
            Whitespace::None => Ok(true),
            Whitespace::IgnoreAscii => {
                Ok(!matches!(byte, 9 | 10 | 11 | 12 | 13 | 32))
            }
            Whitespace::Wrapped { width, ending } => {
                match self.wrap_state {
                    WrapState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = WrapState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        Ok(false)
                    }
                    WrapState::NeedSymbol { .. } => {
                        if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = WrapState::Normal;
                        self.row_symbols = 0;
                        self.row_symbols += 1;
                        Ok(true)
                    }
                    WrapState::Normal => {
                        if self.row_symbols == width {
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.wrap_state = WrapState::NeedSymbol {
                                        separator_offset: offset,
                                    };
                                    Ok(false)
                                }
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.wrap_state = WrapState::NeedLf { cr_offset: offset };
                                    Ok(false)
                                }
                                _ => Err(Error::BadWrap { offset }),
                            }
                        } else if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                            Err(Error::BadWrap { offset })
                        } else {
                            self.row_symbols += 1;
                            Ok(true)
                        }
                    }
                }
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.absolute_offset;
            self.absolute_offset += 1;

            let scan = match self.frame_byte(byte, offset) {
                Ok(true) => self.scan_symbol(byte, offset, &mut output),
                Ok(false) => Ok(()),
                Err(error) => Err(error),
            };
            if let Err(error) = scan {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let error = match self.wrap_state {
            WrapState::NeedLf { cr_offset } => Some(Error::BadWrap { offset: cr_offset }),
            WrapState::NeedSymbol { separator_offset } => {
                Some(Error::BadWrap { offset: separator_offset })
            }
            WrapState::Normal => None,
        };
        if let Some(error) = error {
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.symbols.len() == 2 && self.symbols[1].0 == b'=' {
            let error = Error::BadPadding {
                offset: self.symbols[1].1,
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let mut output = Vec::new();
        if !self.symbols.is_empty() {
            match (self.options.padding, self.symbols.len()) {
                (Padding::Forbidden | Padding::Optional, 1) => {
                    let error = Error::Truncated {
                        offset: self.symbols[0].1,
                    };
                    self.sticky_error = Some(error.clone());
                    return Err(error);
                }
                (Padding::Forbidden | Padding::Optional, 2) => {
                    let a = alphabet_digit(self.symbols[0].0, self.options.alphabet).unwrap();
                    let b = alphabet_digit(self.symbols[1].0, self.options.alphabet).unwrap();
                    output.push((a << 2) | (b >> 4));
                }
                (Padding::Forbidden | Padding::Optional, 3) => {
                    let a = alphabet_digit(self.symbols[0].0, self.options.alphabet).unwrap();
                    let b = alphabet_digit(self.symbols[1].0, self.options.alphabet).unwrap();
                    let c = alphabet_digit(self.symbols[2].0, self.options.alphabet).unwrap();
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                }
                (Padding::Required, _) => {
                    let error = Error::Truncated {
                        offset: self.symbols[0].1,
                    };
                    self.sticky_error = Some(error.clone());
                    return Err(error);
                }
                _ => unreachable!(),
            }
        }

        self.symbols.clear();
        self.closed = true;
        Ok(output)
    }
}
