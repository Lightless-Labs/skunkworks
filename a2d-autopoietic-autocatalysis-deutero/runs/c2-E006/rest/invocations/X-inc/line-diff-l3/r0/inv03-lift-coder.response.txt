#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
enum Row {
    Context(String),
    Remove(String),
    Add(String),
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return None;
    }
    let mut n = 0usize;
    for b in s.bytes() {
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let inner = line.strip_prefix("@@ -")?.strip_suffix(" @@")?;
    let (old, new) = inner.split_once(" +")?;
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    let (a, b, c, d) = (
        parse_number(a)?,
        parse_number(b)?,
        parse_number(c)?,
        parse_number(d)?,
    );
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn full_start(anchor: usize, count: usize) -> usize {
    if count > 0 { anchor - 1 } else { anchor }
}

fn physical_lines(s: &str) -> Vec<&str> {
    if s.is_empty() {
        return Vec::new();
    }
    let mut lines: Vec<&str> = s.split_terminator('\n').collect();
    if !s.ends_with('\n') {
        // The final unterminated segment is still a physical line.
    }
    lines.shrink_to_fit();
    lines
}

fn abs_diff_signed(a: i128, b: i128) -> u128 {
    (a - b).unsigned_abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let patch_lines = physical_lines(patch);
    if !patch.is_empty() && !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    if patch.is_empty() {
        return apply_identity(source);
    }

    if patch_lines.get(0) != Some(&"--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if patch_lines.get(1) != Some(&"+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if patch_lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_old_end: i128 = 0;
    let mut preceding_net: i128 = 0;

    while i < patch_lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(patch_lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while i < patch_lines.len() && !patch_lines[i].starts_with("@@") {
            let line = patch_lines[i];
            let row_line = i + 1;
            if line.as_bytes().contains(&b'\r') || line.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &line[1..];
            match line.as_bytes()[0] {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                    rows.push(Row::Context(payload.to_owned()));
                }
                b'-' => {
                    old_seen += 1;
                    rows.push(Row::Remove(payload.to_owned()));
                }
                b'+' => {
                    new_seen += 1;
                    rows.push(Row::Add(payload.to_owned()));
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            }
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| matches!(r, Row::Add(_) | Row::Remove(_))) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_start_i = old_start as i128;
        if old_start_i < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start_i + preceding_net;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_end = old_start_i + b as i128;
        preceding_net += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() == 17 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut working = validate_source(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| matches!(r, Row::Context(_))).count();
        let trail = hunk.rows.iter().rev().take_while(|r| matches!(r, Row::Context(_))).count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut best_starts = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                let mut match_payloads = Vec::<&str>::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    match row {
                        Row::Context(s) | Row::Remove(s) => match_payloads.push(s),
                        Row::Add(_) => {}
                    }
                }
                let span = hunk.old_count - l - r;
                if match_payloads.len() != span {
                    continue;
                }
                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_diff_signed(s as i128, expected);
                    if distance > options.max_offset as u128 {
                        continue;
                    }
                    if match_payloads
                        .iter()
                        .enumerate()
                        .all(|(j, p)| working[s + l + j] == *p)
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best_starts.push((s, l, r));
                            }
                            Some(old) if rank < old => {
                                best_rank = Some(rank);
                                best_starts.clear();
                                best_starts.push((s, l, r));
                            }
                            Some(old) if rank == old => best_starts.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch { hunk: index + 1 });
        }

        let mut distinct: Vec<usize> = best_starts.iter().map(|x| x.0).collect();
        distinct.sort_unstable();
        distinct.dedup();
        if distinct.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }

        let s = distinct[0];
        let (chosen_l, chosen_r) = best_starts
            .iter()
            .filter(|x| x.0 == s)
            .map(|x| (x.1, x.2))
            .min_by_key(|x| x.0)
            .unwrap();

        let mut replacement = Vec::<String>::new();
        for row in &hunk.rows[chosen_l..hunk.rows.len() - chosen_r] {
            match row {
                Row::Context(v) | Row::Add(v) => replacement.push(v.clone()),
                Row::Remove(_) => {}
            }
        }
        let begin = s + chosen_l;
        let end = s + hunk.old_count - chosen_r;
        working.splice(begin..end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut out = String::new();
    for line in working {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}

fn apply_identity(source: &str) -> Result<String, ApplyError> {
    let lines = validate_source(source)?;
    let mut out = String::new();
    for line in lines {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}

fn validate_source(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source
        .strip_suffix('\n')
        .unwrap()
        .split('\n')
        .map(str::to_owned)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}
