    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },

    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },

struct Row {
    kind: u8,
    payload: String,
}

#[derive(Clone)]
struct Hunk {
    line: usize,
    old_start: i64,
    old_count: usize,
    new_start: i64,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_decimal(bytes: &[u8]) -> Option<usize> {
    if bytes.is_empty() || (bytes.len() > 1 && bytes[0] == b'0') {
        return None;
    }
    let mut n = 0usize;
    for &b in bytes {
        if !b.is_ascii_digit() {
            return None;
        }
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(i64, usize, i64, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let bytes = line.as_bytes();
    if !line.starts_with("@@ -") || !line.ends_with(" @@") {
        return Err(bad());
    }
    let inner = &line[4..line.len() - 3];
    let Some((old, new)) = inner.split_once(" +") else {
        return Err(bad());
    };
    if new.contains(' ') {
        return Err(bad());
    }
    let Some((a, b)) = old.split_once(',') else {
        return Err(bad());
    };
    let Some((c, d)) = new.split_once(',') else {
        return Err(bad());
    };
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return Err(bad());
    }
    let (Some(a), Some(b), Some(c), Some(d)) = (
        parse_decimal(a.as_bytes()),
        parse_decimal(b.as_bytes()),
        parse_decimal(c.as_bytes()),
        parse_decimal(d.as_bytes()),
    ) else {
        return Err(bad());
    };
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    let _ = bytes;
    let old_start = if b > 0 { a as i64 - 1 } else { a as i64 };
    let new_start = if d > 0 { c as i64 - 1 } else { c as i64 };
    Ok((old_start, b, new_start, d))
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }
    let lines = patch_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() <= 2 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut idx = 2;
    let mut prev_old_end = 0i64;
    let mut net = 0i64;

    while idx < lines.len() {
        if !lines[idx].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: idx + 1 });
        }
        let header_line = idx + 1;
        let (old_start, old_count, new_start, new_count) =
            parse_hunk_header(lines[idx], header_line)?;
        idx += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while idx < lines.len() && !lines[idx].starts_with("@@") {
            let row_line = idx + 1;
            let bytes = lines[idx].as_bytes();
            if bytes.is_empty()
                || !matches!(bytes[0], b' ' | b'-' | b'+')
                || bytes.contains(&b'\r')
            {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let kind = bytes[0];
            let payload = &lines[idx][1..];
            rows.push(Row {
                kind,
                payload: payload.to_owned(),
            });
            if kind == b' ' || kind == b'-' {
                old_seen += 1;
            }
            if kind == b' ' || kind == b'+' {
                new_seen += 1;
            }
            idx += 1;
        }

        if old_seen != old_count || new_seen != new_count {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < prev_old_end || new_start != old_start + net {
            return Err(ApplyError::Misordered { line: header_line });
        }
        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        prev_old_end = old_start + old_count as i64;
        net += new_count as i64 - old_count as i64;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count,
            new_start,
            new_count,
            rows,
        });
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_diff(a: i64, b: i64) -> u64 {
    (a as i128 - b as i128).unsigned_abs() as u64
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net = 0i64;
    let mut drift = 0i64;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
        let expected = hunk.old_start + net + drift;
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);

        let mut best_rank: Option<(usize, u64)> = None;
        let mut best_starts = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > lead + trail {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for (ri, row) in hunk.rows.iter().enumerate() {
                    if ri < l || ri >= hunk.rows.len().saturating_sub(r) {
                        continue;
                    }
                    if row.kind == b' ' || row.kind == b'-' {
                        old_payloads.push(row.payload.as_str());
                    }
                }
                let retained_old = hunk.old_count - l - r;
                if retained_old != old_payloads.len() {
                    continue;
                }
                for s in 0..=working.len() {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let dist = abs_diff(s as i64, expected);
                    if dist > options.max_offset as u64 {
                        continue;
                    }
                    if old_payloads
                        .iter()
                        .enumerate()
                        .all(|(i, p)| working[s + l + i] == *p)
                    {
                        let rank = (l + r, dist);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best_starts.clear();
                                best_starts.push((s, l, r));
                            }
                            Some(prev) if rank < prev => {
                                best_rank = Some(rank);
                                best_starts.clear();
                                best_starts.push((s, l, r));
                            }
                            Some(prev) if rank == prev => best_starts.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }
        best_starts.sort_by_key(|&(s, l, _)| (s, l));
        let mut distinct = best_starts.iter().map(|x| x.0).collect::<Vec<_>>();
        distinct.sort_unstable();
        distinct.dedup();
        if distinct.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }
        let (s, l, r) = best_starts[0];

        let mut replacement = Vec::new();
        for (ri, row) in hunk.rows.iter().enumerate() {
            if ri < l || ri >= hunk.rows.len().saturating_sub(r) {
                continue;
            }
            if row.kind == b' ' || row.kind == b'+' {
                replacement.push(row.payload.clone());
            }
        }
        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
        drift = s as i64 - (hunk.old_start + net);
        net += hunk.new_count as i64 - hunk.old_count as i64;
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

fn parse_decimal(s: &str) -> Option<usize> {

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let inner = line.strip_prefix("@@ -")?.strip_suffix(" @@")?;
    let (old, new) = inner.split_once(" +")?;
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') ||

    let d = parse_decimal(d)?;

        return None;
    }
    Some((a, b, c, d))
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines = split_patch_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_old_end = 0usize;
    let mut net: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let row = lines[i];
            if row.contains('\r') || row.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &row[1..];
            if !payload.is_char_boundary(0) {
                return Err(ApplyError::BadBody { line: row_line });
            }
            match row.as_bytes()[0] {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                    rows.push(Row::Context(payload.to_string()));
                }
                b'-' => {
                    old_seen += 1;
                    rows.push(Row::Remove(payload.to_string()));
                }
                b'+' => {
                    new_seen += 1;
                    rows.push(Row::Add(payload.to_string()));
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            }
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| matches!(r, Row::Add(_) | Row::Remove(_))) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i128 + net;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_start.saturating_add(b);
        net += d as i128 - b as i128;

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source.split_terminator('\n').map(str::to_string).collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_diff_signed(a: i128, b: i128) -> i128 {
    (a - b).abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| matches!(r, Row::Context(_))).count();
        let trail = hunk.rows.iter().rev().take_while(|r| matches!(r, Row::Context(_))).count();
        let expected = hunk.old_start as i128 + net + drift;
        let fuzz = options.fuzz;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=fuzz.min(lead) {
            for r in 0..=fuzz.min(trail) {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len().saturating_sub(r)] {
                    match row {
                        Row::Context(s) | Row::Remove(s) => old_payloads.push(s.as_str()),
                        Row::Add(_) => {}
                    }
                }
                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_diff_signed(s as i128, expected);
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let start = s + l;
                    let end = s + hunk.old_count - r;
                    if old_payloads.len() != end - start ||
                        !old_payloads.iter().zip(&working[start..end]).all(|(a, b)| *a == b) {
                        continue;
                    }
                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best.clear();
                            best.push((s, l, r));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best.clear();
                            best.push((s, l, r));
                        }
                        Some(current) if rank == current => best.push((s, l, r)),
                        _ => {}
                    }
                }
            }
        }

        if best.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_index + 1 });
        }

        let mut starts: Vec<usize> = best.iter().map(|x| x.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        let s = starts[0];
        let (_, l, r) = best.into_iter().filter(|x| x.0 == s).min_by_key(|x| x.1).unwrap();
        let from = s + l;
        let to = s + hunk.old_count - r;

        let trimmed = &hunk.rows[l..hunk.rows.len().saturating_sub(r)];
        let mut replacement = Vec::new();
        for row in trimmed {
            match row {
                Row::Context(text) | Row::Add(text) => replacement.push(text.clone()),
                Row::Remove(_) => {}
            }
        }
        working.splice(from..to, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

    pub fuzz: usize,

pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Row {
    prefix: u8,
    payload: String,
}

#[derive(Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn physical_lines(s: &str) -> Vec<&str> {
    if s.is_empty() {
        return Vec::new();
    }
    let mut lines: Vec<&str> = s.split('\n').collect();
    if lines.last() == Some(&"") {
        lines.pop();
    }
    lines
}

fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty()
        || (s.len() > 1 && s.starts_with('0'))
        || !s.bytes().all(|b| b.is_ascii_digit())
    {
        return None;
    }
    let mut value = 0usize;
    for b in s.bytes() {
        value = value.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if value > 128 {
            return None;
        }
    }
    Some(value)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let inner = line.strip_prefix("@@ -")?.strip_suffix(" @@")?;
    let (old, new) = inner.split_once(" +")?;
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    let a = parse_decimal(a)?;
    let b = parse_decimal(b)?;
    let c = parse_decimal(c)?;
    let d = parse_decimal(d)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 { start - 1 } else { start }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for (idx, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(idx).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: idx + 1 });
        }
    }
    if lines.len() <= 2 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end = 0usize;
    let mut net: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let (a, b, c, d) = parse_hunk_header(lines[i])
            .ok_or(ApplyError::BadHunkHeader { line: header_line })?;
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);
        i += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let text = lines[i];
            let bytes = text.as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') || text.contains('\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            let payload = &text[1..];
            let prefix = bytes[0];
            if prefix == b' ' || prefix == b'-' {
                old_seen += 1;
            }
            if prefix == b' ' || prefix == b'+' {
                new_seen += 1;
            }
            rows.push(Row {
                prefix,
                payload: payload.to_owned(),
            });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }
        if old_start < previous_old_end
            || (new_start as i128) != (old_start as i128 + net)
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_start + b;
        net += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let lines = physical_lines(source);
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines.into_iter().map(str::to_owned).collect())
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (idx, hunk) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < hunk.rows.len() && hunk.rows[lead].prefix == b' ' {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < hunk.rows.len() && hunk.rows[hunk.rows.len() - 1 - trail].prefix == b' ' {
            trail += 1;
        }
        let expected = hunk.old_start as i128 + net + drift;
        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_by_start: Vec<(usize, usize, usize)> = Vec::new();

        let max_left = options.fuzz.min(lead);
        let max_right = options.fuzz.min(trail);
        for s in 0..=working.len() {
            if s + hunk.old_count > working.len() {
                continue;
            }
            let distance = (s as i128 - expected).abs();
            if distance > options.max_offset as i128 {
                continue;
            }
            for left in 0..=max_left {
                for right in 0..=max_right {
                    if left + right > hunk.old_count {
                        continue;
                    }
                    let mut old_payloads = Vec::new();
                    let mut new_payloads = Vec::new();
                    let mut context_index = 0usize;
                    for (row_idx, row) in hunk.rows.iter().enumerate() {
                        if row.prefix == b' ' {
                            let dropped = row_idx < left
                                || row_idx >= hunk.rows.len().saturating_sub(right);
                            if !dropped {
                                old_payloads.push(row.payload.as_str());
                                new_payloads.push(row.payload.as_str());
                            } else {
                                context_index += 1;
                            }
                        } else if row.prefix == b'-' {
                            old_payloads.push(row.payload.as_str());
                        } else {
                            new_payloads.push(row.payload.as_str());
                        }
                    }
                    let start = s + left;
                    let end = s + hunk.old_count - right;
                    if end > working.len() || start > end {
                        continue;
                    }
                    if old_payloads.len() != end - start
                        || !old_payloads
                            .iter()
                            .zip(&working[start..end])
                            .all(|(a, b)| *a == b)
                    {
                        continue;
                    }

                    let rank = (left + right, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, left, right));
                        }
                        Some(old_rank) if rank < old_rank => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, left, right));
                        }
                        Some(old_rank) if rank == old_rank => {
                            if let Some(existing) = best_by_start.iter_mut().find(|x| x.0 == s) {
                                if left < existing.1 {
                                    *existing = (s, left, right);
                                }
                            } else {
                                best_by_start.push((s, left, right));
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_by_start.is_empty() {
            return Err(ApplyError::NoMatch { hunk: idx + 1 });
        }
        best_by_start.sort_by_key(|x| x.0);
        if best_by_start.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: idx + 1,
                first: best_by_start[0].0,
                second: best_by_start[1].0,
            });
        }

        let (s, left, right) = best_by_start[0];
        let mut replacement = Vec::new();
        for (row_idx, row) in hunk.rows.iter().enumerate() {
            if row.prefix == b' ' {
                let dropped = row_idx < left || row_idx >= hunk.rows.len().saturating_sub(right);
                if !dropped {
                    replacement.push(row.payload.clone());
                }
            } else if row.prefix == b'+' {
                replacement.push(row.payload.clone());
            }
        }
        let start = s + left;
        let end = s + hunk.old_count - right;
        working.splice(start..end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let patch_lines = physical_lines(patch);
    if !patch.is_empty() && !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    if patch.is_empty() {
        return apply_identity(source);
    }

    if patch_lines.get(0) != Some(&"--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if patch_lines.get(1) != Some(&"+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if patch_lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_old_end: i128 = 0;
    let mut preceding_net: i128 = 0;

    while i < patch_lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(patch_lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while i < patch_lines.len() && !patch_lines[i].starts_with("@@") {
            let line = patch_lines[i];
            let row_line = i + 1;
            if line.as_bytes().contains(&b'\r') || line.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &line[1..];
            match line.as_bytes()[0] {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                    rows.push(Row::Context(payload.to_owned()));
                }
                b'-' => {
                    old_seen += 1;
                    rows.push(Row::Remove(payload.to_owned()));
                }
                b'+' => {
                    new_seen += 1;
                    rows.push(Row::Add(payload.to_owned()));
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            }
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| matches!(r, Row::Add(_) | Row::Remove(_))) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_start_i = old_start as i128;
        if old_start_i < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start_i + preceding_net;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_end = old_start_i + b as i128;
        preceding_net += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() == 17 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut working = validate_source(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| matches!(r, Row::Context(_))).count();
        let trail = hunk.rows.iter().rev().take_while(|r| matches!(r, Row::Context(_))).count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut best_starts = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                let mut match_payloads = Vec::<&str>::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    match row {
                        Row::Context(s) | Row::Remove(s) => match_payloads.push(s),
                        Row::Add(_) => {}
                    }
                }
                let span = hunk.old_count - l - r;
                if match_payloads.len() != span {
                    continue;
                }
                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_diff_signed(s as i128, expected);
                    if distance > options.max_offset as u128 {
                        continue;
                    }
                    if match_payloads
                        .iter()
                        .enumerate()
                        .all(|(j, p)| working[s + l + j] == *p)
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best_starts.push((s, l, r));
                            }
                            Some(old) if rank < old => {
                                best_rank = Some(rank);
                                best_starts.clear();
                                best_starts.push((s, l, r));
                            }
                            Some(old) if rank == old => best_starts.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch { hunk: index + 1 });
        }

        let mut distinct: Vec<usize> = best_starts.iter().map(|x| x.0).collect();
        distinct.sort_unstable();
        distinct.dedup();
        if distinct.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }

        let s = distinct[0];
        let (chosen_l, chosen_r) = best_starts
            .iter()
            .filter(|x| x.0 == s)
            .map(|x| (x.1, x.2))
            .min_by_key(|x| x.0)
            .unwrap();

        let mut replacement = Vec::<String>::new();
        for row in &hunk.rows[chosen_l..hunk.rows.len() - chosen_r] {
            match row {
                Row::Context(v) | Row::Add(v) => replacement.push(v.clone()),
                Row::Remove(_) => {}
            }
        }
        let begin = s + chosen_l;
        let end = s + hunk.old_count - chosen_r;
        working.splice(begin..end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut out = String::new();
    for line in working {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}

fn apply_identity(source: &str) -> Result<String, ApplyError> {
    let lines = validate_source(source)?;
    let mut out = String::new();
    for line in lines {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}

fn validate_source(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source
        .strip_suffix('\n')
        .unwrap()
        .split('\n')
        .map(str::to_owned)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}
