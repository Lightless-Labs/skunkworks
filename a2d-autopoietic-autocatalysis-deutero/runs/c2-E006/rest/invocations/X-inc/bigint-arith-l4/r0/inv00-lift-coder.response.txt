```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64
                + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut digit = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if digit < 0 {
                digit += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(digit as u32);
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_small_abs(a: &[u32], m: u32) -> Vec<u32> {
        if m == 0 || a.is_empty() {
            return Vec::new();
        }
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u64;
        for &x in a {
            let v = x as u64 * m as u64 + carry;
            out.push((v % Self::BASE) as u32);
            carry = v / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    fn add_small_abs(a: &mut Vec<u32>, v: u32) {
        let mut carry = v as u64;
        let mut i = 0;
        while carry != 0 {
            if i == a.len() {
                a.push(0);
            }
            let n = a[i] as u64 + carry;
            a[i] = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
            i += 1;
        }
    }

    fn div_small_abs(a: &[u32], d: u32) -> (Vec<u32>, u32) {
        let mut out = vec![0; a.len()];
        let mut rem = 0u64;
        for i in (0..a.len()).rev() {
            let n = rem * Self::BASE + a[i] as u64;
            out[i] = (n / d as u64) as u32;
            rem = n % d as u64;
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        (out, rem as u32)
    }

    fn mul_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        if a.is_empty() || b.is_empty() {
            return Vec::new();
        }
        let mut out = vec![0u32; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.iter().enumerate() {
                let k = i + j;
                let n = out[k] as u64 + x as u64 * y as u64 + carry;
                out[k] = (n % Self::BASE) as u32;
                carry = n / Self::BASE;
            }
            let mut k = i + b.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let n = out[k] as u64 + carry;
                out[k] = (n % Self::BASE) as u32;
                carry = n / Self::BASE;
                k += 1;
            }
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn div_rem_abs(a: &[u32], b: &[u32]) -> (Vec<u32>, Vec<u32>) {
        if a.len() < b.len() {
            return (Vec::new(), a.to_vec());
        }
        if b.len() == 1 {
            let (q, r) = Self::div_small_abs(a, b[0]);
            return (q, if r == 0 { Vec::new() } else { vec![r] });
        }

        let n = b.len();
        let m = a.len() - n;
        let norm = (Self::BASE / (b[n - 1] as u64 + 1)) as u32;
        let v = Self::mul_small_abs(b, norm);
        let mut u = Self::mul_small_abs(a, norm);
        u.resize(a.len() + 1, 0);
        let mut q = vec![0u32; m + 1];

        for j in (0..=m).rev() {
            let numerator = u[j + n] as u64 * Self::BASE + u[j + n - 1] as u64;
            let mut qhat = (numerator / v[n - 1] as u64).min(Self::BASE - 1);
            let mut rhat = numerator - qhat * v[n - 1] as u64;

            if n > 1 {
                while qhat * v[n - 2] as u64
                    > rhat * Self::BASE + u[j + n - 2] as u64
                {
                    qhat -= 1;
                    rhat += v[n - 1] as u64;
                    if rhat >= Self::BASE {
                        break;
                    }
                }
            }

            let mut borrow = 0u64;
            for i in 0..n {
                let product = qhat * v[i] as u64 + borrow;
                let low = product % Self::BASE;
                borrow = product / Self::BASE;
                if u[j + i] as u64 >= low {
                    u[j + i] = (u[j + i] as u64 - low) as u32;
                } else {
                    u[j + i] = (u[j + i] as u64 + Self::BASE - low) as u32;
                    borrow += 1;
                }
            }
            let underflow = u[j + n] as u64 < borrow;
            u[j + n] = if underflow {
                (u[j + n] as u64 + Self::BASE - borrow) as u32
            } else {
                (u[j + n] as u64 - borrow) as u32
            };

            if underflow {
                qhat -= 1;
                let mut carry = 0u64;
                for i in 0..n {
                    let sum = u[j + i] as u64 + v[i] as u64 + carry;
                    u[j + i] = (sum % Self::BASE) as u32;
                    carry = sum / Self::BASE;
                }
                u[j + n] = ((u[j + n] as u64 + carry) % Self::BASE) as u32;
            }
            q[j] = qhat as u32;
        }

        while q.last() == Some(&0) {
            q.pop();
        }
        let rem_norm = u[..n].to_vec();
        let (mut rem, _) = Self::div_small_abs(&rem_norm, norm);
        while rem.last() == Some(&0) {
            rem.pop();
        }
        (q, rem)
    }

    fn abs_value(&self) -> Self {
        Self {
            negative: false,
            limbs: self.limbs.clone(),
        }
    }

    fn from_abs(limbs: Vec<u32>, negative: bool) -> Self {
        let mut result = Self { negative, limbs };
        result.normalize();
        result
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }
        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let digit_value = |b: u8| -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a' + 10) as u32),
                b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
                _ => None,
            }
        };

        for (i, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            match digit_value(byte) {
                Some(d) if d < radix => {}
                _ => return Err(Error::InvalidDigit { index: i }),
            }
        }

        let magnitude = &bytes[start..];
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                return Err(Error::Separator { index: i });
            }
        }

        let mut digits = Vec::new();
        let mut digit_offsets = Vec::new();
        for (i, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' {
                digits.push(digit_value(byte).unwrap());
                digit_offsets.push(i);
            }
        }

        if digits.len() > 1 && digits[0] == 0 {
            return Err(Error::LeadingZero {
                index: digit_offsets[0],
            });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_offsets[max_digits],
            });
        }

        let mut limbs = Vec::new();
        for digit in digits {
            limbs = Self::mul_small_abs(&limbs, radix);
            Self::add_small_abs(&mut limbs, digit);
        }
        Ok(Self::from_abs(limbs, negative && !magnitude.is_empty()))
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut n = self.limbs.clone();
        let mut digits = Vec::new();
        if n.is_empty() {
            digits.push(0);
        } else {
            while !n.is_empty() {
                let (q, r) = Self::div_small_abs(&n, radix);
                digits.push(r);
                n = q;
            }
            digits.reverse();
        }

        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(if *digit < 10 {
                (b'0' + *digit as u8) as char
            } else {
                (b'a' + (*digit - 10) as u8) as char
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_abs(Self::add_abs(&self.limbs, &rhs.limbs), self.negative)
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_abs(
                    Self::sub_abs(&self.limbs, &rhs.limbs),
                    self.negative,
                ),
                Ordering::Less => Self::from_abs(
                    Self::sub_abs(&rhs.limbs, &self.limbs),
                    rhs.negative,
                ),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        Self::from_abs(
            Self::mul_abs(&self.limbs, &rhs.limbs),
            self.negative ^ rhs.negative,
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = Self::div_rem_abs(&self.limbs, &rhs.limbs);
        Ok((
            Self::from_abs(q, self.negative ^ rhs.negative),
            Self::from_abs(r, self.negative),
        ))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }
        let mut result = Self::one();
        let mut base = self.clone();
        let mut e = exponent;
        while e != 0 {
            if e & 1 == 1 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs_value();
        let mut b = rhs.abs_value();
        while !b.is_zero() {
            let (_, r) = a.div_rem(&b).unwrap();
            a = b;
            b = r;
        }
        a
    }
}
```
