#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_symbol(alphabet: Alphabet, value: u8) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn lookup(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn is_foreign_alphabet(alphabet: Alphabet, byte: u8) -> bool {
    match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    }
}

fn encode_symbols(bytes: &[u8], alphabet: Alphabet, padding: Padding) -> Vec<u8> {
    let mut out = Vec::with_capacity((bytes.len() + 2) / 3 * 4);
    let mut i = 0;
    while i + 3 <= bytes.len() {
        let a = bytes[i];
        let b = bytes[i + 1];
        let c = bytes[i + 2];
        out.push(alphabet_symbol(alphabet, a >> 2));
        out.push(alphabet_symbol(alphabet, ((a & 3) << 4) | (b >> 4)));
        out.push(alphabet_symbol(alphabet, ((b & 15) << 2) | (c >> 6)));
        out.push(alphabet_symbol(alphabet, c & 63));
        i += 3;
    }
    match bytes.len() - i {
        1 => {
            let a = bytes[i];
            out.push(alphabet_symbol(alphabet, a >> 2));
            out.push(alphabet_symbol(alphabet, (a & 3) << 4));
            if padding != Padding::Forbidden {
                out.extend_from_slice(b"==");
            }
        }
        2 => {
            let a = bytes[i];
            let b = bytes[i + 1];
            out.push(alphabet_symbol(alphabet, a >> 2));
            out.push(alphabet_symbol(alphabet, ((a & 3) << 4) | (b >> 4)));
            out.push(alphabet_symbol(alphabet, (b & 15) << 2));
            if padding != Padding::Forbidden {
                out.push(b'=');
            }
        }
        _ => {}
    }
    out
}

fn wrap_symbols(symbols: &[u8], whitespace: Whitespace) -> String {
    let mut out = Vec::new();
    if let Whitespace::Wrapped { width, ending } = whitespace {
        for (i, &b) in symbols.iter().enumerate() {
            if i != 0 && i % width == 0 {
                match ending {
                    LineEnding::Lf => out.push(b'\n'),
                    LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
                }
            }
            out.push(b);
        }
    } else {
        out.extend_from_slice(symbols);
    }
    String::from_utf8(out).unwrap()
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let symbols = encode_symbols(input, options.alphabet, options.padding);
    Ok(wrap_symbols(&symbols, options.whitespace))
}

#[derive(Clone)]
struct EncState {
    options: Options,
    tail: Vec<u8>,
    column: usize,
    closed: bool,
}

pub struct Encoder {
    state: EncState,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            state: EncState {
                options,
                tail: Vec::new(),
                column: 0,
                closed: false,
            },
        })
    }

    fn emit_symbols(&mut self, symbols: &[u8]) -> String {
        let mut out = Vec::new();
        if let Whitespace::Wrapped { width, ending } = self.state.options.whitespace {
            for &b in symbols {
                if self.state.column == width {
                    match ending {
                        LineEnding::Lf => out.push(b'\n'),
                        LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
                    }
                    self.state.column = 0;
                }
                out.push(b);
                self.state.column += 1;
            }
        } else {
            out.extend_from_slice(symbols);
        }
        String::from_utf8(out).unwrap()
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }
        if input.is_empty() {
            return Ok(String::new());
        }
        self.state.tail.extend_from_slice(input);
        let complete = self.state.tail.len() / 3 * 3;
        let chunk: Vec<u8> = self.state.tail.drain(..complete).collect();
        let symbols = encode_symbols(&chunk, self.state.options.alphabet, Padding::Forbidden);
        Ok(self.emit_symbols(&symbols))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }
        self.state.closed = true;
        let tail = std::mem::take(&mut self.state.tail);
        let symbols = encode_symbols(&tail, self.state.options.alphabet, self.state.options.padding);
        Ok(self.emit_symbols(&symbols))
    }
}

#[derive(Clone, Copy)]
struct Sym {
    value: u8,
    byte: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum WrapState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { start: usize },
}

struct DecState {
    options: Options,
    offset: usize,
    symbols_seen: usize,
    group: Vec<Sym>,
    pending_second_pad: Option<usize>,
    terminal_pad: bool,
    column: usize,
    wrap_state: WrapState,
    closed: bool,
    sticky: Option<Error>,
}

pub struct Decoder {
    state: DecState,
}

fn ignored_ascii(byte: u8) -> bool {
    matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            state: DecState {
                options,
                offset: 0,
                symbols_seen: 0,
                group: Vec::with_capacity(4),
                pending_second_pad: None,
                terminal_pad: false,
                column: 0,
                wrap_state: WrapState::Normal,
                closed: false,
                sticky: None,
            },
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.state.sticky = Some(error.clone());
        error
    }

    fn output_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        let g = &self.state.group;
        if g.len() == 4 && g[2].byte == b'=' {
            let a = g[0].value;
            let b = g[1].value;
            if self.state.options.strict && (b & 15) != 0 {
                return Err(Error::NonCanonical { offset: g[1].offset });
            }
            out.push((a << 2) | (b >> 4));
            self.state.terminal_pad = true;
        } else if g.len() == 4 && g[3].byte == b'=' {
            let a = g[0].value;
            let b = g[1].value;
            let c = g[2].value;
            if self.state.options.strict && (c & 3) != 0 {
                return Err(Error::NonCanonical { offset: g[2].offset });
            }
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            self.state.terminal_pad = true;
        } else if g.len() == 4 {
            let a = g[0].value;
            let b = g[1].value;
            let c = g[2].value;
            let d = g[3].value;
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);
        }
        self.state.group.clear();
        self.state.pending_second_pad = None;
        Ok(())
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.state.options.max_symbols {
            if self.state.symbols_seen >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.state.symbols_seen += 1;

        if self.state.terminal_pad {
            return Err(Error::BadPadding { offset });
        }
        if let Some(pad_offset) = self.state.pending_second_pad {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            if self.state.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            self.state.group.push(Sym { value: 0, byte, offset });
            self.output_group(out)?;
            let _ = pad_offset;
            return Ok(());
        }

        let pos = self.state.group.len();
        if byte == b'=' {
            if self.state.options.padding == Padding::Forbidden || pos < 2 || pos > 3 {
                return Err(Error::BadPadding { offset });
            }
            if pos == 2 {
                self.state.pending_second_pad = Some(offset);
                self.state.group.push(Sym { value: 0, byte, offset });
                return Ok(());
            }
            self.state.group.push(Sym { value: 0, byte, offset });
            self.output_group(out)?;
            return Ok(());
        }

        if self.state.group.len() == 3 && self.state.group[2].byte == b'=' {
            return Err(Error::BadPadding { offset });
        }

        let value = match lookup(self.state.options.alphabet, byte) {
            Some(v) => v,
            None if self.state.options.strict && is_foreign_alphabet(self.state.options.alphabet, byte) => {
                return Err(Error::MixedAlphabet { offset });
            }
            None => return Err(Error::InvalidByte { offset, byte }),
        };
        self.state.group.push(Sym { value, byte, offset });
        if self.state.group.len() == 4 {
            self.output_group(out)?;
        }
        Ok(())
    }

    fn frame_byte(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        match self.state.options.whitespace {
            Whitespace::None => Ok(false),
            Whitespace::IgnoreAscii => Ok(ignored_ascii(byte)),
            Whitespace::Wrapped { width, ending } => {
                match self.state.wrap_state {
                    WrapState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.state.wrap_state = WrapState::NeedSymbol { start: cr_offset };
                        return Ok(true);
                    }
                    WrapState::NeedSymbol { start } => {
                        self.state.wrap_state = WrapState::Normal;
                        if byte == b'\r' || byte == b'\n' || ignored_ascii(byte) {
                            return Err(Error::BadWrap { offset });
                        }
                        let _ = start;
                    }
                    WrapState::Normal => {}
                }

                if self.state.column == width {
                    match ending {
                        LineEnding::Lf => {
                            if byte != b'\n' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.state.column = 0;
                            self.state.wrap_state = WrapState::NeedSymbol { start: offset };
                            return Ok(true);
                        }
                        LineEnding::CrLf => {
                            if byte != b'\r' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.state.wrap_state = WrapState::NeedLf { cr_offset: offset };
                            return Ok(true);
                        }
                    }
                }
                if byte == b'\r' || byte == b'\n' || ignored_ascii(byte) {
                    return Err(Error::BadWrap { offset });
                }
                self.state.column += 1;
                Ok(false)
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = Vec::new();
        for &byte in input {
            let offset = self.state.offset;
            self.state.offset += 1;
            let skipped = match self.frame_byte(byte, offset) {
                Ok(v) => v,
                Err(e) => return Err(self.fail(e)),
            };
            if skipped {
                continue;
            }
            if let Err(e) = self.process_symbol(byte, offset, &mut out) {
                return Err(self.fail(e));
            }
        }
        Ok(out)
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        let mut out = Vec::new();
        match self.state.wrap_state {
            WrapState::NeedLf { cr_offset } => return Err(Error::BadWrap { offset: cr_offset }),
            WrapState::NeedSymbol { start } => return Err(Error::BadWrap { offset: start }),
            WrapState::Normal => {}
        }

        if let Some(offset) = self.state.pending_second_pad {
            return Err(Error::BadPadding { offset });
        }

        if self.state.group.is_empty() {
            return Ok(out);
        }

        let len = self.state.group.len();
        if len == 4 {
            return Ok(out);
        }

        if self.state.options.padding == Padding::Required {
            return Err(Error::Truncated {
                offset: self.state.group[0].offset,
            });
        }

        if len == 1 {
            return Err(Error::Truncated {
                offset: self.state.group[0].offset,
            });
        }

        if len == 2 {
            let a = self.state.group[0].value;
            let b = self.state.group[1].value;
            if self.state.options.strict && (b & 15) != 0 {
                return Err(Error::NonCanonical {
                    offset: self.state.group[1].offset,
                });
            }
            out.push((a << 2) | (b >> 4));
        } else if len == 3 {
            let a = self.state.group[0].value;
            let b = self.state.group[1].value;
            let c = self.state.group[2].value;
            if self.state.options.strict && (c & 3) != 0 {
                return Err(Error::NonCanonical {
                    offset: self.state.group[2].offset,
                });
            }
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }
        match self.finish_inner() {
            Ok(out) => {
                self.state.closed = true;
                Ok(out)
            }
            Err(error) => Err(self.fail(error)),
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
