#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

struct RuntimeJob {
    burst_index: usize,
    remaining: u64,
    level: usize,
    completion: Option<u64>,
    total_cpu: u64,
    total_io: u64,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        return 0;
    }
    let divisor = count as u64;
    (total * 100 + divisor / 2) / divisor
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut runtime: Vec<RuntimeJob> = jobs
        .iter()
        .map(|job| RuntimeJob {
            burst_index: 0,
            remaining: job.bursts[0] as u64,
            level: job.level,
            completion: None,
            total_cpu: job
                .bursts
                .iter()
                .step_by(2)
                .map(|&value| value as u64)
                .sum(),
            total_io: job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&value| value as u64)
                .sum(),
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut completed_count = 0usize;

    fn admit_through(
        events: &mut Vec<Event>,
        through: u64,
        jobs: &[Job],
        queues: &mut [std::collections::VecDeque<usize>],
    ) {
        events.sort_by_key(|event| {
            let kind_order = match event.kind {
                EventKind::Arrival => 0,
                EventKind::Wake { .. } => 1,
            };
            (event.time, kind_order, event.index)
        });
        let count = events.partition_point(|event| event.time <= through);
        let admitted: Vec<Event> = events.drain(..count).collect();
        for event in admitted {
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake { level } => queues[level].push_back(event.index),
            }
        }
    }

    while completed_count < jobs.len() {
        admit_through(&mut events, now, jobs, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let level = if let Some(level) = selected_level {
            level
        } else {
            let next_time = events.iter().map(|event| event.time).min().unwrap();
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                last_dispatched_id = None;
                now = next_time;
            }
            continue;
        };

        let job_index = queues[level].pop_front().unwrap();
        let job_id = jobs[job_index].id;

        if last_dispatched_id.is_some_and(|last_id| last_id != job_id) {
            now += switch_cost as u64;
        }
        admit_through(&mut events, now, jobs, &mut queues);

        let quantum = quanta[level] as u64;
        let consumed = runtime[job_index].remaining.min(quantum);
        let start = now;
        now += consumed;
        let end = now;

        runtime[job_index].remaining -= consumed;
        let burst_finished = runtime[job_index].remaining == 0;
        let job_finished = burst_finished
            && runtime[job_index].burst_index + 1 == jobs[job_index].bursts.len();

        runs.push(Run {
            id: job_id,
            start,
            end,
            finished: job_finished,
            level,
        });

        admit_through(&mut events, end, jobs, &mut queues);

        if job_finished {
            runtime[job_index].completion = Some(end);
            completed_count += 1;
        } else if burst_finished {
            let next_io_index = runtime[job_index].burst_index + 1;
            let io_delay = jobs[job_index].bursts[next_io_index] as u64;
            let next_cpu_index = next_io_index + 1;
            runtime[job_index].burst_index = next_cpu_index;
            runtime[job_index].remaining = jobs[job_index].bursts[next_cpu_index] as u64;
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: end + io_delay,
                kind: EventKind::Wake { level: wake_level },
                index: job_index,
            });
            admit_through(&mut events, end, jobs, &mut queues);
        } else {
            let demoted_level = (level + 1).min(quanta.len() - 1);
            runtime[job_index].level = demoted_level;
            queues[demoted_level].push_back(job_index);
        }

        last_dispatched_id = Some(job_id);
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let completion = runtime[index].completion.unwrap();
            let turnaround = completion - job.arrival as u64;
            let waiting = turnaround - runtime[index].total_cpu - runtime[index].total_io;
            Stat {
                id: job.id,
                completion,
                turnaround,
                waiting,
            }
        })
        .collect();

    let waiting_total: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let turnaround_total: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(waiting_total, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(turnaround_total, jobs.len()),
        end: now,
    })
}
