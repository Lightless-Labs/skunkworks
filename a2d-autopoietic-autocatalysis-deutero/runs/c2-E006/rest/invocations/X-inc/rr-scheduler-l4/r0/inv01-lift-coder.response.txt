```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

#[derive(Clone, Debug)]
struct Process {
    burst_index: usize,
    remaining: u64,
    level: usize,
    completion: Option<u64>,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if !(1..=31).contains(&job.bursts.len()) || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut event_cursor = 0usize;
    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            burst_index: 0,
            remaining: u64::from(job.bursts[0]),
            level: job.level,
            completion: None,
        })
        .collect();
    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed = 0usize;

    fn admit_through(
        time: u64,
        events: &[Event],
        event_cursor: &mut usize,
        jobs: &[Job],
        processes: &mut [Process],
        queues: &mut [VecDeque<usize>],
    ) {
        while *event_cursor < events.len() && events[*event_cursor].time <= time {
            let event = events[*event_cursor];
            let level = if event.kind == 0 {
                jobs[event.index].level
            } else {
                processes[event.index].level
            };
            queues[level].push_back(event.index);
            *event_cursor += 1;
        }
    }

    while completed < jobs.len() {
        admit_through(
            time,
            &events,
            &mut event_cursor,
            jobs,
            &mut processes,
            &mut queues,
        );

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let job_id = jobs[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != job_id) {
            time += u64::from(switch_cost);
        }
        admit_through(
            time,
            &events,
            &mut event_cursor,
            jobs,
            &mut processes,
            &mut queues,
        );

        let quantum = u64::from(quanta[level]);
        let consumed = processes[index].remaining.min(quantum);
        let start = time;
        time += consumed;
        let finished = processes[index].burst_index + 1 == jobs[index].bursts.len()
            && processes[index].remaining == consumed;
        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished,
            level,
        });

        processes[index].remaining -= consumed;
        admit_through(
            time,
            &events,
            &mut event_cursor,
            jobs,
            &mut processes,
            &mut queues,
        );

        if processes[index].remaining > 0 {
            processes[index].level = (level + 1).min(quanta.len() - 1);
            queues[processes[index].level].push_back(index);
        } else if finished {
            processes[index].completion = Some(time);
            completed += 1;
        } else {
            let io_position = processes[index].burst_index + 1;
            let io_delay = u64::from(jobs[index].bursts[io_position]);
            processes[index].burst_index += 2;
            processes[index].remaining =
                u64::from(jobs[index].bursts[processes[index].burst_index]);
            processes[index].level = level.saturating_sub(1);
            events.push(Event {
                time: time + io_delay,
                kind: 1,
                index,
            });
            events[event_cursor..].sort_by_key(|event| (event.time, event.kind, event.index));
        }

        last_dispatch_id = Some(job_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;
    let mut end = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = processes[index].completion.unwrap_or(time);
        let turnaround = completion - u64::from(job.arrival);
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        end = end.max(completion);
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}

use std::collections::VecDeque;
```
