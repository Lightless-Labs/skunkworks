fn validate_dependencies(tasks: &[Task]) -> Result<(), Error> {
    let count = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    Ok(())
}

pub fn schedule(tasks: &[Task], cooldowns: &[u64], capacity: &[u32]) -> Result<Schedule, Error> {

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }

        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    let count = tasks.len();
    let mut children = vec![Vec::<usize>::new(); count];
    let mut indegree = vec![0usize; count];

    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    let mut available = Vec::<usize>::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(task_id);
        }
    }

    let mut topo = Vec::with_capacity(count);
    while !available.is_empty() {
        let smallest_index = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, task_id)| *task_id)
            .map(|(index, _)| index)
            .expect("available is nonempty");
        let task_id = available.swap_remove(smallest_index);
        topo.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if topo.len() != count {
        let mut removed = vec![false; count];
        for &task_id in &topo {
            removed[task_id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..count).filter(|&task_id| !removed[task_id]).collect(),
        });

    for &task_id in topo.iter().rev() {
        let mut longest_successor = 0u64;

            longest_successor = longest_successor.max(tails[child]);

                stage: Stage::Tail,

    let mut used = vec![0u64; capacity.len()];
    let mut trace = String::new();

    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));

    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < count {
        running.sort_by_key(|item| item.task);
        let mut still_running = Vec::with_capacity(running.len());

        for item in running.drain(..) {
            if item.finish == time {
                completed[item.task] = true;
                completed_count += 1;

                    item.worker,

                still_running.push(item);
            }

                    continue;

                    .iter()

                    task: task_id,
                    stage: Stage::Finish,

                    stage: Stage::Turnaround,

                finish,
                worker,

                task_id,
                worker,
                resource_snapshot(&used)

                completed_count += 1;
                makespan = makespan.max(finish);

                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    time,
                    task_id,
                    worker,
                    resource_snapshot(&used)
                ));
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == count {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for task_id in 0..count {
            if !started[task_id] {
                consider(tasks[task_id].release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        match next_time {
            Some(next) => time = next,
            None => {
                // With all graph edges validated and the graph acyclic, an unfinished task
                // always has a future release, running finish, or worker availability event.
                return Err(Error::Cycle {
                    blocked: (0..count).filter(|&task_id| !completed[task_id]).collect(),
                });
            }
        }
    }

    trace.push_str(&format!("makespan={}\n", makespan));

    Ok(Schedule {
        runs: runs
            .into_iter()
            .map(|run| run.expect("every task starts exactly once"))
            .collect(),
        makespan,
        trace,
    })
}

    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Validate resource shapes and demands in task order.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest available ID.
    let mut available = BTreeSet::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(task_id) = available.pop_first() {
        order.push(task_id);
        for &child in &children[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &task_id in &order {
                flags[task_id] = true;
            }
            flags
        };
        let blocked = (0..n).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut worker_available_at = vec![0u64; worker_count];
    let mut worker_busy = vec![false; worker_count];
    let mut running = Vec::<(usize, usize, u64)>::new(); // (task, worker, finish)
    let mut used = vec![0u128; capacity.len()];
    let mut trace = format!("workers={};resources={}\n", worker_count, capacity.len());
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete every task finishing now, in ascending task ID.
        let mut ending: Vec<(usize, usize, u64)> = running
            .iter()
            .copied()
            .filter(|&(_, _, finish)| finish == now)
            .collect();
        ending.sort_by_key(|&(task_id, _, _)| task_id);

        for &(task_id, worker, finish) in &ending {
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] -= need as u128;
            }

            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                now,
                task_id,
                worker,
                resource_string(&used)
            ));

            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            worker_busy[worker] = false;

            for &child in &children[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        if !ending.is_empty() {
            running.retain(|&(_, _, finish)| finish != now);
        }

        // Start tasks one at a time, reevaluating eligibility and fit after each.
        loop {
            let mut selected: Option<usize> = None;

            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > now
                    || remaining_predecessors[task_id] != 0
                {
                    continue;
                }

                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(resource, &need)| {
                        (need as u128) <= (capacity[resource] as u128) - used[resource]
                    });
                if !fits {
                    continue;
                }

                let better = match selected {
                    None => true,
                    Some(current) => {
                        tails[task_id] > tails[current]
                            || (tails[task_id] == tails[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    selected = Some(task_id);
                }
            }

            let Some(task_id) = selected else {
                break;
            };

            // Choose the first available worker in the task's cyclic preference order.
            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| !worker_busy[worker] && worker_available_at[worker] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_busy[worker] = true;
            worker_available_at[worker] = available_at;

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += need as u128;
            }

            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                resource_string(&used)
            ));

            if finish == now {
                // Zero-duration tasks complete immediately before another admission.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= need as u128;
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    resource_string(&used)
                ));

                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                worker_busy[worker] = false;

                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push((task_id, worker, finish));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the next event that can change eligibility or availability.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |current| current.min(time)));
            }
        };

        for &(_, _, finish) in &running {
            consider(finish);
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                consider(tasks[task_id].release);
            }
        }
        for worker in 0..worker_count {
            if !worker_busy[worker] && worker_available_at[worker] > now {
                consider(worker_available_at[worker]);
            }
        }

        let Some(next) = next_time else {
            // A validated acyclic graph always has a future progress event unless
            // all work is complete; retain a defensive non-panicking exit.
            return Err(Error::Cycle {
                blocked: (0..n).filter(|&task_id| !completed[task_id]).collect(),
            });
        };
        now = next;
    }

    trace.push_str(&format!("makespan={}\n", makespan));

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task is started exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shapes and demands task-by-task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &cap)) in task.needs.iter().zip(capacity).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: cap,
                });
            }
        }
    }

    // Kahn traversal with a min-heap represented by a sorted available list.
    let mut indegree: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut available: Vec<usize> = (0..n).filter(|&i| indegree[i] == 0).collect();
    let mut kahn_order = Vec::with_capacity(n);
    while !available.is_empty() {
        let id = available.remove(0);
        kahn_order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let pos = available.binary_search(&child).unwrap_or_else(|p| p);
                available.insert(pos, child);
            }
        }
    }

    if kahn_order.len() != n {
        let removed: Vec<bool> = {
            let mut v = vec![false; n];
            for &id in &kahn_order {
                v[id] = true;
            }
            v
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&i| !removed[i]).collect(),
        });
    }

    // Reverse-Kahn bottom levels.
    let mut tail = vec![0u64; n];
    for &id in kahn_order.iter().rev() {
        let max_child = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut used = vec![0u64; capacity.len()];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut worker_free = vec![0u64; worker_count];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // finish, task, worker
    let mut done_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while done_count < n {
        // Process all completions at this instant in ascending task ID.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|&(finish, _, _)| finish == now)
            .collect();
        finishing.sort_by_key(|&(_, task, _)| task);

        for &(_, task_id, worker) in &finishing {
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] -= u64::from(need);
            }
            completed[task_id] = true;
            done_count += 1;
            makespan = makespan.max(now);

            trace.push_str(&format!("{now}|end|{task_id}|{worker}|"));
            append_used(&mut trace, &used);
            trace.push('\n');

            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }
        }
        if !finishing.is_empty() {
            running.retain(|&(finish, _, _)| finish != now);
        }

        // Admit one task at a time, re-evaluating eligibility after every start.
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    if started[id] || tasks[id].release > now || remaining_deps[id] != 0 {
                        return false;
                    }
                    tasks[id]
                        .needs
                        .iter()
                        .enumerate()
                        .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r])
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tail[b]
                    .cmp(&tail[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&w| worker_free[w] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let free_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_free[worker] = free_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }

            trace.push_str(&format!("{now}|start|{task_id}|{worker}|"));
            append_used(&mut trace, &used);
            trace.push('\n');

            if finish == now {
                // Complete a zero-duration task immediately before selecting again.
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                completed[task_id] = true;
                done_count += 1;
                makespan = makespan.max(now);

                trace.push_str(&format!("{now}|end|{task_id}|{worker}|"));
                append_used(&mut trace, &used);
                trace.push('\n');

                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if done_count == n {
            break;
        }

        // Advance to the next event that can change eligibility or worker availability.
        let mut next = u64::MAX;
        for &(finish, _, _) in &running {
            if finish > now {
                next = next.min(finish);
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = next.min(tasks[id].release);
            }
        }
        for &free_at in &worker_free {
            if free_at > now {
                next = next.min(free_at);
            }
        }

        // A dependency may be waiting on a running task; its finish is already an event.
        if next == u64::MAX {
            // With all tasks started or completed, only pending completions could remain.
            // Otherwise this is unreachable for an acyclic graph.
            if let Some(min_finish) = running.iter().map(|&(f, _, _)| f).min() {
                next = min_finish;
            } else {
                break;
            }
        }
        now = next;
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("acyclic tasks are all scheduled"))
        .collect();

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in input order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Validate resource shapes and demands task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &limit)) in task.needs.iter().zip(capacity).enumerate() {
            if need > limit {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: limit,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal, always selecting the smallest available ID.
    let mut available = std::collections::BTreeSet::<usize>::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }
    let mut kahn_order = Vec::with_capacity(n);
    while let Some(task_id) = available.pop_first() {
        kahn_order.push(task_id);
        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }
    if kahn_order.len() != n {
        let removed = {
            let mut marked = vec![false; n];
            for &id in &kahn_order {
                marked[id] = true;
            }
            marked
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Bottom levels, computed in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &task_id in kahn_order.iter().rev() {
        let max_child = successors[task_id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut trace = format!("workers={worker_count};resources={resource_count}\n");
    let mut used = vec![0u128; resource_count];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete every task finishing now, in task ID order.
        let mut ending: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|&(finish, _, _)| finish == now)
            .collect();
        ending.sort_by_key(|&(_, task_id, _)| task_id);

        for &(finish, task_id, worker) in &ending {
            for resource in 0..resource_count {
                used[resource] -= u128::from(tasks[task_id].needs[resource]);
            }
            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                finish,
                task_id,
                worker,
                resource_snapshot(&used)
            ));
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            for &child in &successors[task_id] {
                remaining_deps[child] -= 1;
            }
        }
        if !ending.is_empty() {
            running.retain(|&(finish, _, _)| finish != now);
        }

        // Start tasks one at a time, reselecting after each admission.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > now
                    || remaining_deps[task_id] != 0
                {
                    continue;
                }
                let fits = (0..resource_count).all(|resource| {
                    u128::from(tasks[task_id].needs[resource])
                        <= u128::from(capacity[resource]) - used[resource]
                });
                if !fits {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tail[task_id] > tail[current]
                            || (tail[task_id] == tail[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for resource in 0..resource_count {
                used[resource] += u128::from(tasks[task_id].needs[resource]);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                resource_snapshot(&used)
            ));

            if finish == now {
                // Immediate completion of a zero-duration task.
                for resource in 0..resource_count {
                    used[resource] -= u128::from(tasks[task_id].needs[resource]);
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    resource_snapshot(&used)
                ));
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for &child in &successors[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later relevant event.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                let release = tasks[task_id].release;
                next_time = Some(next_time.map_or(release, |current| current.min(release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time =
                    Some(next_time.map_or(available_at, |current| current.min(available_at)));
            }
        }

        // If no event was found, unfinished work would be impossible in an acyclic,
        // validated graph; this guard avoids panicking on any unforeseen state.
        let Some(next) = next_time else {
            break;
        };
        now = next;
    }

    if completed_count != n {
        // A defensive fallback; validated acyclic inputs should always complete.
        let blocked = (0..n).filter(|&id| !completed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs: runs
            .into_iter()
            .map(|run| run.expect("every task completes after successful simulation"))
            .collect(),
        makespan,
        trace,
    })
}
