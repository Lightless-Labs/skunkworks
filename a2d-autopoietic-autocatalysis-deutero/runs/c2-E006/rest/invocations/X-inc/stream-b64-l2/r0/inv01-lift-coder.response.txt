#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len != 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                } else {
                    let a = self.tail[0];
                    let b = self.tail[1];
                    let c = input[index];
                    output.push(ALPHABET[(a >> 2) as usize] as char);
                    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
                    output.push(ALPHABET[(c & 0x3f) as usize] as char);
                    self.tail_len = 0;
                    index += 1;
                    break;
                }
                self.tail_len += 1;
                index += 1;
            }
        }

        while index + 3 <= input.len() {
            let a = input[index];
            let b = input[index + 1];
            let c = input[index + 2];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
            output.push(ALPHABET[(c & 0x3f) as usize] as char);
            index += 3;
        }

        let remaining = input.len() - index;
        for i in 0..remaining {
            self.tail[i] = input[index + i];
        }
        self.tail_len = remaining;

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    offset: usize,
    padded: bool,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            offset: 0,
            padded: false,
            finished: false,
            sticky_error: None,
        }
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) {
        let a = digit(self.group[0]).unwrap();
        let b = digit(self.group[1]).unwrap();
        output.push((a << 2) | (b >> 4));

        if self.group[2] != b'=' {
            let c = digit(self.group[2]).unwrap();
            output.push((b << 4) | (c >> 2));
            if self.group[3] != b'=' {
                let d = digit(self.group[3]).unwrap();
                output.push((c << 6) | d);
            }
        }

        if self.group[2] == b'=' || self.group[3] == b'=' {
            self.padded = true;
        }
        self.len = 0;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            let error = if self.padded {
                Some(Error::BadPadding {
                    offset: current_offset,
                })
            } else if self.len == 3 && self.group[2] == b'=' && byte != b'=' {
                Some(Error::BadPadding {
                    offset: current_offset,
                })
            } else if (self.len == 0 || self.len == 1) && byte == b'=' {
                Some(Error::BadPadding {
                    offset: current_offset,
                })
            } else if byte == b'=' {
                if self.len == 2 || self.len == 3 {
                    None
                } else {
                    Some(Error::BadPadding {
                        offset: current_offset,
                    })
                }
            } else if digit(byte).is_none() {
                Some(Error::InvalidByte {
                    offset: current_offset,
                    byte,
                })
            } else {
                None
            };

            if let Some(error) = error {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }

            self.group[self.len] = byte;
            self.offsets[self.len] = current_offset;
            self.len += 1;

            if self.len == 4 {
                if self.group[2] == b'=' && self.group[3] != b'=' {
                    let error = Error::BadPadding {
                        offset: self.offsets[3],
                    };
                    self.sticky_error = Some(error.clone());
                    return Err(error);
                }
                self.emit_group(&mut output);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let result = if self.len == 0 {
            Ok(Vec::new())
        } else if self.len == 3 && self.group[2] == b'=' {
            Err(Error::BadPadding {
                offset: self.offsets[2],
            })
        } else {
            Err(Error::Truncated {
                offset: self.offsets[0],
            })
        };

        match result {
            Ok(output) => {
                self.finished = true;
                Ok(output)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
