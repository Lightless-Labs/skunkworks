#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|n| n as u8)
}

fn encode_group(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        out.push_str(std::str::from_utf8(&encode_group(chunk[0], chunk[1], chunk[2])).unwrap());
    }
    match chunks.remainder() {
        [] => {}
        [a] => {
            let pair = encode_group(*a, 0, 0);
            out.push(pair[0] as char);
            out.push(pair[1] as char);
            out.push_str("==");
        }
        [a, b] => {
            let triple = encode_group(*a, *b, 0);
            out.push(triple[0] as char);
            out.push(triple[1] as char);
            out.push(triple[2] as char);
            out.push('=');
        }
        _ => unreachable!(),
    }
    out
}

#[derive(Clone)]
struct Group {
    values: [u8; 4],
    offsets: [usize; 4],
    len: usize,
}

impl Group {
    fn new() -> Self {
        Self {
            values: [0; 4],
            offsets: [0; 4],
            len: 0,
        }
    }

    fn first_offset(&self) -> usize {
        self.offsets[0]
    }
}

#[derive(Clone)]
struct DecoderState {
    group: Group,
    terminal: bool,
    pending_second_padding: Option<usize>,
    offset: usize,
}

impl DecoderState {
    fn new() -> Self {
        Self {
            group: Group::new(),
            terminal: false,
            pending_second_padding: None,
            offset: 0,
        }
    }

    fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        let mut output = Vec::new();

        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            if self.terminal {
                return Err(Error::BadPadding { offset });
            }

            if self.pending_second_padding.is_some() && byte != b'=' {
                return Err(Error::BadPadding { offset });
            }

            let pos = self.group.len;
            if byte == b'=' {
                match pos {
                    0 | 1 => return Err(Error::BadPadding { offset }),
                    2 => {
                        self.pending_second_padding = Some(offset);
                        self.group.values[pos] = 0;
                        self.group.offsets[pos] = offset;
                        self.group.len += 1;
                    }
                    3 => {
                        self.group.offsets[pos] = offset;
                        self.group.len += 1;
                        let first_two_padded = self.group.values[2] == 0
                            && self.pending_second_padding.is_some();
                        if first_two_padded {
                            output.push((self.group.values[0] << 2)
                                | (self.group.values[1] >> 4));
                        } else {
                            output.push((self.group.values[0] << 2)
                                | (self.group.values[1] >> 4));
                            output.push((self.group.values[1] << 4)
                                | (self.group.values[2] >> 2));
                        }
                        self.group = Group::new();
                        self.pending_second_padding = None;
                        self.terminal = true;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let Some(value) = digit(byte) else {
                return Err(Error::InvalidByte { offset, byte });
            };

            self.group.values[pos] = value;
            self.group.offsets[pos] = offset;
            self.group.len += 1;

            if self.group.len == 4 {
                output.push((self.group.values[0] << 2) | (self.group.values[1] >> 4));
                output.push((self.group.values[1] << 4) | (self.group.values[2] >> 2));
                output.push((self.group.values[2] << 6) | self.group.values[3]);
                self.group = Group::new();
            }
        }

        Ok(output)
    }

    fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(offset) = self.pending_second_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group.len != 0 {
            return Err(Error::Truncated {
                offset: self.group.first_offset(),
            });
        }

        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut state = DecoderState::new();
    state.push(input)?;
    state.finish()
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;

                if self.tail_len == 3 {
                    let group = encode_group(self.tail[0], self.tail[1], input[index - 1]);
                    out.push_str(std::str::from_utf8(&group).unwrap());
                    self.tail_len = 0;
                }
            }
        }

        while index + 3 <= input.len() {
            let group = encode_group(input[index], input[index + 1], input[index + 2]);
            out.push_str(std::str::from_utf8(&group).unwrap());
            index += 3;
        }

        let remaining = &input[index..];
        self.tail_len = remaining.len();
        for (i, &byte) in remaining.iter().enumerate() {
            self.tail[i] = byte;
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut out = String::new();
        match self.tail_len {
            0 => {}
            1 => {
                let group = encode_group(self.tail[0], 0, 0);
                out.push(group[0] as char);
                out.push(group[1] as char);
                out.push_str("==");
            }
            2 => {
                let group = encode_group(self.tail[0], self.tail[1], 0);
                out.push(group[0] as char);
                out.push(group[1] as char);
                out.push(group[2] as char);
                out.push('=');
            }
            _ => unreachable!(),
        }
        self.tail_len = 0;
        Ok(out)
    }
}

pub struct Decoder {
    state: DecoderState,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            state: DecoderState::new(),
            finished: false,
            error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.state.push(input) {
            Ok(output) => Ok(output),
            Err(error) => {
                self.error = Some(error.clone());
                Err(error)
            }
        }
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.state.finish() {
            Ok(output) => {
                self.finished = true;
                Ok(output)
            }
            Err(error) => {
                self.error = Some(error.clone());
                Err(error)
            }
        }
    }
}
