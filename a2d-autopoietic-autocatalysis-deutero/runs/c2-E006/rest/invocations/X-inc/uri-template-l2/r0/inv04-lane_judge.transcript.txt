    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

struct Variable {
    name: String,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    variables: Vec<Variable>,
}

#[derive(Debug, Clone)]

    Literal(String),
    Expression(Expression),
}

fn is_name_continue(byte: u8) -> bool {

fn parse_expression(content: &str, content_start: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let operator = if bytes
        .first()
        .is_some_and(|b| b"+#./;?&".contains(b))
    {
        Some(bytes[0] as char)
    } else {
        None
    };

    if let Some(op @ ('+' | '#')) = operator {
        let _ = op;
    } else if operator.is_some() {
        return Err(Error::UnsupportedOperator {
            pos: content_start,
        });
    }

    let offset = usize::from(operator.is_some());
    if offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut start = offset;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| start + relative)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: content_start + start,
            });
        }

        if !is_name_start(bytes[start]) {
            return Err(Error::InvalidName {
                pos: content_start + start,
            });
        }

        for i in start + 1..end {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName {
                    pos: content_start + i,
                });
            }
        }

        let name = content[start..end].to_string();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable {
                pos: content_start + start,
            });
        }
        seen.push(name.clone());
        variables.push(Variable { name });

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn parse_template(template: &str) -> Result<Vec<Part>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut cursor = 0;
    let mut literal_start = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                if literal_start < cursor {
                    parts.push(Part::Literal(template[literal_start..cursor].to_string()));
                }

                let open = cursor;
                let Some(relative_close) = bytes[cursor + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = cursor + 1 + relative_close;

                if let Some(relative_open) = bytes[cursor + 1..close]
                    .iter()
                    .position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + relative_open,
                    });
                }

                let expression =
                    parse_expression(&template[cursor + 1..close], cursor + 1, close)?;
                parts.push(Part::Expression(expression));
                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    if literal_start < bytes.len() {
        parts.push(Part::Literal(template[literal_start..].to_string()));
    }
    Ok(parts)
}

        .map(|(_, value)| *value)

    )

        b'a'..=b'f' => Some(byte - b'a' + 10),

fn encode_value(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 < bytes.len() && hex_value(bytes[i + 1]).is_some() && hex_value(bytes[i + 2]).is_some() {
                output.push('%');
                output.push(bytes[i + 1].to_ascii_uppercase() as char);
                output.push(bytes[i + 2].to_ascii_uppercase() as char);
                i += 3;
                continue;
            }
            output.push_str("%25");
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let allow_reserved = matches!(expression.operator, Some('+' | '#'));
    let mut fields = Vec::new();

    for variable in &expression.variables {
        if let Some(value) = lookup(&variable.name, vars) {
            fields.push(encode_value(value, allow_reserved));
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let joined = fields.join(",");
    if expression.operator == Some('#') {
        format!("#{joined}")
    } else {
        joined
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = parse_template(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(&text),
            Part::Expression(expression) => output.push_str(&expand_expression(&expression, vars)),
        }
    }

    Ok(output)
}

    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression {
    operator: Option<char>,
    variables: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);

        let reserved = matches!(expression.operator, Some('+') | Some('#'));
        let mut fields = Vec::new();

        for name in expression.variables {
            if let Some(value) = vars.iter().rev().find_map(|(key, value)| {
                if *key == name {
                    Some(*value)
                } else {
                    None
                }
            }) {
                fields.push(encode_value(value, reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        cursor = end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let close = i + 1 + relative_close;

                if let Some(relative_open) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_open,
                    });
                }

                let expression = parse_expression(template, start, close)?;
                expressions.push((start, close, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let byte = bytes[content_start];
        content_start += 1;
        match byte {
            b'+' => Some('+'),
            b'#' => Some('#'),
            _ => return Err(Error::UnsupportedOperator { pos: open + 1 }),
        }
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map_or(close, |offset| spec_start + offset);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|prior| prior == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        variables.push(name);

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, variables })
}

        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
