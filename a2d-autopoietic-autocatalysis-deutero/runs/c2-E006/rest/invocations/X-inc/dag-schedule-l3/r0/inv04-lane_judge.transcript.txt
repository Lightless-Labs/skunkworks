                    task: task_id,
                    stage: Stage::Finish,

                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,

                running.push((finish, task_id, worker_id));
            }

            break;
        }

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }

                let release = tasks[task_id].release;
                next_time = Some(next_time.map_or(release, |t| t.min(release)));

        let Some(next) = next_time else {
            break;
        };
        now = next;

        .map(|run| run.expect("all tasks in an acyclic graph are scheduled"))
        .collect();

        runs,
        makespan,
        events,
    })

            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if finish == now {
                // Zero-duration tasks complete immediately before the next admission.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let t = tasks[id].release;
                next_time = Some(next_time.map_or(t, |cur| cur.min(t)));
            }
        }
        for &t in &worker_available {
            if t > now {
                next_time = Some(next_time.map_or(t, |cur| cur.min(t)));
            }
        }

        // A remaining task must eventually have a release, completion, or worker event.
        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task is started"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal, always selecting the smallest available task ID.
    let mut available = vec![false; n];
    for id in 0..n {
        if indegree[id] == 0 {
            available[id] = true;
        }
    }

    let mut order = Vec::with_capacity(n);
    for _ in 0..n {
        let next = (0..n).find(|&id| available[id]);
        let Some(id) = next else {
            let blocked = (0..n).filter(|&i| !order.contains(&i)).collect();
            return Err(Error::Cycle { blocked });
        };

        available[id] = false;
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available[child] = true;
            }
        }
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut worker_available_at = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // (finish, task, worker)
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every task finishing now, ordered by task ID.
        let mut finished_now: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for entry in running.drain(..) {
            if entry.0 == now {
                finished_now.push(entry);
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;
        finished_now.sort_by_key(|&(_, task_id, _)| task_id);

        for &(finish, task_id, worker_id) in &finished_now {
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
            for &child in &children[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        // Start tasks one at a time, re-evaluating eligibility and ranking each time.
        loop {
            let worker_id = (0..workers).find(|&w| worker_available_at[w] <= now);
            let Some(worker_id) = worker_id else {
                break;
            };

            let best = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_predecessors[id] == 0
                })
                .min_by(|&a, &b| {
                    tail[b]
                        .cmp(&tail[a])
                        .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                        .then_with(|| a.cmp(&b))
                });

            let Some(task_id) = best else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let worker_ready = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available_at[worker_id] = worker_ready;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                // A zero-duration task completes immediately before another start.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the next finish, release, or worker availability event.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |current| current.min(time)));
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        let Some(next) = next_time else {
            // With validated acyclic dependencies and finite releases, this is unreachable.
            break;
        };
        now = next;
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task is started exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build children and indegrees. Dependency order does not affect traversal.
    let mut children = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }
    for list in &mut children {
        list.sort_unstable();
    }

    // Deterministic Kahn traversal, always choosing the smallest available ID.
    let mut available = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(&id) = available.iter().next() {
        available.remove(&id);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut is_removed = vec![false; n];
            for &id in &order {
                is_removed[id] = true;
            }
            is_removed
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_successor_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();

    #[derive(Clone, Copy)]
    struct Running {
        finish: u64,
        task: usize,
        worker: usize,
    }

    let mut running: Vec<Running> = Vec::new();
    let mut worker_available = vec![0u64; workers];
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all currently due tasks in task-ID order.
        let mut due: Vec<Running> = running
            .iter()
            .copied()
            .filter(|r| r.finish == now)
            .collect();
        due.sort_by_key(|r| r.task);

        for item in due {
            if let Some(pos) = running
                .iter()
                .position(|r| r.task == item.task && r.worker == item.worker)
            {
                running.swap_remove(pos);
            }
            completed[item.task] = true;
            completed_count += 1;
            makespan = makespan.max(item.finish);
            events.push(Event {
                time: item.finish,
                task: item.task,
                worker: item.worker,
                kind: EventKind::Finish,
            });
            for &child in &children[item.task] {
                remaining_deps[child] -= 1;
            }
        }

        // Repeatedly admit one task, re-evaluating after each start.
        loop {
            let worker = (0..workers).find(|&w| worker_available[w] <= now);
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now || remaining_deps[id] != 0 {
                    continue;
                }
                let is_better = match best {
                    None => true,
                    Some(prev) => {
                        tail[id] > tail[prev]
                            || (tail[id] == tail[prev]
                                && (tasks[id].priority > tasks[prev].priority
                                    || (tasks[id].priority == tasks[prev].priority
                                        && id < prev)))
                    }
                };
                if is_better {
                    best = Some(id);

                break;

                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,

                    finish,
                    task: task_id,
                    worker,

        let mut next_time: Option<u64> = None;
