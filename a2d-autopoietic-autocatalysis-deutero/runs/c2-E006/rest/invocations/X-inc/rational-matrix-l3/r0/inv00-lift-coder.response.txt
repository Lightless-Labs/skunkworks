```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = (self.num as i128).checked_neg().ok_or(RationalError::Overflow)?;
        rational_from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = b[0] == b'-';
        if negative { i = 1; }
        let num_start = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == num_start {
            return Err(parse_err(i, if i == b.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }
        let num_end = i;
        let slash = if i < b.len() && b[i] == b'/' { Some(i) } else { None };
        let mut den_start = 0;
        let mut den_end = 0;
        if let Some(s) = slash {
            i += 1;
            den_start = i;
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == den_start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = i;
        }
        if i < b.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_digits = &b[num_start..num_end];
        if num_digits.len() > 1 && num_digits[0] == b'0' {
            return Err(parse_err(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if slash.is_some() {
            let digits = &b[den_start..den_end];
            if digits.len() > 1 && digits[0] == b'0' {
                return Err(parse_err(den_start + 1, ParseKind::NonCanonical));
            }
        }

        let parse_mag = |digits: &[u8], limit: u128| -> Option<u128> {
            let mut value = 0u128;
            for &c in digits {
                value = value.checked_mul(10)?.checked_add((c - b'0') as u128)?;
                if value > limit { return None; }
            }
            Some(value)
        };
        let nmag = parse_mag(num_digits, if negative { 1u128 << 63 } else { i64::MAX as u128 })
            .ok_or_else(|| parse_err(0, ParseKind::OutOfRange))?;
        let n = if negative {
            if nmag == 1u128 << 63 { i64::MIN } else { -(nmag as i64) }
        } else { nmag as i64 };

        if let Some(s) = slash {
            let den_digits = &b[den_start..den_end];
            let dmag = parse_mag(den_digits, i64::MAX as u128)
                .ok_or_else(|| parse_err(den_start, ParseKind::OutOfRange))?;
            if dmag == 0 {
                return Err(parse_err(den_start, ParseKind::ZeroDenominator));
            }
            if dmag == 1 {
                return Err(parse_err(s, ParseKind::NonCanonical));
            }
            if n == 0 {
                return Err(parse_err(s, ParseKind::NonCanonical));
            }
            if gcd_u128(n.unsigned_abs() as u128, dmag) > 1 {
                return Err(parse_err(s, ParseKind::NonCanonical));
            }
            Ok(Rational { num: n, den: dmag as i64 })
        } else {
            Ok(Rational { num: n, den: 1 })
        }
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) } else { write!(f, "{}/{}", self.num, self.den) }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for col in 0..candidate_cols {
            if p == m.rows { break; }
            let mut best: Option<usize> = None;
            for row in p..m.rows {
                let value = m.data[row * m.cols + col];
                if value.num == 0 { continue; }
                let better = match best {
                    None => true,
                    Some(br) => {
                        let bv = m.data[br * m.cols + col];
                        (value.den, value.num.unsigned_abs(), row) < (bv.den, bv.num.unsigned_abs(), br)
                    }
                };
                if better { best = Some(row); }
            }
            let Some(q) = best else { continue };
            if q != p {
                for c in 0..m.cols { m.data.swap(p * m.cols + c, q * m.cols + c); }
                swaps.push((p, q));
            }
            let pivot = m.data[p * m.cols + col];
            for c in 0..m.cols {
                let idx = p * m.cols + c;
                m.data[idx] = m.data[idx].div(&pivot).map_err(|source| MatrixError::Arithmetic {
                    row: p, col: c, step: p, source,
                })?;
            }
            for row in 0..m.rows {
                if row == p { continue; }
                let factor = m.data[row * m.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..m.cols {
                    let idx = row * m.cols + c;
                    let product = factor.mul(&m.data[p * m.cols + c]).map_err(|source| MatrixError::Arithmetic {
                        row, col: c, step: p, source,
                    })?;
                    m.data[idx] = m.data[idx].sub(&product).map_err(|source| MatrixError::Arithmetic {
                        row, col: c, step: p, source,
                    })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix: m, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..width {
                data.push(if c < n {
                    self.data[r * n + c]
                } else if c - n == r {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let augmented = Matrix { rows: n, cols: width, data };
        let red = augmented.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                out.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix { rows: n, cols: n, data: out })
    }
}
```
