#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut normalized = String::with_capacity(input.len());
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        match ch {
            '\r' => {
                if chars.peek() == Some(&'\n') {
                    chars.next();
                }
                normalized.push('\n');
            }
            _ => normalized.push(ch),
        }
    }

    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            let run_len = i - run_start;
            if run_len >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                output.extend((1..run_len).map(|_| String::new()));
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let words: Vec<Vec<char>> = segment
        .split(|ch| is_separator(*ch))
        .filter(|part| !part.is_empty())
        .map(|part| part.to_vec())
        .collect();

    let mut line_number = 0usize;
    let mut content = Vec::<char>::new();

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let indent = if line_number == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            let suffix_len = word.len() - offset;

            if suffix_len <= capacity {
                let separator_len = usize::from(!content.is_empty());
                if content.len() + separator_len + suffix_len <= capacity {
                    if separator_len != 0 {
                        content.push(' ');
                    }
                    content.extend_from_slice(&word[offset..]);
                    offset = word.len();
                } else {
                    emit_line(&mut content, indent, output);
                    line_number += 1;
                }
            } else if content.is_empty() {
                content.extend_from_slice(&word[offset..offset + capacity]);
                offset += capacity;
                emit_line(&mut content, indent, output);
                line_number += 1;
            } else {
                let available = capacity - content.len() - 1;
                if available >= 2 {
                    content.push(' ');
                    content.extend_from_slice(&word[offset..offset + available]);
                    offset += available;
                    emit_line(&mut content, indent, output);
                    line_number += 1;
                } else {
                    emit_line(&mut content, indent, output);
                    line_number += 1;
                }
            }
        }
    }

    if !content.is_empty() {
        let indent = if line_number == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        emit_line(&mut content, indent, output);
    }
}

fn emit_line(content: &mut Vec<char>, indent: usize, output: &mut Vec<String>) {
    let mut line = String::with_capacity(indent + content.len());
    line.extend(std::iter::repeat_n(' ', indent));
    line.extend(content.drain(..));
    output.push(line);
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized = input.replace("\r\n", "\n").replace('\r', "\n");
    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            let run_len = i - run_start;
            if run_len >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                output.extend((0..run_len - 1).map(|_| String::new()));
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }

    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut indent = options.first_indent;
    let mut content = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let capacity = options.width - indent;
            let suffix_len = word.len() - offset;
            let empty_line = content_len == 0;

            if suffix_len <= capacity {
                let needed = suffix_len + usize::from(!empty_line);
                if content_len + needed <= capacity {
                    if !empty_line {
                        content.push(' ');
                        content_len += 1;
                    }
                    content.extend(word[offset..].iter());
                    content_len += suffix_len;
                    offset = word.len();
                } else {
                    emit_line(indent, &content, output);
                    first_line = false;
                    indent = options.rest_indent;
                    content.clear();
                    content_len = 0;
                }
            } else if empty_line {
                content.extend(word[offset..offset + capacity].iter());
                offset += capacity;
                emit_line(indent, &content, output);
                first_line = false;
                indent = options.rest_indent;
                content.clear();
                content_len = 0;
            } else {
                let available = capacity - content_len - 1;
                if available >= 2 {
                    content.push(' ');
                    content.extend(word[offset..offset + available].iter());
                    offset += available;
                    emit_line(indent, &content, output);
                    first_line = false;
                    indent = options.rest_indent;
                    content.clear();
                    content_len = 0;
                } else {
                    emit_line(indent, &content, output);
                    first_line = false;
                    indent = options.rest_indent;
                    content.clear();
                    content_len = 0;
                }
            }
        }
    }

    if content_len > 0 {
        emit_line(indent, &content, output);
    }

    // Keep the assignment explicit: every segment begins with first-line settings.
    let _ = first_line;
}

fn emit_line(indent: usize, content: &str, output: &mut Vec<String>) {
    let mut line = String::with_capacity(indent + content.len());
    line.extend(std::iter::repeat(' ').take(indent));
    line.push_str(content);
    output.push(line);
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut normalized = String::with_capacity(input.len());
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        if ch == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            let run_len = i - run_start;
            if run_len >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                output.extend((1..run_len).map(|_| String::new()));
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00a0}'
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }
    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut line_content: Vec<char> = Vec::new();

    for word in words {
        let mut offset = 0;

        while offset < word.len() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;

            if line_content.is_empty() {
                let remaining = word.len() - offset;
                if remaining <= capacity {
                    line_content.extend_from_slice(&word[offset..]);
                    offset = word.len();
                } else {
                    line_content.extend_from_slice(&word[offset..offset + capacity]);
                    offset += capacity;
                    emit_line(
                        &mut line_content,
                        indent,
                        output,
                    );
                    first_line = false;
                }
                continue;
            }

            let remaining = word.len() - offset;
            let needed = 1 + remaining;
            if needed <= capacity - line_content.len() {
                line_content.push(' ');
                line_content.extend_from_slice(&word[offset..]);
                offset = word.len();
            } else if remaining <= capacity {
                emit_line(&mut line_content, indent, output);
                first_line = false;
            } else {
                let available = capacity - line_content.len() - 1;
                if available >= 2 {
                    line_content.push(' ');
                    line_content.extend_from_slice(&word[offset..offset + available]);
                    offset += available;
                    emit_line(&mut line_content, indent, output);
                    first_line = false;
                } else {
                    emit_line(&mut line_content, indent, output);
                    first_line = false;
                }
            }
        }
    }

    if !line_content.is_empty() {
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        emit_line(&mut line_content, indent, output);
    }
}
