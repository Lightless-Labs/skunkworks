use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = (self.num as i128) * a + (rhs.num as i128) * b;
        let d = (self.den as i128) * a;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = (self.num as i128) * a - (rhs.num as i128) * b;
        let d = (self.den as i128) * a;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs(), self.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.num as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.den as i128 / g1);
        Self::from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.num.unsigned_abs()) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.den as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.num as i128 / g1);
        Self::from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn err(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }
        let mut i = 0;
        if bytes[0] == b'-' { i = 1; }
        let ns = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
        if i == ns {
            return Err(err(i, ParseKind::InvalidByte));
        }
        let ne = i;
        let mut ds = None;
        if i < bytes.len() && bytes[i] == b'/' {
            i += 1;
            let start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
            if i == start {
                return Err(err(i, ParseKind::InvalidByte));
            }
            ds = Some((start, i, i - start));
        }
        if i < bytes.len() {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let negative = bytes[0] == b'-';
        let ndigits = ne - ns;
        if ndigits >= 2 && bytes[ns] == b'0' {
            return Err(err(ns + 1, ParseKind::NonCanonical));
        }
        if negative && ndigits == 1 && bytes[ns] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some((start, _, len)) = ds {
            if len >= 2 && bytes[start] == b'0' {
                return Err(err(start + 1, ParseKind::NonCanonical));
            }
        }

        let max_num = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nm = 0u128;
        for &b in &bytes[ns..ne] {
            nm = nm.saturating_mul(10).saturating_add((b - b'0') as u128);
            if nm > max_num {
                return Err(err(0, ParseKind::OutOfRange));
            }
        }
        let num = if negative {
            if nm == (1u128 << 63) { i64::MIN } else { -(nm as i64) }
        } else {
            nm as i64
        };

        let mut den = 1i64;
        if let Some((start, end, _)) = ds {
            let mut dm = 0u128;
            for &b in &bytes[start..end] {
                dm = dm.saturating_mul(10).saturating_add((b - b'0') as u128);
                if dm > i64::MAX as u128 {
                    return Err(err(start, ParseKind::OutOfRange));
                }
            }
            if dm == 0 {
                return Err(err(start, ParseKind::ZeroDenominator));
            }
            den = dm as i64;
            let slash = start - 1;
            if den == 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if num == 0 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(num.unsigned_abs(), den as u128) > 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
        }
        Ok(Self { num, den })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols)
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn reduce(mut matrix: Self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let (rows, cols) = (matrix.rows, matrix.cols);
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for col in 0..candidate_cols {
            if p == rows { break; }
            let mut chosen: Option<usize> = None;
            for r in p..rows {
                let v = matrix.data[r * cols + col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), r);
                if chosen.map_or(true, |q| {
                    let w = matrix.data[q * cols + col];
                    key < (w.den, w.num.unsigned_abs(), q)
                }) {
                    chosen = Some(r);
                }
            }
            let q = match chosen { Some(q) => q, None => continue };
            if q != p {
                for c in 0..cols {
                    matrix.data.swap(p * cols + c, q * cols + c);
                }
                swaps.push((p, q));
            }
            let pivot = matrix.data[p * cols + col];
            for c in 0..cols {
                let old = matrix.data[p * cols + c];
                matrix.data[p * cols + c] = old.div(&pivot).map_err(|source| {
                    MatrixError::Arithmetic { row: p, col: c, step: p, source }
                })?;
            }
            for r in 0..rows {
                if r == p { continue; }
                let factor = matrix.data[r * cols + col];
                if factor.num == 0 { continue; }
                for c in 0..cols {
                    let product = factor.mul(&matrix.data[p * cols + c]).map_err(|source| {
                        MatrixError::Arithmetic { row: r, col: c, step: p, source }
                    })?;
                    matrix.data[r * cols + c] =
                        matrix.data[r * cols + c].sub(&product).map_err(|source| {
                            MatrixError::Arithmetic { row: r, col: c, step: p, source }
                        })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        Self::reduce(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = Rational { num: 1, den: 1 };
        }
        let red = Self::reduce(Self { rows: n, cols: width, data }, n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                out.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        let (m, n, k) = (self.rows, self.cols, rhs.cols);
        if m != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: m, rhs: rhs.rows });
        }
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..m {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            for c in 0..k {
                data[r * width + n + c] = rhs.data[r * k + c];
            }
        }
        let red = Self::reduce(Self { rows: m, cols: width, data }, n)
            .map_err(SolveError::Matrix)?;
        for r in 0..m {
            let mut zero = true;
            for c in 0..n {
                if red.matrix.data[r * width + c].num != 0 {
                    zero = false;
                    break;
                }
            }
            if zero {
                for c in 0..k {
                    if red.matrix.data[r * width + n + c].num != 0 {
                        let witness = red.matrix.data[r * width..(r + 1) * width].to_vec();
                        return Ok(Solution::None { row: r, rhs_col: c, witness });
                    }
                }
            }
        }

        let nk = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; nk];
        for (j, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] = red.matrix.data[j * width + n + c];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular_data };
        if red.pivots.len() == n {
            return Ok(Solution::Unique { values: particular, pivots: red.pivots });
        }

        let mut free = Vec::new();
        let mut pi = 0;
        for c in 0..n {
            if pi < red.pivots.len() && red.pivots[pi] == c {
                pi += 1;
            } else {
                free.push(c);
            }
        }
        let f = free.len();
        let nf = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; nf];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let coefficient = red.matrix.data[i * width + free_col];
                basis_data[pivot_col * f + j] = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col, col: j, step: red.pivots.len(), source
                    })
                })?;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: f, data: basis_data },
            pivots: red.pivots,
        })
    }
}
