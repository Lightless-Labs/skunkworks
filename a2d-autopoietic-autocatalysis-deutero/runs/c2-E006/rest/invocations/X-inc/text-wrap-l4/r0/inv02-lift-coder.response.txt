#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    hyphen_ended: bool,
    raw: bool,
    consumed: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    pre_rendered: Vec<String>,
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn render_line(line: &Line) -> String {
    let mut result = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            result.push(' ');
        }
        result.push_str(piece);
    }
    if line.hyphen_ended {
        result.push('-');
    }
    result
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.pre_rendered < b.pre_rendered
}

fn add_cost(
    suffix_cost: u64,
    gap: usize,
    final_line: bool,
    ending: Ending,
    one_piece_nonfinal: bool,
    width: usize,
) -> u64 {
    let g = gap as u64;
    let mut cost = suffix_cost + if final_line { g * g / 4 } else { g * g };
    match ending {
        Ending::Marker => cost += 0, // Added by caller using the configured penalty.
        Ending::Raw => cost += (2 * width + 1) as u64,
        Ending::Complete => {}
    }
    if one_piece_nonfinal && !final_line {
        cost += 3;
    }
    cost
}

#[derive(Clone, Copy)]
enum Ending {
    Complete,
    Marker,
    Raw,
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut memo: Vec<Vec<Option<Solution>>> = words
        .iter()
        .map(|w| vec![None; w.chars.len() + 1])
        .collect();

    fn solve(
        wi: usize,
        offset: usize,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Vec<Option<Solution>>],
    ) -> Solution {
        if wi == words.len() {
            return Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                pre_rendered: Vec::new(),
            };
        }
        if let Some(s) = &memo[wi][offset] {
            return s.clone();
        }

        let indent = if offset == 0 && wi == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let mut best: Option<Solution> = None;

        let mut pieces: Vec<String> = Vec::new();
        let mut used = 0usize;
        let mut cur_wi = wi;
        let mut cur_off = offset;

        loop {
            if cur_wi >= words.len() {
                break;
            }
            let word = &words[cur_wi];
            let remaining = word.chars.len() - cur_off;
            if !pieces.is_empty() {
                if used + 1 >= capacity {
                    break;
                }
                used += 1;
            }
            let avail = capacity - used;

            // Completed-word endings, including the current word if it fits.
            if remaining <= avail {
                let text: String = word.chars[cur_off..].iter().collect();
                pieces.push(text);
                let line_used = used + remaining;
                let next_wi = cur_wi + 1;
                let tail = solve(next_wi, 0, words, options, memo);
                let gap = options.width - (indent + line_used);
                let line = Line {
                    indent,
                    pieces: pieces.clone(),
                    hyphen_ended: false,
                    raw: false,
                    consumed: line_used,
                };
                let is_final = next_wi == words.len();
                let mut candidate = prepend(
                    line,
                    tail,
                    add_cost(0, gap, is_final, Ending::Complete, pieces.len() == 1, options.width),
                    0,
                    line_used,
                );
                if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                    best = Some(candidate);
                }
                pieces.pop();
            }

            // Marker endings in the current word.
            for &mark in word.markers.iter().filter(|&&m| m > cur_off) {
                let prefix_len = mark - cur_off;
                if prefix_len + 1 > avail {
                    continue;
                }
                let mut trial_pieces = pieces.clone();
                trial_pieces.push(word.chars[cur_off..mark].iter().collect());
                let line_used = used + prefix_len + 1;
                let tail = solve(cur_wi, mark + 1, words, options, memo);
                let gap = options.width - (indent + line_used);
                let line = Line {
                    indent,
                    pieces: trial_pieces.clone(),
                    hyphen_ended: true,
                    raw: false,
                    consumed: used + prefix_len,
                };
                let candidate = prepend(
                    line,
                    tail,
                    add_cost(0, gap, false, Ending::Marker, trial_pieces.len() == 1, options.width)
                        + options.hyphen_cost as u64,
                    1,
                    used + prefix_len,
                );
                if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                    best = Some(candidate);
                }
            }

            // Raw split is allowed only when this is the sole piece and no marker fits.
            if pieces.is_empty() && remaining > capacity {
                let marker_fits = word
                    .markers
                    .iter()
                    .any(|&m| m > cur_off && m - cur_off + 1 <= capacity);
                if !marker_fits {
                    let raw_text: String = word.chars[cur_off..cur_off + capacity].iter().collect();
                    let next_offset = cur_off + capacity;
                    let tail = solve(cur_wi, next_offset, words, options, memo);
                    let line = Line {
                        indent,
                        pieces: vec![raw_text],
                        hyphen_ended: false,
                        raw: true,
                        consumed: capacity,
                    };
                    let gap = 0;
                    let candidate = prepend(
                        line,
                        tail,
                        add_cost(0, gap, false, Ending::Raw, true, options.width),
                        0,
                        capacity,
                    );
                    if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }
            }

            // Extend the line with the next word only if the current word completed.
            if remaining > avail {
                break;
            }
            used += remaining;
            cur_wi += 1;
            cur_off = 0;
            if cur_wi >= words.len() {
                break;
            }
            // Next iteration adds the interword space.
        }

        let result = best.unwrap_or_else(|| {
            // Capacity-one progress is guaranteed by the raw-split rule.
            let word = &words[wi];
            let text: String = word.chars[offset..offset + 1].iter().collect();
            let line = Line {
                indent,
                pieces: vec![text],
                hyphen_ended: false,
                raw: true,
                consumed: 1,
            };
            let tail = solve(wi, offset + 1, words, options, memo);
            prepend(
                line,
                tail,
                (2 * options.width + 1) as u64,
                0,
                1,
            )
        });
        memo[wi][offset] = Some(result.clone());
        result
    }

    fn prepend(
        line: Line,
        mut tail: Solution,
        line_cost: u64,
        marker_inc: usize,
        consumed: usize,
    ) -> Solution {
        let mut lines = vec![line.clone()];
        lines.append(&mut tail.lines);
        let mut vector = vec![consumed];
        vector.append(&mut tail.consumed);
        let mut rendered = vec![render_line(&line)];
        rendered.append(&mut tail.pre_rendered);
        Solution {
            cost: line_cost + tail.cost,
            marker_lines: marker_inc + tail.marker_lines,
            lines,
            consumed: vector,
            pre_rendered: rendered,
        }
    }

    let result = solve(0, 0, words, options, &mut memo);
    let mut rendered = Vec::with_capacity(result.lines.len());
    for (i, line) in result.lines.iter().enumerate() {
        let mut s = " ".repeat(line.indent);
        let gaps = line.pieces.len().saturating_sub(1);
        let justify = options.justify && i + 1 < result.lines.len() && gaps > 0;
        if justify {
            let base_width = line.indent
                + line.pieces.iter().map(|p| p.chars().count()).sum::<usize>()
                + gaps
                + usize::from(line.hyphen_ended);
            let extra = options.width - base_width;
            let each = extra / gaps;
            let rem = extra % gaps;
            for (j, piece) in line.pieces.iter().enumerate() {
                if j > 0 {
                    s.push(' ');
                    let start = i % gaps;
                    let distance = (j - 1 + gaps - start) % gaps;
                    s.push_str(&" ".repeat(each + usize::from(distance < rem)));
                }
                s.push_str(piece);
            }
        } else {
            for (j, piece) in line.pieces.iter().enumerate() {
                if j > 0 {
                    s.push(' ');
                }
                s.push_str(piece);
            }
        }
        if line.hyphen_ended {
            s.push('-');
        }
        rendered.push(s);
    }
    rendered
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= original.len() {
                return false;
            }
            let c = original[j as usize].1;
            c != '~' && !separators(c)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] != '\n' {
            i += 1;
            continue;
        }
        let start = i;
        while i < normalized.len() && normalized[i] == '\n' {
            i += 1;
        }
        if i - start >= 2 {
            let segment = &normalized[segment_start..start];
            output.extend(layout_segment(segment, options));
            for _ in 1..(i - start) {
                output.push(String::new());
            }
            segment_start = i;
        }
    }
    output.extend(layout_segment(&normalized[segment_start..], options));
    Ok(output)
}

fn layout_segment(segment: &[char], options: &WrapOptions) -> Vec<String> {
    let mut words = Vec::<Word>::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    let finish_word = |words: &mut Vec<Word>,
                       chars: &mut Vec<char>,
                       markers: &mut Vec<usize>| {
        if !chars.is_empty() {
            words.push(Word {
                chars: std::mem::take(chars),
                markers: std::mem::take(markers),
            });
        } else {
            markers.clear();
        }
    };

    for &c in segment {
        if separators(c) {
            finish_word(&mut words, &mut chars, &mut markers);
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    finish_word(&mut words, &mut chars, &mut markers);

    solve_paragraph(&words, options)
}
