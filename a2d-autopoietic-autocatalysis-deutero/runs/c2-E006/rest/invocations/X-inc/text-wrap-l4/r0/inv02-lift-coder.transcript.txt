fn render_line(line: &Line) -> String {
    let mut result = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            result.push(' ');
        }
        result.push_str(piece);
    }
    if line.hyphen_ended {
        result.push('-');
    }
    result
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.pre_rendered < b.pre_rendered
}

fn add_cost(
    suffix_cost: u64,
    gap: usize,
    final_line: bool,
    ending: Ending,
    one_piece_nonfinal: bool,
    width: usize,
) -> u64 {
    let g = gap as u64;
    let mut cost = suffix_cost + if final_line { g * g / 4 } else { g * g };
    match ending {
        Ending::Marker => cost += 0, // Added by caller using the configured penalty.
        Ending::Raw => cost += (2 * width + 1) as u64,
        Ending::Complete => {}
    }
    if one_piece_nonfinal && !final_line {
        cost += 3;
    }
    cost
}

        return Vec::new();

        .iter()
        .map(|w| vec![None; w.chars.len() + 1])

        wi: usize,
        offset: usize,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Vec<Option<Solution>>],

        if wi == words.len() {

                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                pre_rendered: Vec::new(),

            return s.clone();

                break;

                    break;

                used += 1;

            let avail = capacity - used;

                        raw: true,
                        consumed: capacity,

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= original.len() {
                return false;
            }
            let c = original[j as usize].1;
            c != '~' && !separators(c)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] != '\n' {
            i += 1;
            continue;
        }
        let start = i;
        while i < normalized.len() && normalized[i] == '\n' {
            i += 1;
        }
        if i - start >= 2 {
            let segment = &normalized[segment_start..start];
            output.extend(layout_segment(segment, options));
            for _ in 1..(i - start) {
                output.push(String::new());
            }
            segment_start = i;
        }
    }
    output.extend(layout_segment(&normalized[segment_start..], options));
    Ok(output)
}

fn layout_segment(segment: &[char], options: &WrapOptions) -> Vec<String> {
    let mut words = Vec::<Word>::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    let finish_word = |words: &mut Vec<Word>,
                       chars: &mut Vec<char>,
                       markers: &mut Vec<usize>| {
        if !chars.is_empty() {
            words.push(Word {
                chars: std::mem::take(chars),
                markers: std::mem::take(markers),
            });
        } else {
            markers.clear();
        }
    };

    for &c in segment {
        if separators(c) {
            finish_word(&mut words, &mut chars, &mut markers);
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    finish_word(&mut words, &mut chars, &mut markers);

    solve_paragraph(&words, options)
}
