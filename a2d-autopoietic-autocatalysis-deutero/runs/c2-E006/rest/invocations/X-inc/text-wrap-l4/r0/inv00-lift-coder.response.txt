#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker_end: bool,
    pieces: usize,
    gap: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed_vector: Vec<usize>,
    strings: Vec<String>,
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed_vector != b.consumed_vector {
        return a.consumed_vector > b.consumed_vector;
    }
    a.strings < b.strings
}

fn render_prefix(chars: &[char], start: usize, end: usize) -> String {
    chars[start..end].iter().collect()
}

fn optimize_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    fn rec(
        words: &[Word],
        wi: usize,
        off: usize,
        first: bool,
        options: &WrapOptions,
        memo: &mut Vec<Vec<Option<Solution>>>,
    ) -> Solution {
        if wi >= words.len() {
            return Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                strings: Vec::new(),
            };
        }
        if let Some(s) = &memo[wi][off] {
            return s.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut candidates = Vec::new();

        // Enumerate a line as a sequence of word pieces and optional terminal split.
        struct Build {
            text: String,
            used: usize,
            pieces: usize,
            next_wi: usize,
            next_off: usize,
            marker_end: bool,
            raw: bool,
        }

        fn extend(
            words: &[Word],
            wi: usize,
            off: usize,
            cap: usize,
            used: usize,
            pieces: usize,
            text: String,
            built: &mut Vec<Build>,
        ) {
            let word = &words[wi];
            let remaining = word.chars.len() - off;
            let space = if pieces == 0 { 0 } else { 1 };

            if used + space < cap {
                let available = cap - used - space;
                let finish = off + remaining.min(available);
                if finish == word.chars.len() {
                    let mut t = text.clone();
                    if space != 0 {
                        t.push(' ');
                    }
                    t.push_str(&render_prefix(&word.chars, off, finish));
                    built.push(Build {
                        text: t.clone(),
                        used: used + space + remaining,
                        pieces: pieces + 1,
                        next_wi: wi + 1,
                        next_off: 0,
                        marker_end: false,
                        raw: false,
                    });
                    if wi + 1 < words.len() {
                        extend(
                            words,
                            wi + 1,
                            0,
                            cap,
                            used + space + remaining,
                            pieces + 1,
                            t,
                            built,
                        );
                    }
                } else {
                    let mut t = text.clone();
                    if space != 0 {
                        t.push(' ');
                    }
                    t.push_str(&render_prefix(&word.chars, off, finish));
                    // Marker splits strictly inside the current suffix.
                    for &m in &word.markers {
                        if m > off && m < finish && m - off + 1 <= available {
                            let mut mt = text.clone();
                            if space != 0 {
                                mt.push(' ');
                            }
                            mt.push_str(&render_prefix(&word.chars, off, m));
                            mt.push('-');
                            built.push(Build {
                                text: mt,
                                used: used + space + (m - off) + 1,
                                pieces: pieces + 1,
                                next_wi: wi,
                                next_off: m,
                                marker_end: true,
                                raw: false,
                            });
                        }
                    }

                    // Raw split only on an otherwise empty line and only if no marker fits.
                    if pieces == 0
                        && remaining > cap
                        && !word
                            .markers
                            .iter()
                            .any(|&m| m > off && m < word.chars.len() && m - off + 1 <= cap)
                    {
                        let finish = off + cap;
                        let mut rt = render_prefix(&word.chars, off, finish);
                        built.push(Build {
                            text: std::mem::take(&mut rt),
                            used: cap,
                            pieces: 1,
                            next_wi: wi,
                            next_off: finish,
                            marker_end: false,
                            raw: true,
                        });
                    }
                }
            }
        }

        let mut builds = Vec::new();
        extend(words, wi, off, cap, 0, 0, String::new(), &mut builds);

        for b in builds {
            if b.used == 0 {
                continue;
            }
            let terminal = b.next_wi >= words.len();
            let next = if terminal {
                Solution {
                    cost: 0,
                    marker_lines: 0,
                    lines: Vec::new(),
                    consumed_vector: Vec::new(),
                    strings: Vec::new(),
                }
            } else {
                rec(
                    words,
                    b.next_wi,
                    b.next_off,
                    false,
                    options,
                    memo,
                )
            };
            let displayed = indent + b.used;
            let gap = options.width - displayed;
            let mut cost = next.cost;
            if terminal {
                cost += (gap * gap / 4) as u64;
            } else {
                cost += (gap * gap) as u64;
                if b.pieces == 1 {
                    cost += 3;
                }
            }
            if b.marker_end {
                cost += options.hyphen_cost as u64;
            }
            if b.raw {
                cost += (2 * options.width + 1) as u64;
            }

            let mut line_text = " ".repeat(indent);
            line_text.push_str(&b.text);
            let consumed = b.used - usize::from(b.marker_end);
            let mut lines = vec![Line {
                text: line_text.clone(),
                consumed,
                marker_end: b.marker_end,
                pieces: b.pieces,
                gap,
            }];
            lines.extend(next.lines.clone());
            let mut consumed_vector = vec![consumed];
            consumed_vector.extend(next.consumed_vector.clone());
            let mut strings = vec![line_text];
            strings.extend(next.strings.clone());

            candidates.push(Solution {
                cost,
                marker_lines: usize::from(b.marker_end) + next.marker_lines,
                lines,
                consumed_vector,
                strings,
            });
        }

        let best = candidates
            .into_iter()
            .reduce(|a, b| if better(&a, &b) { a } else { b })
            .unwrap_or(Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                strings: Vec::new(),
            });
        memo[wi][off] = Some(best.clone());
        best
    }

    let mut memo: Vec<Vec<Option<Solution>>> = words
        .iter()
        .map(|w| vec![None; w.chars.len() + 1])
        .collect();
    rec(words, 0, 0, true, options, &mut memo).lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    if original.len() > 512 {
        return Err(WrapError::InputTooLong {
            chars: original.len(),
        });
    }

    for &(index, c) in &original {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..original.len() {
        if original[i].1 == '~' {
            let valid_neighbor = |j: isize| -> bool {
                if j < 0 || j as usize >= original.len() {
                    return false;
                }
                let c = original[j as usize].1;
                c != '~' && !separators(c)
            };
            if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
                return Err(WrapError::InvalidMarker {
                    index: original[i].0,
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                append_segment(&segment, options, &mut output);
                segment.clear();
                for _ in 1..(j - i) {
                    output.push(String::new());
                }
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    append_segment(&segment, options, &mut output);
    Ok(output)
}

fn append_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words = Vec::new();
    let mut i = 0;
    while i < segment.len() {
        while i < segment.len() && separators(segment[i]) {
            i += 1;
        }
        if i >= segment.len() {
            break;
        }
        let mut chars = Vec::new();
        let mut markers = Vec::new();
        while i < segment.len() && !separators(segment[i]) {
            if segment[i] == '~' {
                markers.push(chars.len());
            } else {
                chars.push(segment[i]);
            }
            i += 1;
        }
        words.push(Word { chars, markers });
    }

    if words.is_empty() {
        return;
    }

    let mut lines = optimize_paragraph(&words, options);
    if options.justify {
        for (line_index, line) in lines.iter_mut().enumerate() {
            if line_index + 1 >= lines.len() || line.pieces < 2 {
                continue;
            }
            let indent = if line_index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let content = line.text.chars().skip(indent).collect::<String>();
            let parts: Vec<&str> = content.split(' ').collect();
            let gaps = parts.len().saturating_sub(1);
            if gaps == 0 {
                continue;
            }
            let extra = options.width.saturating_sub(scalar_len(&line.text));
            let each = extra / gaps;
            let rem = extra % gaps;
            let start = line_index % gaps;
            let mut rendered = " ".repeat(indent);
            for (k, part) in parts.iter().enumerate() {
                if k > 0 {
                    let gap_index = k - 1;
                    let add = each + usize::from((gap_index + gaps - start) % gaps < rem);
                    rendered.push_str(&" ".repeat(1 + add));
                }
                rendered.push_str(part);
            }
            line.text = rendered;
        }
    }
    output.extend(lines.into_iter().map(|l| l.text));
}
