#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Modifier {
    Explode,
    Prefix(usize),
}

struct Specifier {
    name: String,
    pos: usize,
    modifier: Option<Modifier>,
}

struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let mut operator = None;

    if content_start < end {
        let b = bytes[content_start];
        if matches!(b, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
            operator = Some(b);
            content_start += 1;
        } else if !is_name_start(b) && b != b',' {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut specs = Vec::new();
    let mut used = Vec::<String>::new();
    let mut part_start = content_start;

    loop {
        let part_end = bytes[part_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| part_start + offset)
            .unwrap_or(end);

        if part_start == part_end {
            return Err(Error::EmptyVariable { pos: part_start });
        }

        let part = &bytes[part_start..part_end];
        let modifier_offset = part.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_offset.map(|i| part_start + i).unwrap_or(part_end);

        if name_end == part_start {
            return Err(Error::InvalidName { pos: part_start });
        }

        let name_bytes = &bytes[part_start..name_end];
        for (i, &b) in name_bytes.iter().enumerate() {
            if (i == 0 && !is_name_start(b)) || (i > 0 && !is_name_continue(b)) {
                return Err(Error::InvalidName { pos: part_start + i });
            }
        }
        let name = template[part_start..name_end].to_string();

        let modifier = if let Some(offset) = modifier_offset {
            let mod_pos = part_start + offset;
            let suffix = &bytes[mod_pos..part_end];
            if suffix == b"*" {
                Some(Modifier::Explode)
            } else if suffix.len() >= 2
                && suffix[0] == b':'
                && suffix.len() <= 5
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut value = 0usize;
                for &digit in &suffix[1..] {
                    value = value * 10 + (digit - b'0') as usize;
                }
                if (1..=9999).contains(&value) {
                    Some(Modifier::Prefix(value))
                } else {
                    return Err(Error::InvalidModifier { pos: mod_pos });
                }
            } else {
                return Err(Error::InvalidModifier { pos: mod_pos });
            }
        } else {
            None
        };

        if used.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: part_start });
        }
        used.push(name.clone());
        specs.push(Specifier {
            name,
            pos: part_start,
            modifier,
        });

        if part_end == end {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        specs,
    })
}

fn scan_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = bytes[start + 1..]
                    .iter()
                    .position(|&b| b == b'}')
                    .map(|offset| start + 1 + offset);
                let Some(end) = close else {
                    return Err(Error::Unclosed { pos: start });
                };
                if let Some(offset) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + offset,
                    });
                }
                expressions.push(parse_expression(template, start, end)?);
                i = end + 1;
            }
            _ => i += 1,
        }
    }
    Ok(expressions)
}

fn reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn encode(input: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if unreserved(b) || (allow_reserved && reserved(b)) {
            out.push(b as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
            i += 1;
        }
    }
    out
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn prefix_chars(value: &str, count: usize) -> &str {
    match value.char_indices().nth(count) {
        Some((index, _)) => &value[..index],
        None => value,
    }
}

fn named_field(name: &str, value: String, operator: u8) -> String {
    if operator == b';' && value.is_empty() {
        name.to_string()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expression.operator.unwrap_or(b'\0');
    let allow_reserved = matches!(op, b'+' | b'#');
    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Some(Modifier::Prefix(_)))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(s) => {
                let truncated = match spec.modifier {
                    Some(Modifier::Prefix(n)) => prefix_chars(s, n),
                    _ => s,
                };
                let encoded = encode(truncated, allow_reserved);
                if matches!(op, b';' | b'?' | b'&') {
                    fields.push(named_field(&spec.name, encoded, op));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Some(Modifier::Explode));
                if exploded {
                    for item in items {
                        let encoded = encode(item, allow_reserved);
                        if matches!(op, b';' | b'?' | b'&') {
                            fields.push(named_field(&spec.name, encoded, op));
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(op, b';' | b'?' | b'&') {
                        fields.push(named_field(&spec.name, joined, op));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|seen| *seen == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }
                if entries.is_empty() {
                    continue;
                }

                let exploded = matches!(spec.modifier, Some(Modifier::Explode));
                if exploded {
                    for (key, value) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(value, allow_reserved);
                        let field = if op == b';' && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, allow_reserved));
                        components.push(encode(value, allow_reserved));
                    }
                    let joined = components.join(",");
                    if matches!(op, b';' | b'?' | b'&') {
                        fields.push(named_field(&spec.name, joined, op));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match op {
        b'#' => (",", "#"),
        b'.' => (".", "."),
        b'/' => ("/", "/"),
        b';' => (";", ";"),
        b'?' => ("&", "?"),
        b'&' => ("&", "&"),
        _ => (",", ""),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = scan_template(template)?;
    let mut out = String::new();
    let mut cursor = 0;

    for expression in &expressions {
        out.push_str(&template[cursor..expression.start]);
        out.push_str(&expand_expression(expression, vars)?);
        cursor = expression.end + 1;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
