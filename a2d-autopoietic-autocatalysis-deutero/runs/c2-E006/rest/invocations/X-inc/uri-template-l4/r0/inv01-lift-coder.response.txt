#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

#[derive(Debug, Clone)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    pos: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut start = open + 1;
    let mut operator = Operator::Simple;

    if start < close {
        operator = match bytes[start] {
            b'+' => Operator::Reserved,
            b'#' => Operator::Fragment,
            b'.' => Operator::Dot,
            b'/' => Operator::Slash,
            b';' => Operator::Semicolon,
            b'?' => Operator::Query,
            b'&' => Operator::Continuation,
            b'=' | b',' | b'!' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: start });
            }
            _ => Operator::Simple,
        };

        if !matches!(operator, Operator::Simple) {
            start += 1;
        }
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = start;

    loop {
        let mut spec_end = spec_start;
        while spec_end < close && bytes[spec_end] != b',' {
            spec_end += 1;
        }

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let mut modifier_start = spec_end;
        for i in spec_start..spec_end {
            if bytes[i] == b'*' || bytes[i] == b':' {
                modifier_start = i;
                break;
            }
        }

        let name_bytes = &bytes[spec_start..modifier_start];
        if name_bytes.is_empty() {
            return Err(Error::InvalidName { pos: spec_start });
        }

        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (offset, byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(*byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }
        let name = String::from_utf8(name_bytes.to_vec()).expect("ASCII name");

        let modifier = if modifier_start == spec_end {
            Modifier::None
        } else {
            let suffix = &bytes[modifier_start..spec_end];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.len() >= 2
                && suffix[0] == b':'
                && suffix.len() <= 5
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut n = 0usize;
                for digit in &suffix[1..] {
                    n = n * 10 + (digit - b'0') as usize;
                }
                if (1..=9999).contains(&n) {
                    Modifier::Prefix(n)
                } else {
                    return Err(Error::InvalidModifier { pos: modifier_start });
                }
            } else {
                return Err(Error::InvalidModifier { pos: modifier_start });
            }
        };

        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            pos: spec_start,
            modifier,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let b = bytes[i];
        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let is_reserved = reserved
            && matches!(
                b,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || is_reserved {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0F) as usize] as char);
        }
        i += 1;
    }

    out
}

fn encode_prefix(value: &str, prefix: Option<usize>, reserved: bool) -> String {
    match prefix {
        Some(n) => encode(&value.chars().take(n).collect::<String>(), reserved),
        None => encode(value, reserved),
    }
}

fn named_field(name: &str, value: String, omit_equals_for_empty: bool) -> String {
    if omit_equals_for_empty && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(expr.operator, Operator::Reserved | Operator::Fragment);
    let (separator, prefix, named, semicolon) = match expr.operator {
        Operator::Simple | Operator::Reserved => (",", "", false, false),
        Operator::Fragment => (",", "#", false, false),
        Operator::Dot => (".", ".", false, false),
        Operator::Slash => ("/", "/", false, false),
        Operator::Semicolon => (";", ";", true, true),
        Operator::Query => ("&", "?", true, false),
        Operator::Continuation => ("&", "&", true, false),
    };

    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let selected = vars.iter().rev().find(|(name, _)| *name == spec.name);
        let Some((_, value)) = selected else {
            continue;
        };

        let prefix_len = match spec.modifier {
            Modifier::Prefix(n) => Some(n),
            _ => None,
        };
        let explode = matches!(spec.modifier, Modifier::Explode);

        match value {
            Value::Scalar(s) => {
                let encoded = encode_prefix(s, prefix_len, reserved);
                fields.push(if named {
                    named_field(&spec.name, encoded, semicolon)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if prefix_len.is_some() {
                    return Err(Error::TypeMismatch { pos: spec.pos });
                }
                if items.is_empty() {
                    continue;
                }
                if explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if named {
                            named_field(&spec.name, encoded, semicolon)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if named {
                        named_field(&spec.name, joined, semicolon)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if prefix_len.is_some() {
                    return Err(Error::TypeMismatch { pos: spec.pos });
                }
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|existing| *existing == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }
                if entries.is_empty() {
                    continue;
                }
                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        fields.push(if semicolon && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if named {
                        named_field(&spec.name, joined, semicolon)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        Ok(String::new())
    } else {
        Ok(format!("{prefix}{}", fields.join(separator)))
    }
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut out = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let mut close = open + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                let expression = parse_expression(template, open, close)?;
                out.push_str(&expand_expression(&expression, vars)?);
                cursor = close + 1;
            }
            _ => {
                let start = cursor;
                while cursor < bytes.len() && bytes[cursor] != b'{' && bytes[cursor] != b'}' {
                    cursor += 1;
                }
                out.push_str(&template[start..cursor]);
            }
        }
    }

    Ok(out)
}
