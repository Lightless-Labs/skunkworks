#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'_'
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    let operator = if content_start < close {
        match bytes[content_start] {
            b'+' => {
                content_start += 1;
                Operator::Plus
            }
            b'#' => {
                content_start += 1;
                Operator::Hash
            }
            b'.' => {
                content_start += 1;
                Operator::Dot
            }
            b'/' => {
                content_start += 1;
                Operator::Slash
            }
            b';' => {
                content_start += 1;
                Operator::Semicolon
            }
            b'?' => {
                content_start += 1;
                Operator::Question
            }
            b'&' => {
                content_start += 1;
                Operator::Ampersand
            }
            b'!' | b'=' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: content_start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let modifier_at = (spec_start..spec_end)
            .find(|&i| bytes[i] == b'*' || bytes[i] == b':')
            .unwrap_or(spec_end);
        let name_start = spec_start;
        let name_end = modifier_at;

        if name_start == name_end || !is_name_start(bytes[name_start]) {
            return Err(Error::InvalidName { pos: name_start });
        }

        let invalid_name_byte = (name_start + 1..name_end)
            .find(|&i| !is_name_continue(bytes[i]));
        if let Some(pos) = invalid_name_byte {
            return Err(Error::InvalidName { pos });
        }

        let name = template[name_start..name_end].to_string();
        let modifier = if modifier_at == spec_end {
            Modifier::None
        } else {
            let suffix = &bytes[modifier_at..spec_end];
            match suffix {
                [b'*'] => Modifier::Explode,
                [b':', digits @ ..] if (1..=4).contains(&digits.len())
                    && digits[0].is_ascii_digit()
                    && digits[0] != b'0'
                    && digits.iter().all(u8::is_ascii_digit) =>
                {
                    let mut value = 0usize;
                    for digit in digits {
                        value = value * 10 + usize::from(*digit - b'0');
                    }
                    if value <= 9999 {
                        Modifier::Prefix(value)
                    } else {
                        return Err(Error::InvalidModifier { pos: modifier_at });
                    }
                }
                _ => return Err(Error::InvalidModifier { pos: modifier_at }),
            }
        };

        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            start: spec_start,
            modifier,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, specs })
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            out.push(byte as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
            i += 1;
        }
    }

    out
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn prefix_value(value: &str, modifier: Modifier) -> String {
    match modifier {
        Modifier::Prefix(count) => value.chars().take(count).collect(),
        Modifier::None | Modifier::Explode => value.to_string(),
    }
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_string()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(expression.operator, Operator::Plus | Operator::Hash);
    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(value) => {
                let value = prefix_value(value, spec.modifier);
                fields.push(match expression.operator {
                    Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                        named_field(&spec.name, encode(&value, reserved), expression.operator)
                    }
                    _ => encode(&value, reserved),
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(match expression.operator {
                            Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                                named_field(&spec.name, encoded, expression.operator)
                            }
                            _ => encoded,
                        });
                    }
                } else {
                    let value = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(match expression.operator {
                        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                            named_field(&spec.name, value, expression.operator)
                        }
                        _ => value,
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|previous| *previous == key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let field = if expression.operator == Operator::Semicolon
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let value = components.join(",");
                    fields.push(match expression.operator {
                        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                            named_field(&spec.name, value, expression.operator)
                        }
                        _ => value,
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Hash => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();

    // Parse and validate every expression before consulting any bindings.
    let mut expressions = Vec::<(usize, usize, Expression)>::new();
    let mut pos = 0;
    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let Some(relative_close) = bytes[pos + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos });
                };
                let close = pos + 1 + relative_close;
                let expression = parse_expression(template, pos, close)?;
                expressions.push((pos, close, expression));
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    for (open, close, expression) in &expressions {
        output.push_str(&template[literal_start..*open]);
        output.push_str(&expand_expression(expression, vars)?);
        literal_start = close + 1;
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}
