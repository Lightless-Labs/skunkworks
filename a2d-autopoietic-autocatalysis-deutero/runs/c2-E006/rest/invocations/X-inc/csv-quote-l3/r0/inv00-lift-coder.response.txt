#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if input.starts_with('\u{feff}') {
        '\u{feff}'.len_utf8()
    } else {
        0
    };
    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        let outside_quote = !matches!(state, State::Quoted { .. });
        if outside_quote && byte == b'\r' {
            if bytes.get(pos + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: pos,
                });
            }
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            pos += 2;
            record_index += 1;
            record_started = false;
            state = State::FieldStart;
            continue;
        }

        match state {
            State::FieldStart => {
                if byte == quote {
                    record_started = true;
                    state = State::Quoted { opening: pos };
                    pos += 1;
                } else if byte == delimiter {
                    fields.push(std::mem::take(&mut field));
                    record_started = true;
                    pos += 1;
                } else if byte == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    pos += 1;
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                } else {
                    field.push(byte as char);
                    record_started = true;
                    state = State::Unquoted;
                    pos += 1;
                }
            }
            State::Unquoted => {
                if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                } else if byte == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    pos += 1;
                } else if byte == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    pos += 1;
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                } else {
                    field.push(byte as char);
                    pos += 1;
                }
            }
            State::Quoted { opening } => {
                if byte == quote {
                    if bytes.get(pos + 1) == Some(&quote) {
                        field.push(quote as char);
                        pos += 2;
                    } else {
                        state = State::AfterQuote;
                        pos += 1;
                    }
                } else {
                    field.push(byte as char);
                    pos += 1;
                }
                let _ = opening;
            }
            State::AfterQuote => {
                if byte == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    pos += 1;
                } else if byte == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    pos += 1;
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started || !fields.is_empty() || !field.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let mut parsed = parse_with_dialect(input, config.delimiter, config.quote)?;

    if !config.headers {
        return Ok(Table {
            header: None,
            rows: parsed,
            duplicates: config.duplicates,
        });
    }

    if parsed.is_empty() {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    }

    let header = parsed.remove(0);
    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen: std::collections::HashMap<&str, usize> = std::collections::HashMap::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name.as_str(), second);
        }
    }

    for (record, row) in parsed.iter().enumerate() {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: record + 1,
                expected: header.len(),
                actual: row.len(),
            });
        }
    }

    Ok(Table {
        header: Some(header),
        rows: parsed,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = build_table(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn valid_integer_syntax(text: &str) -> bool {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    (b'1'..=b'9').contains(&digits[0]) && digits[1..].iter().all(u8::is_ascii_digit)
}

fn valid_decimal_syntax(text: &str) -> bool {
    let bytes = text.as_bytes();
    let unsigned = if bytes.first() == Some(&b'-') {
        &bytes[1..]
    } else {
        bytes
    };
    let Some(dot) = unsigned.iter().position(|&byte| byte == b'.') else {
        return false;
    };
    let integral = &unsigned[..dot];
    let fractional = &unsigned[dot + 1..];
    let integral_ok = integral == b"0"
        || (!integral.is_empty()
            && (b'1'..=b'9').contains(&integral[0])
            && integral[1..].iter().all(u8::is_ascii_digit));
    let fractional_ok = fractional.len() == 2 && fractional.iter().all(u8::is_ascii_digit);
    if !integral_ok || !fractional_ok {
        return false;
    }
    !(bytes.first() == Some(&b'-') && integral == b"0" && fractional == b"00")
}

fn parse_integer(text: &str) -> Result<i64, CellErrorKind> {
    if !valid_integer_syntax(text) {
        return Err(CellErrorKind::InvalidInteger);
    }
    text.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(text: &str) -> Result<i64, CellErrorKind> {
    if !valid_decimal_syntax(text) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = text.starts_with('-');
    let unsigned = if negative { &text[1..] } else { text };
    let dot = unsigned.find('.').expect("validated decimal has a dot");
    let integral = &unsigned[..dot];
    let fractional = &unsigned[dot + 1..];
    let magnitude = integral
        .parse::<i128>()
        .expect("validated decimal integral part fits i128")
        * 100
        + fractional
            .parse::<i128>()
            .expect("validated decimal fractional part fits i128");
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
                output.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let evaluated = if let Some(column) = resolved {
                if let Some(cell) = row.get(column) {
                    if cell.is_empty() {
                        if spec.nullable {
                            Ok(Value::Null)
                        } else if spec.kind == ColumnType::Text {
                            Ok(Value::Text(String::new()))
                        } else {
                            Err((Some(column), CellErrorKind::Empty))
                        }
                    } else {
                        let parsed = match spec.kind {
                            ColumnType::Text => Ok(Value::Text(cell.clone())),
                            ColumnType::Integer => parse_integer(cell).map(Value::Integer),
                            ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
                            ColumnType::Boolean => match cell.as_str() {
                                "yes" | "1" => Ok(Value::Boolean(true)),
                                "no" | "0" => Ok(Value::Boolean(false)),
                                _ => Err(CellErrorKind::InvalidBoolean),
                            },
                        };
                        parsed.map_err(|kind| (Some(column), kind))
                    }
                } else {
                    Err((Some(column), CellErrorKind::MissingCell))
                }
            } else {
                Err((None, CellErrorKind::UnknownColumn))
            };

            match evaluated {
                Ok(value) => output.push(Some(value)),
                Err((column, kind)) => {
                    output.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    row_failed = true;
                    if policy == ErrorPolicy::FirstPerRow || policy == ErrorPolicy::Stop {
                        for _ in spec_index + 1..columns.len() {
                            output.push(None);
                        }
                        break;
                    }
                }
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && row_failed {
            break;
        }
    }

    result
}
