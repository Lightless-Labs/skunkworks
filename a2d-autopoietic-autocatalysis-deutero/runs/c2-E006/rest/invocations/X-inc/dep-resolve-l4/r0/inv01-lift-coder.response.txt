fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(items: &[Constraint]) -> Vec<Constraint> {
    let mut out = items.to_vec();
    out.sort_by_key(|c| constraint_rank(*c));
    out.dedup();
    out
}

fn constraints_text(items: &[Constraint]) -> String {
    canonical_constraints(items)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sort_evidence(items: &[Evidence]) -> Vec<Evidence> {
    let mut out = items.to_vec();
    out.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn evidence_lines(items: &[Evidence]) -> String {
    sort_evidence(items)
        .iter()
        .map(|e| {
            format!(
                "\n  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect::<String>()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

use std::collections::{BTreeMap, BTreeSet};

fn platform_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn platform_text(platform: Platform) -> &'static str {
    match platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        fs.join(",")
    )
}

fn dep_token(source: &str, version: Version, index: usize, d: &Dependency) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        d.package,
        constraint_text(d.constraint),
        sorted_feature_text(&d.features),
        d.gate.as_deref().unwrap_or("*"),
        platform_text(d.platform)
    )
}

fn sorted_feature_text(features: &[String]) -> String {
    let mut fs = features.to_vec();
    fs.sort();
    fs.dedup();
    fs.join(",")
}

fn insert_origin(active: &mut Active, origin: Origin, features: &[String]) {
    active.constraints.push(origin.constraint);
    active.origins.push(origin);
    active.features.extend(features.iter().cloned());
}

fn canonical_path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn compute_active(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
    target: Target,
) -> (BTreeMap<String, Active>, BTreeMap<String, Vec<String>>) {
    let mut active: BTreeMap<String, Active> = BTreeMap::new();

    for (i, root) in roots.iter().enumerate() {
        let a = active.entry(root.package.clone()).or_insert_with(|| Active {
            constraints: Vec::new(),
            origins: Vec::new(),
            features: BTreeSet::new(),
            path: Vec::new(),
        });
        insert_origin(
            a,
            Origin {
                constraint: root.constraint,
                path: vec![root_token(i, root)],
            },
            &root.features,
        );
    }

    // Repeatedly propagate local feature enables and active dependency edges.
    // Every propagation only adds finite facts, so this reaches a least fixed point.
    loop {
        let before = active.clone();

        let names: Vec<String> = active.keys().cloned().collect();
        for name in names {
            let Some(version) = selected.get(&name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&name).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested: Vec<String> = active[&name].features.iter().cloned().collect();
            for feature_name in requested {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    let a = active.get_mut(&name).unwrap();
                    a.features.extend(feature.enables.iter().cloned());
                }
            }

            let activated = active[&name].features.clone();
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }

                let source_path = active[&name].path.clone();
                let token = dep_token(&name, version, j, dep);
                let mut path = source_path;
                path.push(token);
                let a = active.entry(dep.package.clone()).or_insert_with(|| Active {
                    constraints: Vec::new(),
                    origins: Vec::new(),
                    features: BTreeSet::new(),
                    path: Vec::new(),
                });
                insert_origin(
                    a,
                    Origin {
                        constraint: dep.constraint,
                        path,
                    },
                    &dep.features,
                );
            }
        }

        // Recompute canonical incoming paths from all current origins and paths.
        let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
        for (name, a) in &active {
            for origin in &a.origins {
                let candidate = origin.path.clone();
                match paths.get(name) {
                    None => {
                        paths.insert(name.clone(), candidate);
                    }
                    Some(existing) if canonical_path_better(&candidate, existing) => {
                        paths.insert(name.clone(), candidate);
                    }
                    _ => {}
                }
            }
        }
        // Dependency origins may have been formed from a path that became shorter in this
        // iteration; another iteration propagates that improvement.
        for (name, path) in paths {
            active.get_mut(&name).unwrap().path = path;
        }

        if active == before {
            break;
        }
    }

    // Paths can require another relaxation after newly discovered canonical source paths.
    // The closure loop above converges because path propagation only improves finite paths.
    let paths = active
        .iter()
        .map(|(name, a)| (name.clone(), a.path.clone()))
        .collect();
    (active, paths)
}

fn make_evidence(active: &Active) -> Vec<Evidence> {
    let mut evidence: Vec<Evidence> = active
        .origins
        .iter()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    evidence.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    evidence
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: BTreeMap<String, &'a Package>,
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (active, paths) = compute_active(self.roots, selected, &self.catalog, self.target);

        let missing: Vec<String> = active
            .keys()
            .filter(|name| !self.catalog.contains_key(*name))
            .cloned()
            .collect();
        if let Some(package) = missing.into_iter().next() {
            self.record_error(ResolveError::Missing {
                package: package.clone(),
                evidence: make_evidence(&active[&package]),
            });
            return Ok(());
        }

        for (package, version) in selected.iter() {
            let a = &active[package];
            if !a.constraints.iter().all(|c| matches(*c, *version)) {
                self.record_error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: canonical_constraints(&a.constraints),
                    evidence: make_evidence(a),
                });
                return Ok(());
            }
        }

        for (package, a) in &active {
            if selected.contains_key(package) {
                let version = selected[package];
                let release = self.catalog[package]
                    .releases
                    .iter()
                    .find(|r| r.version == version)
                    .unwrap();
                for feature in &a.features {
                    if !release.features.iter().any(|f| f.name == *feature) {
                        self.record_error(ResolveError::MissingFeature {
                            package: package.clone(),
                            selected: version,
                            feature: feature.clone(),
                            path: paths.get(package).cloned().unwrap_or_default(),
                        });
                        return Ok(());
                    }
                }
            }
        }

        for (package, a) in &active {
            if selected.contains_key(package) {
                continue;
            }
            let has_candidate = self.catalog[package].releases.iter().any(|r| {
                a.constraints.iter().all(|c| matches(*c, r.version))
            });
            if !has_candidate {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical_constraints(&a.constraints),
                    evidence: make_evidence(a),
                });
                return Ok(());
            }
        }

        let next = active.keys().find(|p| !selected.contains_key(*p)).cloned();
        let Some(package) = next else {
            let solution: Vec<(String, Version)> =
                selected.iter().map(|(p, v)| (p.clone(), *v)).collect();
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_solution)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        compare_vectors(&solution, best_solution, &self.catalog) == Ordering::Greater
                    }
                }
            };
            if replace {
                self.best = Some((cost, solution));
            }
            return Ok(());
        };

        let a = &active[&package];
        let mut candidates: Vec<Version> = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| a.constraints.iter().all(|c| matches(*c, r.version)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(locked) = self.locks.get(&package).copied() {
            if let Some(i) = candidates.iter().position(|v| *v == locked) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let result = self.visit(selected);
            selected.remove(&package);
            result?;
        }
        Ok(())
    }
}

use std::cmp::Ordering;

fn compare_vectors(
    a: &[(String, Version)],
    b: &[(String, Version)],
    catalog: &BTreeMap<String, &Package>,
) -> Ordering {
    for name in catalog.keys() {
        let av = a.iter().find(|(p, _)| p == name).map(|(_, v)| *v);
        let bv = b.iter().find(|(p, _)| p == name).map(|(_, v)| *v);
        let ordering = match (av, bv) {
            (None, None) => Ordering::Equal,
            (None, Some(_)) => Ordering::Less,
            (Some(_), None) => Ordering::Greater,
            (Some(x), Some(y)) => x.cmp(&y),
        };
        if ordering != Ordering::Equal {
            return ordering;
        }
    }
    Ordering::Equal
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let locks_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: catalog_map,
        locks: locks_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;

    let Some((_, solution)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names: BTreeSet<String> = locks.iter().map(|l| l.package.clone()).collect();
    all_names.extend(selected.iter().map(|s| s.package.clone()));

    let final_selected: BTreeMap<String, Version> = selected
        .iter()
        .map(|s| (s.package.clone(), s.version))
        .collect();
    let (final_active, _) = compute_active(roots, &final_selected, &catalog_map, target);

    let mut changes = Vec::new();
    for package in all_names {
        let before = locks_map.get(&package).copied();
        let after = final_selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active.get(&package).is_some_and(|a| {
                    !a.constraints.iter().all(|c| matches(*c, old))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

#[cfg(test)]
mod tests {
    use super::*;

    fn v(major: u32, minor: u32, patch: u32) -> Version {
        Version { major, minor, patch }
    }
    fn x(n: u32) -> Version {
        v(n, 0, 0)
    }
    fn e(n: u32) -> Constraint {
        Constraint::Exact(x(n))
    }
    fn ge(n: u32) -> Constraint {
        Constraint::AtLeast(x(n))
    }
    fn q(p: &str, c: Constraint) -> Requirement {
        Requirement {
            package: p.into(),
            constraint: c,
            features: vec![],
        }
    }
    fn d(p: &str, c: Constraint) -> Dependency {
        Dependency {
            package: p.into(),
            constraint: c,
            features: vec![],
            gate: None,
            platform: Platform::Any,
        }
    }
    fn r(n: u32, deps: Vec<Dependency>) -> Release {
        Release {
            version: x(n),
            deps,
            features: vec![],
        }
    }
    fn feat(name: &str, enables: &[&str]) -> Feature {
        Feature {
            name: name.into(),
            enables: enables.iter().map(|s| s.to_string()).collect(),
        }
    }
    fn ask(p: &str, c: Constraint, fs: &[&str]) -> Requirement {
        let mut t = q(p, c);
        t.features = fs.iter().map(|s| s.to_string()).collect();
        t
    }
    fn p(name: &str, releases: Vec<Release>) -> Package {
        Package {
            name: name.into(),
            releases,
        }
    }
    fn sel(p: &str, n: u32) -> Selection {
        Selection {
            package: p.into(),
            version: x(n),
        }
    }
    fn change(p: &str, b: Option<u32>, a: Option<u32>, reason: ChangeReason) -> Change {
        Change {
            package: p.into(),
            before: b.map(x),
            after: a.map(x),
            reason,
        }
    }

    #[test]
    fn public_transitive_selection() {
        let c = vec![
            p("engine", vec![r(1, vec![d("util", ge(1))])]),
            p("util", vec![r(1, vec![])]),
        ];
        let out = resolve(&c, &[q("engine", e(1))], &[], Target::Linux, 100).unwrap();
        assert_eq!(out.selected, vec![sel("engine", 1), sel("util", 1)]);
    }

    #[test]
    fn public_ranges() {
        let c = vec![p("codec", vec![r(3, vec![]), r(1, vec![]), r(2, vec![])])];
        let out = resolve(
            &c,
            &[q("codec", ge(1)), q("codec", Constraint::Before(x(3)))],
            &[],
            Target::Linux,
            100,
        )
        .unwrap();
        assert_eq!(out.selected, vec![sel("codec", 2)]);
    }

    fn public_error_format() {

        assert_eq!(
            error_text(&error),
            "invalid-name|site=root%5B0%5D|name=Bad%20Name"
        );
    }

    fn public_direct_feature() {
        let mut release = r(1, vec![]);
        release.features.push(feat("extras", &[]));
        let mut dep = d("plugin", e(1));
        dep.gate = Some("extras".into());
        release.deps.push(dep);
        let c = vec![p("app", vec![release]), p("plugin", vec![r(1, vec![])])];
        let out = resolve(
            &c,
            &[ask("app", e(1), &["extras"])],
            &[],
            Target::Linux,
            10,
        )
        .unwrap();
        assert_eq!(out.selected, vec![sel("app", 1), sel("plugin", 1)]);
        assert_eq!(
            out.changes,
            vec![
                change("app", None, Some(1), ChangeReason::Added),
                change("plugin", None, Some(1), ChangeReason::Added)
            ]
        );
    }
