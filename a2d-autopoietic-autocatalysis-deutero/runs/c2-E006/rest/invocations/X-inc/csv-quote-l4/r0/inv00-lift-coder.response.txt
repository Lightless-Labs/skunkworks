#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|cell| cell == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|cell| cell == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    if pos == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if !in_quotes && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote as char);
                    pos += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    pos += 1;
                }
            } else {
                let start = pos;
                pos += 1;
                while pos < bytes.len()
                    && bytes[pos] != quote
                    && bytes[pos] != 0
                {
                    pos += 1;
                }
                field.push_str(std::str::from_utf8(&bytes[start..pos]).unwrap());
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                pos += 1;
            } else if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                field_started = false;
                after_quote = false;
                if b == b'\r' {
                    pos += 2;
                } else {
                    pos += 1;
                }
            } else {
                return Err(CsvError::TrailingByte {
                    record: record_index,
                    byte: pos,
                });
            }
            continue;
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            pos += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            field_started = false;
            if b == b'\r' {
                pos += 2;
            } else {
                pos += 1;
            }
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
            in_quotes = true;
            field_started = true;
            quote_start = pos;
            pos += 1;
        } else {
            let start = pos;
            while pos < bytes.len()
                && bytes[pos] != delimiter
                && bytes[pos] != quote
                && bytes[pos] != b'\n'
                && bytes[pos] != b'\r'
                && bytes[pos] != 0
            {
                pos += 1;
            }
            field.push_str(std::str::from_utf8(&bytes[start..pos]).unwrap());
            field_started = true;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if after_quote || field_started || !record.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    let mut iter = records.into_iter();
    let Some(header) = iter.next() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen = std::collections::HashMap::<&str, usize>::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name, second);
        }
    }

    let mut rows = Vec::new();
    for (record, row) in iter.enumerate() {
        let input_record = record + 1;
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: input_record,
                expected: header.len(),
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn valid_integer_spelling(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let mut i = usize::from(b[0] == b'-');
    if i == b.len() {
        return false;
    }
    if b[i] == b'0' {
        return i + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[i]) {
        return false;
    }
    i += 1;
    b[i..].iter().all(u8::is_ascii_digit)
}

fn valid_decimal_spelling(s: &str) -> bool {
    let b = s.as_bytes();
    let mut i = usize::from(b.first() == Some(&b'-'));
    if i >= b.len() {
        return false;
    }
    let integral_start = i;
    if b[i] == b'0' {
        i += 1;
    } else if (b'1'..=b'9').contains(&b[i]) {
        i += 1;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
    } else {
        return false;
    }
    if i >= b.len() || b[i] != b'.' {
        return false;
    }
    let integral = &b[integral_start..i];
    i += 1;
    if b.len() - i != 2 || !b[i..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    !(b.first() == Some(&b'-') && integral == b"0" && b[i..] == *b"00")
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => {
            if !valid_integer_spelling(cell) {
                return Err(CellErrorKind::InvalidInteger);
            }
            cell.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !valid_decimal_spelling(cell) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = cell.starts_with('-');
            let body = if negative { &cell[1..] } else { cell };
            let (whole, fraction) = body.split_once('.').unwrap();
            let magnitude_text = format!("{whole}{fraction}");
            let magnitude = magnitude_text
                .parse::<u64>()
                .map_err(|_| CellErrorKind::DecimalOverflow)?;
            if negative {
                if magnitude > (i64::MAX as u64) + 1 {
                    return Err(CellErrorKind::DecimalOverflow);
                }
                if magnitude == (i64::MAX as u64) + 1 {
                    Ok(Value::Decimal(i64::MIN))
                } else {
                    Ok(Value::Decimal(-(magnitude as i64)))
                }
            } else if magnitude > i64::MAX as u64 {
                Err(CellErrorKind::DecimalOverflow)
            } else {
                Ok(Value::Decimal(magnitude as i64))
            }
        }
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut out_rows = Vec::new();
    let mut errors = Vec::new();

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut failed_in_row = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed_in_row {
                out.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let Some(column_index) = resolved else {
                out.push(None);
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                failed_in_row = true;
                if policy == ErrorPolicy::Stop {
                    out.resize(columns.len(), None);
                    out_rows.push(out);
                    return TypedResult { rows: out_rows, errors };
                }
                continue;
            };

            let Some(cell) = row.get(column_index) else {
                out.push(None);
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::MissingCell,
                });
                failed_in_row = true;
                if policy == ErrorPolicy::Stop {
                    out.resize(columns.len(), None);
                    out_rows.push(out);
                    return TypedResult { rows: out_rows, errors };
                }
                continue;
            };

            match evaluate_cell(cell, spec) {
                Ok(value) => out.push(Some(value)),
                Err(kind) => {
                    out.push(None);
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    failed_in_row = true;
                    if policy == ErrorPolicy::Stop {
                        out.resize(columns.len(), None);
                        out_rows.push(out);
                        return TypedResult { rows: out_rows, errors };
                    }
                }
            }

            if policy == ErrorPolicy::FirstPerRow && failed_in_row {
                while out.len() < columns.len() {
                    out.push(None);
                }
                break;
            }
        }
        out_rows.push(out);
    }

    TypedResult {
        rows: out_rows,
        errors,
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn duplicate_header_error(header: &[String]) -> Option<WriteError> {
    let mut seen = std::collections::HashMap::<&str, usize>::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name, second);
    }
    None
}

fn quote_field(field: &str, delimiter: u8, quote: u8, first_field: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let starts_or_ends_space_tab =
        bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
            || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let starts_bom = first_field && field.starts_with('\u{feff}');
    let should_quote = field.is_empty() && sole
        || starts_or_ends_space_tab
        || starts_bom
        || bytes.iter().any(|b| {
            *b == delimiter || *b == quote || *b == b'\r' || *b == b'\n'
        });

    if !should_quote {
        return field.to_string();
    }

    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

fn row_contains_nul(row: &[String], record: usize) -> Option<WriteError> {
    for (column, field) in row.iter().enumerate() {
        if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
            return Some(WriteError::Nul {
                record,
                column,
                byte,
            });
        }
    }
    None
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        match left.as_bytes().cmp(right.as_bytes()) {
            std::cmp::Ordering::Equal => {}
            other => return other,
        }
    }
    a.len().cmp(&b.len())
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    if let Some(header) = &table.header {
        if let Some(error) = row_contains_nul(header, 0) {
            return Err(error);
        }
    }
    let first_data_record = usize::from(table.header.is_some());
    for (index, row) in table.rows.iter().enumerate() {
        if let Some(error) = row_contains_nul(row, first_data_record + index) {
            return Err(error);
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    if let Some(header) = &table.header {
        if header.is_empty() {
            return Err(WriteError::ZeroColumns { record: 0 });
        }
    }
    for (index, row) in table.rows.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: first_data_record + index,
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(error) = duplicate_header_error(header) {
                    return Err(error);
                }
            }
        }
    }

    let mut output_header = table.header.clone();
    let mut output_rows = table.rows.clone();

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            let mut chosen = std::collections::HashMap::<&str, usize>::new();
            for (index, name) in header.iter().enumerate() {
                match table.duplicates {
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        chosen.entry(name).or_insert(index);
                    }
                    DuplicatePolicy::KeepLast => {
                        chosen.insert(name, index);
                    }
                }
            }
            let mut retained: Vec<(String, usize)> = chosen
                .into_iter()
                .map(|(name, index)| (name.to_string(), index))
                .collect();
            retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

            output_header = Some(retained.iter().map(|(name, _)| name.clone()).collect());
            let indices: Vec<usize> = retained.iter().map(|(_, index)| *index).collect();
            output_rows = table
                .rows
                .iter()
                .map(|row| indices.iter().map(|index| row[*index].clone()).collect())
                .collect();
        }
        output_rows.sort_by(|a, b| compare_rows(a, b));
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    if let Some(header) = output_header {
        records.push(header);
    }
    records.extend(output_rows);

    if records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let terminator = if canonical || config.line_ending == LineEnding::Lf {
        "\n"
    } else {
        "\r\n"
    };
    let final_terminator = canonical || config.final_terminator;

    let mut output = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            output.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                row.len() == 1,
            ));
        }
    }
    if final_terminator {
        output.push_str(terminator);
    }

    Ok(output)
}
