#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|&c| matches(version, c))
}

fn encode(value: &str) -> String {
    let mut out = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => {
            let mut reqs = requirements.clone();
            canonicalize(&mut reqs);
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                reqs.iter()
                    .map(|c| constraint_text(*c))
                    .collect::<Vec<_>>()
                    .join(",")
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => {
            let mut reqs = requirements.clone();
            canonicalize(&mut reqs);
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                reqs.iter()
                    .map(|c| constraint_text(*c))
                    .collect::<Vec<_>>()
                    .join(",")
            )
        }
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn lock_version(&self, name: &str) -> Option<Version> {
        self.locks
            .iter()
            .find(|lock| lock.package == name)
            .map(|lock| lock.version)
    }

    fn inspect(
        &mut self,
        assigned: &mut std::collections::BTreeMap<String, Version>,
        roots: &[Requirement],
    ) -> Result<(), ResolveError> {
        let mut constraints: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();
        for root in roots {
            constraints
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in assigned.iter() {
            if let Some(package) = self.package(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        constraints
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for reqs in constraints.values_mut() {
            canonicalize(reqs);
        }

        if let Some(name) = constraints.keys().find(|name| self.package(name).is_none()) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assigned.iter() {
            if let Some(reqs) = constraints.get(name) {
                if !satisfies(*version, reqs) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: reqs.clone(),
                    });
                }
            }
        }

        let mut candidates_by_name = std::collections::BTreeMap::new();
        for (name, reqs) in &constraints {
            if assigned.contains_key(name) {
                continue;
            }
            let package = self.package(name).expect("missing packages checked");
            let mut candidates: Vec<Version> = package
                .releases
                .iter()
                .map(|r| r.version)
                .filter(|v| satisfies(*v, reqs))
                .collect();
            candidates.sort_by(|a, b| b.cmp(a));
            if candidates.is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: reqs.clone(),
                });
            }
            if let Some(locked) = self.lock_version(name) {
                if let Some(index) = candidates.iter().position(|v| *v == locked) {
                    candidates.remove(index);
                    candidates.insert(0, locked);
                }
            }
            candidates_by_name.insert(name.clone(), candidates);
        }

        let Some((name, candidates)) = candidates_by_name.into_iter().next() else {
            let selected = assigned
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            self.record_solution(selected);
            return Ok(());
        };

        let mut first_branch_error = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), version);
            let result = self.inspect(assigned, roots);
            assigned.remove(&name);

            match result {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => return Err(error),
                Err(error) => {
                    if first_branch_error.is_none() {
                        first_branch_error = Some(error);
                    }
                }
            }
        }

        if let Some(error) = first_branch_error {
            Err(error)
        } else {
            unreachable!()
        }
    }

    fn record_solution(&mut self, selected: Vec<Selection>) {
        let selected_map: std::collections::BTreeMap<&str, Version> = selected
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect();
        let cost = self
            .locks
            .iter()
            .filter(|lock| selected_map.get(lock.package.as_str()) != Some(&lock.version))
            .count();

        let better = match &self.best {
            None => true,
            Some((best_cost, best_selection)) if cost < *best_cost => true,
            Some((best_cost, best_selection)) if cost == *best_cost => {
                let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
                names.sort_unstable();
                let current: std::collections::BTreeMap<&str, Version> = selected
                    .iter()
                    .map(|s| (s.package.as_str(), s.version))
                    .collect();
                let best: std::collections::BTreeMap<&str, Version> = best_selection
                    .iter()
                    .map(|s| (s.package.as_str(), s.version))
                    .collect();
                names
                    .into_iter()
                    .find_map(|name| {
                        let a = current.get(name);
                        let b = best.get(name);
                        if a == b {
                            None
                        } else {
                            Some(match (a, b) {
                                (Some(a), Some(b)) => a > b,
                                (Some(_), None) => true,
                                (None, Some(_)) => false,
                                (None, None) => false,
                            })
                        }
                    })
                    .unwrap_or(false)
            }
            _ => false,
        };
        if better {
            self.best = Some((cost, selected));
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut assigned = std::collections::BTreeMap::new();
    match search.inspect(&mut assigned, roots) {
        Err(error @ ResolveError::Limit { .. }) => return Err(error),
        Err(error) => search.first_error = Some(error),
        Ok(()) => {}
    }

    let Some((_, mut selected)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };
    selected.sort_by(|a, b| a.package.cmp(&b.package));

    let selected_map: std::collections::BTreeMap<&str, Version> = selected
        .iter()
        .map(|s| (s.package.as_str(), s.version))
        .collect();

    let mut final_constraints: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        final_constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for selection in &selected {
        if let Some(package) = catalog.iter().find(|p| p.name == selection.package) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|r| r.version == selection.version)
            {
                for dep in &release.deps {
                    final_constraints
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for reqs in final_constraints.values_mut() {
        canonicalize(reqs);
    }

    let mut all_names = std::collections::BTreeSet::new();
    for lock in locks {
        all_names.insert(lock.package.as_str());
    }
    for selection in &selected {
        all_names.insert(selection.package.as_str());
    }

    let mut changes = Vec::new();
    for name in all_names {
        let before = locks
            .iter()
            .find(|lock| lock.package == name)
            .map(|lock| lock.version);
        let after = selected_map.get(name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == name)
                    .map(|p| p.releases.iter().any(|r| r.version == locked))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if !satisfies(
                    locked,
                    final_constraints.get(name).map(Vec::as_slice).unwrap_or(&[]),
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name.to_string(),
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
