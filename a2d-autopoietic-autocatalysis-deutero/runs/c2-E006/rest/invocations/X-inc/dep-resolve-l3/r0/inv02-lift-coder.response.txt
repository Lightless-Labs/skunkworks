#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut out = constraints.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphabetic()
            || byte.is_ascii_digit()
            || *byte == b'.'
            || *byte == b'_'
            || *byte == b'-'
        {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|c| matches_constraint(version, *c))
}

struct Search<'a> {
    packages: &'a [Package],
    package_indices: Vec<usize>,
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_failure: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn active_constraints(
        &self,
        roots: &[Requirement],
        chosen: &[(usize, usize)],
    ) -> Vec<(String, Vec<Constraint>)> {
        let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

        fn add(active: &mut Vec<(String, Vec<Constraint>)>, req: &Requirement) {
            if let Some((_, constraints)) = active.iter_mut().find(|(name, _)| name == &req.package)
            {
                constraints.push(req.constraint);
            } else {
                active.push((req.package.clone(), vec![req.constraint]));
            }
        }

        for req in roots {
            add(&mut active, req);
        }
        for &(package_index, release_index) in chosen {
            for req in &self.packages[package_index].releases[release_index].deps {
                add(&mut active, req);
            }
        }
        for (_, constraints) in &mut active {
            *constraints = canonical_constraints(constraints);
        }
        active.sort_by(|a, b| a.0.cmp(&b.0));
        active
    }

    fn package_index(&self, name: &str) -> Option<usize> {
        self.package_indices
            .iter()
            .copied()
            .find(|&i| self.packages[i].name == name)
    }

    fn selected_release<'b>(
        &'b self,
        name: &str,
        chosen: &'b [(usize, usize)],
    ) -> Option<(usize, usize)> {
        chosen
            .iter()
            .copied()
            .find(|&(pi, _)| self.packages[pi].name == name)
    }

    fn inspect(
        &mut self,
        roots: &[Requirement],
        chosen: &mut Vec<(usize, usize)>,
    ) -> Result<(), ResolveError> {
        let active = self.active_constraints(roots, chosen);

        for (name, _) in &active {
            if self.package_index(name).is_none() {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                });
            }
        }

        for (name, constraints) in &active {
            if let Some((pi, ri)) = self.selected_release(name, chosen) {
                let version = self.packages[pi].releases[ri].version;
                if !matches_all(version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if self.selected_release(name, chosen).is_some() {
                continue;
            }
            let pi = self.package_index(name).unwrap();
            if !self.packages[pi]
                .releases
                .iter()
                .any(|r| matches_all(r.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active
            .iter()
            .find(|(name, _)| self.selected_release(name, chosen).is_none())
            .cloned();

        let Some((name, constraints)) = next else {
            let selected: Vec<Selection> = chosen
                .iter()
                .map(|&(pi, ri)| Selection {
                    package: self.packages[pi].name.clone(),
                    version: self.packages[pi].releases[ri].version,
                })
                .collect();

            let mut cost = 0usize;
            for lock in self.locks {
                let retained = selected
                    .iter()
                    .any(|s| s.package == lock.package && s.version == lock.version);
                if !retained {
                    cost += 1;
                }
            }

            let replace = match &self.best {
                None => true,
                Some((best_cost, best_selected)) => {
                    if cost < *best_cost {
                        true
                    } else if cost > *best_cost {
                        false
                    } else {
                        self.compare_vectors(&selected, best_selected).is_gt()
                    }
                }
            };
            if replace {
                self.best = Some((cost, selected));
            }
            return Ok(());
        };

        let pi = self.package_index(&name).unwrap();
        let mut candidates: Vec<usize> = self.packages[pi]
            .releases
            .iter()
            .enumerate()
            .filter(|(_, release)| matches_all(release.version, &constraints))
            .map(|(ri, _)| ri)
            .collect();
        candidates.sort_by(|&a, &b| {
            self.packages[pi].releases[b]
                .version
                .cmp(&self.packages[pi].releases[a].version)
        });

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(pos) = candidates
                .iter()
                .position(|&ri| self.packages[pi].releases[ri].version == lock.version)
            {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        let mut first_error = None;
        for ri in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            chosen.push((pi, ri));
            let result = self.inspect(roots, chosen);
            chosen.pop();

            match result {
                Ok(()) => {}
                Err(ResolveError::Limit { .. }) => return result,
                Err(error) => {
                    if first_error.is_none() {
                        first_error = Some(error);
                    }
                }
            }
        }

        if let Some(error) = first_error {
            if self.first_failure.is_none() {
                self.first_failure = Some(error.clone());
            }
            Err(error)
        } else {
            Err(ResolveError::NoMatch {
                package: name,
                requirements: constraints,
            })
        }
    }

    fn compare_vectors(&self, a: &[Selection], b: &[Selection]) -> std::cmp::Ordering {
        for &pi in &self.package_indices {
            let name = &self.packages[pi].name;
            let av = a.iter().find(|s| &s.package == name).map(|s| s.version);
            let bv = b.iter().find(|s| &s.package == name).map(|s| s.version);
            match (av, bv) {
                (None, None) => {}
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (Some(av), Some(bv)) => match av.cmp(&bv) {
                    std::cmp::Ordering::Equal => {}
                    other => return other,
                },
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = std::collections::HashSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::HashSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::HashSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut package_indices: Vec<usize> = (0..catalog.len()).collect();
    package_indices.sort_by(|&a, &b| catalog[a].name.cmp(&catalog[b].name));

    let mut search = Search {
        packages: catalog,
        package_indices,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_failure: None,
    };

    let result = search.inspect(roots, &mut Vec::new());
    if let Err(ResolveError::Limit { attempts }) = result {
        return Err(ResolveError::Limit { attempts });
    }

    let Some((_, selected)) = search.best else {
        return Err(search.first_failure.unwrap_or_else(|| match result {
            Err(error) => error,
            Ok(()) => ResolveError::NoMatch {
                package: String::new(),
                requirements: Vec::new(),
            },
        }));
    };

    let final_constraints = search.active_constraints(
        roots,
        &selected
            .iter()
            .map(|selection| {
                let pi = search.package_index(&selection.package).unwrap();
                let ri = catalog[pi]
                    .releases
                    .iter()
                    .position(|release| release.version == selection.version)
                    .unwrap();
                (pi, ri)
            })
            .collect::<Vec<_>>(),
    );

    let mut changes = Vec::new();
    let mut names = std::collections::BTreeSet::new();
    for lock in locks {
        names.insert(lock.package.clone());
    }
    for selection in &selected {
        names.insert(selection.package.clone());
    }

    for name in names {
        let before = locks
            .iter()
            .find(|lock| lock.package == name)
            .map(|lock| lock.version);
        let after = selected
            .iter()
            .find(|selection| selection.package == name)
            .map(|selection| selection.version);
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .package_index(&name)
                    .map(|pi| catalog[pi].releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = final_constraints
                        .iter()
                        .find(|(package, _)| package == &name)
                        .map(|(_, constraints)| constraints.as_slice())
                        .unwrap_or(&[]);
                    if !matches_all(old, constraints) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };

        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
