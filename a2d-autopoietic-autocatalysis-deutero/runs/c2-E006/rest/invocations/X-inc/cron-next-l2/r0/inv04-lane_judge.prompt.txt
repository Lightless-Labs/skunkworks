fn parse_number(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

                field,
                offset: item_start,

            .iter()
            .enumerate()

            return Err(ParseError::InvalidSyntax {
                field,

                        field,
                        offset: item_start,
                    });
                }
                if step_bytes.is_none() {
                    end = start;
                }
            }

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 31],
        vec![false; 12],
        vec![false; 7],
    ];

    for (i, (&field, part)) in FIELDS.iter().zip(parts.iter()).enumerate() {
        parse_field(part, field, &mut selected[i])?;
    }

    Ok(Cron {
        selected,
        day_star: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (minimum, _) = limits(field);
        let values = &self.selected[field_index(field)];
        values
            .iter()
            .enumerate()
            .filter_map(|(i, &allowed)| {
                if allowed {
                    Some(minimum as u8 + i as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            current = advance_minute(current);
            if current.year > 2399 {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let weekday = weekday(at.year, at.month, at.day);
        self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][(at.day - 1) as usize]
            && self.selected[3][(at.month - 1) as usize]
            && self.selected[4][weekday as usize]
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

            (min, max, false)
        };

        let step = if let Some((step_rel, step_bytes)) = step_part {
            parse_number(step_bytes, field, item_start + step_rel, true)?
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        let _ = numeric_base;
        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {

        selected: selected.try_into().ok().unwrap(),

        day_of_week_is_star: pieces[4] == "*",
    })

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let (minimum, maximum) = field_domain(field);
    let item = &bytes[item_start..item_end];

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let base_end = match slash_positions.as_slice() {
        [] => item.len(),
        [slash] => *slash,
        _ => {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    };

    let base = &item[..base_end];
    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let (range_start, range_end, number_spans) = if is_star {
        (minimum, maximum, None)
    } else {
        match hyphens.as_slice() {
            [] if !base.is_empty() => {
                if !base.iter().all(u8::is_ascii_digit) {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                let span = (0, base.len());
                (0, 0, Some((span, None)))
            }
            [hyphen] if *hyphen > 0 && *hyphen + 1 < base.len() => {
                if !base[..*hyphen].iter().all(u8::is_ascii_digit)
                    || !base[*hyphen + 1..].iter().all(u8::is_ascii_digit)
                {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                (0, 0, Some(((0, *hyphen), Some((*hyphen + 1, base.len())))))
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }
    };

    if slash_positions.len() == 1 && slash_positions[0] + 1 == item.len() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let mut start_value = range_start;
    let mut end_value = range_end;

    if let Some((start_span, end_span)) = number_spans {
        let value = parse_number(
            bytes,
            item_start + start_span.0,
            item_start + start_span.1,
            field,
        )?;
        if value < minimum || value > maximum {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_start + start_span.0,
            });
        }
        start_value = value;
        end_value = value;

        if let Some((span_start, span_end)) = end_span {
            let value = parse_number(
                bytes,
                item_start + span_start,
                item_start + span_end,
                field,
            )?;
            if value < minimum || value > maximum {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + span_start,
                });
            }
            if start_value > value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + span_start,
                });
            }
            end_value = value;
        } else if slash_positions.len() == 1 {
            end_value = maximum;
        }
    }

    let step = if let Some(&slash) = slash_positions.first() {
        let step_start = item_start + slash + 1;
        let step_end = item_end;
        if step_start == step_end
            || !bytes[step_start..step_end]
                .iter()
                .all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let value = parse_number(bytes, step_start, step_end, field)?;
        let width = maximum - minimum + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_start,
            });
        }
        value
    } else {
        1
    };

    let mut value = start_value;
    while value <= end_value {
        selected[(value - minimum) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_domain(field);
    let mut selected = vec![false; (maximum - minimum + 1) as usize];
    let mut start = 0;

    for end in 0..=bytes.len() {
        if end == bytes.len() || bytes[end] == b',' {
            if start == end {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: start,
                });
            }
            parse_item(bytes, start, end, field, &mut selected)?;
            start = end + 1;
        }
    }

    Ok(selected)
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if step {
        let (_, max) = domain(field);
        if value == 0 || value > max {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }

    Ok(value)
}

fn parse_field(input: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0;
    for item in input.split(',') {
        let item_bytes = item.as_bytes();
        if item_bytes.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base, step_part) = if let Some(&slash) = slash_positions.first() {
            (&item_bytes[..slash], Some((slash + 1, &item_bytes[slash + 1..])))
        } else {
            (item_bytes, None)
        };

        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let syntax_valid = if base == b"*" {
            hyphens.is_empty()
        } else if hyphens.is_empty() {
            !base.is_empty() && base.iter().all(u8::is_ascii_digit)
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            h > 0
                && h + 1 < base.len()
                && base[..h].iter().all(u8::is_ascii_digit)
                && base[h + 1..].iter().all(u8::is_ascii_digit)
        } else {
            false
        };

        if !syntax_valid || base.is_empty() || step_part.is_some_and(|(_, s)| s.is_empty()) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (start, end) = if base == b"*" {
            (min, max)
        } else if hyphens.is_empty() {
            let value = parse_number(base, field, item_start, false)?;
            (value, value)
        } else {
            let h = hyphens[0];
            let start = parse_number(&base[..h], field, item_start, false)?;
            let end = parse_number(&base[h + 1..], field, item_start + h + 1, false)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + h + 1,
                });
            }
            (start, end)
        };

        let step = if let Some((step_offset, step_bytes)) = step_part {
            Some(parse_number(
                step_bytes,
                field,
                item_start + step_offset,
                true,
            )?)
        } else {
            None
        };

        let (range_start, range_end) = if step.is_some() && hyphens.is_empty() && base != b"*" {
            (start, max)
        } else {
            (start, end)
        };

        if let Some(step) = step {
            let mut value = range_start;
            while value <= range_end {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        } else {
            for value in range_start..=range_end {
                selected[value as usize] = true;
            }
        }

        item_start += item.len() + 1;
    }

    Ok(selected)
}

            .enumerate()
            .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return true;
    }

    false
}
