#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let (minimum, maximum) = field_domain(field);
    let item = &bytes[item_start..item_end];

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let base_end = match slash_positions.as_slice() {
        [] => item.len(),
        [slash] => *slash,
        _ => {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    };

    let base = &item[..base_end];
    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let (range_start, range_end, number_spans) = if is_star {
        (minimum, maximum, None)
    } else {
        match hyphens.as_slice() {
            [] if !base.is_empty() => {
                if !base.iter().all(u8::is_ascii_digit) {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                let span = (0, base.len());
                (0, 0, Some((span, None)))
            }
            [hyphen] if *hyphen > 0 && *hyphen + 1 < base.len() => {
                if !base[..*hyphen].iter().all(u8::is_ascii_digit)
                    || !base[*hyphen + 1..].iter().all(u8::is_ascii_digit)
                {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                (0, 0, Some(((0, *hyphen), Some((*hyphen + 1, base.len())))))
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }
    };

    if slash_positions.len() == 1 && slash_positions[0] + 1 == item.len() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let mut start_value = range_start;
    let mut end_value = range_end;

    if let Some((start_span, end_span)) = number_spans {
        let value = parse_number(
            bytes,
            item_start + start_span.0,
            item_start + start_span.1,
            field,
        )?;
        if value < minimum || value > maximum {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_start + start_span.0,
            });
        }
        start_value = value;
        end_value = value;

        if let Some((span_start, span_end)) = end_span {
            let value = parse_number(
                bytes,
                item_start + span_start,
                item_start + span_end,
                field,
            )?;
            if value < minimum || value > maximum {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + span_start,
                });
            }
            if start_value > value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + span_start,
                });
            }
            end_value = value;
        } else if slash_positions.len() == 1 {
            end_value = maximum;
        }
    }

    let step = if let Some(&slash) = slash_positions.first() {
        let step_start = item_start + slash + 1;
        let step_end = item_end;
        if step_start == step_end
            || !bytes[step_start..step_end]
                .iter()
                .all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let value = parse_number(bytes, step_start, step_end, field)?;
        let width = maximum - minimum + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_start,
            });
        }
        value
    } else {
        1
    };

    let mut value = start_value;
    while value <= end_value {
        selected[(value - minimum) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_domain(field);
    let mut selected = vec![false; (maximum - minimum + 1) as usize];
    let mut start = 0;

    for end in 0..=bytes.len() {
        if end == bytes.len() || bytes[end] == b',' {
            if start == end {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: start,
                });
            }
            parse_item(bytes, start, end, field, &mut selected)?;
            start = end + 1;
        }
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        selected[i] = parse_field(FIELDS[i], parts[i])?;
    }

    Ok(Cron {
        selected,
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (minimum, _) = field_domain(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(minimum as u8 + i as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;

        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][(at.day - 1) as usize]
            && self.selected[3][(at.month - 1) as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            current = match advance_minute(current) {
                Some(next) => next,
                None => return Ok(None),
            };
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn advance_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        return Some(CivilDateTime {
            minute: at.minute + 1,
            ..at
        });
    }

    let mut next = CivilDateTime {
        minute: 0,
        ..at
    };
    if next.hour < 23 {
        next.hour += 1;
        return Some(next);
    }

    next.hour = 0;
    if next.day < days_in_month(next.year, next.month) {
        next.day += 1;
        return Some(next);
    }

    next.day = 1;
    if next.month < 12 {
        next.month += 1;
        return Some(next);
    }

    if next.year < 2399 {
        next.month = 1;
        next.year += 1;
        return Some(next);
    }

    None
}
