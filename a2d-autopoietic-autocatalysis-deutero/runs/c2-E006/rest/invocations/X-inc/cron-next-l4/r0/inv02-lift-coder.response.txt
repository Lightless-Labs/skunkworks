#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_day: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = field_index(field);
        self.selected[i]
            .iter()
            .enumerate()
            .filter_map(|(n, yes)| if *yes { Some(n as u8 + field_min(field)) } else { None })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(&at)?;
        Ok(self.matches_valid(&at))
    }

    fn matches_valid(&self, at: &CivilDateTime) -> bool {
        let minute = self.selected[0][at.minute as usize];
        let hour = self.selected[1][at.hour as usize];
        let month = self.selected[3][at.month as usize - 1];
        if !(minute && hour && month) {
            return false;
        }
        let dom = self.selected[2][at.day as usize - 1];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];
        match self.unrestricted_day {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(&after)?;
        let mut day = after;
        loop {
            let (hours, minutes) = self.allowed_times();
            if self.selected[3][day.month as usize - 1]
                && self.date_matches(&day)
            {
                for &hour in &hours {
                    for &minute in &minutes {
                        let candidate = CivilDateTime { hour, minute, ..day };
                        if candidate > after && self.matches_valid(&candidate) {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            match increment_date(&mut day) {
                Some(()) => {}
                None => return Ok(None),
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(&before)?;
        let mut day = before;
        loop {
            let (hours, minutes) = self.allowed_times();
            if self.selected[3][day.month as usize - 1] && self.date_matches(&day) {
                for &hour in hours.iter().rev() {
                    for &minute in minutes.iter().rev() {
                        let candidate = CivilDateTime { hour, minute, ..day };
                        if candidate < before && self.matches_valid(&candidate) {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            match decrement_date(&mut day) {
                Some(()) => {}
                None => return Ok(None),
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(&start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(&end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut cursor = start;
        loop {
            match self.next_fire(cursor.clone()) {
                Ok(Some(fire)) if fire < end => {
                    if fires.len() == cap {
                        return Ok(FireWindow { fires, truncated: true });
                    }
                    fires.push(fire.clone());
                    cursor = fire;
                }
                _ => return Ok(FireWindow { fires, truncated: false }),
            }
        }
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::new();
        for i in 0..5 {
            if i == 2 && self.unrestricted_day[0] {
                fields.push("*".to_string());
                continue;
            }
            if i == 4 && self.unrestricted_day[1] {
                fields.push("*".to_string());
                continue;
            }
            let min = field_min_by_index(i) as usize;
            let max = field_max_by_index(i) as usize;
            let selected: Vec<usize> = self.selected[i]
                .iter()
                .enumerate()
                .filter_map(|(n, b)| if *b { Some(n + min) } else { None })
                .collect();
            if selected.len() == max - min + 1 {
                if (i == 2 || i == 4) {
                    fields.push("*/1".to_string());
                } else {
                    fields.push("*".to_string());
                }
            } else {
                fields.push(selected.iter().map(|n| n.to_string()).collect::<Vec<_>>().join(","));
            }
        }
        fields.join(" ")
    }

    fn allowed_times(&self) -> (Vec<u8>, Vec<u8>) {
        (self.allowed(Field::Hour), self.allowed(Field::Minute))
    }

    fn date_matches(&self, at: &CivilDateTime) -> bool {
        let dom = self.selected[2][at.day as usize - 1];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];
        match self.unrestricted_day {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) { dom && dow } else { dom || dow }
            }
        }
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_min(field: Field) -> u8 {
    match field {
        Field::DayOfMonth | Field::Month => 1,
        _ => 0,
    }
}

fn field_max(field: Field) -> u8 {
    match field {
        Field::Minute => 59,
        Field::Hour => 23,
        Field::DayOfMonth => 31,
        Field::Month => 12,
        Field::DayOfWeek => 6,
    }
}

fn field_min_by_index(i: usize) -> u8 {
    field_min([Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek][i])
}

fn field_max_by_index(i: usize) -> u8 {
    field_max([Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek][i])
}

fn parse_name(field: Field, s: &str) -> Option<u32> {
    let names: &[(&str, u32)] = match field {
        Field::Month => &[
            ("JAN", 1), ("FEB", 2), ("MAR", 3), ("APR", 4), ("MAY", 5), ("JUN", 6),
            ("JUL", 7), ("AUG", 8), ("SEP", 9), ("OCT", 10), ("NOV", 11), ("DEC", 12),
        ],
        Field::DayOfWeek => &[
            ("SUN", 0), ("MON", 1), ("TUE", 2), ("WED", 3), ("THU", 4), ("FRI", 5), ("SAT", 6),
        ],
        _ => return None,
    };
    names.iter().find_map(|(name, value)| if s.eq_ignore_ascii_case(name) { Some(*value) } else { None })
}

fn parse_num(field: Field, s: &str, offset: usize, step: bool) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n.checked_mul(10).and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if step {
        let width = (field_max(field) - field_min(field) + 1) as u32;
        if n < 1 || n > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else if n < field_min(field) as u32 || n > field_max(field) as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(n)
}

fn atom_value(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    if s.as_bytes().iter().all(u8::is_ascii_digit) {
        parse_num(field, s, offset, false)
    } else if let Some(n) = parse_name(field, s) {
        Ok(n)
    } else {
        Err(ParseError::InvalidSyntax { field, offset })
    }
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit() || b.is_ascii_alphabetic() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut selected = vec![false; (field_max(field) - field_min(field) + 1) as usize];
    for (item_offset, item) in split_with_offsets(text, b',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: item_offset });
        }
        let slash_parts: Vec<&str> = item.split('/').collect();
        let base = slash_parts[0];
        let step_text = if slash_parts.len() == 2 { Some(slash_parts[1]) } else { None };
        if slash_parts.len() > 2 || base.is_empty() || step_text == Some("") {
            return Err(ParseError::InvalidSyntax { field, offset: item_offset });
        }

        let (start, end, base_kind, end_offset) = if base == "*" {
            (field_min(field) as u32, field_max(field) as u32, 0, item_offset)
        } else {
            let hyphens: Vec<usize> = base.bytes().enumerate().filter_map(|(i,b)| if b == b'-' { Some(i) } else { None }).collect();
            if hyphens.len() > 1 {
                return Err(ParseError::InvalidSyntax { field, offset: item_offset });
            }
            if let Some(&h) = hyphens.first() {
                let left = &base[..h];
                let right = &base[h+1..];
                if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                    return Err(ParseError::InvalidSyntax { field, offset: item_offset });
                }
                let start = atom_value(field, left, item_offset)?;
                let end_offset = item_offset + h + 1;
                let end = atom_value(field, right, end_offset)?;
                (start, end, 1, end_offset)
            } else {
                let start = atom_value(field, base, item_offset)?;
                (start, if step_text.is_some() { field_max(field) as u32 } else { start }, 2, item_offset)
            }
        };

        if base_kind == 1 && start > end {
            return Err(ParseError::ReversedRange { field, offset: end_offset });
        }
        let step = if let Some(st) = step_text {
            let step_off = item_offset + base.len() + 1;
            parse_num(field, st, step_off, true)?
        } else {
            1
        };
        let actual_end = if base_kind == 2 && step_text.is_some() { field_max(field) as u32 } else { end };
        let mut n = start;
        while n <= actual_end {
            selected[(n - field_min(field) as u32) as usize] = true;
            match n.checked_add(step) {
                Some(next) => n = next,
                None => break,
            }
        }
    }
    Ok(selected)
}

fn split_with_offsets(text: &str, separator: u8) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let mut start = 0;
    for (i, b) in text.bytes().enumerate() {
        if b == separator {
            out.push((start, &text[start..i]));
            start = i + 1;
        }
    }
    out.push((start, &text[start..]));
    out
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let input = if input.as_bytes().first() == Some(&b'@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }
    let kinds = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut selected = [
        Vec::new(), Vec::new(), Vec::new(), Vec::new(), Vec::new(),
    ];
    for i in 0..5 {
        selected[i] = parse_field(kinds[i], fields[i])?;
    }
    Ok(Cron {
        selected,
        unrestricted_day: [fields[2] == "*", fields[4] == "*"],
    })
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: &CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn increment_date(at: &mut CivilDateTime) -> Option<()> {
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
    } else if at.month < 12 {
        at.day = 1;
        at.month += 1;
    } else if at.year < 2399 {
        at.day = 1;
        at.month = 1;
        at.year += 1;
    } else {
        return None;
    }
    at.hour = 0;
    at.minute = 0;
    Some(())
}

fn decrement_date(at: &mut CivilDateTime) -> Option<()> {
    if at.day > 1 {
        at.day -= 1;
    } else if at.month > 1 {
        at.month -= 1;
        at.day = days_in_month(at.year, at.month);
    } else if at.year > 2000 {
        at.year -= 1;
        at.month = 12;
        at.day = 31;
    } else {
        return None;
    }
    at.hour = 0;
    at.minute = 0;
    Some(())
}
