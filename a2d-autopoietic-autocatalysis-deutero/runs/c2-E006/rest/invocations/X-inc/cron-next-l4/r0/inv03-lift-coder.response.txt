#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32, usize) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n: u32 = 0;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(n)
}

fn parse_atom(
    text: &str,
    field: Field,
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if !text.is_empty() && text.bytes().all(|b| b.is_ascii_digit()) {
        return Ok((parse_number(text, field, offset)?, true));
    }
    if let Some(names) = names(field) {
        if text.len() == 3 && text.bytes().all(|b| b.is_ascii_alphabetic()) {
            let upper = text.to_ascii_uppercase();
            if let Some(i) = names.iter().position(|n| *n == upper) {
                return Ok((if field == Field::Month { i as u32 + 1 } else { i as u32 }, false));
            }
        }
    }
    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_field(
    text: &str,
    field: Field,
) -> Result<(Vec<bool>, bool), ParseError> {
    let (min, max, width) = bounds(field);
    for (i, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut selected = vec![false; width];
    for (i, item) in text.split(',').enumerate() {
        let start = item.as_ptr() as usize - text.as_ptr() as usize;
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: start });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        let (base, step_text) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((&item[pos + 1..], pos + 1)))
        } else {
            (item, None)
        };
        if base.is_empty() || step_text.is_some_and(|(s, _)| s.is_empty()) {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }

        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        if hyphen_count > 1 || (base.contains('*') && base != "*") {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }

        let (base_kind, left, right, right_offset) = if base == "*" {
            (0, min, max, start)
        } else if let Some(pos) = base.find('-') {
            let left_text = &base[..pos];
            let right_text = &base[pos + 1..];
            if left_text.is_empty() || right_text.is_empty() {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            let (l, ln) = parse_atom(left_text, field, start)?;
            let (r, rn) = parse_atom(right_text, field, start + pos + 1)?;
            (2, (l, ln), (r, rn), start + pos + 1)
        } else {
            let (v, numeric) = parse_atom(base, field, start)?;
            (1, (v, numeric), (v, numeric), start)
        };

        let (lo, hi) = if base_kind == 0 {
            (min, max)
        } else if base_kind == 1 {
            (left.0, right.0)
        } else {
            (left.0, right.0)
        };

        if base_kind != 0 {
            if left.1 {
                if left.0 < min || left.0 > max {
                    return Err(ParseError::OutOfRange { field, offset: start });
                }
            }
            if base_kind == 2 && right.1 && (right.0 < min || right.0 > max) {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: right_offset,
                });
            }
            if lo > hi {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: right_offset,
                });
            }
        }

        let step = if let Some((step_text, rel)) = step_text {
            if !step_text.bytes().all(|b| b.is_ascii_digit()) || step_text.is_empty() {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            let n = parse_number(step_text, field, start + rel)?;
            if n == 0 || n as usize > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: start + rel,
                });
            }
            n
        } else {
            1
        };

        let end = if step_text.is_some() && base_kind == 1 {
            max
        } else {
            hi
        };
        let mut value = lo;
        while value <= end {
            selected[(value - min) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        let _ = i;
    }

    Ok((selected, text == "*"))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let source = if input.starts_with('@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let pieces: Vec<&str> = source.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    let mut unrestricted = [false; 2];
    for i in 0..5 {
        let (values, star) = parse_field(pieces[i], FIELDS[i])?;
        selected[i] = values;
        if i == 2 {
            unrestricted[0] = star;
        } else if i == 4 {
            unrestricted[1] = star;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_days: unrestricted,
    })
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _, _) = bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    fn day_matches(&self, dt: CivilDateTime) -> bool {
        let dom = self.selected[2][dt.day as usize - 1];
        let dow = self.selected[4][weekday(dt.year, dt.month, dt.day) as usize];
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if dt.day == days_in_month(dt.year, dt.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn time_matches(&self, dt: CivilDateTime) -> bool {
        self.selected[0][dt.minute as usize]
            && self.selected[1][dt.hour as usize]
            && self.selected[3][dt.month as usize - 1]
            && self.day_matches(dt)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.time_matches(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = date_index(after.year, after.month, after.day);
        let last_date = date_index(2399, 12, 31);
        while date <= last_date {
            let (year, month, day) = date_from_index(date);
            if self.selected[3][month as usize - 1]
                && self.day_matches(CivilDateTime {
                    year,
                    month,
                    day,
                    hour: 0,
                    minute: 0,
                })
            {
                let min_hour = if (year, month, day) == (after.year, after.month, after.day) {
                    after.hour
                } else {
                    0
                };
                for hour in min_hour..24 {
                    if !self.selected[1][hour as usize] {
                        continue;
                    }
                    let min_minute = if (year, month, day) == (after.year, after.month, day)
                        && hour == after.hour
                    {
                        after.minute.saturating_add(1)
                    } else {
                        0
                    };
                    for minute in min_minute..60 {
                        if self.selected[0][minute as usize] {
                            return Ok(Some(CivilDateTime {
                                year,
                                month,
                                day,
                                hour,
                                minute,
                            }));
                        }
                    }
                }
            }
            date += 1;
        }
        Ok(None)
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = date_index(before.year, before.month, before.day);
        loop {
            let (year, month, day) = date_from_index(date);
            if self.selected[3][month as usize - 1]
                && self.day_matches(CivilDateTime {
                    year,
                    month,
                    day,
                    hour: 0,
                    minute: 0,
                })
            {
                for hour in (0..=before.hour).rev() {
                    if !self.selected[1][hour as usize] {
                        continue;
                    }
                    let max_minute = if (year, month, day) == (before.year, before.month, before.day)
                        && hour == before.hour
                    {
                        before.minute.saturating_sub(1)
                    } else {
                        59
                    };
                    if max_minute < 60 {
                        for minute in (0..=max_minute).rev() {
                            if self.selected[0][minute as usize] {
                                return Ok(Some(CivilDateTime {
                                    year,
                                    month,
                                    day,
                                    hour,
                                    minute,
                                }));
                            }
                        }
                    }
                }
            }
            if date == 0 {
                break;
            }
            date -= 1;
        }
        Ok(None)
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut cursor = Some(start);
        while let Some(current) = cursor {
            if current >= end {
                break;
            }
            let Some(next) = self.next_fire(current).map_err(|_| WindowError::ReversedWindow)? else {
                break;
            };
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(next);
            cursor = Some(next);
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for i in 0..5 {
            let field = FIELDS[i];
            let values = self.allowed(field);
            let (_, _, width) = bounds(field);
            let full = values.len() == width;
            let unrestricted = (i == 2 && self.unrestricted_days[0])
                || (i == 4 && self.unrestricted_days[1]);
            if unrestricted || (full && !matches!(i, 2 | 4)) {
                fields.push("*".to_string());
            } else if full && matches!(i, 2 | 4) {
                fields.push("*/1".to_string());
            } else {
                fields.push(
                    values
                        .iter()
                        .map(u8::to_string)
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        fields.join(" ")
    }
}

fn validate(dt: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&dt.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&dt.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if dt.day < 1 || dt.day > days_in_month(dt.year, dt.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if dt.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if dt.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn days_before_year(year: u16) -> u32 {
    let y = year as u32 - 2000;
    365 * y + (y + 3) / 4 - (y + 99) / 100 + (y + 399) / 400
}

fn date_index(year: u16, month: u8, day: u8) -> u32 {
    let mut index = days_before_year(year);
    for m in 1..month {
        index += days_in_month(year, m) as u32;
    }
    index + day as u32 - 1
}

fn date_from_index(mut index: u32) -> (u16, u8, u8) {
    let mut year = 2000u16;
    while year < 2399 {
        let n = if leap(year) { 366 } else { 365 };
        if index < n {
            break;
        }
        index -= n;
        year += 1;
    }
    let mut month = 1u8;
    loop {
        let n = days_in_month(year, month) as u32;
        if index < n {
            break;
        }
        index -= n;
        month += 1;
    }
    (year, month, index as u8 + 1)
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    ((date_index(year, month, day) + 6) % 7) as u8
}
