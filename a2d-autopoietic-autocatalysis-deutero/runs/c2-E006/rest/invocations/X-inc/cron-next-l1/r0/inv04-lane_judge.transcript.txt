    let base_end = slash.unwrap_or(item_end);
    let step_span = slash.map(|i| (i + 1, item_end));

    let mut hyphen = None;
    let mut is_star = false;
    if base_end > item_start && bytes[item_start] == b'*' {
        is_star = true;
        if base_end != item_start + 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    } else {
        for i in item_start..base_end {
            if bytes[i] == b'-' {
                if hyphen.is_some() {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                hyphen = Some(i);
            }
        }
        if let Some(h) = hyphen {
            if h == item_start || h + 1 == base_end {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            if !bytes[item_start..h].iter().all(u8::is_ascii_digit)
                || !bytes[h + 1..base_end].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        } else if !bytes[item_start..base_end].iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    }

    let (start, end) = if is_star {
        (bounds.min as u32, bounds.max as u32)
    } else if let Some(h) = hyphen {
        let start = checked_domain_number(bytes, item_start, h, field, bounds)?;
        let end = checked_domain_number(bytes, h + 1, base_end, field, bounds)?;
        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: h + 1,
            });
        }
        (start, end)
    } else {
        let start = checked_domain_number(bytes, item_start, base_end, field, bounds)?;
        (start, if step_span.is_some() { bounds.max as u32 } else { start })
    };

    let step = if let Some((step_start, step_end)) = step_span {
        if step_start == step_end
            || !bytes[step_start..step_end].iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let value = parse_number(bytes, step_start, step_end, field)?;
        if value == 0 || value > bounds.width as u32 {
            return Err(ParseError::BadStep {
                field,
                offset: step_start,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - bounds.min as u32) as usize] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_field(
    text: &str,
    field: Field,
    bounds: Domain,
) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; bounds.width as usize];
    let day_wildcard = matches!(field, Field::DayOfMonth | Field::DayOfWeek)
        && bytes == b"*";

    let mut start = 0;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(bytes, start, i, field, bounds, &mut selected)?;
            start = i + 1;
        }
    }

    let values = selected
        .iter()
        .enumerate()
        .filter_map(|(i, &yes)| {
            if yes {
                Some(bounds.min + i as u8)
            } else {
                None
            }
        })
        .collect();

    Ok((values, day_wildcard))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let field_order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_wildcards = [false; 2];

    for (i, (&field, text)) in field_order.iter().zip(fields.iter()).enumerate() {
        let (values, wildcard) = parse_field(text, field, domain(field))?;
        selected[i] = values;
        if field == Field::DayOfMonth {
            day_wildcards[0] = wildcard;
        } else if field == Field::DayOfWeek {
            day_wildcards[1] = wildcard;
        }
    }

    Ok(Cron {
        selected,
        day_wildcards,
    })
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    max: u8,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(error(field, offset, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| error(field, offset, ErrorKind::NumberOverflow))?;
    }

    if value > max as u32 {
        return Err(error(field, offset, ErrorKind::OutOfRange));
    }
    Ok(value)
}

fn parse_step(
    bytes: &[u8],
    field: Field,
    offset: usize,
    width: u8,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(error(field, offset, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| error(field, offset, ErrorKind::NumberOverflow))?;
    }

    if value == 0 || value > width as u32 {
        return Err(error(field, offset, ErrorKind::BadStep));
    }
    Ok(value)
}

fn parse_field(input: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = input.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(error(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let (domain_min, domain_max) = field_domain(field);
    let width = domain_max - domain_min + 1;
    let mut selected = vec![false; 60];
    let mut item_start = 0usize;

    for item in input.split(',') {
        let item_len = item.len();
        if item.is_empty() {
            return Err(error(field, item_start, ErrorKind::EmptyItem));
        }

        let item_bytes = item.as_bytes();
        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_bytes, step_offset) = match slash_positions.as_slice() {
            [] => (item_len, None, 0),
            [slash] => (*slash, Some(&item_bytes[slash + 1..]), item_start + slash + 1),
            _ => return Err(error(field, item_start, ErrorKind::InvalidSyntax)),
        };

        let base = &item_bytes[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        enum Base {
            Star,
            Single { value: u32, offset: usize },
            Range {
                start: u32,
                end: u32,
                start_offset: usize,
                end_offset: usize,
            },
        }

        let parsed_base = if base == b"*" {
            Base::Star
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            let value = parse_number(base, field, item_start, domain_max)?;
            Base::Single {
                value,
                offset: item_start,
            }
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if left.is_empty()
                || right.is_empty()
                || !left.iter().all(u8::is_ascii_digit)
                || !right.iter().all(u8::is_ascii_digit)
            {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }

            let start = parse_number(left, field, item_start, domain_max)?;
            let end_offset = item_start + hyphen + 1;
            let end = parse_number(right, field, end_offset, domain_max)?;
            if start > end {
                return Err(error(field, end_offset, ErrorKind::ReversedRange));
            }
            Base::Range {
                start,
                end,
                start_offset: item_start,
                end_offset,
            }
        } else {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        };

        let step = if let Some(step_bytes) = step_bytes {
            if step_bytes.is_empty() || !step_bytes.iter().all(u8::is_ascii_digit) {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
            Some(parse_step(step_bytes, field, step_offset, width)?)
        } else {
            None
        };

        let (start, end) = match parsed_base {
            Base::Star => (domain_min as u32, domain_max as u32),
            Base::Single { value, offset } => {
                let end = if step.is_some() {
                    domain_max as u32
                } else {
                    value
                };
                let _ = offset;
                (value, end)
            }
            Base::Range {
                start,
                end,
                start_offset,
                end_offset,
            } => {
                let _ = (start_offset, end_offset);
                (start, end)
            }
        };

        let increment = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item_len + 1;
    }

    let values = (domain_min..=domain_max)
        .filter(|&v| selected[v as usize])
        .collect();
    let exact_star = input.as_bytes() == b"*";
    Ok((values, exact_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {

    if parts.len() != 5 {
