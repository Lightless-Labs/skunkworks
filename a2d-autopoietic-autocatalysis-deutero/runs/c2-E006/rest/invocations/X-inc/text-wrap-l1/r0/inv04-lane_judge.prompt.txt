pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(32..=126).contains(&byte) && byte != b'\t' && byte != b'\n' && byte != b'\r' {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();
    let mut start = 0;

    while start < bytes.len() {
        while start < bytes.len() && is_separator(bytes[start]) {
            start += 1;
        }
        if start == bytes.len() {
            break;
        }

        let mut end = start;
        while end < bytes.len() && !is_separator(bytes[end]) {
            end += 1;
        }
        let word = &bytes[start..end];
        let mut offset = 0;

        while offset < word.len() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = &word[offset..];
            let remaining_len = remaining.len();

            if remaining_len <= width {
                let needed = remaining_len + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(remaining).expect("validated ASCII"));
                    offset = word.len();
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                line.push_str(
                    std::str::from_utf8(&remaining[..width]).expect("validated ASCII"),
                );
                lines.push(std::mem::take(&mut line));
                offset += width;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(
                        std::str::from_utf8(&remaining[..available]).expect("validated ASCII"),
                    );
                    lines.push(std::mem::take(&mut line));
                    offset += available;
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }

        start = end;
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}

    byte == b' ' || byte == b'\t' || byte == b'\n' || byte == b'\r'

pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, &byte) in input.as_bytes().iter().enumerate() {
        if !(32..=126).contains(&byte) && !matches!(byte, b'\t' | b'\n' | b'\r') {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r')).filter(|word| !word.is_empty()) {
        let mut suffix = word.as_bytes();

        while !suffix.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if suffix.len() <= width {
                let needed = suffix.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(suffix).expect("validated ASCII"));
                    suffix = &[];
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                let (chunk, rest) = suffix.split_at(width);
                lines.push(std::str::from_utf8(chunk).expect("validated ASCII").to_string());
                suffix = rest;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    let (chunk, rest) = suffix.split_at(available);
                    line.push_str(std::str::from_utf8(chunk).expect("validated ASCII"));
                    lines.push(std::mem::take(&mut line));
                    suffix = rest;
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}

pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, &byte) in input.as_bytes().iter().enumerate() {
        if !(32..=126).contains(&byte) && !matches!(byte, b'\t' | b'\n' | b'\r') {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r'))
        .filter(|word| !word.is_empty())
    {
        let mut remaining = word;

        while !remaining.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if remaining.len() <= width {
                let needed = remaining.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(remaining);
                    break;
                } else {
                    lines.push(std::mem::take(&mut line));
                    continue;
                }
            }

            if line.is_empty() {
                lines.push(remaining[..width].to_string());
                remaining = &remaining[width..];
                continue;
            }

            let available = width - line.len() - 1;
            if available >= 2 {
                line.push(' ');
                line.push_str(&remaining[..available]);
                lines.push(std::mem::take(&mut line));
                remaining = &remaining[available..];
            } else {
                lines.push(std::mem::take(&mut line));
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
