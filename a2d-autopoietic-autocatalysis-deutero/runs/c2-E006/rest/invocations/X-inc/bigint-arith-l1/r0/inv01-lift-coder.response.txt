#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits; zero is represented by [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = usize::from(bytes[0] == b'+');
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over separator and magnitude errors.
        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        // Find the first underscore not flanked by digits within the magnitude.
        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess = None;
        let mut digits = Vec::new();

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }

            if first_digit.is_none() {
                first_digit = Some((index, byte));
            }
            digit_count += 1;
            if digit_count == max_digits.saturating_add(1) {
                first_excess = Some(index);
            }
            digits.push(byte - b'0');
        }

        if digit_count > 1 && first_digit.is_some_and(|(_, digit)| digit == b'0') {
            return Err(Error::LeadingZero {
                index: first_digit.unwrap().0,
            });
        }

        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap_or_else(|| {
                    // This branch is only reachable if the count overflowed usize.
                    bytes.len()
                }),
            });
        }

        digits.reverse();
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let mut out = String::with_capacity(width.saturating_add(width / group.max(1)));
        let padding = width - natural_len;

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }

            if position < padding {
                out.push('0');
            } else {
                let digit_index = natural_len - 1 - (position - padding);
                out.push(char::from(b'0' + self.digits[digit_index]));
            }
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        use std::cmp::Ordering;

        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (&left, &right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(&right) {
                        Ordering::Equal => {}
                        ordering => return ordering,
                    }
                }
                Ordering::Equal
            }
            ordering => ordering,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let length = self.digits.len().max(rhs.digits.len());
        let mut result = Vec::with_capacity(length + 1);
        let mut carry = 0u8;

        for i in 0..length {
            let left = self.digits.get(i).copied().unwrap_or(0);
            let right = rhs.digits.get(i).copied().unwrap_or(0);
            let sum = left + right + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            result.push(carry);
        }

        Self { digits: result }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        use std::cmp::Ordering;

        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;

        for i in 0..self.digits.len() {
            let left = i16::from(self.digits[i]) - borrow;
            let right = i16::from(rhs.digits.get(i).copied().unwrap_or(0));

            if left < right {
                result.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                result.push((left - right) as u8);
                borrow = 0;
            }
        }

        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }

        Ok(Self { digits: result })
    }
}
