```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_signed_parts(negative: bool, magnitude: u128, denominator: u128) -> Result<Rational, RationalError> {
    let g = gcd_u128(magnitude, denominator);
    let mag = magnitude / g;
    let den = denominator / g;

    if den == 0 || den > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }

    let num = if negative {
        if mag > (1u128 << 63) {
            return Err(RationalError::Overflow);
        }
        if mag == (1u128 << 63) {
            i64::MIN
        } else {
            -(mag as i64)
        }
    } else {
        if mag > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        mag as i64
    };

    Ok(Rational {
        num,
        den: den as i64,
    })
}

fn abs_i64(n: i64) -> u128 {
    if n < 0 {
        (-(n as i128)) as u128
    } else {
        n as u128
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let negative = (num < 0) ^ (den < 0);
        from_signed_parts(negative, abs_i64(num), abs_i64(den))
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_fraction(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_fraction(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_fraction(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        from_i128_fraction(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let pos = i;
            i += 1;
            Some(pos)
        } else {
            None
        };

        let denominator_start;
        let denominator_end;
        if slash.is_some() {
            denominator_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            denominator_end = i;
        } else {
            denominator_start = i;
            denominator_end = i;
        }

        if i < bytes.len() {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let negative = bytes[0] == b'-';
        let numerator_digits = &bytes[numerator_start..numerator_end];

        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && numerator_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if slash.is_some()
            && denominator_end - denominator_start >= 2
            && bytes[denominator_start] == b'0'
        {
            return Err(parse_error(denominator_start + 1, ParseKind::NonCanonical));
        }

        let numerator_magnitude = parse_magnitude(numerator_digits);
        let numerator_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        if numerator_magnitude > numerator_limit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let denominator_magnitude = if slash.is_some() {
            let magnitude = parse_magnitude(&bytes[denominator_start..denominator_end]);
            if magnitude > i64::MAX as u128 {
                return Err(parse_error(denominator_start, ParseKind::OutOfRange));
            }
            if magnitude == 0 {
                return Err(parse_error(denominator_start, ParseKind::ZeroDenominator));
            }
            magnitude
        } else {
            1
        };

        if let Some(slash_at) = slash {
            if denominator_magnitude == 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, denominator_magnitude) > 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
        }

        from_signed_parts(negative, numerator_magnitude, denominator_magnitude)
    }
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(digits: &[u8]) -> u128 {
    let mut value = 0u128;
    for &digit in digits {
        value = value.saturating_mul(10).saturating_add((digit - b'0') as u128);
    }
    value
}

fn from_i128_fraction(n: i128, d: i128) -> Result<Rational, RationalError> {
    if d == 0 {
        return Err(RationalError::Overflow);
    }
    let negative = (n < 0) ^ (d < 0);
    let magnitude = if n < 0 { (-n) as u128 } else { n as u128 };
    let denominator = if d < 0 { (-d) as u128 } else { d as u128 };
    from_signed_parts(negative, magnitude, denominator)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
