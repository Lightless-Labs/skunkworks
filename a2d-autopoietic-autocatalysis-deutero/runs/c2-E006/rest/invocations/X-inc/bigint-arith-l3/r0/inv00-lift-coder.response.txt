```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented as [0] and is never negative.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let magnitude = &bytes[start..];

        if let Some(offset) = magnitude
            .iter()
            .position(|&b| b != b'_' && !b.is_ascii_digit())
        {
            return Err(Error::InvalidDigit {
                index: start + offset,
            });
        }

        for i in 0..magnitude.len() {
            if magnitude[i] == b'_'
                && (i == 0
                    || i + 1 == magnitude.len()
                    || !magnitude[i - 1].is_ascii_digit()
                    || !magnitude[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: start + i });
            }
        }

        let digit_count = magnitude.iter().filter(|&&b| b != b'_').count();
        let first_digit_offset = magnitude.iter().position(|&b| b != b'_').unwrap();

        if digit_count > 1 && magnitude[first_digit_offset] == b'0' {
            return Err(Error::LeadingZero {
                index: start + first_digit_offset,
            });
        }

        if digit_count > max_digits {
            let excess_offset = magnitude
                .iter()
                .enumerate()
                .filter(|(_, b)| **b != b'_')
                .nth(max_digits)
                .map(|(i, _)| i)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: start + excess_offset,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in magnitude.iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }

        Ok(Self::from_parts(negative, digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let pad = width - self.digits.len();
        let mut out = String::with_capacity(
            width + if group > 0 { (width - 1) / group } else { 0 } + usize::from(self.negative),
        );

        if self.negative {
            out.push('-');
        }

        for i in (0..width).rev() {
            if group > 0 && i != width - 1 && (i + 1) % group == 0 {
                out.push('_');
            }
            let digit = if i >= self.digits.len() {
                0
            } else {
                self.digits[i]
            };
            let _ = pad;
            out.push((b'0' + digit) as char);
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_abs(&self.digits, &rhs.digits),
            (true, true) => cmp_abs(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        match (self.negative, rhs.negative) {
            (false, false) => Self::from_parts(false, add_abs(&self.digits, &rhs.digits)),
            (true, true) => Self::from_parts(true, add_abs(&self.digits, &rhs.digits)),
            _ => match cmp_abs(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    sub_abs(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    sub_abs(&rhs.digits, &self.digits),
                ),
            },
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let index = i + j;
                let value = result[index] as u16 + a as u16 * b as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
            }

            let mut index = i + rhs.digits.len();
            while carry > 0 {
                let value = result[index] as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
                index += 1;
                if index == result.len() && carry > 0 {
                    result.push(0);
                }
            }
        }

        Self::from_parts(self.negative ^ rhs.negative, result)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient, remainder) = div_rem_abs(&self.digits, &rhs.digits);
        Ok((
            Self::from_parts(self.negative ^ rhs.negative, quotient),
            Self::from_parts(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }
}

fn cmp_abs(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for i in (0..a.len()).rev() {
                match a[i].cmp(&b[i]) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry > 0 {
        result.push(carry);
    }
    result
}

fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_abs(a, b) != Ordering::Less);
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i8;

    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(digit as u8);
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn mul_abs_digit(value: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }

    let mut result = Vec::with_capacity(value.len() + 1);
    let mut carry = 0u16;
    for &n in value {
        let product = n as u16 * digit as u16 + carry;
        result.push((product % 10) as u8);
        carry = product / 10;
    }
    while carry > 0 {
        result.push((carry % 10) as u8);
        carry /= 10;
    }
    result
}

fn div_rem_abs(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_abs(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0];

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        normalize(&mut remainder);

        let mut qdigit = 0u8;
        for candidate in (1..=9).rev() {
            let product = mul_abs_digit(divisor, candidate);
            if cmp_abs(&product, &remainder) != Ordering::Greater {
                qdigit = candidate;
                remainder = sub_abs(&remainder, &product);
                break;
            }
        }
        quotient_be.push(qdigit);
    }

    while quotient_be.len() > 1 && quotient_be[0] == 0 {
        quotient_be.remove(0);
    }
    quotient_be.reverse();
    normalize(&mut remainder);

    (quotient_be, remainder)
}

fn normalize(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}
```
