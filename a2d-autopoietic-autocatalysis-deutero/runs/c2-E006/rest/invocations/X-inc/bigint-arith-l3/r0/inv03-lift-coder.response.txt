use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] with negative=false.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        let mut digit_indices = Vec::new();
        let mut first_bad_separator = None;
        for i in start..bytes.len() {
            if bytes[i].is_ascii_digit() {
                digit_indices.push(i);
            } else {
                let prev_is_digit = i > start && bytes[i - 1].is_ascii_digit();
                let next_is_digit = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !(prev_is_digit && next_is_digit) && first_bad_separator.is_none() {
                    first_bad_separator = Some(i);
                }
            }
        }

        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        if digit_indices.len() > 1 && bytes[digit_indices[0]] == b'0' {
            return Err(Error::LeadingZero {
                index: digit_indices[0],
            });
        }

        if digit_indices.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_indices[max_digits],
            });
        }

        let digits: Vec<u8> = digit_indices
            .iter()
            .rev()
            .map(|&i| bytes[i] - b'0')
            .collect();
        Ok(Self::from_parts(negative, digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let mut magnitude = String::with_capacity(width + width / group.max(1));
        let padding = width - self.digits.len();

        for _ in 0..padding {
            magnitude.push('0');
        }
        for &digit in self.digits.iter().rev() {
            magnitude.push((b'0' + digit) as char);
        }

        if group == 0 || magnitude.len() <= group {
            return if self.negative {
                format!("-{magnitude}")
            } else {
                magnitude
            };
        }

        let mut grouped = String::with_capacity(magnitude.len() + magnitude.len() / group);
        let first_group = magnitude.len() % group;
        let first_group = if first_group == 0 { group } else { first_group };
        grouped.push_str(&magnitude[..first_group]);
        for chunk in magnitude[first_group..].as_bytes().chunks(group) {
            grouped.push('_');
            grouped.push_str(std::str::from_utf8(chunk).expect("ASCII digits"));
        }

        if self.negative {
            format!("-{grouped}")
        } else {
            grouped
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        match (self.negative, rhs.negative) {
            (false, false) => Self::from_parts(false, add_magnitude(&self.digits, &rhs.digits)),
            (true, true) => Self::from_parts(true, add_magnitude(&self.digits, &rhs.digits)),
            _ => match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    sub_magnitude(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    sub_magnitude(&rhs.digits, &self.digits),
                ),
            },
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::from_parts(
            self.negative != rhs.negative,
            mul_magnitude(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient, remainder) = div_magnitude(&self.digits, &rhs.digits);
        Ok((
            Self::from_parts(self.negative != rhs.negative, quotient),
            Self::from_parts(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let digits = normalize(digits);
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }
}

fn normalize(mut digits: Vec<u8>) -> Vec<u8> {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        vec![0]
    } else {
        digits
    }
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for (&x, &y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(&y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        out.push(carry);
    }
    normalize(out)
}

fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_magnitude(a, b) != Ordering::Less);
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(digit as u8);
    }
    normalize(out)
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut accum = vec![0u32; a.len() + b.len()];
    for (i, &x) in a.iter().enumerate() {
        for (j, &y) in b.iter().enumerate() {
            accum[i + j] += (x as u32) * (y as u32);
        }
    }

    let mut out = Vec::with_capacity(accum.len() + 1);
    let mut carry = 0u32;
    for value in accum {
        let total = value + carry;
        out.push((total % 10) as u8);
        carry = total / 10;
    }
    while carry != 0 {
        out.push((carry % 10) as u8);
        carry /= 10;
    }
    normalize(out)
}

fn mul_magnitude_digit(a: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }
    let mut out = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u16;
    for &value in a {
        let total = value as u16 * digit as u16 + carry;
        out.push((total % 10) as u8);
        carry = total / 10;
    }
    while carry != 0 {
        out.push((carry % 10) as u8);
        carry /= 10;
    }
    normalize(out)
}

fn div_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0];

    for &digit in dividend.iter().rev() {
        remainder.insert(0, digit);
        remainder = normalize(remainder);

        let mut qdigit = 0u8;
        for candidate in (1..=9).rev() {
            let product = mul_magnitude_digit(divisor, candidate);
            if cmp_magnitude(&product, &remainder) != Ordering::Greater {
                qdigit = candidate;
                remainder = sub_magnitude(&remainder, &product);
                break;
            }
        }
        quotient_be.push(qdigit);
    }

    quotient_be.reverse();
    (normalize(quotient_be), normalize(remainder))
}
