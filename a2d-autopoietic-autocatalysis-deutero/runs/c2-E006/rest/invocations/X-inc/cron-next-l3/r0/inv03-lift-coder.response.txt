#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    Ok(value)
}

fn named_value(field: Field, s: &str) -> Option<u32> {
    let upper = s.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn parse_atom(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if s.is_empty() {
        return Err(parse_error(field, offset, 0));
    }
    if s.as_bytes().iter().all(u8::is_ascii_digit) {
        let (min, max) = domain(field);
        let n = parse_number(s, field, offset)?;
        if n < min || n > max {
            return Err(parse_error(field, offset, 3));
        }
        return Ok(n);
    }
    if s.as_bytes().iter().all(u8::is_ascii_alphabetic) {
        return named_value(field, s).ok_or_else(|| parse_error(field, offset, 0));
    }
    Err(parse_error(field, offset, 0))
}

fn parse_item(
    item: &str,
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(parse_error(field, item_offset, 6));
    }

    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    if slash_count > 1 {
        return Err(parse_error(field, item_offset, 0));
    }
    let (base, step_text, step_offset) = if let Some(pos) = item.find('/') {
        (&item[..pos], Some(&item[pos + 1..]), Some(item_offset + pos + 1))
    } else {
        (item, None, None)
    };

    let (start_text, end_text) = if base == "*" {
        (None, None)
    } else {
        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        if hyphen_count > 1 || base.is_empty() {
            return Err(parse_error(field, item_offset, 0));
        }
        if let Some(pos) = base.find('-') {
            let left = &base[..pos];
            let right = &base[pos + 1..];
            if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                return Err(parse_error(field, item_offset, 0));
            }
            (Some(left), Some(right))
        } else {
            (Some(base), None)
        }
    };

    if step_text == Some("") {
        return Err(parse_error(field, item_offset, 0));
    }

    let start = if let Some(text) = start_text {
        Some(parse_atom(text, field, item_offset)?)
    } else {
        None
    };
    let end = if let Some(text) = end_text {
        let end_offset = item_offset + base.find('-').unwrap() + 1;
        Some(parse_atom(text, field, end_offset)?)
    } else {
        None
    };

    if let (Some(a), Some(b)) = (start, end) {
        if a > b {
            let end_offset = item_offset + base.find('-').unwrap() + 1;
            return Err(parse_error(field, end_offset, 4));
        }
    }

    let step = if let (Some(text), Some(offset)) = (step_text, step_offset) {
        if text.is_empty() || !text.as_bytes().iter().all(u8::is_ascii_digit) {
            return Err(parse_error(field, item_offset, 0));
        }
        let n = parse_number(text, field, offset)?;
        let (_, max) = domain(field);
        if n == 0 || n > max - domain(field).0 + 1 {
            return Err(parse_error(field, offset, 5));
        }
        n
    } else {
        1
    };

    let (min, max) = domain(field);
    let from = start.unwrap_or(min);
    let to = match (start, end, step_text.is_some()) {
        (_, Some(end), _) => end,
        (Some(n), None, true) => max.max(n),
        (Some(n), None, false) => n,
        (None, None, _) => max,
    };

    let mut value = from;
    while value <= to {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (min, max) = domain(FIELDS[i]);
        vec![false; (max - min + 1) as usize]
    });
    let mut unrestricted_days = [false; 2];

    for i in 0..5 {
        let field = FIELDS[i];
        let part = parts[i];

        for (offset, b) in part.bytes().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if field == Field::DayOfMonth && part == "*" {
            unrestricted_days[0] = true;
        }
        if field == Field::DayOfWeek && part == "*" {
            unrestricted_days[1] = true;
        }

        let mut offset = 0;
        for item in part.split(',') {
            parse_item(item, offset, field, &mut selected[i])?;
            offset += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, max) = domain(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some((min as usize + i) as u8))
            .take((max - min + 1) as usize)
            .collect()
    }

    fn has(&self, field: Field, value: u8) -> bool {
        let (min, max) = domain(field);
        if value as u32 < min || value as u32 > max {
            return false;
        }
        self.selected[field_index(field)][(value as u32 - min) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.has(Field::Minute, at.minute)
            || !self.has(Field::Hour, at.hour)
            || !self.has(Field::Month, at.month)
        {
            return false;
        }

        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut cur = after;
        while let Some(next) = increment_minute(cur) {
            if self.matches_valid(next) {
                return Ok(Some(next));
            }
            cur = next;
        }
        Ok(None)
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        return Some(CivilDateTime {
            minute: at.minute + 1,
            ..at
        });
    }
    if at.hour < 23 {
        return Some(CivilDateTime {
            hour: at.hour + 1,
            minute: 0,
            ..at
        });
    }
    if at.day < days_in_month(at.year, at.month) {
        return Some(CivilDateTime {
            day: at.day + 1,
            hour: 0,
            minute: 0,
            ..at
        });
    }
    if at.month < 12 {
        return Some(CivilDateTime {
            month: at.month + 1,
            day: 1,
            hour: 0,
            minute: 0,
            ..at
        });
    }
    if at.year < 2399 {
        return Some(CivilDateTime {
            year: at.year + 1,
            month: 1,
            day: 1,
            hour: 0,
            minute: 0,
            ..at
        });
    }
    None
}
