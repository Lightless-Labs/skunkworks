#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    start: usize,
}

#[derive(Debug, Clone)]
struct Expression {
    specifiers: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator {
            pos: content_start,
        });
    }

    let mut specifiers = Vec::new();
    let mut seen = Vec::<&str>::new();
    let mut start = content_start;

    loop {
        let end = (start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let name_bytes = &bytes[start..end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: start });
        }

        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName { pos: start + offset });
            }
        }

        let name = &template[start..end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(name);
        specifiers.push(Specifier {
            name: name.to_owned(),
            start,
        });

        if end == close {
            break;
        }
        start = end + 1;
    }

    content_start = close;
    let _ = content_start;

    Ok(Expression { specifiers })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;

                if let Some(relative_open) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_open,
                    });
                }

                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn encode_value(value: &str) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            let hi = bytes[i + 1];
            let lo = bytes[i + 2];
            if hi.is_ascii_hexdigit() && lo.is_ascii_hexdigit() {
                output.push('%');
                output.push((hi as char).to_ascii_uppercase());
                output.push((lo as char).to_ascii_uppercase());
                i += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut i = 0;
    let mut expression_index = 0;

    while i < bytes.len() {
        if bytes[i] != b'{' {
            let start = i;
            while i < bytes.len() && bytes[i] != b'{' {
                i += 1;
            }
            output.push_str(&template[start..i]);
            continue;
        }

        let close = i + 1 + bytes[i + 1..].iter().position(|&b| b == b'}').unwrap();
        let expression = &expressions[expression_index];
        expression_index += 1;

        let mut first = true;
        for specifier in &expression.specifiers {
            let _ = specifier.start;
            let mut value = None;
            for &(name, bound_value) in vars {
                if name == specifier.name {
                    value = Some(bound_value);
                }
            }
            if let Some(value) = value {
                if !first {
                    output.push(',');
                }
                output.push_str(&encode_value(value));
                first = false;
            }
        }
        i = close + 1;
    }

    Ok(output)
}
