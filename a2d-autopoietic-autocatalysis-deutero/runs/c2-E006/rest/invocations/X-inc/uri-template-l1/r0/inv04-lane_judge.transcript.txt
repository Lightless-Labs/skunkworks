pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let close = match bytes[pos + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => pos + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[pos + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: pos + 1 + offset });
                }

                let mut content_start = start + 1;
                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                }

                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &bytes[spec_start..spec_end];
                    let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
                    let valid_rest = |b: u8| b.is_ascii_alphanumeric() || b == b'_';

                    if !valid_first(spec[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    if let Some(offset) = spec[1..].iter().position(|&b| !valid_rest(b)) {
                        return Err(Error::InvalidName {
                            pos: spec_start + 1 + offset,
                        });
                    }

                    let name = String::from_utf8(spec.to_vec()).expect("validated ASCII name");
                    if seen.iter().any(|existing| existing == &name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name.clone());
                    names.push((spec_start, name));

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end: close + 1,
                    names,
                });
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

        output.push_str(&template[literal_start..expression.start]);

            }
        }
        output.push_str(&fields.join(","));

        literal_start = expression.end;
    }
    output.push_str(&template[literal_start..]);

    Ok(output)
}

fn encode_value(value: &str) -> String {

    let mut encoded = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

            encoded.push('%');
            encoded.push((bytes[pos + 1] as char).to_ascii_uppercase());

            pos += 1;

        }
    }

    encoded
}

        return Err(Error::NestedOpen {
            pos: open + 1 + relative,
        });
    }

    let mut content_start = open + 1;
    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator {
            pos: content_start,
        });

    let mut names = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map(|relative| spec_start + relative)
            .unwrap_or(close);

            return Err(Error::EmptyVariable { pos: spec_start });

        if !is_name_start(spec[0]) {

                    pos: spec_start + index,
                });
            }

            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push((spec_start, name));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: content_start,
        end: close,
        names,
    })
}

            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);

        index += 1;
    }

                    output.push(',');
                }
                encode_value(value, &mut output);
                first_field = false;
            }
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator {
            pos: content_start,
        });
    }

    let mut specifiers = Vec::new();
    let mut seen = Vec::<&str>::new();
    let mut start = content_start;

    loop {
        let end = (start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let name_bytes = &bytes[start..end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: start });
        }

        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName { pos: start + offset });
            }
        }

        let name = &template[start..end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(name);
        specifiers.push(Specifier {
            name: name.to_owned(),
            start,
        });

        if end == close {
            break;
        }
        start = end + 1;
    }

    content_start = close;
    let _ = content_start;

    Ok(Expression { specifiers })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {

            b'}' => return Err(Error::UnexpectedClose { pos: i }),

                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {

                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

        }

                let end = index + 1 + relative_end;

                if let Some(relative_open) = template[index + 1..end].find('{') {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + relative_open,
                    });
                }

                expressions.push(validate_expression(template, start, end)?);
                index = end + 1;
            }
            _ => index += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields = Vec::new();
        for (_, name) in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(encode_value(value));
            }
        }

        if !fields.is_empty() {
            output.push_str(&fields.join(","));
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
