#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::<u8>::with_capacity(bytes.len());
    let mut positions = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<Run>::new();

    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut active_run: Option<usize> = None;

    // Cleanup quote protection begins at the first retained byte only.
    let mut quote_protection = false;
    let mut quote_closed = false;

    while i < bytes.len() {
        let b = bytes[i];

        if quote_protection && !quote_closed {
            clean.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' && clean.len() > 1 {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        let outside_comment = depth == 0;

        if outside_comment && b == b'"' && clean.is_empty() {
            if let Some(start) = active_run.take() {
                runs.push(Run { start, end: i });
            }
            quote_protection = true;
            clean.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if depth == 0 && (is_hws(b) || b == b'(') {
            if active_run.is_none() {
                active_run = Some(i);
            }
            if is_hws(b) {
                i += 1;
                continue;
            }
            outer_open = i;
            depth = 1;
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                if i + 1 >= bytes.len() || !printable(bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }

            if b == b'\r' || b == b'\n' {
                if b != b'\r'
                    || i + 2 >= bytes.len()
                    || bytes[i + 1] != b'\n'
                    || !is_hws(bytes[i + 2])
                {
                    return Err(EmailError::BadFold { at: i });
                }
                i += 2;
                while i < bytes.len() && is_hws(bytes[i]) {
                    i += 1;
                }
                continue;
            }

            if !printable(b) && !is_hws(b) {
                return Err(EmailError::CommentChar { at: i });
            }

            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
            } else if b == b')' {
                depth -= 1;
                if depth == 0 {
                    i += 1;
                    continue;
                }
            }
            i += 1;
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if b == b'\r' || b == b'\n' {
            if b != b'\r'
                || i + 2 >= bytes.len()
                || bytes[i + 1] != b'\n'
                || !is_hws(bytes[i + 2])
            {
                return Err(EmailError::BadFold { at: i });
            }
            if active_run.is_none() {
                active_run = Some(i);
            }
            i += 2;
            while i < bytes.len() && is_hws(bytes[i]) {
                i += 1;
            }
            continue;
        }

        if is_hws(b) {
            if active_run.is_none() {
                active_run = Some(i);
            }
            i += 1;
            continue;
        }

        if let Some(start) = active_run.take() {
            runs.push(Run { start, end: i });
        }
        if clean.is_empty() && b == b'"' {
            quote_protection = true;
        }
        clean.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = active_run.take() {
        runs.push(Run {
            start,
            end: bytes.len(),
        });
    }

    // Boundary validation follows complete lexical cleanup.
    for run in runs {
        let left_at = clean.last() == Some(&b'@')
            && positions.last().is_some_and(|&p| p < run.start);
        let right_at = clean.first() == Some(&b'@')
            && positions.first().is_some_and(|&p| p >= run.end);

        // Find retained bytes immediately adjacent to this run in original order.
        let left_adjacent = positions
            .iter()
            .rposition(|&p| p < run.start)
            .is_some_and(|idx| clean[idx] == b'@');
        let right_adjacent = positions
            .iter()
            .position(|&p| p >= run.end)
            .is_some_and(|idx| clean[idx] == b'@');

        let at_boundary = !positions.iter().any(|&p| p < run.start)
            || !positions.iter().any(|&p| p >= run.end);

        let _ = (left_at, right_at);
        if !at_boundary && !left_adjacent && !right_adjacent {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let end_pos = input.len();
    let pos = |idx: usize| -> usize { positions.get(idx).copied().unwrap_or(end_pos) };

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: pos(0) });
        }
    }

    let mut separators = Vec::new();
    for j in 0..clean.len() {
        if clean[j] == b'@' && !quote_end.is_some_and(|q| j <= q) {
            separators.push(j);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(separators[1]),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: pos(sep) });
    }

    let mut local = Vec::<u8>::new();
    let mut quoted = false;
    if clean[0] == b'"' {
        quoted = true;
        let q = quote_end.unwrap();
        if q + 1 != sep {
            return Err(EmailError::TrailingLocal { at: pos(q + 1) });
        }
        let mut j = 1;
        let mut origins = Vec::<usize>::new();
        while j < q {
            if clean[j] == b'\\' {
                if j + 1 >= q
                    || !matches!(clean[j + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape { at: pos(j) });
                }
                local.push(clean[j + 1]);
                origins.push(pos(j));
                j += 2;
            } else {
                if !printable(clean[j]) {
                    return Err(EmailError::QuotedChar { at: pos(j) });
                }
                local.push(clean[j]);
                origins.push(pos(j));
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(q) });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: pos(64) });
        }
        let mut previous_dot = false;
        for j in 0..sep {
            let b = clean[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || previous_dot {
                    return Err(EmailError::LocalDot { at: pos(j) });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !b.is_ascii_alphanumeric()
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
                {
                    return Err(EmailError::LocalChar { at: pos(j) });
                }
            }
        }
        local.extend_from_slice(&clean[..sep]);
    }

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_start < domain_end && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(domain_end);
    }

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: if domain_start < clean.len() {
                pos(domain_start)
            } else {
                end_pos
            },
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(domain_start + 253),
        });
    }

    let mut labels = Vec::<(usize, usize)>::new();
    let mut start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean[j] == b'.' {
            if start == j {
                let at = if j == domain_end {
                    root_dot.map(pos).unwrap_or_else(|| pos(j))
                } else {
                    pos(j)
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((start, j));
            start = j + 1;
        }
    }

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong { at: pos(start + 63) });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(start) });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(end - 1) });
        }
        for j in start..end {
            let b = clean[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: pos(j) });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: if domain_start < clean.len() {
                pos(domain_start)
            } else {
                end_pos
            },
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(last_start) });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local: String::from_utf8(local).unwrap(),
        domain,
        quoted,
    })
}
