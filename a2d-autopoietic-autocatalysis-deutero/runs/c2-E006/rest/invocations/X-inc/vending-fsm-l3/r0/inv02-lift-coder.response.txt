#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    state: State,
    events: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = std::collections::HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            state: State {
                credit: 0,
                selected: None,
                escrow: Vec::new(),
                tubes,
            },
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        self.state.clone()
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }

        let new_credit = u32::from(self.state.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        let credit = new_credit as u16;
        self.state.escrow.push(coin);
        self.state.credit = credit;
        self.log(EventKind::Inserted { coin, credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.product_index(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.state.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        let id = id.to_owned();
        self.state.selected = Some(id.clone());
        self.log(EventKind::Selected { id });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = match self.state.selected.as_deref() {
            Some(id) => id.to_owned(),
            None => return Err(Error::NoSelection),
        };
        let index = self.product_index(&id).expect("selected product exists");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }

        let price = product.price;
        let credit = self.state.credit;
        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }

        let amount = credit - price;
        let change = make_change(amount, &self.state.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.state.tubes;
        for (index, denomination) in DENOMINATIONS.iter().enumerate() {
            let used = change.iter().filter(|coin| *coin == denomination).count() as u16;
            proposed_tubes[index] -= used;
        }
        for coin in &self.state.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|denomination| denomination == coin)
                .expect("escrow contains accepted denominations");
            let proposed = u32::from(proposed_tubes[index]) + 1;
            if proposed > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
            proposed_tubes[index] = proposed as u16;
        }

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };

        self.products[index].stock -= 1;
        self.state.tubes = proposed_tubes;
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        self.log(EventKind::Vended {
            id,
            price,
            change,
        });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.state.escrow.iter().rev().copied().collect();
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = self
            .product_index(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if self.state.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = self
            .product_index(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        let stock = stock as u16;
        self.products[index].stock = stock;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|product| product.id == id)
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn make_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index];
        let take = tubes[index].min(remaining / denomination);
        greedy[index] = take;
        remaining -= take * denomination;
    }

    let counts = if remaining == 0 {
        greedy
    } else {
        let mut best: Option<([u16; 4], u32)> = None;
        search_change(3, amount, tubes, [0; 4], &mut best);
        best?.0
    };

    let mut coins = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    Some(coins)
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: [u16; 4],
    best: &mut Option<([u16; 4], u32)>,
) {
    if index < 0 {
        if remaining == 0 {
            let total: u32 = counts.iter().map(|count| u32::from(*count)).sum();
            let replace = match best {
                None => true,
                Some((best_counts, best_total)) => {
                    if total < *best_total {
                        true
                    } else if total > *best_total {
                        false
                    } else {
                        (0..4)
                            .rev()
                            .find_map(|i| {
                                if counts[i] != best_counts[i] {
                                    Some(counts[i] > best_counts[i])
                                } else {
                                    None
                                }
                            })
                            .unwrap_or(false)
                    }
                }
            };
            if replace {
                *best = Some((counts, total));
            }
        }
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i];
    let max = tubes[i].min(remaining / denomination);
    for count in 0..=max {
        let mut next_counts = counts;
        next_counts[i] = count;
        search_change(
            index - 1,
            remaining - count * denomination,
            tubes,
            next_counts,
            best,
        );
    }
}
