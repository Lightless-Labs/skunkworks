#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = self.pending.clone();
        combined.extend_from_slice(input);

        let complete_len = (combined.len() / 5) * 5;
        let mut output = String::new();
        for chunk in combined[..complete_len].chunks_exact(5) {
            encode_symbols(chunk, self.config.alphabet, &mut output);
        }

        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let symbols = encode_symbols(&self.pending, self.config.alphabet, &mut output);
            if self.config.padding == Padding::Required {
                let padding_count = (8 - symbols % 8) % 8;
                output.extend(std::iter::repeat_n('=', padding_count));
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_symbols(input: &[u8], alphabet: Alphabet, output: &mut String) -> usize {
    let alphabet = alphabet_bytes(alphabet);
    let bit_len = input.len() * 8;
    let symbol_count = (bit_len + 4) / 5;

    for symbol_index in 0..symbol_count {
        let mut value = 0u8;
        for bit_offset in 0..5 {
            let bit_index = symbol_index * 5 + bit_offset;
            value <<= 1;
            if bit_index < bit_len {
                let byte = input[bit_index / 8];
                value |= (byte >> (7 - bit_index % 8)) & 1;
            }
        }
        output.push(alphabet[value as usize] as char);
    }

    symbol_count
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    padding_count: usize,
    first_padding_offset: Option<usize>,
    expected_padding: Option<usize>,
    accumulator: u8,
    pending_bits: u8,
    last_data_value: u8,
    last_data_offset: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            padding_count: 0,
            first_padding_offset: None,
            expected_padding: None,
            accumulator: 0,
            pending_bits: 0,
            last_data_value: 0,
            last_data_offset: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                let residue = next.data_count % 8;
                let expected = match residue {
                    0 => return Err(StreamError::Padding { offset }),
                    2 => 6,
                    4 => 4,
                    5 => 3,
                    7 => 1,
                    _ => return Err(StreamError::Padding { offset }),
                };

                if let Some(expected_already) = next.expected_padding {
                    if next.padding_count >= expected_already {
                        return Err(StreamError::Padding { offset });
                    }
                } else {
                    next.expected_padding = Some(expected);
                    next.first_padding_offset = Some(offset);
                }
                next.padding_count += 1;
            } else {
                let value = match decode_value(byte, next.config.alphabet) {
                    Some(value) => value,
                    None => return Err(StreamError::InvalidByte { offset, byte }),
                };

                let canonical = alphabet_bytes(next.config.alphabet)[value as usize];
                if next.config.canonical && byte != canonical {
                    return Err(StreamError::NonCanonical {
                        offset,
                        expected: canonical,
                    });
                }

                if next.first_padding_offset.is_some() {
                    return Err(StreamError::Padding { offset });
                }

                next.data_count += 1;
                next.last_data_value = value;
                next.last_data_offset = offset;
                next.accumulator = (next.accumulator << 5) | value;
                next.pending_bits += 5;

                while next.pending_bits >= 8 {
                    next.pending_bits -= 8;
                    output.push((next.accumulator >> next.pending_bits) as u8);
                    if next.pending_bits == 0 {
                        next.accumulator = 0;
                    } else {
                        next.accumulator &= (1u8 << next.pending_bits) - 1;
                    }
                }
            }

            next.position += 1;
        }

        self.copy_state_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let legal = matches!(residue, 0 | 2 | 4 | 5 | 7);
        if !legal {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => unreachable!(),
        };
        if unused_bits != 0 {
            let mask = (1u8 << unused_bits) - 1;
            if self.last_data_value & mask != 0 {
                return Err(StreamError::TrailingBits {
                    offset: self.last_data_offset,
                });
            }
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            padding_count: self.padding_count,
            first_padding_offset: self.first_padding_offset,
            expected_padding: self.expected_padding,
            accumulator: self.accumulator,
            pending_bits: self.pending_bits,
            last_data_value: self.last_data_value,
            last_data_offset: self.last_data_offset,
            closed: self.closed,
        }
    }

    fn copy_state_from(&mut self, other: Self) {
        self.position = other.position;
        self.data_count = other.data_count;
        self.padding_count = other.padding_count;
        self.first_padding_offset = other.first_padding_offset;
        self.expected_padding = other.expected_padding;
        self.accumulator = other.accumulator;
        self.pending_bits = other.pending_bits;
        self.last_data_value = other.last_data_value;
        self.last_data_offset = other.last_data_offset;
        self.closed = other.closed;
    }
}

fn decode_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let upper = byte.to_ascii_uppercase();
    alphabet_bytes(alphabet)
        .iter()
        .position(|&candidate| candidate == upper)
        .map(|value| value as u8)
}
