#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

pub struct Decoder {
    config: Config,
    bit_buffer: u8,
    bit_count: u8,
    data_count: usize,
    total_consumed: usize,
    last_data_offset: Option<usize>,
    padding_started: bool,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }
        if input.is_empty() {
            return Ok(String::new());
        }

        let mut bytes = Vec::with_capacity(self.pending.len() + input.len());
        bytes.extend_from_slice(&self.pending);
        bytes.extend_from_slice(input);

        let complete_len = (bytes.len() / 5) * 5;
        let output = encode_bytes(&bytes[..complete_len], self.config.alphabet);
        self.pending = bytes[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let data_symbols = (self.pending.len() * 8 + 4) / 5;
            output = encode_bytes(&self.pending, self.config.alphabet);
            if self.config.padding == Padding::Required {
                let padded_len = ((data_symbols + 7) / 8) * 8;
                output.extend(std::iter::repeat('=').take(padded_len - data_symbols));
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            bit_buffer: 0,
            bit_count: 0,
            data_count: 0,
            total_consumed: 0,
            last_data_offset: None,
            padding_started: false,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }
        if input.is_empty() {
            return Ok(Vec::new());
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.total_consumed;
            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }
                if next.padding_started {
                    if next.padding_count >= next.expected_padding {
                        return Err(StreamError::Padding { offset });
                    }
                    next.padding_count += 1;
                } else {
                    let residue = next.data_count % 8;
                    let expected = match residue {
                        2 => 6,
                        4 => 4,
                        5 => 3,
                        7 => 1,
                        _ => return Err(StreamError::Padding { offset }),
                    };
                    next.padding_started = true;
                    next.first_padding_offset = Some(offset);
                    next.expected_padding = expected;
                    next.padding_count = 1;
                }
                next.total_consumed += 1;
                continue;
            }

            let value = match decode_symbol(byte, next.config.alphabet) {
                Some(value) => value,
                None => return Err(StreamError::InvalidByte { offset, byte }),
            };

            if next.config.canonical && byte.is_ascii_lowercase() {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: byte.to_ascii_uppercase(),
                });
            }
            if next.padding_started {
                return Err(StreamError::Padding { offset });
            }

            next.data_count += 1;
            next.last_data_offset = Some(offset);
            next.bit_buffer = (next.bit_buffer << 5) | value;
            next.bit_count += 5;

            while next.bit_count >= 8 {
                next.bit_count -= 8;
                output.push((next.bit_buffer >> next.bit_count) & 0xff);
                if next.bit_count == 0 {
                    next.bit_buffer = 0;
                } else {
                    next.bit_buffer &= (1 << next.bit_count) - 1;
                }
            }
            next.total_consumed += 1;
        }

        self.copy_state_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => {
                return Err(StreamError::InvalidLength {
                    offset: self.total_consumed,
                })
            }
        };

        if self.config.padding == Padding::Required
            && self.padding_count != expected_padding
        {
            return Err(StreamError::Padding {
                offset: self
                    .first_padding_offset
                    .unwrap_or(self.total_consumed),
            });
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => unreachable!(),
        };
        if unused_bits != 0 && self.bit_buffer & ((1 << unused_bits) - 1) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.total_consumed),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

impl Decoder {
    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            data_count: self.data_count,
            total_consumed: self.total_consumed,
            last_data_offset: self.last_data_offset,
            padding_started: self.padding_started,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }

    fn copy_state_from(&mut self, other: Self) {
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.data_count = other.data_count;
        self.total_consumed = other.total_consumed;
        self.last_data_offset = other.last_data_offset;
        self.padding_started = other.padding_started;
        self.first_padding_offset = other.first_padding_offset;
        self.padding_count = other.padding_count;
        self.expected_padding = other.expected_padding;
        self.closed = other.closed;
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(input: &[u8], alphabet: Alphabet) -> String {
    let symbols = alphabet_bytes(alphabet);
    let symbol_count = (input.len() * 8 + 4) / 5;
    let mut output = String::with_capacity(symbol_count);

    for symbol_index in 0..symbol_count {
        let start_bit = symbol_index * 5;
        let mut value = 0u8;
        for bit in 0..5 {
            value <<= 1;
            let bit_index = start_bit + bit;
            if bit_index < input.len() * 8 {
                let byte = input[bit_index / 8];
                value |= (byte >> (7 - (bit_index % 8))) & 1;
            }
        }
        output.push(symbols[value as usize] as char);
    }
    output
}

fn decode_symbol(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let upper = byte.to_ascii_uppercase();
    alphabet_bytes(alphabet)
        .iter()
        .position(|&symbol| symbol == upper)
        .map(|value| value as u8)
}
