    NestedOpen { pos: usize },

enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Question),
            b'&' => Some(Self::Ampersand),
            _ => None,
        }
    }

    fn separator(self) -> &'static str {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ",",
            Self::Dot => ".",
            Self::Slash => "/",
            Self::Semicolon | Self::Question | Self::Ampersand => "&",
        }
    }

    fn prefix(self) -> &'static str {
        match self {
            Self::Simple | Self::Reserved => "",
            Self::Fragment => "#",
            Self::Dot => ".",
            Self::Slash => "/",
            Self::Semicolon => ";",
            Self::Question => "?",
            Self::Ampersand => "&",
        }
    }

    fn allows_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

    }
}

    )
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                const HEX: &[u8; 16] = b"0123456789ABCDEF";
                out.push('%');
                out.push(HEX[high as usize] as char);
                out.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        if unreserved || (allow_reserved && is_reserved(byte)) {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

    let bytes = template.as_bytes();

    if let Some(relative) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + relative,
        });
    }

    let mut operator = Operator::Simple;

    if content_start < close {
        if let Some(found) = Operator::from_byte(bytes[content_start]) {
            operator = found;
            content_start += 1;
        } else if matches!(bytes[content_start], b'=' | b'!' | b'@' | b'|' | b';' | b'<' | b'>') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specifiers = Vec::new();
    let mut start = content_start;
    loop {
        let end = bytes[start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| start + relative)
            .unwrap_or(close);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let name_bytes = &bytes[start..end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: start });
        }
        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName { pos: start + offset });
            }
        }

        let name = &template[start..end];
        if specifiers.iter().any(|(_, existing): &(usize, &str)| *existing == name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        specifiers.push((start, name));

        if end == close {
            break;
        }
        start = end + 1;
    }

    Ok((operator, specifiers))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let close = bytes[i + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                    .map(|relative| i + 1 + relative)
                    .ok_or(Error::Unclosed { pos: i })?;
                let parsed = parse_expression(template, i, close)?;
                expressions.push((i, close + 1, parsed));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;

    for (open, end, (operator, specifiers)) in expressions {
        out.push_str(&template[cursor..open]);

        let mut fields = Vec::new();
        for (_, name) in specifiers {
            if let Some((_, value)) = vars.iter().rev().find(|(binding_name, _)| *binding_name == name) {
                let encoded = encode_value(value, operator.allows_reserved());

                let field = match operator {
                    Operator::Semicolon if encoded.is_empty() => name.to_owned(),
                    Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                        format!("{name}={encoded}")
                    }
                    _ => encoded,
                };
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            out.push_str(operator.prefix());
            out.push_str(&fields.join(operator.separator()));
        }

        cursor = end;
    }

    out.push_str(&template[cursor..]);
    Ok(out)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },

    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

    name: String,
    start: usize,
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<Variable>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(offset) = bytes[open + 1..close]
        .iter()
        .position(|&byte| byte == b'{')
    {
        return Err(Error::NestedOpen {
            pos: open + 1 + offset,
        });
    }

    let content_start = open + 1;
    let mut spec_start = content_start;
    let mut operator = None;

    if content_start < close {
        let first = bytes[content_start] as char;
        if matches!(first, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            operator = Some(first);
            spec_start += 1;
        } else if matches!(first, '=' | '!' | '@' | '|' | '$') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    if spec_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut expression_names = Vec::<String>::new();
    let mut item_start = spec_start;

    loop {
        let item_end = bytes[item_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |offset| item_start + offset);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let name_bytes = &bytes[item_start..item_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: item_start });
        }

        let invalid_offset = name_bytes
            .iter()
            .enumerate()
            .skip(1)
            .find_map(|(index, &byte)| {
                (!is_name_continue(byte)).then_some(index)
            });

        if let Some(offset) = invalid_offset {
            return Err(Error::InvalidName {
                pos: item_start + offset,
            });
        }

        let name = template[item_start..item_end].to_string();
        if expression_names.iter().any(|used| used == &name) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        expression_names.push(name.clone());
        variables.push(Variable {
            name,
            start: item_start,
        });

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        variables,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open = index;
                let Some(offset) = bytes[index + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = index + 1 + offset;
                expressions.push(parse_expression(template, open, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            output.push('%');
            output.push(hex_upper(bytes[index + 1]) as char);
            output.push(hex_upper(bytes[index + 2]) as char);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        index += 1;
    }

    output
}

fn binding_value<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|(candidate, value)| (*candidate == name).then_some(*value))
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some('+') | Some('#'));
    let mut fields = Vec::<String>::new();

    for variable in &expression.variables {
        let Some(value) = binding_value(&variable.name, vars) else {
            continue;
        };

        let encoded = encode(value, allow_reserved);
        let field = match operator {
            Some(';' | '?' | '&') if encoded.is_empty() && operator == Some(';') => {
                variable.name.clone()
            }
            Some(';' | '?' | '&') => format!("{}={}", variable.name, encoded),
            _ => encoded,
        };
        fields.push(field);
    }

    if fields.is_empty() {
        return String::new();
    }

    let separator = match operator {
        Some('.') => ".",
        Some('/') => "/",
        Some(';') => ";",
        Some('?') | Some('&') => "&",
        _ => ",",
    };

    let prefix = match operator {
        Some('#') => "#",
        Some('.') => ".",
        Some('/') => "/",
        Some(';') => ";",
        Some('?') => "?",
        Some('&') => "&",
        _ => "",
    };

    format!("{}{}", prefix, fields.join(separator))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        output.push_str(&expand_expression(&expression, vars));
        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },

enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon | Self::Query | Self::Continuation => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continuation => Some('&'),
        }
    }

    fn is_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }

    fn named_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continuation)
    }

    fn semicolon_fields(self) -> bool {
        matches!(self, Self::Semicolon)
    }
}

struct Expression {
    operator: Operator,
    names: Vec<String>,
}

fn parse_expression(content: &str, start: usize, close: usize) -> Result<Expression, Error> {
    let mut body_start = 0;
    let mut operator = Operator::Simple;

    if let Some(first) = content.chars().next() {
        let op_len = first.len_utf8();
        let recognized = match first {
            '+' => Some(Operator::Reserved),
            '#' => Some(Operator::Fragment),
            '.' => Some(Operator::Dot),
            '/' => Some(Operator::Slash),
            ';' => Some(Operator::Semicolon),
            '?' => Some(Operator::Query),
            '&' => Some(Operator::Continuation),
            _ => None,
        };
        if let Some(op) = recognized {
            operator = op;
            body_start = op_len;
        }
    }

    let body = &content[body_start..];
    let body_pos = start + body_start;
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut offset = 0;

    for specifier in body.split(',') {
        let spec_pos = body_pos + offset;
        if specifier.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let bytes = specifier.as_bytes();
        let valid_first = bytes[0].is_ascii_alphabetic() || bytes[0] == b'_';
        if !valid_first {
            return Err(Error::InvalidName { pos: spec_pos });

        for (i, byte) in bytes.iter().enumerate().skip(1) {
            if !(byte.is_ascii_alphanumeric() || *byte == b'_') {
                return Err(Error::InvalidName { pos: spec_pos + i });
            }

        if seen.iter().any(|name| *name == specifier) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }
        seen.push(specifier);
        names.push(specifier.to_owned());
        offset += specifier.len() + 1;
    }

    Ok(Expression { operator, names })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = i + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(inner) = (open + 1..close).find(|&j| bytes[j] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                let content = &template[open + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);

        let mut fields: Vec<String> = Vec::new();
        for name in &expression.names {
            if let Some(value) = lookup(name, vars) {
                let encoded = encode(value, expression.operator.is_reserved());
                if expression.operator.named_fields() {
                    if expression.operator.semicolon_fields() && encoded.is_empty() {
                        fields.push(name.clone());
                    } else {
                        fields.push(format!("{name}={encoded}"));
                    }
                } else {
                    fields.push(encoded);
                }
            }
        }

        if !fields.is_empty() {
            if let Some(prefix) = expression.operator.prefix() {
                output.push(prefix);
            }
            let separator = expression.operator.separator().to_string();
            output.push_str(&fields.join(&separator));
        }

        cursor = end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
