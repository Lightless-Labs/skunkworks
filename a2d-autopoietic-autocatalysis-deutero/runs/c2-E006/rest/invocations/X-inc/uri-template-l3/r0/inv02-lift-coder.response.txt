#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[open + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                let mut operator = None;
                if content_start < close {
                    let ch = bytes[content_start] as char;
                    if matches!(ch, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
                        operator = Some(ch);
                        content_start += 1;
                    } else if matches!(ch, '!' | '@' | '|' | '$' | '=' | ',') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut variables = Vec::new();
                let mut seen: Vec<String> = Vec::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let mut name_end = spec_start;
                    while name_end < spec_end {
                        let b = bytes[name_end];
                        let valid = if name_end == spec_start {
                            b.is_ascii_alphabetic() || b == b'_'
                        } else {
                            b.is_ascii_alphanumeric() || b == b'_'
                        };
                        if !valid {
                            return Err(Error::InvalidName { pos: name_end });
                        }
                        name_end += 1;
                    }

                    let name = template[spec_start..spec_end].to_owned();
                    if seen.iter().any(|existing| existing == &name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name.clone());
                    variables.push((name, spec_start));

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start: open,
                    end: close + 1,
                    operator,
                    variables,
                });
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let operator = expression.operator.unwrap_or('\0');
        let reserved = matches!(operator, '+' | '#');
        let (separator, prefix) = match operator {
            '.' => ('.', Some('.')),
            '/' => ('/', Some('/')),
            ';' => (';', Some(';')),
            '?' => ('&', Some('?')),
            '&' => ('&', Some('&')),
            _ => (',', None),
        };

        let mut fields = Vec::new();
        for (name, _) in expression.variables {
            if let Some(value) = vars
                .iter()
                .rev()
                .find_map(|(binding_name, value)| (*binding_name == name).then_some(*value))
            {
                let encoded = encode(value, reserved);
                let field = match operator {
                    ';' if encoded.is_empty() => name,
                    ';' | '?' | '&' => format!("{name}={encoded}"),
                    _ => encoded,
                };
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            if let Some(prefix) = prefix {
                output.push(prefix);
            }
            for (index, field) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(separator);
                }
                output.push_str(field);
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];

        if b == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if is_unreserved(b) || (reserved && is_reserved(b)) {
            output.push(b as char);
        } else {
            output.push('%');
            output.push(hex_digit(b >> 4));
            output.push(hex_digit(b & 0x0f));
        }
        i += 1;
    }

    output
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_hex(b: u8) -> bool {
    b.is_ascii_hexdigit()
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
