#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon | Self::Query | Self::Continuation => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continuation => Some('&'),
        }
    }

    fn is_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }

    fn named_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continuation)
    }

    fn semicolon_fields(self) -> bool {
        matches!(self, Self::Semicolon)
    }
}

struct Expression {
    operator: Operator,
    names: Vec<String>,
}

fn parse_expression(content: &str, start: usize, close: usize) -> Result<Expression, Error> {
    let mut body_start = 0;
    let mut operator = Operator::Simple;

    if let Some(first) = content.chars().next() {
        let op_len = first.len_utf8();
        let recognized = match first {
            '+' => Some(Operator::Reserved),
            '#' => Some(Operator::Fragment),
            '.' => Some(Operator::Dot),
            '/' => Some(Operator::Slash),
            ';' => Some(Operator::Semicolon),
            '?' => Some(Operator::Query),
            '&' => Some(Operator::Continuation),
            _ => None,
        };
        if let Some(op) = recognized {
            operator = op;
            body_start = op_len;
        }
    }

    let body = &content[body_start..];
    let body_pos = start + body_start;
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut offset = 0;

    for specifier in body.split(',') {
        let spec_pos = body_pos + offset;
        if specifier.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let bytes = specifier.as_bytes();
        let valid_first = bytes[0].is_ascii_alphabetic() || bytes[0] == b'_';
        if !valid_first {
            return Err(Error::InvalidName { pos: spec_pos });
        }
        for (i, byte) in bytes.iter().enumerate().skip(1) {
            if !(byte.is_ascii_alphanumeric() || *byte == b'_') {
                return Err(Error::InvalidName { pos: spec_pos + i });
            }
        }

        if seen.iter().any(|name| *name == specifier) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }
        seen.push(specifier);
        names.push(specifier.to_owned());
        offset += specifier.len() + 1;
    }

    Ok(Expression { operator, names })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = i + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(inner) = (open + 1..close).find(|&j| bytes[j] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                let content = &template[open + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);

        let mut fields: Vec<String> = Vec::new();
        for name in &expression.names {
            if let Some(value) = lookup(name, vars) {
                let encoded = encode(value, expression.operator.is_reserved());
                if expression.operator.named_fields() {
                    if expression.operator.semicolon_fields() && encoded.is_empty() {
                        fields.push(name.clone());
                    } else {
                        fields.push(format!("{name}={encoded}"));
                    }
                } else {
                    fields.push(encoded);
                }
            }
        }

        if !fields.is_empty() {
            if let Some(prefix) = expression.operator.prefix() {
                output.push(prefix);
            }
            let separator = expression.operator.separator().to_string();
            output.push_str(&fields.join(&separator));
        }

        cursor = end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
