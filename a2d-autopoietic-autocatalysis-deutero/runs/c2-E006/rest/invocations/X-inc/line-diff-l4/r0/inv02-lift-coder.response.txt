#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
}

#[derive(Debug, Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<usize>,
}

fn lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn lcs_table(old: &[String], new: &[String]) -> Vec<Vec<usize>> {
    let mut table = vec![vec![0; new.len() + 1]; old.len() + 1];
    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            table[i][j] = if old[i] == new[j] {
                1 + table[i + 1][j + 1]
            } else {
                table[i + 1][j].max(table[i][j + 1])
            };
        }
    }
    table
}

fn changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let lcs = lcs_table(old, new);
    let (mut i, mut j) = (0, 0);
    let mut out = Vec::new();

    while i < old.len() || j < new.len() {
        if i < old.len() && j < new.len() && old[i] == new[j] {
            i += 1;
            j += 1;
            continue;
        }

        let start = i;
        let mut replacement = Vec::new();

        loop {
            if i >= old.len() && j >= new.len() {
                break;
            }
            if i < old.len() && j < new.len() && old[i] == new[j] {
                break;
            }

            let delete = if i >= old.len() {
                false
            } else if j >= new.len() {
                true
            } else {
                let d = lcs[i + 1][j];
                let ins = lcs[i][j + 1];
                if d > ins {
                    true
                } else if ins > d {
                    false
                } else {
                    (i + j) % 2 == 0
                }
            };

            if delete {
                i += 1;
            } else {
                replacement.push(new[j].clone());
                j += 1;
            }
        }

        out.push(Change {
            start,
            end: i,
            replacement,
            side,
        });
    }

    out
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[Change],
    indices: &[usize],
) -> Vec<String> {
    let mut selected: Vec<&Change> = indices.iter().map(|&i| &changes[i]).collect();
    selected.sort_by_key(|c| (c.start, c.end));

    let mut out = Vec::new();
    let mut cursor = start;
    for change in selected {
        if change.start > cursor {
            out.extend(base[cursor..change.start].iter().cloned());
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        out.extend(base[cursor..end].iter().cloned());
    }
    out
}

fn append_lines(output: &mut Vec<String>, payload: &[String]) {
    output.extend(payload.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut all_changes = changes(&base_lines, &left_lines, 0);
    all_changes.extend(changes(&base_lines, &right_lines, 1));

    let n = all_changes.len();
    let mut parent: Vec<usize> = (0..n).collect();

    fn root(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = root(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..n {
        for j in i + 1..n {
            if interacts(&all_changes[i], &all_changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<Vec<usize>> = Vec::new();
    for i in 0..n {
        let r = root(&mut parent, i);
        if let Some(group) = groups.iter_mut().find(|g| root(&mut parent, g[0]) == r) {
            group.push(i);
        } else {
            groups.push(vec![i]);
        }
    }

    let mut components: Vec<Component> = groups
        .into_iter()
        .map(|indices| Component {
            start: indices
                .iter()
                .map(|&i| all_changes[i].start)
                .min()
                .unwrap(),
            end: indices.iter().map(|&i| all_changes[i].end).max().unwrap(),
            changes: indices,
        })
        .collect();

    components.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0;

    for component in components {
        if cursor < component.start {
            append_lines(&mut output, &base_lines[cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_indices: Vec<usize> = component
            .changes
            .iter()
            .copied()
            .filter(|&i| all_changes[i].side == 0)
            .collect();
        let right_indices: Vec<usize> = component
            .changes
            .iter()
            .copied()
            .filter(|&i| all_changes[i].side == 1)
            .collect();

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &all_changes,
            &left_indices,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &all_changes,
            &right_indices,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let region_len = component.end - component.start;
                    let base_start_display = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_start_display},{region_len}"
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len =
                        4 + left_projection.len() + base_projection.len() + right_projection.len();
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len,
                    });
                }
            }
        }
        cursor = component.end;
    }

    if cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[cursor..]);
    }

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
