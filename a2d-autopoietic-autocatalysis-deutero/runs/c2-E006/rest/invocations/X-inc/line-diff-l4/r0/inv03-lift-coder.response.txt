#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone, Debug)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Clone, Debug)]
struct Component {
    changes: Vec<(bool, usize)>,
    start: usize,
    end: usize,
}

fn split_lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input
        .strip_suffix('\n')
        .unwrap_or(input)
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut i = 0;
    let mut j = 0;
    let mut base_pos = 0;
    let mut changes = Vec::new();
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            run_start.get_or_insert(base_pos);
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            run_start.get_or_insert(base_pos);
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            let choose_delete = if del != ins {
                del > ins
            } else {
                (i + j) % 2 == 0
            };
            run_start.get_or_insert(base_pos);
            if choose_delete {
                deletes += 1;
                i += 1;
                base_pos += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;
    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn project(base: &[String], start: usize, end: usize, changes: &[&Change]) -> Vec<String> {
    let mut output = Vec::new();
    let mut cursor = start;
    for change in changes {
        if change.start > cursor {
            output.extend(base[cursor..change.start].iter().cloned());
        }
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        output.extend(base[cursor..end].iter().cloned());
    }
    output
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let mut nodes = Vec::new();
    for (side, changes) in [(true, &left_changes), (false, &right_changes)] {
        for (index, change) in changes.iter().enumerate() {
            nodes.push((side, index, change));
        }
    }

    let mut parent: Vec<usize> = (0..nodes.len()).collect();
    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }
    for i in 0..nodes.len() {
        for j in i + 1..nodes.len() {
            if interacts(nodes[i].2, nodes[j].2) {
                let a = find(&mut parent, i);
                let b = find(&mut parent, j);
                if a != b {
                    parent[b] = a;
                }
            }
        }
    }

    let mut grouped: Vec<Component> = Vec::new();
    for (node_index, (is_left, change_index, change)) in nodes.iter().enumerate() {
        let root = find(&mut parent, node_index);
        if let Some(component) = grouped.iter_mut().find(|c| c.changes.first().is_some_and(|&(s, idx)| {
            let first = if s { &left_changes[idx] } else { &right_changes[idx] };
            let first_node = nodes.iter().position(|n| std::ptr::eq(n.2, first)).unwrap();
            find(&mut parent, first_node) == root
        })) {
            component.changes.push((*is_left, *change_index));
            component.start = component.start.min(change.start);
            component.end = component.end.max(change.end);
        } else {
            grouped.push(Component {
                changes: vec![(*is_left, *change_index)],
                start: change.start,
                end: change.end,
            });
        }
    }

    grouped.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output = Vec::<String>::new();
    let mut regions = Vec::new();
    let mut base_cursor = 0;

    for component in grouped {
        if base_cursor < component.start {
            output.extend(base_lines[base_cursor..component.start].iter().cloned());
        }

        let component_left: Vec<&Change> = component
            .changes
            .iter()
            .filter(|(is_left, _)| *is_left)
            .map(|(_, index)| &left_changes[*index])
            .collect();
        let component_right: Vec<&Change> = component
            .changes
            .iter()
            .filter(|(is_left, _)| !*is_left)
            .map(|(_, index)| &right_changes[*index])
            .collect();

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component_left,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component_right,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = regions.len() + 1;
                    let output_start = output.len();
                    let base_start_display = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection);
                    output.push(format!(
                        "||||||| BASE {base_start_display},{}",
                        component.end - component.start
                    ));
                    output.extend(base_projection);
                    output.push("=======".to_owned());
                    output.extend(right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    regions.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + (component.end - component.start)
                            + right_projection.len(),
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        output.extend(base_lines[base_cursor..].iter().cloned());
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}
