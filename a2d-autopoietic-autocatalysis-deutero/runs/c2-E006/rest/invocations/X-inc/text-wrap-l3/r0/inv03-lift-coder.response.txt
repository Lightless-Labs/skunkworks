#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    word_id: usize,
}

struct Line {
    indent: usize,
    pieces: Vec<Piece>,
}

impl Line {
    fn content_len(&self) -> usize {
        self.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn push_piece(&mut self, text: String, word_id: usize) {
        self.pieces.push(Piece { text, word_id });
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &c in segment {
        if is_separator(c) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }

    words
}

fn render_line(line: &Line, options: &WrapOptions, is_final: bool, line_index: usize) -> String {
    let mut out = " ".repeat(line.indent);
    let gaps = line.pieces.len().saturating_sub(1);
    let pre_width = line.indent + line.content_len();

    let justify = options.justify && !is_final && gaps > 0 && pre_width < options.width;
    let (base_extra, remainder, start_gap) = if justify {
        let extra = options.width - pre_width;
        (extra / gaps, extra % gaps, line_index % gaps)
    } else {
        (0, 0, 0)
    };

    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            out.push(' ');
            if justify {
                out.push_str(&" ".repeat(base_extra));
                let cyclic_position = (i - 1 + gaps - start_gap) % gaps;
                if cyclic_position < remainder {
                    out.push(' ');
                }
            }
        }
        out.push_str(&piece.text);
    }

    out
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut lines = Vec::new();
    let mut current = Line {
        indent: options.first_indent,
        pieces: Vec::new(),
    };
    let mut next_indent = options.rest_indent;
    let mut word_id = 0;

    for word in words {
        let mut offset = 0usize;
        let mut first_fragment = true;

        while offset < word.chars.len() {
            let capacity = options.width - current.indent;
            let current_len = current.content_len();
            let nonempty = !current.pieces.is_empty();
            let room = capacity.saturating_sub(current_len + usize::from(nonempty));
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room {
                let text: String = word.chars[offset..].iter().collect();
                current.push_piece(text, word_id);
                offset = word.chars.len();
                continue;
            }

            let available = room;
            let rightmost = word
                .markers
                .iter()
                .copied()
                .filter(|&m| m > offset && m < word.chars.len())
                .filter(|&m| m - offset + 1 <= available)
                .max();

            let accepted = if let Some(marker) = rightmost {
                if !nonempty {
                    Some(marker)
                } else {
                    let used = marker - offset + 1;
                    let unused = capacity - (current_len + 1 + used);
                    let baseline = capacity - current_len;
                    let lhs = (unused as u128) * (unused as u128)
                        + options.hyphen_cost as u128;
                    let rhs = (baseline as u128) * (baseline as u128);
                    (lhs < rhs).then_some(marker)
                }
            } else {
                None
            };

            if let Some(marker) = accepted {
                let mut text: String = word.chars[offset..marker].iter().collect();
                text.push('-');
                current.push_piece(text, word_id);
                lines.push(current);
                current = Line {
                    indent: next_indent,
                    pieces: Vec::new(),
                };
                next_indent = options.rest_indent;
                offset = marker + 1;
                first_fragment = false;
                continue;
            }

            if nonempty {
                lines.push(current);
                current = Line {
                    indent: next_indent,
                    pieces: Vec::new(),
                };
                next_indent = options.rest_indent;
                continue;
            }

            let capacity = options.width - current.indent;
            if suffix_len > capacity {
                let end = offset + capacity;
                let text: String = word.chars[offset..end].iter().collect();
                current.push_piece(text, word_id);
                lines.push(current);
                current = Line {
                    indent: next_indent,
                    pieces: Vec::new(),
                };
                next_indent = options.rest_indent;
                offset = end;
                first_fragment = false;
            } else {
                let text: String = word.chars[offset..].iter().collect();
                current.push_piece(text, word_id);
                offset = word.chars.len();
            }
        }

        word_id += 1;
        let _ = first_fragment;
    }

    if !current.pieces.is_empty() {
        lines.push(current);
    }

    let final_index = lines.len().saturating_sub(1);
    lines
        .iter()
        .enumerate()
        .map(|(i, line)| render_line(line, options, i == final_index, i))
        .collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        let (index, c) = original[i];
        if c != '~' {
            continue;
        }
        let valid_side = |side: Option<char>| {
            side.is_some_and(|ch| ch != '~' && !is_separator(ch))
        };
        let left = i.checked_sub(1).map(|j| original[j].1);
        let right = original.get(i + 1).map(|entry| entry.1);
        if !valid_side(left) || !valid_side(right) {
            return Err(WrapError::InvalidMarker { index });
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let run_len = i - start;
            if run_len >= 2 {
                let words = parse_words(&segment);
                output.extend(layout_paragraph(&words, options));
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(run_len - 1));
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[i]);
            i += 1;
        }
    }

    let words = parse_words(&segment);
    output.extend(layout_paragraph(&words, options));
    Ok(output)
}
