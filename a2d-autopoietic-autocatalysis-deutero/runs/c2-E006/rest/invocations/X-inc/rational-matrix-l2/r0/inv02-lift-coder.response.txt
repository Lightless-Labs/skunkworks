```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let gcd = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / gcd;
        let right_scale = self.den as i128 / gcd;
        let numerator = self.num as i128 * left_scale + rhs.num as i128 * right_scale;
        let denominator = self.den as i128 * left_scale;
        Self::from_i128(numerator, denominator)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let gcd = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / gcd;
        let right_scale = self.den as i128 / gcd;
        let numerator = self.num as i128 * left_scale - rhs.num as i128 * right_scale;
        let denominator = self.den as i128 * left_scale;
        Self::from_i128(numerator, denominator)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let left_abs = self.num.unsigned_abs();
        let right_abs = rhs.num.unsigned_abs();
        let cross_left = gcd_u128(left_abs, rhs.den as u128);
        let cross_right = gcd_u128(right_abs, self.den as u128);

        let numerator = (self.num as i128 / cross_left as i128)
            * (rhs.num as i128 / cross_right as i128);
        let denominator = (self.den as i128 / cross_right as i128)
            * (rhs.den as i128 / cross_left as i128);
        Self::from_i128(numerator, denominator)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let left_abs = self.num.unsigned_abs();
        let right_abs = rhs.num.unsigned_abs();
        let cross_num = gcd_u128(left_abs, right_abs);
        let cross_den = gcd_u128(self.den as u128, rhs.den as u128);

        let numerator = (self.num as i128 / cross_num as i128)
            * (rhs.den as i128 / cross_den as i128);
        let denominator = (self.den as i128 / cross_den as i128)
            * (rhs.num as i128 / cross_num as i128);
        Self::from_i128(numerator, denominator)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let num = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut index = 0;
        let negative = bytes[0] == b'-';
        if negative {
            index = 1;
        }

        if index == bytes.len() {
            return Err(parse_error(index, ParseKind::InvalidByte));
        }

        let numerator_start = index;
        while index < bytes.len() && bytes[index].is_ascii_digit() {
            index += 1;
        }

        if index == numerator_start {
            return Err(parse_error(index, ParseKind::InvalidByte));
        }

        let numerator_end = index;
        let mut denominator_start = None;
        let mut denominator_end = None;

        if index < bytes.len() {
            if bytes[index] != b'/' {
                return Err(parse_error(index, ParseKind::InvalidByte));
            }
            index += 1;
            denominator_start = Some(index);

            if index == bytes.len() {
                return Err(parse_error(index, ParseKind::InvalidByte));
            }

            while index < bytes.len() && bytes[index].is_ascii_digit() {
                index += 1;
            }
            if index == denominator_start.unwrap() {
                return Err(parse_error(index, ParseKind::InvalidByte));
            }

            denominator_end = Some(index);
            if index < bytes.len() {
                return Err(parse_error(index, ParseKind::InvalidByte));
            }
        }

        let numerator_digits = &bytes[numerator_start..numerator_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_error(
                numerator_start + 1,
                ParseKind::NonCanonical,
            ));
        }
        if negative && numerator_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let Some(start) = denominator_start {
            let end = denominator_end.unwrap();
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let numerator_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let numerator_magnitude =
            parse_bounded_magnitude(numerator_digits, numerator_limit)
                .ok_or_else(|| parse_error(0, ParseKind::OutOfRange))?;

        let numerator = if negative {
            if numerator_magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        let Some(start) = denominator_start else {
            return Ok(Self {
                num: numerator,
                den: 1,
            });
        };
        let end = denominator_end.unwrap();
        let denominator_digits = &bytes[start..end];
        let denominator_magnitude =
            parse_bounded_magnitude(denominator_digits, i64::MAX as u128)
                .ok_or_else(|| parse_error(start, ParseKind::OutOfRange))?;

        if denominator_magnitude == 0 {
            return Err(parse_error(start, ParseKind::ZeroDenominator));
        }

        let slash = start - 1;
        if denominator_magnitude == 1 {
            return Err(parse_error(slash, ParseKind::NonCanonical));
        }
        if numerator == 0 {
            return Err(parse_error(slash, ParseKind::NonCanonical));
        }
        if gcd_u128(numerator.unsigned_abs() as u128, denominator_magnitude) > 1 {
            return Err(parse_error(slash, ParseKind::NonCanonical));
        }

        Ok(Self {
            num: numerator,
            den: denominator_magnitude as i64,
        })
    }

    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let divisor = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= divisor;
        den /= divisor;

        if num < i64::MIN as i128
            || num > i64::MAX as i128
            || den < 1
            || den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let remainder = a % b;
        a = b;
        b = remainder;
    }
    a
}

fn parse_bounded_magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &digit in digits {
        let digit = (digit - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for col in 0..self.cols {
            for row in 0..self.rows {
                data.push(self.data[row * self.cols + col]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;

        let zero = Rational { num: 0, den: 1 };
        let mut data = Vec::with_capacity(output_len);

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut accumulator = zero;
                for step in 0..self.cols {
                    let left_index = row * self.cols + step;
                    let right_index = step * rhs.cols + col;
                    let term = self.data[left_index]
                        .mul(&rhs.data[right_index])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col,
                            step,
                            source,
                        })?;
                    accumulator = accumulator.add(&term).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step,
                            source,
                        }
                    })?;
                }
                data.push(accumulator);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
```
