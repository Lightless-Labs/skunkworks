```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        let gcd = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= gcd;
        den /= gcd;

        if num < i64::MIN as i128
            || num > i64::MAX as i128
            || den < 1
            || den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let c = self.den as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(c as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = c / g;
        let num = a * left_scale + b * right_scale;
        let den = c * left_scale;
        Self::from_i128(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let c = self.den as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(c as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = c / g;
        let num = a * left_scale - b * right_scale;
        let den = c * left_scale;
        Self::from_i128(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let mut a = self.num as i128;
        let mut b = rhs.num as i128;
        let mut c = self.den as i128;
        let mut d = rhs.den as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(b.unsigned_abs(), c as u128) as i128;
        a /= g1;
        d /= g1;
        b /= g2;
        c /= g2;

        Self::from_i128(a * b, c * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let mut a = self.num as i128;
        let mut b = rhs.den as i128;
        let mut c = self.den as i128;
        let mut d = rhs.num as i128;

        if d < 0 {
            b = -b;
            d = -d;
        }

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(b as u128, c as u128) as i128;
        a /= g1;
        d /= g1;
        b /= g2;
        c /= g2;

        Self::from_i128(a * b, c * d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        Self::from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &byte) in bytes.iter().enumerate() {
            if byte == b'/' {
                if slash.is_some() {
                    return Err(parse_error(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if byte.is_ascii_digit() {
                continue;
            } else if byte == b'-' && i == 0 {
                continue;
            } else {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let numerator_end = slash.unwrap_or(bytes.len());
        let numerator_start = if bytes[0] == b'-' { 1 } else { 0 };
        if numerator_start == numerator_end {
            return Err(parse_error(bytes.len(), ParseKind::InvalidByte));
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(parse_error(bytes.len(), ParseKind::InvalidByte));
            }
        }

        let num_digits = &bytes[numerator_start..numerator_end];
        let den_digits = slash.map(|s| &bytes[s + 1..]);

        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits.len() == 1 && num_digits[0] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }
        if let Some(digits) = den_digits {
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_error(slash.unwrap() + 2, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if bytes[0] == b'-' {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_magnitude = parse_magnitude(num_digits);
        if num_magnitude > num_limit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let den_magnitude = if let Some(digits) = den_digits {
            let value = parse_magnitude(digits);
            if value > i64::MAX as u128 {
                return Err(parse_error(slash.unwrap() + 1, ParseKind::OutOfRange));
            }
            Some(value)
        } else {
            None
        };

        if let (Some(value), Some(s)) = (den_magnitude, slash) {
            if value == 0 {
                return Err(parse_error(s + 1, ParseKind::ZeroDenominator));
            }
        }

        if let (Some(value), Some(s)) = (den_magnitude, slash) {
            if value == 1 {
                return Err(parse_error(s, ParseKind::NonCanonical));
            }
            if num_magnitude == 0 {
                return Err(parse_error(s, ParseKind::NonCanonical));
            }
        }

        if let (Some(value), Some(s)) = (den_magnitude, slash) {
            if gcd_u128(num_magnitude, value) > 1 {
                return Err(parse_error(s, ParseKind::NonCanonical));
            }
        }

        let signed_num = if bytes[0] == b'-' {
            if num_magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_magnitude as i64)
            }
        } else {
            num_magnitude as i64
        };
        let den = den_magnitude.unwrap_or(1) as i64;
        Ok(Self {
            num: signed_num,
            den,
        })
    }
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn parse_magnitude(digits: &[u8]) -> u128 {
    let mut value = 0u128;
    for &byte in digits {
        value = value
            .saturating_mul(10)
            .saturating_add((byte - b'0') as u128);
    }
    value
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(
        rows: usize,
        cols: usize,
        data: Vec<Rational>,
    ) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;

        let mut data = Vec::with_capacity(output_len);
        if output_len == 0 {
            return Ok(Self {
                rows: self.rows,
                cols: rhs.cols,
                data,
            });
        }

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
```
