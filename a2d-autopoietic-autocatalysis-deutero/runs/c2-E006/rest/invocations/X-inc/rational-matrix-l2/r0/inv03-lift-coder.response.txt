use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(parse_err(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if b.is_ascii_digit() {
                continue;
            } else if b == b'-' && i == 0 {
                continue;
            } else {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let numerator_end = slash.unwrap_or(bytes.len());
        let numerator_start = usize::from(bytes[0] == b'-');
        if numerator_start == numerator_end {
            return Err(parse_err(text.len(), ParseKind::InvalidByte));
        }

        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(parse_err(text.len(), ParseKind::InvalidByte));
            }
        }

        let numerator_digits = &bytes[numerator_start..numerator_end];
        let denominator_digits = slash.map(|s| &bytes[s + 1..]);

        if numerator_digits.len() > 1 && numerator_digits[0] == b'0' {
            return Err(parse_err(numerator_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && numerator_digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let Some(digits) = denominator_digits {
            if digits.len() > 1 && digits[0] == b'0' {
                return Err(parse_err(slash.unwrap() + 2, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let limit: u128 = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut numerator_mag = 0u128;
        let mut numerator_out = false;
        for &b in numerator_digits {
            let digit = (b - b'0') as u128;
            if numerator_mag > (limit.saturating_sub(digit)) / 10 {
                numerator_out = true;
                break;
            }
            numerator_mag = numerator_mag * 10 + digit;
        }
        if numerator_out {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }

        let mut denominator = 1u128;
        if let Some(digits) = denominator_digits {
            let mut value = 0u128;
            for &b in digits {
                let digit = (b - b'0') as u128;
                if value > ((i64::MAX as u128).saturating_sub(digit)) / 10 {
                    return Err(parse_err(slash.unwrap() + 1, ParseKind::OutOfRange));
                }
                value = value * 10 + digit;
            }
            denominator = value;
        }

        if let Some(digits) = denominator_digits {
            if digits == b"0" {
                return Err(parse_err(slash.unwrap() + 1, ParseKind::ZeroDenominator));
            }
            if digits == b"1" {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if numerator_mag == 0 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_mag, denominator) > 1 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
        }

        let num = if negative {
            if numerator_mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(numerator_mag as i64)
            }
        } else {
            numerator_mag as i64
        };
        Ok(Self { num, den: denominator as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row * self.cols + col;
        Ok(self.data[index])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for col in 0..self.cols {
            for row in 0..self.rows {
                data.push(self.data[row * self.cols + col]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let output_len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);
        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut sum = Rational { num: 0, den: 1 };
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                    sum = sum.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(sum);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }
}
