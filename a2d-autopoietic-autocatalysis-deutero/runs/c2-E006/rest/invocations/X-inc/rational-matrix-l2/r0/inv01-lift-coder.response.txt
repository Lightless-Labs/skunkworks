use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let mut n = num;
        let mut d = den;
        if d < 0 {
            n = -n;
            d = -d;
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128
            || n > i64::MAX as i128
            || d < 1
            || d > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    fn from_i128_pair(num: i128, den: i128) -> Result<Self, RationalError> {
        Self::from_i128(num, den)
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;

        let g = gcd_u128(b as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = b / g;
        let n = a * left_scale + c * right_scale;
        let den = b * left_scale;
        Self::from_i128_pair(n, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;

        let g = gcd_u128(b as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = b / g;
        let n = a * left_scale - c * right_scale;
        let den = b * left_scale;
        Self::from_i128_pair(n, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let mut a = self.num as i128;
        let mut b = self.den as i128;
        let mut c = rhs.num as i128;
        let mut d = rhs.den as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(c.unsigned_abs(), b as u128) as i128;
        a /= g1;
        d /= g1;
        c /= g2;
        b /= g2;

        Self::from_i128_pair(a * c, b * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let mut a = self.num as i128;
        let mut b = self.den as i128;
        let mut c = rhs.den as i128;
        let mut d = rhs.num as i128;

        if d < 0 {
            c = -c;
            d = -d;
        }

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(c as u128, b as u128) as i128;
        a /= g1;
        d /= g1;
        c /= g2;
        b /= g2;

        Self::from_i128_pair(a * c, b * d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        Self::from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let mut slash = None;
        let mut denominator_start = 0;
        let mut denominator_end = 0;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            denominator_start = i;

            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            denominator_end = i;

            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';

        if numerator_end - numerator_start >= 2 && bytes[numerator_start] == b'0' {
            return Err(parse_err(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && numerator_end - numerator_start == 1 && bytes[numerator_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if slash.is_some()
            && denominator_end - denominator_start >= 2
            && bytes[denominator_start] == b'0'
        {
            return Err(parse_err(denominator_start + 1, ParseKind::NonCanonical));
        }

        let numerator_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0u128;
        for &byte in &bytes[numerator_start..numerator_end] {
            numerator_magnitude = numerator_magnitude * 10 + (byte - b'0') as u128;
            if numerator_magnitude > numerator_limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let mut denominator_magnitude = 1u128;
        if slash.is_some() {
            denominator_magnitude = 0;
            for &byte in &bytes[denominator_start..denominator_end] {
                denominator_magnitude = denominator_magnitude * 10 + (byte - b'0') as u128;
                if denominator_magnitude > i64::MAX as u128 {
                    return Err(parse_err(denominator_start, ParseKind::OutOfRange));
                }
            }
            if denominator_magnitude == 0 {
                return Err(parse_err(denominator_start, ParseKind::ZeroDenominator));
            }
        }

        if let Some(slash_at) = slash {
            if denominator_magnitude == 1 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, denominator_magnitude) > 1 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
        }

        let numerator = if negative {
            if numerator_magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        if slash.is_some() {
            Ok(Self {
                num: numerator,
                den: denominator_magnitude as i64,
            })
        } else {
            Ok(Self {
                num: numerator,
                den: 1,
            })
        }
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(
        rows: usize,
        cols: usize,
        data: Vec<Rational>,
    ) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for col in 0..self.cols {
            for row in 0..self.rows {
                data.push(self.data[row * self.cols + col]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for step in 0..self.cols {
                    let left_index = row * self.cols + step;
                    let right_index = step * rhs.cols + col;

                    let term = self.data[left_index]
                        .mul(&rhs.data[right_index])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col,
                            step,
                            source,
                        })?;

                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
