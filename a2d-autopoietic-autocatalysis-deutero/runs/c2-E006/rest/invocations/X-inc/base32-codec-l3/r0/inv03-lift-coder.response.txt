#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    MissingCheck { offset: usize },
    InvalidByte { offset: usize, byte: u8 },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    CheckMismatch { offset: usize, expected: u8, found: u8 },
}

const DATA_ALPHABET: &[u8; 32] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
const CHECK_ALPHABET: &[u8; 37] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";

pub fn encode(input: &[u8], checked: bool) -> String {
    let mut output = String::new();
    let mut remainder = 0u8;
    let mut accumulator = 0u16;
    let mut bits = 0u8;

    for &byte in input {
        accumulator = (accumulator << 8) | byte as u16;
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let value = ((accumulator >> bits) & 0x1f) as u8;
            output.push(DATA_ALPHABET[value as usize] as char);
            remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
        }
    }

    if bits != 0 {
        let value = (((accumulator << (5 - bits)) & 0x1f) as u8);
        output.push(DATA_ALPHABET[value as usize] as char);
        remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
    }

    if checked {
        output.push(CHECK_ALPHABET[remainder as usize] as char);
    }

    output
}

fn data_value(byte: u8) -> Option<u8> {
    match byte.to_ascii_uppercase() {
        b'0' | b'O' => Some(0),
        b'1' | b'I' | b'L' => Some(1),
        b'2'..=b'9' => Some(byte - b'0'),
        b'A'..=b'H' => Some(byte.to_ascii_uppercase() - b'A' + 10),
        b'J'..=b'K' => Some(byte.to_ascii_uppercase() - b'J' + 18),
        b'M'..=b'N' => Some(byte.to_ascii_uppercase() - b'M' + 20),
        b'P'..=b'T' => Some(byte.to_ascii_uppercase() - b'P' + 22),
        b'V'..=b'Z' => Some(byte.to_ascii_uppercase() - b'V' + 27),
        _ => None,
    }
}

fn check_value(byte: u8) -> Option<u8> {
    match byte {
        b'*' => Some(32),
        b'~' => Some(33),
        b'$' => Some(34),
        b'=' => Some(35),
        b'U' | b'u' => Some(36),
        _ => data_value(byte),
    }
}

pub fn decode(input: &str, checked: bool) -> Result<Vec<u8>, DecodeError> {
    let significant: Vec<(usize, u8)> = input
        .bytes()
        .enumerate()
        .filter(|&(_, byte)| byte != b'-')
        .collect();

    if checked && significant.is_empty() {
        return Err(DecodeError::MissingCheck {
            offset: input.len(),
        });
    }

    let data_end = if checked {
        significant.len() - 1
    } else {
        significant.len()
    };

    let mut values = Vec::with_capacity(data_end);
    for &(offset, byte) in &significant[..data_end] {
        match data_value(byte) {
            Some(value) => values.push((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    }

    let reserved_check = if checked {
        let (offset, byte) = significant[data_end];
        match check_value(byte) {
            Some(value) => Some((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    } else {
        None
    };

    let residue = values.len() % 8;
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => {
            return Err(DecodeError::InvalidLength {
                offset: input.len(),
            });
        }
    };

    if unused_bits != 0 {
        let (offset, value) = values[values.len() - 1];
        if value & ((1 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset });
        }
    }

    let mut remainder = 0u8;
    for &(_, value) in &values {
        remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
    }

    if let Some((offset, found)) = reserved_check {
        if found != remainder {
            return Err(DecodeError::CheckMismatch {
                offset,
                expected: remainder,
                found,
            });
        }
    }

    let mut output = Vec::with_capacity(values.len() * 5 / 8);
    let mut accumulator = 0u16;
    let mut bits = 0u8;

    for &(_, value) in &values {
        accumulator = (accumulator << 5) | value as u16;
        bits += 5;
        if bits >= 8 {
            bits -= 8;
            output.push((accumulator >> bits) as u8);
        }
    }

    Ok(output)
}
