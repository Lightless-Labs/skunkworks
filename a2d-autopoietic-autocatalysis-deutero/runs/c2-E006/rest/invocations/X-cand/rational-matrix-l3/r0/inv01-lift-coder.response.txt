```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_fraction(mut num: i128, mut den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    if den < 0 {
        num = num.checked_neg().ok_or(RationalError::Overflow)?;
        den = den.checked_neg().ok_or(RationalError::Overflow)?;
    }
    let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    num /= g;
    den /= g;
    if num < i64::MIN as i128 || num > i64::MAX as i128 || den < 1 || den > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: num as i64,
        den: den as i64,
    })
}

fn from_i128_pair(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_i128_pair(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = (self.num as i128) * a + (rhs.num as i128) * b;
        let d = (self.den as i128) * a;
        from_i128_pair(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = (self.num as i128) * a - (rhs.num as i128) * b;
        let d = (self.den as i128) * a;
        from_i128_pair(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs(), self.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.num as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.den as i128 / g1);
        from_i128_pair(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.num.unsigned_abs()) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.den as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.num as i128 / g1);
        from_i128_pair(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num: n, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
            if i == bytes.len() || !bytes[i].is_ascii_digit() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
        } else if !bytes[i].is_ascii_digit() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::InvalidByte });
        }
        let nstart = if bytes[0] == b'-' { 1 } else { 0 };
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        let nend = i;
        let mut slash = None;
        let mut dstart = 0;
        let mut dend = 0;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            slash = Some(i);
            i += 1;
            dstart = i;
            if i == bytes.len() || !bytes[i].is_ascii_digit() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            dend = i;
            if i < bytes.len() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
        }

        let negative = bytes[0] == b'-';
        let ndigits = nend - nstart;
        if ndigits >= 2 && bytes[nstart] == b'0' {
            return Err(RationalError::Parse { at: nstart + 1, kind: ParseKind::NonCanonical });
        }
        if negative && ndigits == 1 && bytes[nstart] == b'0' {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::NonCanonical });
        }
        if slash.is_some() && dend - dstart >= 2 && bytes[dstart] == b'0' {
            return Err(RationalError::Parse { at: dstart + 1, kind: ParseKind::NonCanonical });
        }

        let nmag = parse_bounded_magnitude(&bytes[nstart..nend], if negative { 1u128 << 63 } else { i64::MAX as u128 });
        let nmag = match nmag {
            Some(v) => v,
            None => return Err(RationalError::Parse { at: 0, kind: ParseKind::OutOfRange }),
        };
        let num = if negative {
            if nmag == (1u128 << 63) { i64::MIN } else { -(nmag as i64) }
        } else {
            nmag as i64
        };

        if let Some(s) = slash {
            let dmag = match parse_bounded_magnitude(&bytes[dstart..dend], i64::MAX as u128) {
                Some(v) => v,
                None => return Err(RationalError::Parse { at: dstart, kind: ParseKind::OutOfRange }),
            };
            if dmag == 0 {
                return Err(RationalError::Parse { at: dstart, kind: ParseKind::ZeroDenominator });
            }
            if dmag == 1 {
                return Err(RationalError::Parse { at: s, kind: ParseKind::NonCanonical });
            }
            if num == 0 {
                return Err(RationalError::Parse { at: s, kind: ParseKind::NonCanonical });
            }
            if gcd_u128(num.unsigned_abs() as u128, dmag) > 1 {
                return Err(RationalError::Parse { at: s, kind: ParseKind::NonCanonical });
            }
            return Ok(Rational { num, den: dmag as i64 });
        }
        Ok(Rational { num, den: 1 })
    }
}

fn parse_bounded_magnitude(bytes: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        let d = (b - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(d)?;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let aug = Matrix { rows: n, cols: width, data };
        let red = eliminate(aug, n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            result.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Matrix { rows: n, cols: n, data: result })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

fn eliminate(mut matrix: Matrix, pivot_cols: usize) -> Result<Reduction, MatrixError> {
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..pivot_cols {
        if p == matrix.rows {
            break;
        }
        let mut chosen: Option<usize> = None;
        for row in p..matrix.rows {
            let v = matrix.data[row * matrix.cols + col];
            if v.num == 0 {
                continue;
            }
            let key = (v.den, v.num.unsigned_abs(), row);
            if chosen.map_or(true, |old| {
                let o = matrix.data[old * matrix.cols + col];
                key < (o.den, o.num.unsigned_abs(), old)
            }) {
                chosen = Some(row);
            }
        }
        let Some(q) = chosen else { continue };
        if q != p {
            for c in 0..matrix.cols {
                matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * matrix.cols + col];
        for c in 0..matrix.cols {
            let idx = p * matrix.cols + c;
            matrix.data[idx] = matrix.data[idx].div(&pivot).map_err(|source| MatrixError::Arithmetic {
                row: p, col: c, step: p, source,
            })?;
        }

        for row in 0..matrix.rows {
            if row == p { continue; }
            let factor = matrix.data[row * matrix.cols + col];
            if factor.num == 0 { continue; }
            for c in 0..matrix.cols {
                let pivot_entry = matrix.data[p * matrix.cols + c];
                let product = factor.mul(&pivot_entry).map_err(|source| MatrixError::Arithmetic {
                    row, col: c, step: p, source,
                })?;
                let idx = row * matrix.cols + c;
                matrix.data[idx] = matrix.data[idx].sub(&product).map_err(|source| MatrixError::Arithmetic {
                    row, col: c, step: p, source,
                })?;
            }
        }

        pivots.push(col);
        p += 1;
    }

    Ok(Reduction { matrix, pivots, swaps })
}
```
