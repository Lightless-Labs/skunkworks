use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        rational_from_i128(a * d + c * b, b * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        rational_from_i128(a * d - c * b, b * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i += 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }
        let numerator_end = i;

        let mut slash = None;
        let mut denominator_start = None;
        let mut denominator_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            let start = i;
            denominator_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            denominator_end = Some(i);
        }

        if i < bytes.len() {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_digits = &bytes[numerator_start..numerator_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_error(
                numerator_start + 1,
                ParseKind::NonCanonical,
            ));
        }
        if negative && numerator_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            let digits = &bytes[start..end];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let numerator_limit: u128 = if negative {
            1_u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0_u128;
        for &b in numerator_digits {
            numerator_magnitude = numerator_magnitude * 10 + (b - b'0') as u128;
            if numerator_magnitude > numerator_limit {
                return Err(parse_error(0, ParseKind::OutOfRange));
            }
        }

        let mut denominator_magnitude = 1_u128;
        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            for &b in &bytes[start..end] {
                denominator_magnitude =
                    denominator_magnitude * 10 + (b - b'0') as u128;
                if denominator_magnitude > i64::MAX as u128 {
                    return Err(parse_error(start, ParseKind::OutOfRange));
                }
            }

            if denominator_magnitude == 0 {
                return Err(parse_error(start, ParseKind::ZeroDenominator));
            }

            let slash_at = slash.unwrap();
            if denominator_magnitude == 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, denominator_magnitude) > 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
        }

        let n = if negative {
            if numerator_magnitude == (1_u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        Ok(Rational {
            num: n,
            den: denominator_magnitude as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut accumulator = zero;
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                    accumulator = accumulator
                        .add(&term)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col,
                            step,
                            source,
                        })?;
                }
                data.push(accumulator);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.reduce(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for row in 0..n {
            for col in 0..n {
                data.push(self.data[row * n + col]);
            }
            for col in 0..n {
                data.push(if row == col {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }

        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduction = augmented.reduce(n)?;
        let rank = reduction.rank();
        if rank < n {
            return Err(MatrixError::Singular { rank });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for row in 0..n {
            for col in 0..n {
                inverse_data.push(reduction.matrix.data[row * width + n + col]);
            }
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }

    fn reduce(&self, pivot_columns: usize) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;

        for pivot_col in 0..pivot_columns {
            if p == matrix.rows {
                break;
            }

            let mut selected: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + pivot_col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), row);
                if selected
                    .map(|chosen| {
                        let current = matrix.data[chosen * matrix.cols + pivot_col];
                        key < (current.den, current.num.unsigned_abs(), chosen)
                    })
                    .unwrap_or(true)
                {
                    selected = Some(row);
                }
            }

            let Some(q) = selected else {
                continue;
            };

            if q != p {
                for col in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + col, q * matrix.cols + col);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + pivot_col];
            for col in 0..matrix.cols {
                let index = p * matrix.cols + col;
                matrix.data[index] =
                    matrix.data[index]
                        .div(&pivot)
                        .map_err(|source| MatrixError::Arithmetic {
                            row: p,
                            col,
                            step: p,
                            source,
                        })?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + pivot_col];
                if factor.num == 0 {
                    continue;
                }
                for col in 0..matrix.cols {
                    let pivot_entry = matrix.data[p * matrix.cols + col];
                    let product = factor.mul(&pivot_entry).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step: p,
                            source,
                        }
                    })?;
                    let index = row * matrix.cols + col;
                    matrix.data[index] = matrix.data[index].sub(&product).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step: p,
                            source,
                        }
                    })?;
                }
            }

            pivots.push(pivot_col);
            p += 1;
        }

        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}
