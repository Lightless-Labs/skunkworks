use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{self:?}")
    }
}

impl std::error::Error for Error {}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn digit(byte: u8, alphabet: Alphabet, strict: bool, offset: usize) -> Result<u8, Error> {
    let value = match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    };
    if let Some(value) = value {
        return Ok(value);
    }
    let foreign = match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    };
    if strict && foreign {
        Err(Error::MixedAlphabet { offset })
    } else {
        Err(Error::InvalidByte { offset, byte })
    }
}

fn encode_digit(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 if alphabet == Alphabet::Standard => b'+',
        63 if alphabet == Alphabet::Standard => b'/',
        62 => b'-',
        _ => b'_',
    }
}

fn emit_symbol(out: &mut String, byte: u8) {
    out.push(byte as char);
}

fn append_wrapped_symbol(
    out: &mut String,
    symbol: u8,
    whitespace: Whitespace,
    column: &mut usize,
) {
    if let Whitespace::Wrapped { width, ending } = whitespace {
        if *column == width {
            match ending {
                LineEnding::Lf => out.push('\n'),
                LineEnding::CrLf => out.push_str("\r\n"),
            }
            *column = 0;
        }
        emit_symbol(out, symbol);
        *column += 1;
    } else {
        emit_symbol(out, symbol);
    }
}

fn encode_triple(a: u8, b: u8, c: u8, alphabet: Alphabet) -> [u8; 4] {
    [
        encode_digit(a >> 2, alphabet),
        encode_digit(((a & 0x03) << 4) | (b >> 4), alphabet),
        encode_digit(((b & 0x0f) << 2) | (c >> 6), alphabet),
        encode_digit(c & 0x3f, alphabet),
    ]
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut output = encoder.push(input)?;
    output.push_str(&encoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    options: Options,
    buffer: Vec<u8>,
    closed: bool,
    column: usize,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffer: Vec::with_capacity(2),
            closed: false,
            column: 0,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut i = 0;

        if !self.buffer.is_empty() {
            while self.buffer.len() < 3 && i < input.len() {
                self.buffer.push(input[i]);
                i += 1;
            }
            if self.buffer.len() == 3 {
                let symbols = encode_triple(
                    self.buffer[0],
                    self.buffer[1],
                    self.buffer[2],
                    self.options.alphabet,
                );
                for symbol in symbols {
                    append_wrapped_symbol(
                        &mut output,
                        symbol,
                        self.options.whitespace,
                        &mut self.column,
                    );
                }
                self.buffer.clear();
            }
        }

        while i + 3 <= input.len() {
            let symbols = encode_triple(
                input[i],
                input[i + 1],
                input[i + 2],
                self.options.alphabet,
            );
            for symbol in symbols {
                append_wrapped_symbol(
                    &mut output,
                    symbol,
                    self.options.whitespace,
                    &mut self.column,
                );
            }
            i += 3;
        }

        self.buffer.extend_from_slice(&input[i..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;

        let mut output = String::new();
        if self.buffer.len() == 1 {
            let a = self.buffer[0];
            let symbols = [
                encode_digit(a >> 2, self.options.alphabet),
                encode_digit((a & 0x03) << 4, self.options.alphabet),
            ];
            for symbol in symbols {
                append_wrapped_symbol(
                    &mut output,
                    symbol,
                    self.options.whitespace,
                    &mut self.column,
                );
            }
            if self.options.padding != Padding::Forbidden {
                for _ in 0..2 {
                    append_wrapped_symbol(
                        &mut output,
                        b'=',
                        self.options.whitespace,
                        &mut self.column,
                    );
                }
            }
        } else if self.buffer.len() == 2 {
            let a = self.buffer[0];
            let b = self.buffer[1];
            let symbols = [
                encode_digit(a >> 2, self.options.alphabet),
                encode_digit(((a & 0x03) << 4) | (b >> 4), self.options.alphabet),
                encode_digit((b & 0x0f) << 2, self.options.alphabet),
            ];
            for symbol in symbols {
                append_wrapped_symbol(
                    &mut output,
                    symbol,
                    self.options.whitespace,
                    &mut self.column,
                );
            }
            if self.options.padding != Padding::Forbidden {
                append_wrapped_symbol(
                    &mut output,
                    b'=',
                    self.options.whitespace,
                    &mut self.column,
                );
            }
        }
        self.buffer.clear();
        Ok(output)
    }
}

#[derive(Clone, Copy)]
enum FrameState {
    Row,
    NeedLf { cr_offset: usize, sep_offset: usize },
    NeedSymbol { sep_offset: usize },
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbols_seen: usize,
    group: Vec<(u8, usize)>,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    frame: FrameState,
    closed: bool,
    sticky: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols_seen: 0,
            group: Vec::with_capacity(4),
            pending_second_padding: None,
            terminal_padding: false,
            frame: FrameState::Row,
            closed: false,
            sticky: None,
        })
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn consume_symbol(&mut self, byte: u8, at: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols_seen >= limit {
                return Err(Error::Overlong { offset: at, limit });
            }
        }
        self.symbols_seen = self.symbols_seen.saturating_add(1);

        if self.terminal_padding {
            return Err(Error::BadPadding { offset: at });
        }
        if self.pending_second_padding.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset: at });
        }

        let position = self.group.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden || position < 2 || position > 3 {
                return Err(Error::BadPadding { offset: at });
            }
            if position == 2 {
                self.pending_second_padding = Some(at);
                self.group.push((byte, at));
                return Ok(());
            }

            let second_pad = self.pending_second_padding.take();
            let vals = self.group.iter().map(|(b, _)| *b).collect::<Vec<_>>();
            if vals.len() == 2 && vals[1] == b'=' {
                let a = digit(vals[0], self.options.alphabet, self.options.strict, self.group[0].1)?;
                let b = digit(vals[1], self.options.alphabet, self.options.strict, self.group[1].1)?;
                if self.options.strict && b & 0x0f != 0 {
                    return Err(Error::NonCanonical { offset: self.group[1].1 });
                }
                output.push((a << 2) | (b >> 4));
            } else if vals.len() == 3 && vals[2] != b'=' {
                let a = digit(vals[0], self.options.alphabet, self.options.strict, self.group[0].1)?;
                let b = digit(vals[1], self.options.alphabet, self.options.strict, self.group[1].1)?;
                let c = digit(vals[2], self.options.alphabet, self.options.strict, self.group[2].1)?;
                if self.options.strict && c & 0x03 != 0 {
                    return Err(Error::NonCanonical { offset: self.group[2].1 });
                }
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
            } else {
                return Err(Error::BadPadding { offset: at });
            }
            self.group.clear();
            self.pending_second_padding = None;
            self.terminal_padding = true;
            let _ = second_pad;
            return Ok(());
        }

        let value = digit(byte, self.options.alphabet, self.options.strict, at)?;
        self.group.push((byte, at));
        if self.group.len() == 4 {
            let a = digit(self.group[0].0, self.options.alphabet, self.options.strict, self.group[0].1)?;
            let b = digit(self.group[1].0, self.options.alphabet, self.options.strict, self.group[1].1)?;
            let c = digit(self.group[2].0, self.options.alphabet, self.options.strict, self.group[2].1)?;
            let d = value;
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
            self.group.clear();
        }
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, at: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => {}
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Ok(());
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.frame {
                    FrameState::NeedLf { cr_offset, sep_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset: at });
                        }
                        self.frame = FrameState::NeedSymbol { sep_offset };
                        let _ = cr_offset;
                        return Ok(());
                    }
                    FrameState::NeedSymbol { .. } => {
                        self.frame = FrameState::Row;
                    }
                    FrameState::Row => {}
                }

                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset: at });
                }

                let col = self.symbols_seen % width;
                if col == 0 && self.symbols_seen != 0 {
                    let sep_offset = at;
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.frame = FrameState::NeedSymbol { sep_offset };
                            return Ok(());
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.frame = FrameState::NeedLf {
                                cr_offset: at,
                                sep_offset,
                            };
                            return Ok(());
                        }
                        _ => return Err(Error::BadWrap { offset: at }),
                    }
                }
                if matches!(self.frame, FrameState::NeedSymbol { .. }) {
                    self.frame = FrameState::Row;
                }
            }
        }

        self.consume_symbol(byte, at, output)
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        match self.frame {
            FrameState::NeedLf { cr_offset, .. } => return Err(Error::BadWrap { offset: cr_offset }),
            FrameState::NeedSymbol { sep_offset } => return Err(Error::BadWrap { offset: sep_offset }),
            FrameState::Row => {}
        }

        if let Some(offset) = self.pending_second_padding {
            return Err(Error::BadPadding { offset });
        }

        let mut output = Vec::new();
        match self.group.len() {
            0 => {}
            1 => return Err(Error::Truncated { offset: self.group[0].1 }),
            2 => {
                if self.options.padding == Padding::Required {
                    return Err(Error::Truncated { offset: self.group[0].1 });
                }
                let a = digit(self.group[0].0, self.options.alphabet, self.options.strict, self.group[0].1)?;
                let b = digit(self.group[1].0, self.options.alphabet, self.options.strict, self.group[1].1)?;
                if self.options.strict && b & 0x0f != 0 {
                    return Err(Error::NonCanonical { offset: self.group[1].1 });
                }
                output.push((a << 2) | (b >> 4));
            }
            3 => {
                if self.options.padding == Padding::Required {
                    return Err(Error::Truncated { offset: self.group[0].1 });
                }
                let a = digit(self.group[0].0, self.options.alphabet, self.options.strict, self.group[0].1)?;
                let b = digit(self.group[1].0, self.options.alphabet, self.options.strict, self.group[1].1)?;
                let c = digit(self.group[2].0, self.options.alphabet, self.options.strict, self.group[2].1)?;
                if self.options.strict && c & 0x03 != 0 {
                    return Err(Error::NonCanonical { offset: self.group[2].1 });
                }
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
            }
            _ => unreachable!(),
        }
        self.group.clear();
        Ok(output)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = self.sticky.clone() {
            return Err(error);
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let at = self.offset;
            self.offset = self.offset.saturating_add(1);
            if let Err(error) = self.process_byte(byte, at, &mut output) {
                return self.fail(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = self.sticky.clone() {
            return Err(error);
        }
        if self.closed {
            return self.fail(Error::AlreadyFinished);
        }
        match self.finish_inner() {
            Ok(output) => {
                self.closed = true;
                Ok(output)
            }
            Err(error) => self.fail(error),
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
