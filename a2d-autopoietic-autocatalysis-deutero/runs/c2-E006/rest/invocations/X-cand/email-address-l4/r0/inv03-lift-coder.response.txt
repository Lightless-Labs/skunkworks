use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, false)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let parsed = parse_with_policy(input, policy.allow_single_label)?;
    normalize_parsed(input, parsed, policy)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = keys.get(&key) {
                    result.duplicate_of[index] = Some(first_index);
                } else {
                    keys.insert(key, index);
                    result.addresses.push(address);
                    result.source_indices.push(index);
                }
            }
        }
    }
    result
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize, usize)>, // cleaned insertion point, original start, original end
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &str) -> Result<Cleaned, EmailError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut positions = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut in_quote = false;
    let mut quote_closed = false;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;

    while i < bytes.len() {
        if in_quote {
            let b = bytes[i];
            out.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    out.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    in_quote = false;
                    quote_closed = true;
                }
                i += 1;
            }
            continue;
        }

        if comment_depth > 0 {
            let b = bytes[i];
            if b == b'\\' {
                let slash = i;
                if i + 1 >= bytes.len() || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: slash });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(bytes, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if !(0x20..=0x7e).contains(&b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let b = bytes[i];
        if is_hws(b) || b == b'(' || b == b'\r' || b == b'\n' {
            let start = i;
            let insertion = out.len();
            let mut end = i;

            loop {
                if end >= bytes.len() {
                    break;
                }
                if is_hws(bytes[end]) {
                    end += 1;
                    continue;
                }
                if bytes[end] == b'\r' || bytes[end] == b'\n' {
                    if let Some(next) = fold_end(bytes, end) {
                        end = next;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: end });
                }
                if bytes[end] == b'(' {
                    outer_comment = end;
                    comment_depth = 1;
                    end += 1;
                    while end < bytes.len() && comment_depth > 0 {
                        let cb = bytes[end];
                        if cb == b'\\' {
                            if end + 1 >= bytes.len()
                                || !(0x20..=0x7e).contains(&bytes[end + 1])
                            {
                                return Err(EmailError::CommentEscape { at: end });
                            }
                            end += 2;
                            continue;
                        }
                        if cb == b'\r' || cb == b'\n' {
                            if let Some(next) = fold_end(bytes, end) {
                                end = next;
                                continue;
                            }
                            return Err(EmailError::BadFold { at: end });
                        }
                        if cb == b'(' {
                            if comment_depth == 3 {
                                return Err(EmailError::CommentDepth { at: end });
                            }
                            comment_depth += 1;
                            end += 1;
                            continue;
                        }
                        if cb == b')' {
                            comment_depth -= 1;
                            end += 1;
                            continue;
                        }
                        if !(0x20..=0x7e).contains(&cb) && cb != b'\t' {
                            return Err(EmailError::CommentChar { at: end });
                        }
                        end += 1;
                    }
                    if comment_depth > 0 {
                        return Err(EmailError::UnclosedComment { at: outer_comment });
                    }
                    continue;
                }
                break;
            }

            i = end;
            runs.push((insertion, start, end));
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if !quote_closed && out.is_empty() && b == b'"' {
            in_quote = true;
            out.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        out.push(b);
        positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    for &(point, run_start, _) in &runs {
        let at_start = point == 0;
        let at_end = point == out.len();
        let adjacent_at = (point > 0 && out[point - 1] == b'@')
            || (point < out.len() && out[point] == b'@');
        if !at_start && !at_end && !adjacent_at {
            return Err(EmailError::CfwsBoundary { at: run_start });
        }
    }

    Ok(Cleaned {
        bytes: out,
        positions,
        runs,
    })
}

fn original_pos(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn parse_with_policy(input: &str, allow_single_label: bool) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(input)?;
    let bytes = &cleaned.bytes;
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let input_len = input.len();
    let mut quote_close = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1usize;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: original_pos(&cleaned, 0, input_len),
            });
        }
    }

    let protected_end = quote_close.map_or(0, |i| i + 1);
    let ats: Vec<usize> = (protected_end..bytes.len())
        .filter(|&i| bytes[i] == b'@')
        .collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: original_pos(&cleaned, ats[1], input_len),
        });
    }

    let at_index = ats[0];
    if at_index == 0 {
        return Err(EmailError::EmptyLocal {
            at: original_pos(&cleaned, at_index, input_len),
        });
    }

    let (local, quoted, local_origins) = if let Some(close) = quote_close {
        if close + 1 != at_index {
            return Err(EmailError::TrailingLocal {
                at: original_pos(&cleaned, close + 1, input_len),
            });
        }
        decode_quoted(bytes, &cleaned, close, input_len)?
    } else {
        if at_index > 64 {
            return Err(EmailError::LocalTooLong {
                at: original_pos(&cleaned, 64, input_len),
            });
        }
        validate_atom(&bytes[..at_index], &cleaned, input_len)?;
        (
            String::from_utf8(bytes[..at_index].to_vec()).unwrap(),
            false,
            (0..at_index)
                .map(|i| original_pos(&cleaned, i, input_len))
                .collect(),
        )
    };

    let domain_start = at_index + 1;
    let mut domain_end = bytes.len();
    let mut root_dot = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(domain_end);
    }

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: original_pos(&cleaned, domain_start, input_len),
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: original_pos(&cleaned, domain_start + 253, input_len),
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || bytes[i] == b'.' {
            if start == i {
                let error_at = if i == domain_end {
                    root_dot
                        .map(|d| original_pos(&cleaned, d, input_len))
                        .unwrap_or_else(|| original_pos(&cleaned, i, input_len))
                } else {
                    original_pos(&cleaned, i, input_len)
                };
                return Err(EmailError::EmptyLabel { at: error_at });
            }
            label_starts.push(start);
            label_ends.push(i);
            start = i + 1;
        }
    }

    for (&start, &end) in label_starts.iter().zip(&label_ends) {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: original_pos(&cleaned, start + 63, input_len),
            });
        }
        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&cleaned, start, input_len),
            });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&cleaned, end - 1, input_len),
            });
        }
        for i in start..end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: original_pos(&cleaned, i, input_len),
                });
            }
        }
    }

    if label_starts.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: original_pos(&cleaned, domain_start, input_len),
        });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: original_pos(&cleaned, last_start, input_len),
        });
    }

    let local_origins = local_origins;
    let _ = local_origins;
    Ok(Address {
        local,
        domain: String::from_utf8(bytes[domain_start..domain_end].to_ascii_lowercase()).unwrap(),
        quoted,
    })
}

fn decode_quoted(
    bytes: &[u8],
    cleaned: &Cleaned,
    close: usize,
    input_len: usize,
) -> Result<(String, bool, Vec<usize>), EmailError> {
    let mut decoded = Vec::new();
    let mut origins = Vec::new();
    let mut i = 1usize;

    while i < close {
        let b = bytes[i];
        if b == b'\\' {
            if i + 1 >= close {
                return Err(EmailError::BadEscape {
                    at: original_pos(cleaned, i, input_len),
                });
            }
            let escaped = bytes[i + 1];
            if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                return Err(EmailError::BadEscape {
                    at: original_pos(cleaned, i, input_len),
                });
            }
            decoded.push(escaped);
            origins.push(original_pos(cleaned, i, input_len));
            i += 2;
        } else {
            if !(0x20..=0x7e).contains(&b) {
                return Err(EmailError::QuotedChar {
                    at: original_pos(cleaned, i, input_len),
                });
            }
            decoded.push(b);
            origins.push(original_pos(cleaned, i, input_len));
            i += 1;
        }
    }

    if decoded.is_empty() {
        return Err(EmailError::EmptyLocal {
            at: original_pos(cleaned, close, input_len),
        });
    }
    if decoded.len() > 64 {
        return Err(EmailError::LocalTooLong { at: origins[64] });
    }

    Ok((
        String::from_utf8(decoded).unwrap(),
        true,
        origins,
    ))
}

fn validate_atom(
    local: &[u8],
    cleaned: &Cleaned,
    input_len: usize,
) -> Result<(), EmailError> {
    const PUNCT: &[u8] = b"!#$%&'*+-/=?^_`{|}~";
    for i in 0..local.len() {
        let b = local[i];
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot {
                    at: original_pos(cleaned, i, input_len),
                });
            }
        } else if !b.is_ascii_alphanumeric() && !PUNCT.contains(&b) {
            return Err(EmailError::LocalChar {
                at: original_pos(cleaned, i, input_len),
            });
        }
    }
    Ok(())
}

fn valid_dot_atom(local: &[u8]) -> bool {
    if local.is_empty() {
        return false;
    }
    const PUNCT: &[u8] = b"!#$%&'*+-/=?^_`{|}~";
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return false;
            }
        } else if !b.is_ascii_alphanumeric() && !PUNCT.contains(&b) {
            return false;
        }
    }
    true
}

fn normalize_parsed(
    input: &str,
    mut address: Address,
    policy: &Policy,
) -> Result<Address, EmailError> {
    let mut bytes = address.local.into_bytes();
    if policy.local_case == LocalCase::Lower {
        bytes.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = bytes.iter().position(|&b| b == b'+') {
            if plus == 0 {
                let pos = original_plus_position(input, &address, policy);
                return Err(EmailError::EmptyNormalizedLocal { at: pos });
            }
            bytes.truncate(plus);
        }
    }

    let safe = valid_dot_atom(&bytes);
    address.local = String::from_utf8(bytes).unwrap();
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

fn original_plus_position(input: &str, address: &Address, _policy: &Policy) -> usize {
    // Reconstruct the source position for the leading decoded plus where possible.
    let cleaned = match cleanup(input) {
        Ok(c) => c,
        Err(_) => return 0,
    };
    let bytes = &cleaned.bytes;
    let mut close = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
    }
    if let Some(close) = close {
        let mut i = 1;
        while i < close {
            if bytes[i] == b'\\' {
                if bytes.get(i + 1) == Some(&b'+') {
                    return cleaned.positions[i];
                }
                i += 2;
            } else {
                if bytes[i] == b'+' {
                    return cleaned.positions[i];
                }
                i += 1;
            }
        }
    } else {
        let _ = address;
        if let Some(pos) = bytes.iter().position(|&b| b == b'+') {
            return cleaned.positions[pos];
        }
    }
    0
}
