fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_protected = false;
    let mut quote_closed = false;
    let mut quote_escape = false;
    let mut comment_depth = 0usize;
    let mut comment_outer = 0usize;
    let mut comment_stack: Vec<usize> = Vec::new();

    while i < input.len() {
        let b = input[i];

        if quote_protected && !quote_closed {
            bytes.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        let in_comment = comment_depth > 0;

        if in_comment {
            if b == b'\\' {
                let Some(&next) = input.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !(0x20..=0x7e).contains(&next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }

            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }

            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }

            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                comment_stack.push(i);
            } else if b == b')' {
                comment_depth -= 1;
                comment_stack.pop();
            }
            i += 1;
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                let run_start = i;
                i = end;
                while i < input.len() {
                    let c = input[i];
                    if is_hws(c) {
                        i += 1;
                    } else if c == b'(' {
                        if comment_depth == 0 {
                            comment_outer = i;
                        }
                        if comment_depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });

                        comment_depth += 1;
                        comment_stack.push(i);
                        i += 1;
                        break;
                    } else if c == b'\r' || c == b'\n' {
                        if let Some(next_end) = fold_end(input, i) {
                            i = next_end;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    } else {
                        break;
                    }
                }
                runs.push(CfwsRun { start: run_start, end: i });
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }

                if input[i] == b'(' {
                    if comment_depth == 0 {
                        comment_outer = i;
                    }
                    if comment_depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    comment_depth += 1;
                    comment_stack.push(i);
                    i += 1;
                    break;
                } else if input[i] == b'\r' || input[i] == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if is_hws(input[i]) {
                    i += 1;
                } else {
                    break;
                }
            }
            runs.push(CfwsRun { start: run_start, end: i });
            continue;
        }

        if b == b'(' {
            let run_start = i;

            comment_depth = 1;
            comment_stack.push(i);

fn valid_atom(local: &[u8]) -> bool {
    if local.is_empty() || local[0] == b'.' || local[local.len() - 1] == b'.' {
        return false;
    }
    let mut prev_dot = false;
    for &b in local {
        if b == b'.' {
            if prev_dot {
                return false;
            }
            prev_dot = true;
        } else {
            prev_dot = false;
            if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return false;
            }
        }
    }
    true
}

fn parse_cleaned(

    cleaned: &Cleaned,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {

    let input_len = input.len();

    for run in &cleaned.runs {
        let before = cleaned.positions.partition_point(|&p| p < run.start);
        let after = cleaned.positions.partition_point(|&p| p < run.end);
        let at_boundary = before == 0
            || after == bytes.len()
            || bytes.get(before.wrapping_sub(1)) == Some(&b'@')
            || bytes.get(after) == Some(&b'@');
        if !at_boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i = (i + 2).min(bytes.len());
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: original_pos(cleaned, 0, input_len),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: original_pos(cleaned, separators[1], input_len),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: original_pos(cleaned, sep, input_len),
        });
    }

    let local_bytes = &bytes[..sep];
    let (decoded, origins, quoted) = if local_bytes[0] == b'"' {
        let end = quote_end.unwrap_or(0);
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: original_pos(cleaned, end + 1, input_len),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = local_bytes[j];
            if b == b'\\' {
                let Some(&escaped) = local_bytes.get(j + 1) else {
                    return Err(EmailError::BadEscape {
                        at: original_pos(cleaned, j, input_len),
                    });
                };
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: original_pos(cleaned, j, input_len),
                    });
                }
                decoded.push(escaped);
                origins.push(original_pos(cleaned, j, input_len));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: original_pos(cleaned, j, input_len),
                    });
                }
                decoded.push(b);
                origins.push(original_pos(cleaned, j, input_len));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: original_pos(cleaned, end, input_len),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        (decoded, origins, true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: original_pos(cleaned, 64, input_len),
            });
        }
        let mut prev_dot = false;
        for (idx, &b) in local_bytes.iter().enumerate() {
            let pos = original_pos(cleaned, idx, input_len);
            if b == b'.' {
                if idx == 0 || idx + 1 == local_bytes.len() || prev_dot {
                    return Err(EmailError::LocalDot { at: pos });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !(b.is_ascii_alphanumeric()
                    || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
                {
                    return Err(EmailError::LocalChar { at: pos });
                }
            }
        }
        (
            local_bytes.to_vec(),
            (0..local_bytes.len())
                .map(|idx| original_pos(cleaned, idx, input_len))
                .collect(),
            false,
        )
    };

    let raw_domain_start = sep + 1;
    let domain_start_pos = original_pos(cleaned, raw_domain_start, input_len);
    let mut domain_end = bytes.len();
    let mut root_dot_pos = None;
    if domain_end > raw_domain_start && bytes[domain_end - 1] == b'.' {
        root_dot_pos = Some(original_pos(cleaned, domain_end - 1, input_len));
        domain_end -= 1;
    }

    if raw_domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    let domain_bytes = &bytes[raw_domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: original_pos(cleaned, raw_domain_start + 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;
    for (idx, &b) in domain_bytes.iter().enumerate() {
        if b == b'.' {
            labels.push((label_start, idx));
            label_start = idx + 1;
        }
    }
    labels.push((label_start, domain_bytes.len()));

    for &(start, end) in &labels {
        if start == end {
            let at = if start == domain_bytes.len() {
                root_dot_pos.unwrap_or(input_len)
            } else {
                original_pos(cleaned, raw_domain_start + start, input_len)
            };
            return Err(EmailError::EmptyLabel { at });
        }

        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: original_pos(cleaned, raw_domain_start + start + 63, input_len),
            });
        }

        if domain_bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(cleaned, raw_domain_start + start, input_len),
            });
        }
        if domain_bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(cleaned, raw_domain_start + end - 1, input_len),
            });
        }

        for idx in start..end {
            let b = domain_bytes[idx];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar {
                    at: original_pos(cleaned, raw_domain_start + idx, input_len),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if domain_bytes[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: original_pos(cleaned, raw_domain_start + last_start, input_len),
        });
    }

    let local = String::from_utf8(decoded).expect("ASCII local");
    let domain = domain_bytes
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    let domain = String::from_utf8(domain).expect("ASCII domain");
    Ok((
        Address {
            local,
            domain,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = input.as_bytes().iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(input.as_bytes())?;
    parse_cleaned(input, &cleaned, false).map(|(a, _)| a)
}

fn dot_atom_valid(local: &str) -> bool {
    valid_atom(local.as_bytes())
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = input.as_bytes().iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(input.as_bytes())?;
    let (mut address, mut origins) =
        parse_cleaned(input, &cleaned, policy.allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                let at = origins.get(plus).copied().unwrap_or(input.len());
                return Err(EmailError::EmptyNormalizedLocal { at });
            }
            address.local.truncate(plus);
            origins.truncate(plus);
        }
    }

    let safe = dot_atom_valid(&address.local);
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (idx, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((idx, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    result.duplicate_of[idx] = Some(*first_index);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), idx));
                    result.source_indices.push(idx);
                    result.addresses.push(address);
                }
            }
        }
    }

    result
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)

fn cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut map = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut quote_prefix = false;

    while i < bytes.len() {
        if quote_prefix {
            let b = bytes[i];
            out.push(b);
            map.push(i);
            i += 1;
            if b == b'\\' {
                if i < bytes.len() {
                    out.push(bytes[i]);
                    map.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                quote_prefix = false;
            }
            continue;
        }

        if depth == 0 && b'\"' == bytes[i] && out.is_empty() {
            quote_prefix = true;
            out.push(bytes[i]);
            map.push(i);
            i += 1;
            continue;
        }

        if depth == 0 && bytes[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        let is_cfws_start = if depth == 0 {
            is_hws(bytes[i]) || bytes[i] == b'\r' || bytes[i] == b'\n' || bytes[i] == b'('
        } else {
            is_hws(bytes[i])
                || bytes[i] == b'\r'
                || bytes[i] == b'\n'
                || bytes[i] == b'('
                || bytes[i] == b')'
        };

        if is_cfws_start {
            let run_start = i;
            let mut run_depth = depth;
            if run_depth == 0 && bytes[i] == b'(' {
                run_depth = 1;
                outer_open = i;
                i += 1;
            }

            while i < bytes.len() {
                if run_depth > 0 {
                    match bytes[i] {
                        b'\\' => {
                            let slash = i;
                            if i + 1 >= bytes.len() || !is_printable(bytes[i + 1]) {
                                return Err(EmailError::CommentEscape { at: slash });
                            }
                            i += 2;
                        }
                        b'\r' | b'\n' => {
                            if let Some(end) = fold_end(bytes, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        }
                        b'(' => {
                            if run_depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            run_depth += 1;
                            i += 1;
                        }
                        b')' => {
                            run_depth -= 1;
                            i += 1;
                            if run_depth == 0 && (i >= bytes.len() || !is_hws(bytes[i]) && bytes[i] != b'\r' && bytes[i] != b'\n' && bytes[i] != b'(') {
                                break;
                            }
                        }
                        b' ' | b'\t' => i += 1,
                        b if is_printable(b) => i += 1,
                        _ => return Err(EmailError::CommentChar { at: i }),
                    }
                } else {
                    match bytes[i] {
                        b' ' | b'\t' => i += 1,
                        b'\r' | b'\n' => {
                            if let Some(end) = fold_end(bytes, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        }
                        b'(' => {
                            run_depth = 1;
                            outer_open = i;
                            i += 1;
                        }
                        _ => break,
                    }
                }
            }
            if run_depth > 0 {
                return Err(EmailError::UnclosedComment { at: outer_open });
            }
            depth = 0;
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if depth > 0 {
            unreachable!();
        }
        out.push(bytes[i]);
        map.push(i);
        i += 1;
    }

    Ok((out, map, runs))
}

fn pos(map: &[usize], index: usize, input_len: usize) -> usize {
    map.get(index).copied().unwrap_or(input_len)
}

fn dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let allowed_punct = b"!#$%&'*+-/=?^_`{|}~";
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if b.is_ascii_alphanumeric() || allowed_punct.contains(&b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_clean(
    input: &str,
    bytes: &[u8],
    map: &[usize],
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let n = bytes.len();
    let mut quote_close = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        while i < n {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(map, 0, input.len()),
            });
        }
    }

    let protected_end = quote_close.map(|x| x + 1).unwrap_or(0);
    let mut ats = Vec::new();
    for i in 0..n {
        if i >= protected_end && bytes[i] == b'@' {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(map, ats[1], input.len()),
        });
    }
    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(map, at, input.len()),
        });
    }

    let mut local = Vec::new();
    let mut origins = Vec::new();
    let mut quoted = false;
    if bytes[0] == b'"' {
        let close = quote_close.unwrap();
        if close + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: pos(map, close + 1, input.len()),
            });
        }
        quoted = true;
        let mut i = 1;
        while i < close {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape {
                        at: pos(map, i, input.len()),
                    });
                }
                let next = bytes[i + 1];
                if !matches!(next, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: pos(map, i, input.len()),
                    });
                }
                local.push(next);
                origins.push(pos(map, i, input.len()));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(map, i, input.len()),
                    });
                }
                local.push(b);
                origins.push(pos(map, i, input.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(map, close, input.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        if at > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(map, 64, input.len()),
            });
        }
        let allowed_punct = b"!#$%&'*+-/=?^_`{|}~";
        for i in 0..at {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == at || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: pos(map, i, input.len()),
                    });
                }
            } else if !(b.is_ascii_alphanumeric() || allowed_punct.contains(&b)) {
                return Err(EmailError::LocalChar {
                    at: pos(map, i, input.len()),
                });
            }
            local.push(b);
            origins.push(pos(map, i, input.len()));
        }
    }

    let domain_start = at + 1;
    let raw_domain = &bytes[domain_start..];
    let mut domain_end = n;
    let mut removed_root = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_root = Some(pos(map, domain_end, input.len()));
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: pos(map, domain_start, input.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(map, domain_start + 253, input.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if bytes[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    labels.push((start, domain_end));

    for (idx, &(s, e)) in labels.iter().enumerate() {
        if s == e {
            let p = if idx + 1 == labels.len() {
                removed_root.unwrap_or_else(|| pos(map, s, input.len()))
            } else if idx == 0 {
                pos(map, s, input.len())
            } else {
                pos(map, s - 1, input.len())
            };
            return Err(EmailError::EmptyLabel { at: p });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(map, s + 63, input.len()),
            });
        }
        if bytes[s] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(map, s, input.len()),
            });
        }
        if bytes[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(map, e - 1, input.len()),
            });
        }
        for i in s..e {
            if !(bytes[i].is_ascii_alphanumeric() || bytes[i] == b'-') {
                return Err(EmailError::DomainChar {
                    at: pos(map, i, input.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: pos(map, domain_start, input.len()),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(map, last_start, input.len()),
        });
    }

    let domain = String::from_utf8(bytes[domain_start..domain_end].iter().map(u8::to_ascii_lowercase).collect())
        .unwrap_or_default();
    let address = Address {
        local: String::from_utf8(local).unwrap_or_default(),
        domain,
        quoted,
    };
    let _ = raw_domain;
    Ok((address, origins))
}

fn parse_with_policy(input: &str, policy: &Policy, allow_single_label: bool) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = input.as_bytes().iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let (bytes, map, runs) = cleanup(input)?;
    for run in runs {
        let left = map.iter().position(|&p| p >= run.start);
        let right = map.iter().position(|&p| p >= run.end);
        let left_byte = left.and_then(|i| i.checked_sub(1).map(|j| bytes[j]));
        let right_byte = right.and_then(|i| bytes.get(i).copied());
        if left.is_some_and(|i| i > 0)
            && right.is_some_and(|i| i < bytes.len())
            && left_byte != Some(b'@')
            && right_byte != Some(b'@')
        {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let (mut address, origins) = parse_clean(input, &bytes, &map, allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }
    if !dot_atom(address.local.as_bytes()) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), false)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, policy.allow_single_label)
}

pub fn format_email(address: &Address) -> String {
    if !address.quoted {
        return format!("{}@{}", address.local, address.domain);
    }
    let mut out = String::with_capacity(address.local.len() + address.domain.len() + 3);
    out.push('"');
    for ch in address.local.chars() {
        if ch == '"' || ch == '\\' {
            out.push('\\');
        }
        out.push(ch);
    }
    out.push('"');
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: std::collections::HashMap<(String, String), usize> =
        std::collections::HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = keys.get(&key) {
                    result.duplicate_of[index] = Some(first_index);
                } else {
                    keys.insert(key, index);
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }
    result
}
