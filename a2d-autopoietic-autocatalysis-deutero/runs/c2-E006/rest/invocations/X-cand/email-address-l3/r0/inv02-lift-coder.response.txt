#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Retained {
    byte: u8,
    pos: usize,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
    pos: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn parse_cleanup(input: &str) -> Result<(Vec<Retained>, Vec<Run>), EmailError> {
    let bytes = input.as_bytes();
    let mut retained = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_protected = false;

    while i < bytes.len() {
        if quote_protected {
            let b = bytes[i];
            retained.push(Retained { byte: b, pos: i });
            i += 1;
            if b == b'\\' {
                if i < bytes.len() {
                    retained.push(Retained { byte: bytes[i], pos: i });
                    i += 1;
                }
            } else if b == b'"' {
                quote_protected = false;
            }
            continue;
        }

        if is_hws(bytes[i]) || bytes[i] == b'\r' || bytes[i] == b'\n' || bytes[i] == b'(' {
            let run_start = i;
            let mut comment_depth = 0usize;
            let mut outer_open = None;

            loop {
                if i >= bytes.len() {
                    break;
                }
                let b = bytes[i];

                if comment_depth > 0 {
                    match b {
                        b'\\' => {
                            if i + 1 >= bytes.len()
                                || !(0x20..=0x7e).contains(&bytes[i + 1])
                            {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        }
                        b'(' => {
                            if comment_depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            comment_depth += 1;
                            i += 1;
                        }
                        b')' => {
                            comment_depth -= 1;
                            i += 1;
                        }
                        b'\r' | b'\n' => {
                            if let Some(end) = fold_end(bytes, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        }
                        b'\t' => i += 1,
                        _ if (0x20..=0x7e).contains(&b) => i += 1,
                        _ => return Err(EmailError::CommentChar { at: i }),
                    }
                    continue;
                }

                if is_hws(b) {
                    i += 1;
                } else if b == b'\r' || b == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if b == b'(' {
                    outer_open = Some(i);
                    comment_depth = 1;
                    i += 1;
                } else {
                    break;
                }
            }

            if comment_depth > 0 {
                return Err(EmailError::UnclosedComment {
                    at: outer_open.unwrap_or(run_start),
                });
            }

            runs.push(Run {
                start: retained.len(),
                end: retained.len(),
                pos: run_start,
            });
            continue;
        }

        if bytes[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        retained.push(Retained {
            byte: bytes[i],
            pos: i,
        });
        if retained.len() == 1 && bytes[i] == b'"' {
            quote_protected = true;
        }
        i += 1;
    }

    Ok((retained, runs))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, mut runs) = parse_cleanup(input)?;

    for run in &mut runs {
        run.end = run.start;
    }
    for (idx, r) in runs.iter_mut().enumerate() {
        let next_start = runs.get(idx + 1).map(|r| r.start).unwrap_or(clean.len());
        r.end = next_start.min(clean.len());
    }

    for run in &runs {
        let left_at = run.start > 0 && clean[run.start - 1].byte == b'@';
        let right_at = run.start < clean.len() && clean[run.start].byte == b'@';
        let edge = run.start == 0 || run.start == clean.len();
        if !(edge || left_at || right_at) {
            return Err(EmailError::CfwsBoundary { at: run.pos });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if clean[0].byte == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i].byte == b'\\' {
                i += 2;
            } else if clean[i].byte == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote { at: clean[0].pos });
        }
    }

    let protected_end = quote_close.map(|i| i + 1).unwrap_or(0);
    let separators: Vec<usize> = (protected_end..clean.len())
        .filter(|&i| clean[i].byte == b'@')
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: clean[separators[1]].pos,
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: clean[sep].pos });
    }

    let local_bytes = &clean[..sep];
    let mut local = String::new();
    let mut quoted = false;

    if local_bytes[0].byte == b'"' {
        quoted = true;
        let close = quote_close.expect("quoted prefix has closing quote");
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: clean[close + 1].pos,
            });
        }
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = clean[i].byte;
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: clean[i].pos });
                }
                let escaped = clean[i + 1].byte;
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: clean[i].pos });
                }
                local.push(escaped as char);
                origins.push(clean[i].pos);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: clean[i].pos });
                }
                local.push(b as char);
                origins.push(clean[i].pos);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: clean[close].pos,
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_bytes[64].pos,
            });
        }
        for (i, item) in local_bytes.iter().enumerate() {
            let b = item.byte;
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes[i - 1].byte == b'.'
                {
                    return Err(EmailError::LocalDot { at: item.pos });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: item.pos });
            }
            local.push(b as char);
        }
    }

    let domain_start = sep + 1;
    let raw_domain = &clean[domain_start..];
    let mut domain_end = clean.len();
    let root_dot = if raw_domain.last().map(|r| r.byte) == Some(b'.') {
        domain_end -= 1;
        Some(clean[domain_end].pos)
    } else {
        None
    };

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: clean.get(domain_start).map(|r| r.pos).unwrap_or(input.len()),
        });
    }

    let domain_slice = &clean[domain_start..domain_end];
    if domain_slice.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_slice[253].pos,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if clean[i].byte == b'.' {
            label_starts.push(start);
            label_ends.push(i);
            start = i + 1;
        }
    }
    label_starts.push(start);
    label_ends.push(domain_end);

    for idx in 0..label_starts.len() {
        let start = label_starts[idx];
        let end = label_ends[idx];
        if start == end {
            let at = if idx + 1 == label_starts.len() {
                root_dot.unwrap_or_else(|| {
                    clean.get(start).map(|r| r.pos).unwrap_or(input.len())
                })
            } else {
                clean.get(start).map(|r| r.pos).unwrap_or(input.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: clean[start + 63].pos,
            });
        }
        if clean[start].byte == b'-' {
            return Err(EmailError::LabelHyphen { at: clean[start].pos });
        }
        if clean[end - 1].byte == b'-' {
            return Err(EmailError::LabelHyphen {
                at: clean[end - 1].pos,
            });
        }
        for item in &clean[start..end] {
            if !item.byte.is_ascii_alphanumeric() && item.byte != b'-' {
                return Err(EmailError::DomainChar { at: item.pos });
            }
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: clean[domain_start].pos,
        });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if clean[last_start..last_end]
        .iter()
        .all(|item| item.byte.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: clean[last_start].pos,
        });
    }

    let domain: String = domain_slice
        .iter()
        .map(|item| item.byte.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
