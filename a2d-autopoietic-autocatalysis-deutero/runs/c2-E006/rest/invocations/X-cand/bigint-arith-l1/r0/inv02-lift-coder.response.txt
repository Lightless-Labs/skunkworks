```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian decimal digits. Zero is represented as [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = usize::from(bytes[0] == b'+');
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        // Check separators in the magnitude, including sign-relative offsets.
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }

        let mut first_digit_index = None;
        let mut digit_count = 0usize;
        let mut digits = Vec::new();

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' {
                if first_digit_index.is_none() {
                    first_digit_index = Some(index);
                }
                digit_count += 1;
                digits.push(byte - b'0');
            }
        }

        let first_index = first_digit_index.expect("nonempty magnitude has a digit");
        if digit_count > 1 && bytes[first_index] == b'0' {
            return Err(Error::LeadingZero { index: first_index });
        }

        if digit_count > max_digits {
            // Avoid max_digits + 1, which could overflow at usize::MAX.
            let excess_number = max_digits.saturating_add(1);
            let mut seen = 0usize;
            let excess_index = (start..bytes.len())
                .find(|&i| {
                    if bytes[i].is_ascii_digit() {
                        seen += 1;
                        seen == excess_number
                    } else {
                        false
                    }
                })
                .expect("digit_count exceeds the limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: excess_index,
            });
        }

        digits.reverse();
        Ok(Self::from_le_digits(digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut raw = String::with_capacity(self.digits.len().max(min_digits));
        for &digit in self.digits.iter().rev() {
            raw.push(char::from(b'0' + digit));
        }

        let width = raw.len().max(min_digits);
        let padding = width - raw.len();

        if group == 0 {
            let mut output = String::with_capacity(width);
            output.extend(std::iter::repeat_n('0', padding));
            output.push_str(&raw);
            return output;
        }

        let separator_count = (width - 1) / group;
        let mut output = String::with_capacity(width + separator_count);
        let first_group_len = if width % group == 0 {
            group
        } else {
            width % group
        };

        for i in 0..width {
            if i != 0 && (i - first_group_len) % group == 0 {
                output.push('_');
            }
            if i < padding {
                output.push('0');
            } else {
                output.push(raw.as_bytes()[i - padding] as char);
            }
        }

        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (&left, &right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(&right) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = Vec::with_capacity(self.digits.len().max(rhs.digits.len()) + 1);
        let mut carry = 0u8;
        let length = self.digits.len().max(rhs.digits.len());

        for i in 0..length {
            let left = self.digits.get(i).copied().unwrap_or(0);
            let right = rhs.digits.get(i).copied().unwrap_or(0);
            let sum = left + right + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            result.push(carry);
        }

        Self::from_le_digits(result)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i8;

        for i in 0..self.digits.len() {
            let left = self.digits[i] as i8 - borrow;
            let right = rhs.digits.get(i).copied().unwrap_or(0) as i8;
            if left < right {
                result.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                result.push((left - right) as u8);
                borrow = 0;
            }
        }

        Ok(Self::from_le_digits(result))
    }

    fn from_le_digits(mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        Self { digits }
    }
}
```
