fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(field, start, |field, offset| ParseError::LeadingZero {
            field,
            offset,
        }));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| {
                error(field, start, |field, offset| ParseError::NumberOverflow {
                    field,
                    offset,
                })
            })?;
    }

    let (min, max, width) = field_spec(field);
    if is_step {
        if value == 0 || value > width {
            return Err(error(field, start, |field, offset| ParseError::BadStep {
                field,
                offset,
            }));
        }
    } else if value < min || value > max {
        return Err(error(field, start, |field, offset| ParseError::OutOfRange {
            field,
            offset,
        }));
    }

    Ok(value)
}

                    field,
                    offset: item_start,

                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            None => None,

            0 => (domain_min, domain_max),
            1 => {

                let value = parse_number(item, start, end, field, false)?;
                (value, if step_bounds.is_some() { domain_max } else { value })

                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + end_start,

        .into_iter()
        .enumerate()

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected = Vec::with_capacity(5);
    for (text, field) in fields.iter().zip(order) {
        selected.push(parse_field(text, field)?);

    Ok(Cron {
        selected: selected.try_into().expect("exactly five fields"),
        day_of_month_is_exact_star: fields[2].as_bytes() == b"*",
        day_of_week_is_exact_star: fields[4].as_bytes() == b"*",
    })
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let slash = bytes[item_start..item_end]
        .iter()
        .position(|&b| b == b'/')
        .map(|p| item_start + p);

    let base_end = slash.unwrap_or(item_end);
    let step_span = slash.map(|p| (p + 1, item_end));

    let is_star = base_end == item_start + 1 && bytes[item_start] == b'*';
    let hyphen = bytes[item_start..base_end]
        .iter()
        .position(|&b| b == b'-')
        .map(|p| item_start + p);

    let base_valid = if is_star {
        true
    } else if let Some(h) = hyphen {
        h > item_start
            && h + 1 < base_end
            && bytes[item_start..h].iter().all(u8::is_ascii_digit)
            && bytes[h + 1..base_end].iter().all(u8::is_ascii_digit)
    } else {
        item_start < base_end
            && bytes[item_start..base_end].iter().all(u8::is_ascii_digit)
    };

    let step_valid = match step_span {
        None => true,
        Some((start, end)) => {
            start < end && bytes[start..end].iter().all(u8::is_ascii_digit)
        }
    };

    if !base_valid || !step_valid {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let (min, max) = domain(field);
    let (start, end) = if is_star {
        (min, max)
    } else if let Some(h) = hyphen {
        let range_start = parse_number(bytes, item_start, h, field, false)?;
        let range_end = parse_number(bytes, h + 1, base_end, field, false)?;
        if range_start > range_end {
            return Err(ParseError::ReversedRange {
                field,
                offset: h + 1,
            });
        }
        (range_start, range_end)
    } else {
        let value = parse_number(bytes, item_start, base_end, field, false)?;
        (value, if step_span.is_some() { max } else { value })
    };

    let step = if let Some((step_start, step_end)) = step_span {
        parse_number(bytes, step_start, step_end, field, true)?
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(field_text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = field_text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });

        parse_item(bytes, item_start, item_end, field, &mut selected)?;

    }

    Ok(selected)
}

            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let is_star = base == b"*";
        let is_range = range_positions.len() == 1;
        let is_number = !base.is_empty() && base.iter().all(u8::is_ascii_digit);

        if !(is_star || is_number || is_range) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (start_num, end_num, start_offset, end_offset) = if is_star {
            (None, None, 0, 0)
        } else if is_range {
            let hyphen = range_positions[0];
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if left.is_empty()
                || right.is_empty()
                || !left.iter().all(u8::is_ascii_digit)
                || !right.iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (
                Some(left),
                Some(right),
                item_start,
                item_start + hyphen + 1,
            )
        } else {
            (Some(base), None, item_start, 0)
        };

        let (start, end) = if is_star {
            (domain_min as u32, domain_max as u32)
        } else {
            let start_bytes = start_num.unwrap();
            let start_value =
                parse_number(start_bytes, start_offset, field, false)?;
            let end_value = if let Some(end_bytes) = end_num {
                let value = parse_number(end_bytes, end_offset, field, false)?;
                if start_value > value {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
                value
            } else if step_bytes.is_some() {
                domain_max as u32
            } else {
                start_value
            };
            (start_value, end_value)
        };

        let step = if let Some(step_bytes) = step_bytes {
            if step_bytes.is_empty() || !step_bytes.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            parse_number(step_bytes, step_offset, field, true)?
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item_len + 1;
    }

    Ok((domain_min..=domain_max)
        .filter(|&v| selected[v as usize])
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_types = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        parsed[i] = parse_field(fields[i], field_types[i])?;
    }

    Ok(Cron {
        selected: parsed,
        day_of_month_is_star: fields[2] == "*",
        day_of_week_is_star: fields[4] == "*",
    })
}
