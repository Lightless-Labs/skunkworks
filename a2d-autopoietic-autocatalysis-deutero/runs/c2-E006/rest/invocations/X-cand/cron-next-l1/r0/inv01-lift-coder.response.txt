#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_of_month_is_exact_star: bool,
    day_of_week_is_exact_star: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_spec(field: Field) -> (u32, u32, u32) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn error(field: Field, offset: usize, make: fn(Field, usize) -> ParseError) -> ParseError {
    make(field, offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(field, start, |field, offset| ParseError::LeadingZero {
            field,
            offset,
        }));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| {
                error(field, start, |field, offset| ParseError::NumberOverflow {
                    field,
                    offset,
                })
            })?;
    }

    let (min, max, width) = field_spec(field);
    if is_step {
        if value == 0 || value > width {
            return Err(error(field, start, |field, offset| ParseError::BadStep {
                field,
                offset,
            }));
        }
    } else if value < min || value > max {
        return Err(error(field, start, |field, offset| ParseError::OutOfRange {
            field,
            offset,
        }));
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (domain_min, domain_max, _) = field_spec(field);
    let mut selected = vec![false; (domain_max + 1) as usize];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_bounds) = if slash_positions.is_empty() {
            (item.len(), None)
        } else if slash_positions.len() == 1 {
            let slash = slash_positions[0];
            (slash, Some((slash + 1, item.len())))
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        let base = &item[..base_end];
        let (base_kind, first_bounds, second_bounds) = if base == b"*" {
            (0u8, None, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();

            if hyphens.is_empty() {
                if base.iter().all(u8::is_ascii_digit) {
                    (1, Some((0, base.len())), None)
                } else {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
            } else if hyphens.len() == 1 {
                let hyphen = hyphens[0];
                if hyphen > 0
                    && hyphen + 1 < base.len()
                    && base[..hyphen].iter().all(u8::is_ascii_digit)
                    && base[hyphen + 1..].iter().all(u8::is_ascii_digit)
                {
                    (
                        2,
                        Some((0, hyphen)),
                        Some((hyphen + 1, base.len())),
                    )
                } else {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
            } else {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let step_bounds = match step_bounds {
            Some((start, end)) if start < end && item[start..end].iter().all(u8::is_ascii_digit) => {
                Some((start, end))
            }
            Some(_) => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            None => None,
        };

        let (start_value, end_value) = match base_kind {
            0 => (domain_min, domain_max),
            1 => {
                let (start, end) = first_bounds.unwrap();
                let value = parse_number(item, start, end, field, false)?;
                (value, if step_bounds.is_some() { domain_max } else { value })
            }
            _ => {
                let (start_start, start_end) = first_bounds.unwrap();
                let (end_start, end_end) = second_bounds.unwrap();
                let start_value = parse_number(item, start_start, start_end, field, false)?;
                let end_value = parse_number(item, end_start, end_end, field, false)?;
                if start_value > end_value {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + end_start,
                    });
                }
                (start_value, end_value)
            }
        };

        let step = if let Some((start, end)) = step_bounds {
            Some(parse_number(item, start, end, field, true)?)
        } else {
            None
        };

        let increment = step.unwrap_or(1);
        let mut value = start_value;
        loop {
            selected[value as usize] = true;
            if value > end_value || end_value - value < increment {
                break;
            }
            value += increment;
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .into_iter()
        .enumerate()
        .filter_map(|(value, is_selected)| is_selected.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected = Vec::with_capacity(5);
    for (text, field) in fields.iter().zip(order) {
        selected.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: selected.try_into().expect("exactly five fields"),
        day_of_month_is_exact_star: fields[2].as_bytes() == b"*",
        day_of_week_is_exact_star: fields[4].as_bytes() == b"*",
    })
}
