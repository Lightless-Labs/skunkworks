#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::FieldStart;
    let mut record_index = 0usize;
    let mut i = start;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        // Outside quoted fields, CR is valid only as part of CRLF.
        if !matches!(state, State::Quoted { .. }) && b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            finish_field(&mut field, &mut record);
            records.push(std::mem::take(&mut record));
            record_index += 1;
            state = State::FieldStart;
            i += 2;
            continue;
        }

        match state {
            State::FieldStart => match b {
                b',' => {
                    finish_field(&mut field, &mut record);
                    i += 1;
                }
                b'\n' => {
                    finish_field(&mut field, &mut record);
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    i += 1;
                }
                b'"' => {
                    state = State::Quoted { opening: i };
                    i += 1;
                }
                _ => {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    state = State::Unquoted;
                    i += ch.len_utf8();
                }
            },
            State::Unquoted => match b {
                b',' => {
                    finish_field(&mut field, &mut record);
                    state = State::FieldStart;
                    i += 1;
                }
                b'\n' => {
                    finish_field(&mut field, &mut record);
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += 1;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                }
                _ => {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            },
            State::Quoted { opening } => {
                if b == b'"' {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'"' {
                        field.push('"');
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                    state = State::Quoted { opening };
                }
            }
            State::AfterQuote => match b {
                b',' => {
                    finish_field(&mut field, &mut record);
                    state = State::FieldStart;
                    i += 1;
                }
                b'\n' => {
                    finish_field(&mut field, &mut record);
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            },
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    // A final outside-quote record terminator already emitted its record.
    if i > start && matches!(bytes.last(), Some(b'\n')) {
        return Ok(records);
    }

    finish_field(&mut field, &mut record);
    records.push(record);
    Ok(records)
}

fn finish_field(field: &mut String, record: &mut Vec<String>) {
    record.push(std::mem::take(field));
}
