#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    let mut output = String::new();
    let mut value = 0u8;
    let mut bits = 0u8;

    for &byte in input {
        for shift in (0..8).rev() {
            value = (value << 1) | ((byte >> shift) & 1);
            bits += 1;

            if bits == 5 {
                output.push(ALPHABET[value as usize] as char);
                value = 0;
                bits = 0;
            }
        }
    }

    if bits != 0 {
        value <<= 5 - bits;
        output.push(ALPHABET[value as usize] as char);
    }

    while output.len() % 8 != 0 {
        output.push('=');
    }

    output
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    // Lexical validation.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && !ALPHABET.contains(&byte) {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    // Padding placement.
    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);
    if let Some(relative) = bytes[p..].iter().position(|&byte| byte != b'=') {
        return Err(DecodeError::Padding {
            offset: p + relative,
        });
    }

    // Data length.
    let residue = p % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    // Padding shape.
    let required_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => unreachable!(),
    };
    if n % 8 != 0 || n - p != required_padding {
        return Err(DecodeError::Padding { offset: p });
    }

    // Trailing bits.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits != 0 {
        let value = ALPHABET
            .iter()
            .position(|&symbol| symbol == bytes[p - 1])
            .unwrap() as u8;
        if value & ((1 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset: p - 1 });
        }
    }

    // Decode complete bytes, most significant bit first.
    let mut output = Vec::with_capacity(p.saturating_mul(5) / 8);
    let mut value = 0u8;
    let mut bits = 0u8;

    for &symbol in &bytes[..p] {
        let symbol_value = ALPHABET
            .iter()
            .position(|&candidate| candidate == symbol)
            .unwrap() as u8;

        for shift in (0..5).rev() {
            value = (value << 1) | ((symbol_value >> shift) & 1);
            bits += 1;

            if bits == 8 {
                output.push(value);
                value = 0;
                bits = 0;
            }
        }
    }

    Ok(output)
}
