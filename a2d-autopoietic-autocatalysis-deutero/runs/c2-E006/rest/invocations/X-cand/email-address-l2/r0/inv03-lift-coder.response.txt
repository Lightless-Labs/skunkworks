#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_close = if bytes[0] == b'"' {
        let mut i = 1;
        let mut found = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    found = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match found {
            Some(pos) => Some(pos),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' {
            if let Some(close) = quoted_close {
                if i <= close {
                    continue;
                }
            }
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, quoted) = if let Some(close) = quoted_close {
        if close >= sep {
            // The closing quote is part of the local prefix and must precede the separator.
            return Err(EmailError::TrailingLocal { at: sep });
        }
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: i });
                }
                let escaped = bytes[i + 1];
                match escaped {
                    b'"' | b'\\' | b'@' => {
                        decoded.push(escaped);
                        origins.push(i);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: i }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                decoded.push(b);
                origins.push(i);
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let local_bytes = &bytes[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !is_ascii_alnum(b) && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                return Err(EmailError::LocalChar { at: i });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let (domain_bytes, root_dot_pos) = if raw_domain.last() == Some(&b'.') {
        (&raw_domain[..raw_domain.len() - 1], Some(input.len() - 1))
    } else {
        (raw_domain, None)
    };

    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain {
            at: if root_dot_pos.is_some() {
                domain_start
            } else {
                input.len()
            },
        });
    }

    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels = Vec::new();
    let mut label_start = 0;
    for i in 0..=domain_bytes.len() {
        if i == domain_bytes.len() || domain_bytes[i] == b'.' {
            if i == label_start {
                let at = if i == domain_bytes.len() {
                    root_dot_pos.unwrap_or(domain_start + i)
                } else {
                    domain_start + i
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label = &domain_bytes[label_start..i];
            let abs_start = domain_start + label_start;

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong { at: abs_start + 63 });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: abs_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: abs_start + label.len() - 1,
                });
            }
            for (offset, &b) in label.iter().enumerate() {
                if !is_ascii_alnum(b) && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: abs_start + offset,
                    });
                }
            }

            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    let last_label = &domain_bytes[last_start..last_end];
    if last_label.iter().all(|b| b.is_ascii_digit()) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let domain = domain_bytes
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

fn is_ascii_alnum(b: u8) -> bool {
    b.is_ascii_alphanumeric()
}
