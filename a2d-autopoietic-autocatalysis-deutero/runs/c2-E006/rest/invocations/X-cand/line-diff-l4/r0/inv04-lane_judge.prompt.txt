fn validate(
    base: &str,
    left: &str,
    right: &str,
) -> Result<(Vec<String>, Vec<String>, Vec<String>), MergeError> {
    let inputs = [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ];

    for (input, side) in &inputs {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn {
                side: side.clone(),
                byte,
            });
        }
    }

    for (input, side) in &inputs {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side: side.clone() });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    Ok((base_lines, left_lines, right_lines))
}

fn edit_script(old: &[String], new: &[String]) -> Vec<Edit> {
    let rows = old.len() + 1;
    let cols = new.len() + 1;
    let mut lcs = vec![0usize; rows * cols];

    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            let value = if old[i] == new[j] {
                1 + lcs[(i + 1) * cols + j + 1]
            } else {
                lcs[(i + 1) * cols + j].max(lcs[i * cols + j + 1])
            };
            lcs[i * cols + j] = value;
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut script = Vec::with_capacity(old.len() + new.len());

    while i < old.len() && j < new.len() {
        if old[i] == new[j] {
            script.push(Edit::Keep);
            i += 1;
            j += 1;
        } else {
            let delete_lcs = lcs[(i + 1) * cols + j];
            let insert_lcs = lcs[i * cols + j + 1];
            if delete_lcs > insert_lcs || (delete_lcs == insert_lcs && (i + j) % 2 == 0) {
                script.push(Edit::Delete);
                i += 1;
            } else {
                script.push(Edit::Insert);
                j += 1;
            }
        }
    }

    while i < old.len() {
        script.push(Edit::Delete);
        i += 1;
    }
    while j < new.len() {
        script.push(Edit::Insert);
        j += 1;
    }

    script
}

fn changes_from_script(old: &[String], new: &[String]) -> Vec<Change> {
    let script = edit_script(old, new);
    let (mut old_i, mut new_i) = (0usize, 0usize);
    let mut changes = Vec::new();
    let mut active_start = 0usize;
    let mut active_deletes = 0usize;
    let mut active_replacement = Vec::new();
    let mut active = false;

    for edit in script {
        match edit {
            Edit::Keep => {
                if active {
                    changes.push(Change {
                        start: active_start,
                        end: active_start + active_deletes,
                        replacement: std::mem::take(&mut active_replacement),
                    });
                    active_deletes = 0;
                    active = false;
                }
                old_i += 1;
                new_i += 1;
            }
            Edit::Delete => {
                if !active {
                    active = true;
                    active_start = old_i;
                }
                active_deletes += 1;
                old_i += 1;
            }
            Edit::Insert => {
                if !active {
                    active = true;
                    active_start = old_i;
                }
                active_replacement.push(new[new_i].clone());
                new_i += 1;
            }
        }
    }

    if active {
        changes.push(Change {
            start: active_start,
            end: active_start + active_deletes,
            replacement: active_replacement,
        });
    }

    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_positive = a.start < a.end;
    let b_positive = b.start < b.end;

    match (a_positive, b_positive) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn build_components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let mut all: Vec<(bool, usize)> = Vec::with_capacity(left.len() + right.len());
    all.extend((0..left.len()).map(|i| (true, i)));
    all.extend((0..right.len()).map(|i| (false, i)));

    let mut parent: Vec<usize> = (0..all.len()).collect();

    fn root(parent: &mut [usize], mut i: usize) -> usize {
        while parent[i] != i {
            parent[i] = parent[parent[i]];
            i = parent[i];
        }
        i
    }

    for i in 0..all.len() {
        for j in i + 1..all.len() {
            let (is_left_i, idx_i) = all[i];
            let (is_left_j, idx_j) = all[j];

            let a = if is_left_i {
                &left[idx_i]
            } else {
                &right[idx_i]
            };
            let b = if is_left_j {
                &left[idx_j]
            } else {
                &right[idx_j]
            };

            if interact(a, b) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut grouped: Vec<(usize, Vec<usize>)> = Vec::new();
    for i in 0..all.len() {
        let r = root(&mut parent, i);
        if let Some((_, indices)) = grouped.iter_mut().find(|(key, _)| *key == r) {
            indices.push(i);
        } else {
            grouped.push((r, vec![i]));
        }
    }

    let mut components = Vec::new();
    for (_, members) in grouped {
        let mut start = usize::MAX;
        let mut end = 0usize;
        let mut base_indices = Vec::new();
        let mut left_indices = Vec::new();
        let mut right_indices = Vec::new();

        for member in members {
            let (is_left, index) = all[member];
            let change = if is_left {
                left_indices.push(index);
                &left[index]
            } else {
                right_indices.push(index);
                &right[index]
            };
            start = start.min(change.start);
            end = end.max(change.end);
            base_indices.push(member);
        }

        left_indices.sort_unstable();
        right_indices.sort_unstable();
        components.push(Component {
            start,
            end,
            base_indices,
            left_indices,
            right_indices,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start < a.end).cmp(&(b.start < b.end)))
    });
    components
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change], indices: &[usize]) -> Vec<String> {
    let mut out = Vec::new();
    let mut cursor = start;

    for &index in indices {
        let change = &changes[index];
        if change.start < cursor || change.start > end || change.end > end {
            continue;
        }

        out.extend(base[cursor..change.start].iter().cloned());
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    out.extend(base[cursor..end].iter().cloned());
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let (base_lines, left_lines, right_lines) = validate(base, left, right)?;
    let left_changes = changes_from_script(&base_lines, &left_lines);
    let right_changes = changes_from_script(&base_lines, &right_lines);
    let components = build_components(&left_changes, &right_changes);

    let mut output_lines = Vec::new();
    let mut regions = Vec::new();
    let mut cursor = 0usize;

    for component in components {
        let start = component.start;
        let end = component.end;

        if start > cursor {
            output_lines.extend(base_lines[cursor..start].iter().cloned());
            cursor = start;
        }

        let base_projection = base_lines[start..end].to_vec();
        let left_projection = projection(
            &base_lines,
            start,
            end,
            &left_changes,
            &component.left_indices,
        );
        let right_projection = projection(
            &base_lines,
            start,
            end,
            &right_changes,
            &component.right_indices,
        );

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = regions.len() + 1;
                    let output_start = output_lines.len();
                    let base_len = end - start;
                    let base_marker_start = if base_len > 0 { start + 1 } else { start };

                    output_lines.push(format!("<<<<<<< LEFT {ordinal}"));
                    output_lines.extend(left_projection.iter().cloned());
                    output_lines.push(format!("||||||| BASE {base_marker_start},{base_len}"));
                    output_lines.extend(base_projection.iter().cloned());
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection.iter().cloned());
                    output_lines.push(format!(">>>>>>> RIGHT {ordinal}"));

                    regions.push(Conflict {
                        ordinal,
                        base_start: start,
                        base_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        if end > cursor {
            cursor = end;
        }
    }

    output_lines.extend(base_lines[cursor..].iter().cloned());

    let mut text = String::new();
    for line in &output_lines {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}

fn canonical_changes(old: &[String], new: &[String], side: MergeSide) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::new();
    let (mut i, mut j) = (0, 0);

    while i < n && j < m {
        if old[i] == new[j] {
            edits.push(Edit::Keep);
            i += 1;
            j += 1;
        } else {
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            if delete_len > insert_len || (delete_len == insert_len && (i + j) % 2 == 0) {
                edits.push(Edit::Delete);
                i += 1;
            } else {
                edits.push(Edit::Insert);
                j += 1;
            }
        }
    }

    while i < n {
        edits.push(Edit::Delete);
        i += 1;
    }
    while j < m {
        edits.push(Edit::Insert);
        j += 1;
    }

    let mut changes = Vec::new();
    let (mut base_pos, mut new_pos, mut cursor) = (0, 0, 0);

    while cursor < edits.len() {
        if matches!(edits[cursor], Edit::Keep) {
            base_pos += 1;
            new_pos += 1;
            cursor += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0;
        let mut replacement = Vec::new();

        while cursor < edits.len() && !matches!(edits[cursor], Edit::Keep) {
            match edits[cursor] {
                Edit::Delete => {
                    deleted += 1;
                    base_pos += 1;
                }
                Edit::Insert => {
                    replacement.push(new[new_pos].clone());
                    new_pos += 1;
                }
                Edit::Keep => unreachable!(),
            }
            cursor += 1;
        }

        changes.push(Change {
            start,
            end: start + deleted,
            replacement,
            side: side.clone(),
        });
    }

    changes
}

    }

    if position < end {
        output.extend(base[position..end].iter().cloned());
    }

    output
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = canonical_changes(&base_lines, &left_lines, MergeSide::Left);
    changes.extend(canonical_changes(
        &base_lines,
        &right_lines,
        MergeSide::Right,
    ));

    let count = changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            let grandparent = parent[parent[x]];
            parent[x] = grandparent;
            x = grandparent;
        }
        x
    }

    for i in 0..count {
        for j in (i + 1)..count {
            if interact(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut grouped: Vec<Vec<usize>> = Vec::new();
    for i in 0..count {
        let r = root(&mut parent, i);
        let group_index = grouped.iter().position(|g| {
            g.first()
                .map(|&member| root(&mut parent, member) == r)
                .unwrap_or(false)
        });
        if let Some(index) = group_index {
            grouped[index].push(i);
        } else {
            grouped.push(vec![i]);
        }
    }

    let mut components: Vec<Vec<usize>> = grouped;
    components.sort_by_key(|members| {
        let start = members
            .iter()
            .map(|&i| changes[i].start)
            .min()
            .unwrap_or(0);
        let end = members
            .iter()
            .map(|&i| changes[i].end)
            .max()
            .unwrap_or(start);
        (start, usize::from(start < end))
    });

    let mut output_lines = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for members in components {
        let region_start = members
            .iter()
            .map(|&i| changes[i].start)
            .min()
            .unwrap_or(base_cursor);
        let region_end = members
            .iter()
            .map(|&i| changes[i].end)
            .max()
            .unwrap_or(region_start);

        if base_cursor < region_start {
            append_lines(&mut output_lines, &base_lines[base_cursor..region_start]);
        }

        let base_projection = base_lines[region_start..region_end].to_vec();

        let mut left_changes: Vec<&Change> = members
            .iter()
            .map(|&i| &changes[i])
            .filter(|change| change.side == MergeSide::Left)
            .collect();
        let mut right_changes: Vec<&Change> = members
            .iter()
            .map(|&i| &changes[i])
            .filter(|change| change.side == MergeSide::Right)
            .collect();

        left_changes.sort_by_key(|change| (change.start, change.end));
        right_changes.sort_by_key(|change| (change.start, change.end));

        let left_projection = project(
            &base_lines,
            region_start,
            region_end,
            &left_changes,
        );
        let right_projection = project(
            &base_lines,
            region_start,
            region_end,
            &right_changes,
        );

        if left_projection == right_projection {
            append_lines(&mut output_lines, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output_lines, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output_lines, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output_lines, &left_projection),
                Resolution::PreferRight => append_lines(&mut output_lines, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines.len();
                    output_lines.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output_lines, &left_projection);
                    let base_label_start = if region_start < region_end {
                        region_start + 1
                    } else {
                        region_start
                    };
                    output_lines.push(format!(
                        "||||||| BASE {base_label_start},{}",
                        region_end - region_start
                    ));
                    append_lines(&mut output_lines, &base_projection);
                    output_lines.push("=======".to_owned());
                    append_lines(&mut output_lines, &right_projection);
                    output_lines.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: region_start,
                        base_len: region_end - region_start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = base_cursor.max(region_end);
    }

    if base_cursor < base_lines.len() {
        append_lines(&mut output_lines, &base_lines[base_cursor..]);
    }

    let mut text = String::new();
    for line in &output_lines {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change<'a> {
    start: usize,
    end: usize,
    replacement: Vec<&'a str>,
}

#[derive(Clone)]
struct Component {
    members: Vec<(bool, usize)>,
    start: usize,
    end: usize,
}

fn lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }

    text[..text.len() - 1].split('\n').collect()
}

fn edit_script<'a>(old: &[&'a str], new: &[&'a str]) -> Vec<Change<'a>> {
    let rows = old.len() + 1;
    let cols = new.len() + 1;
    let mut lcs = vec![vec![0usize; cols]; rows];

    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut ops: Vec<(u8, Option<&'a str>)> = Vec::new();
    let (mut i, mut j) = (0, 0);

    while i < old.len() || j < new.len() {
        if i < old.len() && j < new.len() && old[i] == new[j] {
            ops.push((0, None));
            i += 1;
            j += 1;
        } else if i == old.len() {
            ops.push((2, Some(new[j])));
            j += 1;
        } else if j == new.len() {
            ops.push((1, None));
            i += 1;
        } else {
            let delete_lcs = lcs[i + 1][j];
            let insert_lcs = lcs[i][j + 1];
            if delete_lcs > insert_lcs
                || (delete_lcs == insert_lcs && (i + j) % 2 == 0)
            {
                ops.push((1, None));
                i += 1;
            } else {
                ops.push((2, Some(new[j])));
                j += 1;
            }
        }
    }

    let mut changes = Vec::new();
    let mut base_pos = 0;
    let mut cursor = 0;

    while cursor < ops.len() {
        if ops[cursor].0 == 0 {
            base_pos += 1;
            cursor += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0;
        let mut replacement = Vec::new();

        while cursor < ops.len() && ops[cursor].0 != 0 {
            match ops[cursor] {
                (1, _) => {
                    deleted += 1;
                    base_pos += 1;
                }
                (2, Some(line)) => replacement.push(line),
                _ => unreachable!(),
            }
            cursor += 1;
        }

        changes.push(Change {
            start,
            end: start + deleted,
            replacement,
        });
    }

    changes
}

fn interacts(a: &Change<'_>, b: &Change<'_>) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;

    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection<'a>(
    base: &[&'a str],
    start: usize,
    end: usize,
    changes: &[&Change<'a>],
) -> Vec<&'a str> {
    let mut output = Vec::new();
    let mut cursor = start;

    for change in changes {
        output.extend_from_slice(&base[cursor..change.start]);
        output.extend_from_slice(&change.replacement);
        cursor = change.end;
    }

    output.extend_from_slice(&base[cursor..end]);
    output
}

fn append_line(output: &mut String, line: &str) {
    output.push_str(line);
    output.push('\n');
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let mut nodes: Vec<(bool, usize)> = (0..left_changes.len())
        .map(|i| (true, i))
        .chain((0..right_changes.len()).map(|i| (false, i)))
        .collect();
    let mut parent: Vec<usize> = (0..nodes.len()).collect();

    fn root(parent: &mut [usize], mut i: usize) -> usize {
        while parent[i] != i {
            parent[i] = parent[parent[i]];
            i = parent[i];
        }
        i
    }

    for i in 0..nodes.len() {
        for j in (i + 1)..nodes.len() {
            let (left_i, index_i) = nodes[i];
            let (left_j, index_j) = nodes[j];
            let a = if left_i {
                &left_changes[index_i]
            } else {
                &right_changes[index_i]
            };
            let b = if left_j {
                &left_changes[index_j]
            } else {
                &right_changes[index_j]
            };

            if interacts(a, b) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut components: Vec<Component> = Vec::new();
    for (node, member) in nodes.drain(..).enumerate() {
        let r = root(&mut parent, node);
        if let Some(component) = components.iter_mut().find(|c| c.members.first().map(|m| {
            let index = if m.0 { m.1 } else { left_changes.len() + m.1 };
            root(&mut parent, index) == r
        }).unwrap_or(false)) {
            component.members.push(member);
        } else {
            let change = if member.0 {
                &left_changes[member.1]
            } else {
                &right_changes[member.1]
            };
            components.push(Component {
                members: vec![member],
                start: change.start,
                end: change.end,
            });
        }
    }

    for component in &mut components {
        for &(is_left, index) in &component.members {
            let change = if is_left {
                &left_changes[index]
            } else {
                &right_changes[index]
            };
            component.start = component.start.min(change.start);
            component.end = component.end.max(change.end);
        }
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
    });

    let mut output = String::new();
    let mut output_line_count = 0usize;
    let mut base_cursor = 0usize;
    let mut conflicts = Vec::new();

    for component in components {
        for line in &base_lines[base_cursor..component.start] {
            append_line(&mut output, line);
            output_line_count += 1;
        }

        let base_projection = &base_lines[component.start..component.end];
        let mut left_component_changes: Vec<&Change<'_>> = component
            .members
            .iter()
            .filter_map(|&(is_left, index)| is_left.then_some(&left_changes[index]))
            .collect();
        let mut right_component_changes: Vec<&Change<'_>> = component
            .members
            .iter()
            .filter_map(|&(is_left, index)| (!is_left).then_some(&right_changes[index]))
            .collect();
        left_component_changes.sort_by_key(|c| c.start);
        right_component_changes.sort_by_key(|c| c.start);

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_component_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_component_changes,
        );

        let emitted: Vec<&str>;
        if left_projection == right_projection {
            emitted = left_projection;
        } else if left_projection == base_projection {
            emitted = right_projection;
        } else if right_projection == base_projection {
            emitted = left_projection;
        } else {
            match resolution {
                Resolution::PreferLeft => emitted = left_projection,
                Resolution::PreferRight => emitted = right_projection,
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_line_count;
                    let base_len = component.end - component.start;
                    let base_number = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    append_line(&mut output, &format!("<<<<<<< LEFT {ordinal}"));
                    output_line_count += 1;
                    for line in &left_projection {
                        append_line(&mut output, line);
                        output_line_count += 1;
                    }
                    append_line(
                        &mut output,
                        &format!("||||||| BASE {base_number},{base_len}"),
                    );
                    output_line_count += 1;
                    for line in base_projection {
                        append_line(&mut output, line);
                        output_line_count += 1;
                    }
                    append_line(&mut output, "=======");
                    output_line_count += 1;
                    for line in &right_projection {
                        append_line(&mut output, line);
                        output_line_count += 1;
                    }
                    append_line(&mut output, &format!(">>>>>>> RIGHT {ordinal}"));
                    output_line_count += 1;

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                    base_cursor = component.end;
                    continue;
                }
            }
        }

        for line in emitted {
            append_line(&mut output, line);
            output_line_count += 1;
        }
        base_cursor = component.end;
    }

    for line in &base_lines[base_cursor..] {
        append_line(&mut output, line);
    }

    Ok(MergeResult {
        text: output,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
