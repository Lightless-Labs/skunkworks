#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn validate_expression(template: &str, open: usize, close: usize) -> Result<(), Error> {
    let bytes = template.as_bytes();
    let content_start = open + 1;

    if let Some(inner) = bytes[content_start..close]
        .iter()
        .position(|&byte| byte == b'{')
    {
        return Err(Error::NestedOpen {
            pos: content_start + inner,
        });
    }

    let mut start = content_start;
    if matches!(
        bytes.get(start),
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    ) {
        return Err(Error::UnsupportedOperator { pos: start });
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut seen: Vec<&str> = Vec::new();

    loop {
        let end = bytes[start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |offset| start + offset);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let name_bytes = &bytes[start..end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: start });
        }

        if let Some(offset) = name_bytes[1..]
            .iter()
            .position(|&byte| !is_name_continue(byte))
        {
            return Err(Error::InvalidName {
                pos: start + 1 + offset,
            });
        }

        let name = &template[start..end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(name);

        if end == close {
            break;
        }
        start = end + 1;
    }

    Ok(())
}

fn validate_template(template: &str) -> Result<(), Error> {
    let bytes = template.as_bytes();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open = pos;
                let close = match bytes[pos + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(offset) => pos + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                validate_expression(template, open, close)?;
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(())
}

fn encode_value(value: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == b'%' && pos + 2 < bytes.len() {
            let high = bytes[pos + 1];
            let low = bytes[pos + 2];
            if high.is_ascii_hexdigit() && low.is_ascii_hexdigit() {
                output.push('%');
                output.push((high as char).to_ascii_uppercase());
                output.push((low as char).to_ascii_uppercase());
                pos += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }

        pos += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    validate_template(template)?;

    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        if bytes[pos] != b'{' {
            let next = bytes[pos..]
                .iter()
                .position(|&byte| byte == b'{')
                .map_or(bytes.len(), |offset| pos + offset);
            output.push_str(&template[pos..next]);
            pos = next;
            continue;
        }

        let close = pos + 1
            + bytes[pos + 1..]
                .iter()
                .position(|&byte| byte == b'}')
                .expect("template syntax was validated");

        let content_start = pos + 1;
        let mut start = content_start;
        let mut first = true;

        loop {
            let end = bytes[start..close]
                .iter()
                .position(|&byte| byte == b',')
                .map_or(close, |offset| start + offset);
            let name = &template[start..end];

            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                if !first {
                    output.push(',');
                }
                output.push_str(&encode_value(value));
                first = false;
            }

            if end == close {
                break;
            }
            start = end + 1;
        }

        pos = close + 1;
    }

    Ok(output)
}
