```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct JobState {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    level: usize,
    burst_pos: usize,
    remaining: u32,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

fn schedule_mean(total: u64, count: u64) -> u64 {
    (total * 100 + count / 2) / count
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut states: Vec<JobState> = jobs
        .iter()
        .map(|job| JobState {
            id: job.id,
            arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            level: job.level,
            burst_pos: 0,
            remaining: job.bursts[0],
            completion: None,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut events = Vec::new();
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut next_event = 0usize;
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let admit_through = |limit: u64,
                         events: &[Event],
                         next_event: &mut usize,
                         queues: &mut Vec<VecDeque<usize>>| {
        while *next_event < events.len() && events[*next_event].time <= limit {
            let event = events[*next_event];
            queues[event.level].push_back(event.index);
            *next_event += 1;
        }
    };

    while states.iter().any(|state| state.completion.is_none()) {
        admit_through(time, &events, &mut next_event, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if next_event >= events.len() {
                break;
            }
            let next_time = events[next_event].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let job_index = queues[level].pop_front().unwrap();
        let job_id = states[job_index].id;
        let cost = if last_dispatch_id.is_some_and(|last_id| last_id != job_id) {
            u64::from(switch_cost)
        } else {
            0
        };
        time += cost;
        admit_through(time, &events, &mut next_event, &mut queues);

        let run_start = time;
        let consumed = states[job_index].remaining.min(quanta[level]);
        time += u64::from(consumed);
        states[job_index].remaining -= consumed;

        let burst_finished = states[job_index].remaining == 0;
        let job_finished = burst_finished && states[job_index].burst_pos + 1 == states[job_index].bursts.len();

        runs.push(Run {
            id: job_id,
            start: run_start,
            end: time,
            finished: job_finished,
            level,
        });

        admit_through(time, &events, &mut next_event, &mut queues);

        if job_finished {
            states[job_index].completion = Some(time);
        } else if burst_finished {
            let io_delay = u64::from(states[job_index].bursts[states[job_index].burst_pos + 1]);
            let wake_level = level.saturating_sub(1);
            states[job_index].burst_pos += 2;
            states[job_index].remaining = states[job_index].bursts[states[job_index].burst_pos];
            events.push(Event {
                time: time + io_delay,
                kind: 1,
                index: job_index,
                level: wake_level,
            });
            events[next_event..].sort_by_key(|event| (event.time, event.kind, event.index));
            admit_through(time, &events, &mut next_event, &mut queues);
        } else {
            let demoted = (level + 1).min(quanta.len() - 1);
            states[job_index].level = demoted;
            queues[demoted].push_back(job_index);
        }

        last_dispatch_id = Some(job_id);
    }

    let end = states
        .iter()
        .filter_map(|state| state.completion)
        .max()
        .unwrap_or(0);

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for state in &states {
        let completion = state.completion.unwrap_or(0);
        let turnaround = completion - state.arrival;
        let work: u64 = state.bursts.iter().map(|&burst| u64::from(burst)).sum();
        let waiting = turnaround - work;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: state.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: if count == 0 {
            0
        } else {
            schedule_mean(total_waiting, count)
        },
        mean_turnaround_x100: if count == 0 {
            0
        } else {
            schedule_mean(total_turnaround, count)
        },
        end,
    })
}

use std::collections::VecDeque;
```
