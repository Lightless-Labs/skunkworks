#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

struct Running {
    task: usize,
    worker: usize,
    finish: u64,
}

fn append_usage(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
        return;
    }

    for (i, value) in used.iter().enumerate() {
        if i > 0 {
            trace.push(',');
        }
        trace.push_str(&value.to_string());
    }
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependencies in input order, preserving the specified error precedence.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resources task-by-task, checking each shape before its demands.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = Vec::<usize>::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        order.push(id);

        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut was_removed = vec![false; n];
            for &id in &order {
                was_removed[id] = true;
            }
            was_removed
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];
    let mut running: Vec<Running> = Vec::new();
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut completing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(i, entry)| (entry.finish == now).then_some(i))
            .collect();
        completing.sort_by_key(|&i| running[i].task);

        for index in completing.into_iter().rev() {
            let entry = running.remove(index);
            for (resource, &need) in tasks[entry.task].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            completed[entry.task] = true;
            completed_count += 1;

            trace.push_str(&format!(
                "{}|end|{}|{}|",
                now, entry.task, entry.worker
            ));
            append_usage(&mut trace, &used);
            trace.push('\n');
        }

        loop {
            let worker = (0..worker_count)
                .filter(|&w| worker_available[w] <= now)
                .min_by_key(|&w| {
                    // Cyclic distance from each candidate task is task-dependent;
                    // worker selection is performed after choosing the task below.
                    w
                });

            let mut best_task: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || !tasks[id].deps.iter().all(|&dep| completed[dep])
                {
                    continue;
                }

                let fits = tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| {
                        u64::from(need) <= u64::from(capacity[r]) - used[r]
                    });
                if !fits {
                    continue;
                }

                let Some(w) = worker else {
                    break;
                };
                let _ = w;

                let is_better = match best_task {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if is_better {
                    best_task = Some(id);
                }
            }

            let Some(task_id) = best_task else {
                break;
            };

            // Choose the first available worker in cyclic order from task_id % W.
            let preferred = task_id % worker_count;
            let assigned_worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&w| worker_available[w] <= now);

            let Some(worker_id) = assigned_worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let turnaround = finish
                .checked_add(cooldowns[worker_id])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker_id] = turnaround;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }

            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });

            trace.push_str(&format!(
                "{}|start|{}|{}|",
                now, task_id, worker_id
            ));
            append_usage(&mut trace, &used);
            trace.push('\n');

            makespan = makespan.max(finish);

            if finish == now {
                // Zero-duration tasks complete immediately before the next admission.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;

                trace.push_str(&format!(
                    "{}|end|{}|{}|",
                    now, task_id, worker_id
                ));
                append_usage(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running {
                    task: task_id,
                    worker: worker_id,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now && next_time.map_or(true, |current| candidate < current) {
                next_time = Some(candidate);
            }
        };

        for entry in &running {
            consider(entry.finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > now {
                consider(worker_available[worker]);
            }
        }

        let next = next_time.expect("unfinished tasks always have a future event");
        now = next;
    }

    trace.push_str(&format!("makespan={makespan}\n"));

    Ok(Schedule {
        runs: runs
            .into_iter()
            .map(|run| run.expect("every task must have started"))
            .collect(),
        makespan,
        trace,
    })
}
