use std::str::FromStr;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn width(field: Field) -> u32 {
    let (a, b) = bounds(field);
    b - a + 1
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    dom_unrestricted: bool,
    dow_unrestricted: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (lo, hi) = bounds(field);
        let set = &self.sets[field_index(field)];
        (lo..=hi)
            .filter(|&v| set[v as usize])
            .map(|v| v as u8)
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(&at)?;
        Ok(self.matches_valid(&at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(&after)?;
        let mut date = (after.year, after.month, after.day);
        loop {
            if self.date_matches(date.0, date.1, date.2) {
                for h in self.allowed(Field::Hour) {
                    for m in self.allowed(Field::Minute) {
                        let candidate = CivilDateTime {
                            year: date.0,
                            month: date.1,
                            day: date.2,
                            hour: h,
                            minute: m,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if !advance_date(&mut date.0, &mut date.1, &mut date.2) {
                return Ok(None);
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(&before)?;
        let mut date = (before.year, before.month, before.day);
        loop {
            if self.date_matches(date.0, date.1, date.2) {
                let hours = self.allowed(Field::Hour);
                let minutes = self.allowed(Field::Minute);
                for &h in hours.iter().rev() {
                    for &m in minutes.iter().rev() {
                        let candidate = CivilDateTime {
                            year: date.0,
                            month: date.1,
                            day: date.2,
                            hour: h,
                            minute: m,
                        };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if !retreat_date(&mut date.0, &mut date.1, &mut date.2) {
                return Ok(None);
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(&start).map_err(|e| WindowError::InvalidStart {
            component: match e {
                DateError::InvalidDateTime { component } => component,
            },
        })?;
        validate_datetime(&end).map_err(|e| WindowError::InvalidEnd {
            component: match e {
                DateError::InvalidDateTime { component } => component,
            },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(next) = self.next_fire(cursor).map_err(|_| unreachable!())? {
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(next);
            cursor = next;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut out = Vec::new();
        for field in FIELDS {
            let (lo, hi) = bounds(field);
            let vals = self.allowed(field);
            let full = vals.len() == (hi - lo + 1) as usize;
            let text = match field {
                Field::DayOfMonth if self.dom_unrestricted => "*".to_string(),
                Field::DayOfWeek if self.dow_unrestricted => "*".to_string(),
                Field::DayOfMonth | Field::DayOfWeek if full => "*/1".to_string(),
                _ if full => "*".to_string(),
                _ => vals.iter().map(u8::to_string).collect::<Vec<_>>().join(","),
            };
            out.push(text);
        }
        out.join(" ")
    }

    fn matches_valid(&self, at: &CivilDateTime) -> bool {
        self.sets[0][at.minute as usize]
            && self.sets[1][at.hour as usize]
            && self.sets[3][at.month as usize]
            && self.day_rule(at.year, at.month, at.day)
    }

    fn date_matches(&self, year: u16, month: u8, day: u8) -> bool {
        self.sets[3][month as usize] && self.day_rule(year, month, day)
    }

    fn day_rule(&self, year: u16, month: u8, day: u8) -> bool {
        let dom = self.sets[2][day as usize];
        let dow = self.sets[4][weekday(year, month, day) as usize];
        match (self.dom_unrestricted, self.dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if day == days_in_month(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate_datetime(at: &CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_date(year: &mut u16, month: &mut u8, day: &mut u8) -> bool {
    if *day < days_in_month(*year, *month) {
        *day += 1;
    } else if *month < 12 {
        *month += 1;
        *day = 1;
    } else if *year < 2399 {
        *year += 1;
        *month = 1;
        *day = 1;
    } else {
        return false;
    }
    true
}

fn retreat_date(year: &mut u16, month: &mut u8, day: &mut u8) -> bool {
    if *day > 1 {
        *day -= 1;
    } else if *month > 1 {
        *month -= 1;
        *day = days_in_month(*year, *month);
    } else if *year > 2000 {
        *year -= 1;
        *month = 12;
        *day = 31;
    } else {
        return false;
    }
    true
}

fn name_value(field: Field, atom: &str) -> Option<u32> {
    if atom.len() != 3 || !atom.bytes().all(|b| b.is_ascii_alphabetic()) {
        return None;
    }
    let upper = atom.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn valid_atom_shape(field: Field, atom: &str) -> bool {
    !atom.is_empty()
        && (atom.bytes().all(|b| b.is_ascii_digit()) || name_value(field, atom).is_some())
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(field: Field, text: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (lo, hi) = bounds(field);
    if value < lo || value > hi {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(value)
}

fn parse_step(field: Field, text: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if value == 0 || value > width(field) {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_atom_value(field: Field, atom: &str, offset: usize) -> Result<u32, ParseError> {
    if atom.bytes().all(|b| b.is_ascii_digit()) {
        parse_number(field, atom, offset)
    } else {
        Ok(name_value(field, atom).expect("atom shape checked"))
    }
}

fn parse_item(
    field: Field,
    item: &str,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    if slash_count > 1 {
        return Err(syntax_error(field, item_offset));
    }
    let (base, step_text, step_offset) = if let Some(pos) = item.find('/') {
        (&item[..pos], Some(&item[pos + 1..]), item_offset + pos + 1)
    } else {
        (item, None, item_offset)
    };
    if base.is_empty() || step_text == Some("") {
        return Err(syntax_error(field, item_offset));
    }

    let (base_kind, first_text, second_text, second_offset) = if base == "*" {
        (0u8, "", None, 0usize)
    } else if base.contains('*') {
        return Err(syntax_error(field, item_offset));
    } else {
        let hyphens = base.bytes().filter(|&b| b == b'-').count();
        if hyphens > 1 {
            return Err(syntax_error(field, item_offset));
        }
        if hyphens == 0 {
            if !valid_atom_shape(field, base) {
                return Err(syntax_error(field, item_offset));
            }
            (1u8, base, None, 0usize)
        } else {
            let pos = base.find('-').unwrap();
            let left = &base[..pos];
            let right = &base[pos + 1..];
            if !valid_atom_shape(field, left) || !valid_atom_shape(field, right) {
                return Err(syntax_error(field, item_offset));
            }
            (2u8, left, Some(right), pos + 1)
        }
    };

    if let Some(step) = step_text {
        if !step.bytes().all(|b| b.is_ascii_digit()) {
            return Err(syntax_error(field, item_offset));
        }
    }

    let (lo, hi) = bounds(field);
    let mut start = lo;
    let mut end = hi;
    if base_kind == 1 || base_kind == 2 {
        start = parse_atom_value(field, first_text, item_offset)?;
        if let Some(second) = second_text {
            end = parse_atom_value(field, second, item_offset + second_offset)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_offset + second_offset,
                });
            }
        } else {
            end = if step_text.is_some() { hi } else { start };
        }
    }

    let step = if let Some(step) = step_text {
        parse_step(field, step, step_offset)?
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let text = if input.as_bytes().first() == Some(&b'@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let parts: Vec<&str> = text.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (lo, hi) = bounds(FIELDS[i]);
        vec![false; (hi + 1) as usize.max((lo + 1) as usize)]
    });

    for (index, part) in parts.iter().enumerate() {
        let field = FIELDS[index];
        for (offset, byte) in part.bytes().enumerate() {
            let legal = byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b',');
            if !legal {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut start = 0usize;
        for item in part.split(',') {
            parse_item(field, item, start, &mut sets[index])?;
            start += item.len() + 1;
        }
    }

    Ok(Cron {
        sets,
        dom_unrestricted: parts[2] == "*",
        dow_unrestricted: parts[4] == "*",
    })
}
