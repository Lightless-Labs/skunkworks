```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains(&product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.as_str());

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = match DENOMINATIONS.iter().position(|&d| d == coin) {
            Some(index) => index,
            None => return Err(Error::BadCoin { coin }),
        };

        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(DENOMINATIONS[index]);
        self.credit = new_credit as u16;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let product = match self.products.iter().find(|product| product.id == id) {
            Some(product) => product,
            None => {
                return Err(Error::UnknownProduct {
                    id: id.to_owned(),
                });
            }
        };

        if self.selected.as_deref() == Some(product.id.as_str()) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(product.id.clone());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_deref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };

        let product = match self.products.iter().find(|product| product.id == selected_id) {
            Some(product) => product,
            None => return Err(Error::NoSelection),
        };
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change_counts = match Self::choose_change(amount, &self.tubes) {
            Some(counts) => counts,
            None => return Err(Error::NoChange { amount }),
        };

        let mut proposed_tubes = self.tubes;
        for i in 0..4 {
            proposed_tubes[i] -= change_counts[i];
        }
        for &coin in &self.escrow {
            if let Some(index) = DENOMINATIONS.iter().position(|&d| d == coin) {
                proposed_tubes[index] += 1;
            }
        }

        for (index, &count) in proposed_tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change_counts[i] {
                change.push(DENOMINATIONS[i]);
            }
        }

        let receipt = Receipt {
            id: selected_id.to_owned(),
            price,
            change,
        };

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(refunded)
    }

    fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
        if amount == 0 {
            return Some([0; 4]);
        }

        let mut remaining = amount;
        let mut greedy = [0u16; 4];
        for i in (0..4).rev() {
            let coin = DENOMINATIONS[i];
            let take = tubes[i].min(remaining / coin);
            greedy[i] = take;
            remaining -= take * coin;
        }
        if remaining == 0 {
            return Some(greedy);
        }

        let mut best: Option<[u16; 4]> = None;
        let mut best_count = u32::MAX;

        fn search(
            index: isize,
            remaining: u16,
            tubes: &[u16; 4],
            current: &mut [u16; 4],
            best: &mut Option<[u16; 4]>,
            best_count: &mut u32,
        ) {
            if remaining == 0 {
                let count: u32 = current.iter().map(|&n| n as u32).sum();
                let better = count < *best_count
                    || (count == *best_count
                        && best
                            .as_ref()
                            .map(|old| (0..4).rev().find_map(|i| {
                                if current[i] != old[i] {
                                    Some(current[i] > old[i])
                                } else {
                                    None
                                }
                            }).unwrap_or(false))
                            .unwrap_or(true));
                if better {
                    *best_count = count;
                    *best = Some(*current);
                }
                return;
            }
            if index < 0 {
                return;
            }

            let i = index as usize;
            let coin = DENOMINATIONS[i];
            let max_take = tubes[i].min(remaining / coin);
            for take in (0..=max_take).rev() {
                current[i] = take;
                search(
                    index - 1,
                    remaining - take * coin,
                    tubes,
                    current,
                    best,
                    best_count,
                );
            }
            current[i] = 0;
        }

        search(
            3,
            amount,
            tubes,
            &mut [0; 4],
            &mut best,
            &mut best_count,
        );
        best
    }
}
```
