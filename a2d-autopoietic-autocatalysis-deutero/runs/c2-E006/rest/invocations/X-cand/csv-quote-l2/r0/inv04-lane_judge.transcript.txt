    delimiter: u8,
    quote: u8,

        3
    } else {
        0
    };

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut record_started = false;

        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

                    record_started = true;
                    i += 1;
                }

                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            State::Quoted { opening } => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;

                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {

                    field.push(ch);
                    i += ch.len_utf8();
                    let _ = opening;
                }

                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;

                    i += if b == b'\r' { 2 } else { 1 };

                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,

        }
    }

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    if records.is_empty() {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    }

    let mut iter = records.into_iter();
    let header = iter.next().unwrap();

    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen: std::collections::HashMap<&str, usize> =
            std::collections::HashMap::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name.as_str(), second);
        }
    }

    let expected = header.len();
    let mut rows = Vec::new();
    for (record_index, row) in iter.enumerate() {
        if row.len() != expected {
            return Err(CsvError::HeaderWidth {
                record: record_index + 1,
                expected,
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }

        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

                    });
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::FieldStart => {
            if !row.is_empty() {
                row.push(String::new());
                records.push(row);
            }
        }
        State::Unquoted | State::AfterQuote => {
            row.push(String::from_utf8(field).unwrap());
            records.push(row);
        }
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn make_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    let mut iter = records.into_iter();
    let Some(header) = iter.next() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        for second in 0..header.len() {
            if let Some(first) = header[..second]
                .iter()
                .position(|item| item == &header[second])
            {
                return Err(CsvError::DuplicateHeader {
                    name: header[second].clone(),
                    first,
                    second,
                });
            }
        }
    }

    let mut rows = Vec::new();
    for (index, row) in iter.enumerate() {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: index + 1,
                expected: header.len(),
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

    }

        self.result = None;
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    let records = parse_records(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut records = records.into_iter();
            let header = records.next().unwrap();
            let rows: Vec<Vec<String>> = records.collect();

            if config.duplicates == DuplicatePolicy::Reject {
                for second in 0..header.len() {
                    if let Some(first) = header[..second]
                        .iter()
                        .position(|value| value == &header[second])
                    {
                        return Err(CsvError::DuplicateHeader {
                            name: header[second].clone(),
                            first,
                            second,
                        });
                    }
                }
            }

            for (index, row) in rows.iter().enumerate() {
                if row.len() != header.len() {
                    return Err(CsvError::HeaderWidth {
                        record: index + 1,
                        expected: header.len(),
                        actual: row.len(),
                    });
                }
            }

            (Some(header), rows)
        }
    } else {
        (None, records)
    };

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })

enum ParseState {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut state = ParseState::FieldStart;
    let mut opening_quote = 0usize;
    let mut record_index = 0usize;
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut has_record_content = false;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if state != ParseState::Quoted && byte == b'\r' {
            if pos + 1 >= bytes.len() || bytes[pos + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: pos,
                });
            }

            row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            records.push(std::mem::take(&mut row));
            state = ParseState::FieldStart;
            has_record_content = false;
            record_index += 1;
            pos += 2;
            continue;
        }

        match state {
            ParseState::FieldStart => {
                if byte == delimiter {
                    row.push(String::new());
                    has_record_content = true;
                    pos += 1;
                } else if byte == quote {
                    state = ParseState::Quoted;
                    opening_quote = pos;
                    has_record_content = true;
                    pos += 1;
                } else if byte == b'\n' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    state = ParseState::FieldStart;
                    has_record_content = false;
                    record_index += 1;
                    pos += 1;
                } else {
                    field.push(byte);
                    state = ParseState::Unquoted;
                    has_record_content = true;
                    pos += 1;
                }
            }
            ParseState::Unquoted => {
                if byte == delimiter {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = ParseState::FieldStart;
                    has_record_content = true;
                    pos += 1;
                } else if byte == b'\n' {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut row));
                    state = ParseState::FieldStart;
                    has_record_content = false;
                    record_index += 1;
                    pos += 1;
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                } else {
                    field.push(byte);
                    pos += 1;
                }
            }
            ParseState::Quoted => {
                if byte == quote {
                    if pos + 1 < bytes.len() && bytes[pos + 1] == quote {
                        field.push(quote);
                        pos += 2;
                    } else {
                        state = ParseState::AfterQuote;
                        pos += 1;
                    }
                } else {
                    field.push(byte);
                    pos += 1;
                }
            }
            ParseState::AfterQuote => {
                if byte == delimiter {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = ParseState::FieldStart;
                    has_record_content = true;
                    pos += 1;
                } else if byte == b'\n' {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut row));
                    state = ParseState::FieldStart;
                    has_record_content = false;
                    record_index += 1;
                    pos += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
        }
    }

    if state == ParseState::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if has_record_content || !row.is_empty() || !field.is_empty() {
        row.push(String::from_utf8(field).unwrap());
        records.push(row);
    }

    Ok(records)
}
