#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    let records = parse_records(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut records = records.into_iter();
            let header = records.next().unwrap();
            let rows: Vec<Vec<String>> = records.collect();

            if config.duplicates == DuplicatePolicy::Reject {
                for second in 0..header.len() {
                    if let Some(first) = header[..second]
                        .iter()
                        .position(|value| value == &header[second])
                    {
                        return Err(CsvError::DuplicateHeader {
                            name: header[second].clone(),
                            first,
                            second,
                        });
                    }
                }
            }

            for (index, row) in rows.iter().enumerate() {
                if row.len() != header.len() {
                    return Err(CsvError::HeaderWidth {
                        record: index + 1,
                        expected: header.len(),
                        actual: row.len(),
                    });
                }
            }

            (Some(header), rows)
        }
    } else {
        (None, records)
    };

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum ParseState {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut state = ParseState::FieldStart;
    let mut opening_quote = 0usize;
    let mut record_index = 0usize;
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut has_record_content = false;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if state != ParseState::Quoted && byte == b'\r' {
            if pos + 1 >= bytes.len() || bytes[pos + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: pos,
                });
            }

            row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            records.push(std::mem::take(&mut row));
            state = ParseState::FieldStart;
            has_record_content = false;
            record_index += 1;
            pos += 2;
            continue;
        }

        match state {
            ParseState::FieldStart => {
                if byte == delimiter {
                    row.push(String::new());
                    has_record_content = true;
                    pos += 1;
                } else if byte == quote {
                    state = ParseState::Quoted;
                    opening_quote = pos;
                    has_record_content = true;
                    pos += 1;
                } else if byte == b'\n' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    state = ParseState::FieldStart;
                    has_record_content = false;
                    record_index += 1;
                    pos += 1;
                } else {
                    field.push(byte);
                    state = ParseState::Unquoted;
                    has_record_content = true;
                    pos += 1;
                }
            }
            ParseState::Unquoted => {
                if byte == delimiter {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = ParseState::FieldStart;
                    has_record_content = true;
                    pos += 1;
                } else if byte == b'\n' {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut row));
                    state = ParseState::FieldStart;
                    has_record_content = false;
                    record_index += 1;
                    pos += 1;
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                } else {
                    field.push(byte);
                    pos += 1;
                }
            }
            ParseState::Quoted => {
                if byte == quote {
                    if pos + 1 < bytes.len() && bytes[pos + 1] == quote {
                        field.push(quote);
                        pos += 2;
                    } else {
                        state = ParseState::AfterQuote;
                        pos += 1;
                    }
                } else {
                    field.push(byte);
                    pos += 1;
                }
            }
            ParseState::AfterQuote => {
                if byte == delimiter {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = ParseState::FieldStart;
                    has_record_content = true;
                    pos += 1;
                } else if byte == b'\n' {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut row));
                    state = ParseState::FieldStart;
                    has_record_content = false;
                    record_index += 1;
                    pos += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
        }
    }

    if state == ParseState::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if has_record_content || !row.is_empty() || !field.is_empty() {
        row.push(String::from_utf8(field).unwrap());
        records.push(row);
    }

    Ok(records)
}
