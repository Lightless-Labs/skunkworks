#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Variable {
    name: String,
    start: usize,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    operator_pos: Option<usize>,
    variables: Vec<Variable>,
}

#[derive(Debug, Clone)]
enum Part {
    Literal(String),
    Expression(Expression),
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = parse(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(&text),
            Part::Expression(expression) => {
                expand_expression(&mut output, &expression, vars);
            }
        }
    }

    Ok(output)
}

fn parse(template: &str) -> Result<Vec<Part>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[open + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                if open > cursor {
                    parts.push(Part::Literal(template[cursor..open].to_owned()));
                }

                let expression = parse_expression(template, open + 1, close)?;
                parts.push(Part::Expression(expression));
                cursor = close + 1;
            }
            _ => {
                cursor += 1;
            }
        }
    }

    if cursor > 0 {
        let last_end = match parts.last() {
            Some(Part::Expression(_)) => {
                let mut end = 0;
                for part in &parts {
                    match part {
                        Part::Literal(text) => end += text.len(),
                        Part::Expression(_) => end += 2,
                    }
                }
                end
            }
            _ => cursor,
        };
        let _ = last_end;
    }

    // Preserve the trailing literal span, if any.
    let consumed = parts.iter().map(|part| match part {
        Part::Literal(text) => text.len(),
        Part::Expression(expression) => {
            // Recover source span length from variable positions is not possible;
            // the cursor at the end of parsing is the authoritative boundary.
            let _ = expression;
            0
        }
    }).sum::<usize>();
    let _ = consumed;

    // Re-scan solely to append any remaining literal span after the last expression.
    // The primary scan leaves cursor at the end for literals or immediately after
    // the last expression.
    let mut end = 0;
    for part in &parts {
        match part {
            Part::Literal(text) => end += text.len(),
            Part::Expression(_) => {
                // Expression source lengths are represented by the parser cursor
                // bookkeeping below, so this branch is unused for trailing text.
            }
        }
    }
    let _ = end;

    // The loop above advances through literal bytes but does not add them until it
    // meets an expression. Add the suffix after the final expression by finding it.
    let suffix_start = find_suffix_start(template);
    if suffix_start < template.len() {
        parts.push(Part::Literal(template[suffix_start..].to_owned()));
    }

    Ok(parts)
}

fn find_suffix_start(template: &str) -> usize {
    let bytes = template.as_bytes();
    let mut cursor = 0;
    let mut last_close_end = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'{' => {
                if let Some(offset) = bytes[cursor + 1..].iter().position(|&b| b == b'}') {
                    let close = cursor + 1 + offset;
                    last_close_end = close + 1;
                    cursor = close + 1;
                } else {
                    break;
                }
            }
            _ => cursor += 1,
        }
    }

    if last_close_end == 0 {
        0
    } else {
        last_close_end
    }
}

fn parse_expression(template: &str, content_start: usize, close: usize) -> Result<Expression, Error> {
    let content = &template.as_bytes()[content_start..close];
    if content.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let first = content[0];
    let operator = match first {
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' => Some(first as char),
        _ => None,
    };
    let operator_pos = operator.map(|_| content_start);
    let names_start = content_start + usize::from(operator.is_some());

    if names_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut spec_start = names_start;

    while spec_start <= close {
        let spec_end = template.as_bytes()[spec_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &template.as_bytes()[spec_start..spec_end];
        for (offset, &byte) in name_bytes.iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        operator_pos,
        variables,
    })
}

fn expand_expression(output: &mut String, expression: &Expression, vars: &[(&str, &str)]) {
    let operator = expression.operator;
    let mut fields: Vec<(String, String)> = Vec::new();

    for variable in &expression.variables {
        let _ = variable.start;
        if let Some(value) = vars
            .iter()
            .rev()
            .find_map(|(name, value)| (*name == variable.name).then_some(*value))
        {
            let encoded = encode(value, matches!(operator, Some('+') | Some('#')));
            fields.push((variable.name.clone(), encoded));
        }
    }

    if fields.is_empty() {
        return;
    }

    let separator = match operator {
        Some('.') => '.',
        Some('/') => '/',
        Some(';') => ';',
        Some('?') | Some('&') => '&',
        _ => ',',
    };

    match operator {
        Some('#') => output.push('#'),
        Some('.') => output.push('.'),
        Some('/') => output.push('/'),
        Some(';') => output.push(';'),
        Some('?') => output.push('?'),
        Some('&') => output.push('&'),
        _ => {}
    }

    for (index, (name, value)) in fields.iter().enumerate() {
        if index > 0 {
            output.push(separator);
        }

        match operator {
            Some(';') => {
                output.push_str(name);
                if !value.is_empty() {
                    output.push('=');
                    output.push_str(value);
                }
            }
            Some('?') | Some('&') => {
                output.push_str(name);
                output.push('=');
                output.push_str(value);
            }
            _ => output.push_str(value),
        }
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
