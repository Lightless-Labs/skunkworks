pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    start: usize,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    specifiers: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn supported_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn parse_expression(content: &str, content_start: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let mut operator = None;
    let mut offset = 0;

    if let Some(&first) = bytes.first() {
        if supported_operator(first) {
            operator = Some(first as char);
            offset = 1;
        }
    }

    if offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specifiers = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut item_start = offset;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(Error::EmptyVariable {
                pos: content_start + item_start,
            });
        }

        let name_bytes = &bytes[item_start..item_end];
        let mut invalid = None;

        if !is_name_start(name_bytes[0]) {
            invalid = Some(item_start);
        } else if let Some(relative) = name_bytes[1..]
            .iter()
            .position(|&b| !is_name_continue(b))
        {
            invalid = Some(item_start + 1 + relative);
        }

        if let Some(relative) = invalid {
            return Err(Error::InvalidName {
                pos: content_start + relative,
            });
        }

        let name = content[item_start..item_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable {
                pos: content_start + item_start,
            });
        }

        seen.push(name.clone());
        specifiers.push(Specifier {
            name,
            start: content_start + item_start,
        });

        if item_end == bytes.len() {
            break;
        }

        item_start = item_end + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open_pos = index;
                let close_pos = match bytes[index + 1..].iter().position(|&b| b == b'}') {
                    Some(relative) => index + 1 + relative,

            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            output.push('%');
            output.push(hex_upper(bytes[index + 1]) as char);
            output.push(hex_upper(bytes[index + 2]) as char);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

    vars.iter()

fn expand_expression(expression: &Expression, vars: &[(&str, &str)], output: &mut String) {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some('+' | '#'));
    let named = matches!(operator, Some(';' | '?' | '&'));

    let mut fields = Vec::<String>::new();

    for specifier in &expression.specifiers {
        if let Some(value) = lookup(&specifier.name, vars) {
            let encoded = encode(value, allow_reserved);
            if named {
                if operator == Some(';') && encoded.is_empty() {
                    fields.push(specifier.name.clone());
                } else {
                    fields.push(format!("{}={}", specifier.name, encoded));
                }
            } else {
                fields.push(encoded);
            }
        }
    }

    if fields.is_empty() {
        return;
    }

    let (separator, prefix) = match operator {
        Some('#') => (",", "#"),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("&", "?"),
        Some('&') => ("&", "&"),
        _ => (",", ""),
    };

    output.push_str(prefix);
    output.push_str(&fields.join(separator));
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::with_capacity(template.len());
    let mut index = 0;

    while index < template.len() {
        if template.as_bytes()[index] == b'{' {
            let close_pos = template.as_bytes()[index + 1..]
                .iter()
                .position(|&b| b == b'}')
                .map(|relative| index + 1 + relative)
                .expect("validated expression must have a closing brace");

            let expression = expressions
                .get(
                    template.as_bytes()[..index]
                        .iter()
                        .filter(|&&b| b == b'{')
                        .count(),
                )
                .expect("validated expression must exist");

            expand_expression(expression, vars, &mut output);
            index = close_pos + 1;
        } else {
            let next_open = template.as_bytes()[index..]
                .iter()
                .position(|&b| b == b'{')
                .map(|relative| index + relative)
                .unwrap_or(template.len());
            output.push_str(&template[index..next_open]);
            index = next_open;
        }
    }

    Ok(output)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

    name: String,
    start: usize,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<u8>,
    specs: Vec<Specifier>,
}

#[derive(Debug, Clone)]
enum Segment {
    Literal { start: usize, end: usize },
    Expression(Expression),
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn validate_expression(content: &str, content_start: usize, close: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let mut operator = None;
    let mut spec_start = 0;

    if let Some(&first) = bytes.first() {
        if matches!(first, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
            operator = Some(first);
            spec_start = 1;
        }
    }

    if spec_start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = spec_start;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: content_start + start,
            });
        }

        let name_bytes = &bytes[start..end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName {
                pos: content_start + start,
            });
        }
        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_byte(byte) {
                return Err(Error::InvalidName {
                    pos: content_start + start + offset,
                });
            }
        }

        let name = content[start..end].to_owned();
        if seen.iter().any(|prior| prior == &name) {
            return Err(Error::DuplicateVariable {
                pos: content_start + start,
            });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            start: content_start + start,
        });

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::EmptyVariable {
                pos: content_start + start,
            });
        }
    }

    Ok(Expression { operator, specs })
}

fn validate_template(template: &str) -> Result<Vec<Segment>, Error> {
    let bytes = template.as_bytes();
    let mut segments = Vec::new();
    let mut cursor = 0;
    let mut literal_start = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                if literal_start < cursor {
                    segments.push(Segment::Literal {
                        start: literal_start,
                        end: cursor,
                    });
                }

                let open = cursor;
                let close = bytes[open + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                    .map(|offset| open + 1 + offset)
                    .ok_or(Error::Unclosed { pos: open })?;

                if let Some(offset) = bytes[open + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let content_start = open + 1;
                let content = &template[content_start..close];
                segments.push(Segment::Expression(validate_expression(
                    content,
                    content_start,
                    close,
                )?));

                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    if literal_start < bytes.len() {
        segments.push(Segment::Literal {
            start: literal_start,
            end: bytes.len(),
        });
    }

    Ok(segments)
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' {
            if i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
                output.push('%');
                output.push(upper_hex(bytes[i + 1]) as char);
                output.push(upper_hex(bytes[i + 2]) as char);
                i += 3;
                continue;
            }
            push_percent_encoded(&mut output, byte);
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            push_percent_encoded(&mut output, byte);
        }

        i += 1;
    }

    output
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {

        .rev()

fn expand_expression(expression: &Expression, vars: &[(&str, &str)], output: &mut String) {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some(b'+') | Some(b'#'));
    let mut fields: Vec<(&Specifier, String)> = Vec::new();

    for spec in &expression.specs {
        if let Some(value) = lookup(vars, &spec.name) {
            fields.push((spec, encode_value(value, allow_reserved)));
        }
    }

    if fields.is_empty() {
        return;
    }

    match operator {
        Some(b'#') | Some(b'.') | Some(b'/') | Some(b';') | Some(b'?') | Some(b'&') => {
            output.push(operator.unwrap() as char);
        }
        _ => {}
    }

    let separator = match operator {
        Some(b'.') => '.',
        Some(b'/') => '/',
        Some(b';') => ';',
        Some(b'?') | Some(b'&') => '&',
        _ => ',',
    };

    for (index, (spec, value)) in fields.iter().enumerate() {
        if index != 0 {
            output.push(separator);
        }

        match operator {
            Some(b';') => {
                output.push_str(&spec.name);
                if !value.is_empty() {
                    output.push('=');
                    output.push_str(value);
                }
            }
            Some(b'?') | Some(b'&') => {
                output.push_str(&spec.name);
                output.push('=');
                output.push_str(value);
            }
            _ => output.push_str(value),
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let segments = validate_template(template)?;
    let mut output = String::new();

    for segment in segments {
        match segment {
            Segment::Literal { start, end } => output.push_str(&template[start..end]),
            Segment::Expression(expression) => {
                expand_expression(&expression, vars, &mut output);
            }
        }
    }

    Ok(output)
}
