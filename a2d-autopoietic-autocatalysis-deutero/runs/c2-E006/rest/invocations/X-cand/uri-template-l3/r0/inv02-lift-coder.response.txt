#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    start: usize,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    specifiers: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn supported_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn parse_expression(content: &str, content_start: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let mut operator = None;
    let mut offset = 0;

    if let Some(&first) = bytes.first() {
        if supported_operator(first) {
            operator = Some(first as char);
            offset = 1;
        }
    }

    if offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specifiers = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut item_start = offset;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(Error::EmptyVariable {
                pos: content_start + item_start,
            });
        }

        let name_bytes = &bytes[item_start..item_end];
        let mut invalid = None;

        if !is_name_start(name_bytes[0]) {
            invalid = Some(item_start);
        } else if let Some(relative) = name_bytes[1..]
            .iter()
            .position(|&b| !is_name_continue(b))
        {
            invalid = Some(item_start + 1 + relative);
        }

        if let Some(relative) = invalid {
            return Err(Error::InvalidName {
                pos: content_start + relative,
            });
        }

        let name = content[item_start..item_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable {
                pos: content_start + item_start,
            });
        }

        seen.push(name.clone());
        specifiers.push(Specifier {
            name,
            start: content_start + item_start,
        });

        if item_end == bytes.len() {
            break;
        }

        item_start = item_end + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open_pos = index;
                let close_pos = match bytes[index + 1..].iter().position(|&b| b == b'}') {
                    Some(relative) => index + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open_pos }),
                };

                if let Some(relative) = bytes[index + 1..close_pos]
                    .iter()
                    .position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + relative,
                    });
                }

                let content_start = open_pos + 1;
                let content = &template[content_start..close_pos];
                expressions.push(parse_expression(content, content_start, close_pos)?);
                index = close_pos + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_upper(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            output.push('%');
            output.push(hex_upper(bytes[index + 1]) as char);
            output.push(hex_upper(bytes[index + 2]) as char);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)], output: &mut String) {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some('+' | '#'));
    let named = matches!(operator, Some(';' | '?' | '&'));

    let mut fields = Vec::<String>::new();

    for specifier in &expression.specifiers {
        if let Some(value) = lookup(&specifier.name, vars) {
            let encoded = encode(value, allow_reserved);
            if named {
                if operator == Some(';') && encoded.is_empty() {
                    fields.push(specifier.name.clone());
                } else {
                    fields.push(format!("{}={}", specifier.name, encoded));
                }
            } else {
                fields.push(encoded);
            }
        }
    }

    if fields.is_empty() {
        return;
    }

    let (separator, prefix) = match operator {
        Some('#') => (",", "#"),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("&", "?"),
        Some('&') => ("&", "&"),
        _ => (",", ""),
    };

    output.push_str(prefix);
    output.push_str(&fields.join(separator));
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::with_capacity(template.len());
    let mut index = 0;

    while index < template.len() {
        if template.as_bytes()[index] == b'{' {
            let close_pos = template.as_bytes()[index + 1..]
                .iter()
                .position(|&b| b == b'}')
                .map(|relative| index + 1 + relative)
                .expect("validated expression must have a closing brace");

            let expression = expressions
                .get(
                    template.as_bytes()[..index]
                        .iter()
                        .filter(|&&b| b == b'{')
                        .count(),
                )
                .expect("validated expression must exist");

            expand_expression(expression, vars, &mut output);
            index = close_pos + 1;
        } else {
            let next_open = template.as_bytes()[index..]
                .iter()
                .position(|&b| b == b'{')
                .map(|relative| index + relative)
                .unwrap_or(template.len());
            output.push_str(&template[index..next_open]);
            index = next_open;
        }
    }

    Ok(output)
}
