```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs. Zero is represented by [].
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn abs(&self) -> Self {
        let mut n = self.clone();
        n.negative = false;
        n
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let av = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = av + bv + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self { negative: false, limbs: out }
    }

    // Requires a >= b by magnitude.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let av = a.limbs[i] as i64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut value = av - bv - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u32);
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn mul_small_abs(&mut self, multiplier: u32) {
        if multiplier == 0 || self.is_zero() {
            self.limbs.clear();
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * multiplier as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_abs(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn div_rem_abs_small(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = vec![0u32; self.limbs.len()];
        let mut rem = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let value = rem * Self::BASE + self.limbs[i] as u64;
            quotient[i] = (value / divisor as u64) as u32;
            rem = value % divisor as u64;
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        (q, rem as u32)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            match Self::digit_value(bytes[i]) {
                Some(d) if d < radix => {}
                _ => {
                    invalid = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > start
                    && bytes[i - 1] != b'_'
                    && Self::digit_value(bytes[i - 1]).is_some_and(|d| d < radix);
                let right_ok = i + 1 < bytes.len()
                    && bytes[i + 1] != b'_'
                    && Self::digit_value(bytes[i + 1]).is_some_and(|d| d < radix);
                if !(left_ok && right_ok) {
                    separator = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let first_value = Self::digit_value(bytes[first_digit_index]).unwrap();
        if digit_count > 1 && first_value == 0 {
            return Err(Error::LeadingZero { index: first_digit_index });
        }
        if digit_count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen > max_digits {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut result = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            result.mul_small_abs(radix);
            result.add_small_abs(Self::digit_value(byte).unwrap());
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    fn magnitude_digits(&self, radix: u32) -> Vec<u8> {
        let alphabet = b"0123456789abcdefghijklmnopqrstuvwxyz";
        if self.is_zero() {
            return vec![b'0'];
        }
        let mut value = self.abs();
        let mut digits = Vec::new();
        while !value.is_zero() {
            let (q, r) = value.div_rem_abs_small(radix);
            digits.push(alphabet[r as usize]);
            value = q;
        }
        digits.reverse();
        digits
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut digits = self.magnitude_digits(radix);
        if digits.len() < min_digits {
            let mut padded = vec![b'0'; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut result = String::with_capacity(digits.len() + digits.len() / group.max(1) + 1);
        if self.negative && !self.is_zero() {
            result.push('-');
        }
        for (i, &d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                result.push('_');
            }
            result.push(d as char);
        }
        Ok(result)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::sub_abs(self, rhs);
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = Self::sub_abs(rhs, self);
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let value = out[k] as u64 + a as u64 * b as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let value = out[k] as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self { negative: self.negative ^ rhs.negative, limbs: out };
        result.normalize();
        result
    }

    fn shift_base_add(&mut self, limb: u32) {
        if self.is_zero() {
            if limb != 0 {
                self.limbs.push(limb);
            }
        } else {
            self.limbs.insert(0, limb);
        }
    }

    fn div_rem_abs(dividend: &Self, divisor: &Self) -> (Self, Self) {
        let mut quotient = vec![0u32; dividend.limbs.len()];
        let mut remainder = Self::zero();
        for i in (0..dividend.limbs.len()).rev() {
            remainder.shift_base_add(dividend.limbs[i]);
            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let mut product = divisor.clone();
                product.mul_small_abs(mid as u32);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }
            quotient[i] = best as u32;
            if best != 0 {
                let mut product = divisor.clone();
                product.mul_small_abs(best as u32);
                remainder = Self::sub_abs(&remainder, &product);
            }
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        (q, remainder)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = Self::div_rem_abs(&self.abs(), &rhs.abs());
        q.negative = !q.is_zero() && (self.negative ^ rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: Self::POW_LIMIT });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power != 0 {
            if power & 1 != 0 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, r) = Self::div_rem_abs(&a, &b);
            a = b;
            b = r;
        }
        a
    }
}
```
