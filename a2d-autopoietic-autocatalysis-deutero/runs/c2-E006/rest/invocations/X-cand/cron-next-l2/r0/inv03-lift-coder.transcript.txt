fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    offset: usize,
    field: Field,
    minimum: u8,
    maximum: u8,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let (base_end, step_range) = if let Some(&slash) = slash_positions.first() {
        if slash == 0 || slash + 1 == bytes.len() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (slash, Some((slash + 1, bytes.len())))
    } else {
        (bytes.len(), None)
    };

    let base = &bytes[..base_end];
    let star = base == b"*";
    let mut hyphens = Vec::new();
    for (i, &b) in base.iter().enumerate() {
        if b == b'-' {
            hyphens.push(i);
        }
    }

    let (number_ranges, is_range) = if star {
        if !hyphens.is_empty() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (None, false)
    } else if hyphens.is_empty() {
        if base.is_empty() || !base.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (Some((0, None)), false)
    } else if hyphens.len() == 1 {
        let h = hyphens[0];
        if h == 0

            || !base[..h].iter().all(u8::is_ascii_digit)
            || !base[h + 1..].iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (Some((0, Some(h))), true)
    } else {
        return Err(ParseError::InvalidSyntax { field, offset });
    };

    let mut start_value = minimum as u32;
    let mut end_value = maximum as u32;

    if let Some((start_begin, end_begin)) = number_ranges {
        let start_end = end_begin.unwrap_or(base.len());
        start_value = parse_number(base, start_begin, start_end, field)?;
        if start_value < minimum as u32 || start_value > maximum as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: offset + start_begin,
            });
        }

        if let Some(h) = end_begin {
            end_value = parse_number(base, h + 1, base.len(), field)?;
            if end_value < minimum as u32 || end_value > maximum as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: offset + h + 1,
                });
            }
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: offset + h + 1,
                });
            }
        } else if step_range.is_some() {
            end_value = maximum as u32;
        } else {
            end_value = start_value;
        }

        let value = parse_number(bytes, begin, end, field)?;
        let width = (maximum - minimum + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: offset + begin,
            });
        }
        value
    } else {
        1
    };

        end_value = start_value;
    }

    while value <= end_value {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())

fn parse_field(field_text: &str, field: Field, index: usize) -> Result<Vec<u8>, ParseError> {
    let bytes = field_text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = [false; 60];
    let mut item_start = 0;
    for item in field_text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }
        parse_item(
            item.as_bytes(),
            item_start,
            field,
            MINIMUMS[index],
            MAXIMUMS[index],
            &mut selected,
        )?;
        item_start += item.len() + 1;
    }

    Ok((MINIMUMS[index]..=MAXIMUMS[index])
        .filter(|&v| selected[v as usize])
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut sets = Vec::with_capacity(5);
    for i in 0..5 {
        sets.push(parse_field(pieces[i], FIELDS[i], i)?);
    }

    Ok(Cron {
        sets: sets.try_into().ok().unwrap(),
        exact_day_stars: [pieces[2] == "*", pieces[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at);
        Ok(self.sets[0].contains(&at.minute)
            && self.sets[1].contains(&at.hour)
            && self.sets[2].contains(&at.day)
            && self.sets[3].contains(&at.month)
            && self.sets[4].contains(&weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            current = match next_minute(current) {
                Some(value) => value,
                None => return Ok(None),
            };
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if !(1..=days_in_month(at.year, at.month)).contains(&at.day) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if is_leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += (at.day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn next_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        at.minute += 1;
        return Some(at);
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return Some(at);
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return Some(at);
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return Some(at);
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return Some(at);
    }
    None
}
