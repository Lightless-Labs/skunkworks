fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    let error = |kind| match kind {
        0 => ParseError::LeadingZero { field, offset: start },
        1 => ParseError::NumberOverflow { field, offset: start },
        2 => ParseError::BadStep { field, offset: start },
        _ => ParseError::OutOfRange { field, offset: start },
    };

    if end <= start {
        return Err(ParseError::InvalidSyntax { field, offset: start });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(0));
    }

    let mut value: u32 = 0;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| error(1))?;
    }

    let (min, max, width) = field_bounds(field);
    if is_step {
        if value < 1 || value > width as u32 {
            return Err(error(2));
        }
    } else if value < min || value > max {
        return Err(error(3));
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max, _) = field_bounds(field);
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base_end, step_range) = if let Some(&slash) = slash_positions.first() {
            (slash, Some((slash + 1, item.len())))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let syntax_valid = if is_star {
            hyphens.is_empty()
        } else if hyphens.is_empty() {
            !base.is_empty() && base.iter().all(u8::is_ascii_digit)
        } else {
            hyphens.len() == 1
                && hyphens[0] > 0
                && hyphens[0] + 1 < base.len()
                && base[..hyphens[0]].iter().all(u8::is_ascii_digit)
                && base[hyphens[0] + 1..].iter().all(u8::is_ascii_digit)
        };

        if !syntax_valid {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (start, end, end_offset) = if is_star {
            (min, max, item_start)
        } else if hyphens.is_empty() {
            let n = parse_number(bytes, item_start, item_start + base.len(), field, false)?;
            (n, if step_range.is_some() { max } else { n }, item_start)
        } else {
            let hyphen = hyphens[0];
            let start = parse_number(bytes, item_start, item_start + hyphen, field, false)?;
            let end_offset = item_start + hyphen + 1;
            let end = parse_number(
                bytes,
                end_offset,
                item_start + base.len(),
                field,
                false,
            )?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (start, end, end_offset)
        };

        let step = if let Some((step_start, step_end)) = step_range {
            Some(parse_number(
                bytes,
                item_start + step_start,
                item_start + step_end,
                field,
                true,
            )?)
        } else {
            None
        };

        let stride = step.unwrap_or(1);
        let mut value = start;
        loop {
            selected[value as usize] = true;
            if value >= end || end - value < stride {
                break;
            }
            value += stride;
        }

        let _ = end_offset;
        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    for i in 0..5 {
        selected.push(parse_field(parts[i], fields[i])?);
    }

    Ok(Cron {
        selected: selected.try_into().unwrap(),
        exact_day_stars: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(n, &yes)| yes.then_some(n as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    if current.day < days_in_month(current.year, current.month) {
                        current.day += 1;
                    } else {
                        current.day = 1;
                        if current.month < 12 {
                            current.month += 1;
                        } else {
                            if current.year == 2399 {
                                return Ok(None);
                            }
                            current.year += 1;
                            current.month = 1;
                        }
                    }
                }
            }

            if self.matches(current)? {
                return Ok(Some(current));
            }
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        1 | 3 | 5 | 7 | 8 | 10 | 12 => 31,
        4 | 6 | 9 | 11 => 30,
        2 if is_leap_year(year) => 29,
        2 => 28,
        _ => 0,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days % 7) % 7) as u8
}

                });
            }
            (left, right, right_start)
        };

        let step = if let Some((step_start, step_end)) = step_range {
            let abs_start = item_start + step_start;
            Some(parse_number(bytes, abs_start, item_start + step_end, field, true)?)
        } else {
            None
        };

        let increment = step.unwrap_or(1);
        let mut value = start;
        loop {
            selected[value as usize] = true;
            if value >= end || end - value < increment {
                break;
            }
            value += increment;
        }

        let _ = end_offset;

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());

    for i in 0..5 {
        selected[i] = parse_field(fields[i], pieces[i])?;
    }

    Ok(Cron {
        selected,
        exact_day_star: [pieces[2] == "*", pieces[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .filter(|&value| {
                let (min, max) = DOMAINS[index];
                (value as u32) >= min && (value as u32) <= max
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !advance_minute(&mut current) {
                return Ok(None);
            }
            if self.selected[0][current.minute as usize]
                && self.selected[1][current.hour as usize]
                && self.selected[3][current.month as usize]
                && self.selected[2][current.day as usize]
                && self.selected[4][weekday(current) as usize]
            {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if is_leap(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += (at.day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return true;
    }
    false
}

    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

pub struct Cron {

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

const MINIMUMS: [u8; 5] = [0, 0, 1, 1, 0];
const MAXIMUMS: [u8; 5] = [59, 23, 31, 12, 6];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    offset: usize,
    field: Field,
    minimum: u8,
    maximum: u8,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let (base_end, step_range) = if let Some(&slash) = slash_positions.first() {
        if slash == 0 || slash + 1 == bytes.len() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (slash, Some((slash + 1, bytes.len())))
    } else {
        (bytes.len(), None)
    };

    let base = &bytes[..base_end];
    let star = base == b"*";
    let mut hyphens = Vec::new();
    for (i, &b) in base.iter().enumerate() {
        if b == b'-' {
            hyphens.push(i);
        }
    }

    let (number_ranges, is_range) = if star {
        if !hyphens.is_empty() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (None, false)
    } else if hyphens.is_empty() {
        if base.is_empty() || !base.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (Some((0, None)), false)
    } else if hyphens.len() == 1 {
        let h = hyphens[0];
        if h == 0
            || h + 1 == base.len()
            || !base[..h].iter().all(u8::is_ascii_digit)
            || !base[h + 1..].iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (Some((0, Some(h))), true)
    } else {
        return Err(ParseError::InvalidSyntax { field, offset });
    };

    let mut start_value = minimum as u32;
    let mut end_value = maximum as u32;

    if let Some((start_begin, end_begin)) = number_ranges {
        let start_end = end_begin.unwrap_or(base.len());
        start_value = parse_number(base, start_begin, start_end, field)?;
        if start_value < minimum as u32 || start_value > maximum as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: offset + start_begin,
            });
        }

        if let Some(h) = end_begin {
            end_value = parse_number(base, h + 1, base.len(), field)?;
            if end_value < minimum as u32 || end_value > maximum as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: offset + h + 1,
                });
            }
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: offset + h + 1,
                });
            }
        } else if step_range.is_some() {
            end_value = maximum as u32;
        } else {
            end_value = start_value;
        }
    }

    let step = if let Some((begin, end)) = step_range {
        let value = parse_number(bytes, begin, end, field)?;
        let width = (maximum - minimum + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: offset + begin,
            });
        }
        value
    } else {
        1
    };

    if !is_range && !star && step_range.is_none() {
        end_value = start_value;
    }

    let mut value = start_value;
    while value <= end_value {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(field_text: &str, field: Field, index: usize) -> Result<Vec<u8>, ParseError> {
    let bytes = field_text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = [false; 60];
    let mut item_start = 0;
    for item in field_text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }
        parse_item(
            item.as_bytes(),
            item_start,
            field,
            MINIMUMS[index],
            MAXIMUMS[index],
            &mut selected,
        )?;
        item_start += item.len() + 1;
    }

    Ok((MINIMUMS[index]..=MAXIMUMS[index])
        .filter(|&v| selected[v as usize])
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut sets = Vec::with_capacity(5);
    for i in 0..5 {
        sets.push(parse_field(pieces[i], FIELDS[i], i)?);
    }

    Ok(Cron {
        sets: sets.try_into().ok().unwrap(),
        exact_day_stars: [pieces[2] == "*", pieces[4] == "*"],
    })
}

impl Cron {

    }

        validate_datetime(at)?;
        let weekday = weekday(at);
        Ok(self.sets[0].contains(&at.minute)
            && self.sets[1].contains(&at.hour)
            && self.sets[2].contains(&at.day)
            && self.sets[3].contains(&at.month)
            && self.sets[4].contains(&weekday))
    }

        &self,

                Some(value) => value,

            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
