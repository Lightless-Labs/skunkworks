#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    last_value: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut bytes = Vec::with_capacity(self.pending.len() + input.len());
        bytes.extend_from_slice(&self.pending);
        bytes.extend_from_slice(input);

        let complete_len = bytes.len() / 5 * 5;
        let output = encode_bytes(&bytes[..complete_len], self.config.alphabet);
        self.pending = bytes[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = encode_bytes(&self.pending, self.config.alphabet);
        if self.config.padding == Padding::Required && !self.pending.is_empty() {
            while output.len() % 8 != 0 {
                output.push('=');
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(input: &[u8], alphabet: Alphabet) -> String {
    let symbols = (input.len() * 8 + 4) / 5;
    let mut output = String::with_capacity(symbols);
    let alphabet = alphabet_bytes(alphabet);

    for symbol_index in 0..symbols {
        let mut value = 0u8;
        for bit_index in 0..5 {
            value <<= 1;
            let absolute_bit = symbol_index * 5 + bit_index;
            if absolute_bit < input.len() * 8 {
                let byte = input[absolute_bit / 8];
                value |= (byte >> (7 - absolute_bit % 8)) & 1;
            }
        }
        output.push(alphabet[value as usize] as char);
    }
    output
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            last_value: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let value = match decode_value(byte, next.config.alphabet) {
                Some(value) => value,
                None if byte == b'=' => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }
                    if next.first_padding_offset.is_none() {
                        let expected = match next.data_count % 8 {
                            2 => 6,
                            4 => 4,
                            5 => 3,
                            7 => 1,
                            _ => return Err(StreamError::Padding { offset }),
                        };
                        next.first_padding_offset = Some(offset);
                        next.expected_padding = expected;
                    }
                    if next.padding_count >= next.expected_padding {
                        return Err(StreamError::Padding { offset });
                    }
                    next.padding_count += 1;
                    next.position += 1;
                    continue;
                }
                None => {
                    return Err(StreamError::InvalidByte { offset, byte });
                }
            };

            if byte.is_ascii_lowercase() && next.config.canonical {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: byte.to_ascii_uppercase(),
                });
            }
            if next.first_padding_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count += 1;
            next.last_value = value;
            next.last_data_offset = Some(offset);
            next.bit_buffer = (next.bit_buffer << 5) | value;
            next.bit_count += 5;

            while next.bit_count >= 8 {
                next.bit_count -= 8;
                output.push((next.bit_buffer >> next.bit_count) & 0xff);
                if next.bit_count == 0 {
                    next.bit_buffer = 0;
                } else {
                    next.bit_buffer &= (1u8 << next.bit_count) - 1;
                }
            }
            next.position += 1;
        }

        *self = next;
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => unreachable!(),
        };
        if unused_bits != 0 && self.last_value & ((1 << unused_bits) - 1) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

fn decode_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let upper = byte.to_ascii_uppercase();
    alphabet_bytes(alphabet)
        .iter()
        .position(|&candidate| candidate == upper)
        .map(|value| value as u8)
}
